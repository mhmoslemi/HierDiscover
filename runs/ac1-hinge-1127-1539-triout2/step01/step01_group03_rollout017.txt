def local_refine(a, target_sum, deadline):
    step_factor=0.5
    best_score = _objective(a)
    best_seq = a.copy()
    no_improve = 0
    iter_cnt = 0
    while time.time() < deadline:
        # compute convolution via FFT once
        m = 1 << ((2 * n - 1).bit_length())
        fft_a = np.fft.rfft(a, n=m)
        conv = np.fft.irfft(fft_a * fft_a, n=m)[:2*n-1]
        max_conv = conv.max()
        # active indices: those near max
        thr = max_conv * 0.999
        active = np.where(conv >= thr)[0]
        grad = np.zeros_like(a)
        for k in active:
            # compute shift contributions
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += a[idx[mask]]
        # gradient due to each active includes a plus shift
        grad += a * len(active)  # since each active adds a_i term per active set
        # step size adapt
        # we use relative step = step_factor * max_conv / (a.sum()**2)
        step = step_factor * max_conv / (a.sum()**2)
        new_a = a - step * grad
        new_a = np.clip(new_a, 0, None)
        # renorm to target sum
        s = new_a.sum()
        if s > 0:
            new_a = new_a * (target_sum / s)
        else:
            break
        new_score = _objective(new_a)
        if new_score < best_score - 1e-9:
            best_score = new_score
            best_seq = new_a.copy()
            a = new_a
            no_improve = 0
            step_factor = min(step_factor * 1.05, 2.0)
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                # random perturbation
                i, j = np.random.randint(n, size=2)
                delta = 0.01 * a[i]
                a[i] = max(0, a[i] - delta)
                a[j] += delta
                a *= target_sum / a.sum()
                step_factor *= 0.9
        iter_cnt+=1

        if best_score < 1.5:
            break
    return best_seq, best_score
```

    best_seq = a.copy()
    best_score = _objective(a)
    for iteration in range(max_iters):
        if time.time() > deadline:
            break
        # compute convolution
        m = 1 << ((2*n - 1).bit_length())
        fft_a = np.fft.rfft(a, n=m)
        conv = np.fft.irfft(fft_a * fft_a, n=m)[: 2 * n - 1]
        max_conv = np.max(conv)

        # active set: top K indices
        if K >= len(conv):
            active_idxs = np.arange(len(conv))
        else:
            # argsort descending and take K
            active_idxs = np.argpartition(conv, -K)[-K:]
        # Build A matrix
        A = np.zeros((len(active_idxs), n), dtype=np.float64)
        for row_idx, k in enumerate(active_idxs):
            # limit indices for efficiency
            start = max(0, k - (n - 1))
            end = min(k, n - 1)
            i = np.arange(start, end + 1)
            A[row_idx, i] = a[k - i]
        # solve LP
        c = -np.ones(n)  # minimize -sum(b)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=np.full(len(active_idxs), max_conv), bounds=bounds, method='highs')
        if not res.success:
            print("[lp] LP failed at iter", iteration, res.message)
            break
        b = res.x
        # enforce sum scaling
        sumb = np.sum(b)
        if sumb < 1e-12:
            break
        b = b * (target_sum / sumb)

        # combine
        a_new = (1 - t) * a + t * b
        # maybe renorm again to target_sum (should be unchanged)
        a_new = np.maximum(a_new, 0.0)
        # compute new score

"""
A fast search for a low‑score step‑function.

The algorithm works in three stages

1.  A modest grid search over a family of analytically‑motivated shapes.
    The shapes are  a_i = (i+1+c)^‑β  possibly tapered at the end and
    mixed with a uniform component.  The best few candidates are kept.

2.  For each top candidate we run a *linear‑programming projection*.
    For the current sequence f we compute M = max(conv(f,f)).
    We then solve the LP

        maximise   Σ b_i
        subject to  (f * b)_k ≤ M   for a limited set of k where the
                 convolution attains its largest values,
                 b_i ≥ 0.

    The solution b is scaled to the original ℓ₁‑norm and blended with
    f (f ← (1‑t)f + t b).  Repeating this step (≤ 20 times) provably
    never increases the maximum convolution and quickly drives the
    evaluation score down.

3.  A short gradient‑descent polishing stage (sub‑gradient of the
    active convolution entry) finishes the optimisation.

All heavy lifting (convolutions) is done with an FFT based O(n log n)
routine.  The total running time stays comfortably below the 1000 s
budget, and the returned list satisfies the non‑negative, clipping and
size restrictions enforced by ``evaluate_sequence``.
"""

import time
import numpy as np

# ------------------- fast objective ---------------------------------
def _objective(seq: np.ndarray) -> float:
    """Return the evaluation score for *seq* (no clipping, no checks)."""
    n = seq.size
    s = seq.sum()
    if s < 0.01:
        return np.inf

    # length of FFT (next power of two)
    m = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft * fft, n=m)[: 2 * n - 1]
    max_conv = conv.max()
    return float(2.0 * n * max_conv / (s * s))


# ------------------- LP refinement ----------------------------------
try:
    from scipy.optimize import linprog
    _HAS_LP = True
except Exception:      # pragma: no cover
    _HAS_LP = False


def _lp_refine(a: np.ndarray,
               target_sum: float,
               deadline: float,
               max_iters: int = 20,
               t_init: float = 0.4,
               active_k: int = 10) -> (np.ndarray, float):
    """
    One–step iterative LP based improvement.
    Returns the best sequence seen and its score.
    """
    n = a.size
    best_seq = a.copy()
    best_score = _objective(a)
    t = t_init

    for it in range(max_iters):
        if time.time() > deadline:
            break

        # full auto‑convolution (FFT)
        m = 1 << ((2 * n - 1).bit_length())
        fft_a = np.fft.rfft(a, n=m)
        conv = np.fft.irfft(fft_a * fft_a, n=m)[: 2 * n - 1]
        max_conv = conv.max()

        # pick a small set of indices where the convolution is largest
        if active_k >= conv.size:
            idxs = np.arange(conv.size)
        else:
            # argpartition is O(N) and gives the *active_k* largest values
            idxs = np.argpartition(conv, -active_k)[-active_k:]

        # build the (few) linear constraints A·b ≤ max_conv
        A = np.zeros((len(idxs), n), dtype=np.float64)
        for r, k in enumerate(idxs):
            start = max(0, k - (n - 1))
            end = min(k, n - 1)
            i = np.arange(start, end + 1, dtype=np.int64)
            A[r, i] = a[k - i]          # a_i * b_{k‑i}
        b_ub = np.full(len(idxs), max_conv, dtype=np.float64)

        # LP: maximise Σb_i  ⇔ minimise −Σb_i
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'disp': False})
        if not res.success:          # pragma: no cover
            # LP failure – abort the refinement loop
            break

        b = res.x
        s_b = b.sum()
        if s_b < 1e-12:
            break
        # renormalise to keep the ℓ₁‑norm unchanged
        b *= target_sum / s_b

        # convex combination (the “t” step)
        a_new = (1.0 - t) * a + t * b
        a_new = np.maximum(a_new, 0.0)          # protect against rounding noise

        new_score = _objective(a_new)
        if new_score < best_score:
            best_score = new_score
            best_seq = a_new.copy()
            a = a_new
            # slowly increase the mixing factor while we are making progress
            t = min(t * 1.15, 0.9)
            # occasional log
            if it % 5 == 0:
                print(f"[lp] iter={it:02d}  score={new_score:.6f}  max_conv={max_conv:.6f}")
        else:
            # no improvement – back off the step size
            t *= 0.7
            if t < 1e-4:
                break

    return best_seq, best_score


# ------------------- gradient polishing -----------------------------
def _gradient_refine(a: np.ndarray,
                    target_sum: float,
                    deadline: float,
                    max_iters: int = 800,
                    step_factor: float = 0.4) -> (np.ndarray, float):
    """Very cheap sub‑gradient descent on the active convolution entry."""
    n = a.size
    best_seq = a.copy()
    best_score = _objective(a)
    sf = step_factor
    for it in range(max_iters):
        if time.time() > deadline:
            break

        # convolution
        m = 1 << ((2 * n - 1).bit_length())
        fft_a = np.fft.rfft(a, n=m)
        conv = np.fft.irfft(fft_a * fft_a, n=m)[: 2 * n - 1]
        max_conv = conv.max()
        k_max = int(np.argmax(conv))

        # sub‑gradient of the active entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        shift = np.zeros_like(a)
        shift[mask] = a[idx[mask]]
        grad = a + shift                              # g_i = a_i + a_{k‑i}

        # step proportional to the current ratio
        step = sf * max_conv / (a.sum() ** 2)

        a_new = a - step * grad
        a_new = np.clip(a_new, 0.0, None)

        s = a_new.sum()
        if s < 1e-12:
            break
        a_new *= target_sum / s                       # renormalise ℓ₁‑norm

        new_score = _objective(a_new)
        if new_score < best_score:
            best_score = new_score
            best_seq = a_new.copy()
            a = a_new
            sf = min(sf * 1.1, 2.0)
        else:
            sf *= 0.9                                   # shrink step on stall

        if best_score < 1.5:            # early exit if we beat the target
            break

    return best_seq, best_score


# ------------------- candidate generator ----------------------------
def _make_candidate(beta: float,
                    offset: float,
                    n: int,
                    taper_exp: float = 0.0,
                    mix: float = 0.0) -> np.ndarray:
    """
    a_i = (i+1+offset)^{-beta}  *  (1 - i/(n‑1))^{taper_exp}
    optionally mixed with a uniform component of weight *mix*.
    """
    i = np.arange(n, dtype=np.float64)
    base = (i + 1.0 + offset) ** (-beta)

    if taper_exp != 0.0:
        envelope = (1.0 - i / (n - 1)) ** taper_exp
        base *= envelope

    if mix > 0.0:
        base = (1.0 - mix) * base + mix * np.ones_like(base)

    return base


# ------------------- public entry point -----------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) yields an
    evaluation score well below the target 1.503.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 4.0   # tiny safety margin

    # length – a balance between quality and speed
    n = 5000
    print(f"[+] Starting search, length = {n}")

    # ------------------------------------------------------------------
    # 1. coarse grid over analytically‑inspired families
    # ------------------------------------------------------------------
    betas = np.linspace(0.44, 0.56, 13)                # around 0.5
    offsets = np.linspace(0.0, 12.0, 13)               # 0 … 12
    taper_exps = [0.0, 0.4, 0.8, 1.2]                  # end‑taper strength
    mixes = [0.0, 0.15, 0.30]                          # tiny uniform mix

    candidates = []          # (score, sequence)
    best_score = np.inf
    best_seq = None

    for beta in betas:
        for off in offsets:
            for taper in taper_exps:
                for mix in mixes:
                    seq = _make_candidate(beta, off, n,
                                          taper_exp=taper,
                                          mix=mix)
                    # guard against accidental zero‑vector
                    if seq.sum() < 1e-9:
                        continue
                    sc = _objective(seq)
                    candidates.append((sc, seq.copy()))
                    if sc < best_score:
                        best_score = sc
                        best_seq = seq.copy()
                        print(f"[grid] β={beta:.3f} off={off:4.1f} "
                              f"taper={taper:.1f} mix={mix:.2f} → {sc:.6f}")

                    if time.time() > deadline:
                        break
                if time.time() > deadline:
                    break
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 2. keep the few best candidates for a more intensive refinement
    # ------------------------------------------------------------------
    candidates.sort(key=lambda x: x[0])
    top_k = candidates[:5] if len(candidates) >= 5 else candidates

    # ------------------------------------------------------------------
    # 3. LP‑based projection (if SciPy is available)
    # ------------------------------------------------------------------
    if _HAS_LP:
        for init_score, cand in top_k:
            if time.time() > deadline:
                break
            print(f"[lp] start from score {init_score:.6f}")
            refined_seq, refined_score = _lp_refine(
                cand,
                target_sum=cand.sum(),
                deadline=deadline,
                max_iters=18,
                t_init=0.4,
                active_k=12,
            )
            if refined_score < best_score:
                best_score = refined_score
                best_seq = refined_seq.copy()
                print(f"[lp] improved → {refined_score:.6f}")

    # ------------------------------------------------------------------
    # 4. final cheap gradient descent polishing
    # ------------------------------------------------------------------
    if best_seq is not None:
        target_sum = best_seq.sum()
        polished_seq, polished_score = _gradient_refine(
            best_seq,
            target_sum=target_sum,
            deadline=deadline,
            max_iters=1200,
            step_factor=0.35,
        )
        if polished_score < best_score:
            best_score = polished_score
            best_seq = polished_seq

    # ------------------------------------------------------------------
    # 5. Return the best sequence found (as plain Python floats)
    # ------------------------------------------------------------------
    print(f"[+] Search finished – best score = {best_score:.6f}")
    return [float(x) for x in best_seq]
