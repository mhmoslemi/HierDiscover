def propose_candidate(seed=0, budget_s=1000):
    import time, math, numpy as np

    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0

    # Determine length
    try:
        n = len(height_sequence_1)
    except Exception:
        n = 12000

    target_sum = math.sqrt(2.0 * n)

    # helper function produce a sequence from parameters
    def generate(p, shift):
        i = np.arange(n, dtype=np.float64)
        base = (i + shift + 1) ** p - (i + shift) ** p
        a = base
        a_sum = a.sum()
        if a_sum == 0:  # avoid
            return np.full(n, target_sum / n)
        a = a * (target_sum / a_sum)
        return a

    # quick scoring function
    L = 1 << ((2 * n - 1).bit_length())
    def score_of(a):
        conv = np.fft.irfft(np.fft.rfft(a, n=L) * np.fft.rfft(a, n=L), n=L)[:2*n-1]
        return conv.max()

    # Initial candidate using sqrt difference (p=0.5 shift=0)
    best_a = generate(0.5, 0.0)
    best_score = score_of(best_a)

    # random parameter search
    for _ in range(200):
        p = np.random.uniform(0.35, 0.55)  # maybe include near sqrt
        shift = np.random.uniform(0.0, 2.0)
        a = generate(p, shift)
        sc = score_of(a)
        if sc < best_score:
            best_score = sc
            best_a = a
    print(f"After random search score {best_score:.6f}")

    # Adam optimizer
    a = best_a.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr_base = 0.2  # base learning rate
    iteration = 0

    while time.time() < deadline:
        # compute conv and max
        conv = np.fft.irfft(np.fft.rfft(a, n=L) * np.fft.rfft(a, n=L), n=L)[:2*n-1]
        max_b = conv.max()
        cur_score = max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"Iter {iteration} new best {best_score:.6f}")

        # compute subgradient
        eps = 1e-12 * max_b
        ks = np.where(conv >= max_b - eps)[0]
        grad = np.zeros_like(a)
        for k in ks:
            _add_gradient_for_k(grad, a, k)

        # Adam update
        iteration += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        lr = lr_base / (np.sqrt(iteration) + 1)  # decaying
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # projection
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, target_sum / n)
        else:
            a = a * (target_sum / s)

        # occasional jitter
        if iteration % 200 == 0:
            sigma = target_sum * 5e-4 / math.sqrt(n)  # small noise
            noise = np.random.randn(n) * sigma
            a = np.maximum(a + noise, 0.0)
            a = a * (target_sum / a.sum())
    print(f"Finished after {iteration} iter best score {best_score:.6f}")
    return best_a.tolist()

"""
Propose a non‑negative step function with a low self‑convolution supremum.

The algorithm works as follows:

1. **Length choice** – use a reasonably large n (≈ 1.5 × 10⁴).  A larger
   n allows the discrete problem to approach its continuous limit, where
   the theoretical optimum is ≈ 1.0.  The length is taken from the
   supplied ``height_sequence_1`` if it exists, otherwise a default of
   15000 is used.

2. **Parametric warm‑start** – generate candidates of the form
      a_i = (i+δ+1)^p − (i+δ)^p
   (the discrete derivative of a power function).  Random values of the
   exponent *p* ∈ [0.30, 0.60] and the shift δ ∈ [0, 3] are sampled,
   each candidate is scaled so that Σa = √(2 n) (the scaling does not
   affect the final score) and the best one (smallest max‑convolution)
   is kept.

3. **Projected Adam optimisation** – starting from the best parametric
   candidate, perform many iterations of a projected Adam update:
   * compute the self‑convolution with an FFT,
   * collect all indices where the maximum is attained (sub‑gradient of
     the ∞‑norm),
   * update the coefficients with Adam’s adaptive moments,
   * project back onto the feasible set (non‑negative and fixed L₁‑norm).

   Every few hundred iterations a tiny Gaussian jitter is added to
   escape possible plateaus.

The whole loop runs until the time budget (≈ 1000 s) is exhausted.
The best sequence found so far is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – must be top‑level (no closures)
# ----------------------------------------------------------------------


def _add_gradient_for_k(grad: np.ndarray, a: np.ndarray, k: int) -> None:
    """
    Add the sub‑gradient contribution for a particular shift `k`
    to the gradient array `grad`.

    The convolution (a * a)[k] = Σ_i a_i a_{k‑i}.  Its derivative w.r.t.
    a_i, for a given k, is 2·a_{k‑i} (when the index is inside the
    allowed range).  This routine adds those terms efficiently.
    """
    n = a.size
    if k < n:
        # indices 0 … k  (i from 0 to k, a_{k-i} runs backwards)
        grad[: k + 1] += 2.0 * a[k::-1]
    else:
        # indices k‑(n‑1) … n‑1
        start = k - n + 1
        length = 2 * n - 1 - k
        grad[start:] += 2.0 * a[: length][::-1]


def _self_convolution(a: np.ndarray, L: int) -> np.ndarray:
    """
    Return the (full) self‑convolution of `a` using an FFT of length `L`.
    The result has length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * a.size - 1]
    return conv


def _scale_to_target(a: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale the vector so that its L1‑norm equals `target_sum`."""
    s = a.sum()
    if s == 0.0:
        # fallback to the uniform distribution
        return np.full_like(a, target_sum / a.size)
    return a * (target_sum / s)


# ----------------------------------------------------------------------
# Main entry point required by the judge
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) minimises the
    provided `evaluate_sequence` function.

    Parameters
    ----------
    seed : int
        Random seed to make the search deterministic.
    budget_s : int
        Time budget in seconds (the function stops a little early).

    Returns
    -------
    list[float]
        The best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0   # small safety margin

    # ------------------------------------------------------------------
    # 1) Choose a length.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)               # type: ignore[name-defined]
    except Exception:
        n = 15000                                 # comfortable default

    target_sum = math.sqrt(2.0 * n)               # we keep Σa = √(2n)

    # Pre‑compute the FFT length (next power of two ≥ 2n‑1)
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # 2) Helper to build a candidate from parameters (p, shift)
    # ------------------------------------------------------------------
    def make_candidate(p: float, shift: float) -> np.ndarray:
        i = np.arange(n, dtype=np.float64)
        # power‑law derivative a_i = (i+shift+1)^p – (i+shift)^p
        a = (i + shift + 1.0) ** p - (i + shift) ** p
        # Numerical safety – ensure no negatives due to rounding
        a = np.maximum(a, 0.0)
        return _scale_to_target(a, target_sum)

    # ------------------------------------------------------------------
    # 3) Fast scoring routine – the objective after fixing Σa = √(2n)
    #    is simply max(conv).
    # ------------------------------------------------------------------
    def score_of(a: np.ndarray) -> float:
        conv = _self_convolution(a, L)
        return float(conv.max())

    # ------------------------------------------------------------------
    # 4) Initialise with the classic sqrt‑difference sequence (p=0.5)
    # ------------------------------------------------------------------
    best_a = make_candidate(0.5, 0.0)
    best_score = score_of(best_a)
    print(f"Initial (sqrt‑diff) score: {best_score:.6f}")

    # If a previous construction is available, try it as a warm start as
    # well (after normalisation).
    try:
        prev = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        prev = _scale_to_target(prev, target_sum)
        sc = score_of(prev)
        if sc < best_score:
            best_score = sc
            best_a = prev.copy()
            print(f"Previous global sequence improved score to {best_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 5) Random sampling of (p, shift) – cheap global exploration.
    # ------------------------------------------------------------------
    for _ in range(250):
        p = np.random.uniform(0.30, 0.60)
        shift = np.random.uniform(0.0, 3.0)
        cand = make_candidate(p, shift)
        sc = score_of(cand)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"Random search improved to {best_score:.6f} (p={p:.4f},Δ={shift:.4f})")

    # ------------------------------------------------------------------
    # 6) Projected Adam optimisation starting from the best candidate.
    # ------------------------------------------------------------------
    a = best_a.copy()
    m = np.zeros_like(a)          # first moment estimate
    v = np.zeros_like(a)          # second moment estimate

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_lr = 0.25                # base learning rate (will decay)

    iteration = 0
    while time.time() < deadline:
        # ----- self‑convolution and current score -----
        conv = _self_convolution(a, L)
        max_b = float(conv.max())
        cur_score = max_b

        # keep the best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"Iter {iteration:6d} – new best {best_score:.6f}")

        # ----- sub‑gradient of the ∞‑norm (all active indices) -----
        eps = 1e-12 * max_b
        active = np.where(conv >= max_b - eps)[0]
        grad = np.zeros_like(a)
        for k in active:
            _add_gradient_for_k(grad, a, k)

        # ----- Adam update (projected) -----
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # simple decay of the base step size
        lr = base_lr / (math.sqrt(iteration) + 1.0)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- Projection onto the feasible set -----
        a = np.maximum(a, 0.0)
        a = _scale_to_target(a, target_sum)

        # ----- Occasional tiny jitter – helps to escape flat regions -----
        if iteration % 250 == 0:
            sigma = target_sum * 5e-4 / math.sqrt(n)
            noise = np.random.randn(n) * sigma
            a = np.maximum(a + noise, 0.0)
            a = _scale_to_target(a, target_sum)

    # ------------------------------------------------------------------
    # Final reporting
    # ------------------------------------------------------------------
    print(f"Finished after {iteration} iterations – best score {best_score:.6f}")

    # Return the best sequence found as a plain Python list.
    return best_a.tolist()
