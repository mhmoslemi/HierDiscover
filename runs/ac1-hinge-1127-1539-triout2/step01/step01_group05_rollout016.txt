```
def propose_candidate(seed=42,budget_s=1000,**kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # Choose lengths to try
    lengths = [4000,5000,6000,8000] (bounded by maybe max 10000)
    # Use sequences for each length.
    best_score = float('inf')
    best_seq = None

    for n in lengths:
        # initialize a as described shape, plus maybe variant.
        a = init_shape(n)
        best_a_local = a.copy()
        best_score_local = compute_score(a)
        # ADAM
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        eta0 = 0.2
        gamma = 5.0
        iteration = 0
        while time.time() < deadline:
            iteration += 1
            # compute conv
            conv = fft_convolve(a)
            max_conv = conv.max()
            # compute weights
            w = np.exp(gamma * (conv - max_conv))
            w /= w.sum()
            # gradient = 2 * (w cross-correlation a_rev)
            grad = compute_grad(w,a)
            # Adam
            m = 0.9*m + (1-0.9)*grad
            v = 0.999*v + (1-0.999)*(grad**2)
            m_hat = m/(1-0.9**iteration)
            v_hat = v/(1-0.999**iteration)
            a_new = a - eta0 * m_hat/(np.sqrt(v_hat)+1e-8)
            a_new = np.maximum(a_new,0)
            # renorm
            a_new *= target_sum / a_new.sum()
            a = a_new
            # schedule
            if iteration % 10 == 0:
                gamma = min(gamma*1.01, 200.0)
                eta0 = max(eta0*0.995, 0.0005)
            # evaluate periodically
            if iteration % 50 == 0:
                score = compute_score(a)
                if score < best_score_local:
                    best_score_local=score
                    best_a_local = a.copy()
                    # log
        # after loop, compare to global best
        if best_score_local < best_score:
            best_score = best_score_local; best_seq = best_a_local.copy()
    # Return best_seq.tolist()
```

import numpy as np
import time

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _compute_score(a: np.ndarray) -> float:
    """Fast evaluation of the objective (no security checks)."""
    n = a.shape[0]
    nfft = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    return float(2.0 * n * max_c / (s * s))

def _gradient(a: np.ndarray, gamma: float,
              nfft_conv: int, nfft_corr: int) -> np.ndarray:
    """Return the gradient of the soft‑max of the convolution."""
    n = a.shape[0]
    # convolution a*a
    af = np.fft.rfft(a, n=nfft_conv)
    conv = np.fft.irfft(af * af, n=nfft_conv)[: 2 * n - 1]
    max_c = conv.max()
    # soft‑max weights
    w = np.exp(gamma * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # cross‑correlation w ⊛ a_rev
    wf = np.fft.rfft(w, n=nfft_corr)
    af_rev = np.fft.rfft(a[::-1], n=nfft_corr)
    corr_full = np.fft.irfft(wf * af_rev, n=nfft_corr)
    grad = 2.0 * corr_full[n - 1 : n - 1 + n]
    return grad

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    2·n·max(conv(a,a)) / (sum(a)²).  The routine combines a quick
    analytical initialisation (a_i ≈ 1/(i+½)^p) with an Adam‑style
    gradient descent on a soft‑max approximation of the maximal convolution.
    Several problem sizes and several initial exponents are tried within the
    allotted time; the best sequence found is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------

              nfft_conv: int, nfft_corr: int) -> np.ndarray:
