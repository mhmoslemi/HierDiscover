Implementation details for projection onto simplex:

```
def proj_simplex(v):
    # v is np array, will project onto simplex sum=1, v>=0.
    # using algorithm from https://arxiv.org/abs/1101.6081.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

Then after update compute a = proj_simplex(a + α*d). This ensures we stay on simplex constraints.

"""
Subgradient‐projected descent for the L∞ self‑convolution problem.

We minimise   2·n·max_k (a∗a)[k]   with a_i≥0, Σa_i=1.
The algorithm repeatedly:

* evaluates the full linear convolution a∗a with an FFT,
* obtains the index k* where the maximum is attained,
* builds the sub‑gradient of the maximal entry:
        g_i = 2·a_{k*−i}   (0 ≤ k*−i < n, else 0)
* moves opposite to the centred gradient (so the L1‑sum stays unchanged)
  and projects back onto the probability simplex.
* a small random jitter is injected from time to time to escape plateaus.

The method starts from the provided `height_sequence_1` (if any) and otherwise
from a random Dirichlet draw.  It runs for the allotted budget (≤ 1000 s)
and returns the best sequence seen.  The external score is computed as
`2·n·max_conv` which equals the value returned by ``evaluate_sequence`` for
a normalised sum of 1.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _conv_max(a: np.ndarray) -> tuple[float, int]:
    """
    Full linear convolution a*a via FFT.
    Returns (maximum value, index of that maximum).
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()   # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    max_idx = int(np.argmax(conv))
    return float(conv[max_idx]), max_idx


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex {x ≥ 0, Σx = 1}.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("simplex projection works on 1‑D arrays")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient_step(a: np.ndarray, eta: float) -> tuple[np.ndarray, float]:
    """
    One projected sub‑gradient step that reduces the maximal
    convolution entry.  Returns the new vector and the current max entry.
    """
    n = a.shape[0]
    cur_max, k = _conv_max(a)

    # sub‑gradient of the maximal entry (g_i = 2·a_{k−i} when defined)
    idx = np.arange(n)
    partner = k - idx
    valid = (partner >= 0) & (partner < n)

    g = np.zeros_like(a)
    g[valid] = 2.0 * a[partner[valid]]

    # direction that keeps Σa = 1 (centered gradient)
    d = -(g - np.mean(g))

    # additive update, then project back onto the simplex
    a_new = a + eta * d
    a_new = _proj_simplex(a_new)

    return a_new, cur_max


def _initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a feasible start vector of length n with Σ=1.
    50 % of the time the existing global `height_sequence_1` is used,
    otherwise a Dirichlet(1,…,1) draw is taken.
    """
    if "height_sequence_1" in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=float)
            if base.size >= n:
                a = base[:n].copy()
            else:
                a = np.pad(base, (0, n - base.size), constant_values=0.0)
            if rng.random() < 0.5:
                return _proj_simplex(a)   # keep the provided shape
        except Exception:
            pass
    # random Dirichlet – uniform on the simplex
    a = rng.random(n)
    return _proj_simplex(a)


def _optimize_subgradient(
    n: int,
    seed: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Run the projected sub‑gradient descent until the deadline.
    Returns the best sequence found and its external score.
    """
    rng = np.random.default_rng(seed)

    # start point
    a = _initial_sequence(n, rng)

    # initial external score (2·n·max_conv because Σa = 1)
    cur_max, _ = _conv_max(a)
    best_score = 2.0 * n * cur_max
    best_seq = a.copy()
    print(f"[init] n={n}, score={best_score:.6f}")

    it = 0
    eta0 = 0.7           # initial step size (tuned empirically)

    # parameters for occasional random jitter
    jitter_prob = 0.02
    jitter_scale = 0.04   # relative to the current mean

    while time.time() < deadline:
        it += 1
        # simple diminishing step size
        eta = eta0 / np.sqrt(it)

        a, cur_max = _subgradient_step(a, eta)
        cur_score = 2.0 * n * cur_max

        # keep the best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[improve] it={it:,} score={best_score:.6f}")

        # occasional jitter to escape flat regions
        if rng.random() < jitter_prob:
            sigma = a.mean() * jitter_scale
            noise = rng.normal(scale=sigma, size=n)
            a_jitter = a + noise
            a = _proj_simplex(a_jitter)

    return best_seq, best_score
