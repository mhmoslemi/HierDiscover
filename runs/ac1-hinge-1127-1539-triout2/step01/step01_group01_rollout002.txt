"""
Propose a step‑function (non‑negative list) that minimizes the
self‑convolution bound used by `evaluate_sequence`.

The optimiser works in several independent restarts:

* A start vector is built either from the provided `height_sequence_1`
  (if present) or from a simple symmetric “U‑shape’’ (linear,
  quadratic or exponential) with a randomly chosen strength parameter.
* Each restart runs a projected Adam‑style gradient descent on the
  sub‑gradient of the maximum entry of the self‑convolution.
  The direction is orthogonalised so that the ℓ₁‑norm (the required sum)
  stays constant; after the step the vector is clipped, symmetrised
  and re‑scaled to the normalisation `sqrt(2·n)`.
* A tiny fraction of random mass transfers and occasional Gaussian
  jitter are injected to escape plateaus.
* The active set for the sub‑gradient is enlarged gradually
  (`rel_tol` grows from 0 to ~0.05) which smoothes the descent.
* The best sequence over all restarts is returned.  The objective is
  evaluated internally via an FFT‑based convolution (much faster than
  the naïve `np.convolve` used inside `evaluate_sequence`), but the
  final sequence complies with all checks performed by `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution of a with itself using a power‑of‑two FFT.
    Returns the full convolution of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """
    Scale a non‑negative vector to have the prescribed ℓ₁‑norm.
    Zero entries stay zero; if the whole vector is zero a uniform
    distribution is returned.
    """
    s = v.sum()
    if s > 0:
        return v * (target / s)
    return np.full_like(v, target / v.shape[0])


def generate_U_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
) -> np.ndarray:
    """
    Produce a symmetric “U‑shaped’’ sequence of length n.
    kind ∈ {linear, quadratic, exponential}.
    param ≥ 0 controls the edge‑enhancement.
    The result is scaled so that the ℓ₁‑norm equals target_sum.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_to_sum(raw, target_sum)


def enforce_symmetry(a: np.ndarray) -> np.ndarray:
    """Force a to be symmetric: a[i] = a[-i-1]."""
    return (a + a[::-1]) * 0.5


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a tiny amount of mass from one random index to another.
    The total sum is preserved (up to a re‑normalisation).
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0.0:
        return a
    a[i] -= delta
    a[j] += delta
    # re‑scale to keep the exact target sum (helps against rounding errors)
    return project_to_sum(a, target_sum)


def compute_score(a: np.ndarray, target_sum: float) -> float:
    """
    Fast surrogate for `evaluate_sequence`: 2·n·max(conv(a,a)) / (target_sum**2).
    All entries are assumed to be non‑negative and below 1000, which holds
    for the vectors produced by the optimiser.
    """
    n = a.shape[0]
    conv = conv_self_fft(a)
    max_val = conv.max()
    return 2.0 * n * max_val / (target_sum ** 2)

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:

    Search for a non‑negative height sequence that minimises
    `evaluate_sequence`.  The function runs for at most *budget_s*
    seconds and returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Build a list of initial candidates (different lengths / shapes)
    # ------------------------------------------------------------------
    start_candidates = []

    # 1) reuse the provided construction if it exists
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base):
        n0 = len(base)
        a0 = np.array(base, dtype=np.float64)
        target0 = np.sqrt(2.0 * n0)
        a0 = project_to_sum(a0, target0)
        a0 = enforce_symmetry(a0)
        start_candidates.append((n0, a0, "provided"))

    for _ in range(5):
        n = np.random.randint(7000, 10001)                 # manageable size
        target = np.sqrt(2.0 * n)
        kind = np.random.choice(["linear", "quadratic", "exponential"])
        param = np.random.uniform(0.0, 5.0)
        a = generate_U_shape(n, target, kind=kind, param=param)
        a = enforce_symmetry(a)
        start_candidates.append((n, a, f"{kind}_{param:.2f}"))

    # ------------------------------------------------------------------
    # Main optimisation loops – one independent run per start candidate
    # ------------------------------------------------------------------
    for n, a_start, descr in start_candidates:
        if time.time() >= deadline:
            break

        print(f"[{time.time():.2f}] Starting run ({descr}) – length {n}")

        a = a_start.copy()

        lr = 0.02 * avg                 # initial learning rate (≈2 % of avg entry)

        v = np.zeros_like(a)

            a = project_to_sum(a, target_sum)  # restore the required ℓ₁‑norm

            # Occasional random pairwise mass transfer (helps escape flat regions)
            if np.random.rand() < 0.01:
                a = random_pair_transfer(a, target_sum, scale=0.01)

            # Current objective value (fast surrogate)
            cur_score = compute_score(a, target_sum)

            # Record improvement
            if cur_score + 1e-12 < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"Iter {t_iter:06d} ({descr}) – new global best {best_score:.6f}")
                no_improve = 0
            else:
                no_improve += 1

            # Parameter adaptation
            if t_iter % 1000 == 0:
                lr = max(lr * 0.9, 1e-6)               # gentle decay
                rel_tol = min(0.05, rel_tol + 0.005)   # broaden active set slowly

            # If we stagnate for a while, inject a tiny Gaussian jitter
            if no_improve >= 5000:
                jitter = np.random.randn(n) * 0.001
                a = np.clip(a + jitter, 0.0, None)
                a = enforce_symmetry(a)
                a = project_to_sum(a, target_sum)
                no_improve = 0
                print(f"Iter {t_iter:06d} ({descr}) – jitter applied")

            # Upper bound on iterations per restart (prevents endless loops)
            if t_iter >= 20000:
                break

        print(f"Run ({descr}) finished – best score so far {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 8000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        print("Fallback to uniform sequence.")

    print(f"Returning sequence of length {len(best_seq)} with score {best_score:.6f}")
    return best_seq.tolist()
