"""
Propose a step‐function sequence that minimizes
    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm works in three stages:

1. **Coarse parametric search** –  a family a_i = (i + s)^(-p) is explored
   over a few lengths (≈3‑12k) and a grid of (p, s).  The best
   candidate (lowest true max‑convolution) is kept as a warm start.

2. **Smoothed‑max gradient descent** –  Starting from the best candidate,
   we iteratively minimise the max‑convolution.
   * For early iterations a soft‑max (temperature γ) is used,
     giving a smooth approximation of the true max.
   * After γ is large enough we switch to the exact sub‑gradient
     (uniform weights on all arg‑max positions).
   * The gradient is obtained via an FFT based cross‑correlation,
     the step is projected onto the non‑negative orthant and the
     vector is rescaled to keep `sum(a) = sqrt(2·n)`.
   * Step size η decays slowly, γ grows, and occasional tiny jitter is
     added to escape plateaux.

3. **Final refinement** –  a short random local‑search phase tries tiny
   mass transfers between two coordinates; any improvement is kept.

The whole procedure respects the 1000 s budget and returns the best
sequence found as a plain Python list.
"""

import numpy as np
import time
from typing import List


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _compute_score(a: np.ndarray) -> float:
    """Fast evaluation of the objective (without the security wrapper)."""
    n = a.shape[0]
    nfft = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    # With the normalisation sum(a)=sqrt(2n) the denominator equals 2n,
    # therefore the score equals max_c.
    return float(max_c)


def _parametric_shape(n: int, p: float, s: float) -> np.ndarray:
    """
    Return a decreasing shape a_i = (i + s) ** (-p) normalised so that
    sum(a) = sqrt(2·n).
    """
    idx = np.arange(n, dtype=np.float64) + s
    a = idx ** (-p)
    target = np.sqrt(2.0 * n)
    a *= target / a.sum()
    return a


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """Return a list of non‑negative heights with a low evaluation score."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # a tiny safety margin

    best_a = None
    best_score = np.inf
    best_n = None

    # ------------------------------------------------------------------
    # 1. Coarse parametric search over a few lengths and (p, s) values.
    # ------------------------------------------------------------------
    lengths = [3000, 5000, 8000, 12000]               # reasonable sizes
    p_vals = np.linspace(0.20, 0.80, 31)              # 0.20 … 0.80
    s_vals = [0.0, 0.5, 1.0]

    for n in lengths:
        if time.time() >= deadline:
            break
        for s in s_vals:
            if time.time() >= deadline:
                break
            for p in p_vals:
                if time.time() >= deadline:
                    break
                a = _parametric_shape(n, p, s)
                score = _compute_score(a)
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    best_n = n
                    print(
                        f"[grid] n={n:5d}, s={s:3.1f}, p={p:5.3f} → score {score:.6f}"
                    )

    # ------------------------------------------------------------------
    # 2. If the grid failed (should not happen), fall back to the classic
    #    1/√(i+0.5) shape.
    # ------------------------------------------------------------------
    if best_a is None:
        n = 5000
        a = 1.0 / np.sqrt(np.arange(n, dtype=np.float64) + 0.5)
        target_sum = np.sqrt(2.0 * n)
        a *= target_sum / a.sum()
    else:
        a = best_a.copy()
        n = best_a.shape[0]
        target_sum = np.sqrt(2.0 * n)

    # Pre‑compute FFT sizes (they stay constant)
    nfft_conv = _next_pow2(2 * n - 1)
    nfft_corr = _next_pow2(3 * n - 2)

    # Hyper‑parameters for the smoothed‑max descent
    eta = 0.30          # initial step size
    eta_min = 0.001
    eta_decay = 0.9995

    gamma = 5.0         # soft‑max temperature
    gamma_max = 200.0
    gamma_inc = 1.001
    gamma_switch = 0.9 * gamma_max   # when to switch to exact arg‑max

    iteration = 0
    last_improve_iter = 0

    # ------------------------------------------------------------------
    # 2. Gradient descent on the true max‑convolution.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # ----- convolution of the current sequence -----
        af = np.fft.rfft(a, n=nfft_conv)
        conv = np.fft.irfft(af * af, n=nfft_conv)[: 2 * n - 1]
        max_conv = conv.max()          # current objective (since sum fixed)

        # ----- compute the weight vector w -----
        if gamma < gamma_switch:
            # smooth soft‑max
            w = np.exp(gamma * (conv - max_conv))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
        else:
            # exact sub‑gradient: uniform over all arg‑max positions
            idxs = np.flatnonzero(conv >= max_conv - 1e-12)
            w = np.zeros_like(conv)
            if idxs.size > 0:
                w[idxs] = 1.0 / idxs.size

        # ----- gradient via cross‑correlation (FFT) -----
        wf = np.fft.rfft(w, n=nfft_corr)
        af_rev = np.fft.rfft(a[::-1], n=nfft_corr)
        corr_full = np.fft.irfft(wf * af_rev, n=nfft_corr)
        # valid part corresponds to indices n‑1 … n‑1+n‑1
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]

        # ----- gradient step with projection and renormalisation -----
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.full_like(a_new, target_sum / n)
        else:
            a_new *= target_sum / s

        # occasional tiny random jitter (helps escape flat regions)
        if iteration % 73 == 0:
            jitter = 1e-4 * np.random.randn(n)
            a_new = np.maximum(a_new + jitter, 0.0)
            a_new *= target_sum / a_new.sum()

        # ----- evaluate the new candidate -----
        af_new = np.fft.rfft(a_new, n=nfft_conv)
        conv_new = np.fft.irfft(af_new * af_new, n=nfft_conv)[: 2 * n - 1]
        max_conv_new = conv_new.max()
        score_new = max_conv_new                     # same as max_conv_new

        # ----- keep the best overall sequence -----
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            last_improve_iter = iteration
            elapsed = time.time() - start_time
            print(
                f"Iter {iteration:6d}: new best {best_score:.6f}  (elapsed {elapsed:.1f}s)"
            )

        # ----- simple acceptance rule (always accept improvements) -----
        if score_new <= max_conv or np.random.rand() < 0.001:
            a = a_new

        # ----- hyper‑parameter updates -----
        eta = max(eta * eta_decay, eta_min)
        gamma = min(gamma * gamma_inc, gamma_max)

        # ----- early termination of the descent if stagnated -----
        if iteration - last_improve_iter > 5000:
            # no improvement for a while -> break to final refinement
            break

    # ------------------------------------------------------------------
    # 3. Very short random local search to polish the result.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # number of local moves limited by remaining time
        max_moves = int(1e5 * (deadline - time.time()) / budget_s)
        max_moves = max(2000, min(max_moves, 50000))
        a = best_a.copy()
        for _ in range(max_moves):
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            # propose moving a tiny amount δ from i to j
            delta = 1e-5 * (np.random.rand() - 0.5) * a[i]
            if delta > a[i]:
                delta = a[i] - 1e-12
            if delta < -a[j]:
                delta = -a[j] + 1e-12
            if abs(delta) < 1e-12:
                continue
            a_candidate = a.copy()
            a_candidate[i] -= delta
            a_candidate[j] += delta
            # no need to re‑normalise – sum is unchanged
            # evaluate
            af_c = np.fft.rfft(a_candidate, n=nfft_conv)
            conv_c = np.fft.irfft(af_c * af_c, n=nfft_conv)[: 2 * n - 1]
            score_c = conv_c.max()
            if score_c < best_score - 1e-12:
                best_score = score_c
                best_a = a_candidate.copy()
                a = a_candidate
                # occasional log
                if _ % 5000 == 0:
                    elapsed = time.time() - start_time
                    print(
                        f"Local move {_:6d}: improved to {best_score:.6f} (elapsed {elapsed:.1f}s)"
                    )

    # Final sanity check – ensure non‑negativity and correct normalisation
    best_a = np.maximum(best_a, 0.0)
    best_a *= (np.sqrt(2.0 * n) / best_a.sum())
    return best_a.tolist()
