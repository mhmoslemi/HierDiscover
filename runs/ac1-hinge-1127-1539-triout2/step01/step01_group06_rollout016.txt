def _project_simplex(v, s):
    # ensure non-negative after shift
    if s <= 0: return np.zeros_like(v)
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    # find rho = max {j | u_j - cssv_j/j > 0}
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative vector) that minimises
evaluate_sequence() = 2·n·max(conv(a,a))/sum(a)^2.

The algorithm consists of two phases:

1. **Randomised shape search** – vectors of the form
       a_i ∝ (i+β)^(-p)·exp(-q·i)
   are generated with random (p,q,β).  Each candidate is
   normalised to the required ℓ1‑norm √(2n) and evaluated by a
   fast FFT based convolution.  The best candidate seen is stored.

2. **Projected sub‑gradient descent** – starting from the best
   candidate, we iteratively reduce the current maximum convolution
   value.  The sub‑gradient of the max‑convolution is
       g_i = 2·a_{k_max-i}   (only for indices participating in the
   maximal convolution entry k_max).  A step in the opposite
   direction is taken, the vector is clipped to non‑negativity and
   renormalised (projection onto the simplex of sum √(2n)).  The step
   size follows the classic 1/√t schedule.  Occasionally a small
   random jitter is added to escape plateaus.

Both phases run until the overall time budget (default 1000 s) is
exhausted.  The best sequence found is returned as a plain Python
list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all are top‑level, no closures or lambdas
# ----------------------------------------------------------------------


def _fft_size(n: int) -> int:
    """Return the next power‑of‑two length suitable for a convolution of length n."""
    return 1 << ((2 * n - 1).bit_length())


def _full_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution a*a using a real FFT of given size."""
    # real‑valued FFT is faster than complex FFT
    fa = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(fa * fa, n=size_fft)[: 2 * a.size - 1]
    return conv


def _max_convolution(a: np.ndarray, size_fft: int) -> float:
    """Maximum entry of a*a (FFT based)."""
    conv = _full_convolution(a, size_fft)
    return float(conv.max())


def _initial_powerlaw(n: int, p: float) -> np.ndarray:
    """Deterministic start: a_i ∝ (i+1)^{-p}, renormalised to √(2n)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = np.sqrt(2.0 * n)
    a *= s / a.sum()
    return a


def _make_shape(n: int, p: float, q: float, beta: float) -> np.ndarray:
    """
    Generate a candidate shape a_i = (i+β)^{-p}·exp(-q·i) and renormalise.
    Parameters:
        p    – power‑law exponent (≈0.5)
        q    – exponential decay rate (≈0 … 0.02)
        beta – shift of the index (≈0 … 5)
    """
    idx = np.arange(n, dtype=np.float64)
    # avoid division by zero when beta is zero
    a = (idx + beta) ** (-p)
    if q != 0.0:
        a *= np.exp(-q * idx)
    total = a.sum()
    if total == 0.0:
        return None
    a *= np.sqrt(2.0 * n) / total
    return a


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}.
    Standard O(n log n) algorithm (see Duchi et al. 2008).
    """
    if s <= 0:
        return np.zeros_like(v)
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    # find the last index where u_i > cssv_i / (i+1)
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
    else:
        rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient_descent(
    a0: np.ndarray,
    size_fft: int,
    deadline: float,
    step_factor: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent for min max(conv(a,a)).
    Returns the best vector encountered and its max‑convolution value.
    """
    a = a0.copy()
    n = a.size
    s = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _max_convolution(a, size_fft)

    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        # full convolution to locate the active index
        conv = _full_convolution(a, size_fft)
        k_max = int(conv.argmax())
        cur_val = float(conv.max())
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}]  sub‑grad iter {it} new best = {best_val:.6f}")

        # sub‑gradient of the active quadratic form
        g = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        g[idx] = 2.0 * a[k_max - idx]

        norm = np.linalg.norm(g)
        if norm == 0.0:
            break

        # classic step size = step_factor / (‖g‖·√t)
        step = step_factor / (norm * np.sqrt(it + 1))

        # gradient step, non‑negativity, and projection onto the simplex
        v = a - step * g
        v = np.maximum(v, 0.0)
        a = _project_onto_simplex(v, s)

        it += 1

        # occasional small random jitter to avoid stagnation
        if it % 500 == 0:
            noise = rng.normal(scale=0.01 * s, size=n)
            a = _project_onto_simplex(a + noise, s)

    return best_a, best_val


def _local_mass_transfer(
    a0: np.ndarray,
    size_fft: int,
    deadline: float,
    max_frac: float = 0.12,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing (mass‑transfer) used as a final polish.
    Only moves that strictly improve the objective are accepted.
    """
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_val = _max_convolution(a, size_fft)
    cur_val = best_val
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        frac = rng.random() * max_frac
        delta = a[i] * frac
        if delta <= 0.0:
            continue
        a[i] -= delta
        a[j] += delta
        new_val = _max_convolution(a, size_fft)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}]  local move improved to {best_val:.6f}")
        else:
            a[j] -= delta
            a[i] += delta

    return best_a, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function sequence that minimises evaluate_sequence().
    The routine spends the allotted time performing a randomised shape search
    followed by a projected sub‑gradient descent and a brief local polish.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0  # tiny safety margin

    # ------------------------------------------------------------------
    # Configuration
    # ------------------------------------------------------------------
    n = 3000                        # length of the vector (adjustable)
    size_fft = _fft_size(n)         # FFT length reused throughout
    best_seq = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # Phase 1 – randomised shape sampling
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    while time.time() < deadline - 30.0:   # leave ~30 s for later phases
        # draw parameters (slightly broader than the power‑law region)
        p = rng.uniform(0.35, 0.70)       # power exponent
        q = rng.uniform(0.0, 0.02)        # exponential decay
        beta = rng.uniform(0.0, 5.0)      # index shift
        a = _make_shape(n, p, q, beta)
        if a is None:
            continue
        val = _max_convolution(a, size_fft)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[{time.time():.1f}]  random shape p={p:.3f} q={q:.4f} β={beta:.2f} => {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – sub‑gradient refinement starting from the best so far
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback to a plain power‑law if random sampling failed
        best_seq = _initial_powerlaw(n, 0.5)

    sub_deadline = min(deadline, time.time() + 350.0)   # up to ~350 s
    refined_seq, refined_val = _subgradient_descent(
        best_seq, size_fft, sub_deadline, step_factor=0.5
    )
    if refined_val < best_val:
        best_val = refined_val
        best_seq = refined_seq.copy()
        print(f"[{time.time():.1f}]  after sub‑gradient best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – short local mass‑transfer polish (uses the same FFT size)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        local_deadline = deadline
        polished_seq, polished_val = _local_mass_transfer(
            best_seq, size_fft, local_deadline, max_frac=0.10
        )
        if polished_val < best_val:
            best_val = polished_val
            best_seq = polished_seq.copy()
            print(f"[{time.time():.1f}]  final polished best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing – strip trailing zeros and return as Python list
    # ------------------------------------------------------------------
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    return best_seq.tolist()
