def project_to_simplex(v, s):
    # assume v is np.ndarray.
    # implementation from https://arxiv.org/abs/1101.6081 (Algorithm 1)
    if s < 0:
        raise ValueError
    # Sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    # Find rho
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def softmax_step(a, target_sum, alpha, eta):
    # compute conv
    conv = conv_self_fft(a)
    # compute softmax weights
    max_c = conv.max()
    exps = np.exp(alpha * (conv - max_c))
    sum_exps = exps.sum()
    # compute correlation gradient
    # compute correlation of exps with a, normalized
    L = 1 << ((len(exps) + len(a) - 1).bit_length())
    # FFT for correlation
    Ex = np.fft.rfft(exps, n=L)
    A = np.fft.rfft(a, n=L)
    # correlation = ifft(conj(Ex)*A) -> real array
    corr = np.fft.irfft(np.conj(Ex) * A, n=L)[:len(a)]
    grad = 2 * corr / sum_exps

    a_new = a - eta * grad

    a_new = np.maximum(a_new, 0)
    a_new = project_to_simplex(a_new, target_sum)
    return a_new

"""
Propose a non‑negative height sequence that minimises the
self‑convolution bound used in `evaluate_sequence`.

The optimiser works in three stages:

1️⃣ **Parametric sweep** – simple symmetric “U‑shapes’’ (linear,
   quadratic and exponential) are tried.  The best of those shapes
   (or the provided `height_sequence_1`) is used as a start.

2️⃣ **Soft‑max gradient descent** – the objective
   `max_k (a*a)[k]` is approximated by a smooth log‑sum‑exp:
   `L(a)= (1/α)·log Σ exp(α·conv)`.  Its gradient can be written as a
   weighted correlation of the soft‑max weights with the current
   sequence.  Using FFT the gradient is computed cheaply, a projected
   gradient step is taken and the vector is projected onto the simplex
   `sum a = √(2 n)`.  The temperature `α` is increased gradually, the
   step size is decayed, and occasional random jitter is added to
   escape plateaus.

3️⃣ **Optional sub‑gradient polishing** – a few classic sub‑gradient
   steps (active‑set on the maximal convolution index) are performed
   with a small step size to fine‑tune the final answer.

All numbers stay non‑negative, the ℓ₁‑norm is fixed, and the routine
stops after the allotted time budget (≈1000 s), returning the best
sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution of a with itself using FFT.
    Returns the full convolution of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())            # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { w | w_i ≥ 0 , Σ w_i = s }.
    Implements the algorithm of Duchi et al. (2008).
    """
    if s <= 0:
        raise ValueError("Simplex sum must be positive")
    # If already feasible, return unchanged (fast path)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def generate_U_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
) -> np.ndarray:
    """
    Produce a symmetric “U‑shaped’’ sequence of length n.
    kind ∈ {linear, quadratic, exponential}.
    param controls the strength of the edge‑enhancement (≥0).
    The result is scaled so that the ℓ₁‑norm equals target_sum.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_to_simplex(raw, target_sum)


def softmax_gradient_step(
    a: np.ndarray,
    target_sum: float,
    alpha: float,
    eta: float,
    L_fft: int,
) -> np.ndarray:
    """
    One projected gradient step on the smooth approximation
          L(a) = (1/α) log Σ_k exp(α·(a*a)[k]).
    The gradient is 2·corr(p, a) / Σ p where p_k = exp(α·conv_k) .
    """
    conv = conv_self_fft(a)                # length 2n‑1
    max_c = conv.max()
    # soft‑max weights (numerically stable)
    w = np.exp(alpha * (conv - max_c))
    w_sum = w.sum()
    # correlation of w with a (linear‑time via FFT)
    # correlation = ifft(conj(fft(w)) * fft(a))
    W = np.fft.rfft(w, n=L_fft)
    A = np.fft.rfft(a, n=L_fft)
    corr = np.fft.irfft(np.conj(W) * A, n=L_fft)[: a.shape[0]]
    grad = 2.0 * corr / w_sum

    # gradient descent step
    a_new = a - eta * grad
    a_new = np.maximum(a_new, 0.0)
    a_new = project_to_simplex(a_new, target_sum)
    return a_new


def subgradient_step(
    a: np.ndarray,
    target_sum: float,
    eta: float,
) -> np.ndarray:
    """Classic sub‑gradient step using only the maximal‑index."""
    n = a.shape[0]
    conv = conv_self_fft(a)
    idx = np.argmax(conv)               # single active index
    # derivative of (a*a)[idx] w.r.t. a_i is 2·a_{idx‑i} if in range
    i = np.arange(n, dtype=np.int64)
    j = idx - i
    mask = (j >= 0) & (j < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[j[mask]]
    # make the direction orthogonal to the all‑ones vector
    direction = -g + np.mean(g)
    a_new = a + eta * direction
    a_new = np.maximum(a_new, 0.0)
    a_new = project_to_simplex(a_new, target_sum)
    return a_new
