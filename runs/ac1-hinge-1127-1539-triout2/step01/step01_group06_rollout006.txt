def directed_search(a, deadline, rng):
    n = len(a)
    best = a.copy()
    best_val = _score(a)
    cur_val = best_val
    step_frac = 0.15
    eps = 1e-12
    while time.time() < deadline:
        # compute conv and max
        conv = _conv_full(a) # use _max_convolution but also need conv array to get indices
        max_val = conv.max()
        max_idxs = np.where(np.abs(conv - max_val) < 1e-12)[0]  # building list
        # compute gradient
        grad = np.zeros_like(a)
        for k in max_idxs:
            i_start = max(0, k - (n-1))
            i_end = min(n-1, k)
            i_range = np.arange(i_start, i_end+1)
            j_range = k - i_range
            grad[i_range] += a[j_range]

        # choose source and dest
        src = np.argmax(grad)
        dst = np.argmin(grad)

        if a[src] <= eps:
            # pick random src?
            src = rng.integers(0, n)
            continue

        # propose delta
        delta = a[src] * step_frac * rng.random()
        a[src] -= delta
        a[dst] += delta

        new_val = _score(a)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = a.copy()
                print(f"{time.time():.1f} Improved = {best_val:.6f}")
        else:
            # reject
            a[dst] -= delta
            a[src] += delta

    return best, best_val

def _full_convolution(a: np.ndarray) -> np.ndarray:
    size = 1 << ((2*a.size - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa*fa, n=size)[:2*a.size-1]
    return conv

def _initial_sequence(p: float, n: int) -> np.ndarray:
    a = (np.arange(1, n+1, dtype=np.float64) ** (-p))
    a *= np.sqrt(2.0*n) / a.sum()
    return a

def _score(a: np.ndarray) -> float:
    n = a.size
    conv = _full_convolution(a)
    max_val = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0*n*max_val/(s*s)

def _descent_search(a: np.ndarray, deadline: float, rng) -> tuple[np.ndarray,float]:
    n = a.size
    s = a.sum()
    # compute initial conv
    conv = _full_convolution(a)
    max_val = conv.max()
    cur_score = 2.0*n*max_val/(s*s)
    best_seq = a.copy()
    best_score = cur_score

    # parameters
    while time.time() < deadline:
        # compute gradient for current max indices
        max_val = conv.max()
        # threshold: treat values within 1e-12 of max
        max_idxs = np.where(conv >= max_val - 1e-12)[0]
        grad = np.zeros_like(a)
        # accumulate contributions
        for k in max_idxs:
            i_start = max(0, k-(n-1))
            i_end = min(n-1, k)
            i_range = np.arange(i_start, i_end+1)
            j_range = k - i_range
            grad[i_range] += a[j_range]

        # candidate source/dest
        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))

        if a[src] <= 0:
            # trivial fallback to random
            src = rng.integers(0, n)
            continue

        # propose delta
        frac = rng.random() * 0.2  # up to 20% of source
        delta = a[src] * frac
        if delta <= 0:
            continue

        a[src] -= delta
        a[dst] += delta

        # recompute conv and score
        new_conv = _full_convolution(a)
        new_max = new_conv.max()
        new_score = 2.0*n*new_max/(s*s)

        # accept if better (strict) else revert
        if new_score < cur_score - 1e-12:
            cur_score = new_score
            conv = new_conv
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] improved to {best_score:.6f}")
        else:
            # revert
            a[dst] -= delta
            a[src] += delta

        # occasional random perturbation to escape stagnation
        if rng.random() < 0.01:
            # random move
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0:

                a[j] += delta2

    return best_seq, best_score

"""
Step‑function optimizer for the convolution bound.

We minimise  
    2·n·max(conv(a,a)) / (sum(a)²)  

by keeping the ℓ¹‑norm fixed to √(2·n) (so the objective equals the
maximum convolution value).  
The algorithm combines:

* deterministic power‑law initialisations a_i ∝ (i+1)^‑p,
* a fast FFT‑based full convolution,
* a greedy “gradient” descent that repeatedly moves a small fraction of
  mass from the index that contributes most to the current peak of the
  convolution to the index that contributes the least,
* occasional random perturbations and many restarts with different
  lengths and exponents.

The search stops after the allotted time (default 1000 s) and returns the
best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a using a power‑of‑two FFT.
    Returns the full array of length 2·len(a)‑1.
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Compute the objective used by the competition for a non‑negative
    vector a.  The ℓ¹‑norm is not forced to √(2·n) here; the formula
    matches evaluate_sequence.
    """
    n = a.size
    conv = _full_convolution(a)
    max_val = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_val / (s * s)


def _initial_sequence(p: float, n: int) -> np.ndarray:
    """
    Power‑law start: a_i ∝ (i+1)^‑p, renormalised so that
    sum(a)=√(2·n).  All entries are non‑negative.
    """
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= np.sqrt(2.0 * n) / a.sum()
    return a


def _descent_search(a: np.ndarray, deadline: float, rng: np.random.Generator
                    ) -> tuple[np.ndarray, float]:
    """
    Greedy descent that respects the total sum.
    At each step we identify the convolution index where the current peak
    occurs, compute a simple gradient (the contribution of every entry to
    that peak) and move a fraction of mass from the highest‑gradient
    entry to the lowest‑gradient entry.  Moves are kept only if they
    strictly improve the objective.  Random perturbations are added
    occasionally to escape plateaus.
    """
    a = a.copy()
    n = a.size
    s = a.sum()                     # constant (floating point noise only)
    conv = _full_convolution(a)
    cur_score = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    best_score = cur_score

    # parameters
    max_frac = 0.20                 # up to 20 % of a source entry per move
    random_chance = 0.01           # probability of a pure random perturbation
    eps = 1e-12

    while time.time() < deadline:
        # locate the (possibly several) indices attaining the current max
        max_val = conv.max()
        max_idxs = np.where(conv >= max_val - 1e-12)[0]

        # gradient: how much each a[i] contributes to the current peak(s)
        grad = np.zeros_like(a)
        for k in max_idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1, dtype=np.int32)
            j_range = k - i_range
            grad[i_range] += a[j_range]

        # source = index with largest contribution, destination = smallest
        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))

        # if the source is effectively empty, fall back to a random choice
        if a[src] <= eps:
            src = rng.integers(0, n)
            if a[src] <= eps:
                continue

        # propose a mass transfer
        frac = rng.random() * max_frac
        delta = a[src] * frac
        if delta <= 0.0:
            continue

        a[src] -= delta
        a[dst] += delta

        # recompute convolution and score
        new_conv = _full_convolution(a)
        new_score = 2.0 * n * new_conv.max() / (s * s)

        if new_score < cur_score - 1e-12:
            # accept improvement
            conv = new_conv
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}]  improved score = {best_score:.6f}")
        else:
            # reject – restore the previous state
            a[dst] -= delta
            a[src] += delta

        # occasional random move to avoid getting stuck
        if rng.random() < random_chance:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > eps:
                delta2 = a[i] * rng.random() * max_frac
                a[i] -= delta2
                a[j] += delta2
                conv = _full_convolution(a)
                cur_score = 2.0 * n * conv.max() / (s * s)

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises evaluate_sequence().
    The routine runs for at most ``budget_s`` seconds, trying several
    lengths and power‑law exponents, then refines each candidate with the
    greedy descent described above.  The best sequence discovered is
    returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # If a previous construction is available, use it as a starting point.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            prev = np.array(globals()['height_sequence_1'], dtype=np.float64)
            prev = np.maximum(prev, 0.0)
            if prev.size > 0 and prev.sum() > 0.0:
                n0 = prev.size
                prev = prev * np.sqrt(2.0 * n0) / prev.sum()
                prev_score = _score(prev)
                best_seq = prev.copy()
                best_score = prev_score
                print(f"Loaded previous sequence, score = {prev_score:.6f}")
        except Exception:
            pass

    # --------------------------------------------------------------
    # Parameter grid – longer vectors tend to give lower bounds.
    # --------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000]          # stay well below 20000
    exponents = [0.45, 0.5, 0.55, 0.6, 0.65]

    # ------------------------------------------------------------------
    # Main optimisation loop.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        for n in lengths:
            for p in exponents:
                if time.time() >= deadline:
                    break

                # initialise from a power‑law profile
                a = _initial_sequence(p, n)
                cur_score = _score(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"Start (n={n}, p={p})  score = {cur_score:.6f}")

                # allocate a short time slice for local refinement
                sub_deadline = min(deadline, time.time() + 5.0)
                a_opt, opt_score = _descent_search(a, sub_deadline, rng)

                if opt_score < best_score:
                    best_score = opt_score
                    best_seq = a_opt.copy()
                    print(f"After descent (n={n}, p={p})  best = {opt_score:.6f}")

            if time.time() >= deadline:
                break

        # ------------------------------------------------------------------
        # If we still have time, perform a few random restarts.
        # ------------------------------------------------------------------
        if time.time() < deadline:
            n = rng.integers(2000, 12001)
            a = rng.random(n)
            a *= np.sqrt(2.0 * n) / a.sum()
            cur_score = _score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"Random init (n={n})  score = {cur_score:.6f}")

            sub_deadline = min(deadline, time.time() + 3.0)
            a_opt, opt_score = _descent_search(a, sub_deadline, rng)
            if opt_score < best_score:
                best_score = opt_score
                best_seq = a_opt.copy()
                print(f"Random descent  best = {opt_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _initial_sequence(0.5, 2000)

    # Trim trailing zeros (tiny values) for neatness
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    return best_seq.tolist()
