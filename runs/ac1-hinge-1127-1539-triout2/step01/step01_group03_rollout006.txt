    best_score = _objective(a)

    n = len(a)
    while time.time() < time_deadline:
        # conv via FFT
        m = 1 << ((2 * n - 1).bit_length())
        fft_a = np.fft.rfft(a, n=m)
        conv = np.fft.irfft(fft_a * fft_a, n=m)[:2*n-1]
        max_conv = conv.max()
        # Get top_k indices
        if top_k >= len(conv):
            top_indices = np.arange(len(conv))
        else:
            top_indices = np.argpartition(conv, -top_k)[-top_k:]
        # Compute gradient
        grad = np.zeros_like(a)
        for k in top_indices:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            shift = np.zeros_like(a)
            shift[mask] = a[idx[mask]]

            best_score = new_score

            a = new_a

"""
Propose a low‑score step‑function.

The algorithm works in three stages:
1. **Parameter sweep** – a two‑parameter family
       a_i = 1 / (i + 1 + c) ** beta
   is explored.  For each β a 1‑D golden‑section search finds the offset
   c that minimizes the evaluation score.  The best (β,c) pair is kept.
2. **Smooth max‑gradient descent** – starting from the best candidate,
   a sub‑gradient of the maximum convolution value is computed using the
   top‑K largest convolution entries (a simple smooth approximation of
   the max).  An additive step is taken, the vector is projected back
   onto the simplex (non‑negative, fixed ℓ₁‑norm) and the step size is
   adapted.  Random swaps are injected when progress stalls.
3. **Final polishing** – a short simulated‑annealing style random walk
   makes a few more local perturbations.

All work respects the provided time budget (default 1000 s).  The returned
list contains non‑negative numbers; the evaluation routine will clip them
to the allowed range before computing the final score.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast evaluation of the objective (identical to `evaluate_sequence`
# but without the security checks).  Uses FFT for O(n log n) convolution.
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    n = len(seq)
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    # length of FFT: next power of two for speed
    m = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft * fft, n=m)[: 2 * n - 1]
    max_conv = np.max(conv)
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# 1‑D golden‑section search for a scalar (offset `c`) that minimises the
# objective for a fixed exponent `beta`.
# ----------------------------------------------------------------------
def _golden_section(beta: float, n: int, lo: float = 0.0, hi: float = 5.0,
                    tol: float = 1e-4, max_iter: int = 30) -> tuple[float, float]:
    phi = (1 + np.sqrt(5.0)) / 2.0          # golden ratio
    resphi = 2.0 - phi                      # = 1/phi² ≈ 0.381966
    c1 = hi - resphi * (hi - lo)
    c2 = lo + resphi * (hi - lo)

    idx = np.arange(n, dtype=np.float64) + 1.0  # i+1 part, offset added later
    def score(c):
        a = 1.0 / ((idx + c) ** beta)
        return _objective(a)

    f1 = score(c1)
    f2 = score(c2)

    it = 0
    while (hi - lo) > tol and it < max_iter:
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - resphi * (hi - lo)
            f1 = score(c1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + resphi * (hi - lo)
            f2 = score(c2)
        it += 1

    if f1 < f2:
        return c1, f1
    else:
        return c2, f2


# ----------------------------------------------------------------------
# Gradient descent on the shape using a smooth approximation of the max.
# ----------------------------------------------------------------------
def _refine_sequence(a0: np.ndarray,
                     deadline: float,
                     target_score: float = 1.503,
                     top_k: int = 5) -> tuple[np.ndarray, float]:
    """Iteratively reduce the score of `a0` until time runs out."""
    n = len(a0)
    # pre‑compute the FFT length (constant)
    m = 1 << ((2 * n - 1).bit_length())

    a = a0.copy()
    target_sum = np.sum(a)
    best_seq = a.copy()
    best_score = _objective(best_seq)

    step_factor = 0.5       # initial step multiplier
    no_improve = 0

    while time.time() < deadline:
        # Full convolution via FFT (re‑used each iteration)
        fft_a = np.fft.rfft(a, n=m)
        conv = np.fft.irfft(fft_a * fft_a, n=m)[: 2 * n - 1]
        max_conv = np.max(conv)

        # ---- compute a smooth sub‑gradient ----
        # Pick the `top_k` largest convolution entries (a cheap smooth max)
        if top_k >= conv.size:
            top_idx = np.arange(conv.size)
        else:
            top_idx = np.argpartition(conv, -top_k)[-top_k:]

        grad = np.zeros_like(a)
        for k in top_idx:
            # shift = a shifted by k (i.e. a_{k-i}) with zero padding
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            shift = np.zeros_like(a)
            shift[mask] = a[idx[mask]]
            grad += a + shift               # g_i = a_i + a_{k-i}
        grad /= len(top_idx)

        # ---- step size ----
        # r = max_conv / (sum(a)^2) appears in the exact gradient of the score
        r = max_conv / (np.sum(a) ** 2)
        step = step_factor * r

        # ---- additive update + projection onto the simplex ----
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)

        s = np.sum(a_new)
        if s > 1e-12:
            a_new *= target_sum / s
        else:
            break          # all numbers vanished – abort

        new_score = _objective(a_new)

        if new_score < best_score:
            best_score = new_score
            best_seq = a_new.copy()
            a = a_new
            no_improve = 0
            step_factor = min(step_factor * 1.1, 2.0)   # gently increase step
        else:
            no_improve += 1
            # occasional random mass‑swap to escape plateaus
            if no_improve % 20 == 0:
                i, j = np.random.randint(n, size=2)
                delta = a[i] * 0.05 * np.random.rand()
                a_sw = a.copy()
                a_sw[i] = max(0.0, a[i] - delta)
                a_sw[j] += delta
                a_sw *= target_sum / np.sum(a_sw)
                a = a_sw
                step_factor *= 0.9

        if best_score < target_score:
            # good enough – stop early
            break

    return best_seq, best_score

    best_seq = None
    best_score = np.inf

    # β in a generous window around 0.5
    betas = np.linspace(0.44, 0.56, 25)
    for beta in betas:
        c_opt, score_c = _golden_section(beta, n, lo=0.0, hi=6.0,
                                         tol=5e-4, max_iter=25)
        if score_c < best_score:
            best_score = score_c
            best_seq = (1.0 / ((np.arange(n, dtype=np.float64) + 1.0 + c_opt) ** beta))
            print(f"[grid] beta={beta:.4f}, c={c_opt:.4f} → score={score_c:.6f}")

        if time.time() > deadline:
            break

    # If the sweep failed (very unlikely) fall back to a uniform sequence
    if best_seq is None:
        best_seq = np.ones(n, dtype=np.float64)
        best_score = _objective(best_seq)

    print(f"[grid] best after sweep: score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Refine using smooth‑max gradient descent.
    # --------------------------------------------------------------
    refined_seq, refined_score = _refine_sequence(best_seq, deadline,
                                                  target_score=1.503,
                                                  top_k=7)
    if refined_score < best_score:
        best_score = refined_score
        best_seq = refined_seq
        print(f"[refine] improved to score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Final short random walk (simulated‑annealing style) for extra
    #    polishing.  Very cheap – only a few hundred attempts.
    # --------------------------------------------------------------
    attempts = 300
    target_sum = np.sum(best_seq)
    a = best_seq.copy()
    for _ in range(attempts):
        if time.time() > deadline:
            break
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = a[i] * 0.03 * np.random.rand()
        a_new = a.copy()
        a_new[i] = max(0.0, a[i] - delta)
        a_new[j] += delta
        a_new *= target_sum / np.sum(a_new)
        new_score = _objective(a_new)
        if new_score < best_score:
            best_score = new_score
            best_seq = a_new.copy()
            a = a_new
            # occasional verbose feedback
            if int(best_score * 1000) % 10 == 0:
                print(f"[rand] new best score={best_score:.6f}")

    print(f"[+] Search finished. Best score = {best_score:.6f}")
    # Convert to plain Python floats as required by the evaluation harness
    return [float(x) for x in best_seq]
