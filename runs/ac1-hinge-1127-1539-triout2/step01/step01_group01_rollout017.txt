"""
search for a non‑negative step function that minimises `evaluate_sequence`.

The programme follows three stages:

1. **Broad initialisation** – a collection of parametric “U‑shapes” and a few
   random Dirichlet vectors is examined; the best among them is kept.
2. **Deterministic sub‑gradient descent** – repeatedly move the current
   sequence opposite to the sub‑gradient of the maximal auto‑convolution
   (only the (almost) maximal index is used, which makes the update cheap).
   The step size is large at the beginning and decays slowly; the ℓ₁‑norm
   (the sum) is kept constant by orthogonal projection.
3. **Stochastic exploration** – occasional random pair‑wise mass transfers,
   Gaussian jitter and a simple simulated‑annealing acceptance rule are used
   to escape plateaus.  All changes preserve the ℓ₁‑norm (up to tiny rounding
   errors that are re‑projected).

The whole search respects a user‑provided time budget (≈ 1000 s).  The
function returns the best sequence found as a plain Python list of floats.
"""

import time
import numpy as np


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution of a with itself using an FFT of power‑of‑two length."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())        # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector to have ℓ₁‑norm = target."""
    s = v.sum()
    if s > 0.0:
        return v * (target / s)
    # fallback – uniform distribution (should never happen)
    return np.full_like(v, target / v.shape[0])


def generate_U_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
) -> np.ndarray:
    """
    Produce a symmetric “U‑shape’’ of length n.
    kind ∈ {linear, quadratic, exponential, power}.
    param ≥ 0 controls the edge emphasis; for ``power`` the exponent is
    taken from param (the multiplicative factor is fixed to 1).
    The result is scaled to ℓ₁‑norm = target_sum.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "power":
        # param plays the role of the exponent (≥1)
        raw = 1.0 + d ** max(param, 0.0)
    else:
        raise ValueError(f"unknown shape kind {kind!r}")

    raw = np.clip(raw, 0.0, None)
    return project_to_sum(raw, target_sum)


def subgradient_step(
    a: np.ndarray,
    target_sum: float,
    eta: float,
    rel_tol: float = 0.0,
) -> np.ndarray:
    """
    One projected sub‑gradient step that reduces the current maximum of the
    self‑convolution.  *rel_tol* determines the active set:
    if >0, all convolution entries ≥ (1‑rel_tol)·max are treated as active.
    """
    n = a.shape[0]
    conv = conv_self_fft(a)
    max_val = conv.max()

    if rel_tol <= 0.0:
        act = np.array([np.argmax(conv)], dtype=np.int64)
    else:
        act = np.where(conv >= max_val * (1.0 - rel_tol))[0]

    # build the sub‑gradient (see the derivation above)
    g = np.zeros_like(a)
    idx = np.arange(n, dtype=np.int64)
    for k in act:
        j = k - idx
        mask = (j >= 0) & (j < n)
        # each contribution appears twice (a_i a_j and a_j a_i)
        g[mask] += 2.0 * a[j[mask]]

    # make the direction orthogonal to the all‑ones vector (preserve sum)
    direction = -g + np.mean(g)
    a_new = a + eta * direction
    a_new = np.clip(a_new, 0.0, None)
    a_new = project_to_sum(a_new, target_sum)
    return a_new


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.02) -> np.ndarray:
    """
    Transfer a small fraction of mass from a random source index to a random
    destination index.  The total ℓ₁‑norm stays unchanged (up to rounding).
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    if a[i] <= 0.0:
        return a
    delta = scale * a[i]                     # a fraction of the source entry
    a[i] -= delta
    a[j] += delta
    return project_to_sum(a, target_sum)


def random_dirichlet(n: int, target_sum: float) -> np.ndarray:
    """Generate a random non‑negative vector with the prescribed ℓ₁‑norm."""
    raw = np.random.exponential(1.0, size=n)
    return project_to_sum(raw, target_sum)

    # ------------------------------------------------------------------

    temperature = 0.02 * (best_score - 1.0 + 1e-3)   # a few percent of the excess over 1
    stagnation = 0

    print(f"[{time.time():.2f}] Starting stochastic annealing (temp={temperature:.6e})")
    while time.time() < deadline:
        r = np.random.rand()
        if r < 0.55:
            # deterministic sub‑gradient step (smaller eta now)
            a_candidate = subgradient_step(cur_seq, target_sum, eta * 0.5, rel_tol=rel_tol)
        elif r < 0.80:
            # random pair transfer
            a_candidate = random_pair_transfer(cur_seq, target_sum, scale=0.022)
        else:
            # Gaussian jitter on the whole vector
            jitter = np.random.randn(n) * (0.0015 * avg)
            a_candidate = np.clip(cur_seq + jitter, 0.0, None)
            a_candidate = project_to_sum(a_candidate, target_sum)

        cand_score = evaluate_sequence(a_candidate.tolist())

        # Acceptance rule (always accept improvements)
        if cand_score < cur_score:
            cur_seq, cur_score = a_candidate, cand_score
            stagnation = 0
        else:
            # Metropolis‑type acceptance
            if temperature > 0.0 and np.random.rand() < np.exp(-(cand_score - cur_score) / temperature):
                cur_seq, cur_score = a_candidate, cand_score
                stagnation = 0
            else:
                stagnation += 1

        # keep the global best
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = cur_seq.copy()
            print(f"Anneal iter {iteration:06d} – new best: {best_score:.6f}")

        # slow cooling
        if iteration % 500 == 0:
            temperature = max(temperature * 0.97, 1e-6)
            eta = max(eta * 0.96, 1e-9)

        # occasional rejuvenation after long stagnation
        if stagnation >= 3000:
            jitter = np.random.randn(n) * (0.003 * avg)
            cur_seq = np.clip(best_seq + jitter, 0.0, None)
            cur_seq = project_to_sum(cur_seq, target_sum)
            cur_score = evaluate_sequence(cur_seq.tolist())
            stagnation = 0
            print(f"Anneal iter {iteration:06d} – rejuvenation applied")

        iteration += 1

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as a plain Python list)
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Search finished – best score {best_score:.6f}")
    return best_seq.tolist()
