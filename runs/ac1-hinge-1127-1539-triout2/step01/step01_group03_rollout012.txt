"""
Propose a low‑score step function.

Algorithm
---------
1. **Parameterised seeding** –  a family a_i = (i+1+c)^(-β) is tried on a grid
   of exponents β∈[0.40,0.80] and offsets c∈[0,20].  Each candidate is
   normalised to Σa = √(2n) (this makes the evaluation equal to the maximal
   auto‑convolution value) and the best is kept.

2. **Active‑set sub‑gradient descent** –  at every iteration the convolution
   `c = a*a` is computed by FFT.  All indices whose value is within a tiny
   relative tolerance of the current maximum constitute the *active set*.
   The sub‑gradient of the maximum is the sum of the gradients of these
   active constraints:
       ∂(a*a)_k/∂a_i = a_i + a_{k-i}  (zero when the index is out of range).

   A step in the negative direction is taken, re‑normalising the ℓ₁‑norm
   after each update (the objective is scale‑invariant).

3. **Escaping plateaus** –  when no improvement is seen for a while a random
   mass transfer between two positions is performed and the step size is
   slightly reduced.

The whole search runs for the allotted time (≤ budget_s seconds) and returns
the best sequence found.  The code uses only NumPy and SciPy (for a fast
sparse linear‑programming boost is optional but not required).  It is fully
self‑contained and respects the security checks of ``evaluate_sequence``.
"""

import time
import numpy as np
import random

# --------------------------------------------------------------
# Fast objective used during the search (identical to evaluate_sequence
# up to the clipping performed inside that function).
def _objective(seq: np.ndarray, m_fft: int) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)**2) using an FFT."""
    n = len(seq)
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    # full linear convolution via FFT (real‑valued signal)
    fft = np.fft.rfft(seq, n=m_fft)
    conv = np.fft.irfft(fft * fft, n=m_fft)[: 2 * n - 1]
    return float(2.0 * n * np.max(conv) / (s * s))


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds.  The function returns well before this limit.

    Returns
    -------
    list[float]
        The best sequence found.
    """
    np.random.seed(seed)
    random.seed(seed)

    deadline = time.time() + budget_s - 2.0   # tiny safety margin

    # ----------------------------------------------------------
    # 1. Choose a length.
    #    Larger n gives a finer approximation of the continuous
    #    optimum, but increases the cost of each FFT.  n = 8000 is a
    #    good trade‑off for the allotted time.
    # ----------------------------------------------------------
    n = 8000
    print(f"[+] Starting search, length = {n}")

    # length of the zero‑padded FFT (next power of two)
    m_fft = 1 << ((2 * n - 1).bit_length())

    # ----------------------------------------------------------
    # 2. Grid search over the family a_i = (i+1+c)^(-β).
    #    The target ℓ₁‑norm is set to sqrt(2·n); this makes the
    #    evaluation equal to the maximal convolution entry.
    # ----------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)

    best_seq = None
    best_score = np.inf

    betas = np.linspace(0.40, 0.80, 41)        # include exponents > 0.5
    offsets = np.linspace(0.0, 20.0, 21)      # explore larger additive shifts

    idx_base = np.arange(n, dtype=np.float64) + 1.0   # i+1 term (avoids 0⁰)

    for beta in betas:
        for c in offsets:
            cand_raw = 1.0 / ((idx_base + c) ** beta)
            # normalise to the prescribed ℓ₁‑norm
            scale = target_sum / cand_raw.sum()
            cand = cand_raw * scale

            score = _objective(cand, m_fft)

            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"[grid] β={beta:.3f}, c={c:5.2f} → score={score:.6f}")

            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    # fallback (should never happen)
    if best_seq is None:
        best_seq = np.ones(n, dtype=np.float64) * (target_sum / n)
        best_score = _objective(best_seq, m_fft)

    # ----------------------------------------------------------
    # 3. Active‑set sub‑gradient descent.
    #    For each iteration we identify the indices where the
    #    convolution is (almost) maximal, compute the corresponding
    #    sub‑gradient, and take a scaled step.
    # ----------------------------------------------------------
    a = best_seq.copy()
    best_global_seq = a.copy()
    best_global_score = best_score

    # hyper‑parameters
    step_factor = 0.5          # initial step‑size multiplier
    no_improve = 0
    iter_cnt = 0
    max_no_improve = 200      # after this we inject a random perturbation
    rel_tol = 1e-6            # tolerance to define the active set
    momentum = 0.0            # simple momentum term (optional)

    while time.time() < deadline:
        # ----- convolution and active set -----
        fft_a = np.fft.rfft(a, n=m_fft)
        conv = np.fft.irfft(fft_a * fft_a, n=m_fft)[: 2 * n - 1]
        max_conv = np.max(conv)

        # active indices: all entries within (1‑rel_tol) of the maximum
        # (normally a handful of indices)
        active_mask = conv >= max_conv * (1.0 - rel_tol)
        active_idxs = np.nonzero(active_mask)[0]

        # ----- sub‑gradient of the max‑conv -----
        # gradient = Σ_{k∈active} (a + shift_k)
        grad = np.zeros_like(a)
        for k in active_idxs:
            idx = k - np.arange(n)               # positions of a_{k-i}
            mask = (idx >= 0) & (idx < n)
            # contribution a_i (present for every active k)
            grad[mask] += a[idx[mask]]            # a_{k-i} term
        grad += len(active_idxs) * a              # a_i term for each active k

        # ----- step computation -----
        # r = max_conv / sum(a)^2 ;  with our normalisation sum(a)=target_sum,
        # r = max_conv / (2·n)
        r = max_conv / (target_sum ** 2)
        step = step_factor * r

        # optional momentum (helps to smooth oscillations)
        if iter_cnt > 0:
            step += momentum * 0.1

        new_a = a - step * grad
        new_a = np.maximum(new_a, 0.0)

        # renormalise to keep the ℓ₁‑norm fixed (scale does not affect the score)
        s = np.sum(new_a)
        if s > 1e-12:
            new_a *= target_sum / s
        else:
            break      # degenerated to all zeros – should not happen

        new_score = _objective(new_a, m_fft)

        # ----- acceptance -----
        if new_score < best_global_score:
            best_global_score = new_score
            best_global_seq = new_a.copy()
            a = new_a
            no_improve = 0
            step_factor = min(step_factor * 1.05, 2.0)   # gradually enlarge step
            momentum = 0.9 * momentum + step            # update momentum
            if iter_cnt % 100 == 0:
                print(f"[grad] iter={iter_cnt:5d}, score={new_score:.6f}")
        else:
            no_improve += 1
            # decay step size slowly when we are not improving
            step_factor = max(step_factor * 0.97, 0.1)
            momentum = 0.5 * momentum

            # --- escape from plateaus ---
            if no_improve >= max_no_improve:
                i, j = np.random.randint(0, n, size=2)
                # move a small fraction (≤5 %) of mass from i to j
                delta = min(a[i], a[i] * np.random.rand() * 0.05)
                a[i] -= delta
                a[j] += delta
                a *= target_sum / np.sum(a)     # renormalise
                no_improve = 0
                step_factor = max(step_factor * 0.9, 0.1)

        iter_cnt += 1

        # early exit if the target benchmark is reached
        if best_global_score < 1.503:
            print(f"[!] Target reached: score={best_global_score:.6f} "
                  f"after {iter_cnt} iterations.")
            break

    print(f"[+] Search finished. Best score = {best_global_score:.6f}")
    # ensure plain Python floats for the final return value
    return [float(x) for x in best_global_seq]
