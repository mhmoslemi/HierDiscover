```
def project_simplex(v, s):
    # ensure v is numpy array
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

assistantfinal**Solution Overview**

The score to minimise is  

\[
\text{score}= \frac{2n\;\max(a*a)}{(\sum a)^2}\; .
\]

Because the denominator is fixed (we normalise the ℓ₁‑norm to  
\(\sqrt{2n}\)), minimising the score is exactly the same as minimising the
maximum entry of the linear convolution `a*a`.

The algorithm consists of three parts

1. **Initialisation** – a power‑law vector  `a_i ∝ (i+1)^{-p}` (with a
   random exponent `p∈[0.4,0.6]`) is generated and scaled to the required
   ℓ₁‑norm.  
   Several lengths (`n≈2000 … 5000`) and several restarts are used.

2. **Gradient‑guided mass transfer** – for the current vector we compute
   the convolution, locate the index `k*` where the maximum occurs,
   and evaluate the “gradient”  

   \[
   d_i = a_{k*-i}\quad (i\text{ that contributes to }k*)
   \]

   (the derivative of the maximal convolution w.r.t. `a_i`).  
   Moving a small fraction of the mass from an index with a large
   gradient to an index with a small gradient reduces the peak.
   The move is accepted only if the maximum actually decreases.
   This step is repeated many thousands of times; a simple
   step‑size schedule (`move_frac / √iteration`) makes the search
   increasingly fine‑grained.

3. **Projection onto the simplex** – after every move we project the
   vector onto the simplex  
   \(\{x\ge0,\;\sum x = \sqrt{2n}\}\) so the ℓ₁‑norm stays exactly the
   normalisation required by the competition.  The Euclidean projection
   onto the simplex is performed with the classic `O(n log n)` algorithm.

The whole search runs until the time budget (≈ 1000 s) is exhausted.
The best sequence found over all lengths and restarts is returned.

The code below follows the description above, uses only `numpy`,
`time` and `math`, contains no closures or lambda‑functions, and prints
useful progress information.

```python
"""
Gradient‑guided mass‑transfer optimizer for the convolution bound.

The algorithm works on a simplex (non‑negative entries, fixed ℓ₁‑norm).
For the current vector a it finds the convolution peak k*,
computes the gradient d_i = a_{k*‑i} (the influence of a_i on the peak)
and moves a small fraction of mass from a high‑gradient index to a
low‑gradient index.  After each move the vector is projected back onto
the simplex.  Several lengths, random power‑law starts and restarts are
tried; the best sequence seen is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _max_convolution_fft(a: np.ndarray, size: int) -> tuple[np.ndarray, float]:
    """
    Compute the linear convolution a*a using a power‑of‑two FFT.
    Returns the full convolution array and its maximum value.
    """
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * a.size - 1]
    return conv, float(conv.max())


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x >= 0, sum(x) = s}.
    Standard O(n log n) algorithm (see Duchi et al., 2008).
    """
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sort descending
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform simplex point
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _initial_powerlaw(p: float, n: int, total: float) -> np.ndarray:
    """
    Return a deterministic power‑law start a_i ∝ (i+1)^{-p}
    and scale it to have ℓ₁‑norm = total.
    """
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= total / a.sum()
    return a

# Main optimisation routine
# ----------------------------------------------------------------------

                a = _project_simplex(a, total)

                # ---- schedule the step size ------------------------------------------------
                iteration += 1
                if iteration % 10000 == 0:
                    move_frac *= move_frac_decay
                    if move_frac < 0.01:
                        move_frac = 0.01

    # ------------------------------------------------------------------
    # Fallback in the unlikely case that nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        total = math.sqrt(2.0 * n)
        best_seq = np.full(n, total / n, dtype=np.float64)

    # truncate trailing zeros (the outer harness accepts any length)
    nz = np.where(best_seq > eps_zero)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # return a plain Python list as required by the competition harness
    return best_seq.tolist()
```
