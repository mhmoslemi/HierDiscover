Fast FFT‑based projected gradient descent with momentum.

The routine works in three stages:

1. **Parametric U‑shape sweep** – a few simple families (linear,
   quadratic, exponential) are tried and the best of them is kept as a
   starting point.  If a previous construction ``height_sequence_1`` is
   available it is also considered.

2. **Projected gradient descent** – the maximal entry of the
   self‑convolution is reduced by moving against a sub‑gradient of the
   max‑convolution while preserving the ℓ₁‑norm (the required sum).  The
   update uses a momentum term, an adaptive step size and a slowly
   widening active set (indices whose convolution value is close to the
   current maximum).  Random pairwise mass transfers and occasional
   jitter help to escape plateaus.

3. **Final cheap random search** – after the main descent a short
   random‑transfer phase tries to squeeze out any remaining improvement.

All operations are performed with FFTs ( O(n log n) ) so the algorithm
can run many thousands of iterations within the 1000 s budget.  The
best sequence found is returned as a plain Python list.
"""

import time
import numpy as np


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution of ``a`` with itself using a power‑of‑two FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector so that its ℓ₁‑norm equals ``target``."""
    s = v.sum()
    if s > 0:
        return v * (target / s)
    # fallback – uniform distribution
    return np.full_like(v, target / v.shape[0])


def generate_U_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
) -> np.ndarray:
    """Produce a symmetric “U‑shaped’’ sequence of length ``n``."""
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid               # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_to_sum(raw, target_sum)


def compute_max_conv(a: np.ndarray) -> float:
    """Return max entry of the linear self‑convolution of ``a``."""
    conv = conv_self_fft(a)
    return float(conv.max())

    """Indices where ``conv`` is within ``rel_tol`` of its maximum."""
    max_val = conv.max()
    if rel_tol <= 0.0:
        return np.array([np.argmax(conv)], dtype=np.int64)
    return np.where(conv >= max_val * (1.0 - rel_tol))[0]


def compute_gradient(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution at ``a``.
    For every active convolution index ``k`` we add 2 · a_{k‑i}
    to the gradient entry ``i``.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k - i
        g[i] += 2.0 * a[j]
    return g

    return project_to_sum(a, target_sum)

    # Phase 0 – small parametric sweep (fast, FFT‑based)
    # ------------------------------------------------------------------
    best_a = project_to_sum(np.full(n, target_sum / n), target_sum)   # uniform fallback
    best_val = compute_max_conv(best_a)

    if a0 is not None:
        a0 = project_to_sum(a0, target_sum)
        val0 = compute_max_conv(a0)
        if val0 < best_val:
            best_val = val0
            best_a = a0.copy()

    shape_kinds = ("linear", "quadratic", "exponential")
    param_grids = {
        "linear": np.linspace(0.0, 5.0, 26),          # 0 … 5   step ≈0.2
        "quadratic": np.linspace(0.0, 4.0, 21),      # 0 … 4   step ≈0.2
        "exponential": np.linspace(0.0, 3.0, 16),    # 0 … 3   step ≈0.2
    }

    print(f"[{time.time():.2f}] Starting parametric sweep (n={n})")
    for kind in shape_kinds:
        for p in param_grids[kind]:
            cand = generate_U_shape(n, target_sum, kind=kind, param=p)
            val = compute_max_conv(cand)
            if val < best_val:
                best_val = val
                best_a = cand.copy()
                print(f"  New best after {kind} p={p:.2f}: {best_val:.6f}")
            if time.time() >= deadline:
                print("Time exhausted during sweep.")
                return best_a.tolist()

    # ------------------------------------------------------------------
    # Phase 1 – projected gradient descent with momentum
    # ------------------------------------------------------------------
    a = best_a.copy()
    momentum = np.zeros_like(a)
    beta = 0.9                         # momentum decay
    avg = target_sum / n
    step_size = 0.05 * avg            # initial aggressive step
    rel_tol = 0.0
    no_improve = 0
    iteration = 0

    print(f"[{time.time():.2f}] Entering gradient descent (step={step_size:.6f})")
    while time.time() < deadline:
        iteration += 1

        # current convolution and active set
        conv = conv_self_fft(a)
        max_val = conv.max()
        act = active_indices(conv, rel_tol)

        # sub‑gradient of the current max‑convolution
        grad = compute_gradient(a, act)
        direction = -grad + np.mean(grad)          # orthogonal to all‑ones

        # try the step; if it worsens the max we halve the step a few times
        accepted = False
        trial_step = step_size
        for _ in range(4):
            trial_momentum = beta * momentum + trial_step * direction
            trial_a = a + trial_momentum
            trial_a = np.clip(trial_a, 0.0, None)
            trial_a = project_to_sum(trial_a, target_sum)
            trial_val = compute_max_conv(trial_a)

            if trial_val <= max_val + 1e-12:
                momentum = trial_momentum
                a = trial_a
                accepted = True
                break
            trial_step *= 0.5

        if not accepted:
            # we are stuck – reset momentum and shrink step size
            momentum = np.zeros_like(a)
            step_size *= 0.7
            continue

        # record any improvement over the global best
        if trial_val + 1e-12 < best_val:
            best_val = trial_val
            best_a = a.copy()
            print(f"Iter {iteration:06d} – new best {best_val:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # scheduling
        if iteration % 2000 == 0:
            step_size *= 0.9
            rel_tol = min(0.05, rel_tol + 0.005)
            if step_size < 1e-7:
                step_size = 1e-7

        # occasional stochastic mass transfer
        if np.random.rand() < 0.02:
            a = random_pair_transfer(a, target_sum, scale=0.005)

        # stagnation handling – small random jitter
        if no_improve >= 5000:
            jitter = np.random.randn(n) * 0.001 * avg
            a = np.clip(a + jitter, 0.0, None)
            a = project_to_sum(a, target_sum)
            momentum = np.zeros_like(a)
            no_improve = 0
            print(f"Iter {iteration:06d} – jitter applied")

    print(f"[{time.time():.2f}] Gradient phase finished; best value {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – final cheap random transfers
    # ------------------------------------------------------------------
    a = best_a.copy()
    while time.time() < deadline:
        a = random_pair_transfer(a, target_sum, scale=0.002)
        val = compute_max_conv(a)
        if val + 1e-12 < best_val:
            best_val = val
            best_a = a.copy()
            print(f"Random transfer – improvement to {best_val:.6f}")

    print(f"[{time.time():.2f}] Search concluded; final best {best_val:.6f}")
    return best_a.tolist()
