def compute_conv_and_gradient(a, beta, target_sum, Nfft, L, total_len):
    fa = np.fft.rfft(a, n=Nfft)
    conv_full = np.fft.irfft(fa * fa, n=Nfft)[:L]  # L = 2n-1
    # compute softmax weights
    maxc = conv_full.max()

    sumexp = exp_weights.sum()
    w = exp_weights / sumexp
    # gradient

    # extract g vector
    g = 2.0 * grad_full[(L - 1) - np.arange(len(a))]
    # smooth max objective:
    smooth = maxc + np.log(sumexp) / beta
    return conv_full, smooth, g

"""
Optimiser for the self‑convolution bound using a smooth‑max gradient.
The algorithm works in three phases:

1. **Initialisation** – start from a known good sequence (if available) or
   from a random / simple “U‑shaped’’ vector of length *n*.  The ℓ₁‑norm is
   fixed to √(2 n) so that `evaluate_sequence` returns exactly the maximum
   entry of the self‑convolution.

2. **Smooth‑max gradient descent** – minimise a differentiable surrogate
   of max (conv) :
        f_β(a) = (1/β)·log(∑_k e^{β·(a∗a)_k})
   Its gradient is the correlation of the current vector with the soft‑max
   weights w_k ∝ e^{β·(a∗a)_k}.  After each step we project onto the
   non‑negative simplex (preserving the required sum).

   The smoothing parameter β is increased slowly (β≈1 → β≈200) so that the
   optimisation first sees a smooth landscape and finally focuses on the
   true maximum.  The step size η is reduced together with β.

3. **Escapes & restarts** – occasional random pair transfers and a small
   Gaussian jitter prevent the search from stagnating.  Several restarts
   with different initial shapes are performed until the time budget (~1000 s)
   expires.  The best sequence encountered is returned.

All heavy work (convolutions and gradients) is performed with FFTs, giving
≈O(n log n) per iteration and allowing thousands of iterations within the
budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray, nfft: int, out_len: int) -> np.ndarray:
    """
    Linear convolution of a with itself using a fixed FFT length.
    Returns the first ``out_len`` entries (normally 2·len(a)-1).
    """
    fa = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(fa * fa, n=nfft)[:out_len]
    return conv


def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """
    Scale a non‑negative vector to have the prescribed ℓ₁‑norm.
    Zero entries stay zero; if the whole vector is zero a uniform
    distribution is returned.
    """
    s = v.sum()
    if s > 0.0:
        return v * (target / s)
    return np.full_like(v, target / v.shape[0])


def generate_U_shape(n: int, target_sum: float,
                     kind: str = "linear", param: float = 0.0) -> np.ndarray:
    """
    Produce a symmetric “U‑shaped’’ sequence.
    kind ∈ {linear, quadratic, exponential}.
    param ≥ 0 controls the edge‑enhancement strength.
    The result is scaled to have ℓ₁‑norm = target_sum.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid          # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_to_sum(raw, target_sum)


# ----------------------------------------------------------------------
# Core optimisation (smooth‑max gradient)
# ----------------------------------------------------------------------


def smooth_max_and_gradient(a: np.ndarray, beta: float,
                            nfft: int, L: int, total_len: int):
    """
    For a given vector ``a`` compute
        * conv = a ∗ a   (full linear convolution, length L)
        * smooth_max = (1/β)·log Σ exp(β·conv)  (a smooth approximation of max)
        * gradient of smooth_max w.r.t. ``a`` (size = len(a))

    The gradient formula is
        g = 2 * corr(a, w)    where   w_k ∝ exp(β·conv_k)
    and ``corr`` is the (non‑circular) cross‑correlation.
    FFTs of size ``nfft`` are used for efficiency.
    """
    # ----- convolution a*a -------------------------------------------------
    fa = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(fa * fa, n=nfft)[:L]            # length L = 2·n-1

    # ----- soft‑max weights -----------------------------------------------
    maxc = conv.max()
    exp_w = np.exp(beta * (conv - maxc))                # stabilised
    sumexp = exp_w.sum()
    w = exp_w / sumexp                                 # length L, Σ w = 1

    # ----- gradient (2 * correlation) ------------------------------------
    w_rev_fft = np.fft.rfft(w[::-1], n=nfft)           # FFT of reversed weights
    grad_full = np.fft.irfft(fa * w_rev_fft, n=nfft)[:total_len]
    # correlation indices: g_j = 2 * conv_full[L‑1‑j]
    g = 2.0 * grad_full[(L - 1) - np.arange(a.shape[0])]

    # ----- smooth max value ------------------------------------------------
    smooth = maxc + np.log(sumexp) / beta

    return conv, smooth, g

# ----------------------------------------------------------------------
