def propose_candidate(seed=42, budget_s=1000, **kwargs):

    start_time = time.time()

    else:

    # normalise sum

    # optionally add random jitter to diversify
    a += 1e-6 * np.random.randn(n)
    a = np.maximum(a,0)
    a = a * target_sum / a.sum()

    # optimum variables
    best_a = a.copy()
    best_score = _compute_score(best_a)
    print(f"Initial score: {best_score:.6f}")

    # precompute FFT sizes
    nfft_conv = _next_pow2(2*n-1)
    nfft_corr = _next_pow2(3*n-2)

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    eta = 0.1   # base learning rate
    eta_min = 0.001

    gamma = 5.0
    gamma_max = 80.0

    iteration=0
    last_improve_iter=0

    while time.time() < deadline:
        iteration+=1
        # compute conv
        af = np.fft.rfft(a, n=nfft_conv)
        conv = np.fft.irfft(af*af, n=nfft_conv)[:2*n-1]
        max_conv = conv.max()
        # weights
        if gamma < gamma_max:
            w = np.exp(gamma * (conv - max_conv))
            w_sum = w.sum()
            if w_sum==0.0:
                w = np.full_like(w, 1.0/w.size)
                w_sum = 1.0
            w /= w_sum
        else:
            # hardmax: uniform over all entries nearly max
            tol = 1e-12
            active = np.where(conv >= max_conv - tol)[0]
            if len(active)==0:
                active = np.array([np.argmax(conv)])
            w = np.zeros_like(conv)
            w[active] = 1.0/len(active)

        # gradient via cross-correlation
        wf = np.fft.rfft(w, n=nfft_corr)
        af_rev = np.fft.rfft(a[::-1], n=nfft_corr)
        corr_full = np.fft.irfft(wf * af_rev, n=nfft_corr)
        grad = 2.0 * corr_full[n-1:n-1+n]

        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)

        # enforce monotone non‑increasing
        # Reorder to ensure descending; keep sum by scaling after.
        a_new = np.maximum.accumulate(a_new[::-1])[::-1]

        # renorm
        s = a_new.sum()
        if s == 0.0:
            a_new = np.full_like(a_new, target_sum / n)
        else:
            a_new = a_new * target_sum / s

        # occasional jitter
        if iteration % 100 == 0:
            jitter = 1e-5 * np.random.randn(n)
            a_new = np.maximum(a_new + jitter, 0.0)
            a_new = a_new * target_sum / a_new.sum()

        # Evaluate
        score_new = _compute_score(a_new)

        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            last_improve_iter = iteration
            elapsed = time.time() - start_time
            print(f"Iter {iteration:6d}: new best {best_score:.6f} (elapsed {elapsed:.1f}s)")
        
        # Accept move if improvement else accept with probability (optional)
        if score_new < best_score * 1.001: # accept small uphill? maybe not needed
            a = a_new
        else:
            # maybe accept with small prob if temperature high
            # Use simple criterion: if iteration - last_improve_iter > 5000 then accept random restart
            if iteration - last_improve_iter > 5000:
                # restart with jitter around best
                a = best_a.copy()
                a += 1e-3 * np.random.randn(n)
                a = np.maximum(a,0)
                a = a * target_sum / a.sum()
                m.fill(0)
                v.fill(0)
                last_improve_iter = iteration
                print(f"Iter {iteration:6d}: restart after stagnation")
                # continue
            else:
                # keep old a (stay)
                a = a

        # update hyperparameters
        gamma = min(gamma*1.003, gamma_max)
        eta = max(eta*0.999, eta_min)

    return best_a.tolist()

Propose a sequence of non‑negative heights that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The implementation builds on the soft‑max gradient descent used previously
but adds a number of enhancements:

* **Adam optimisation** – adaptive first/second moment estimates give a
  more stable and faster descent than a plain constant step size.

* **Hard‑max sub‑gradient** – once the soft‑max temperature has reached a
  preset ceiling we switch to a true sub‑gradient of the (non‑smooth)
  maximum by putting equal weight on all lag positions that attain the
  current maximal convolution value.

* **Dynamic schedules** – the temperature `γ` is increased very slowly,
  while the base learning rate is annealed.  Periodic tiny jitter is
  injected to escape plateaus.

* **Stagnation handling** – if no improvement is observed for many
  iterations the algorithm restarts from a slightly perturbed copy of the
  best‑so‑far sequence.

* **Monotonicity (optional)** – after each update the sequence is forced
  to be non‑increasing (a simple `np.maximum.accumulate` on the reversed
  array).  This keeps the shape close to the analytically motivated
  decreasing profiles without hurting the optimisation.

The routine runs until the supplied time budget (default 1000 s) expires
and returns the best sequence encountered as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the objective for a given array `a` (no security checks).
    The score equals max(conv(a, a)) when the array is normalised to
    sum = sqrt(2 * n).
    """
    n = a.shape[0]
    nfft = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    # When sum = sqrt(2 n) the factor 2*n/(s**2) becomes 1.
    # We keep the generic formula for safety.
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights with a low evaluation
    score.  The function respects the 1000 s time limit; the best sequence
    found so far is returned as a Python list.
    """

    start_time = time.time()
    # Small safety margin to ensure we return before the hard timeout.
    deadline = start_time + budget_s - 5.0

    # 1. Determine the problem size `n` and initialise a candidate.
    # ------------------------------------------------------------------

        n = init_seq.shape[0]
    except Exception:

        n = 10000
        init_seq = None

    target_sum = np.sqrt(2.0 * n)  # the normalisation used by the benchmark

    if init_seq is not None:
        a = np.maximum(init_seq, 0.0)
    else:
        # Analytic start shape a_i = 1 / sqrt(i + 0.5)
        a = 1.0 / np.sqrt(np.arange(n, dtype=np.float64) + 0.5)

    # Normalise to the prescribed L1‑norm.
    a = a * target_sum / a.sum()

    # Tiny random perturbation to avoid pathological symmetry.
    a += 1e-6 * np.random.randn(n)
    a = np.maximum(a, 0.0)
    a = a * target_sum / a.sum()

    # ------------------------------------------------------------------
    # 2. Keep track of the best solution found so far.
    # ------------------------------------------------------------------
    best_a = a.copy()
    best_score = _compute_score(best_a)
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Pre‑compute FFT sizes that remain constant throughout the run.
    # ------------------------------------------------------------------
    nfft_conv = _next_pow2(2 * n - 1)   # for a * a
    nfft_corr = _next_pow2(3 * n - 2)   # for w ⊛ a_rev

    # ------------------------------------------------------------------
    # 4. Adam optimiser state.

    m = np.zeros_like(a)   # first moment estimate
    v = np.zeros_like(a)   # second moment estimate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    # Learning‑rate schedule (base learning rate before Adam scaling).
    eta = 0.12          # initial learning rate

    # Soft‑max temperature schedule.
    gamma = 5.0

    gamma_factor = 1.002  # slow increase

    # Iteration counters for stagnation detection.
    iteration = 0

    stagnation_limit = 8000   # restart if no improvement for this many steps

    # ------------------------------------------------------------------

    while time.time() < deadline:

        af = np.fft.rfft(a, n=nfft_conv)
        conv = np.fft.irfft(af * af, n=nfft_conv)[: 2 * n - 1]
        max_conv = conv.max()

        if gamma < gamma_max:

            w = np.exp(gamma * (conv - max_conv))

        # Enforce a non‑increasing profile (helps keep the shape well‑behaved).
        a_new = np.maximum.accumulate(a_new[::-1])[::-1]

        # ------- renormalise to the prescribed L1‑norm -------
        s = a_new.sum()

        else:
            a_new = a_new * target_sum / s

            a_new = np.maximum(a_new + jitter, 0.0)
            a_new = a_new * target_sum / a_new.sum()

        score_new = _compute_score(a_new)

            best_score = score_new
            best_a = a_new.copy()
            last_improve_iter = iteration
            elapsed = time.time() - start_time
            print(f"[iter {iteration:6d}] new best {best_score:.6f} (elapsed {elapsed:.1f}s)")

        if score_new <= best_score * 1.001:
            a = a_new
        else:
            # If we have been stuck for a long time, restart from the best found.
            if iteration - last_improve_iter > stagnation_limit:
                a = best_a.copy()
                # Add a modest perturbation before continuing.
                a += 5e-4 * np.random.randn(n)
                a = np.maximum(a, 0.0)
                a = a * target_sum / a.sum()

                v.fill(0.0)
                beta1_pow = 1.0
                beta2_pow = 1.0
                last_improve_iter = iteration
                print(f"[iter {iteration:6d}] restart after stagnation")
            # otherwise keep the previous `a` (reject the move)

        # ------- update hyper‑parameters -------
        gamma = min(gamma * gamma_factor, gamma_max)
        eta = max(eta * 0.998, eta_min)

    # ------------------------------------------------------------------
    # 6. Return the best sequence discovered (as a plain Python list).
    # ------------------------------------------------------------------
    print(f"[final] best score={best_score:.6f} after {iteration:,} iterations")
    return best_a.tolist()
