"""
Propose a low‑evaluation step‑function using a simple alternating LP scheme.

The algorithm works in three phases:

1. **Initialisation** – start from a known good construction (the global
   `height_sequence_1` if present, otherwise a power‑law family) and
   normalise the vector to unit L1‑norm.

2. **Alternating convex optimisation** – keep the current sequence `f`,
   compute M = max(f*f).  Solve a linear programme
       maximise   Σ g_i
       subject to 0 ≤ g_i,
                  (f * g)_k ≤ M   for all convolution indices k.
   The LP pushes mass into positions where `f` is small, flattening the
   self‑convolution.  The optimiser returns a vector `g`; after rescaling
   it to the same L1‑norm as `f` we replace `f ← (1‑t)·f + t·g` with a
   moderate step size `t`.  Repeating this step quickly reduces the
   maximal convolution entry and thus the evaluation score.

3. **Final up‑scaling & fine stochastic refinement** – the best vector found
   for a moderate length (≈2000) is interpolated to the target length
   (5675, the size used by the competition).  A tiny random‑transfer
   refinement is performed until the time budget expires.

All work respects the constraints of `evaluate_sequence`: non‑negative
entries, finite numbers and a total sum > 0.  The code uses only NumPy and
SciPy (the `highs` LP solver) and keeps the total runtime well within the
provided budget.
"""

    """Return a non‑negative vector of length n from the power‑law differences."""
    i = np.arange(n, dtype=float)
    a = (np.power(i + gamma + 1.0, p) - np.power(i + gamma, p))
    total = np.power(n + gamma, p) - np.power(gamma, p)
    if total == 0:
        # avoid division by zero – fall back to uniform
        return np.full(n, 1.0 / n)
    a /= total
    return a


# ----------------------------------------------------------------------
# Build the (2n‑1)×n convolution matrix C such that C·g = f * g.
def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """Construct the convolution matrix for a fixed vector f."""
    n = f.size
    rows = 2 * n - 1
    # Each column i contains the constant coefficient f[i] on rows i … i+n‑1
    C = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        C[i:i + n, i] = f[i]
    return C


# ----------------------------------------------------------------------
def _solve_lp_max_sum(f: np.ndarray, M: float, time_limit: float | None = None) -> np.ndarray | None:
    """
    Solve   maximise Σ g_i   st.  g_i ≥ 0,  (f * g)_k ≤ M.
    Returns the optimal g (not normalised) or None on failure.
    """
    n = f.size
    # Build the constraint matrix only once per call (cost O(n²) but n≈2000)
    A = _conv_matrix(f)
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # Linear programme: minimise -Σ g_i   (i.e. maximise Σ g_i)
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0.0, None) for _ in range(n)]

    # SciPy's 'highs' backend respects a 'time_limit' option (seconds).
    # If the caller provides None we let the solver run without a limit.
    options = {"presolve": True}
    if time_limit is not None:
        options["time_limit"] = time_limit

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options=options)

    if not res.success or res.x is None:
        return None
    return res.x


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Find a step‑function with evaluation < 1.503.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (default 1000).  The function returns as soon
        as the budget (minus a safety margin) is exhausted.

    Returns
    -------
    list[float]
        The best sequence found, ready to be passed to `evaluate_sequence`.
    """
    np.random.seed(seed)
    # Safety margin to avoid cutting off the very last call.
    deadline = time.time() + budget_s - 0.5

    # ------------------------------------------------------------------
    # Phase 1 – initialise a reasonable starting vector (length ≈2000).
    n_small = 2000
    try:
        # Prefer the globally supplied best construction if it exists.
        init_seq = np.asarray(height_sequence_1, dtype=float)
        if init_seq.ndim != 1:
            raise ValueError
        if init_seq.size >= n_small:
            f = init_seq[:n_small].copy()
        else:
            # Pad with zeros and then renormalise.
            f = np.zeros(n_small, dtype=float)
            f[:init_seq.size] = init_seq
    except Exception:
        # Fallback: a power‑law shape that is known to be decent.
        f = _generate_powerlaw(n_small, p=0.40, gamma=20.0)

    # Enforce non‑negativity and normalise to unit L1‑norm.
    f = np.maximum(0.0, f)
    total = f.sum()
    if total <= 1e-12:
        f = np.full(n_small, 1.0 / n_small)
    else:
        f = f / total

    best_f = f.copy()
    best_score = evaluate_sequence(best_f.tolist())
    print(f"[init] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – alternating LP optimisation (a few iterations).
    iteration = 0
    # Step size for convex combination (can be adapted later).
    t = 0.5

    while time.time() < deadline:
        iteration += 1
        # Current maximal self‑convolution.
        conv_ff = np.convolve(best_f, best_f)
        M = conv_ff.max()

        # Remaining time for this LP (a generous fraction of the whole budget).
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # Give the LP at most half of the remaining time.
        lp_time = min(2.0, remaining * 0.5)

        g = _solve_lp_max_sum(best_f, M, time_limit=lp_time)
        if g is None:
            print("[lp] LP failed – aborting LP phase.")
            break

        # Normalise g to the same L1‑norm (which is 1 after the normalisation above).
        g_sum = g.sum()
        if g_sum <= 0:
            break
        g = g / g_sum

        # Convex update.
        new_f = (1.0 - t) * best_f + t * g
        new_f = np.maximum(0.0, new_f)          # safety
        new_f = new_f / new_f.sum()             # keep sum = 1

        new_score = evaluate_sequence(new_f.tolist())
        print(f"[iter {iteration:3d}] score={new_score:.6f}  M={M:.8e}")

        if new_score < best_score - 1e-12:
            best_f = new_f
            best_score = new_score
            print(f"[iter {iteration}]  ** new best **")
            # If improvement is strong we keep the same step size,
            # otherwise we slowly reduce it to fine‑tune.
            t = min(0.9, t * 1.05)
        else:
            # No improvement – shrink the step size to make the update gentler.
            t = max(0.1, t * 0.7)

        # Limit number of LP iterations to keep within the overall budget.
        if iteration >= 15:
            break

    # ------------------------------------------------------------------
    # Phase 3 – up‑scale to the target length (5675) and fine random refinement.
    n_big = 5675
    # Linear interpolation of the final small vector to the larger size.
    x_small = np.linspace(0.0, 1.0, n_small, endpoint=False)
    x_big   = np.linspace(0.0, 1.0, n_big,   endpoint=False)
    seq_big = np.interp(x_big, x_small, best_f)
    seq_big = np.maximum(0.0, seq_big)
    seq_big = seq_big / seq_big.sum()        # unit L1 again

    # One quick evaluation after up‑scaling.
    best_score = evaluate_sequence(seq_big.tolist())
    print(f"[scale] up‑scaled score={best_score:.6f}")

    seq = seq_big.copy()

    # Tiny stochastic refinement (same spirit as the original code).
    while time.time() < deadline:
        i, j = np.random.choice(len(seq), 2, replace=False)
        delta = (np.random.rand() - 0.5) * 1e-4   # signed tiny transfer
        if seq[i] + delta < 0.0 or seq[j] - delta < 0.0:
            continue
        cand = seq.copy()
        cand[i] += delta
        cand[j] -= delta
        # Keep L1 unchanged automatically.
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < best_score:
            best_score = cand_score
            seq = cand
            print(f"[refine] improved score -> {best_score:.6f}")

    print(f"[final] best score={best_score:.6f} (length={len(seq)})")
    return seq.tolist()
