Optimisation of a non‑negative step sequence for the given evaluate_sequence.

The routine proceeds in three stages inside the 1000 s time budget.

1. **Broad parametric sweep** – families of simple analytic shapes are
   sampled (power law, exponential and a mixed power/exponential).  Every
   candidate is normalised to unit ℓ¹‑norm (the objective is scale‑free) and
   evaluated.  The best candidate so far is kept.

2. **Projected sub‑gradient descent with momentum** – starting from the best
   candidate we iteratively reduce the maximum entry of the auto‑convolution
   (the only term that appears in the objective).  For the current vector
   `x` the sub‑gradient of the ℓ∞‑norm of `x*x` is computed from all active
   (maximising) convolution indices.  An additive update with a decreasing
   step‑size and Nesterov‑type momentum is performed and the result is
   projected back onto the probability simplex (the ℓ¹‑norm is kept equal to
   one).  Random jitter and occasional restarts help escape plateaus.

3. **Length growth** – after a fixed number of iterations a few tiny entries
   are appended to the vector (the length `n` grows up to a preset maximum).
   This allows the mass to spread further, which can reduce the factor
   `n·max(conv)` in the objective.

def _generate_powerlaw(p: float, sigma: float,
                       max_len: int = 5000, eps: float = 1e-10) -> np.ndarray:
    """a_i = (i + sigma) ** (-p)   truncated when a_i < eps."""
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p)
    return a[a > eps]


def _generate_exponential(lam: float,
                          max_len: int = 5000, eps: float = 1e-10) -> np.ndarray:
    """a_i = exp(-lam * i)   truncated when a_i < eps."""
    i = np.arange(max_len, dtype=np.float64)
    a = np.exp(-lam * i)
    return a[a > eps]


def _generate_mixed(p: float, sigma: float, lam: float,
                    max_len: int = 5000, eps: float = 1e-10) -> np.ndarray:
    """a_i = (i+sigma)^(-p) * exp(-lam*i)   truncated when a_i < eps."""
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p) * np.exp(-lam * i)
    return a[a > eps]


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum(x)=1}.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection only works on 1‑D arrays")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient(x: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of the function   g(x) = max_k (x*x)_k
    with respect to the vector x (on the simplex).  All active
    convolution indices are used.
    """
    n = x.shape[0]
    conv = np.convolve(x, x)                # length 2n‑1
    m = conv.max()
    # list of indices where the max is attained (within numerical tolerance)
    active = np.where(conv >= m - 1e-14)[0]
    g = np.zeros_like(x)
    for k in active:
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        i = np.arange(i_min, i_max + 1, dtype=int)
        j = k - i
        # derivative of conv[k] w.r.t x_i is x_j (the factor 2 is absorbed in step‑size)
        g[i] += x[j]
    return g

                print('Target bound reached – terminating early.')
                break

        # restart with a fresh random Dirichlet if no progress for a while
        if iteration % 6000 == 0:
            n = x.shape[0]
            rnd = np.random.rand(n)
            x = 0.6 * x + 0.4 * rnd
            x = x / x.sum()
            momentum = np.zeros_like(x)

    print('final best score:', best_score, 'length:', len(best_seq))
    return best_seq
