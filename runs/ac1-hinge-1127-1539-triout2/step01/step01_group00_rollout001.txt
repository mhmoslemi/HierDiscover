def project_simplex(v, s):
    """Project onto simplex {x >= 0, sum x = s}."""
    n = len(v)
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    # find rho: max i s.t. u_i - cssv_i / (i+1) > 0
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho)==0:
        # all v negative or small; set uniform
        return np.full_like(v, s / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Simple smooth‑max gradient descent for the self‑convolution L∞ problem.

The algorithm works on a vector `a` (length n) with fixed L1‑norm  
S = sqrt(2·n).  The objective is max_k (a∗a)[k]; we minimise a smooth
approximation   Jβ(a)=log ∑ₖexp(β·(a∗a)[k])   (β is a temperature that is
increased during the run).

For a given β the gradient of Jβ is

    ∂J/∂a = 2·conv(softmaxβ(b), rev(a))

where `b = a∗a` and `rev` denotes the reversed vector.
A multiplicative update `a ← a·exp(−η·∂J/∂a)` keeps all entries non‑negative;
after the update we simply rescale to the required L1‑norm.

The routine runs several random restarts (or starts from the provided
`height_sequence_1`) until the time budget expires, keeping the best
vector seen (the external score equals the maximal convolution entry,
so we can read it directly from the internal computation).  Small
random jitter is injected occasionally to avoid tiny plateaus.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries would be zero – return the uniform point
        return np.full_like(v, s / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real 1‑D arrays using FFT."""
    n = x.shape[0]
    m = y.shape[0]
    conv_len = n + m - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[:conv_len]
    return conv


def _full_self_conv(a: np.ndarray) -> np.ndarray:
    """Return the full linear self‑convolution a∗a."""
    return _conv_fft(a, a)


def _softmax_weights(b: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max of the vector b with temperature 1/β (β≥0)."""
    # shift by max for numerical stability
    shift = b - b.max()
    # when β is zero we just take a uniform distribution
    if beta == 0:
        w = np.full_like(b, 1.0 / b.size)
        return w
    exp_shift = np.exp(beta * shift)
    total = exp_shift.sum()
    if total == 0.0:
        # all under‑flowed → uniform
        return np.full_like(b, 1.0 / b.size)
    return exp_shift / total


def _gradient_softmax(p: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute ∇Jβ = 2·conv(p, rev(a)) restricted to the original length.
    `p` has length 2·n‑1, `a` has length n.
    """
    n = a.shape[0]
    rev = a[::-1]
    conv_full = _conv_fft(p, rev)                 # length = (2n‑1)+n‑1 = 3n‑2
    # the relevant slice is positions n‑1 … n‑1+n‑1
    slice_start = n - 1
    slice_end = slice_start + n
    grad_slice = conv_full[slice_start:slice_end]
    return 2.0 * grad_slice


def _initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a random feasible start (Dirichlet) or reuse the global one."""
    S = np.sqrt(2.0 * n)

    # 50 % chance to seed from the previously discovered construction
    if 'height_sequence_1' in globals() and rng.random() < 0.5:
        base = np.asarray(height_sequence_1, dtype=float)
        if base.size >= n:
            a = base[:n].copy()
        else:
            a = np.pad(base, (0, n - base.size), constant_values=0.0)
    else:
        a = rng.random(n)                     # uniform on simplex before scaling

    # project onto the simplex of radius S
    a = _project_simplex(a, S)
    return a

# Core optimisation loop
# ----------------------------------------------------------------------


def _optimize_one_restart(
    n: int,
    rng: np.random.Generator,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Perform the smooth‑max descent for a single random restart.
    Returns the best sequence found in this restart and its score.
    """
    S = np.sqrt(2.0 * n)
    a = _initial_sequence(n, rng)

    best_local = a.copy()
    best_score = np.max(_full_self_conv(a))

    # hyper‑parameters (tuned empirically)
    eta = 0.5            # multiplicative step size
    beta = 0.1           # start temperature
    beta_factor = 1.005  # geometric schedule
    beta_max = 20.0
    jitter_prob = 0.008
    eps_jitter = 0.01    # relative amplitude of jitter

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1. current self‑convolution and max entry (the exact external score)
        # --------------------------------------------------------------
        b = _full_self_conv(a)
        cur_max = float(b.max())

        if cur_max < best_score:
            best_score = cur_max
            best_local = a.copy()
            print(f"[restart‑improve] n={n} score={best_score:.6f}")

        # --------------------------------------------------------------
        # 2. soft‑max weights and gradient
        # --------------------------------------------------------------
        p = _softmax_weights(b, beta)
        grad = _gradient_softmax(p, a)

        # --------------------------------------------------------------
        # 3. multiplicative update + renormalisation
        # --------------------------------------------------------------
        a = a * np.exp(-eta * grad)

        total = a.sum()
        if total == 0.0:                     # safety – should never happen
            a = np.full_like(a, S / n)
        else:
            a *= (S / total)

        # --------------------------------------------------------------
        # 4. occasional random jitter to escape flat spots
        # --------------------------------------------------------------
        if np.random.random() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * eps_jitter
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()

        # --------------------------------------------------------------
        # 5. schedule updates
        # --------------------------------------------------------------
        beta = min(beta * beta_factor, beta_max)

    return best_local, best_score
