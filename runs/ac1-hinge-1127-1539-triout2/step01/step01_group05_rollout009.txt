"""
Propose a step‑function sequence that minimizes
    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works in three phases:

1. **Fast parametric warm‑start** – a small grid search over
   `a_i = (i + offset) ** (-p)` is performed (with the same length as
   ``height_sequence_1`` when it exists).  The best shape is kept as the
   initial point.

2. **Adam‑based smoothed‑max descent** – the objective is the true
   maximum of the convolution, approximated by a soft‑max (temperature
   ``γ``).  A gradient of the smooth‐max is
        g = 2 * (w ⊛ rev(a))
   where ``w`` are the soft‑max weights.  Adam (first‑ and second‑moment
   adaptation) is used together with an exponential decay of the learning
   rate and a slowly increasing ``γ``.  After each step the vector is
   projected onto the non‑negative orthant and renormalised so that
   ``sum(a) = sqrt(2·n)`` – under this normalisation the evaluation function
   equals the maximum of the convolution.

3. **Exploration & polishing** – occasional Gaussian jitter is added,
   and uphill moves are occasionally accepted (simulated‑annealing style)
   to escape shallow plateaus.  The best vector seen during the whole run
   is returned.

The routine respects the 1000 s time budget, prints progress whenever a
new best value is found and finally returns the best sequence as a plain
Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, top‑level as required)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_max(a: np.ndarray) -> float:
    """
    Compute max(a * a) using a real FFT.
    Length of `a` is n, the convolution length is 2·n‑1.
    """
    n = a.shape[0]
    nfft = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
    return float(conv.max())


def _softmax_weights(conv: np.ndarray, gamma: float) -> np.ndarray:
    """Return soft‑max weights w_i = exp(gamma*(conv_i - max(conv))) / sum."""
    max_c = conv.max()
    w = np.exp(gamma * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        # Numerical fallback – uniform weights
        w = np.full_like(w, 1.0 / w.size)
        w_sum = 1.0
    w /= w_sum
    return w


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient of the smooth maximum w.r.t. a:
        g = 2 * ( w ⊛ rev(a) )
    """
    n = a.shape[0]
    nfft = _next_pow2(3 * n - 2)          # length for cross‑correlation
    wf = np.fft.rfft(w, n=nfft)
    arev_f = np.fft.rfft(a[::-1], n=nfft)
    corr_full = np.fft.irfft(wf * arev_f, n=nfft)
    # valid part starts at index n‑1 and has length n
    grad = 2.0 * corr_full[n - 1 : n - 1 + n]
    return grad


def _renormalise(a: np.ndarray, target_sum: float) -> np.ndarray:
    """Project onto the non‑negative orthant and scale to the target sum."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        # Recover a uniform vector if everything collapsed
        a = np.full_like(a, target_sum / a.shape[0])
    else:
        a *= target_sum / s
    return a

    """
    Search for a non‑negative step‑function sequence with a low evaluation
    score.  The function runs for at most ``budget_s`` seconds and returns
    the best sequence found as a plain Python list.
    """

    # If a previous construction exists we reuse its length, otherwise we
    # fall back to a moderate size that still fits comfortably in the time
    # budget.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        n = 8000                     # a safe default for fresh runs

    target_sum = math.sqrt(2.0 * n)

    #    a_i = (i + offset) ** (-p)

    best_a = None

    p_vals = np.linspace(0.45, 0.85, 9)          # exponent

        # --------------------------------------------------------------

            best_a = a.copy()
            elapsed = time.time() - start_time

        max_conv = float(conv.max())

        conv_cand = np.fft.irfft(cand_fft * cand_fft, n=_next_pow2(2 * n - 1))[: 2 * n - 1]
