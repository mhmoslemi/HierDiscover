def _gradient(a, beta):
    n = a.size
    # compute convolution a*a via FFT
    size = 1 << ((2*n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
    b = conv  # shape (2n-1,)
    # Softmax probabilities p
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)
    # cross-correlation: g_j = 2 * sum_k p_k * a_{k - j}
    # Equivalent to conv of a and reversed p, then slice
    p_rev = p[::-1]
    # compute FFT size for conv a with p_rev
    size2 = 1 << ((n + (2*n-1) - 1).bit_length())
    # maybe reuse size, but compute new for safety
    A_fft = np.fft.rfft(a, n=size2)
    Pr_fft = np.fft.rfft(p_rev, n=size2)
    corr = np.fft.irfft(A_fft * Pr_fft, n=size2)[: n + (2*n-1) - 1]  # length = n+L-1 = 3n-2
    # g = 2 * corr[n-1 : n-1 + n] (indices j correspond)
    g = 2.0 * corr[n-1 : n-1 + n]
    return b, p, g

def _project_to_simplex(v, s):
    """Project vector v onto the simplex {x >=0, sum x = s}."""
    # Implementation similar to Duchi et al.
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _optimize(a0, total_time, start_time):

    n = a.size
    target_sum = np.sqrt(2.0 * n)

            # step size adapt

            a = a - lr * g

def _project_to_simplex(v, s):
    """Project onto the simplex {x >=0, sum x = s}."""
    # Based on Duchi et al. (2008)
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    # find rho
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        # all zeros?
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _gradient(a, beta):
    """Compute gradient of softmax-approximated max-convolution w.r.t a."""
    n = a.shape[0]
    # Compute convolution a*a
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[:2 * n - 1]
    b = conv
    # softmax probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)
    # cross-correlation gradient: g = 2 * (a ⋆ p)
    p_rev = p[::-1]
    size2 = 1 << ((n + (2 * n - 1) - 1).bit_length())
    A_fft = np.fft.rfft(a, n=size2)
    Pr_fft = np.fft.rfft(p_rev, n=size2)
    corr = np.fft.irfft(A_fft * Pr_fft, n=size2)[: n + (2 * n - 1) - 1]  # length 3n-2
    # slice
    g = 2.0 * corr[n - 1 : n - 1 + n]
    return g, b

"""
Gradient‑descent + random hill‑climbing search for the step function bound.

The objective is  2·n·max(conv(a,a)) / (sum(a)^2) .
Because the value is scale‑invariant we keep the ℓ¹‑norm fixed to √(2·n).
The algorithm works as follows:

1.  Initialise a decreasing power‑law sequence a_i ∝ (i+1)^(-p) with a
    random exponent p≈0.5 and project it onto the simplex
    {a≥0, Σa = √(2·n)}.

2.  Repeatedly minimise an smoothed approximation of max(conv(a,a)).
    For a temperature β we use the log‑sum‑exp:
            Mβ(a) = (1/β)·log Σ_k exp(β·b_k) ,  b = a*a .
    Its gradient w.r.t. a is
            ∇Mβ = 2·(a ⋆ p) ,   p_k = exp(β·b_k)/Σ exp(β·b_k)
    (⋆ = forward cross‑correlation).  The gradient is computed with FFTs,
    then a gradient step is taken and the result is projected back onto
    the simplex.

    β is increased gradually (annealing) while the step size is scaled
    with 1/(β·‖∇Mβ‖) so that the updates stay stable.

3.  After the annealing phase a short stochastic “hill‑climbing” phase
    moves a random fraction of mass between two positions and keeps
    improving moves.  This refines the solution locally.

4.  The whole routine is run for several vector lengths and random
    restarts; the best sequence found before the time budget expires is
    returned.

All helper functions are top‑level, there are no closures or lambdas,
and the code works with a single CPU.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """Fast version of the competition objective (no safety checks)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:                                     # safety – should never happen
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    { x | x_i ≥ 0, Σ x_i = s }.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of the soft‑max approximation of the maximum convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Return (g, b) where
        b = a*a (full convolution, length 2·n‑1)
        g = ∇ (log‑sum‑exp_{β}(b))  = 2·(a ⋆ p)    with p_k ∝ exp(β·b_k)
    Both are computed with FFTs for speed.
    """
    n = a.shape[0]

    # ---- b = a*a -------------------------------------------------------
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ---- soft‑max probabilities p ---------------------------------------
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)                # shape (2·n‑1,)

    # ---- gradient g = 2·(a ⋆ p) ---------------------------------------
    # cross‑correlation a ⋆ p can be obtained as the central part of
    # conv(a_rev, p) where a_rev = a[::-1]
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1        # = 3·n‑2
    size2 = 1 << ((conv_len).bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    # slice the required part: indices n‑1 … n‑1+n‑1 (total n entries)
    g = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return g, b

# Stochastic hill‑climbing (the cheap local search from the previous run)
# ----------------------------------------------------------------------


def _random_hill_climb(a: np.ndarray, deadline: float):
    """
    Very cheap hill‑climber that preserves the ℓ¹‑norm.
    Randomly transfers a fraction of mass from i to j and keeps improving moves.
    """
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    cur_val = best_val

    max_frac = 0.12
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        frac = rng.random() * max_frac
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        new_val = _score(a)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}]  hill‑climb improved = {best_val:.6f}")
        else:
            a[j] -= delta
            a[i] += delta
    return best_a, best_val

# Main optimiser (gradient descent with annealing)

def _annealed_gradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected gradient descent on the soft‑max approximation of the maximum.
    The temperature β is increased gradually (annealing) while the step
    size adapts to the gradient magnitude.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 256.0
    step_base = target_sum * 0.20          # base step magnitude
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        # gradient at current temperature
        g, _ = _gradient_softmax(a, beta)

        # step size proportional to 1/(β·‖g‖) (the factor β keeps later stages safe)
        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            lr = step_base / (norm_g * (beta + 1.0))
            a = a - lr * g

        # projection onto the simplex (keeps non‑negativity and L¹‑norm)
        a = _project_to_simplex(a, target_sum)

        # occasional tiny random jitter to help escape plateaus
        if it % 250 == 0:

            a = a + jitter

            a = _project_to_simplex(a, target_sum)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}]  beta={beta:.2f}  score={best_val:.6f}")

        # annealing schedule
        if it % 500 == 0 and beta < max_beta:

        it += 1

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """

    `evaluate_sequence` objective.  The routine runs for at most `budget_s`
    seconds, trying several vector lengths and random restarts.  Each
    restart consists of a gradient‑descent phase (annealed soft‑max) followed
    by a short stochastic hill‑climb.  The best sequence found is returned.
    """

    rng = np.random.default_rng(seed)

    best_score = np.inf

    # vector lengths – larger lengths give more flexibility but cost more time
    lengths = [1500, 2500, 3500, 4500, 5000]

        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

            if time.time() >= deadline:

            a0 = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
            a0 *= target_sum / a0.sum()

            a0 += rng.normal(scale=target_sum * 0.001, size=n)
            a0 = np.maximum(a0, 0.0)
            a0 = _project_to_simplex(a0, target_sum)

            # allocate a chunk of the remaining budget for this run
            sub_deadline = min(deadline, time.time() + 20.0)   # at most 20 s per restart

            # ---- gradient descent phase -------------------------------------------------
            a_opt, opt_score = _annealed_gradient_descent(a0, sub_deadline)

            # ---- short hill‑climbing refinement -----------------------------------------
            if time.time() < deadline:
                a_opt, opt_score = _random_hill_climb(a_opt, deadline)

            if opt_score < best_score:
                best_score = opt_score
                best_seq = a_opt.copy()
                print(f"** NEW GLOBAL BEST **  n={n}  p={p:.4f}  restart={restart}  score={best_score:.6f}")

    # fallback – should never be needed
    if best_seq is None:
        n = 1000
        a = np.ones(n, dtype=np.float64)
        a *= np.sqrt(2.0 * n) / n
        best_seq = a

    # trim trailing zeros (just in case)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    return best_seq.tolist()
