def softmax_grad(x, q):
    conv = np.convolve(x, x)
    # compute weights w = exp(q*conv) normalized
    # for numerical stability subtract max
    logits = q * conv
    maxlog = np.max(logits)
    exp_logits = np.exp(logits - maxlog)
    w = exp_logits / np.sum(exp_logits)
    rev_x = x[::-1]
    w_rev_conv = np.convolve(w, rev_x)
    g = 2 * w_rev_conv[len(w)-1 : len(w)-1 + len(x)]
    return g

def _softmax_update(x, eta, q):
    conv = np.convolve(x, x)
    logits = q * conv
    max_log = np.max(logits)
    w = np.exp(logits - max_log)
    w = w / w.sum()
    rev_x = x[::-1]
    # convolution w * rev_x
    cw = np.convolve(w, rev_x)
    gradient = 2.0 * cw[len(w)-1 : len(w)-1 + len(x)]
    y = x * np.exp(-eta * gradient)
    if y.sum() == 0:
        y = np.abs(np.random.randn(len(x)))
    return y / y.sum()

"""
Propose a step‑function (non‑negative list of heights) that minimizes
`evaluate_sequence`.  The algorithm works in three stages:

1. **Broad parametric sweep** – generate many candidate families
   (power‑law, exponential, mixed power‑law·exponential) with a wide
   range of exponents and offsets.  Every candidate is evaluated and the
   best one is kept.

2. **Soft‑max guided mirror descent** – smooth the ℓ∞‑norm of the
   auto‑convolution by replacing the hard maximum with a soft‑max.
   The gradient of the soft‑max is computed analytically (a single
   convolution) and an exponential‑gradient (mirror‑descent) step is
   performed.  The soft‑max temperature `q` is increased gradually so
   that the algorithm first makes large, smooth moves and later focuses
   on the true maximum.

3. **Final polishing** – occasional random jitter and pruning of tiny
   entries avoid stagnation.  The best sequence ever seen is returned
   when the 1000 s budget expires.

The whole routine respects the time limit, prints progress, and works
without any file I/O.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper factories for parametric families
# ----------------------------------------------------------------------


def generate_powerlaw(p: float, sigma: float,
                      max_len: int = 3000, eps: float = 1e-8) -> np.ndarray:
    """a_i = (i+σ)^(-p)   (truncated when below eps)."""
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p)
    return a[a > eps]


def generate_exponential(lam: float,
                         max_len: int = 3000, eps: float = 1e-8) -> np.ndarray:
    """a_i = exp(-λ·i)   (truncated when below eps)."""
    i = np.arange(max_len, dtype=np.float64)
    a = np.exp(-lam * i)
    return a[a > eps]


def generate_mixed(p: float, sigma: float, lam: float,
                   max_len: int = 3000, eps: float = 1e-8) -> np.ndarray:
    """a_i = (i+σ)^(-p)·exp(-λ·i)   (truncated when below eps)."""
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p) * np.exp(-lam * i)
    return a[a > eps]


# ----------------------------------------------------------------------
# Soft‑max based mirror‑descent update
# ----------------------------------------------------------------------


def _softmax_mirror_step(x: np.ndarray, eta: float, q: float) -> np.ndarray:
    """
    One iteration of a soft‑max guided mirror‑descent.
    x must be non‑negative and sum to one.
    """
    n = x.shape[0]

    # full auto‑convolution of the current vector
    conv = np.convolve(x, x)                     # length 2n‑1

    # soft‑max weights w_k ∝ exp(q·conv_k)
    logits = q * conv
    max_log = np.max(logits)
    w = np.exp(logits - max_log)                 # avoid overflow
    w /= w.sum()

    # gradient g_i = 2·∑_k w_k·x_{k‑i}  (zero outside the overlap)
    rev_x = x[::-1]
    cw = np.convolve(w, rev_x)                   # length 3n‑2
    start = len(w) // 2                          # = n‑1
    g = 2.0 * cw[start:start + n]

    # exponential (multiplicative) update + re‑normalisation onto the simplex
    y = x * np.exp(-eta * g)
    if y.sum() == 0.0:                           # pathological case – random restart
        y = np.abs(np.random.randn(n))
    y /= y.sum()
    return y
