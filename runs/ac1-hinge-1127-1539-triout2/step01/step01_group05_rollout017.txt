def project_onto_simplex(v, target_sum):
    """Project vector v onto the simplex {x >= 0, sum x = target_sum}."""
    # If sum(v) == target_sum and all >=0 -> return v
    # else use algorithm.
    v = np.asarray(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative sequence) with a low value of

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm consists of three stages:

1. **Initialisation & random search** – generate a few power‑law
   candidates (a_i = 1/(i+offset)^p) and keep the best one (also
   re‑use the globally supplied ``height_sequence_1`` if present).

2. **Adam optimisation on a smooth‑max surrogate** – the true
   objective `max(conv)` is approximated by a soft‑max with a
   temperature `γ`.  The gradient of this surrogate is
   `g = 2 * (w ⊛ a_rev)` where `w` are the soft‑max weights.
   An Adam optimiser (first‑ and second‑moment adaptation) is used,
   and after each update the vector is projected onto the simplex
   `{x ≥ 0, sum(x)=sqrt(2 n)}` so that the normalisation never changes.

3. **Fine‑tuning** – after the main loop a short phase with a very
   large `γ` (almost the true max) and a tiny constant step size is
   performed to polish the solution.

All arithmetic is performed with NumPy and FFTs for speed.
The function returns the best sequence found before the time budget
expires (a small safety margin is kept).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two greater than or equal to *x*."""
    return 1 << (x - 1).bit_length()


def _compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the metric (identical to the provided
    `evaluate_sequence` but without the security wrapper).  The input
    vector is assumed to be non‑negative and well‑behaved.
    """
    n = a.shape[0]
    nfft = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))


def project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of *v* onto the simplex
        { x : x_i ≥ 0 , sum_i x_i = target_sum }.
    """
    # Shift by the mean to guarantee non‑negativity after clipping
    # (standard algorithm, see e.g. Duchi et al., ICML 2008).
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(len(v)) + 1  # 1‑based indexing
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Main optimisation routine
# ----------------------------------------------------------------------
