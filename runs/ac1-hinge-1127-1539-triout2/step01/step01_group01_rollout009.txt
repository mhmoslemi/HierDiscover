def project_onto_simplex(v, target):
    # v: 1D numpy array
    # target: sum constraint >0
    # returns projection onto {x >=0, sum(x)=target}
    n = v.shape[0]
    # shift to have sum=target ignoring non-neg?
    # algorithm: sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

the evaluation function

    2 * n * max(conv(a,a)) / (sum(a)**2)

where `conv` is the linear self‑convolution.  The algorithm consists of

   occasionally; any improvement is kept.

All operations are performed with FFT‑based convolutions, which makes a
single iteration O(n log n) and allows thousands of iterations within the
1000 s budget.  The algorithm always returns the best sequence seen so
far.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a with itself using a power‑of‑two FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def generate_U_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
) -> np.ndarray:
    """
    Produce a symmetric “U‑shaped’’ sequence of length n.
    kind ∈ {linear, quadratic, exponential}.
    param ≥ 0 controls the edge enhancement.
    The result is scaled to have ℓ₁‑norm `target_sum`.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex
        {x | x_i ≥ 0, sum_i x_i = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)
    # make a copy to avoid modifying the input
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero; return uniform distribution
        return np.full_like(v, target / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a tiny amount of mass from one random index to another,
    then project back onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def smooth_gradient_step(
    a: np.ndarray,
    target_sum: float,
    eta: float,
    tau_factor: float,
) -> tuple[np.ndarray, float]:
    """
    Perform one gradient step on the smoothed objective.
    Returns the new vector and the current max‑convolution value.
    """
    n = a.shape[0]

    # 1) self‑convolution and max entry
    conv = conv_self_fft(a)                     # length 2n‑1
    max_conv = conv.max()

    # 2) smoothing weights (log‑sum‑exp)
    tau = max(tau_factor * max_conv, 1e-12)
    # stability: subtract max before exponentiation
    w = np.exp((conv - max_conv) / tau)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # 3) gradient = (2/τ) * convolution(w, a_rev) restricted to indices n‑1 … 2n‑2
    a_rev = a[::-1]                             # reversed a
    L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]

    # slice that yields ∂/∂a_i of the smoothed max
    grad = (2.0 / tau) * conv_full[n - 1 : n - 1 + n]

    # 4) gradient descent step followed by projection onto simplex
    a_new = a - eta * grad
    a_new = project_onto_simplex(np.maximum(a_new, 0.0), target_sum)
    return a_new, max_conv


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    `evaluate_sequence`.  The search respects a time budget (≈1000 s)
    and returns the best sequence observed.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # a thin safety margin

    # ------------------------------------------------------------------
    # Determine problem size (use existing construction if available)
    # ------------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        init_from_base = np.array(base, dtype=np.float64)
    else:
        # a large but manageable length (≈8 k)
        n = np.random.randint(7500, 8700)
        init_from_base = None

    target_sum = np.sqrt(2.0 * n)               # conventional normalisation

    # ------------------------------------------------------------------
    # Phase 1 – simple parametric sweep (U‑shapes)
    # ------------------------------------------------------------------
    best_seq = np.full(n, target_sum / n)       # uniform start
    best_score = float("inf")

    shape_kinds = ("linear", "quadratic", "exponential")
    param_grids = {
        "linear": np.linspace(0.0, 4.0, 41),      # step 0.1
        "quadratic": np.linspace(0.0, 3.0, 31),  # step 0.1
        "exponential": np.linspace(0.0, 2.0, 21),  # step 0.1
    }

    print(f"[{time.time():.2f}] Starting parametric sweep (n={n})")
    for kind in shape_kinds:
        for p in param_grids[kind]:
            cand = generate_U_shape(n, target_sum, kind=kind, param=p)
            score = evaluate_sequence(cand.tolist())
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"  New best {kind} (p={p:.2f}): {best_score:.6f}")
            if time.time() >= deadline:
                print("Time exhausted during sweep.")
                return best_seq.tolist()

    # also try the provided baseline if any
    if init_from_base is not None:
        score = evaluate_sequence(init_from_base.tolist())
        if score < best_score:
            best_score = score
            best_seq = init_from_base.copy()
            print(f"  Baseline improves initial score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – smoothed gradient descent with occasional random jumps
    # ------------------------------------------------------------------
    a = best_seq.copy()
    # initial hyper‑parameters
    eta = 0.2 * (target_sum / n)          # fairly aggressive step size
    tau_factor = 0.05                     # τ = τ_factor * max_conv
    iteration = 0
    no_improve = 0

    print(f"[{time.time():.2f}] Entering smoothed gradient descent")

    while time.time() < deadline:
        # one gradient step
        a, cur_max = smooth_gradient_step(a, target_sum, eta, tau_factor)
        cur_score = cur_max                     # because sum = √(2n) ⇒
                                               # evaluate = max_conv
        iteration += 1

        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"Iter {iteration:06d} – improved to {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive schedules
        if iteration % 800 == 0:
            eta *= 0.95               # slowly decay step size
            tau_factor = max(0.001, tau_factor * 0.96)   # tighten smoothing

        # occasional random pair transfer (helps escaping plateaus)
        if np.random.rand() < 0.015:
            a = random_pair_transfer(a, target_sum, scale=0.008)

        # heavy jitter after long stagnation
        if no_improve >= 3000:
            jitter = np.random.randn(n) * 0.0005
            a = project_onto_simplex(np.maximum(a + jitter, 0.0), target_sum)
            no_improve = 0
            print(f"Iter {iteration:06d} – jitter applied")

    # ------------------------------------------------------------------
    # Phase 3 – final brief random perturbations (optional)
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Gradient descent finished; best score {best_score:.6f}")

    # One last quick pass of tiny random transfers to polish the result
    for _ in range(30):
        if time.time() >= deadline:
            break
        candidate = random_pair_transfer(best_seq.copy(), target_sum, scale=0.005)
        cand_score = evaluate_sequence(candidate.tolist())
        if cand_score < best_score:
            best_score = cand_score
            best_seq = candidate.copy()
            print(f"Polish transfer improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found as a plain Python list
    # ------------------------------------------------------------------
    return best_seq.tolist()
