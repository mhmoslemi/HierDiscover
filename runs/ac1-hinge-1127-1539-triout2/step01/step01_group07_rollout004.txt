Compute function compute_score(seq):

def compute_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    sum_a = a.sum()
    if sum_a <= 0:
        return float('inf')
    # compute conv using FFT
    m = 1 << (2*n-1).bit_length()  # next power of two >= 2n-1
    fa = np.fft.rfft(a, m)
    conv = np.fft.irfft(fa * fa, m)[:2*n-1]
    max_b = conv.max()
    score = 2.0 * n * max_b / (sum_a * sum_a)
    return float(score)

But note that we might want to ensure conv doesn't have negative rounding errors; we can use np.maximum(conv, 0). The maximum is positive anyway.

"""
Propose a step‑function with a very low evaluation score.

The algorithm explores two simple parametric families:

1. Power‑law differences:
       a_i = (i+γ+1)^p - (i+γ)^p     (0 < p < 1, γ ≥ 0)
   The vector is normalised to L1‑norm 1.

2. Geometric decay:
       a_i = (1‑q)·q**i          (0 < q < 1)

Both families are examined on a modest set of lengths `n`.  For each
parameter triple (n,p,γ) or (n,q) we compute the evaluation score
directly (using an FFT based convolution – much faster than the
reference `evaluate_sequence`).  The best candidates are kept and then
refined with a simple stochastic hill‑climbing on the continuous
parameters.  Finally the best sequence is returned (as a plain Python
list) and its true score is printed using the supplied `evaluate_sequence`
function.

The search is time‑boxed (default 1000 s) and respects all constraints of
`evaluate_sequence`.  The whole procedure runs on a single CPU core.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast computation of the evaluation function (identical to
# evaluate_sequence but without the defensive checks).  This is used
# during the search to keep it cheap.
# ----------------------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Return the same value that evaluate_sequence would return for `a`."""
    n = a.shape[0]
    if n == 0:
        return float('inf')
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float('inf')
    # convolution via FFT (real input → rfft is sufficient)
    m = 1 << (2 * n - 1).bit_length()        # next power‑of‑2 ≥ 2n‑1
    fa = np.fft.rfft(a, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def _powerlaw_seq(n: int, p: float, gamma: float) -> np.ndarray:
    """Power‑law difference sequence, normalised to L1‑norm 1."""
    i = np.arange(n, dtype=np.float64)
    a = (np.power(i + gamma + 1.0, p) - np.power(i + gamma, p))
    total = np.power(n + gamma, p) - np.power(gamma, p)
    # guard against division by zero (should never happen for p>0)
    if total == 0:
        return np.zeros_like(a)
    return a / total


def _geometric_seq(n: int, q: float) -> np.ndarray:
    """Geometric decay sequence, already normalised (∑a_i = 1)."""
    if q <= 0.0 or q >= 1.0:
        return np.zeros(n, dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - q) * np.power(q, i)
    return a  # sum is exactly 1 (up to floating‑point noise)


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function with evaluation < 1.503.

    The search proceeds in three stages:
      1. Coarse grid over a few lengths and the two families.
      2. Keep the best few candidates and perform a stochastic refinement
         on the continuous parameters (p, gamma) or (q).
      3. Return the best sequence (as a Python list) after a final call to
         `evaluate_sequence` for reporting.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_score = float('inf')
    best_seq = None
    best_info = None   # tuple describing the family and its parameters

    # --------------------------------------------------------------
    # 1. Coarse grid search
    # --------------------------------------------------------------
    # candidate lengths (small enough to keep the FFT cheap, large enough
    # to give a non‑trivial shape)
    n_candidates = [200, 300, 500, 800, 1200, 2000, 3000, 4000]
    # power‑law grid
    p_grid = np.linspace(0.05, 0.70, 27)               # step ≈0.025
    gamma_grid = np.linspace(0.0, 200.0, 41)          # step = 5
    # geometric grid
    q_grid = np.linspace(0.60, 0.999, 35)             # very decay‑slow

    for n in n_candidates:
        # -------- power‑law family ----------
        for p in p_grid:
            for gamma in gamma_grid:
                if time.time() > deadline:
                    break
                seq = _powerlaw_seq(n, float(p), float(gamma))
                sc = _fast_score(seq)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    best_info = ('pow', n, float(p), float(gamma))
                    print(f"[grid‑pow] new best score={sc:.6f} (n={n}, p={p:.3f}, γ={gamma:.1f})")
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

        # -------- geometric family ----------
        for q in q_grid:
            if time.time() > deadline:
                break
            seq = _geometric_seq(n, float(q))
            sc = _fast_score(seq)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                best_info = ('geo', n, float(q))
                print(f"[grid‑geo] new best score={sc:.6f} (n={n}, q={q:.3f})")
        if time.time() > deadline:
            break

    # --------------------------------------------------------------
    # 2. Local stochastic refinement around the current best.
    # --------------------------------------------------------------
    if best_seq is None:
        # very unlikely, but fall back to a tiny uniform sequence
        best_seq = np.full(200, 1.0 / 200.0)
        best_score = _fast_score(best_seq)

    # Parameters extracted from best_info for convenient updates
    family = best_info[0]
    n_best = best_info[1]

    # initial parameters (they will be overwritten during refinement)
    if family == 'pow':
        p_cur, gamma_cur = best_info[2], best_info[3]
    else:
        q_cur = best_info[2]

    # refinement schedule – start with larger steps, then shrink
    step_info = [
        (0.02, 5.0, 0.2),   # (σ_p, σ_γ, prob_update)
        (0.008, 2.0, 0.5),
        (0.002, 0.5, 0.8),
    ]

    while time.time() < deadline:
        # choose a schedule entry according to elapsed time
        elapsed = time.time() + budget_s - deadline
        # simply cycle through step_info
        for sigma_p, sigma_gamma, prob in step_info:
            if time.time() >= deadline:
                break
            if family == 'pow':
                if np.random.rand() > prob:
                    continue
                p_new = p_cur + (np.random.rand() - 0.5) * 2 * sigma_p
                gamma_new = gamma_cur + (np.random.rand() - 0.5) * 2 * sigma_gamma
                if p_new <= 0.01 or p_new >= 0.99 or gamma_new < 0:
                    continue
                seq_new = _powerlaw_seq(n_best, p_new, gamma_new)
                sc_new = _fast_score(seq_new)
                if sc_new < best_score:
                    best_score = sc_new
                    best_seq = seq_new.copy()
                    p_cur, gamma_cur = p_new, gamma_new
                    print(f"[refine‑pow] improved score={sc_new:.6f} (n={n_best}, p={p_new:.4f}, γ={gamma_new:.2f})")
            else:   # geometric family
                if np.random.rand() > prob:
                    continue
                q_new = q_cur + (np.random.rand() - 0.5) * 2 * sigma_p   # reuse sigma_p as σ_q
                if not (0.01 < q_new < 0.999):
                    continue
                seq_new = _geometric_seq(n_best, q_new)
                sc_new = _fast_score(seq_new)
                if sc_new < best_score:
                    best_score = sc_new
                    best_seq = seq_new.copy()
                    q_cur = q_new
                    print(f"[refine‑geo] improved score={sc_new:.6f} (n={n_best}, q={q_new:.4f})")

    # --------------------------------------------------------------
    # 3. Final sanity check using the official evaluator and return.
    # --------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        real_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence returned {real_score:.6f}")
    except Exception as e:
        print("[final] error calling evaluate_sequence:", e)
        real_score = None

    print("Best parameters:", best_info)
    print("Best score (fast estimate) :", best_score)
    print("Length of returned sequence :", len(final_seq))
    return final_seq
