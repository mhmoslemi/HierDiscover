"""
Propose a step‑function (non‑negative vector) that minimizes
`evaluate_sequence`.  The search consists of three parts:

1. **Rich set of initial guesses** – uniform, several parametric U‑shapes,
   and a family of “arcsine’’ (U‑shaped) distributions
   a_i ∝ ( (i+½)(n‑i‑½) )^{-β/2} for various β.  The best of these
   starts the optimisation.

2. **Projected sub‑gradient descent** – exactly as in the reference
   implementation, but with a simple momentum term and a built‑in
   early‑stop when the time budget is exhausted.

3. **Smoothed‑max gradient descent** – we minimise a smooth approximation
   of the max‑convolution,
        L_α(a) = (1/α)·log∑_k exp(α·(a*a)_k) .
   Its gradient is computed analytically via a second FFT convolution.
   The parameter α is annealed from 1 to ≈15, and a back‑tracking line‑search
   adapts the step size.  Random pairwise transfers and occasional jitter
   keep the search from stagnating.

All convolutions are performed with FFTs (O(n log n)) and the current
objective value is obtained directly from the self‑convolution (the
`evaluate_sequence` value equals the maximal convolution entry when the
ℓ₁‑norm is fixed to √(2n)).  The algorithm respects the 1000 s time budget
and always returns the best sequence found.
"""

import time

def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution of a with itself using a power‑of‑two FFT.
    Returns the full convolution of length 2*len(a)-1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())        # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of a and b using a power‑of‑two FFT."""
    n = a.shape[0]
    m = b.shape[0]
    L = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.ifft(fa * fb, n=L).real
    return conv[: n + m - 1]


def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector to have ℓ₁‑norm equal to *target*."""
    s = v.sum()
    if s > 0.0:
        return v * (target / s)
    # fallback – uniform distribution
    return np.full_like(v, target / v.shape[0])


def generate_U_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
) -> np.ndarray:
    """Symmetric “U‑shaped’’ sequence of given length."""
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # normalized distance ∈[0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_to_sum(raw, target_sum)


def generate_arcsine_shape(
    n: int,
    target_sum: float,
    beta: float = 1.0,
) -> np.ndarray:
    """Discrete approximation of ( (i+½)(n‑i‑½) )^{-β/2}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    prod = i * (n - i)                     # (i+½)(n‑i‑½)
    raw = prod ** (-beta / 2.0)
    raw = np.clip(raw, 0.0, None)
    return project_to_sum(raw, target_sum)


def subgradient_step(
    a: np.ndarray,
    target_sum: float,
    eta: float,
    rel_tol: float = 0.0,
) -> np.ndarray:
    """One projected sub‑gradient step on the max‑convolution."""
    n = a.shape[0]
    conv = conv_self_fft(a)
    max_val = conv.max()
    if rel_tol <= 0.0:
        act = np.array([np.argmax(conv)], dtype=np.int64)
    else:
        act = np.where(conv >= max_val * (1.0 - rel_tol))[0]

    g = np.zeros_like(a)
    idx = np.arange(n, dtype=np.int64)
    for k in act:
        j = k - idx
        mask = (j >= 0) & (j < n)
        g[mask] += 2.0 * a[j[mask]]

    # keep the sum unchanged
    direction = -g + np.mean(g)
    a_new = a + eta * direction
    a_new = np.clip(a_new, 0.0, None)
    a_new = project_to_sum(a_new, target_sum)
    return a_new


def smooth_gradient_step(
    a: np.ndarray,
    target_sum: float,
    eta: float,
    alpha: float,
) -> np.ndarray:
    """
    Gradient step for the smoothed max objective:
        L_α(a) = (1/α)·log∑_k exp(α·(a*a)_k)
    """
    n = a.shape[0]
    conv = conv_self_fft(a)                     # length 2n‑1
    maxc = conv.max()
    # numerically stable softmax over the convolution entries
    w = np.exp(alpha * (conv - maxc))
    w /= w.sum()                                # w_k ≥ 0, ∑ w = 1

    # gradient of L_α : 2 * (a * w_rev) evaluated at the appropriate indices
    w_rev = w[::-1]                             # reverse the weight vector
    grad_full = conv_fft(a, w_rev)              # length 3n‑2
    M = 2 * n - 2
    grad = 2.0 * grad_full[M - np.arange(n)]

    # project to the hyperplane sum(a)=target_sum
    direction = -grad + np.mean(grad)
    a_new = a + eta * direction
    a_new = np.clip(a_new, 0.0, None)
    a_new = project_to_sum(a_new, target_sum)
    return a_new


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0.0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_to_sum(a, target_sum)


def compute_score(a: np.ndarray) -> float:
    """Fast surrogate for evaluate_sequence when the ℓ₁‑norm is fixed."""
    return float(conv_self_fft(a).max())

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    `evaluate_sequence`.  The search respects *budget_s* seconds and
    always returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5        # 0.5 s safety margin

    # ------------------------------------------------------------------
    # Choose length – reuse known solution if available, otherwise random.
    # ------------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
    else:
        n = np.random.randint(8000, 12000)

    target_sum = np.sqrt(2.0 * n)                  # normalisation used by evaluate_sequence

    # ------------------------------------------------------------------
    # Phase 0 – generate a diverse set of seeds and keep the best.
    # ------------------------------------------------------------------
    best_seq = np.full(n, target_sum / n)          # uniform start
    best_score = compute_score(best_seq)

    # optional: use the provided sequence if present
    if isinstance(base, (list, tuple)):
        cand = np.array(base, dtype=np.float64)
        cand = project_to_sum(cand, target_sum)
        cand_score = compute_score(cand)
        if cand_score < best_score:
            best_score, best_seq = cand_score, cand.copy()
            print(f"[init] Base sequence improves to {best_score:.6f}")

    # --- simple U‑shapes -------------------------------------------------
    shape_kinds = ("linear", "quadratic", "exponential")
    param_grids = {
        "linear": np.linspace(0.0, 4.0, 41),
        "quadratic": np.linspace(0.0, 3.0, 31),
        "exponential": np.linspace(0.0, 2.0, 21),
    }
    for kind in shape_kinds:
        for p in param_grids[kind]:
            cand = generate_U_shape(n, target_sum, kind=kind, param=p)
            cand_score = compute_score(cand)
            if cand_score < best_score:
                best_score, best_seq = cand_score, cand.copy()
                print(f"[init] U‑shape {kind} p={p:.2f} → {best_score:.6f}")
            if time.time() >= deadline:
                print("Time exhausted during initial sweep.")
                return best_seq.tolist()

    # --- arcsine‑type shapes --------------------------------------------
    for beta in (0.5, 0.75, 1.0, 1.25, 1.5):
        cand = generate_arcsine_shape(n, target_sum, beta=beta)
        cand_score = compute_score(cand)
        if cand_score < best_score:
            best_score, best_seq = cand_score, cand.copy()
            print(f"[init] Arcsine β={beta:.2f} → {best_score:.6f}")
        if time.time() >= deadline:
            print("Time exhausted during arcsine sweep.")
            return best_seq.tolist()

    # ------------------------------------------------------------------
    # Phase 1 – projected sub‑gradient descent (with momentum style).
    # ------------------------------------------------------------------
    a = best_seq.copy()
    eta = 0.02 * (target_sum / n)          # ~2 % of the average entry
    rel_tol = 0.0
    iteration = 0
    no_improve = 0
    momentum = np.zeros_like(a)

    print(f"[{time.time():.2f}] Starting sub‑gradient phase (n={n})")
    while time.time() < deadline and iteration < 25000:
        # classic sub‑gradient step
        g = subgradient_step(a, target_sum, eta, rel_tol)
        # simple momentum (exponential moving average of the change)
        momentum = 0.8 * momentum + 0.2 * (g - a)
        a = g + momentum
        a = np.clip(a, 0.0, None)
        a = project_to_sum(a, target_sum)

        iteration += 1
        cur_score = compute_score(a)

        if cur_score + 1e-12 < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"Iter {iteration:05d} – sub‑grad improved to {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # step‑size and active‑set schedule
        if iteration % 500 == 0:
            eta *= 0.9
            eta = max(eta, 1e-6)
            rel_tol = min(0.05, rel_tol + 0.005)

        # occasional random transfer
        if np.random.rand() < 0.02:
            a = random_pair_transfer(a, target_sum, scale=0.005)

        # jitter after long stagnation
        if no_improve >= 3000:
            jitter = np.random.randn(n) * 0.001
            a = np.clip(a + jitter, 0.0, None)
            a = project_to_sum(a, target_sum)
            no_improve = 0
            print(f"Iter {iteration:05d} – jitter applied")

    # ------------------------------------------------------------------
    # Phase 2 – smoothed‑max gradient descent with annealing α.
    # ------------------------------------------------------------------
    a = best_seq.copy()
    eta = 0.01 * (target_sum / n)
    alpha = 1.0
    alpha_max = 15.0
    iter2 = 0
    print(f"[{time.time():.2f}] Starting smoothed‑max phase")
    while time.time() < deadline and iter2 < 18000:
        # increase α gradually
        if iter2 % 500 == 0 and alpha < alpha_max:
            alpha = min(alpha_max, alpha * 1.3)

        improved = False
        # try a few step‑size shrinkings (back‑tracking)
        for factor in (1.0, 0.5, 0.25):
            cand = smooth_gradient_step(a, target_sum, eta * factor, alpha)
            cand_score = compute_score(cand)
            if cand_score + 1e-12 < best_score:
                best_score, best_seq = cand_score, cand.copy()
                a = cand
                improved = True
                print(f"Iter {iteration+iter2:05d} – smooth α={alpha:.2f} → {best_score:.6f}")
                break
        if not improved:
            # no gain → shrink step size or inject a tiny random move
            eta *= 0.7
            if eta < 1e-7:
                a = random_pair_transfer(a, target_sum, scale=0.002)
                eta = 0.01 * (target_sum / n)

        # occasional random pair transfer to keep diversity
        if np.random.rand() < 0.01:
            a = random_pair_transfer(a, target_sum, scale=0.005)

        iter2 += 1

    # ------------------------------------------------------------------
    # Phase 3 – final quick random perturbations (optional).
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        a = random_pair_transfer(a, target_sum, scale=0.003)
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"Final pert. new best {best_score:.6f}")

    print(f"[{time.time():.2f}] Search finished. Best score {best_score:.6f}")
    return best_seq.tolist()
