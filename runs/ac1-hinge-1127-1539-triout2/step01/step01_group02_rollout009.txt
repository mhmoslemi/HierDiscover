"""
Propose a step‑function (non‑negative sequence) with a low value of
evaluate_sequence.  The algorithm runs in three stages:

1.  Quick parametric sweep (power‑law, exponential, random Dirichlet) to get
    a decent starting point.
2.  A smooth “soft‑max” mirror‑descent that reduces the largest entries of
    the convolution.  The gradient is built from a weighted sum of the
    convolution entries (higher weight for larger values) and the update is
    multiplicative followed by a projection onto the probability simplex.
    Several restarts with different initialisations are performed until the
    time budget expires.
3.  Post‑processing: try truncating the best sequence to shorter lengths
    (renormalising each time) – shorter vectors may give a smaller
    2·n·max(conv) value.

The routine respects the 1000 s budget and returns the best sequence found.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex {x>=0, sum=1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _generate_powerlaw(p: float, sigma: float, max_len: int = 5000,
                       eps: float = 1e-8) -> np.ndarray:
    """Return a truncated power‑law a_i = (i+σ)^{‑p}."""
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p)
    mask = a > eps
    return a[mask]


def _generate_exponential(lam: float, max_len: int = 5000,
                          eps: float = 1e-8) -> np.ndarray:
    """Return a truncated exponential a_i = exp(-λ·i)."""
    i = np.arange(max_len, dtype=np.float64)
    a = np.exp(-lam * i)
    mask = a > eps
    return a[mask]


def _softmax_gradient(x: np.ndarray,
                      conv: np.ndarray,
                      weight_power: float,
                      top_k: int = None) -> np.ndarray:
    """
    Build a gradient for the soft‑max version of the ℓ∞‑norm.
    If top_k is supplied, only the *top_k* convolution entries are used
    (binary weights).  Otherwise, weights are taken as conv**weight_power
    (normalised to sum to one).
    Returns the gradient vector (length = len(x)).
    """
    if top_k is not None and top_k < conv.shape[0]:
        # binary mask on the largest entries
        idx = np.argpartition(-conv, top_k)[:top_k]
        w = np.zeros_like(conv)
        w[idx] = 1.0
    else:
        # power‑weighted soft max
        w = np.power(conv, weight_power)
        s = w.sum()
        if s > 0:
            w /= s
    # gradient of 2* n * soft‑max(conv) w.r.t. x:
    # ∂conv_k/∂x_i = 2 * x_{k-i} (zero outside)
    grad = 2.0 * np.correlate(w, x, mode='valid')
    return grad


def _mirror_descent(x: np.ndarray,
                    it: int,
                    weight_power: float,
                    top_k_frac: float) -> np.ndarray:
    """
    One mirror‑descent iteration using a soft‑max / top‑K gradient.
    """
    n = x.shape[0]
    conv = np.convolve(x, x)        # full convolution, length 2n-1
    # choose the number of top elements to consider
    top_k = max(1, int(top_k_frac * conv.shape[0]))
    grad = _softmax_gradient(x, conv, weight_power, top_k=top_k)
    # step size – decays like 1/√t, with a moderate base
    eta = 0.7 / np.sqrt(it + 1.0)
    # multiplicative (exponential) update, then project onto the simplex
    y = x * np.exp(-eta * grad)
    y = _project_simplex(y)
    return y


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **_):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The function respects the given time budget and returns the best
    sequence discovered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # a safety margin

    best_score = float('inf')
    best_seq = None

    # Helper to evaluate a candidate (normalise to sum‑1 first)
    # ------------------------------------------------------------------
    def _eval(seq_arr: np.ndarray):
        if seq_arr.size == 0:
            return float('inf')
        seq = seq_arr / seq_arr.sum()
        try:
            return float(evaluate_sequence(seq.tolist()))
        except Exception:
            return float('inf')

    # ------------------------------------------------------------------

    max_len = 3000

            if best_score < 1.503:

    # exponential sweep
    if best_score >= 1.503 and time.time() < deadline:

            sc = _eval(seq)
            if sc < best_score:

            if best_score < 1.503:

                    print(f'  new best (dirichlet) α={conc:.2f} n={n} score={best_score:.6f}')
                if best_score < 1.503:
                    break
            if best_score < 1.503 or time.time() >= deadline:
                break

    # ------------------------------------------------------------------
    print('=== Phase 2 : soft‑max mirror descent ===')
    restart_counter = 0
    while time.time() < deadline:
        restart_counter += 1
        # ------------------------------------------------------------------
        # Initialise a fresh candidate for this restart
        # ------------------------------------------------------------------
        if best_seq is not None and restart_counter % 3 == 0:
            # occasionally warm‑start from the current best (scaled to sum‑1)
            x = np.asarray(best_seq, dtype=np.float64)
        else:
            # random Dirichlet start with a random length
            n = rng.integers(1500, max_len + 1)
            conc = rng.uniform(0.2, 5.0)
            x = rng.dirichlet(np.full(n, conc))
        x = _project_simplex(x)  # guarantee sum‑1

        # ------------------------------------------------------------------
        # Run mirror‑descent iterations for this restart
        # ------------------------------------------------------------------
        no_improve = 0
        it = 0
        max_iter_restart = 6000
        while it < max_iter_restart and time.time() < deadline:
            it += 1
            # weight_power slowly grows to focus more on the true peak
            weight_power = 1.0 + min(5.0, it / 1500.0)
            top_k_frac = 0.005  # use the top 0.5 % of convolution entries
            x = _mirror_descent(x, it, weight_power, top_k_frac)

            # occasional jitter to escape plateaus
            if rng.random() < 0.02:

                delta = (rng.random() - 0.5) * 0.04 * x[idx]
                x[idx] = max(0.0, x[idx] + delta)
                x = _project_simplex(x)

            cur_score = _eval(x)
            if cur_score < best_score - 1e-9:
                best_score, best_seq = cur_score, x.tolist()
                no_improve = 0
                print(f'  restart {restart_counter:02d} it {it:04d} -> '
                      f'new best {best_score:.6f} (len={len(x)})')
                if best_score < 1.503:
                    print('Target reached – terminating early.')
                    return best_seq
            else:
                no_improve += 1

            # if we have gone many steps without improvement, break early
            if no_improve >= 2000:

    # ------------------------------------------------------------------
    print('=== Phase 3 : truncation refinement ===')
    if best_seq is not None:

        L = full.shape[0]

        for new_len in np.linspace(int(L * 0.5), L, num=15, dtype=int):

            if seq.sum() < 1e-12:

            seq = _project_simplex(seq)   # re‑normalise to sum‑1
            sc = _eval(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.tolist()
                print(f'  truncation len={new_len} improved score={best_score:.6f}')
                if best_score < 1.503:
                    print('Target reached after truncation.')
                    return best_seq

    print('final best score:', best_score, 'length:', len(best_seq) if best_seq else None)
    return best_seq
