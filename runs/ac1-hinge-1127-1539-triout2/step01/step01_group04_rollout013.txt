def propose_candidate(seed=42, budget_s=1000):
    import time, math, numpy as np

    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0

    # Sequence lengths to try
    lengths = [500, 1000, 2000, 4000, 8000]  # adapt if time left
    best_score = float('inf')
    best_seq = None

    for n in lengths:
        # prepare L and Lgrad
        L = 1 << ((2*n - 1).bit_length())
        Lgrad = 1 << ((3*n - 2).bit_length())
        # initialize a using sqrt-differences scaled to target sum
        idx = np.arange(n, dtype=np.float64)
        a = np.sqrt(idx+1) - np.sqrt(idx)
        target_sum = math.sqrt(2.0 * n)
        a = a * (target_sum / a.sum())
        # maybe add small random perturbation to start
        a += np.random.randn(n) * target_sum * 1e-4
        a = np.clip(a, 0, None)
        a *= (target_sum / a.sum())

        best_local_seq = a.copy()
        # compute initial score via evaluate_sequence:
        try:
            cur_score = evaluate_sequence(a.tolist())
        except Exception:
            cur_score = float('inf')
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()

        iteration = 0
        alpha = 5.0
        alpha_inc = (30.0 - alpha) / 20000  # increase to 30 over many iter

        while time.time() < deadline:
            # Compute self-convolution using FFT
            fa = np.fft.rfft(a, n=L)
            conv = np.fft.irfft(fa*fa, n=L)[:2*n-1]
            # compute max directly
            max_val = conv.max()

            # Compute weight vector w
            # shift for numeric stability
            shifted = conv - max_val
            # compute exponent scaled by alpha
            # to avoid overflow, we can clamp large exponents: but after shift you are <=0, so safe
            w = np.exp(alpha * shifted)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.ones_like(w) / w.size
            else:
                w = w / w_sum

            # compute gradient using convolution with reversed w
            w_rev = w[::-1]
            fa2 = np.fft.rfft(w_rev, n=Lgrad)
            fb2 = np.fft.rfft(a, n=Lgrad)
            conv_grad_raw = np.fft.irfft(fa2 * fb2, n=Lgrad)
            # conv_grad_raw length Lgrad, but we need first 3n-2 elements (full conv)
            # We'll slice the valid part: while rfft yields length Lgrad, we should take first 3n-2 value (since other zero-padded tail)
            conv_grad_raw = conv_grad_raw[: 3*n - 2]
            # compute gradient: grad_i = 2 * conv_grad_raw[2*n-2 - i]
            idx_rev = (2*n - 2) - np.arange(n)
            grad = 2.0 * conv_grad_raw[idx_rev]

            # step size schedule
            lr = 0.5 / math.sqrt(iteration + 1.0)  # tuning constant
            a = a - lr * grad

            # projection
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s == 0.0:
                a = np.full(n, target_sum / n, dtype=np.float64)
            else:
                a = a * (target_sum / s)

            # jitter every 200 iter
            if iteration % 200 == 0 and iteration > 0:
                sigma = target_sum * 1e-3 / math.sqrt(n)
                a += np.random.randn(n) * sigma
                a = np.maximum(a, 0.0)
                a = a * (target_sum / a.sum())

            # evaluate via evaluation function
            try:
                cur_score = evaluate_sequence(a.tolist())
            except Exception:
                cur_score = float('inf')
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"n={n} iter={iteration} new best {best_score:.6f}")

            # update alpha
            if alpha < 30.0:
                alpha += alpha_inc

            iteration += 1
            # exit condition possibly enable early break if iteration too large
            if iteration > 200000:  # safety many iterations
                break

        # maybe break if time low
        if time.time() >= deadline - 5:
            break

    # return best_seq
    return best_seq.tolist()

"""
Propose a high‑quality non‑negative step function.

The algorithm works in two phases.

1) **Smoothed gradient descent** – we minimise a smooth approximation
   of the infinity‑norm of the self‑convolution:
       F(a) = (1/α)·log Σ_k exp(α·(a*a)_k) .
   The gradient of F is 2·(a * w_rev) where w is the normalised
   soft‑max of the convolution values.  Using a decaying learning rate,
   a simple momentum term and a projection onto the feasible set
   (non‑negative and fixed ℓ₁‑norm = √(2·n)) we obtain a fast
   non‑convex optimiser that continuously improves the max‑convolution.

2) **Local random mass‑transfer refinement** – after the main descent we
   spend any remaining time performing cheap random transfers of a tiny
   amount of mass from one index to another; a move is kept if it lowers
   the true evaluation (the maximum convolution).  This helps to escape
   the very flat regions that the smooth objective may miss.

All heavy work (convolutions, gradients) is performed with FFTs, so the
routine stays well within the 1000 s budget even for several‑thousand‑item
sequences.  The best sequence found is returned as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – pure top‑level, no closures
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x (x>0)."""
    return 1 << (x - 1).bit_length()


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full self‑convolution of a real vector using FFT length L."""
    Fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Fa * Fa, n=L)
    return conv[: 2 * a.size - 1]


def _conv_fft(a: np.ndarray, b: np.ndarray, L: int) -> np.ndarray:
    """Full convolution a ⋆ b using FFT length L."""
    Fa = np.fft.rfft(a, n=L)
    Fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Fa * Fb, n=L)
    return conv[: a.size + b.size - 1]

# ----------------------------------------------------------------------

    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.

    Returns
    -------
    list[float]
        The best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Choose a length.  If a previous construction is available we reuse it,
    # otherwise fall back to a moderate default.
    # ------------------------------------------------------------------
    try:
        # height_sequence_1 may be defined by the surrounding system
        a0 = np.array(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        n = a0.size
    except Exception:
        n = 8000                # a comfortable size for the FFT based method
        idx = np.arange(n, dtype=np.float64)
        a0 = np.sqrt(idx + 1.0) - np.sqrt(idx)   # √(i+1)‑√i
    # ------------------------------------------------------------------
    # Normalise the ℓ₁‑norm to √(2·n) (this makes the evaluation equal to the
    # maximum entry of the self‑convolution).
    # ------------------------------------------------------------------
    target_sum = math.sqrt(2.0 * n)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, target_sum / n, dtype=np.float64)
    else:
        a *= (target_sum / s)

    # Keep the best ever seen (score = max self‑convolution because of the
    # normalisation).
    conv = _conv_self_fft(a, _next_pow2(2 * n - 1))
    best_score = float(conv.max())
    best_seq = a.copy()
    print(f"Initial score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Pre‑compute FFT lengths for the two convolutions we need each iteration.
    # ------------------------------------------------------------------
    L_self = _next_pow2(2 * n - 1)            # length for a*a
    L_grad = _next_pow2(3 * n - 2)            # length for a * w_rev

    # Hyper‑parameters (tuned heuristically)
    step0 = 0.15                               # base learning‑rate factor
    momentum_coeff = 0.9                       # momentum term
    alpha = 5.0                                # start of soft‑max temperature
    alpha_target = 30.0
    # increase α linearly over the first 60 000 iterations
    max_alpha_iters = 60000
    alpha_inc = (alpha_target - alpha) / max_alpha_iters

    momentum = np.zeros_like(a, dtype=np.float64)
    iteration = 0

    # ------------------------------------------------------------------
    # Phase 1 – smoothed gradient descent
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ----- self‑convolution and current score -----
        conv_self = _conv_self_fft(a, L_self)
        M = conv_self.max()
        cur_score = float(M)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"Iter {iteration:7d} – new best {best_score:.6f}")

        # ----- soft‑max weights (smooth approximation of the max) -----
        # shifting by M prevents overflow; all exponentials are ≤ 1
        shifted = conv_self - M
        w = np.exp(alpha * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        w_rev = w[::-1]
        Fa = np.fft.rfft(a, n=L_grad)
        Fwr = np.fft.rfft(w_rev, n=L_grad)
        conv_grad_full = np.fft.irfft(Fa * Fwr, n=L_grad)
        conv_grad_full = conv_grad_full[: 3 * n - 2]          # linear convolution length
        # gradient_i = 2 * conv_grad_full[2·n‑2‑i]
        idx_rev = (2 * n - 2) - np.arange(n)
        grad = 2.0 * conv_grad_full[idx_rev]

        # ----- momentum update -----
        momentum = momentum_coeff * momentum + (1.0 - momentum_coeff) * grad

        # ----- step size (decays like 1/√t) -----
        lr = step0 / math.sqrt(iteration + 1.0)

        # ----- projected gradient step -----
        a = a - lr * momentum
        a = np.maximum(a, 0.0)            # non‑negativity
        s = a.sum()
        if s == 0.0:
            a = np.full(n, target_sum / n, dtype=np.float64)
        else:
            a *= (target_sum / s)        # renormalise ℓ₁‑norm

        # ----- occasional tiny random jitter (helps escape plateaus) -----
        if iteration % 200 == 0 and iteration > 0:
            sigma = target_sum * 1e-3 / math.sqrt(n)
            a += np.random.randn(n) * sigma
            a = np.maximum(a, 0.0)
            a *= (target_sum / a.sum())

        # ----- temperature schedule -----
        if iteration < max_alpha_iters:
            alpha += alpha_inc

        iteration += 1

        # safety break if we have already spent a large part of the budget
        if iteration >= 200000:
            break

    # ------------------------------------------------------------------
    # Phase 2 – cheap random mass‑transfer refinement (uses exact score)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # pick two distinct indices
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j or a[i] <= 0.0:
            continue
        # transfer a small fraction of the mass at i to j
        delta = a[i] * np.random.rand() * 0.03      # up to 3 % of a[i]
        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        # exact max‑convolution (still via FFT for speed)
        conv_cand = _conv_self_fft(a_candidate, L_self)
        cand_score = float(conv_cand.max())

        if cand_score < best_score:
            best_score = cand_score
            best_seq = a_candidate.copy()
            a = a_candidate
            print(f"Local move {iteration:7d} – new best {best_score:.6f}")

        iteration += 1

    print(f"Finished – best score {best_score:.6f} (length {n})")
    return best_seq.tolist()
