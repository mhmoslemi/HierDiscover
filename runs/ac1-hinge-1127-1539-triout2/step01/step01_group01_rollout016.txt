"""
Propose a high‑quality non‑negative height sequence for the
self‑convolution bound.

Algorithm
---------
1. **Coarse parametric sweep** – generate a few symmetric “U‑shaped’’
   families (linear, quadratic, exponential, inverse) and keep the best
   as a warm start.

2. **Smoothed max‑gradient descent** – at each iteration the maximum of
   the self‑convolution is approximated by a soft‑max
   (log‑sum‑exp) with a temperature β.
   The gradient of the soft‑max is a weighted correlation of the current
   sequence with the weight vector w∝exp(β·b).  A projected (sum‑preserving)
   gradient step is taken with a decreasing step size.
   β is slowly increased (annealing) so the method starts with a smooth
   landscape and ends up refining the true max‑convolution.

3. **Exploration tricks** – occasional random pair transfers,
   occasional Gaussian jitter when progress stalls, and a few
   sub‑gradient steps (the “hard’’ max) at the very end.

All operations are O(n log n) via FFT, allowing n≈8‑10 k to be handled
comfortably within the 1000 s budget.  The routine always returns the
best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution (full) using a power‑of‑two FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())            # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector to have ℓ₁‑norm = target."""
    s = v.sum()
    if s > 0:
        return v * (target / s)
    # fallback – uniform distribution (should never happen)
    return np.full_like(v, target / v.shape[0])


def generate_U_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
) -> np.ndarray:
    """Symmetric “U‑shaped’’ sequence of length n, scaled to target_sum."""
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "inverse":
        # decreasing towards the edges (opposite of a U)
        raw = 1.0 / (1.0 + param * d)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_to_sum(raw, target_sum)


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute  ∇_a  (∑_k w_k (a*a)_k) = 2 * correlation(a, w).
    Implemented via FFT: correlation = conv_rev where a_rev is reversed a.
    """
    n = a.shape[0]
    a_rev = a[::-1]                     # reversed copy
    total_len = a_rev.shape[0] + w.shape[0] - 1
    L = 1 << (total_len.bit_length())
    FA = np.fft.rfft(a_rev, n=L)
    FW = np.fft.rfft(w, n=L)
    conv_rev = np.fft.irfft(FA * FW, n=L)[:total_len]
    start = n - 1                         # offset that yields Σ a_j w_{i+j}
    g = 2.0 * conv_rev[start:start + n]
    return g


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """Transfer a tiny fraction of mass from i → j while preserving the sum."""
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0.0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_to_sum(a, target_sum)


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    `evaluate_sequence`.  Runs for at most *budget_s* seconds and
    returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Choose a length – reuse known construction if present
    # ------------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
    else:
        n = np.random.randint(7800, 9200)           # a large but manageable size

    target_sum = np.sqrt(2.0 * n)                    # normalisation used in the paper

    # ------------------------------------------------------------------
    # Phase 1 – simple parametric sweep (fast)
    # ------------------------------------------------------------------
    best_seq = np.full(n, target_sum / n)            # uniform start
    best_score = float("inf")

    shape_kinds = ("linear", "quadratic", "exponential", "inverse")
    param_grids = {
        "linear": np.linspace(0.0, 4.0, 41),         # 0 … 4 step 0.1
        "quadratic": np.linspace(0.0, 3.0, 31),     # 0 … 3
        "exponential": np.linspace(0.0, 2.0, 21),   # 0 … 2
        "inverse": np.linspace(0.0, 4.0, 41),       # 0 … 4
    }

    print(f"[{time.time():.2f}] Starting parametric sweep (n={n})")
    for kind in shape_kinds:
        for p in param_grids[kind]:
            cand = generate_U_shape(n, target_sum, kind=kind, param=p)
            # fast inline score (no evaluate_sequence call)
            b = conv_self_fft(cand)
            score = 2.0 * n * b.max() / (target_sum * target_sum)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"  New best {kind} (p={p:.2f}) → {best_score:.6f}")
            if time.time() >= deadline:
                print("Time exhausted during sweep.")
                return best_seq.tolist()

    # ------------------------------------------------------------------
    # Phase 2 – smoothed max‑gradient descent with annealing
    # ------------------------------------------------------------------
    a = best_seq.copy()
    # initial step size ≈ 5 % of the average entry
    eta0 = 0.05 * (target_sum / n)
    beta = 0.5           # start temperature (soft max)
    beta_max = 20.0
    iteration = 0
    no_improve = 0
    last_improve_iter = 0

    # Pre‑compute the first convolution (used for the first score)
    b = conv_self_fft(a)
    best_score = 2.0 * n * b.max() / (target_sum * target_sum)
    best_seq = a.copy()
    print(f"[{time.time():.2f}] Beginning smoothed gradient descent "
          f"(eta0={eta0:.5f}, beta={beta}) – initial score {best_score:.6f}")

    while time.time() < deadline:
        iteration += 1

        # ----- compute soft‑max weights -----
        max_b = b.max()
        shifted = b - max_b                     # shift to avoid overflow
        # exponentiate with current temperature
        exp_vals = np.exp(beta * shifted)
        sum_exp = exp_vals.sum()
        if sum_exp == 0.0:                     # numerical safety
            w = np.zeros_like(b)
            w[np.argmax(b)] = 1.0
        else:
            w = exp_vals / sum_exp               # normalized weights

        # ----- gradient of the smoothed objective -----
        g = gradient_from_weights(a, w)          # shape (n,)

        # direction that keeps the ℓ₁‑norm unchanged
        direction = -g + np.mean(g)

        # step size decays like 1/√iteration
        step = eta0 / np.sqrt(iteration)
        a = a + step * direction

        # enforce feasibility
        a = np.clip(a, 0.0, None)
        a = project_to_sum(a, target_sum)

        # ----- new convolution & true score -----
        b = conv_self_fft(a)
        cur_score = 2.0 * n * b.max() / (target_sum * target_sum)

        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
            last_improve_iter = iteration
            print(f"Iter {iteration:06d} – improved to {best_score:.6f}")
        else:
            no_improve += 1

        # ----- annealing of β -----
        if beta < beta_max:
            beta = min(beta_max, beta * 1.001)

        # ----- occasional random pair transfer (exploration) -----
        if np.random.rand() < 0.02:
            a = random_pair_transfer(a, target_sum, scale=0.005)
            b = conv_self_fft(a)   # recompute after random move

        # ----- stagnation handling -----
        if no_improve >= 5000:
            jitter = np.random.randn(n) * 0.001 * (target_sum / n)
            a = np.clip(a + jitter, 0.0, None)
            a = project_to_sum(a, target_sum)
            b = conv_self_fft(a)
            no_improve = 0
            print(f"Iter {iteration:06d} – jitter applied (stagnation)")

        # Periodic progress report
        if iteration % 2000 == 0:
            elapsed = time.time() - (deadline - budget_s + 0.5)
            print(f"[{time.time():.2f}] Iter {iteration:06d}, "
                  f"score {cur_score:.6f}, beta {beta:.3f}, elapsed {elapsed:.1f}s")

    # ------------------------------------------------------------------
    # Phase 3 – optional short sub‑gradient refinement (hard max)
    # ------------------------------------------------------------------
    # One or two extra sub‑gradient steps can sometimes shave off a tad.
    def subgradient_step(a_vec: np.ndarray, target: float, eta: float) -> np.ndarray:
        """Classic sub‑gradient step on the true max (no smoothing)."""
        n_loc = a_vec.shape[0]
        b_loc = conv_self_fft(a_vec)
        max_val = b_loc.max()
        act = np.where(b_loc >= max_val)[0][:5]      # a few active indices
        g_loc = np.zeros_like(a_vec)
        idx = np.arange(n_loc, dtype=np.int64)
        for k in act:
            j = k - idx
            mask = (j >= 0) & (j < n_loc)
            g_loc[mask] += 2.0 * a_vec[j[mask]]
        direction = -g_loc + np.mean(g_loc)
        a_next = a_vec + eta * direction
        a_next = np.clip(a_next, 0.0, None)
        return project_to_sum(a_next, target)

    # perform a handful of sub‑gradient steps if time permits
    sub_iter = 0
    sub_eta = 0.01 * (target_sum / n)
    while time.time() < deadline and sub_iter < 50:
        sub_iter += 1
        a = subgradient_step(a, target_sum, sub_eta)
        b = conv_self_fft(a)
        cur_score = 2.0 * n * b.max() / (target_sum * target_sum)
        if cur_score + 1e-13 < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"Sub‑grad {sub_iter:02d} – improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found (plain Python list)
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Search finished – best score {best_score:.6f}")
    return best_seq.tolist()
