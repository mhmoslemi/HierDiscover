"""
Propose a step‑function (non‑negative sequence) with a low value of

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works in two phases:

1. **Parameterised random search** – many simple power‑law shapes
   a_i = 1/(i+d)^p (and a few exponential shapes) are sampled, normalised
   to sum = sqrt(2n) and evaluated with a fast FFT based scorer.
   The best shape found becomes the starting point for the second phase.

2. **Soft‑max gradient descent with Adam** – the true objective contains a
   hard maximum; we approximate it by a soft‑max with an increasing temperature
   γ.  The gradient of the smoothed objective is
        g = 2 * ( w ⊛ a_rev )
   where w are the soft‑max weights on the convolution entries.
   Adam updates, non‑negativity projection and renormalisation keep the
   iterates feasible.  γ is increased slowly (focus on the true max) while
   the learning rate decays.  Random jitter is added occasionally to escape
   plateaus.

Both phases respect the overall time budget (default 1000 s).  The best
sequence discovered is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fast_score(a: np.ndarray, nfft_conv: int, target_sum: float) -> float:
    """
    Compute the evaluation score (without the security wrapper) using FFT.
    a must already be normalised so that sum(a) = target_sum.
    """
    n = a.shape[0]
    conv = np.fft.irfft(
        np.fft.rfft(a, n=nfft_conv) * np.fft.rfft(a, n=nfft_conv), n=nfft_conv
    )[: 2 * n - 1]
    max_conv = conv.max()
    # With the normalisation sum(a) == target_sum the denominator is target_sum**2
    return float(2.0 * n * max_conv / (target_sum * target_sum))


def _generate_power_shape(p: float, d: float, n: int, target_sum: float) -> np.ndarray:
    """a_i = 1 / (i + d) ** p, normalised to have the required sum."""
    idx = np.arange(n, dtype=np.float64)
    a = 1.0 / np.power(idx + d, p)
    s = a.sum()
    if s == 0.0:
        a = np.full_like(a, target_sum / n)
    else:
        a = a * (target_sum / s)
    return a


def _generate_exp_shape(alpha: float, n: int, target_sum: float) -> np.ndarray:
    """a_i = exp(-alpha * i), normalised."""
    a = np.exp(-alpha * np.arange(n, dtype=np.float64))
    s = a.sum()
    if s == 0.0:
        a = np.full_like(a, target_sum / n)
    else:
        a = a * (target_sum / s)
    return a

    # Determine length of the problem (use previous best if available)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        n = 5000  # fallback size

    target_sum = np.sqrt(2.0 * n)          # we keep sum(a) = sqrt(2n)
    len_conv = 2 * n - 1
    nfft_conv = _next_pow2(len_conv)       # size for a*a convolution
    nfft_corr = _next_pow2(3 * n - 2)      # size for w ⊛ a_rev correlation

    # ------------------------------------------------------------------
    # Phase 1 – cheap random search over simple parametric families
    # ------------------------------------------------------------------
    best_a = None
    best_score = float("inf")

    # Reserve a few seconds for the final optimisation phase
    random_search_time = 30.0
    rand_deadline = deadline - random_search_time

    print(f"Starting random search (n={n}) …")
    while time.time() < rand_deadline:
        # Choose a family
        family = np.random.choice(["power", "exp"])
        if family == "power":
            # power‑law shape: a_i = 1/(i+d)^p
            p = np.random.uniform(0.35, 0.85)   # exponent
            d = np.random.uniform(0.05, 5.0)   # offset
            a = _generate_power_shape(p, d, n, target_sum)
        else:
            # simple exponential decay
            alpha = np.random.uniform(5e-4, 2e-2)
            a = _generate_exp_shape(alpha, n, target_sum)

        score = _fast_score(a, nfft_conv, target_sum)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            elapsed = time.time() - start_time
            print(
                f"  Random search new best: {best_score:.6f}  "
                f"(elapsed {elapsed:.1f}s, family={family})"
            )

        # also occasionally try the classic 1/√(i+0.5) shape as a sanity check
        if np.random.rand() < 0.001:
            a0 = 1.0 / np.sqrt(np.arange(n, dtype=np.float64) + 0.5)
            a0 = a0 * (target_sum / a0.sum())
            s0 = _fast_score(a0, nfft_conv, target_sum)
            if s0 < best_score:
                best_score = s0
                best_a = a0.copy()
                elapsed = time.time() - start_time
                print(
                    f"  Random search new best (baseline): {best_score:.6f} "
                    f"(elapsed {elapsed:.1f}s)"
                )

    # If, for any reason, random search found nothing, fall back to the
    # analytical 1/√(i+0.5) construction.
    if best_a is None:
        best_a = 1.0 / np.sqrt(np.arange(n, dtype=np.float64) + 0.5)
        best_a = best_a * (target_sum / best_a.sum())
        best_score = _fast_score(best_a, nfft_conv, target_sum)

    print(f"Random search finished. Best score so far: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – soft‑max gradient descent with Adam
    # ------------------------------------------------------------------
    a = best_a.copy()
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # Hyper‑parameters for the soft‑max and the optimiser
    gamma = 5.0                # start temperature for the soft‑max
    gamma_max = 200.0
    lr = 0.12                  # initial Adam learning‑rate (effective step size)
    lr_min = 0.002

    iteration = 0
    print("Starting gradient descent …")
    while time.time() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # 1. Full convolution a * a (FFT)
        # --------------------------------------------------------------
        af = np.fft.rfft(a, n=nfft_conv)
        conv = np.fft.irfft(af * af, n=nfft_conv)[:len_conv]
        max_conv = conv.max()

        # --------------------------------------------------------------
        # 2. Soft‑max weights for the smooth maximum
        # --------------------------------------------------------------
        # subtract max to keep numbers ≤ 0 and avoid overflow
        w = np.exp(gamma * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:                     # safety fallback
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --------------------------------------------------------------
        # 3. Gradient of the smooth maximum: g = 2 * (w ⊛ a_rev)
        # --------------------------------------------------------------
        wf = np.fft.rfft(w, n=nfft_corr)
        af_rev = np.fft.rfft(a[::-1], n=nfft_corr)
        corr_full = np.fft.irfft(wf * af_rev, n=nfft_corr)
        # valid slice starts at index n‑1 and contains n elements
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]

        # --------------------------------------------------------------
        # 4. Adam update
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)   # project onto the non‑negative orthant

        # --------------------------------------------------------------
        # 5. Renormalise to keep sum(a) = target_sum (scale invariant)
        # --------------------------------------------------------------
        s = a_new.sum()
        if s == 0.0:
            a_new = np.full_like(a_new, target_sum / n)
        else:
            a_new = a_new * (target_sum / s)

        # --------------------------------------------------------------
        # 6. Occasionally inject a tiny random jitter (helps escape plateaux)
        # --------------------------------------------------------------
        if iteration % 97 == 0:
            jitter = 1e-4 * np.random.randn(n)
            a_new = np.maximum(a_new + jitter, 0.0)
            a_new = a_new * (target_sum / a_new.sum())

        # --------------------------------------------------------------
        # 7. Evaluate new candidate (fast scorer)
        # --------------------------------------------------------------
        score_new = _fast_score(a_new, nfft_conv, target_sum)

        # --------------------------------------------------------------
        # 8. Keep the best solution seen so far
        # --------------------------------------------------------------
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            elapsed = time.time() - start_time
            print(
                f"Iter {iteration:6d}: new best score {best_score:.6f} "
                f"(elapsed {elapsed:.1f}s)"
            )
            # If we already beat the target we can stop early.
            if best_score <= 1.503:
                print("Target bound reached – stopping early.")
                break

        # --------------------------------------------------------------
        # 9. Prepare for next iteration
        # --------------------------------------------------------------
        a = a_new

        # slowly focus the soft‑max on the true maximum
        gamma = min(gamma * 1.005, gamma_max)

        # decay the learning‑rate gently
        lr = max(lr * 0.999, lr_min)

        # optional early exit if we are running out of time
        if time.time() > deadline:
            break

    print(f"Search finished. Best score obtained: {best_score:.6f}")
    # Return the best sequence as a plain Python list
    return best_a.tolist()
