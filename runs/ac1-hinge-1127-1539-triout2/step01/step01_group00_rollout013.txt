"""
Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
The method runs several independent projected multiplicative‑gradient
optimisations.  Instead of the raw ∞‑norm of the self‑convolution we use a
soft‑max (temperature β) to obtain a smooth approximation of the maximum.
β starts small (smooth) and is increased slowly, while the step size η is
decayed.  Random jitter is injected occasionally to escape plateaus.
The best sequence seen before the time budget expires is returned.

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution / correlation utilities (FFT‑based)
# ----------------------------------------------------------------------


def _conv_full(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a (full length 2 n‑1) using an FFT.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_max(a: np.ndarray) -> tuple[np.ndarray, float, int]:
    """
    Returns (full_convolution, max_value, argmax_index).
    """
    conv = _conv_full(a)
    max_val = float(conv.max())
    max_idx = int(np.argmax(conv))
    return conv, max_val, max_idx


def _valid_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Valid part of the cross‑correlation `correlate(w, a, mode='valid')`.
    Both vectors are real, the result has length len(a).
    """
    n = a.shape[0]
    conv_len = len(w) + n - 1          # = (2n‑1) + n‑1 = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fw = np.fft.rfft(w, n=fft_len)
    Fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    conv_full = np.fft.irfft(Fw * Fa_rev, n=fft_len)[:conv_len]
    start = n - 1
    return conv_full[start:start + n]


# ----------------------------------------------------------------------
# Gradient of the soft‑max approximation of the max‑convolution entry
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float) -> tuple[np.ndarray, float]:
    """
    Compute a sub‑gradient of the smoothed objective.
    Returns (g, current_max_convolution).
    """
    # full self‑convolution
    b = _conv_full(a)
    max_b = b.max()

    # soft‑max weights (temperature β)
    w = np.exp(beta * (b - max_b))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # C_i = Σ_j w_{i+j} a_j  (valid correlation)
    C = _valid_corr(w, a)

    # diagonal contribution (i = j ↔ k = 2i)
    idx = np.arange(a.shape[0], dtype=int)
    diag = w[2 * idx] * a

    # sub‑gradient (matches the original one for large β)
    g = 2.0 * C - diag
    return g, max_b

def _multiplicative_update(a: np.ndarray, g: np.ndarray, eta: float) -> np.ndarray:
    """
    One multiplicative step: a ← a·exp(−η·g)  followed by L1‑normalisation.

    total = a_new.sum()
    if total == 0.0:
        a_new = np.full(n, S / n)
    else:
        a_new *= (S / total)
    return a_new

    temperature β while decaying the step size η.  Random jitter is added
    sporadically.  The best sequence discovered before the time budget
    expires is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # small safety margin

    # Lengths to explore (including the known construction length)
    candidate_lengths = [2000, 4000, 5641, 8000]

    best_overall_seq = None
    best_overall_score = float('inf')
    run_idx = 0

    while time.time() < deadline and run_idx < len(candidate_lengths):
        n = candidate_lengths[run_idx]

        # allocate ≈ 30 % of the remaining time to the current run
        remaining = deadline - time.time()
        run_deadline = time.time() + remaining * 0.3

        # possibly initialise from the current best solution
        start_seq = None
        if best_overall_seq is not None and np.random.rand() < 0.35:
            cur = np.asarray(best_overall_seq, dtype=float)
            if cur.size > n:
                start_seq = cur[:n].copy()
            else:
                start_seq = np.pad(cur, (0, n - cur.size), constant_values=0.0)

        seq, score = _optimize_run(n, seed + run_idx * 137, run_deadline,
                                   start_seq=start_seq)

        if score < best_overall_score:
            best_overall_score = score
            best_overall_seq = seq.copy()
            print(f"[global improve] run={run_idx} n={n} score={best_overall_score:.6f}")

        run_idx += 1

    # --------------------------------------------------------------
    # Final refinement on the best sequence found so far (if time allows)
    # --------------------------------------------------------------
    if best_overall_seq is not None and time.time() < deadline - 5:
        a = np.array(best_overall_seq, copy=True)
        n = a.shape[0]
        eta = 0.3
        beta = 10.0
        for i in range(200):
            if time.time() >= deadline:
                break
            g, _ = _gradient_softmax(a, beta)
            a = _multiplicative_update(a, g, eta)
            # tiny decay of η to stabilise
            eta *= 0.99
            cur = evaluate_sequence(a.tolist())
            if cur < best_overall_score:
                best_overall_score = cur
                best_overall_seq = a.copy()
                print(f"[final refine] iter={i} score={best_overall_score:.6f}")

    # --------------------------------------------------------------
    # Fallback: use the supplied global construction if nothing else succeeded
    # --------------------------------------------------------------
    if best_overall_seq is None:
        if 'height_sequence_1' in globals():
            best_overall_seq = height_sequence_1
            best_overall_score = evaluate_sequence(best_overall_seq)
        else:
            # a trivial two‑point uniform sequence as a last resort
            best_overall_seq = [0.5, 0.5]
            best_overall_score = evaluate_sequence(best_overall_seq)

    print(f"[final] best_score={best_overall_score:.6f} length={len(best_overall_seq)}")
    return best_overall_seq.tolist()
