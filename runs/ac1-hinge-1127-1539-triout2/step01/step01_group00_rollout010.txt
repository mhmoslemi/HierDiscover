def _simplex_projection(v, S):
    # v is np.ndarray.
    n = len(v)
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - S
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] - 1 # zero-index?
    theta = cssv[cond][-1] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w
```

def _simplex_projection(v, S):
    # Ensure v is np.ndarray
    if S <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Optimise a non‑negative step‑function (a vector of heights) to minimise
the evaluation function defined in the competition.  The objective after
normalising the L¹‑norm to sqrt(2·n) is simply the maximum entry of the
self‑convolution a*a.  We employ a fast projected multiplicative gradient
descent with:

* a sub‑gradient that averages over *all* active (tight) convolution
  indices, not just a single arg‑max;
* exponential moving‑average momentum (β≈0.9) to stabilise the descent;
* a decaying global step‑size η;
* occasional “jitter’’ – a small random perturbation to escape plateaus;
* many short independent restarts (each run gets a time slice) so that
  several basins of attraction are explored.

All updates keep the vector non‑negative and re‑normalise it to the
required L¹‑norm, which is equivalent to staying inside the feasible set.
Only when the internal surrogate (the convolution maximum) improves do
we call the expensive ``evaluate_sequence`` function.  The best sequence
found before the budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _conv_max(a: np.ndarray) -> tuple[np.ndarray, float, int]:
    """
    Linear convolution a*a using FFT.
    Returns (full convolution array, maximal value, index of the maximum).
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    max_idx = int(np.argmax(conv))
    return conv, float(conv[max_idx]), max_idx


def _gradient(a: np.ndarray, active_idxs: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry.
    The gradient is averaged over all active indices (those attaining the
    maximal value up to a tiny tolerance).
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    # vector of indices 0..n-1 for broadcasting
    idx = np.arange(n)

    for k in active_idxs:
        partner = k - idx
        mask = (partner >= 0) & (partner < n)

        # off‑diagonal contributions: 2·a[partner]
        g[mask] += 2.0 * a[partner[mask]]

        # diagonal term (i=j) should contribute only a[i], not doubled
        if k % 2 == 0:
            d = k // 2
            if 0 <= d < n:
                g[d] -= a[d]      # cancel the extra contribution

    if active_idxs.size > 1:
        g /= float(active_idxs.size)

    return g


def _initial_sequence(n: int, seed: int) -> np.ndarray:
    """
    Produce a non‑negative start vector of length n whose entries sum to
    sqrt(2·n).  With 50 % probability the global ``height_sequence_1`` is
    used (if it exists and is long enough); otherwise a uniform Dirichlet
    draw is returned.
    """
    rng = np.random.default_rng(seed)
    S = np.sqrt(2.0 * n)

    use_existing = False
    if "height_sequence_1" in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=float)
            if base.size >= n:
                a = base[:n].copy()
            else:
                a = np.pad(base, (0, n - base.size), constant_values=0.0)
            use_existing = True
        except Exception:
            use_existing = False

    if use_existing and rng.random() < 0.5:
        a = a.copy()
    else:
        # Dirichlet(1,…,1) – i.e. uniform on the simplex
        a = rng.random(n)

    # Normalise to the required L¹‑norm
    a = a / a.sum() * S
    return a


# ----------------------------------------------------------------------
# Core optimisation routine (one independent run)
# ----------------------------------------------------------------------


def _optimize_once(n: int, seed: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform a single optimisation run on a vector of length ``n``.
    Returns the best sequence found and its external score.
    """
    rng = np.random.default_rng(seed)

    # required L¹‑norm
    S = np.sqrt(2.0 * n)

    # initialise
    a = _initial_sequence(n, seed)

    # external score of the start point
    best_seq = a.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    conv, cur_max, _ = _conv_max(a)
    best_internal = cur_max
    print(f"[run {seed}] init n={n} score={best_score:.6f}")

    # algorithmic hyper‑parameters
    eta = 0.45                # initial step‑size
    decay = 0.9997            # multiplicative decay per iteration
    beta = 0.9                # momentum coefficient
    jitter_prob = 0.015       # probability of a random perturbation
    jitter_scale = 0.015      # relative size of the jitter
    tol_rel = 1e-12           # tolerance for detecting active constraints

    momentum = np.zeros_like(a)

    while time.time() < deadline:
        # ----- compute surrogate gradient -----
        conv, max_val, _ = _conv_max(a)
        # active (tight) indices – a tiny absolute tolerance works well
        active = np.where(conv >= max_val - max(tol_rel, max_val * tol_rel))[0]

        g = _gradient(a, active)

        # ----- momentum and multiplicative update -----
        momentum = beta * momentum + (1.0 - beta) * g
        a = a * np.exp(-eta * momentum)

        # ----- re‑normalise to the L¹‑simplex (keep positivity) -----
        total = a.sum()
        if total <= 0.0:
            a = np.full(n, S / n)
        else:
            a *= (S / total)

        # ----- occasional jitter to avoid stagnation -----
        if rng.random() < jitter_prob:
            noise = rng.normal(scale=jitter_scale * a.mean(), size=n)
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()

        # ----- check for improvement -----
        if max_val < best_internal - 1e-12:
            # Only invoke the expensive external evaluator when the cheap
            # surrogate improves.
            ext_score = float(evaluate_sequence(a.tolist()))
            if ext_score < best_score:
                best_score = ext_score
                best_seq = a.copy()
                best_internal = max_val
                print(f"[improve] t={time.time():.1f}s  score={best_score:.6f}")

        # ----- schedule update -----
        eta *= decay

    return best_seq, best_score

# ----------------------------------------------------------------------
