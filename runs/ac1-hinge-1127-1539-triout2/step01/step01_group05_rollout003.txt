import numpy as np
import time
import math

def _next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()

def compute_score(a: np.ndarray, target_sum: float) -> float:
    # compute evaluation function
    n = a.size
    nfft = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[:2 * n - 1]
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    return float(2.0 * n * max_c / (s * s))

def max_conv(a: np.ndarray) -> float:
    n = a.shape[0]
    nfft = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[:2 * n - 1]
    return float(conv.max())

def softmax_weights(conv: np.ndarray, gamma: float) -> np.ndarray:
    max_c = conv.max()
    w = np.exp(gamma * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
        w_sum = 1.0
    return w / w_sum

def grad_softmax(a: np.ndarray, gamma: float, nfft_conv: int, nfft_corr: int) -> np.ndarray:
    # compute convolution a*a
    af = np.fft.rfft(a, n=nfft_conv)
    conv = np.fft.irfft(af * af, n=nfft_conv)[:2 * len(a) - 1]
    w = softmax_weights(conv, gamma)
    # correlation w * a_rev
    wf = np.fft.rfft(w, n=nfft_corr)
    af_rev = np.fft.rfft(a[::-1], n=nfft_corr)
    corr_full = np.fft.irfft(wf * af_rev, n=nfft_corr)
    n = len(a)
    grad = 2.0 * corr_full[n - 1 : n - 1 + n]
    return grad

def adam_optimize(initial_a: np.ndarray, target_sum: float,
                  time_deadline: float) -> tuple[np.ndarray, float]:
    # hyperparameters
    a = initial_a.copy()
    n = a.size
    nfft_conv = _next_pow2(2 * n - 1)
    nfft_corr = _next_pow2(3 * n - 2)

    gamma = 5.0
    gamma_max = 400.0
    lr = 0.12
    lr_min = 0.0005

    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = compute_score(best_a, target_sum)

    start_time = time.time()
    # keep iterating until deadline
    while time.time() < time_deadline:
        t += 1
        # compute gradient
        grad = grad_softmax(a, gamma, nfft_conv, nfft_corr)

        # Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0:
            a_new = np.full_like(a_new, target_sum / n)
        else:
            a_new *= target_sum / s

        # evaluate

        # simple acceptance: accept if better, else keep old

            best_score = score_new
            best_a = a_new.copy()
            elapsed = time.time() - start_time

        # always move to a_new (like gradient descent) if better; else sometimes accept?

            a = a_new

        # update hyperparameters

        lr = max(lr * 0.998, lr_min)

import math
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _eval_score(a: np.ndarray, target_sum: float) -> float:
    """
    Fast version of the evaluation function used for internal bookkeeping.
    It assumes the sum of ``a`` should be ``target_sum`` (the value used in the
    theoretical normalisation  √(2·n)).  The returned value equals the true
    ``evaluate_sequence`` score when the sum matches ``target_sum``.
    """
    n = a.shape[0]
    nfft = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))


def _softmax_weights(conv: np.ndarray, gamma: float) -> np.ndarray:
    """
    Compute the soft‑max probabilities over the convolution entries.
    The distribution is stable because the maximal entry is subtracted.
    """
    m = conv.max()
    w = np.exp(gamma * (conv - m))
    w_sum = w.sum()
    if w_sum == 0.0:                      # safety fallback
        w = np.full_like(w, 1.0 / w.size)
        w_sum = 1.0
    return w / w_sum


def _grad_softmax(a: np.ndarray, gamma: float,
                  nfft_conv: int, nfft_corr: int) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    Implements the formula  g = 2·(w ⊛ a_rev)  where w are the soft‑max
    weights and ⊛ denotes cross‑correlation.
    """
    n = a.shape[0]

    # --- convolution a*a -------------------------------------------------
    af = np.fft.rfft(a, n=nfft_conv)
    conv = np.fft.irfft(af * af, n=nfft_conv)[: 2 * n - 1]

    # --- soft‑max weights -------------------------------------------------
    w = _softmax_weights(conv, gamma)

    # --- cross‑correlation w ⊛ a_rev ------------------------------------
    wf = np.fft.rfft(w, n=nfft_corr)
    af_rev = np.fft.rfft(a[::-1], n=nfft_corr)
    corr_full = np.fft.irfft(wf * af_rev, n=nfft_corr)

    # valid part of the correlation has length n and starts at index n‑1
    grad = 2.0 * corr_full[n - 1 : n - 1 + n]
    return grad

# Phase 1 – Adam optimiser on the soft‑max surrogate
# ----------------------------------------------------------------------


def _adam_optimize(initial_a: np.ndarray, target_sum: float,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the soft‑max surrogate of the max‑convolution.
    The sum of the vector is kept equal to ``target_sum`` after every step.
    """
    a = initial_a.copy()
    n = a.size
    nfft_conv = _next_pow2(2 * n - 1)
    nfft_corr = _next_pow2(3 * n - 2)

    # Adam hyper‑parameters
    lr = 0.12          # initial learning rate
    lr_min = 5e-4
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8

    # soft‑max temperature schedule
    gamma = 5.0
    gamma_max = 400.0

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    step_counter = 0

    best_a = a.copy()
    best_score = _eval_score(best_a, target_sum)

    start = time.time()
    while time.time() < deadline:
        step_counter += 1

        # gradient of the soft‑max surrogate
        grad = _grad_softmax(a, gamma, nfft_conv, nfft_corr)

        # Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

        m_hat = m / (1 - beta1 ** step_counter)
        v_hat = v / (1 - beta2 ** step_counter)

        update = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - update
        a_new = np.maximum(a_new, 0.0)                     # non‑negative orthant

        # renormalise to the prescribed sum (scaling does not affect the score)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.full_like(a_new, target_sum / n)
        else:
            a_new *= target_sum / s

        # evaluate
        score_new = _eval_score(a_new, target_sum)

        # keep the best ever‑seen vector
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            elapsed = time.time() - start
            print(
                f"Adam iter {step_counter:6d}: new best {best_score:.6f} "
                f"(elapsed {elapsed:.1f}s)"
            )

        # simple acceptance rule: keep the new point if it improves,
        # otherwise stay where we are (the optimiser is robust enough).
        if score_new < _eval_score(a, target_sum):
            a = a_new

        # schedule updates
        gamma = min(gamma * 1.003, gamma_max)
        lr = max(lr * 0.998, lr_min)

    return best_a, best_score

def _simulated_annealing(start_a: np.ndarray, start_score: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Very lightweight SA that preserves the total sum exactly.
    At each step a tiny amount of mass is moved from one index to another.
    The temperature is linearly cooled.
    """
    a = start_a.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    n = a.size
    nfft = _next_pow2(2 * n - 1)

    # Pre‑compute a scale for the random moves
    scale = max(1.0, a.max())

    start = time.time()
    total_time = deadline - start
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        frac = elapsed / total_time
        # temperature goes from 0.02 → 0
        temperature = (1.0 - frac) * 0.02

        # pick two distinct indices
        i = np.random.randint(0, n)
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1

        # propose moving a small amount of mass
        delta = temperature * np.random.randn() * scale
        if delta > 0.0:
            move = min(delta, a[i])
            a_i = a[i] - move
            a_j = a[j] + move
        else:
            move = min(-delta, a[j])
            a_i = a[i] + move
            a_j = a[j] - move

        a_new = a.copy()
        a_new[i] = a_i
        a_new[j] = a_j

        # evaluate the candidate (full convolution)
        af = np.fft.rfft(a_new, n=nfft)
        conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
        new_score = float(2.0 * n * conv.max() / (a_new.sum() ** 2))

        # Metropolis acceptance
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / (temperature + 1e-12))
            accept = np.random.rand() < prob

        if accept:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                print(
                    f"SA iter {it:6d}: new best {best_score:.6f} "
                    f"(elapsed {elapsed:.1f}s)"
                )

    return best_a, best_score

    """

    ``2·n·max(conv(a,a))/(sum(a)**2)``.
    The algorithm consists of two stages:

    1.  Adam optimisation on a smooth (soft‑max) approximation of the

        the total sum.  The temperature is linearly cooled.

    entry exceed the limits imposed by the evaluation wrapper.
    The best vector seen during the whole run is returned as a plain
    Python list.
    """

    start_time = time.time()
    # leave a tiny safety margin for the surrounding harness
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Determine the length of the construction and a sensible initial guess.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
        have_global = True
    except Exception:
        n = 5000
        have_global = False

    target_sum = math.sqrt(2.0 * n)

    # analytic shape a_i = 1 / sqrt(i+0.5)
    a_shape = 1.0 / np.sqrt(np.arange(n, dtype=np.float64) + 0.5)
    a_shape *= target_sum / a_shape.sum()

    if have_global:

        # make sure it respects the constraints
        a_global = np.maximum(a_global, 0.0)

        start_a = a_global if _eval_score(a_global, target_sum) < _eval_score(
            a_shape, target_sum
        ) else a_shape
    else:
        start_a = a_shape

    # Phase 1 – Adam optimisation (≈ 65 % of the budget)
