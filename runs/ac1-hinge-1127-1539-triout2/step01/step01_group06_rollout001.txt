"""
Step‑function minimiser using an iterative linear‑programming update.

The evaluation metric simplifies (after the required ℓ¹‑normalisation)
to the maximal entry of the auto‑convolution a∗a.  We start from a
power‑law profile a_i ∝ (i+1)^‑p (p≈0.5) and then repeatedly solve a
small linear programme:

    maximise   Σ b_i
    subject to b_i ≥ 0 ,   (a ∗ b)_k ≤ M  for all k,
    where M = max(a ∗ a).

The optimum b can carry more ℓ¹‑mass than the current a while keeping the
cross‑convolution below the current bound M.  After scaling b to the
required norm we move a a little towards the new shape (blend factor t≈0.2).
Each successful step strictly reduces the maximal convolution, and the
process is repeated until the time budget expires.  The result is then
post‑processed by a cheap local hill‑climber to squeeze out any remaining
improvement.

The whole routine respects the 1000 s budget, works for any length up to a
few thousand and falls back to the original stochastic optimiser if
CVXPY is unavailable.
"""

import time
import numpy as np

# Try to import CVXPY – if it is not available we fall back to the
# pure‑stochastic optimiser defined later.
try:
    import cvxpy as cp
    _HAS_CVXPY = True
except Exception:  # pragma: no cover
    _HAS_CVXPY = False

# ----------------------------------------------------------------------
# Core numerical helpers (no closures, no lambdas)
# ----------------------------------------------------------------------


def max_convolution(a: np.ndarray) -> float:
    """Maximum value of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def score(a: np.ndarray) -> float:
    """
    Evaluation function used by the competition (equal to the maximal
    convolution after ℓ¹‑normalisation to sqrt(2·n)).
    """
    n = a.size
    max_b = max_convolution(a)
    s = a.sum()
    if s < 0.01:            # safety – should never happen
        return np.inf
    return 2.0 * n * max_b / (s * s)


def initial_sequence(p: float, n: int) -> np.ndarray:
    """Power‑law start a_i ∝ (i+1)⁻ᵖ normalised to ℓ¹‑norm √(2 n)."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= np.sqrt(2.0 * n) / a.sum()
    return a


# ----------------------------------------------------------------------
# Linear‑programming step (requires CVXPY)
# ----------------------------------------------------------------------


def lp_maximise_mass(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve
        maximise Σ b_i
        subject to b_i ≥ 0 and conv(f, b) ≤ M (element‑wise)
    and return the optimal b (or None on failure).
    """
    if not _HAS_CVXPY:                     # pragma: no cover
        return None

    n = f.shape[0]

    # Parameter holding the current f (constant during the solve)
    f_param = cp.Parameter(n, nonneg=True, value=f)

    # Decision variable
    b = cp.Variable(n, nonneg=True)

    # Linear convolution f ∗ b (length 2n‑1)
    conv_fb = cp.conv(f_param, b)

    # Constraints: every entry of the cross‑convolution must be ≤ M
    constraints = [conv_fb <= M]

    # Objective: maximise total ℓ¹‑mass of b
    problem = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    # ECOS is fast for dense problems of this size
    try:
        problem.solve(solver=cp.ECOS, warm_start=True,
                      abstol=1e-8, reltol=1e-8, feastol=1e-8,
                      max_iters=2000)
    except Exception:                     # pragma: no cover
        return None

    if b.value is None:
        return None

    # Return a plain NumPy array (copy to detach from CVXPY internal storage)
    return np.array(b.value, dtype=np.float64).flatten()


def lp_iterative_improve(a: np.ndarray,
                         S: float,
                         deadline: float,
                         blend_factor: float = 0.2,
                         max_iters: int = 200) -> tuple[np.ndarray, float]:
    """
    Repeatedly apply the LP‑based update to a, stopping when the deadline
    is reached or when no further progress is observed.
    """
    best_a = a.copy()
    best_val = score(best_a)

    # Simple safeguard – we never let the mass drop below the required norm.
    cur_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        M = max_convolution(cur_a)

        b = lp_maximise_mass(cur_a, M)
        if b is None:
            break

        sum_b = b.sum()
        if sum_b <= 0.0:
            break

        # Scale b back to the required ℓ¹‑norm S
        scale = S / sum_b
        g = b * scale

        # Two candidate updates: a full replacement and a modest blend.
        cand_full = g
        cand_blend = (1.0 - blend_factor) * cur_a + blend_factor * g

        # Evaluate both candidates and keep the best one.
        for cand in (cand_full, cand_blend):
            val = score(cand)
            if val < best_val - 1e-12:
                best_val = val
                best_a = cand.copy()
                cur_a = cand.copy()
                print(f"[{time.time():.1f}]  LP iter {it:3d} improved score = {best_val:.6f}")
                break
        else:
            # No improvement – still move the current iterate a little towards g
            # to avoid stalling, but keep a feasible point.
            cur_a = cand_blend

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climber (fallback / final polishing)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap random mass‑transfer hill‑climber that preserves the ℓ¹‑norm.
    Only strictly improving moves are kept.
    """
    n = a.size
    best_a = a.copy()
    best_val = score(best_a)

    cur_val = best_val
    max_frac = 0.12
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        frac = rng.random() * max_frac
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        new_val = score(a)

        if new_val < cur_val - 1e-12:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}]  Hill‑climb improved score = {best_val:.6f}")
        else:
            a[j] -= delta
            a[i] += delta

    return best_a, best_val

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises `evaluate_sequence`.
    The routine mixes a linear‑programming based refinement with a fast
    stochastic hill‑climber and respects the global time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # tiny safety margin

    best_seq = None
    best_val = np.inf

    # Try a few different lengths – larger lengths give more freedom but
    # make each LP slower.  The chosen set stays comfortably below 4000.
    lengths = [800, 1200, 1800, 2500, 3200]
    # Power‑law exponents to explore around the known optimum p≈0.5.
    exponents = [0.42, 0.45, 0.48, 0.50, 0.52, 0.55, 0.58]

    for n in lengths:
        if time.time() >= deadline:
            break
        S = np.sqrt(2.0 * n)          # required ℓ¹‑norm

        for p in exponents:
            if time.time() >= deadline:
                break

            # ---- 1. initialise -------------------------------------------------
            a = initial_sequence(p, n)
            cur_val = score(a)

            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[{time.time():.1f}] start (n={n}, p={p:.2f}) score = {cur_val:.6f}")

            # ---- 2. LP‑based iterative improvement -------------------------------
            # Allocate a modest slice of the remaining budget for the LP loop.
            sub_deadline = min(deadline, time.time() + 12.0)
            a_lp, val_lp = lp_iterative_improve(a, S, sub_deadline,
                                                blend_factor=0.2,
                                                max_iters=80)

            if val_lp < best_val:
                best_val = val_lp
                best_seq = a_lp.copy()
                print(f"[{time.time():.1f}] after LP (n={n}, p={p:.2f}) best = {val_lp:.6f}")

            # ---- 3. Final polishing with cheap hill‑climber --------------------
            # Use any leftover time for a short stochastic refinement.
            remaining = deadline - time.time()
            if remaining > 1.0:
                a_hc, val_hc = hill_climb(a_lp.copy(), time.time() + min(7.0, remaining))
                if val_hc < best_val:
                    best_val = val_hc
                    best_seq = a_hc.copy()
                    print(f"[{time.time():.1f}] after hill‑climb (n={n}, p={p:.2f}) best = {val_hc:.6f}")

    # ----------------------------------------------------------------------
    # Safety net – should never happen
    # ----------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.ones(1000, dtype=np.float64)
        best_seq *= np.sqrt(2.0 * best_seq.size) / best_seq.sum()

    # Trim trailing near‑zeros (helps a bit with the evaluation routine)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Return a plain Python list as required by the outer harness
    return best_seq.tolist()
