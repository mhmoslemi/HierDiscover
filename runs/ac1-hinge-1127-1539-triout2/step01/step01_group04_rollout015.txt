def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0

    # Determine length
    n = 1024  # power of two for fast FFT
    target_sum = math.sqrt(2.0*n)

    # Initialize a with sqrt-difference
    idx = np.arange(n, dtype=np.float64)
    a = np.sqrt(idx + 1) - np.sqrt(idx)
    a *= target_sum / a.sum()

    best_a = a.copy()
    conv = np.convolve(a, a)
    best_score = conv.max()
    print(f"Initial score {best_score}")

    # Precompute FFT length
    L = 1 << ((2*n - 1).bit_length())

    iter_num = 0
    t = 0.2   # mixing factor
    while time.time() < deadline:
        # Compute current max
        conv = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(a, L), n=L)[:2*n-1]
        M = conv.max()
        # Record best
        if M < best_score:
            best_score = M
            best_a = a.copy()
            print(f"Iter {iter_num} new best {best_score:.6f}")
        # Build LP constraints
        # Build A_ub matrix as dense np array
        rows = 2*n-1
        A = np.zeros((rows, n), dtype=np.float64)
        for k in range(rows):
            # j valid range: max(0, k-(n-1)) to min(k, n-1)
            j_start = max(0, k-(n-1))
            j_end = min(k, n-1)
            # f indexes are i = k - j
            # for j in [j_start, j_end], i = k - j in [0, n-1]
            # fill A[k, j] = f[i]
            # vectorized:
            idx_j = np.arange(j_start, j_end+1)
            A[k, idx_j] = a[k - idx_j]
        # Solve LP maximise sum(b) -> minimize -sum(b)
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(rows, M, dtype=np.float64)
        bounds = [(0, None) for _ in range(n)]
        # Use high-performance method
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            print(f"LP failed at iter {iter_num}: {res.message}")
            break
        b = res.x
        sum_b = b.sum()
        # Rescale to target sum
        if sum_b == 0:
            break
        g = b * (target_sum / sum_b)
        # convex combination
        a = (1.0 - t) * a + t * g
        # Ensure nonnegativity (should be) and normalise sum exactly (might have rounding)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full(n, target_sum / n)
        else:
            a *= target_sum / a_sum
        iter_num += 1

        # Optionally adjust mixing factor if improvement slows
        if iter_num % 10 == 0:
            # Could reduce t
            t = max(0.05, t * 0.95)

    # Return best sequence
    print(f"Finished after {iter_num} iterations, best score {best_score:.6f}")
    return best_a.tolist()

"""Propose a step‐function sequence with a low self‑convolution bound.

The algorithm alternates two operations:

1. **Current worst case** – compute M = max_k (a*a)_k for the
   current sequence a (the evaluation value after normalisation).

2. **Linear‑programming update** – maximise the ℓ¹‑norm of a new
   non‑negative vector b under the linear constraints
   (a * b)_k ≤ M for every shift k.  The constraints are linear because
   the convolution with a fixed a is a linear map in b.
   The optimiser returns b with the largest possible total mass while
   keeping the mixed convolution below M.  After scaling b to the
   required ℓ¹‑norm (sqrt(2·n)) we replace a by a convex combination
   (1‑t)·a + t·b.  This never raises the worst‑case convolution, but
   usually lowers it.

Repeating the LP step (with a slowly decreasing mixing weight *t*)
drives the maximal self‑convolution towards the theoretical optimum.
The routine stops when the time budget is exhausted and returns the
best sequence ever observed.

The implementation uses NumPy for fast FFT‑based convolutions and SciPy
‘highs’ linear programming solver.  All arithmetic stays in double
precision and the sequence is projected onto the feasible set
(non‑negative, fixed ℓ¹‑norm) after every update.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog


def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Self‑convolution of a real vector using an FFT length L."""
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * a.size - 1]
    return conv

    ``evaluate_sequence`` metric.

    Parameters

        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.

    Returns

    list[float]
        The best sequence found.
    """

    deadline = time.time() + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------

    # leads to a few dozen MB of constraint matrix and a solve time of
    # a few seconds on a single core).  If a previous construction is
    # short enough we reuse its length, otherwise we fall back to 2048.
    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    a = np.sqrt(idx + 1.0) - np.sqrt(idx)          # a_i = √(i+1) – √i
    a *= target_sum / a.sum()                      # normalise to √(2n)

            print(f"Iter {iteration:4d} – new best M = {best_M:.6f}")
