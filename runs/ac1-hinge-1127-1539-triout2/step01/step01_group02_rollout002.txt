def project_simplex(v):
    """Projection of vector v onto probability simplex (non-negative, sum=1)."""
    # using algorithm from Duchi et al.
    v = np.asarray(v, dtype=np.float64)
    if v.sum() == 1 and np.all(v >= 0):
        return v
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Non‑negative step‑function optimizer.

The goal is to minimise

    f(a) = 2 * n * max(conv(a, a)) / (sum(a)**2)

which is scale‑invariant.  After normalising the candidate to a
probability vector (sum = 1) the score simplifies to
    score = 2 * n * max(conv(p, p))

where `conv` is the full (linear) convolution.
The algorithm proceeds in two phases:

1.  A large pool of seed vectors is generated (power‑law, exponential,
    uniform, random Dirichlet and the previous best `height_sequence_1`).
2.  Each seed is refined by a mirror‑descent / exponentiated‑gradient
    scheme that repeatedly reduces the largest convolution coefficient.
    The sub‑gradient of the ℓ∞‑norm of the convolution at the currently
    highest positions is used; a multiplicative update keeps the vector
    non‑negative and a normalisation step projects it onto the simplex.

Fast FFT‑based convolution is used, and the score is computed directly
instead of calling `evaluate_sequence` in the inner loop (the external
function is only used for the final check).  The routine respects a
1000 s time budget and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _generate_powerlaw(p: float, sigma: float, max_len: int = 4000,
                       eps: float = 1e-9) -> np.ndarray:
    """
    Return a truncated power‑law sequence a_i = (i + sigma) ** (-p).
    The sequence is cut when values fall below `eps` or when `max_len`
    entries have been generated.
    """
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p)
    mask = a > eps
    return a[mask]


def _full_conv_fft(p: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a real vector with itself using FFT.
    Returns a real array of length 2*len(p)-1.
    """
    n = p.shape[0]
    # next power of two for efficient FFT
    L = 1 << ((2 * n - 2).bit_length())
    ft = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(ft * ft, n=L)[: 2 * n - 1]
    conv = np.real(conv)
    # numerical noise may produce tiny negatives – clip them
    conv[conv < 0] = 0.0
    return conv


def _subgradient(p: np.ndarray, conv: np.ndarray, top_k: int = 5) -> np.ndarray:
    """
    Sub‑gradient of max(conv(p,p)) w.r.t. a probability vector `p`.
    The gradient is summed over the `top_k` largest convolution entries.
    """
    n = p.shape[0]
    g = np.zeros_like(p)

    if top_k >= conv.size:
        ks = np.arange(conv.size)
    else:
        ks = np.argpartition(conv, -top_k)[-top_k:]

    for k in ks:
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        idx = np.arange(i_min, i_max + 1, dtype=int)
        # contribution p_{k-i}
        g[idx] += p[k - idx]

    g *= 2.0          # each pair appears twice in the derivative
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]

    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find rho
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The routine respects the 1000 s time limit and
    returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # 1) Gather a diverse set of starting points
    # ------------------------------------------------------------------
    seeds = []

    # previous best (if supplied by the environment)
    if "height_sequence_1" in globals():
        seeds.append(np.asarray(height_sequence_1, dtype=np.float64))

    # power‑law families
    for p_exp in np.linspace(0.40, 0.60, 9):
        for sigma in np.logspace(-3.0, 0.5, 15):
            seq = _generate_powerlaw(p_exp, sigma, max_len=4000, eps=1e-9)
            if seq.size >= 200:            # ignore trivially short vectors
                seeds.append(seq)

    # exponential families
    for lam in np.linspace(0.0005, 0.005, 10):
        i = np.arange(4000, dtype=np.float64)
        seq = np.exp(-lam * i)
        seq = seq[seq > 1e-9]
        if seq.size >= 200:
            seeds.append(seq)

    # uniform vectors of various lengths
    for n in (500, 1000, 2000, 3000, 4000):
        seeds.append(np.ones(n, dtype=np.float64))

    # random Dirichlet‑like seeds
    for n in (500, 1000, 2000, 3000, 4000):
        rnd = np.random.rand(n)
        seeds.append(rnd)

    # ------------------------------------------------------------------
    # 2) Local refinement for each seed
    # ------------------------------------------------------------------
    for base in seeds:
        if time.time() >= deadline:
            break

        p = np.asarray(base, dtype=np.float64)
        if p.sum() == 0.0:
            continue
        # Normalise to the simplex
        p = p / p.sum()
        n = p.shape[0]

        # limit the length to keep the per‑iteration cost sane
        if n > 4000:
            p = p[:4000]
            n = p.shape[0]
            p = p / p.sum()

        # Evaluate the initial candidate (score = 2*n*max(conv))
        conv = _full_conv_fft(p)
        cur_score = 2.0 * n * np.max(conv)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = p.tolist()
            print(f"Initial seed | n={n} | score={best_score:.6f}")

        # Parameters for the mirror‑descent loop
        max_iters = 5000                     # per seed – may be cut by time
        eta0 = 0.30
        top_k = 5

        it = 0
        while time.time() < deadline and it < max_iters:
            it += 1

            # 1) compute convolution and current score
            conv = _full_conv_fft(p)
            max_val = np.max(conv)
            cur_score = 2.0 * n * max_val

            # 2) record improvement
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = p.tolist()
                print(f"Iter {it:5d} | n={n} | new best {best_score:.6f}")

                if best_score < 1.503:
                    print("Target achieved – terminating early.")
                    return best_seq

            # 3) sub‑gradient for the active (largest) convolution entries
            g = _subgradient(p, conv, top_k=top_k)

            # 4) multiplicative (exponentiated‑gradient) update
            eta = eta0 / np.sqrt(it)
            p = p * np.exp(-eta * g)

            # 5) re‑normalise onto the simplex
            sum_p = p.sum()
            if sum_p == 0.0 or not np.isfinite(sum_p):
                # Re‑initialise if we collapsed to the zero vector
                p = np.abs(np.random.randn(n))
                sum_p = p.sum()
            p = p / sum_p

            # 6) occasional random jitter to escape plateaus
            if np.random.rand() < 0.02:
                idx = np.random.randint(n)
                delta = (np.random.rand() - 0.5) * 0.01 * p[idx]
                p[idx] = max(p[idx] + delta, 0.0)
                p = p / p.sum()

        # end of refinement for this seed

    # ------------------------------------------------------------------
    # 3) Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    print("Final best score:", best_score, "length:", len(best_seq) if best_seq else None)
    return best_seq if best_seq is not None else []  # empty list signals failure
