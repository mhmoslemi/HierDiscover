def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    best_seq = None; best_score = np.inf
    # potential lengths list
    lengths = [500, 1000, 2000, 4000]
    # exponents random on each restart
    while time.time() < deadline:
        # pick length random
        n = np.random.choice(lengths)
        # generate start sequence
        if best_seq is not None and np.random.rand() < 0.3:
            # use best truncated/padded to n
            seq = np.array(best_seq[:n], dtype=np.float64)
            if seq.size < n:
                # pad with small values to maintain sum
                extra = n - seq.size
                # add small uniform to each entry
                seq = np.concatenate([seq, np.full(extra, 1e-12)])
            seq = seq * (np.sqrt(2*n) / seq.sum())
        else:
            p = np.random.uniform(0.4, 0.6)
            seq = _initial_sequence(p, n)
        # run annealing
        seq_opt, score_opt = _anneal(seq, deadline)
        if score_opt < best_score:
            best_score = score_opt; best_seq = seq_opt
            print(f"New best score {best_score:.6f} for n={n}")
    # after loop, final cleanup (cut zeros)
    if best_seq is None:
        # fallback uniform
        n=1000; best_seq = np.ones(n); best_seq = best_seq * (np.sqrt(2*n)/best_seq.sum())
    # trim trailing zeros
    nz = np.where(best_seq>1e-12)[0]
    if nz.size:
        best_seq = best_seq[:nz[-1]+1]
    return best_seq.tolist()

def _anneal(a, deadline):
    n = a.size
    cur = a.copy()
    cur_val = _score(cur)
    best = cur.copy()
    best_val = cur_val
    rng = np.random.default_rng()
    # temperature init
    T = 0.001 * cur_val  # scaled
    # step fraction parameters
    max_frac = 0.3
    # cooling per iteration
    cooling = 0.9995

    # pre-calc conv full to get kmax maybe.
    while time.time() < deadline:
        # compute conv to get max index:
        conv = _conv_full(cur)
        k = int(conv.argmax())
        # decide move type:
        if rng.random() < 0.7:
            # random mass transfer
            i = rng.integers(0,n)
            if cur[i] <= 1e-15:
                continue
            j = rng.integers(0,n)
            if i == j: continue
            frac = rng.random()*max_frac
            delta = cur[i]*frac
            cur[i] -= delta
            cur[j] += delta
            # compute new score
            new_val = _score(cur)
            # decide acceptance
            if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val)/max(T, 1e-12)):
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = cur.copy()
                # keep changes
            else:
                cur[j] -= delta
                cur[i] += delta
        else:
            # guided move using gradient at max index
            # compute contributions
            # contributions for i where partner exists.
            i_vals = np.arange(n)
            partner = k - i_vals
            mask = (partner >=0) & (partner < n)
            contrib = np.zeros(n, dtype=np.float64)
            partner_idx = partner[mask]
            contrib[mask] = cur[i_vals[mask]] * cur[partner_idx]
            # select i weighted by contrib+small epsilon
            total_contrib = contrib.sum()
            if total_contrib <= 0:
                # fallback random move
                continue
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
            # destination j not equal to i or partner
            j = rng.integers(0,n)
            while j == i or (k - i == j):
                j = rng.integers(0,n)
            # move delta fraction from i
            if cur[i] <= 1e-15: continue
            frac = rng.random()*max_frac
            delta = cur[i]*frac
            cur[i] -= delta
            cur[j] += delta
            new_val = _score(cur)
            if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val)/max(T, 1e-12)):
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = cur.copy()
            else:
                cur[j] -= delta
                cur[i] += delta
        T *= cooling
        if T < 1e-9:
            T = 1e-9
    return best, best_val

def _conv_full(a: np.ndarray) -> np.ndarray:
    n = a.size
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa*fa, n=size)[:2*n-1]
    return conv

def _anneal(a, deadline):
    n = a.shape[0]
    cur = a.copy()
    cur_val = _score(cur)
    best = cur.copy()
    best_val = cur_val
    max_frac = 0.2
    rng = np.random.default_rng()
    T = 0.001 * cur_val  # scale
    cooling = 0.9995
    while time.time() < deadline:
        # Determine move type: 70% random, 30% gradient
        if rng.random() < 0.7:
            # random mass transfer
            i = rng.integers(0, n)
            if cur[i] <= 1e-15:
                continue
            j = rng.integers(0, n)
            if i == j:
                continue
            frac = rng.random() * max_frac
            delta = cur[i] * frac
            cur[i] -= delta
            cur[j] += delta
            new_val = _score(cur)
            if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12)):
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = cur.copy()
                # accept
            else:
                cur[j] -= delta
                cur[i] += delta
        else:
            # gradient move
            conv = _conv_full(cur)
            k = int(conv.argmax())
            partner = k - np.arange(n)
            mask = (partner >= 0) & (partner < n)
            contrib = np.zeros(n, dtype=np.float64)
            contrib[mask] = cur[np.arange(n)[mask]] * cur[partner[mask]]
            total = contrib.sum()
            if total <= 0.0:
                continue
            # weighted choice
            cum = np.cumsum(contrib)
            r = rng.random() * total
            i = int(np.searchsorted(cum, r))
            # pick destination j
            j = rng.integers(0, n)
            attempts = 0
            while (j == i) or (k - i == j) or (j < 0) or (j >= n):
                j = rng.integers(0, n)
                attempts += 1
                if attempts > 5:
                    break
            if i == j:
                continue
            if cur[i] <= 1e-15:
                continue
            frac = rng.random() * max_frac
            delta = cur[i] * frac
            cur[i] -= delta
            cur[j] += delta
            new_val = _score(cur)
            if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12)):
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = cur.copy()
            else:
                cur[j] -= delta
                cur[i] += delta
        T *= cooling
        if T < 1e-9:
            T = 1e-9
    return best, best_val
```

def _lp_improve(f, t_mix, max_iters, deadline):
    n = f.shape[0]
    cur = f.copy()
    cur_val = _score(cur)
    best = cur.copy()
    best_val = cur_val
    rng = np.random.default_rng()
    for _ in range(max_iters):
        if time.time() >= deadline:
            break
        M = _max_convolution(cur)
        # Build constraint matrix
        A = _conv_matrix(cur)
        b = np.full(A.shape[0], M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)  # maximize sum(g)
        # Solve LP
        try:
            res = scipy.optimize.linprog(c, A_ub=A, b_ub=b,
                                         bounds=[(0, None)] * n,
                                         method='highs')
        except Exception:
            break
        if not res.success:
            break
        g = np.maximum(res.x, 0.0)
        sum_g = g.sum()
        if sum_g <= 0.0:
            break
        # scale to have correct L1 norm
        scale = np.sqrt(2.0 * n) / sum_g
        g *= scale
        # mix
        cur = (1.0 - t_mix) * cur + t_mix * g
        # enforce nonnegativity, renormalize to correct sum (numeric errors)
        cur = np.maximum(cur, 0.0)
        cur_sum = cur.sum()
        if cur_sum > 0:
            cur *= np.sqrt(2.0 * n) / cur_sum
        new_val = _score(cur)
        # simple acceptance (always accept as we aim to reduce M)
        cur_val = new_val
        if new_val < best_val:
            best_val = new_val
            best = cur.copy()
        # optionally narrow mixing factor gradually? Not needed.
    return best, best_val
```

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """Search for a step‑function sequence that minimises evaluate_sequence().
    Uses a combination of power‑law initialisation, simulated annealing,
    and a linear‑programming refinement that redistributes mass to the tail.
    The scheduler respects the given time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0

    best_seq = None
    best_score = np.inf

    # Prepare possible lengths
    lengths = [500, 800, 1200, 1800, 2500]
    exponents = [0.45, 0.48, 0.5, 0.52, 0.55]
    offsets = [1.0, 2.0, 3.0, 5.0]

    # If a previously discovered solution is available, keep a copy.
    try:
        global_seq = np.array(height_sequence_1, dtype=np.float64)
    except Exception:
        global_seq = None

    # Outer loops
    for n in lengths:
        for p in exponents:
            for offset in offsets:
                if time.time() >= deadline:
                    break
                # initialise
                a = _initial_sequence_params(p, offset, n)
                cur_score = _score(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"Init (n={n}, p={p:.3f}, off={offset}) score={cur_score:.6f}")

                # local hill‑climbing (simulated annealing) for a few seconds
                sub_deadline = min(deadline, time.time() + 3.0)
                a_opt, opt_score = _anneal(a, sub_deadline)
                if opt_score < best_score:
                    best_score = opt_score
                    best_seq = a_opt.copy()
                    print(f"After anneal (n={n}) score={opt_score:.6f}")

                # LP refinement if SciPy is available
                if _has_scipy:
                    # restrict total time per LP block
                    sub_deadline_lp = min(deadline, time.time() + 4.0)
                    a_lp, lp_score = _lp_improve(a_opt, t_mix=0.5, max_iters=3, deadline=sub_deadline_lp)
                    if lp_score < best_score:
                        best_score = lp_score
                        best_seq = a_lp.copy()
                        print(f"After LP (n={n}) score={lp_score:.6f}")

                    # final short anneal after LP
                    sub_deadline2 = min(deadline, time.time() + 2.0)
                    a_final, final_score = _anneal(a_lp, sub_deadline2)
                    if final_score < best_score:
                        best_score = final_score
                        best_seq = a_final.copy()
                        print(f"After final anneal (n={n}) score={final_score:.6f}")

        if time.time() >= deadline:
            break

    # If we still have time, run a final optimisation on the best sequence found.
    if time.time() < deadline and best_seq is not None:

"""
Step‑function optimizer.

The routine searches a vector a ≥ 0 with fixed ℓ1‑norm √(2·n) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2) .

It combines three ideas:

1.  Power‑law initialisation a_i ∝ (i+offset)^{-p}.
2.  A simulated‑annealing / hill‑climbing refinement that moves
    a small amount of mass between positions.
3.  An optional linear‑programming refinement (if SciPy is available) that
    solves

        maximise   Σ g_j
        subject to  (a * g)_k ≤ M   for every convolution index k,
                    g_j ≥ 0,

    where M = max(conv(a,a)).  The optimum g concentrates mass where a is
    small, then the current vector is mixed with the scaled g.  A final
    short anneal is applied.

All work respects a global time budget (default 1000 s) and prints progress.
The best sequence found is returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports – the LP part is only used if they are present.
# ----------------------------------------------------------------------
try:
    import scipy.linalg
    import scipy.optimize
    _HAS_SCIPY = True
except Exception:  # pragma: no cover
    _HAS_SCIPY = False

# ----------------------------------------------------------------------
# Core helper functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def _max_convolution(a: np.ndarray) -> float:
    """Maximum value of the linear convolution a*a (real FFT, fast)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (real FFT, fast)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Evaluation function used by the competition (identical to
    evaluate_sequence for a valid non‑negative array)."""
    n = a.size
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _initial_sequence(p: float, offset: float, n: int) -> np.ndarray:
    """
    Power‑law starter a_i ∝ (i+offset)^{-p} renormalised to ℓ1‑norm √(2·n).
    """
    idx = np.arange(n, dtype=np.float64) + offset
    a = idx ** (-p)
    a_sum = a.sum()
    a *= np.sqrt(2.0 * n) / a_sum
    return a


def _anneal(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climber.
    Random mass‑transfer moves (accept if better or with probability
    exp(-(Δ)/T)).  Occasionally a “gradient” move is performed that
    targets the index where the convolution attains its maximum.
    """
    n = a.size
    cur = a.copy()
    cur_val = _score(cur)
    best = cur.copy()
    best_val = cur_val
    max_frac = 0.20          # at most 20 % of source entry is moved
    rng = np.random.default_rng()
    T = 0.001 * cur_val      # initial temperature
    cooling = 0.9995

    while time.time() < deadline:
        if rng.random() < 0.7:
            # ----- pure random transfer -------------------------------------------------
            i = rng.integers(0, n)
            if cur[i] <= 1e-15:
                continue
            j = rng.integers(0, n)
            if i == j:
                continue
            frac = rng.random() * max_frac
            delta = cur[i] * frac
            cur[i] -= delta
            cur[j] += delta
            new_val = _score(cur)

            if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / max(T, 1e-12)):
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = cur.copy()
            else:
                cur[j] -= delta
                cur[i] += delta
        else:
            # ----- gradient‑guided move --------------------------------------------------
            conv = _conv_full(cur)
            k = int(conv.argmax())                     # index of current maximum
            partner = k - np.arange(n)
            mask = (partner >= 0) & (partner < n)
            contrib = np.zeros(n, dtype=np.float64)
            contrib[mask] = cur[np.arange(n)[mask]] * cur[partner[mask]]
            total = contrib.sum()
            if total <= 0.0:
                continue
            cum = np.cumsum(contrib)
            r = rng.random() * total
            i = int(np.searchsorted(cum, r, side="right"))
            # pick a destination that is not i and not the partner of i (avoid cancelling the max)
            j = rng.integers(0, n)
            attempts = 0
            while (j == i) or (k - i == j):
                j = rng.integers(0, n)
                attempts += 1
                if attempts > 5:
                    break
            if i == j or cur[i] <= 1e-15:
                continue
            frac = rng.random() * max_frac
            delta = cur[i] * frac
            cur[i] -= delta
            cur[j] += delta
            new_val = _score(cur)

            if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / max(T, 1e-12)):
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = cur.copy()
            else:
                cur[j] -= delta
                cur[i] += delta

        # temperature schedule
        T *= cooling
        if T < 1e-9:
            T = 1e-9

    return best, best_val


def _conv_constraint_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2·n‑1) × n matrix that maps a vector g to the convolution
    (f * g).  Entry (k, j) = f_{k‑j} when 0 ≤ k‑j < n, otherwise 0.
    """
    n = f.shape[0]
    first_col = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    first_row = np.concatenate([f[:1], np.zeros(n - 1, dtype=f.dtype)])
    return scipy.linalg.toeplitz(first_col, first_row)


def _lp_improve(f: np.ndarray,
                t_mix: float = 0.5,
                max_iters: int = 3,
                deadline: float = 0.0) -> tuple[np.ndarray, float]:
    """
    Linear‑programming refinement.
    For the current vector f, solve:
        maximise Σ g_j
        subject to (f * g)_k ≤ M   for every k,
                    g_j ≥ 0,
    where M = max(conv(f,f)).
    The optimum g is scaled to the required ℓ1‑norm and mixed with f.
    """
    n = f.shape[0]
    cur = f.copy()
    cur_val = _score(cur)
    best = cur.copy()
    best_val = cur_val

    for _ in range(max_iters):
        if time.time() >= deadline:
            break
        M = _max_convolution(cur)
        A = _conv_constraint_matrix(cur)        # (2n‑1) × n
        b = np.full(A.shape[0], M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)       # minimise –Σg  ⇔ maximise Σg
        bounds = [(0, None)] * n

        try:
            res = scipy.optimize.linprog(c,
                                         A_ub=A,
                                         b_ub=b,
                                         bounds=bounds,
                                         method='highs')
        except Exception:                       # pragma: no cover
            break

        if not res.success:
            break

        g = np.maximum(res.x, 0.0)
        sum_g = g.sum()
        if sum_g <= 0.0:
            break

        # Scale g to the correct ℓ1‑norm
        scale = np.sqrt(2.0 * n) / sum_g
        g *= scale

        # Convex combination
        cur = (1.0 - t_mix) * cur + t_mix * g
        cur = np.maximum(cur, 0.0)
        cur_sum = cur.sum()
        if cur_sum > 0:
            cur *= np.sqrt(2.0 * n) / cur_sum

        cur_val = _score(cur)
        if cur_val < best_val:
            best_val = cur_val
            best = cur.copy()

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function sequence that minimises the supplied
    evaluate_sequence() function.  The routine respects the time budget,
    tries several lengths, power‑law exponents and offsets, refines each
    candidate with simulated annealing and (if SciPy is present) a
    linear‑programming mass‑redistribution step, and finally returns the
    best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0  # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 1.  Try a previously discovered construction if it exists.
    # ------------------------------------------------------------------
    try:
        global_seq = np.array(height_sequence_1, dtype=np.float64)
        if global_seq.size > 0:
            n_glob = global_seq.size
            global_seq = global_seq * (np.sqrt(2.0 * n_glob) / global_seq.sum())
            gl_score = _score(global_seq)
            if gl_score < best_score:
                best_score = gl_score
                best_seq = global_seq.copy()
                print(f"[seed] global sequence score = {gl_score:.6f}")

            # quick anneal on the global seed
            sub_deadline = min(deadline, time.time() + 5.0)
            seq_opt, seq_opt_score = _anneal(global_seq, sub_deadline)
            if seq_opt_score < best_score:
                best_score = seq_opt_score
                best_seq = seq_opt.copy()
                print(f"[seed] after anneal score = {seq_opt_score:.6f}")
    except Exception:  # pragma: no cover
        global_seq = None

    # ------------------------------------------------------------------
    # 2.  Systematic search over lengths / exponents / offsets.
    # ------------------------------------------------------------------
    lengths = [500, 800, 1200, 1800, 2500]          # modest sizes, all < 10 000
    exponents = [0.45, 0.48, 0.50, 0.52, 0.55]     # around the known optimum 0.5
    offsets = [1.0, 2.0, 3.0, 5.0]                # shift the singularity

    for n in lengths:
        for p in exponents:
            for off in offsets:
                if time.time() >= deadline:
                    break

                # ---- initialise -------------------------------------------------
                a = _initial_sequence(p, off, n)
                cur_score = _score(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[init] n={n:4d} p={p:.3f} off={off:.1f} → {cur_score:.6f}")

                # ---- short annealing -------------------------------------------
                sub_deadline = min(deadline, time.time() + 3.0)
                a_opt, opt_score = _anneal(a, sub_deadline)
                if opt_score < best_score:
                    best_score = opt_score
                    best_seq = a_opt.copy()
                    print(f"[anneal] n={n:4d} → {opt_score:.6f}")

                # ---- LP refinement (only for moderate n to keep memory sane) ---
                if _HAS_SCIPY and n <= 1200:
                    sub_deadline_lp = min(deadline, time.time() + 4.0)
                    a_lp, lp_score = _lp_improve(a_opt,
                                                 t_mix=0.5,
                                                 max_iters=3,
                                                 deadline=sub_deadline_lp)
                    if lp_score < best_score:
                        best_score = lp_score
                        best_seq = a_lp.copy()
                        print(f"[LP] n={n:4d} → {lp_score:.6f}")

                    # final tiny anneal after LP
                    sub_deadline2 = min(deadline, time.time() + 2.0)
                    a_final, final_score = _anneal(a_lp, sub_deadline2)
                    if final_score < best_score:
                        best_score = final_score
                        best_seq = a_final.copy()
                        print(f"[final anneal] n={n:4d} → {final_score:.6f}")

        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 3.  One last long anneal on the global best (if any time left).
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        a_last, last_score = _anneal(best_seq, time.time() + remaining)
        if last_score < best_score:
            best_score = last_score
            best_seq = a_last.copy()
            print(f"[final] best score improved to {last_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Fallback (should never be needed).
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 800
        best_seq = np.ones(n, dtype=np.float64)
        best_seq *= np.sqrt(2.0 * n) / best_seq.sum()
        print("[fallback] using uniform sequence")

    # Trim trailing near‑zero entries for neatness.
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    return best_seq.tolist()
