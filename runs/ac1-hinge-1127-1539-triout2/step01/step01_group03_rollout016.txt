        obj = cp.Maximize(cp.sum(b))
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.ECOS, warm_start=True, feastol=1e-9, abstol=1e-9, reltol=1e-9, max_iters=500)
        if b.value is None:
            break
        b_opt = np.array(b.value).flatten()
        sum_a = np.sum(a)
        sum_b = np.sum(b_opt)
        if sum_b == 0:
            break
        # scale b to have same sum as a (renorm)
        b_scaled = b_opt * (sum_a / sum_b)
        # blend
        t = 0.2
        a_new = (1 - t) * a + t * b_scaled
        # compute new score
        new_score = compute_score(a_new)
        if new_score < best_score:
            best_score = new_score
            best_seq = a_new.copy()
        # update a for next iteration
        a = a_new
        # update M = max conv of a
        M = np.max(np.convolve(a, a))
        iter += 1
        if best_score < 1.45: break
    return best_seq.tolist()

Propose a low‑score step function.

The search consists of two stages.
1)  A cheap grid search over the two most important
    hyper‑parameters of the classic construction
        a_i = (i+1+offset)^{-beta} .

    candidate of this stage already beats the previous record

2)  A few hundred iterations of a simple projected (sub‑)gradient
    descent that works on the true objective
        F(a) = 2·n·max(conv(a,a)) / (sum(a)^2) .
    The sub‑gradient is taken at the maximal convolution index,
    a small step is taken, negativity is projected away and the
    ℓ₁‑norm is re‑scaled (scaling does not change the score).
    Random pairwise mass transfers are used occasionally to escape
    plateaus.

All heavy work is done with FFT‑based convolutions, which makes a
single objective evaluation O(n log n).  With n=5000 the whole
procedure comfortably fits into the 1000 s budget and returns a
valid list of non‑negative floats.
"""

import time
import numpy as np


def _objective(seq: np.ndarray) -> float:
    """
    Fast evaluation of the score 2·n·max(conv(seq,seq)) / (sum(seq)**2).
    Uses an FFT based full linear convolution.
    Returns np.inf for degenerate sequences (sum < 0.01).
    """
    n = seq.shape[0]
    s = np.sum(seq)
    if s < 0.01:
        return np.inf

    # FFT length: next power of two for speed
    m = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft * fft, n=m)[: 2 * n - 1]
    max_conv = np.max(conv)
    return float(2.0 * n * max_conv / (s * s))


def _conv_and_max(seq: np.ndarray):
    """
    Return (full_convolution, max_value, argmax_index) using FFT.
    """
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft * fft, n=m)[: 2 * n - 1]
    max_val = np.max(conv)
    arg_max = int(np.argmax(conv))
    return conv, max_val, arg_max

    best_global_score = best_score

    while time.time() < deadline:
        conv, max_conv, k_max = _conv_and_max(a)

        # sub‑gradient for the active (max) index
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = a.copy()
        grad[mask] += a[idx[mask]]

        # step size proportional to the current ratio r = max_conv / sum(a)^2
        r = max_conv / (target_sum ** 2)
        step = step_factor * r

        new_a = a - step * grad
        new_a = np.maximum(new_a, 0.0)

        # re‑normalise the ℓ₁‑norm (does not affect the score)
        s = np.sum(new_a)
        if s > 1e-12:
            new_a *= target_sum / s
        else:
            # all entries vanished – abort refinement
            break

        new_score = _objective(new_a)

        if new_score < best_global_score:
            best_global_score = new_score
            best_global_seq = new_a.copy()
            a = new_a
            no_improve = 0
            step_factor = min(step_factor * 1.07, 2.0)   # slowly enlarge steps
            if iter_cnt % 100 == 0:
                print(f"[grad] iter={iter_cnt:4d}, score={new_score:.6f}")
        else:
            no_improve += 1
            # occasional random mass transfer to jump out of plateaus
            if no_improve % 30 == 0:
                i, j = np.random.randint(n, size=2)
                delta = a[i] * 0.05 * np.random.rand()
                a[i] = max(0.0, a[i] - delta)
                a[j] += delta
                a *= target_sum / np.sum(a)
                step_factor *= 0.9

        iter_cnt += 1

        # Early exit once we are safely below the required threshold
        if best_global_score < 1.48:
            print(f"[!]" +
                  f" target comfortably beaten: score={best_global_score:.6f}"
                  f" after {iter_cnt} iterations.")
            break

    print(f"[+] Search finished. Best score = {best_global_score:.6f}")
    # convert to plain python list of floats (the evaluator expects a list)
    return [float(v) for v in best_global_seq]
