def linear_convolution_fft(x, y):
    n = len(x)
    m = len(y)
    conv_len = n + m - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[:conv_len]
    return conv
```

def _linear_conv(a, b):
    n = a.shape[0]; m = b.shape[0]
    L = n+m-1
    N = 1 << (L-1).bit_length()
    Fa = np.fft.rfft(a, n=N)
    Fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(Fa*Fb, n=N)[:L]
    return conv
```

def _smooth_gradient(a, beta):
    n = a.shape[0]
    conv = _linear_conv(a, a)   # length 2n-1
    # softmax
    max_val = np.max(beta * conv)
    # stable: compute exp(beta*conv - max)
    exp_vals = np.exp(beta * conv - max_val)
    w = exp_vals / exp_vals.sum()
    # reverse a
    a_rev = a[::-1]
    # convolution w * a_rev
    grad_conv = _linear_conv(w, a_rev)  # length 3n-2
    # extract gradient
    # g_k = 2 * grad_conv[n-1 + k]
    g = 2.0 * grad_conv[n-1 : n-1 + n]
    return g

"""
Soft‑max‑smoothed exponentiated‑gradient descent on the simplex.

The algorithm works with a sequence `a` that is kept ℓ₁‑normalised to
√(2·n).  For a fixed length n the external score equals the maximal entry of
the self‑convolution `a*a`.  At each iteration we:

1. Compute the full linear convolution `c = a*a`.
2. Form a soft‑max distribution `w ∝ exp(β·c)` (β starts small and grows
   slowly).  This yields a smooth approximation of the max‑operator.
3. The gradient of the smoothed objective is
      g = 2·( w  ⋆  a_rev )[n‑1 : n‑1+n] ,
   where `a_rev` is `a` reversed and `⋆` denotes linear convolution.
   It is obtained again by a FFT‑based convolution.
4. Perform a multiplicative update `a ← a·exp(−η·g)` (which preserves
   non‑negativity) and re‑scale to keep ∑a = √(2·n).
5. Occasionally add a tiny random jitter to escape plateaus.

Multiple random restarts are performed within the allotted time budget.
The best sequence seen (according to the internal max‑convolution value,
which equals the official score for a normalised vector) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear convolution helpers (FFT based, real inputs)
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_linear(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using a real FFT."""
    n = x.shape[0]
    m = y.shape[0]
    conv_len = n + m - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fx = np.fft.rfft(x, n=fft_len)
    Fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(Fx * Fy, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Gradient of the β‑soft‑max approximation of max(a*a)
# ----------------------------------------------------------------------


def _smooth_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute ∇_{a}  ( 1/β·log Σ_k exp(β·(a*a)[k]) )
    efficiently via two FFT‑based convolutions.
    """
    n = a.shape[0]

    # soft‑max weights (w_k ≥ 0, Σ w_k = 1)
    scaled = beta * conv
    max_scaled = np.max(scaled)
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # gradient = 2 * (w ⋆ a_rev)[n‑1 : n‑1+n]
    a_rev = a[::-1]
    grad_full = _conv_linear(w, a_rev)          # length = (2n‑1) + n‑1 = 3n‑2
    g = 2.0 * grad_full[n - 1 : n - 1 + n]      # keep only the n relevant entries
    return g


# ----------------------------------------------------------------------
# Single optimisation run (one random start)
# ----------------------------------------------------------------------


def _optimise_one_start(a: np.ndarray, deadline: float, rng: np.random.Generator):
    """Exponentiated‑gradient descent with soft‑max smoothing until `deadline`."""
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # best local result (tracked by the true max‑convolution value)
    conv = _conv_self(a)
    best_val = float(np.max(conv))
    best_seq = a.copy()

    # hyper‑parameters (somewhat tuned empirically)
    eta = 0.20          # initial step size for the multiplicative update
    eta_decay = 0.9995
    beta = 0.5          # start with a very smooth approximation
    beta_mul = 1.003
    beta_max = 20.0
    jitter_prob = 0.02
    jitter_scale = 0.01

    while time.time() < deadline:
        # ------------------------------------------------------------------
        # 1) current convolution and improvement check
        # ------------------------------------------------------------------
        conv = _conv_self(a)
        cur_max = float(np.max(conv))

        if cur_max < best_val:
            best_val = cur_max
            best_seq = a.copy()

        # ------------------------------------------------------------------
        # 2) gradient of the smoothed max
        # ------------------------------------------------------------------
        g = _smooth_gradient(a, conv, beta)

        # ------------------------------------------------------------------
        # 3) multiplicative update + re‑normalisation
        # ------------------------------------------------------------------
        a = a * np.exp(-eta * g)

        total = a.sum()
        if total == 0.0:                 # (practically never happens)
            a = np.full(n, S / n)
        else:
            a *= (S / total)

        # ------------------------------------------------------------------
        # 4) occasional jitter to help escape tiny plateaus
        # ------------------------------------------------------------------
        if rng.random() < jitter_prob:
            noise = rng.standard_normal(n) * a.mean() * jitter_scale
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()

        # ------------------------------------------------------------------
        # 5) schedule updates
        # ------------------------------------------------------------------
        eta *= eta_decay
        beta = min(beta_max, beta * beta_mul)

    return best_seq, best_val
