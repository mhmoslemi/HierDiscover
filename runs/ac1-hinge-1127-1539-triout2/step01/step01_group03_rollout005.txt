import cvxpy as cp

def solve_lp(a, M):
    n = len(a)
    A = convolution_matrix(a) # shape (2n-1, n)
    b = cp.Variable(n, nonneg=True)
    constraints = [A @ b <= M]
    # Additionally maybe require sum to be something? Not required.
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True)
    opt_b = b.value
    return opt_b

def solve_lp_for_a(a, M):
    n = len(a)
    A = build_conv_matrix(a)
    # Upper bounds vector
    ub = np.full(2*n - 1, M, dtype=np.float64)
    # c: we minimize -sum(b) -> c = -np.ones(n)
    c_vec = -np.ones(n, dtype=np.float64)

    # Use linprog
    res = linprog(c_vec, A_ub=A, b_ub=ub, bounds=[(0, None)] * n, method='highs')
    if res.success:
        b_opt = res.x
        return b_opt
    else:
        print("[LP] Solver failed:", res.message)
        return None

"""
Propose a low‑score step function.

The algorithm combines three ideas:

1.  A cheap grid‑search over the classic 1/√(i+c) shape to obtain a
    decent starting point (score ≈ π/2 ≈ 1.57).

2.  An *LP‑based improvement*:  given the current sequence `a`,
    we maximise the ℓ₁‑norm of a new non‑negative sequence `b`
    under the linear constraints   (a * b)[k] ≤ M  (where M is the
    current maximal auto‑convolution).  After scaling `b` to the
    standard ℓ₁‑norm `sqrt(2 n)` we replace `a` by this improved
    sequence.  Each replacement cannot increase the maximal
    convolution, and usually reduces it substantially.

3.  A short sub‑gradient descent phase (identical to the one used in the
    reference implementation) to polish the final sequence.

The whole search is limited to the supplied time budget (default 1000 s);
the best sequence found so far is returned.  The evaluation function
`evaluate_sequence` is **not** called inside the search – we use a fast
FFT based proxy (`_objective`) that matches the true score exactly for
all admissible sequences.

The routine works for a moderate size (n≈1000) which keeps the linear
program tractable; the resulting sequence can be used directly – the
score is independent of the length once the ℓ₁‑norm is normalised.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog


def _objective(seq: np.ndarray) -> float:
    """Fast computation of the evaluation score.
    Returns 2·n·max(conv(seq,seq)) / (sum(seq)²).
    """
    n = len(seq)
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    # next power‑of‑two for FFT speed
    m = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft * fft, n=m)[: 2 * n - 1]
    return float(2.0 * n * np.max(conv) / (s * s))


def build_convolution_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Construct the (2n‑1)×n matrix `A` such that  (A @ b)ₖ = (a * b)ₖ .
    The (k‑j)‑th entry of `a` appears on the diagonal with offset=j.
    """
    n = len(a)
    # each diagonal offset j contains the constant value a[j]
    data = [np.full(n, a_j, dtype=np.float64) for a_j in a]
    offsets = np.arange(n, dtype=int)
    rows = 2 * n - 1
    A = sp.diags(data, offsets, shape=(rows, n), format="csr")
    return A


def lp_step(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve
        maximise   sum(b)
        subject to A @ b <= M,   b >= 0
    where A encodes the convolution with the current `a`.
    Returns the optimal vector `b` (un‑scaled) or None on failure.
    """
    n = len(a)
    A = build_convolution_matrix(a)

    # Upper bounds for every convolution index
    ub = np.full(2 * n - 1, M, dtype=np.float64)

    # linprog minimises, so we minimise -sum(b)
    c = -np.ones(n, dtype=np.float64)

    # use HiGHS (efficient sparse LP solver)
    res = linprog(
        c,
        A_ub=A,
        b_ub=ub,
        bounds=[(0, None)] * n,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        print("[LP] Solver failed:", res.message)
        return None
    b = np.maximum(res.x, 0.0)  # clip tiny negatives caused by numerics
    return b

        # full auto‑convolution (FFT)
        m_fft = 1 << ((2 * n - 1).bit_length())
        af = np.fft.rfft(a, n=m_fft)
        conv = np.fft.irfft(af * af, n=m_fft)[: 2 * n - 1]
        max_conv = np.max(conv)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the active (max) entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        shift = np.zeros_like(a)
        shift[mask] = a[idx[mask]]
        grad = a + shift                     # g_i = a_i + a_{k‑max‑i}

        # step size proportional to the current ratio
        r = max_conv / (np.sum(a) ** 2)
        step = step_factor * r

        new_a = a - step * grad
        new_a = np.maximum(new_a, 0.0)

        s = np.sum(new_a)
        if s < 1e-12:
            break
        new_a *= target_sum / s              # re‑normalise

        new_score = _objective(new_a)

        if new_score < best_global_score:
            best_global_score = new_score
            best_global_seq = new_a.copy()
            a = new_a
            no_improve = 0
            step_factor = min(step_factor * 1.1, 2.0)
            if iter_gd % 50 == 0:
                print(f"[grad] iter={iter_gd:4d} → score={new_score:.6f}")
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                # random mass transfer to escape plateaus
                i, j = np.random.randint(n, size=2)
                delta = 0.05 * a[i] * np.random.rand()
                a[i] = max(0.0, a[i] - delta)
                a[j] += delta
                a *= target_sum / np.sum(a)
                step_factor *= 0.9
        iter_gd += 1

        if best_global_score < 1.503:
            print(f"[+] Target reached: score={best_global_score:.6f}")
            break

    print(f"[+] Search finished. Best score = {best_global_score:.6f}")
    # Convert to plain Python list as required by the evaluator
    return [float(x) for x in best_global_seq]
