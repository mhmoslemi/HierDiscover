"""
Optimiser for the self‑convolution bound using a smoothed “flatten‑the‑auto‑correlation’’
gradient descent.  The idea is to drive the convolution a∗a towards a constant
value (its average S²/(2·n‑1)) by a gradient step obtained from the weighted
correlation w_k = exp(β·(c_k‑c̄))‑1, where c_k are the current convolution entries.
Positive w_k penalise peaks, negative w_k reward troughs, thus flattening the
shape of the convolution and lowering its maximum.  The update is projected back
onto the simplex (ℓ₁‑norm = √(2n), non‑negative) and a simple momentum term is
added.  The algorithm runs for at most ``budget_s`` seconds, occasionally
injects small random perturbations to escape plateaus and keeps the best
sequence seen.  All helper functions are top‑level (no closures) and the code
conforms to the required interface.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector to have ℓ₁‑norm = target.
    Zero entries stay zero; if the whole vector is zero a uniform
    distribution is returned."""
    s = v.sum()
    if s > 0.0:
        return v * (target / s)
    return np.full_like(v, target / v.shape[0])


def generate_U_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
) -> np.ndarray:
    """Symmetric “U‑shaped’’ vector of length ``n``.
    ``kind`` ∈ {linear, quadratic, exponential}.  ``param`` ≥ 0 controls the
    edge‑enhancement strength.  The result is scaled to the prescribed ℓ₁‑norm."""
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # ∈[0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_to_sum(raw, target_sum)


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """Transfer a tiny amount of mass from a random index to another."""
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0.0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_to_sum(a, target_sum)


# ----------------------------------------------------------------------
# FFT based convolution / correlation (full, linear)
# ----------------------------------------------------------------------


def fft_len(min_len: int) -> int:
    """Next power of two ≥ ``min_len``."""
    return 1 << ((min_len).bit_length())


def full_self_convolution_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a∗a of length 2·len(a)‑1 using a FFT of length ``L``."""
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def correlation_fft(a: np.ndarray, w: np.ndarray, L: int) -> np.ndarray:
    """
    Linear correlation (a ⋆ w)[ℓ] = Σ_i a_i·w_{i+ℓ} for ℓ = 0,…,len(a)-1.
    Implemented via FFT of length ``L`` (≥ len(a)+len(w)‑1).
    """
    fa = np.fft.rfft(a, n=L)
    fw = np.fft.rfft(w, n=L)
    corr = np.fft.irfft(fa * np.conj(fw), n=L)
    return corr[: a.shape[0]]


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence`` (i.e. the maximum entry of the self‑convolution).
    A gradient descent that tries to flatten the auto‑correlation is employed,
    combined with momentum, occasional random pair transfers and jitter
    rescues.  The best sequence found within the time budget is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Choose the length (reuse a known construction if present)
    # ------------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        a = np.array(base, dtype=np.float64)
    else:
        # a reasonably large length that still fits comfortably in the budget
        n = np.random.randint(7500, 9000)
        a = generate_U_shape(n, np.sqrt(2.0 * n), kind="exponential", param=1.0)

    target_sum = np.sqrt(2.0 * n)                     # normalisation used in the paper
    a = project_to_sum(np.maximum(a, 0.0), target_sum)

    # ------------------------------------------------------------------
    # Pre‑compute constants and FFT size
    # ------------------------------------------------------------------
    L = fft_len(3 * n - 2)                           # ≥ n + (2n‑1) – 1 for correlation
    target_conv = target_sum * target_sum / (2 * n - 1)   # average value of a∗a

    # ------------------------------------------------------------------
    # Helper to evaluate cheaply (same formula as evaluate_sequence)
    # ------------------------------------------------------------------
    def fast_score(conv_vec: np.ndarray) -> float:
        return 2.0 * n * conv_vec.max() / (target_sum * target_sum)

    # initialise best
    conv = full_self_convolution_fft(a, L)
    best_score = fast_score(conv)
    best_seq = a.copy()

    print(f"[{time.time():.2f}] start n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optimisation parameters
    # ------------------------------------------------------------------
    step = 0.02 * (target_sum / n)            # initial step ≈2 % of average entry
    beta = 0.5                                 # controls sharpness of weighting
    momentum = np.zeros_like(a)
    no_improve = 0
    it = 0

    # ------------------------------------------------------------------
    # Main loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ----- compute convolution and weighting ---------------------------------
        fa = np.fft.rfft(a, n=L)

        # self‑convolution (full length 2n‑1)
        conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]

        # weight vector w_k = exp(β·(c_k‑c̄))‑1   (negative for troughs, positive for peaks)
        diff = conv - target_conv
        w = np.exp(beta * diff) - 1.0

        # ----- correlation a ⋆ w (gradient of the smoothed max) ------------------
        fw = np.fft.rfft(w, n=L)
        corr = np.fft.irfft(fa * np.conj(fw), n=L)[: n]   # length n

        # gradient of the objective (up to a constant factor 2)
        g = 2.0 * corr

        # direction that preserves the ℓ₁‑norm (project onto hyperplane Σx=0)
        direction = -g + np.mean(g)

        # momentum update
        momentum = 0.9 * momentum + 0.1 * direction

        # ----- tentative step ----------------------------------------------------
        a_new = a + step * momentum
        a_new = np.clip(a_new, 0.0, None)
        a_new = project_to_sum(a_new, target_sum)

        # ----- evaluate -----------------------------------------------------------
        conv_new = np.fft.irfft(np.fft.rfft(a_new, n=L) *
                               np.fft.rfft(a_new, n=L), n=L)[: 2 * n - 1]
        score_new = fast_score(conv_new)

        it += 1

        if score_new + 1e-12 < best_score:
            best_score = score_new
            best_seq = a_new.copy()
            no_improve = 0
            print(f"Iter {it:06d} – new best {best_score:.6f}")
        else:
            no_improve += 1

        a = a_new

        # ----- schedule adjustments ------------------------------------------------
        if it % 1000 == 0:
            step = max(step * 0.93, 1e-6)          # gradual decay, never too tiny
            beta = min(beta + 0.02, 5.0)           # sharpen weight function slowly

        # occasional random pair transfer (helps escape shallow plateaus)
        if np.random.rand() < 0.015:
            a = random_pair_transfer(a, target_sum, scale=0.005)

        # stagnation handling
        if no_improve >= 5000:
            # jitter around the current best solution
            jitter = np.random.randn(n) * 0.001
            a = np.clip(best_seq + jitter, 0.0, None)
            a = project_to_sum(a, target_sum)
            momentum.fill(0.0)
            no_improve = 0
            print(f"Iter {it:06d} – jitter/resample applied")

        # light progress report near the end of the budget
        if (deadline - time.time()) < 60.0 and it % 500 == 0:
            print(f"[{time.time():.2f}] time left ≈{deadline - time.time():.0f}s, best={best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] optimisation finished, best score {best_score:.6f}")
    # ensure the returned list respects the required format
    return best_seq.tolist()
