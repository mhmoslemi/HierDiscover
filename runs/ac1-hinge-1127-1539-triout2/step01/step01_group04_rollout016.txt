```
def proj_simplex(v, s):
    # v is numpy array, want projection onto nonnegative simplex with sum=s
    # Implementation from Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def adam_step(a, grad, m, v, t, alpha, beta1, beta2, eps):
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad ** 2)
    m_hat = m / (1 - beta1 ** t)

    import time, math, numpy as np
    from scipy.optimize import linprog
    from scipy.sparse import diags

    np.random.seed(seed)
    deadline= time.time() + budget_s - 0.5
    # Determine n
    try:
        n = len(height_sequence_1)
    except Exception:
        n = 2000
    S = math.sqrt(2.0 * n)

    # initialize a via sqrt difference
    idx = np.arange(n, dtype=np.float64)
    a = np.sqrt(idx + 1) - np.sqrt(idx)
    a *= S / a.sum()

    best_seq = a.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"Initial score {best_score:.6f}")

    L = 1 << ((2*n - 1).bit_length())
    # Adam params
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_adj = 0
    step_iter = 0

    # function for projection onto simplex
    def project_simplex(v):
        # v is np array, desire sum=S, nonnegative
        # Duchi et al algorithm
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - S
        ind = np.arange(1, len(v)+1)
        cond = u - cssv / ind > 0
        if not np.any(cond):
            # fallback uniform
            return np.full_like(v, S/len(v))
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
        w = np.maximum(v - theta, 0)
        return w

    # function for LP step
    def lp_step(a_vec, M_val):
        # Build convolution matrix A for a_vec
        # Offsets are 0..n-1
        offsets = np.arange(n, dtype=int)
        # create data matrix with each row i constant a_i
        data = np.broadcast_to(a_vec[:, None], (n, n))
        A = diags(data, offsets, shape=(2*n-1, n), format='csr')
        c = -np.ones(n)  # maximize sum(g) -> minimize -sum(g)
        b_ub = np.full(2*n-1, M_val)
        bounds = [(0, None)] * n
        # solve LP
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            return None
        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            return None
        # scale g0 to have sum S
        g = g0 * (S / sum_g0)
        # try mixing with various t values, keep best reduction in M
        best_a = a_vec
        best_M = M_val
        # precompute L for FFT
        for t in (0.5, 0.3, 0.2, 0.1, 0.05):
            a_new = (1 - t) * a_vec + t * g
            a_new = project_simplex(a_new)
            conv_new = _conv_fft(a_new, L)
            M_new = conv_new.max()
            if M_new < best_M:
                best_M = M_new
                best_a = a_new
        if best_M < M_val - 1e-9:
            return best_a
        else:
            return None

    # main loop
    while time.time() < deadline:
        conv = _conv_fft(a, L)
        M = conv.max()
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"Iter {step_iter:8d} – new best {best_score:.6f}")

        # Time check for LP step
        time_left = deadline - time.time()
        # run LP step every, say, 5 iterations if time allows
        if (step_iter % 5 == 0) and time_left > 7.0:
            new_a = lp_step(a, M)
            if new_a is not None:
                a = new_a
                # after LP step skip gradient step this iteration
                step_iter += 1
                continue

        # gradient (subgradient) step
        epsg = 1e-12 * M
        ks = np.where(conv >= M - epsg)[0]
        grad = np.zeros_like(a)
        for k in ks:
            _add_gradient_for_k(grad, a, k)

        # Adam update
        step_iter += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** step_iter)
        v_hat = v / (1 - beta2 ** step_iter)
        lr = 0.4 / (math.sqrt(step_iter) + 1)  # heuristic larger initial lr
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # occasional jitter

            a = np.maximum(a + noise, 0)
            a = project_simplex(a)

    print(f"Finished after {step_iter} iterations – best score {best_score:.6f}")
    return best_seq.tolist()

"""
Hybrid LP–gradient optimizer for the step‑function challenge.

The algorithm starts from the classic √‑difference sequence (which gives a
score ≈ π/2).  It repeatedly does two things while the time budget allows:

1. **Linear‑programming improvement** – for the current vector `a` we
   maximise the L₁‑mass of a non‑negative vector `g` under the
   constraints `a * g ≤ M` (where `M` is the current maximal self‑convolution).
   If `g` carries more total mass than the target L₁‑norm, we mix the two
   vectors (`a ← (1–t) a + t g`) and keep the mixture that yields the greatest
   reduction of the maximal convolution.  This is exactly the iteration
   described in the statement and quickly pushes the score well below the
   √‑difference baseline.

2. **Projected Adam (sub‑gradient) descent** – after each LP step, or whenever
   the remaining time is too short for another LP, we perform a few hundred
   Adam‑style updates using the sub‑gradient of the active convolution peak.
   The updated vector is projected onto the simplex
   `{x ≥ 0, Σx = √(2·n)}` after each step.

Random jitter is added every few hundred iterations to escape plateaus.
The best sequence seen throughout the run is returned.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# Helper routines (top‑level, no closures)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Self‑convolution of a real vector using an FFT of length L (a power of two).
    Returns the first 2·len(a)‑1 entries.
    """
    fa = np.fft.rfft(a, n=L)
    return np.fft.irfft(fa * fa, n=L)[: 2 * a.size - 1]


def _add_gradient_for_k(grad: np.ndarray, a: np.ndarray, k: int) -> None:
    """
    Add the sub‑gradient contribution for a shift `k` to `grad`.
    The contribution is 2·a_{k‑i} for each valid index i.
    """
    n = a.size
    if k < n:
        # indices 0 … k
        grad[: k + 1] += 2.0 * a[k :: -1]
    else:
        # indices k‑(n‑1) … n‑1
        start = k - n + 1
        length = 2 * n - 1 - k
        grad[start:] += 2.0 * a[: length][::-1]


def project_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of `v` onto the simplex {x ≥ 0, Σx = target_sum}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]               # descending sort
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)          # 1‑based indices
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, target_sum / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def lp_step(a_vec: np.ndarray, M_val: float, target_sum: float, L_fft: int) -> np.ndarray | None:
    """
    Linear‑programming refinement step.

    Given the current vector `a_vec` (with Σa = target_sum) and its maximal
    self‑convolution value `M_val`, solve the LP

        maximise   Σ g_j
        subject to g_j ≥ 0,
                   (a_vec * g)[k] ≤ M_val   ∀k.

    If the LP produces a vector with larger L₁‑mass, scale it to the target
    sum, mix it with `a_vec` for a few mixing coefficients and return the
    best mixture (or None if no improvement is found).
    """
    n = a_vec.size

    # Build the convolution matrix A (size (2n‑1) × n) such that (A @ g) = a * g.
    # Row k, column j contains a_{k‑j} if 0 ≤ k‑j < n else 0.
    # This matrix can be built efficiently with `scipy.sparse.diags`.
    offsets = np.arange(n, dtype=int)
    # each diagonal i consists of the constant a_i repeated n times
    data = np.broadcast_to(a_vec[:, None], (n, n))
    A = diags(data, offsets, shape=(2 * n - 1, n), format="csr")

    # LP: maximise Σ g  <=> minimise -Σ g
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_val)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    if not res.success:
        return None
    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 <= target_sum:
        # No additional mass can be packed – no improvement possible
        return None

    # Scale g0 to have the correct L₁‑norm
    g = g0 * (target_sum / sum_g0)

    # Try a few mixing coefficients and keep the best reduction of M
    best_a = a_vec
    best_M = M_val
    for t in (0.5, 0.3, 0.2, 0.1, 0.05):
        a_new = (1.0 - t) * a_vec + t * g
        a_new = project_simplex(a_new, target_sum)
        conv_new = _conv_fft(a_new, L_fft)
        M_new = conv_new.max()
        if M_new < best_M:
            best_M = M_new
            best_a = a_new

    # Return a new vector only if we have strictly improved the bound
    if best_M < M_val - 1e-9:
        return best_a
    return None
