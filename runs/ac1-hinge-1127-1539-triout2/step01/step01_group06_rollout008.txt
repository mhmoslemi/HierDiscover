Important to implement Euclidean projection onto simplex with given sum S (S = sqrt(2 * n)). This algorithm:

def project_onto_simplex(v, s):
    if s < 0: raise error.
    n = len(v)
    u = sorted(v, reverse=True)
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _optimize(a, deadline):
    n = a.size
    S = np.sqrt(2.0 * n)
    # scale to have sum S if not already (should be)
    cur_sum = a.sum()
    if abs(cur_sum - S) > 1e-12:
        a = a * (S / cur_sum)
    best_a = a.copy()
    best_val = _max_convolution(a)  # since sum normalized, value = objective
    cur_val = best_val
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        # compute convolution
        conv = _convolution_array(a)
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]
        # gradient-based move
        lo = max(0, max_idx - (n - 1))
        hi = min(max_idx, n - 1)
        idx_range = np.arange(lo, hi+1)
        # compute gradient g: a_{max_idx - i}
        g = np.zeros_like(a)
        g[idx_range] = a[max_idx - idx_range]
        # choose source weighted by a_i * g_i (product)
        weights = a * g
        total_w = weights.sum()
        if total_w > 0:
            src = rng.choice(n, p=weights / total_w)
        else:
            # fallback random source
            src = rng.integers(lo, hi+1)
        # choose destination with minimal g (preferring low impact)
        # if multiple minima, random among them
        min_g = g.min()
        sink_candidates = np.where(g == min_g)[0]
        if sink_candidates.size > 1:
            dst = rng.choice(sink_candidates)
        else:
            dst = sink_candidates[0] if sink_candidates.size else rng.integers(0, n)
        # ensure src != dst, if equal pick another
        if src == dst:
            dst = (dst + 1) % n
        # propose move
        delta_frac = rng.random() * 0.1  # up to 10% of source mass
        delta = a[src] * delta_frac
        if delta <= 0:
            continue
        a[src] -= delta
        a[dst] += delta
        # evaluate after move
        new_conv = _convolution_array(a)
        new_val = new_conv.max()
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}] improved global best {best_val:.6f}")
        else:
            # reject move
            a[dst] -= delta
            a[src] += delta
        # occasional random perturb - every 100 iterations
        it += 1
        if it % 200 == 0:
            # random shuffle some mass
            i1 = rng.integers(0, n)
            i2 = rng.integers(0, n)
            if i1 != i2:
                # exchange a small portion
                eps = a[i1] * rng.random() * 0.02
                a[i1] -= eps
                a[i2] += eps
        # optionally break early if cur_val close to theoretical lower bound (maybe <1.5)
    return best_a, best_val

"""
Optimization of a non‑negative step function to minimise
    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm explores many random initialisations (power‑law,
exponential and uniform) for a variety of lengths, then refines each
candidate with a fast projected sub‑gradient / hill‑climbing loop that
preserves the ℓ1‑norm.  Moves are biased towards indices that contribute
most to the current maximum convolution, and occasional swaps are tried
to separate large coefficients.  The best sequence seen before the time
budget expires is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers
# ----------------------------------------------------------------------
def _convolution_array(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT (full length)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_convolution_array(a).max())


# ----------------------------------------------------------------------
# Simple simplex projection (keeps sum fixed, enforces non‑negativity)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Initial shape generators
# ----------------------------------------------------------------------
def _init_power_law(p: float, n: int, shift: int = 0) -> np.ndarray:
    """a_i ∝ (i+1+shift)^{-p}, renormalised to ℓ1‑norm √(2n)."""
    i = np.arange(1, n + 1, dtype=np.float64) + float(shift)
    a = i ** (-p)
    a *= math.sqrt(2.0 * n) / a.sum()
    return a


def _init_exponential(lam: float, n: int) -> np.ndarray:
    """a_i ∝ exp(-lam·i), renormalised to ℓ1‑norm √(2n)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    a *= math.sqrt(2.0 * n) / a.sum()
    return a


def _init_random(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random positive vector with the required ℓ1‑norm."""
    a = rng.random(n)
    a *= math.sqrt(2.0 * n) / a.sum()
    return a


# ----------------------------------------------------------------------
# Core optimiser for a single starting vector
# ----------------------------------------------------------------------
def _optimize(start_a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient / hill‑climbing optimiser.
    Keeps the ℓ1‑norm fixed ( √(2·n) ) and never leaves the non‑negative orthant.
    Returns the best vector seen and its associated max‑convolution value.
    """
    n = start_a.size
    target_sum = math.sqrt(2.0 * n)

    # ensure the starting vector respects the norm exactly
    a = start_a * (target_sum / start_a.sum())
    best_a = a.copy()
    best_val = _max_convolution(a)               # equal to the objective when sum is fixed
    cur_val = best_val

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        # -------------------------------------------------------------
        # 1) compute full convolution and locate the currently binding k*
        # -------------------------------------------------------------
        conv = _convolution_array(a)
        k_star = int(conv.argmax())
        max_val = float(conv.max())

        # -------------------------------------------------------------
        # 2) gradient‑biased mass transfer
        # -------------------------------------------------------------
        lo = max(0, k_star - (n - 1))
        hi = min(k_star, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # gradient g_i = a_{k*‑i}   (zero outside the support)
        g = np.zeros_like(a)
        g[idx] = a[k_star - idx]

        # source index weighted by a_i·g_i  (large contribution to the peak)
        src_weights = a * g
        if src_weights.sum() > 0:
            src = rng.choice(n, p=src_weights / src_weights.sum())
        else:
            src = rng.integers(lo, hi + 1)

        # destination index with smallest gradient (least impact on the peak)
        min_g = g.min()
        dst_candidates = np.where(g == min_g)[0]
        if dst_candidates.size:
            dst = rng.choice(dst_candidates)
        else:
            dst = rng.integers(0, n)

        if src == dst:
            dst = (dst + 1) % n

        # move a random fraction (up to 10 %) of the source mass to the destination
        frac = rng.random() * 0.10
        delta = a[src] * frac
        if delta > 0.0:
            a[src] -= delta
            a[dst] += delta

            new_val = _max_convolution(a)
            if new_val < cur_val:
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best_a = a.copy()
                    print(f"[{time.time():.1f}]   improved best {best_val:.6f}")
            else:
                # revert the move
                a[dst] -= delta
                a[src] += delta

        # -------------------------------------------------------------
        # 3) occasional random swap – helps to separate large coefficients
        # -------------------------------------------------------------
        if it % 150 == 0:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                a[i], a[j] = a[j], a[i]          # swap
                new_val = _max_convolution(a)
                if new_val < cur_val:
                    cur_val = new_val
                    if new_val < best_val:
                        best_val = new_val
                        best_a = a.copy()
                        print(f"[{time.time():.1f}]   swap improved best {best_val:.6f}")
                else:
                    # keep the swap only if it improves; otherwise revert
                    a[i], a[j] = a[j], a[i]

        # -------------------------------------------------------------
        # 4) occasional small random perturbation (simulating annealing)
        # -------------------------------------------------------------
        if it % 200 == 0:
            i1 = rng.integers(0, n)
            i2 = rng.integers(0, n)
            if i1 != i2 and a[i1] > 0.0:
                eps = a[i1] * rng.random() * 0.02
                a[i1] -= eps
                a[i2] += eps
                # re‑project onto the simplex to guarantee the sum constraint
                a = _project_simplex(a, target_sum)

                cur_val = _max_convolution(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.1f}]   perturb improved best {best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Public entry point required by the challenge
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises the given evaluation function.
    The routine runs for up to ``budget_s`` seconds, trying several
    randomly generated initial vectors and refining each with a fast
    sub‑gradient hill‑climber.  The best sequence found is returned as a
    plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0          # leave a small safety margin

    # -----------------------------------------------------------------
    # Global best holder
    # -----------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # -----------------------------------------------------------------
    # Try to start from the previously discovered construction, if any
    # -----------------------------------------------------------------
    try:
        # height_sequence_1 may be injected into the global namespace
        prev = globals()["height_sequence_1"]
        prev_arr = np.asarray(prev, dtype=np.float64)
        n0 = prev_arr.size
        target_sum0 = math.sqrt(2.0 * n0)
        prev_arr = prev_arr * (target_sum0 / prev_arr.sum())
        best_seq = prev_arr.copy()
        best_score = _max_convolution(best_seq)
        print(f"Start from supplied height_sequence_1, score = {best_score:.6f}")
    except Exception:
        pass

    # -----------------------------------------------------------------
    # Fallback initialisation (uniform) if nothing else is available
    # -----------------------------------------------------------------
    if best_seq is None:
        n0 = 2000
        best_seq = np.full(n0, math.sqrt(2.0 * n0) / n0, dtype=np.float64)
        best_score = _max_convolution(best_seq)
        print(f"Fallback uniform start, score = {best_score:.6f}")

    # -----------------------------------------------------------------
    # Quick refinement of the current global best
    # -----------------------------------------------------------------
    sub_deadline = min(deadline, time.time() + 8.0)
    refined_seq, refined_score = _optimize(best_seq.copy(), sub_deadline)
    if refined_score < best_score:
        best_score = refined_score
        best_seq = refined_seq.copy()
        print(f"[{time.time():.1f}] refined start, new best {best_score:.6f}")

    # -----------------------------------------------------------------
    # Main stochastic search loop
    # -----------------------------------------------------------------
    lengths = [1800, 2500, 3200, 4000, 5000, 6400, 8000]
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 5.0:
            break

        # pick a length and a shape descriptor
        n = int(rng.choice(lengths))
        shape_type = rng.choice(["power", "exp", "rand"])

        if shape_type == "power":
            p = rng.uniform(0.20, 0.80)
            shift = int(rng.integers(-12, 13))   # moderate shift, may be negative
            a0 = _init_power_law(p, n, shift)
        elif shape_type == "exp":
            lam = rng.uniform(5e-4, 2e-2)
            a0 = _init_exponential(lam, n)
        else:   # completely random
            a0 = _init_random(n, rng)

        # allocate a modest time slice for this candidate
        slice_deadline = min(deadline, time.time() + min(7.0, remaining * 0.85))
        a_opt, score_opt = _optimize(a0, slice_deadline)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] new global best {best_score:.6f} (n={n}, shape={shape_type})")

    # -----------------------------------------------------------------
    # Final tidy‑up
    # -----------------------------------------------------------------
    if best_seq is None:
        # absolute safeguard – should never happen
        best_seq = np.array([math.sqrt(2.0)], dtype=np.float64)

    # Trim trailing negligible entries (helps with runtime of the outer harness)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    return best_seq.tolist()
