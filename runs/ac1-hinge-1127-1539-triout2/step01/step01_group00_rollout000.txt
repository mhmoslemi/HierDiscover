def simulated_annealing(a, C, deadline):
    n = len(a)
    conv_len = 2*n-1
    cur_max = C.max()
    best_seq = a.copy()
    best_score = cur_max
    rng = np.random.default_rng()
    T0 = cur_max * 0.1  # initial temperature relative to max
    T = T0
    decay = 0.99995
    iter = 0
    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i: j+=1
        if a[i] < 1e-12:
            continue
        # Choose delta fraction:
        frac = rng.random() * 0.2  # up to 20%
        delta = a[i] * frac
        if delta <= 0:
            continue
        # Compute new convolution
        # Using copy of C: 
        Cnew = C.copy()
        # shift contributions
        Cnew[i:i+n] -= 2*delta*a
        Cnew[j:j+n] += 2*delta*a
        # delta^2 spikes:
        two_i = 2*i
        two_j = 2*j
        i_j = i + j
        Cnew[two_i] += delta*delta
        Cnew[two_j] += delta*delta
        Cnew[i_j] -= 2*delta*delta
        # Compute new max
        new_max = Cnew.max()
        # Acceptance
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob = np.exp(-(new_max - cur_max)/T)
            if rng.random() < prob:
                accept = True
        if accept:
            # update a and C
            a[i] -= delta
            a[j] += delta
            # Clip small negatives to zero
            if a[i] < 1e-15:
                a[i] = 0.0
            # re-normalize sum (should remain same algebraically)
            # But due to floating errors, adjust:
            cur_sum = a.sum()
            S = np.sqrt(2.0 * n)
            if abs(cur_sum - S) > 1e-9:
                a *= (S / cur_sum)
                # Recompute C from scratch to avoid drift? Could re-evaluate after certain steps.
                C = np.convolve(a, a)
                cur_max = C.max()

                cur_max = new_max

"""
Propose a step‑function sequence with a low self‑convolution ∞‑norm.

Algorithm
---------
1. **Initialisation** – generate a non‑negative vector `a` of length `n`
   (default 8000) whose L¹‑norm equals `sqrt(2·n)`.  If a global
   `height_sequence_1` is available we reuse (and possibly truncate/pad)
   it, otherwise we draw from a Dirichlet distribution.

2. **Phase 1 – multiplicative sub‑gradient descent**  
   Repeatedly minimise the current maximal entry of the self‑convolution
   `a*a` using the sub‑gradient `g_j = 2·a_{k‑j}` (or `a_j` on the
   diagonal) where `k` is the index attaining the maximum.  A
   multiplicative update `a ← a·exp(−η·g)` preserves non‑negativity; after
   each step the vector is re‑normalised to keep the L¹‑norm fixed.
   A modest decay of the step size and occasional jitter help escape
   plateaus.

3. **Phase 2 – simulated annealing on the simplex**  
   Starting from the best vector found in Phase 1 we perform a large number
   of random mass‑transfer moves:
   * pick distinct indices `i, j`
   * move a random fraction of `a[i]` to `a[j]`
   * update the convolution array `C = a*a` using an O(n) incremental
     formula (no FFT per move)
   * accept the move if it reduces the maximal entry, otherwise accept
     with probability `exp(−Δ/T)` where the temperature `T` slowly decays.

   The move keeps the total sum exactly constant, and the convolution is
   kept up‑to‑date so the max can be obtained in O(1).  Periodically we
   recompute `C` from scratch with an FFT to avoid numerical drift.

Both phases respect the 1000 s budget; the best vector seen is printed
and returned as a plain Python list.  The external evaluation
`evaluate_sequence` equals the maximal convolution entry because of the
normalisation, so the internal “max‑convolution” value is a perfect proxy
for the final score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution `a*a` using an FFT (real‑valued)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_max_fft(a: np.ndarray) -> tuple[float, int, np.ndarray]:
    """
    Compute the full convolution `a*a` via FFT, return (max_value, max_index,
    convolution_array).
    """
    conv = _conv_full_fft(a)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx, conv


def _gradient_step(a: np.ndarray, eta: float) -> np.ndarray:
    """
    Perform a single multiplicative sub‑gradient step aimed at reducing the
    maximal entry of the self‑convolution.  `eta` is the step size.
    The vector is re‑normalised to L1‑norm sqrt(2·n).
    """
    n = a.shape[0]
    _, k_max, _ = _conv_max_fft(a)

    # Build sub‑gradient g
    idx = np.arange(n)
    partner = k_max - idx
    valid = (partner >= 0) & (partner < n)

    g = np.zeros_like(a)
    g[valid] = 2.0 * a[partner[valid]]

    # diagonal contribution (i == j) when k_max is even
    if k_max % 2 == 0:
        d = k_max // 2
        if 0 <= d < n:
            g[d] = a[d]

    # multiplicative update
    a_new = a * np.exp(-eta * g)

    # re‑normalise to sqrt(2·n)
    S = np.sqrt(2.0 * n)
    total = a_new.sum()
    if total == 0.0:
        a_new = np.full(n, S / n)
    else:
        a_new *= (S / total)
    return a_new


def _initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce an initial non‑negative sequence of length `n` with
    L1‑norm sqrt(2·n).  50 % of the time we start from the global
    `height_sequence_1` (if present), otherwise from a Dirichlet draw.
    """
    S = np.sqrt(2.0 * n)

    # Try to reuse the existing construction if possible
    if 'height_sequence_1' in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=float)
            if base.size >= n:
                a = base[:n].copy()
            else:
                a = np.pad(base, (0, n - base.size), constant_values=0.0)
            # Normalise
            a = a / a.sum() * S
            return a
        except Exception:
            pass

    # fallback: Dirichlet (all ones) -> uniform on the simplex
    a = rng.random(n)
    a = a / a.sum() * S
    return a


# ----------------------------------------------------------------------
# Phase 2 – simulated annealing (mass‑transfer moves)
# ----------------------------------------------------------------------


def _annealing(
    a: np.ndarray,
    C: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Perform simulated annealing on `a` while maintaining its convolution
    array `C`.  Returns the best sequence seen and the corresponding maximal
    convolution entry.
    """
    n = a.shape[0]
    conv_len = C.shape[0]

    # current objective
    cur_max = float(C.max())
    best_a = a.copy()
    best_val = cur_max

    # temperature schedule
    T = cur_max * 0.1  # start with ~10 % of the current value
    T_decay = 0.99995

    # move size control
    max_frac = 0.2  # at most 20 % of the source mass may be moved

    # bookkeeping
    it = 0
    recompute_every = 2000  # recompute C from scratch to avoid drift

    while time.time() < deadline:
        # pick two distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        if a[i] <= 1e-14:
            continue

        # propose moving a random fraction of a[i] to a[j]
        frac = rng.random() * max_frac
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        # ---- incremental update of the convolution ----
        C_new = C.copy()

        # linear contributions: 2*delta * (shift_j - shift_i)
        # shift_i adds at indices i .. i+n-1
        # shift_j adds at indices j .. j+n-1
        C_new[i : i + n] -= 2.0 * delta * a
        C_new[j : j + n] += 2.0 * delta * a

        # quadratic delta^2 terms
        ii = 2 * i
        jj = 2 * j
        ij = i + j
        C_new[ii] += delta * delta
        C_new[jj] += delta * delta
        C_new[ij] -= 2.0 * delta * delta

        new_max = float(C_new.max())

        # Acceptance test (Metropolis)
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            # probability to accept a worsening move
            prob = np.exp(-(new_max - cur_max) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            # Apply the move to the state
            a[i] -= delta
            a[j] += delta

            # Guard against tiny negative artefacts
            if a[i] < 0.0:
                a[i] = 0.0

            # Update current convolution and objective
            C = C_new
            cur_max = new_max

            # Keep the globally best sequence
            if cur_max < best_val:
                best_val = cur_max
                best_a = a.copy()
                # occasional external evaluation for logging (optional)
                # ext = evaluate_sequence(best_a.tolist())
                # print(f"[anneal‑improve] t={time.time():.1f}s  max={best_val:.6f}")

        # temperature decay
        T *= T_decay
        it += 1

        # periodic full recomputation to curb accumulated round‑off
        if it % recompute_every == 0:
            C = _conv_full_fft(a)
            cur_max = float(C.max())
            if cur_max < best_val:
                best_val = cur_max
                best_a = a.copy()

    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights whose `evaluate_sequence` value
    should be below the current bound of 1.503.  The routine runs for at most
    ``budget_s`` seconds, prints progress, and internally combines a
    projected multiplicative gradient descent with simulated annealing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0  # small safety margin

    # ------------------------------------------------------------------
    # Choose a problem size
    # ------------------------------------------------------------------
    # Larger n gives more flexibility but costs more per iteration.
    # Empirically n in the range 6000–10000 works well within the budget.
    n = 8000
    if 'height_sequence_1' in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=float)
            if base.size >= 5000:
                # Use the longer of the known construction and our default
                n = max(base.size, n)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Phase 1 – multiplicative sub‑gradient descent
    # ------------------------------------------------------------------
    a = _initial_sequence(n, rng)

    # best seen so far (internal objective = maximal convolution entry)
    best_a = a.copy()
    best_val = float(_conv_max_fft(a)[0])

    print(f"[init] length={n}, initial max={best_val:.6f}")

    # time allocation: ~30 % of the budget for phase 1
    phase1_deadline = time.time() + 0.30 * budget_s

    eta = 0.5          # initial step size
    decay = 0.9999
    jitter_prob = 0.02
    while time.time() < phase1_deadline:
        a = _gradient_step(a, eta)

        cur_max, _, _ = _conv_max_fft(a)
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[grad‑improve] time={time.time():.1f}s  max={best_val:.6f}")

        eta *= decay

        # occasional jitter to escape flat regions
        if rng.random() < jitter_prob:
            noise = rng.standard_normal(n) * a.mean() * 0.01
            a = np.clip(a + noise, 0.0, None)
            a *= np.sqrt(2.0 * n) / a.sum()

    # ------------------------------------------------------------------
    # Phase 2 – simulated annealing on the simplex
    # ------------------------------------------------------------------

    C = _conv_full_fft(best_a)

    best_a, best_val = _annealing(best_a, C, anneal_deadline, rng)

    print(f"[final] best max={best_val:.6f}  length={len(best_a)}")
    # The external evaluator will compute the true score; we return the raw list.
    return best_a.tolist()
