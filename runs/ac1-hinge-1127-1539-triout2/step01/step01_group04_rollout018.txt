import time, math, numpy as np

def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0

    best_seq = None
    best_score = float('inf')

    # candidate lengths (sorted)
    lengths = [10, 12, 14, 16, 18, 20, 30, 40, 50, 80, 100, 150, 200, 300, 500, 800, 1200, 2000, 4000, 6000]
    # optionally restrict to within time.

    for n in lengths:
        if time.time() >= deadline:
            break
        # compute target sum
        target_sum = math.sqrt(2.0 * n)

        # init with sqrt-diff shape
        idx = np.arange(n, dtype=np.float64)
        a = np.sqrt(idx + 1.0) - np.sqrt(idx)
        a = a * (target_sum / a.sum())   # normalized

        # optimization parameters
        L = 1 << ((2 * n - 1).bit_length())
        # Adam
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        gamma = 1.0
        gamma_max = 2000.0
        gamma_mul = 1.002  # increase per iteration
        # iteration loop
        it = 0
        max_iters = 20000  # cap
        while time.time() < deadline and it < max_iters:
            # compute conv via FFT
            FA = np.fft.rfft(a, n=L)
            conv = np.fft.irfft(FA * FA, n=L)[: 2*n-1]
            cur_score = float(conv.max())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"n={n} iter={it} new best {best_score:.6f}")

            # compute softmax weights
            maxc = conv.max()
            # maintain numeric stability
            exp_term = np.exp(gamma * (conv - maxc))
            w = exp_term / exp_term.sum()
            # correlation via FFT (linear)
            FW = np.fft.rfft(w, n=L)
            # correlation using conj(FW)
            corr_complex = np.fft.irfft(FA * np.conj(FW), n=L)
            # extract gradient indices
            # Linear correlation length 2n-1 goes from -(n-1) to (n-1)
            # index offset = n-1
            offset = n - 1
            grad = 2.0 * corr_complex[offset:offset + n]   # length n
            # Adam update
            it_one = it + 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** it_one)
            v_hat = v / (1.0 - beta2 ** it_one)
            step = 0.1 / (np.sqrt(v_hat) + eps)   # base lr 0.1 adapt
            a = a - step * m_hat

            # projection to non-negative and renormalize
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s == 0.0:
                a = np.full(n, target_sum / n, dtype=np.float64)
            else:
                a *= (target_sum / s)

            # increase gamma slowly
            if gamma < gamma_max:
                gamma = min(gamma * gamma_mul, gamma_max)

            # jitter every 200 it
            if it % 200 == 0 and it > 0:
                sigma = target_sum * 1e-4 / math.sqrt(n)
                noise = np.random.randn(n) * sigma
                a = np.maximum(a + noise, 0.0)
                a *= (target_sum / a.sum())

            it += 1

        # after finishing this n, optionally compute final score of a and update best.

    if best_seq is None:
        # fallback: uniform
        n = 100
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)

    print(f"Finished search: best score {best_score:.6f}")
    return best_seq.tolist()

"""
Propose a step‑function (non‑negative list) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The score is invariant under scaling of the vector, therefore we can keep the
L1‑norm fixed to √(2·n) at every iteration.  The problem is a
min‑max optimisation of a self‑convolution.  We use a smooth approximation of
the max (soft‑max) and perform Adam‑style updates on the vector while staying
in the simplex (non‑negative, fixed L1‑norm).  The optimisation is carried out
for several problem sizes – small sizes give a lower bound than the
asymptotic √−difference construction (π/2 ≈ 1.571) – and the best sequence
found across all sizes is returned.

The algorithm works completely within the supplied time budget (default 1000 s)
and makes no use of file or network I/O.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper routines – all are top‑level (no closures) as required.
# ----------------------------------------------------------------------


def _self_conv_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using an FFT of minimal power‑of‑two length."""
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())          # next power of two ≥ 2n‑1
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors x and y via FFT."""
    L = 1 << ((x.size + y.size - 1).bit_length())
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(fx * fy, n=L)[: x.size + y.size - 1]
    return conv


def _softmax_weights(conv: np.ndarray, gamma: float) -> np.ndarray:
    """Return a probability vector proportional to exp(gamma*conv)."""
    # subtract the maximum to keep the exponent bounded
    m = conv.max()
    ex = np.exp(gamma * (conv - m))
    return ex / ex.sum()


def _gradient_via_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed max w.r.t. a.
    The formula is   grad_i = 2 * Σ_k w_k * a_{k-i}
    which is a linear correlation of a with the reversed weight vector.
    """
    n = a.size
    # reverse w for correlation
    w_rev = w[::-1]
    # linear correlation = convolution of a with w_rev
    full = _conv_fft(a, w_rev)                      # length = n + (2n‑1) – 1 = 3n‑2
    # the needed slice corresponds to indices 2n‑2‑i  (i = 0…n‑1)
    start = 2 * n - 2
    grad = 2.0 * full[start - np.arange(n)]
    return grad


def _project_to_simplex(a: np.ndarray, target_sum: float) -> np.ndarray:
    """Project a onto the non‑negative orthant and renormalise to target_sum."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:                                 # fall back to a uniform vector
        a = np.full_like(a, target_sum / a.size)
    else:
        a *= (target_sum / s)
    return a


def _adam_step(a: np.ndarray,
               grad: np.ndarray,
               m: np.ndarray,
               v: np.ndarray,
               it: int,
               lr: float = 0.1,
               beta1: float = 0.9,
               beta2: float = 0.999,
               eps: float = 1e-8) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """Perform one Adam update on vector a (in place) and return new (a,m,v)."""
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)
    m_hat = m / (1.0 - beta1 ** (it + 1))
    v_hat = v / (1.0 - beta2 ** (it + 1))
    step = lr / (np.sqrt(v_hat) + eps)
    a = a - step * m_hat
    return a, m, v


# ----------------------------------------------------------------------
# Main entry point required by the runner.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function with a low evaluation score.
    The routine explores several problem sizes, runs an Adam optimiser with a
    soft‑max approximation of the infinity norm and keeps the best sequence
    found before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0   # leave a small safety margin

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # Problem sizes that are cheap enough to explore thoroughly.
    # Very small sizes often beat the asymptotic bound π/2 ≈ 1.571.
    # ------------------------------------------------------------------
    size_pool = [8, 9, 10, 11, 12, 13, 14, 15, 16, 18,
                 20, 22, 25, 30, 35, 40, 50, 60, 80, 100,
                 150, 200, 300, 500, 800, 1200, 2000, 4000]

    for n in size_pool:
        # stop immediately if we run out of time
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)        # L1‑norm that makes the objective equal to max(conv)

        # ---- initialisation -------------------------------------------------
        idx = np.arange(n, dtype=np.float64)
        a = np.sqrt(idx + 1.0) - np.sqrt(idx)   # √(i+1)–√i  (the classic start)
        a = a * (target_sum / a.sum())          # normalise to the required L1‑norm

        # Adam state
        m = np.zeros_like(a)
        v = np.zeros_like(a)

        # schedule for the soft‑max temperature
        gamma = 1.0
        gamma_max = 2500.0
        gamma_mul = 1.002   # slowly increase, making the approximation tighter

        # maximum number of iterations for the current size
        max_iters = 25000 if n <= 150 else 15000
        it = 0
        while it < max_iters and time.time() < deadline:
            # ----- self‑convolution (exact) ---------------------------------
            conv = _self_conv_fft(a)
            cur_score = float(conv.max())          # this is exactly evaluate_sequence(a)

            # keep the overall best sequence seen so far
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[n={n}] iter={it:5d}  new best = {best_score:.6f}")

            # ----- smooth max (soft‑max) ------------------------------------
            w = _softmax_weights(conv, gamma)      # length = 2n‑1, sums to 1

            # ----- gradient of the smooth objective -------------------------
            grad = _gradient_via_correlation(a, w)

            # ----- Adam step -------------------------------------------------
            a, m, v = _adam_step(a, grad, m, v, it)

            # ----- projection back onto feasible set ------------------------
            a = _project_to_simplex(a, target_sum)

            # ----- increase the temperature slowly --------------------------
            if gamma < gamma_max:
                gamma = min(gamma * gamma_mul, gamma_max)

            # ----- occasional jitter to escape shallow plateaus -------------
            if it % 250 == 0 and it > 0:
                sigma = target_sum * 5e-4 / math.sqrt(n)
                noise = np.random.randn(n) * sigma
                a = _project_to_simplex(a + noise, target_sum)

            it += 1

        # end of optimisation for this size
        print(f"Finished size {n:4d} after {it:5d} iters – best score so far {best_score:.6f}")

    # ----------------------------------------------------------------------
    # If nothing was found (should never happen) fall back to a uniform vector.
    # ----------------------------------------------------------------------
    if best_seq is None:
        n = 100
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))   # type: ignore[name-defined]

    print(f"Search completed – final best score {best_score:.6f}")
    # return a plain Python list as required by the evaluator
    return best_seq.tolist()
