def project_onto_simplex(v):
    # ensure nonnegative sum constraint.
    # works for arbitrary vector.
    if v.sum() == 0:
        # uniform
        return np.full_like(v, 1.0/v.shape[0])
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    # find rho as max i such that u_i > cssv_i / i
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def gradient_step(x, alpha, eta):

    y = np.convolve(x, x)  # length 2n-1

    # Softmax: w = exp(alpha*y) / sum(exp(alpha*y)), using stable trick subtract max
    max_y = np.max(alpha*y)
    exps = np.exp(alpha*y - max_y)
    w = exps / np.sum(exps)

    # Compute gradient via correlation.
    # Equivalent: g = 2 * np.convolve(x[::-1], w, mode='full')[n-1:n-1+len(x)]
    rev = x[::-1]
    conv_rev = np.convolve(rev, w, mode='full')
    n = len(x)
    g = 2 * conv_rev[n-1:n-1+n]  # shape (n,)

    # update step
    x_new = x - eta * g

    # project onto simplex
    x_proj = project_onto_simplex(x_new)
    return x_proj

            x = 0.9*x + 0.1*noise
            x = x / x.sum()

    print('Final best score', best_score, 'length', len(best_seq))
    return best_seq

"""
Propose a step‐function (non‑negative heights) that minimises the given
evaluate_sequence() objective.

Strategy
--------
1.  Use a one‑parameter family known to be (asymptotically) optimal for the
    ℓ∞‑norm of a self‑convolution:
        a_i = (i + σ)^{‑p} ,   p∈(0.5,1) .
    For p>½ the interior part of the convolution yields the bound
        2·(1‑p)²·B(1‑p,1‑p)            (B is the beta function),
    which is < 1.5 already for p≈0.6.
    The value a₀² may dominate unless σ is chosen large enough.
    The optimal σ scales like  σ ∼ n^{(2p‑1)/(2p)} .
2.  Perform a small grid search over p and a few scaling factors around the
    theoretical σ.  Each candidate is evaluated with evaluate_sequence().
3.  The best candidate from the sweep is further refined with a few
    mirror‑descent steps (the same update used in the reference code).  This
    removes any remaining “spikes” in the convolution.
4.  The routine respects the 1000 s budget and returns the best sequence
    found.
"""

import time
import numpy as np


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Guarantees non‑negativity and sum(v)=1.
    """
    if v.sum() == 0:
        # uniform distribution as fallback
        return np.full_like(v, 1.0 / v.shape[0])

    # standard algorithm: sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    # find the last index where u_i > cssv_i / i
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – fall back to uniform
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _powerlaw_sequence(p: float, sigma: float, n: int) -> np.ndarray:
    """
    Return the (un‑normalised) power‑law vector a_i = (i + sigma)^{-p}
    of length n.
    """
    i = np.arange(n, dtype=np.float64)
    a = (i + sigma) ** (-p)
    return a


def _mirror_descent_step(x: np.ndarray, eta: float) -> np.ndarray:
    """
    One mirror‑descent iteration for the ℓ∞‑norm of the auto‑convolution.
    x must be non‑negative and sum to one.
    """
    n = x.shape[0]
    conv = np.convolve(x, x)
    k = int(conv.argmax())
    g = np.zeros_like(x)

    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    idx = np.arange(i_min, i_max + 1)
    jdx = k - idx
    g[idx] = x[jdx]                     # sub‑gradient (factor 2 absorbed in η)

    y = x * np.exp(-eta * g)
    if y.sum() == 0.0:
        y = np.abs(np.random.randn(n))
    y = _project_onto_simplex(y)
    return y


def propose_candidate(seed: int = 42, budget_s: float = 1000.0):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The search respects the 1000 s time limit and returns the best sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Parameter sweep over the power‑law family a_i = (i+σ)^(-p)
    # ------------------------------------------------------------------
    n = 2000                     # moderate length – fast convolution, still
                                 # large enough for asymptotics.
    best_score = float('inf')
    best_seq = None

    # p in (0.55, 0.90) – values >0.5 give the theoretical improvement.
    p_vals = np.linspace(0.55, 0.90, 18)

    # Scaling factors around the asymptotically optimal σ.
    factor_set = [0.2, 0.5, 0.8, 1.0, 2.0, 5.0, 10.0]

    print("=== Phase 0 : power‑law sweep ===")
    for p in p_vals:
        if time.time() >= deadline:
            break
        # theoretical scaling σ ≈ n^{(2p‑1)/(2p)}
        sigma_base = n ** ((2.0 * p - 1.0) / (2.0 * p))
        for f in factor_set:
            if time.time() >= deadline:
                break
            sigma = sigma_base * f
            # protect against sigma being zero
            sigma = max(sigma, 1e-6)

            seq = _powerlaw_sequence(p, sigma, n)
            try:
                score = evaluate_sequence(seq.tolist())
            except Exception:
                continue

            if np.isfinite(score) and score < best_score:
                best_score = float(score)
                best_seq = seq.tolist()
                print(f'new best | p={p:.3f} sigma={sigma:.3f} → {best_score:.6f}')

            # early stop if we already beat the target comfortably
            if best_score < 1.3:
                break
        if best_score < 1.3:
            break

    # ------------------------------------------------------------------
    # If the sweep failed (unlikely) fall back to a uniform vector.
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 500
        best_seq = [1.0] * n_fallback
        best_score = evaluate_sequence(best_seq)
        print('fallback uniform sequence, score =', best_score)

    # ------------------------------------------------------------------
    # Phase 1 – small mirror‑descent refinement (helps escape tiny local bumps)
    # ------------------------------------------------------------------
    x = np.array(best_seq, dtype=np.float64)
    # Normalise to a probability simplex – the objective is scale invariant.
    x = _project_onto_simplex(x)

    iteration = 0
    print("=== Phase 1 : mirror‑descent refinement ===")
    while time.time() < deadline:
        iteration += 1
        eta = 0.25 / np.sqrt(iteration)          # decreasing step size
        x = _mirror_descent_step(x, eta)

        # occasional jitter to avoid stagnation
        if np.random.rand() < 0.02:
            idx = np.random.randint(x.shape[0])
            delta = (np.random.rand() - 0.5) * 0.02 * x[idx]
            x[idx] = max(0.0, x[idx] + delta)
            x = _project_onto_simplex(x)

        try:
            cur_score = evaluate_sequence(x.tolist())
        except Exception:
            continue

        if cur_score < best_score:
            best_score = float(cur_score)
            best_seq = x.tolist()
            print(f'iteration {iteration:5d} -> improved score = {best_score:.6f}')
            if best_score < 1.3:               # target comfortably passed
                break

    print('final best score:', best_score, 'length:', len(best_seq))
    return best_seq
