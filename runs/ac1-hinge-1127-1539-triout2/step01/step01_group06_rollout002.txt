def _anneal_search(a, deadline, init_temp=0.01, cooling=0.995):
    cur = a.copy()
    cur_score = _score(cur)
    best = cur.copy()
    best_score = cur_score
    temp = init_temp
    n = a.size
    rng = np.random.default_rng()
    while time.time() < deadline:
        i = rng.integers(0,n)
        j = rng.integers(0,n)
        if i == j: continue
        if cur[i] < 1e-12: continue
        frac = rng.random() * 0.3  # bigger moves
        delta = cur[i] * frac
        cur[i] -= delta
        cur[j] += delta
        new_score = _score(cur)
        # Accept if better or with probability exp(-(new_score - cur_score)/temp)
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # probability
            prob = np.exp(-(new_score - cur_score) / temp)
            if rng.random() < prob:
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
                print(f"[{time.time():.1f}]  anneal improved score = {best_score:.6f}")
        else:
            # revert
            cur[j] -= delta
            cur[i] += delta
        # temperature cooling
        temp *= cooling
        if temp < 1e-9:
            temp = 1e-9
    return best, best_score
```

"""
Propose a step‑function (non‑negative heights) that minimizes the
evaluation function

    2·n·max(conv(a,a)) / (sum(a)^2)

The scaling of the vector cancels out, so we can work with the
ℓ1‑norm fixed to √(2·n) and simply minimise the maximum convolution.
The algorithm consists of three phases:

1. **Exploratory grid search** – evaluate families of analytic shapes
   (power‑law a_i ∝ (i+δ)^‑p and exponential a_i ∝ e^{-λ i}) for several
   lengths n.  The grid covers a range of exponents, offsets and decay rates,
   keeping the total ℓ1‑norm fixed.  The best candidate from this phase is
   kept.

2. **Simulated annealing** – starting from the best grid candidate,
   repeatedly move a random fraction of mass from one position to another.
   Moves that improve the score are always accepted; worsening moves are
   accepted with a probability decreasing with a temperature that is slowly
   cooled.  This explores the neighbourhood of the grid optimum and can
   escape local minima.

3. **Deterministic hill‑climbing** – a final short pass that only keeps
   strictly improving mass‑transfer moves, polishing the solution.

The whole routine respects the supplied time budget (default 1000 s) and
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def _max_convolution(a: np.ndarray) -> float:
    """
    Compute max of the linear convolution a * a using an FFT of power‑of‑two
    length (fast and numerically stable).
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """
    Evaluation function for a normalised vector.
    The vector must be non‑negative; its ℓ1‑norm is assumed arbitrary,
    because the scaling cancels out.  The function therefore returns the
    maximum convolution value.
    """
    n = a.size
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)   # equal to max_b when sum = √(2n)


def _normalise(a: np.ndarray) -> np.ndarray:
    """Scale a to satisfy Σa = √(2·n)."""
    n = a.size
    target = np.sqrt(2.0 * n)
    scale = target / a.sum()
    return a * scale


def _grid_search(best_seq, best_score, deadline):
    """
    Exhaustively evaluate a few analytical families.
    Returns possibly improved (best_seq, best_score).
    """
    lengths = [1500, 2500, 3500, 5000]               # modest sizes
    p_vals = np.linspace(0.35, 0.68, 14)            # power‑law exponents
    # Offsets: linearly spaced near zero and a few larger values
    delta_vals_small = np.linspace(0.0, 5.0, 11)
    delta_vals_big = np.logspace(np.log10(6.0), np.log10(200.0), 6)
    delta_vals = np.concatenate([delta_vals_small, delta_vals_big])
    lam_vals = np.linspace(0.0005, 0.01, 20)        # exponential rates

    for n in lengths:
        if time.time() > deadline:
            break
        sqrt2n = np.sqrt(2.0 * n)
        i = np.arange(1, n + 1, dtype=np.float64)   # 1‑based for power law
        # ----- power‑law family -----
        for p in p_vals:
            for delta in delta_vals:
                if time.time() > deadline:
                    break
                a = (i + delta) ** (-p)
                a = _normalise(a)
                cur = _score(a)
                if cur < best_score:
                    best_score = cur
                    best_seq = a.copy()
                    print(f"[grid] n={n} p={p:.4f} δ={delta:.2f} → {cur:.6f}")
        # ----- exponential family -----
        for lam in lam_vals:
            if time.time() > deadline:
                break
            # use zero‑based index for exponentials
            k = np.arange(n, dtype=np.float64)
            a = np.exp(-lam * k)
            a = _normalise(a)
            cur = _score(a)
            if cur < best_score:
                best_score = cur
                best_seq = a.copy()
                print(f"[grid] n={n} λ={lam:.5f} → {cur:.6f}")

    return best_seq, best_score


def _anneal_search(a: np.ndarray, deadline) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style mass‑transfer search.
    """
    cur = a.copy()
    cur_score = _score(cur)
    best = cur.copy()
    best_score = cur_score

    n = cur.size
    rng = np.random.default_rng()
    temp = 0.02                # initial temperature
    decay = 0.9995             # cooling factor per iteration
    max_frac = 0.4             # move up to 40 % of source entry

    while time.time() < deadline:
        i = rng.integers(0, n)
        if cur[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        frac = rng.random() * max_frac
        delta = cur[i] * frac
        cur[i] -= delta
        cur[j] += delta

        new_score = _score(cur)

        # Accept if better, otherwise probabilistically
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # Metropolis acceptance
            prob = np.exp(-(new_score - cur_score) / temp)
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
                print(f"[anneal] improved score = {best_score:.6f}")
        else:
            # revert
            cur[j] -= delta
            cur[i] += delta

        temp = max(temp * decay, 1e-9)

    return best, best_score


def _hill_climb(a: np.ndarray, deadline) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climbing: moves are kept only if they strictly
    improve the score.
    """
    cur = a.copy()
    cur_score = _score(cur)
    best = cur.copy()
    best_score = cur_score

    n = cur.size
    rng = np.random.default_rng()
    max_frac = 0.2

    while time.time() < deadline:
        i = rng.integers(0, n)
        if cur[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        frac = rng.random() * max_frac
        delta = cur[i] * frac
        cur[i] -= delta
        cur[j] += delta

        new_score = _score(cur)

        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
                print(f"[hill] improved score = {best_score:.6f}")
        else:
            # reject
            cur[j] -= delta
            cur[i] += delta

    return best, best_score
