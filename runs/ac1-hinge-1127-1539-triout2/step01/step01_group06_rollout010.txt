"""
LP‑guided stochastic optimizer for step‑function bound.

import time
import numpy as np
import scipy.optimize as opt
import scipy.linalg as la

def _max_convolution(a):
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())

def _score(a):
    n = a.size
    maxb = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * maxb / (s * s)

def _initial_sequence(p, n):
    a = (np.arange(1, n+1, dtype=np.float64) ** (-p))
    a *= np.sqrt(2.0 * n) / a.sum()
    return a

def _conv_toeplitz_matrix(f):
    n = f.shape[0]
    c = np.concatenate([f, np.zeros(n-1, dtype=f.dtype)])
    r = np.concatenate([f[:1], np.zeros(n-1, dtype=f.dtype)])
    return la.toeplitz(c, r)

def _lp_max_sum(f, M):
    A = _conv_toeplitz_matrix(f)
    n = f.size
    b = np.full(A.shape[0], M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n
    # linprog prefers dense A; this works for moderate n
    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        # Clip tiny negatives caused by numerical noise
        return np.maximum(res.x, 0.0)
    return None

"""
LP‑guided stochastic optimizer for the step‑function bound.

The routine mixes two ideas:
1.  A linear program maximises the ℓ¹‑norm of a vector *g* that satisfies
    the current worst‑case convolution bound  M = max(conv(f,f)).
    The LP improves the total mass while keeping every entry of conv(f,g)
    ≤ M, i.e. it finds a direction that can increase the denominator of the
    objective without increasing the numerator.
2.  The new direction *g* (renormalised to the fixed ℓ¹‑norm √(2 n)) is
    blended into the current candidate *f* with a modest step size.  This
    strictly reduces the max‑convolution in practice.  After a few LP
    iterations a short random hill‑climb (mass‑transfer moves) is applied
    for fine‑tuning.

The whole search respects the supplied time budget (default 1000 s) and
returns a list of non‑negative heights that minimises ``evaluate_sequence``.
"""

import time
import numpy as np
import scipy.optimize as opt
import scipy.linalg as la

# ----------------------------------------------------------------------
# Helper functions (pure, no closures)
# ----------------------------------------------------------------------


def _max_convolution(a: np.ndarray) -> float:
    """Maximum value of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """Evaluation function used inside the optimiser (identical to
    evaluate_sequence up to the safety clamping)."""
    n = a.size
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _initial_sequence(p: float, n: int) -> np.ndarray:
    """Power‑law start a_i ∝ (i+1)^{‑p}, ℓ¹‑normalised to √(2 n)."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= np.sqrt(2.0 * n) / a.sum()
    return a


def _conv_toeplitz_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix A such that conv(f,g) = A @ g.
    Row i contains the coefficients f_{i‑j} for all admissible j.
    """
    n = f.shape[0]
    c = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    r = np.concatenate([f[:1], np.zeros(n - 1, dtype=f.dtype)])
    return la.toeplitz(c, r)


def _lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max Σ g_i   s.t.   g ≥ 0   and   A @ g ≤ M,
    where A is the convolution matrix of *f*.
    Returns the optimal vector g (or None if the solver fails).
    """
    A = _conv_toeplitz_matrix(f)                # dense (2n‑1)×n matrix
    n = f.size
    b = np.full(A.shape[0], M, dtype=np.float64)  # right‑hand side
    c = -np.ones(n, dtype=np.float64)             # minimise –Σg_i  ↔ maximise Σg_i
    bounds = [(0, None)] * n

    # Highs is fast and robust for dense LPs of this size.
    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        # Clip tiny negative artefacts caused by numerical noise.
        return np.maximum(res.x, 0.0)
    return None


def _lp_improve(a_start: np.ndarray,
                deadline: float,
                max_iters: int = 10,
                step: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Repeatedly apply the LP step to the current vector *a*.
    Each iteration:
        • compute the current worst‑case bound M,
        • solve the LP that maximises the ℓ¹‑norm while keeping conv(a,g) ≤ M,
        • rescale the LP solution to the fixed ℓ¹‑norm √(2 n),
        • blend it into *a* with a convex coefficient *step*.
    The best vector seen during the loop is returned.
    """
    a = a_start.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        M = _max_convolution(a)

        g_opt = _lp_max_sum(a, M)
        if g_opt is None:
            break

        sum_g = g_opt.sum()
        # If the LP cannot increase the ℓ¹‑norm we have reached a stationary point.
        if sum_g <= target_sum * (1 - 1e-9):
            break

        # Rescale to keep the ℓ¹‑norm fixed (required by the objective).
        scale = target_sum / sum_g
        g = g_opt * scale

        # Convex combination (step is deliberately modest).
        a_new = (1.0 - step) * a + step * g
        new_val = _score(a_new)

        if new_val < best_val:
            best_val = new_val
            best_a = a_new.copy()
            a = a_new                     # continue from the improved point
            print(f"[{time.time():.1f}] LP improve iter {it}, score={best_val:.6f}")
        else:
            # No improvement – shrink the step size to try a gentler update.
            step = max(step * 0.5, 0.05)

    return best_a, best_val


def _local_search(a_start: np.ndarray,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing that preserves the ℓ¹‑norm.
    Randomly moves a fraction of the mass from one index to another.
    Only strictly improving moves are kept.
    """
    a = a_start.copy()
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    cur_val = best_val

    max_frac = 0.12               # maximal percentage of a[i] to relocate
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        frac = rng.random() * max_frac
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        new_val = _score(a)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}] local search improved = {best_val:.6f}")
        else:
            # revert the move
            a[j] -= delta
            a[i] += delta

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the challenge
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises the
    competition objective.  The algorithm alternates between an LP‑guided
    redistribution step and a short stochastic hill‑climber, respecting the
    overall time budget (default 1000 s).  The best sequence found is
    returned as a plain Python list.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin for the harness

    best_seq = None
    best_score = np.inf

    # Different problem sizes give a trade‑off between quality and speed.
    lengths = [800, 1200, 1800, 2500]         # keep under ~10 000 to stay fast
    exponents = [0.44, 0.47, 0.5, 0.53, 0.56]

    for n in lengths:
        for p in exponents:
            if time.time() >= deadline:
                break

            # ----- start from a power‑law profile ---------------------------------
            a = _initial_sequence(p, n)
            cur_score = _score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"Start n={n}, p={p:.3f}, score={cur_score:.6f}")

            # ----- LP‑guided improvement -----------------------------------------
            lp_deadline = min(deadline, time.time() + 20.0)   # at most 20 s per (n,p)
            a_lp, lp_score = _lp_improve(a, lp_deadline, max_iters=8, step=0.3)
            if lp_score < best_score:
                best_score = lp_score
                best_seq = a_lp.copy()
                print(f"After LP n={n}, p={p:.3f}, best score={lp_score:.6f}")

            # ----- brief stochastic refinement ------------------------------------
            if time.time() < deadline:
                local_deadline = min(deadline, time.time() + 5.0)
                a_loc, loc_score = _local_search(a_lp, local_deadline)
                if loc_score < best_score:
                    best_score = loc_score
                    best_seq = a_loc.copy()
                    print(f"After local n={n}, p={p:.3f}, best score={loc_score:.6f}")

    # final polishing on the overall best candidate (if we still have time)
    if best_seq is not None and time.time() < deadline:
        final_deadline = min(deadline, time.time() + 10.0)
        fin_seq, fin_score = _local_search(best_seq.copy(), final_deadline)
        if fin_score < best_score:
            best_score = fin_score
            best_seq = fin_seq
            print(f"Final polishing improved score={fin_score:.6f}")

    # fallback – should never happen, but ensures a return value
    if best_seq is None:
        best_seq = np.ones(1000, dtype=np.float64)
        best_seq *= np.sqrt(2.0 * best_seq.size) / best_seq.sum()

    # trim trailing zeros (they would be ignored by evaluate_sequence anyway)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    return best_seq.tolist()
