"""
Optimiser for the self‑convolution bound.

The search consists of three phases:

1. **Extensive parametric sweep** – a large collection of symmetric “U‑shaped’’
   families (linear, quadratic, cubic, power‑law, cosine, exponential) is
   explored.  Every candidate is scaled to the normalisation
   ∑a_i = √(2 n) (so that the objective equals the maximal auto‑convolution).

2. **Projected sub‑gradient descent with momentum** – starting from the best
   sweep candidate we repeatedly:
      * compute the self‑convolution by FFT,
      * find the index (or a small active set) attaining the maximum,
      * build a sub‑gradient of the max‑convolution,
      * take a step orthogonal to the all‑ones direction, add momentum,
      * clip, symmetrise and re‑scale to keep the ℓ₁‑norm fixed.
   The step size decays slowly, the active‐set tolerance is gradually enlarged
   and occasional random pair‑wise mass transfers help escape plateaus.

3. **Global restarts** – after a period of stagnation the current iterate is
   jittered and a fresh run is started from a random Dirichlet draw; the
   overall best sequence is kept.

All operations are performed with NumPy and FFT‐based convolutions, which are
much faster than the naïve `np.convolve` used inside `evaluate_sequence`.  The
final candidate is returned as a plain Python list; its quality is verified
once with the official `evaluate_sequence` before returning.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution of a with itself using a power‑of‑two FFT.
    Returns the full convolution of length 2*len(a)-1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector to have ℓ₁‑norm = target.
    Zero entries stay zero; if the whole vector is zero a uniform vector is returned."""
    s = v.sum()
    if s > 0.0:
        return v * (target / s)
    # fallback – uniform distribution
    return np.full_like(v, target / v.shape[0])


def symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry: v[i] = v[-i‑1]."""
    return (v + v[::-1]) * 0.5


def fast_max_conv(a: np.ndarray) -> float:
    """Return the maximal entry of a*a (the objective when ∑a = √(2 n))."""
    conv = conv_self_fft(a)
    return float(conv.max())


def generate_U_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
    exponent: float = 2.0,
) -> np.ndarray:
    """Produce a symmetric “U‑shaped’’ sequence of length n.
    - kind ∈ {linear, quadratic, cubic, power, cosine, exponential}
    - param controls the edge‑enhancement (≥0).
    - exponent is used only for the *power* family (raw = 1+param·d**exponent)."""
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # normalised distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "cubic":
        raw = 1.0 + param * d ** 3
    elif kind == "power":
        raw = 1.0 + param * (d ** exponent)
    elif kind == "cosine":
        # 1‑cos(π d) is 0 at the centre and 2 at the edges
        raw = 1.0 + param * (1.0 - np.cos(math.pi * d))
    elif kind == "exponential":
        raw = np.exp(param * d)
    else:
        raise ValueError(f"unknown shape kind: {kind}")

    raw = np.clip(raw, 0.0, None)
    return project_to_sum(raw, target_sum)


def random_dirichlet(n: int, target_sum: float, alpha: float = 0.5) -> np.ndarray:
    """Draw a random non‑negative vector from Dirichlet(α) and scale to target_sum."""
    # Gamma(α,1) for each component, then normalise.
    x = np.random.gamma(alpha, 1.0, size=n)
    return project_to_sum(x, target_sum)


def subgradient_step(
    a: np.ndarray,
    target_sum: float,
    eta: float,
    momentum: np.ndarray,
    mu: float,
    rel_tol: float,
) -> tuple[np.ndarray, np.ndarray]:
    """
    One sub‑gradient step with momentum.
    Returns the updated vector and the new momentum term.
    """
    n = a.shape[0]
    conv = conv_self_fft(a)
    max_val = conv.max()

    # indices where the convolution is (almost) maximal
    if rel_tol <= 0.0:
        act = np.array([np.argmax(conv)], dtype=np.int64)
    else:
        act = np.where(conv >= max_val * (1.0 - rel_tol))[0]

    # sub‑gradient of the max‑convolution
    g = np.zeros_like(a)
    idx = np.arange(n, dtype=np.int64)
    for k in act:
        j = k - idx
        mask = (j >= 0) & (j < n)
        # contribution a_{k-i} (appears twice in the usual derivative)
        g[idx[mask]] += 2.0 * a[j[mask]]

    # make step orthogonal to the all‑ones direction (preserve sum)
    direction = -g + np.mean(g)                     # sum(direction) == 0

    # momentum update
    new_momentum = mu * momentum + (1.0 - mu) * direction
    a_new = a + eta * new_momentum

    a_new = np.clip(a_new, 0.0, None)
    a_new = symmetrize(a_new)
    a_new = project_to_sum(a_new, target_sum)

    return a_new, new_momentum


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """Transfer a small fraction of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0.0:
        return a
    a[i] -= delta
    a[j] += delta
    a = symmetrize(a)               # keep symmetry
    return project_to_sum(a, target_sum)

            # initialise from a fresh Dirichlet draw
            a = random_dirichlet(n, target_sum, alpha=0.4)
            a = symmetrize(a)               # enforce symmetry from the start
            momentum = np.zeros_like(a)
            eta_r = 0.05 * avg_entry
            rel_tol_r = 0.0
            it_r = 0
            end_r = time.time() + per_restart - 0.5
            while time.time() < end_r:
                a, momentum = subgradient_step(a, target_sum, eta_r, momentum, mu, rel_tol_r)
                it_r += 1
                # cheap random swap every ≈100 steps
                if it_r % 100 == 0:
                    a = random_pair_transfer(a, target_sum, scale=0.003)

                cur = fast_max_conv(a)
                if cur + 1e-12 < best_score:
                    best_score = cur
                    best_seq = a.copy()
                    print(f"Restart {r+1}/{restarts}, iter {it_r:05d} – new best {best_score:.6f}")

                # simple decay
                if it_r % 400 == 0:
                    eta_r *= 0.93
                    rel_tol_r = min(0.06, rel_tol_r + 0.004)

    # ------------------------------------------------------------------
    # Final verification with the official evaluator (optional)
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.2f}] Final verified value = {final_val:.6f}")
    except Exception as exc:
        # should never happen, but we keep the best sequence anyway
        print(f"Verification failed: {exc}")

    # ------------------------------------------------------------------
    # Return the best sequence (plain Python list)
    # ------------------------------------------------------------------
    return best_seq.tolist()
