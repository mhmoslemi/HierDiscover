def cross_correlation_fft(a, w):
    # a length n, w length m (2n-1)
    # returns full correlation length n+m-1.
    L = 1 << ((n + m - 1).bit_length())  # power of 2
    Fa = np.fft.rfft(a, n=L)
    Fw = np.fft.rfft(w, n=L)  # w not reversed
    # correlation: ifft( Fa * conj(Fw) ), yields circular correlation of length L.
    corr = np.fft.irfft(Fa * np.conj(Fw), n=L)
    # extract valid part: need first n+m-1 entries.
    return corr[: n + m - 1]

def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    n = len(height_sequence_1) if defined else 5000
    idx = np.arange(n, dtype=np.float64)
    a = np.sqrt(idx + 1) - np.sqrt(idx)
    target_sum = np.sqrt(2.0 * n)
    a *= target_sum / a.sum()

    best_seq = a.copy()
    best_score = compute_score(a)
    print(f"Initial score: {best_score}")

    # set hyperparameters
    beta0 = 2.0
    beta_max = 80.0
    beta_inc = (beta_max - beta0) / 50000  # increase per iteration
    lr = 0.2  # base LR
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    iteration = 0
    while time.time() < deadline:
        # compute FFT length suitable for this n
        L = 1 << ((3 * n - 2).bit_length())  # constant per n
        # forward FFT of a
        Fa = np.fft.rfft(a, n=L)

        # self-convolution (full)
        conv_full = np.fft.irfft(Fa * Fa, n=L)[: 2 * n - 1]  # length 2n-1

        # compute-score (max conv)
        cur_score = conv_full.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"Iter {iteration} – new best {best_score:.6f}")

        # compute beta schedule
        beta = min(beta_max, beta0 + iteration * beta_inc)

        # compute softmax weights
        # using stable softmax
        scaled = beta * conv_full
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        w = exp_scaled / exp_scaled.sum()

        # FFT of w (zero-padded)
        Fw = np.fft.rfft(w, n=L)

        # correlation (full) using Fa * conj(Fw)
        corr_full = np.fft.irfft(Fa * np.conj(Fw), n=L)[: n + (2 * n - 1) - 1]  # 3n-2

        # extract gradient slice
        start = (2 * n - 2)  # offset
        grad = 2.0 * corr_full[start : start + n]  # shape n

        # Adam update
        iteration += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step

        # projection: nonnegative then rescale sum to target
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0:
            a = np.full(n, target_sum / n)
        else:
            a *= (target_sum / s)

    print(f"Finished after {iteration} iterations – best score {best_score:.6f}")
    return best_seq.tolist()
