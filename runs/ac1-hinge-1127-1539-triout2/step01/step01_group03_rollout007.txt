def _objective(seq: np.ndarray) -> float:
    """Fast evaluation of 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.size
    s = np.sum(seq)

        return np.inf

    conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]

    # power‑law grid
    betas = np.linspace(0.45, 0.55, 41)          # fine grid around 0.5
    offsets = np.linspace(0.0, 10.0, 21)        # fairly wide shift range
    for beta in betas:
        for c in offsets:
            cand = 1.0 / ((idx_base + c) ** beta)
            cand *= target_sum / cand.sum()
            sc = _objective(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    # pure exponential family
    if time.time() < deadline:
        lam_vals = np.linspace(5e-4, 5e-3, 20)
        for lam in lam_vals:
            cand = np.exp(-lam * np.arange(n, dtype=np.float64))
            cand *= target_sum / cand.sum()
            sc = _objective(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
            if time.time() > deadline:
                break

    # random mixtures of two power‑law components
    if time.time() < deadline:
        for _ in range(300):
            beta1 = np.random.uniform(0.4, 0.6)
            c1 = np.random.uniform(0.0, 10.0)
            beta2 = np.random.uniform(0.4, 0.6)
            c2 = np.random.uniform(0.0, 10.0)
            w1 = np.random.rand()
            w2 = 1.0 - w1
            cand = (w1 / ((idx_base + c1) ** beta1) +
                    w2 / ((idx_base + c2) ** beta2))
            cand *= target_sum / cand.sum()
            sc = _objective(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
            if time.time() > deadline:
                break

    # fallback (should never happen)
    if best_seq is None:
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _objective(best_seq)

    # ------------------------------------------------------------------
    # 3. Sub‑gradient descent on the maximal convolution entry.
    # ------------------------------------------------------------------
    a = best_seq.copy()
    cur_score = best_score
    best_global_seq = a.copy()
    best_global_score = cur_score

    # descent hyper‑parameters
    step_factor = 0.2          # controls absolute step size
    no_improve = 0
    iter_cnt = 0
    max_backtrack = 5          # attempts per iteration
    m_fft = 1 << ((2 * n - 1).bit_length())   # fixed FFT length

    while time.time() < deadline:
        # full convolution of the current sequence
        f_fft = np.fft.rfft(a, n=m_fft)
        conv = np.fft.irfft(f_fft * f_fft, n=m_fft)[: 2 * n - 1]
        max_conv = np.max(conv)

        # stop early if the target is already achieved
        if max_conv < 1.503:
            best_global_seq = a.copy()
            best_global_score = max_conv
            print(f"[target] reached score {max_conv:.6f} after {iter_cnt} iters")
            break

        # active set – indices whose convolution value is close to the maximum
        eps = max_conv * 1e-4
        active = np.where(conv >= max_conv - eps)[0]

        # compute the exact sub‑gradient for the active set
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i_range = np.arange(lo, hi + 1)
            shift_idx = k - i_range
            grad[i_range] += 2.0 * a[shift_idx]

        if np.allclose(grad, 0):
            # gradient vanished – nothing more to do
            break

        # back‑tracking line search
        step = step_factor * max_conv / (target_sum * target_sum)
        improved = False
        for _ in range(max_backtrack):
            new_a = a - step * grad
            new_a = np.maximum(new_a, 0.0)
            s = new_a.sum()
            if s > 0:
                new_a *= target_sum / s
            else:
                break
            new_score = _objective(new_a)
            if new_score < cur_score:
                improved = True
                break
            step *= 0.5

        if improved:
            a = new_a
            cur_score = new_score
            if new_score < best_global_score:
                best_global_score = new_score
                best_global_seq = a.copy()
                no_improve = 0
                step_factor = min(step_factor * 1.1, 1.0)
            else:
                no_improve += 1
        else:
            # No improvement – shrink the step factor and occasionally jitter.
            step_factor = max(step_factor * 0.7, 1e-4)
            no_improve += 1
            if no_improve % 15 == 0:
                i, j = np.random.randint(n, size=2)
                if i != j and a[i] > 0:
                    delta = 0.05 * a[i]
                    a[i] -= delta
                    a[j] += delta
                    a *= target_sum / a.sum()
                    cur_score = _objective(a)

        iter_cnt += 1
        if iter_cnt % 50 == 0:
            print(f"[iter] {iter_cnt:4d} cur={cur_score:.6f} best={best_global_score:.6f}")

    print(f"[finished] best score = {best_global_score:.6f}")
    # Convert to plain Python list for the external evaluator.
    return [float(x) for x in best_global_seq]
