"""
Propose a candidate step‐function sequence that improves the current bound.

Algorithm
---------
1. Choose a problem size `n`.  If a previous construction is available
   (`height_sequence_1`) its length is kept, otherwise a larger size (≈8000)
   is used – larger n gives more freedom to reduce the maximal self‑
   convolution.

2. The objective is `max_k (a*a)[k]`.  With the L¹‑norm fixed to
   `S = sqrt(2*n)` the evaluation returned by ``evaluate_sequence`` equals
   this maximum.  The algorithm therefore works **directly on the
   surrogate** `max_k (a*a)[k]` computed with an FFT (O(n log n)).

3. A *projected multiplicative gradient descent* with momentum (exponentiated
   gradient) is run.  The sub‑gradient of the active convolution entry
   `k_max` is `g_j = 2*a_{k_max-j}` (the diagonal term uses a single factor).
   The momentum term is updated with a decay factor `beta` and the update
   is `a ← a·exp(-η·momentum)`.  After each step the vector is renormalised
   to keep the L¹‑norm exactly `S`.

4. When progress stalls a short **simulated‑annealing phase** is entered.
   A random amount of mass is moved from one index to another; the move is
   accepted if it lowers the maximum, otherwise with probability
   `exp(-(Δ)/T)` where the temperature `T` decays linearly during the phase.
   Accepted moves are kept and the global best is updated.

5. The whole budget (≈1000 s) is split among a few independent restarts.
   Each restart begins from a fresh random Dirichlet‑type draw (gamma
   distribution with a random concentration) or, with 50 % probability,
   from a perturbed copy of the current best solution.  The best sequence
   over all restarts is returned.

The method stays fully inside the time limit, uses only NumPy and the
provided ``evaluate_sequence`` for the final validation, and prints occasional
progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level)
# ----------------------------------------------------------------------


def _conv_max(a: np.ndarray) -> tuple[np.ndarray, float, int]:
    """
    Compute the linear convolution a*a using an FFT,
    return (full_convolution, maximum_value, index_of_maximum).
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    # length for efficient FFT (next power of two)
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    max_idx = int(np.argmax(conv))
    return conv, float(conv[max_idx]), max_idx


def _initial_random_sequence(n: int, rng: np.random.Generator, S: float) -> np.ndarray:
    """
    Produce a random non‑negative sequence of length n with L1‑norm S.
    A random concentration parameter (gamma shape) is sampled to obtain
    diverse initial shapes (from spiky to almost uniform).
    """
    # concentration α∈[0.3,3] – low values give sparse vectors,
    # high values give near‑uniform vectors.
    alpha = rng.uniform(0.3, 3.0)
    a = rng.gamma(alpha, 1.0, size=n)
    if a.sum() == 0.0:
        a = np.full(n, S / n)
    else:
        a = a / a.sum() * S
    return a

# Main optimisation routine
# ----------------------------------------------------------------------

    Return a list of non‑negative heights (a step function) that yields a

    """

    if 'height_sequence_1' in globals():

    S = np.sqrt(2.0 * n)            # target L1‑norm (kept invariant)

    best_val = np.inf

    external_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] best internal max={best_val:.6f}, external score={external_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
