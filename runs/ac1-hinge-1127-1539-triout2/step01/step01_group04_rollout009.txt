"""
Propose a new coefficient sequence for the “step‑function” score.

The method is a deterministic variant of the Frank‑Wolfe / LP‑based
iteration described in the problem statement:

1. Initialise a non‑negative vector `a` (length `n`) with the
   square‑root‑difference sequence, scaled so that
   Σa = √(2·n).  For this scaling the evaluation
   `evaluate_sequence` equals the infinity‑norm of the self‑convolution
   of `a`.

2. Repeatedly
   * compute M = max(a * a)   (full linear convolution)
   * solve the linear programme

         maximise   1ᵀ g
         subject to  a ∗ g ≤ M   (point‑wise)
                    g ≥ 0

     (the convolution `a ∗ g` is a linear map; its matrix is built on
     the fly.)  The optimum `g₀` has the largest possible L¹‑norm while
     respecting the current bound `M`.

   * rescale `g = (√(2·n) / Σg₀)·g₀` so that it has the same L¹‑norm as `a`.

   * take a small convex combination `a ← (1‑t)·a + t·g` (t≈0.05).  For a
     sufficiently small `t` the new self‑convolution never exceeds `M`,
     hence the score is non‑increasing.

   * keep the best vector seen so far.

   * every few iterations add a tiny Gaussian jitter to escape plateaus.

The outer loop stops slightly before the supplied time budget; the best
sequence discovered is returned.  The implementation uses FFTs for fast
convolution and `scipy.optimize.linprog` (HiGHS) for the LP step.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: self‑convolution via FFT (real input → real output)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return the linear convolution a * a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Helper: build the convolution matrix A such that (A @ g)[k] = (a ∗ g)[k]
# ----------------------------------------------------------------------
def _build_conv_matrix(a: np.ndarray, A: np.ndarray) -> None:
    """
    Fill the pre‑allocated matrix A (shape (2·n‑1, n)) so that
    A @ g = a ∗ g for any g ≥ 0.
    """
    n = a.size
    # reset the matrix (only the columns that will be overwritten)
    A.fill(0.0)
    for i in range(n):
        A[i:i + n, i] = a[i]          # column i appears on rows i … i+n‑1

    Search for a non‑negative step‑function that minimises the given
    evaluation function.  Returns the best sequence found before the
    time limit expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0   # 1 s safety margin

    # --------------------------------------------------------------
    # Choose a workable length.
    # If a modest existing construction is present we reuse it;
    # otherwise we fall back to n = 2000.
    # --------------------------------------------------------------
    try:
        n_candidate = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        n_candidate = None

    # keep the problem tractable for the LP (≈64 MiB for n≈2000)
    n = n_candidate if (n_candidate is not None and n_candidate <= 2500) else 2000

    # --------------------------------------------------------------
    # 1) Initialise with the square‑root‑difference sequence.
    # --------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)
    a = np.sqrt(idx + 1.0) - np.sqrt(idx)          # a_i = √(i+1) – √i
    target_sum = math.sqrt(2.0 * n)                # required ℓ¹‑norm
    a *= target_sum / a.sum()                      # normalise to √(2 n)

    # remember the best vector seen
    best_seq = a.copy()
    conv = _conv_fft(a, 1 << ((2 * n - 1).bit_length()))
    best_score = float(conv.max())
    print(f"Initial score (n={n}): {best_score:.6f}")

    # Pre‑allocate the convolution matrix for the LP.
    # --------------------------------------------------------------
    A_mat = np.empty((2 * n - 1, n), dtype=np.float64)

    # iteration counters
    iteration = 0
    stagnation = 0

    jitter_interval = 30  # add jitter every ``jitter_interval`` iterations

        # Current bound M = max(a ∗ a)
        # ----------------------------------------------------------
        conv = _conv_fft(a, 1 << ((2 * n - 1).bit_length()))
        M = conv.max()
        if M < best_score - 1e-12:

            best_seq = a.copy()
            stagnation = 0

        # ----------------------------------------------------------

        c = -np.ones(n, dtype=np.float64)          # minimise –Σg  ⇔ maximise Σg

            iteration += 1

        # ----------------------------------------------------------
        g = g0 * (target_sum / sum_g0)       # now Σg = target_sum

        # convex combination (t_mix is deliberately small)
        a_new = (1.0 - t_mix) * a + t_mix * g

        a_new = np.maximum(a_new, 0.0)

        iteration += 1

    return best_seq.tolist()
