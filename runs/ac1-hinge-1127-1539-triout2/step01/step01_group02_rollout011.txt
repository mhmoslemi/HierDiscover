def _project_simplex(v):
    """Projects a vector onto probability simplex (nonnegative, sum=1)."""
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _project_simplex(v):
    """Project onto probability simplex."""
    # v: numpy array
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1

# Helper functions

def _generate_powerlaw(p: float, sigma: float, max_len: int = 5000,
                       eps: float = 1e-12) -> np.ndarray:
    """
    Return a truncated power‑law sequence a_i = (i + sigma) ** (-p).
    Truncation stops when the values fall below `eps` or `max_len` is
    reached.  The sequence is not normalised; scaling is irrelevant for
    `evaluate_sequence`.
    """
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p)
    a = a[a > eps]               # keep only entries larger than eps
    return a


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    { w | w_i >= 0, sum_i w_i = 1 }.
    Implementation follows Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # sorted in descending order
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)

    if best_seq is None:
        n = 1000
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)
        print('fallback to uniform, score', best_score)

    # ------------------------------------------------------------------
    # Phase 2 – soft‑max mirror‑descent refinement
    # ------------------------------------------------------------------
    x = np.array(best_seq, dtype=np.float64)
    # Normalise to the probability simplex (scale irrelevance)
    x = x / x.sum()

    n = x.shape[0]
    beta = 30.0            # initial temperature for the soft‑max
    beta_max = 500.0
    alpha0 = 0.2           # initial step‑size factor
    iteration = 0

    print('=== Phase 2 : soft‑max mirror‑descent refinement ===')
    while time.time() < deadline:
        iteration += 1

        # Full convolution of the current vector with itself
        conv = np.convolve(x, x)
        max_conv = conv.max()
        # With sum(x) == 1 the objective equals 2 * n * max_conv
        cur_score = 2.0 * n * max_conv

        if cur_score < best_score:
            best_score = cur_score
            best_seq = x.tolist()
            print(f'iter {iteration:5d}  new best {best_score:.6f}')
            if best_score < 1.503:
                print('Target reached – stopping early.')
                break

        # ----- soft‑max over convolution entries -----
        # subtract the maximum to keep the exponent numerically stable
        exp_terms = np.exp(beta * (conv - max_conv))
        w = exp_terms / exp_terms.sum()               # probability distribution over lags

        # ----- gradient of the smoothed objective -----
        # grad_i = 2 * sum_j w_{i+j} * x_j
        w_rev = w[::-1]
        corr = np.convolve(x, w_rev)                  # length = 3n‑2
        indices = (2 * n - 2) - np.arange(n)          # (2n‑2‑i) for i=0..n‑1
        grad = 2.0 * corr[indices]

        # ----- exponentiated‑gradient step and projection -----
        eta = alpha0 / np.sqrt(iteration)              # decreasing step size
        x = x * np.exp(-eta * grad)
        x = _project_simplex(x)

        # ----- temperature schedule (sharpen the soft‑max) -----
        beta = min(beta_max, beta * 1.001)

        # ----- occasional random jitter to escape plateaus -----
        if np.random.rand() < 0.01:
            i = np.random.randint(n)
            delta = (np.random.rand() - 0.5) * 0.05 * x[i]
            x[i] = max(0.0, x[i] + delta)
            x = _project_simplex(x)

    print('final best score:', best_score, 'length:', len(best_seq))
    return best_seq
