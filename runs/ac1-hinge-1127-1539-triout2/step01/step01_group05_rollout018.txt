"""Proposes a step‑function heights sequence via a two‑stage search.
Stage 1 performs a fast random search over simple parametrised families
(symmetric power‑law and beta‑distribution shapes).  The best candidate
found is then refined in Stage 2 by a projected gradient descent on a
soft‑max approximation of the convolution supremum.  The algorithm
maintains the sum of the heights = sqrt(2·n) so the evaluation score
equals the maximal convolution value.  A small amount of stochastic
jitter and simulated‑annealing acceptance are used to escape local
minima.
"""
import numpy as np
import time
from typing import List

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _compute_score(a: np.ndarray) -> float:
    """Fast version of evaluate_sequence (no security checks)."""
    n = a.shape[0]
    nfft = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))

    """Return a list of non‑negative heights that (hopefully) yields a low score.

    The function runs for at most *budget_s* seconds and returns the best
    sequence discovered.  The implementation follows the two‑stage approach
    described in the module docstring.

        return v * (target_sum / s)

    best_score = float("inf")

            ref = ref[:n] if ref.shape[0] > n else np.pad(ref, (0, n - ref.shape[0]))
        ref = _renorm(ref)
        score_ref = _compute_score(ref)
        best_a = ref.copy()
        best_score = score_ref

    # ------------------------------------------------------------------

    coords = i + 0.5
    rev_coords = n - i - 0.5

    rand_iter = 0
    # allocate a small budget for the cheap random search (≈15 % of total time)
    rand_deadline = start + 0.15 * budget_s
    while time.time() < rand_deadline:
        rand_iter += 1
        # Choose a family at random

        if family == "dec_sym":
            # a_i = (i+0.5)^(-p) + (n-i-0.5)^(-p)
            p = np.exp(np.random.uniform(np.log(0.1), np.log(1.5)))
            a = (coords ** (-p)) + (rev_coords ** (-p))
        elif family == "dec_prod":
            # a_i = ((i+0.5)*(n-i-0.5))^(-p)
            p = np.exp(np.random.uniform(np.log(0.1), np.log(1.5)))
            a = (coords * rev_coords) ** (-p)

            log_v = np.clip(log_v, -700, 700)  # exp(-700) ≈ 5e-305
            a = np.exp(log_v)

        if np.random.rand() < 0.15:

        if np.random.rand() < 0.1:
            a *= 1.0 + 2e-3 * np.random.randn(n)

        a = np.maximum(a, 0.0)
        a = _renorm(a)

        if sc < best_score:

            best_a = a.copy()

    # ------------------------------------------------------------------

    nfft_conv = _next_pow2(2 * n - 1)          # for a * a

        w = np.exp(gamma * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:
            # fall back to uniform distribution (should hardly happen)
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --------------------------------------------------------------
        # 3) Gradient of the smooth maximum: g = 2 * (w ⊛ a_rev)
        # --------------------------------------------------------------
        wf = np.fft.rfft(w, n=nfft_corr)
        af_rev = np.fft.rfft(a[::-1], n=nfft_corr)

        # --------------------------------------------------------------
        # 4) Gradient step, projection onto the non‑negative orthant, and renorm.
        # --------------------------------------------------------------

        a_new = np.maximum(a_new, 0.0)
        a_new = _renorm(a_new)

        # occasional jitter (helps escaping flat plateaus)
        if gd_iter % 73 == 0:
            jitter = 5e-5 * np.random.randn(n)
            a_new = np.maximum(a_new + jitter, 0.0)
            a_new = _renorm(a_new)

        # --------------------------------------------------------------
        # 5) Evaluate and possibly accept.
        # --------------------------------------------------------------
        score_new = _compute_score(a_new)

        # Simulated‑annealing style acceptance:
        accept = False
        if score_new < best_score:
            accept = True
        else:
            # temperature proportional to the current stepsize
            temp = eta
            if np.random.rand() < np.exp(-(score_new - best_score) / (temp + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if score_new < best_score:
                best_score = score_new
                best_a = a.copy()
                elapsed = time.time() - start
                print(f"GDIter {gd_iter:6d}: new best {best_score:.6f} (elapsed {elapsed:.1f}s)")

        # --------------------------------------------------------------
        # 6) Update hyper‑parameters (slow cooling / focusing).
        # --------------------------------------------------------------
        gamma = min(gamma * 1.003, gamma_max)
        eta = max(eta * 0.998, eta_min)

    # ------------------------------------------------------------------
    # Return the best sequence found (as plain Python list).
    # ------------------------------------------------------------------
    return best_a.tolist()

"""Generate a step‑function (non‑negative heights) with a low evaluation score.

The algorithm works in three phases:

1. **Random parametric search** – a cheap exploration of simple
   symmetric families (power‑law and beta‑type shapes).  The best
   candidate found is kept.

2. **Projected gradient descent** – starting from the best candidate,
   we minimise a smooth approximation of the true objective
   (the maximal self‑convolution) using the soft‑max trick.  After each
   gradient step the vector is projected onto the non‑negative orthant
   and renormalised so that its ℓ₁‑norm equals √(2 n).  The temperature
   of the soft‑max (``gamma``) is slowly increased while the step size
   (``eta``) is annealed.

3. **Local polishing** – a few pairwise mass‑transfer moves are tried
   to escape tiny plateaux.

The function respects the 1000 s time budget and returns the best
sequence found as a plain Python list.
"""

import time
from typing import List

import numpy as np


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _compute_score(a: np.ndarray) -> float:
    """Fast evaluation of `2 * n * max(conv(a, a)) / (sum(a) ** 2)`."""
    n = a.shape[0]
    nfft = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))


def _renorm(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale ``v`` so that its sum becomes ``target_sum`` (non‑negative)."""
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, target_sum / v.shape[0])
    return v * (target_sum / s)

    The routine is limited to ``budget_s`` seconds.  It first performs a
    cheap random search over a few parametric families and then refines
    the best candidate with a smooth‑max gradient descent.

        ref = _renorm(ref, target_sum)
        best_a = ref.copy()
        best_score = _compute_score(best_a)
        print(f"Initial reference score {best_score:.6f}")

        i = np.arange(n, dtype=np.float64)
        left = i + 0.5
        right = n - i - 0.5
        a0 = (left ** -0.5) + (right ** -0.5)

        gamma = min(gamma * 1.003, gamma_max)
        eta = max(eta * 0.998, eta_min)

    # Return the best sequence as a plain Python list.
    # ------------------------------------------------------------------
    return best_a.tolist()
