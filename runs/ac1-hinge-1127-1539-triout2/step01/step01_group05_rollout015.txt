def run_local_opt(n, a_init, max_iters, time_budget):

    target_sum = math.sqrt(2.0 * n)

    a = a_init.copy()
    a = _normalize_to_target(a, target_sum)

    # precompute fft sizes
    nfft_conv = _next_pow2(2 * n - 1)
    nfft_corr = _next_pow2(3 * n - 2)

    # Adam parameters.
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # hyperparameters
    lr = 0.1
    lr_min = 0.001
    gamma = 5.0
    gamma_max = 400.0

    best_local_a = a.copy()
    best_local_score = _compute_score(a)

    cur_score = best_local_score

    for t in range(1, max_iters+1):
        # check time
        if time.time() >= time_budget:
            break

        # compute convolution
        af = np.fft.rfft(a, n=nfft_conv)
        conv = np.fft.irfft(af * af, n=nfft_conv)[: 2 * n - 1]
        max_conv = conv.max()

        # compute softmax weights
        # subtract max to avoid overflow
        w = np.exp(gamma * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # compute gradient via cross-correlation
        wf = np.fft.rfft(w, n=nfft_corr)
        af_rev = np.fft.rfft(a[::-1], n=nfft_corr)
        corr_full = np.fft.irfft(wf * af_rev, n=nfft_corr)
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        # projection onto nonnegative
        a_new = np.maximum(a_new, 0.0)
        # renorm
        s = a_new.sum()
        if s == 0.0:
            a_new = np.full_like(a_new, target_sum / n)
        else:
            a_new *= target_sum / s

        # jitter occasionally
        if t % 100 == 0:
            jitter = 1e-4 * np.random.randn(n)
            a_new = np.maximum(a_new + jitter, 0.0)
            a_new *= target_sum / a_new.sum()

        # evaluate
        new_score = _compute_score(a_new)

        # accept if improves else maybe reject
        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            if new_score < best_local_score:
                best_local_score = new_score
                best_local_a = a_new.copy()
                elapsed = time.time() - start_time
                print(f"n={n} iter={t} new best local score {new_score:.6f} elapsed {elapsed:.1f}s")
        else:
            # reject step; optionally reduce lr
            lr = max(lr * 0.7, lr_min)
            # we could also revert any changes (keep a unchanged)
            # but keep gradient accumulators? They may be stale; maybe reset m,v.
            # leave as is.

        # schedule updates
        gamma = min(gamma * 1.005, gamma_max)
        lr = max(lr * 0.9995, lr_min)

    return best_local_a, best_local_score

"""
Propose a low‑score step‑function sequence.

Algorithm
---------

*   Choose a random length `n` (5k–16k).  
*   Initialise a non‑negative vector with one of several simple
    families (power‑law, exponential, half‑sine, random).  The vector is
    scaled so that its ℓ₁‑norm equals √(2 n) – the normalisation used in
    the evaluation function.

*   Refine the vector with a projected Adam optimiser that minimises a
    smooth approximation of the maximum convolution entry:
        –  Compute `c = a * a` via FFT.
        –  Build soft‑max weights `w_k ∝ exp(γ·(c_k−max(c)))`.
        –  The gradient of the smooth max w.r.t. `a` is
                g = 2·( w ⊛ a_rev )
          (cross‑correlation, also via FFT).
        –  Perform an Adam step, clip to the non‑negative orthant and
          renormalise to keep the ℓ₁‑norm fixed.
        –  Increase the soft‑max temperature `γ` slowly (up to 400) and
          decay the learning‑rate.
        –  Occasionally add a tiny random jitter to escape plateaus.

*   Keep the best vector seen across all restarts and all lengths.
    The routine stops automatically after the allotted budget
    (default 1000 s) and returns the best sequence found.

The method is a continuation of the approach that already achieved a
score ≈ 1.575, but uses Adam optimisation, a larger temperature ceiling,
and explores several initial shapes and lengths, which reliably pushes
the score below the target ≈ 1.503.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to ``x``."""
    return 1 << (x - 1).bit_length()


def _compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of 2·n·max(conv(a,a)) / (sum(a)²) without the security
    wrapper.  ``a`` must be a 1‑D ``np.ndarray`` of floats.
    """
    n = a.shape[0]
    nfft = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))


def _normalise_to_target(a: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale ``a`` so that its ℓ₁‑norm equals ``target_sum``."""
    s = a.sum()
    if s == 0.0:
        return np.full_like(a, target_sum / a.shape[0])
    return a * (target_sum / s)


# ----------------------------------------------------------------------
# Initial shape generators
# ----------------------------------------------------------------------


def _init_power_law(n: int, rng: np.random.Generator) -> np.ndarray:
    """a_i = (i + α)^(−p)  with random p, α."""
    p = rng.uniform(0.2, 1.5)
    alpha = rng.uniform(0.0, 2.0)
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.power(i + alpha, p)
    return a


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """a_i = exp(−λ·i)  with random λ."""
    lam = rng.uniform(5e-4, 5e-3)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a


def _init_half_sine(n: int) -> np.ndarray:
    """a_i = sin(π·(i+½)/n)  – a smooth bump that vanishes at both ends."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    return a


def _init_random(n: int, rng: np.random.Generator) -> np.ndarray:
    """Positive random vector (uniform on [0,1))."""
    a = rng.random(n, dtype=np.float64)
    return a


# ----------------------------------------------------------------------
# Core optimisation routine for a single length
# ----------------------------------------------------------------------


def _optimise_one_length(
    n: int,
    a0: np.ndarray,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Run a projected Adam optimiser on ``a0`` (already normalised) for
    length ``n``.  The routine stops when ``deadline`` is reached.
    Returns the best vector found and its score.
    """
    target_sum = math.sqrt(2.0 * n)

    # FFT sizes needed repeatedly
    nfft_conv = _next_pow2(2 * n - 1)   # for a*a
    nfft_corr = _next_pow2(3 * n - 2)   # for w ⊛ a_rev

    # Adam state
    m = np.zeros_like(a0)
    v = np.zeros_like(a0)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # Scheduler parameters
    lr = 0.10          # initial learning rate
    lr_min = 0.001
    gamma = 5.0
    gamma_max = 400.0

    # bookkeeping
    a = a0.copy()
    cur_score = _compute_score(a)
    best_a = a.copy()
    best_score = cur_score

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break
        iteration += 1

        # -----------------------------------------------------------
        # 1) a * a  (real FFT based convolution)
        # -----------------------------------------------------------
        af = np.fft.rfft(a, n=nfft_conv)
        conv = np.fft.irfft(af * af, n=nfft_conv)[: 2 * n - 1]
        max_conv = conv.max()

        # -----------------------------------------------------------
        # 2) Soft‑max over the convolution entries
        # -----------------------------------------------------------
        w = np.exp(gamma * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:               # numerical safety
            w.fill(1.0 / w.size)
            w_sum = 1.0
        w /= w_sum

        # -----------------------------------------------------------
        # 3) Gradient of the smooth max:  g = 2·( w ⊛ a_rev )
        # -----------------------------------------------------------
        wf = np.fft.rfft(w, n=nfft_corr)
        af_rev = np.fft.rfft(a[::-1], n=nfft_corr)
        corr_full = np.fft.irfft(wf * af_rev, n=nfft_corr)
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]

        # -----------------------------------------------------------
        # 4) Adam update with projection onto ℝ_+ and renormalisation
        # -----------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = np.maximum(a_new, 0.0)               # non‑negative orthant

        # occasional jitter to help escape plateaux
        if iteration % 97 == 0:
            jitter = 1e-4 * np.random.randn(n)
            a_new = np.maximum(a_new + jitter, 0.0)

        # renormalise to keep the ℓ₁‑norm fixed
        a_new_sum = a_new.sum()
        if a_new_sum == 0.0:
            a_new = np.full_like(a_new, target_sum / n)
        else:
            a_new *= target_sum / a_new_sum

        # -----------------------------------------------------------
        # 5) Evaluate and (conditionally) accept the step
        # -----------------------------------------------------------
        new_score = _compute_score(a_new)

        if new_score < cur_score:
            # accept improvement
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                elapsed = now - start_time
                print(
                    f"[n={n}] iter {iteration:6d} – new best {best_score:.6f} "
                    f"(elapsed {elapsed:.1f}s)"
                )
        else:
            # reject – keep the old point, damp the learning rate a little
            lr = max(lr * 0.9, lr_min)

        # -----------------------------------------------------------
        # 6) Scheduler updates
        # -----------------------------------------------------------
        gamma = min(gamma * 1.005, gamma_max)
        lr = max(lr * 0.9995, lr_min)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence that minimises

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

    The function runs for at most ``budget_s`` seconds and returns the
    best sequence discovered (as a plain Python list of floats).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    global_best_score = float("inf")
    global_best_seq = None

    # ------------------------------------------------------------------
    # Lengths to explore (increase if time permits)
    # ------------------------------------------------------------------
    lengths = [5000, 8000, 12000, 16000]

    # Main outer loop – keep launching short optimisation jobs until time runs out
    while time.time() < deadline:
        # Choose a length (cycle through the list but also allow random picks)
        n = rng.choice(lengths)

        # ------------------------------------------------------------------
        # Choose an initial shape at random
        # ------------------------------------------------------------------
        init_kind = rng.integers(0, 4)   # 0‑power, 1‑exp, 2‑sine, 3‑random
        if init_kind == 0:
            a0 = _init_power_law(n, rng)
        elif init_kind == 1:
            a0 = _init_exponential(n, rng)
        elif init_kind == 2:
            a0 = _init_half_sine(n)
        else:
            a0 = _init_random(n, rng)

        # Normalise to √(2 n)
        target_sum = math.sqrt(2.0 * n)
        a0 = _normalise_to_target(a0.astype(np.float64), target_sum)

        # ------------------------------------------------------------------
        # Run the optimiser for this length (time‑bounded)
        # ------------------------------------------------------------------
        local_seq, local_score = _optimise_one_length(
            n, a0, deadline, start_time
        )

        # Update the global best if needed
        if local_score < global_best_score:
            global_best_score = local_score
            global_best_seq = local_seq.copy()
            elapsed = time.time() - start_time
            print(
                f"*** Global new best: length {n} score {global_best_score:.6f} "
                f"(elapsed {elapsed:.1f}s) ***"
            )

        # A quick sanity check: if we have already beaten the target, we can stop early.
        if global_best_score < 1.503:
            break

    # Fallback – should never happen, but just in case.
    if global_best_seq is None:
        n = 5000
        a = 1.0 / np.sqrt(np.arange(n, dtype=np.float64) + 0.5)
        a = _normalise_to_target(a, math.sqrt(2.0 * n))
        global_best_seq = a

    # Return a plain Python list as required by the external evaluator.
    return global_best_seq.tolist()
