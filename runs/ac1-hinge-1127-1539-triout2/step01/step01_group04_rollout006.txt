"""
Algorithm:
- Choose length n (from previous solution if available else default).
- Build several simple initial non‑negative step functions (sqrt‑diff,
  arcsine, uniform, triangular).
- Pick the best by calling evaluate_sequence once.
- Perform a smoothed max‑minimisation using an exponential soft‑max
  (log‑sum‑exp) of the self‑convolution.  The gradient of the smooth
  objective is obtained analytically and evaluated by a second FFT
  convolution.  An Adam optimiser together with projection onto the
  simplex (non‑negative, fixed L1‑norm) evolves the coefficients.
- Occasionally add tiny Gaussian jitter to escape plateaus.
- Keep the best sequence seen and return it.
"""

import time, math, numpy as np

def _next_pow2(x):
    return 1 << (x-1).bit_length()

def _self_convolve(a, L):
    fa = np.fft.rfft(a, n=L)
    return np.fft.irfft(fa * fa, n=L)[: 2 * a.size - 1]

def _grad_from_softmax(conv, a, alpha):
    """Return gradient of log‑sum‑exp objective w.r.t. a.

    Parameters
    ----------
    conv: ndarray, shape (2n‑1,)
        Self‑convolution of a.
    a: ndarray, shape (n,)
    alpha: float
        Smoothing parameter.

    Returns
    -------
    grad: ndarray, shape (n,)
    """
    n = a.size
    M = conv.max()
    # soft‑max weights (normalised)
    w_raw = np.exp(np.clip(alpha * (conv - M), -50, 0))
    w = w_raw / w_raw.sum()
    # cross‑correlation w * a via FFT
    a_rev = a[::-1]
    L2 = _next_pow2(len(w) + n - 1)
    fw = np.fft.rfft(w, n=L2)
    fa = np.fft.rfft(a_rev, n=L2)
    corr = np.fft.irfft(fw * fa, n=L2)
    # extract the n entries where the shift aligns
    s = corr[n-1 : n-1 + n]  # shape (n,)
    # diagonal contribution
    w_diag = w[0::2]        # w at indices 0,2,4,...
    grad = 2.0 * s - w_diag * a
    return grad

"""
Algorithm – smoothed max‑minimisation with Adam

* Choose a length n (re‑use the length of a previously found sequence if
  possible, otherwise use a comfortable default).

* Build a few simple non‑negative initial step functions:
  – the classic √‑difference (the “square‑root derivative”),
  – an arcsine‑type density 1/√(x(1‑x)),
  – uniform,
  – a triangular shape.
  The one with the smallest value of ``evaluate_sequence`` is taken as the
  starting point.

* Optimise the L∞‑norm of the self‑convolution while keeping the L1‑norm
  fixed to √(2 n).  Instead of the non‑smooth max we minimise the
  log‑sum‑exp (soft‑max) of the convolution:
        f(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)
  The gradient of f is
        ∂f/∂a_i = 2·Σ_k p_k·a_{k‑i} – p_{2i}·a_i,
  where p_k are the normalised soft‑max weights.
  The weighted cross‑correlation Σ_k p_k·a_{k‑i} is evaluated with a single
  FFT, so each iteration costs only O(n log n).

* The gradient is fed to an Adam optimiser; after every step the vector is
  projected onto the feasible set (non‑negative and Σa = √(2 n)).
  Every few hundred iterations a tiny Gaussian jitter is added to help
  escape plateaus.

* The best sequence seen so far (according to ``evaluate_sequence``) is
  remembered and finally returned.  The whole routine respects the supplied
  time budget (default 1000 s) and never performs I/O.
"""

import time
import math
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _self_convolve(a: np.ndarray, L: int) -> np.ndarray:
    """Return the (real) self‑convolution a∗a using an FFT of length L."""
    fa = np.fft.rfft(a, n=L)
    return np.fft.irfft(fa * fa, n=L)[: 2 * a.size - 1]

def _grad_from_softmax(conv: np.ndarray, a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smoothed objective  (1/α)·log Σ exp(α·conv)
    w.r.t. the coefficient vector a.

    Parameters
    ----------
    conv : ndarray, shape (2n‑1,)
        The self‑convolution a∗a.
    a : ndarray, shape (n,)
        Current coefficient vector.
    alpha : float
        Smoothing parameter (larger ⇒ closer to the true max).

    Returns
    -------
    grad : ndarray, shape (n,)
        The gradient ∂f/∂a.
    """
    n = a.size
    M = conv.max()
    # soft‑max probabilities, safely clipped to avoid under‑flow
    w_raw = np.exp(np.clip(alpha * (conv - M), -50, 0))
    p = w_raw / w_raw.sum()                # Σ p_k = 1

    # cross‑correlation p⋆a (i.e. Σ_k p_k a_{k‑i})
    a_rev = a[::-1]
    L2 = _next_pow2(len(p) + n - 1)        # length for the second FFT
    fp = np.fft.rfft(p, n=L2)
    fa = np.fft.rfft(a_rev, n=L2)
    corr = np.fft.irfft(fp * fa, n=L2)
    # extract the n entries where the shift aligns (indices n‑1 … n‑1+n‑1)
    s = corr[n - 1 : n - 1 + n]

    # diagonal part: p_{2i}
    p_diag = p[0::2]                       # entries 0,2,4,… (length n)

    grad = 2.0 * s - p_diag * a
    return grad
