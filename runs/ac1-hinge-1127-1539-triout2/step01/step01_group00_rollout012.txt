def _smooth_gradient_step(a, eta, beta):
    n = a.shape[0]
    L = 2*n - 1
    conv_len_target = L
    fft_len = 1 << (conv_len_target + n - 1 - 1).bit_length()  # at least n+L-1
    # compute FFT of a padded
    Fa = np.fft.rfft(a, n=fft_len)
    # convolution a*a
    conv_full = np.fft.irfft(Fa * Fa, n=fft_len)[:L]
    max_conv = conv_full.max()
    # weights
    # use softmax over conv entries
    shifted = conv_full - max_conv  # negative or zero
    # scaling factor for numerical stability
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones_like(w) / w.shape[0]
    else:
        w = w / w_sum
    # compute correlation G = 2 * sum_j a_j * w_{i+j} - a_i * w_{2i}
    # compute w_rev
    w_rev = w[::-1]
    Fw = np.fft.rfft(w_rev, n=fft_len)
    # cross-correlation via FFT: a * w_rev (convolution)
    corr_full = np.fft.irfft(Fa * Fw, n=fft_len)
    # in conv length: corr_full length = fft_len
    # correlation[i] = corr_full[L-1 - i] for i in range(n)
    # ensure indexing correct
    indices = (L - 1) - np.arange(n)
    corr = corr_full[indices]
    # compute G
    G = 2.0 * corr - a * w[2 * np.arange(n)]
    # multiplicative update
    a_new = a * np.exp(-eta * G)
    # renormalize to sum S
    S = np.sqrt(2.0 * n)
    total = a_new.sum()
    if total == 0:
        a_new = np.full(n, S / n)
    else:
        a_new *= (S / total)
    return a_new, max_conv

def _smooth_gradient_step(a, eta, beta):
    n = a.shape[0]
    L = 2*n - 1
    fft_len = 1 << ((n + L - 1) - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)

    conv_full = np.fft.irfft(Fa * Fa, n=fft_len)[:L]
    max_conv = float(conv_full.max())

    # compute softmax-like but with factor beta

def _smooth_gradient_step(a, eta, beta):
    n = a.shape[0]
    L = 2 * n - 1
    fft_len = 1 << ((n + L - 1 - 1).bit_length())
    # FFT of a
    Fa = np.fft.rfft(a, n=fft_len)
    # convolution a*a
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:L]
    max_conv = float(conv.max())
    # compute weights
    shifted = conv - max_conv
    # avoid overflow for large beta; use np.exp safely:
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / L)
    else:
        w /= w_sum
    # correlation step
    w_rev = w[::-1]
    Fw = np.fft.rfft(w_rev, n=fft_len)
    # cross-correlation: a * w_rev
    corr_full = np.fft.irfft(Fa * Fw, n=fft_len)
    # extract correlation for i=0..n-1: conv index L-1-i
    idx = (L - 1) - np.arange(n)
    corr = corr_full[idx]
    # gradient
    G = 2.0 * corr - a * w[2 * np.arange(n)]
    # multiplicative update
    a_new = a * np.exp(-eta * G)
    # renormalize to sum S
    S = np.sqrt(2.0 * n)
    total = a_new.sum()
    if total == 0:
        a_new = np.full(n, S / n)
    else:
        a_new *= (S / total)
    return a_new, max_conv

"""
A smoothed‐gradient descent for minimising the peak of a self‑convolution.

The evaluation score for a non‑negative vector a with Σa = √(2·n) equals
max_k (a∗a)[k].  The algorithm works directly on this objective:

*   At each iteration a “soft‑max” of the convolution entries is formed:
    w_k ∝ exp(β·( (a∗a)[k] – max(a∗a) )).  For β → ∞ this becomes the usual
    max‑operator, for smaller β the gradient is averaged over many indices
    – this dramatically improves the descent behaviour.

*   The gradient of the smooth surrogate is
        G_i = 2·∑_j a_j·w_{i+j} – a_i·w_{2i} .
    The first term is a correlation of a with the weight vector w and can be
    computed with one FFT; the second term is a simple point‑wise product.

*   An exponentiated‑gradient (multiplicative) update
        a ← a·exp(‑η·G)
    keeps all entries non‑negative.  Afterwards the vector is projected
    onto the simplex Σa = √(2·n).

*   The step size η slowly decays, β is increased slowly (so the algorithm
    starts smooth and ends sharp).  Occasional random jitter and a
    restart‑when‑stuck mechanism help to escape plateaus.

The routine runs for at most ``budget_s`` seconds and returns the best
sequence seen.  All heavy work is done with FFTs (O(n log n)), which easily
fits into the 1000 s budget for n≈5600.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_max(a: np.ndarray) -> tuple[np.ndarray, float, int]:
    """
    Linear convolution a∗a via FFT.
    Returns the full convolution array, its maximum value and the index of
    that maximum.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    max_idx = int(np.argmax(conv))
    return conv, float(conv[max_idx]), max_idx


def _initial_sequence(n: int, seed: int) -> np.ndarray:
    """
    Produce a feasible starting vector of length n with Σa = √(2·n).
    With 50 % probability the existing global ``height_sequence_1`` is used,
    otherwise a random Dirichlet draw is taken.
    """
    rng = np.random.default_rng(seed)
    S = np.sqrt(2.0 * n)

    if 'height_sequence_1' in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=float)
            if base.size >= n:
                a = base[:n].copy()
            else:
                a = np.pad(base, (0, n - base.size), constant_values=0.0)
            # use the provided construction half the time
            if rng.random() < 0.5:
                a = a / a.sum() * S
                return a
        except Exception:
            pass

    # uniform Dirichlet (i.e. random point on the simplex)
    a = rng.random(n)
    a = a / a.sum() * S
    return a


# ----------------------------------------------------------------------
# Smoothed gradient step
# ----------------------------------------------------------------------


def _smooth_gradient_step(a: np.ndarray, eta: float, beta: float) -> tuple[np.ndarray, float]:
    """
    Perform one exponentiated‑gradient step using a smooth approximation of
    the ∞‑norm of the self‑convolution.

    Parameters
    ----------
    a : np.ndarray
        Current non‑negative vector (already normalised Σa = √(2·n)).
    eta : float
        Multiplicative step size (positive).
    beta : float
        Smoothing parameter for the soft‑max (β = 0 → uniform, large → max).

    Returns
    -------
    a_new : np.ndarray
        Updated vector (again Σa_new = √(2·n)).
    max_conv : float
        The maximal convolution entry of the *old* vector a (used only for
        diagnostics; the caller should recompute the max for the new vector
        if needed).
    """
    n = a.shape[0]
    L = 2 * n - 1                       # length of a∗a
    fft_len = _next_pow2(n + L - 1)      # enough for both convolutions

    # FFT of the current vector
    Fa = np.fft.rfft(a, n=fft_len)

    # ----- self‑convolution a∗a -----
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:L]
    max_conv = float(conv.max())

    # ----- soft‑max weights -----
    # shift by the maximum for numerical stability
    shifted = conv - max_conv
    w_raw = np.exp(beta * shifted)      # unnormalised weights
    w_sum = w_raw.sum()
    if w_sum == 0.0:
        w = np.full_like(w_raw, 1.0 / L)
    else:
        w = w_raw / w_sum                # normalised (∑w = 1)

    # ----- correlation a ↔ w (used in the gradient) -----
    w_rev = w[::-1]                     # reverse for correlation
    Fw = np.fft.rfft(w_rev, n=fft_len)
    corr_full = np.fft.irfft(Fa * Fw, n=fft_len)

    # extract the n required entries: corr[i] = ∑_j a_j·w_{i+j}
    idx = (L - 1) - np.arange(n)        # L‑1−i
    corr = corr_full[idx]

    # ----- gradient of the smooth surrogate -----
    # derivative of max entry w.r.t a_i is 2·∑_j a_j·w_{i+j} – a_i·w_{2i}
    G = 2.0 * corr - a * w[2 * np.arange(n)]

    # ----- multiplicative update (guarantees non‑negativity) -----
    a_new = a * np.exp(-eta * G)

    # ----- projection onto the simplex Σa = √(2·n) -----
    S = np.sqrt(2.0 * n)
    total = a_new.sum()
    if total == 0.0:
        a_new = np.full(n, S / n)
    else:
        a_new *= (S / total)

    return a_new, max_conv

    a = _initial_sequence(n, seed)

    # Current best (use the max of the convolution as the objective)
    _, cur_max, _ = _conv_max(a)
    best_seq = a.copy()
    best_score = cur_max
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Hyper‑parameters
    # ------------------------------------------------------------------
    eta = 0.5                 # initial step size
    eta_decay = 0.9995        # slow exponential decay
    beta = 5.0                # start with a smooth surrogate
    beta_inc = 1.001          # slowly sharpen the surrogate
    beta_max = 80.0           # cap to avoid overflow

    jitter_prob = 0.005       # occasional random perturbations
    steps_without_improve = 0
    restart_after = 20000    # restart when stuck for many iterations
    iteration = 0

    # ------------------------------------------------------------------
    # Main loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # one smoothed gradient step
        a, _ = _smooth_gradient_step(a, eta, beta)

        # evaluate the true objective (fast via FFT)
        _, cur_max, _ = _conv_max(a)

        # keep the best vector seen so far
        if cur_max < best_score - 1e-9:
            best_score = cur_max
            best_seq = a.copy()
            steps_without_improve = 0
            print(f"[improve] iter={iteration:5d} time={time.time():.1f}s "
                  f"score={best_score:.6f}")
        else:
            steps_without_improve += 1

        # --------------------------------------------------------------
        # Adaptive hyper‑parameters
        # --------------------------------------------------------------
        eta = max(eta * eta_decay, 1e-6)
        beta = min(beta * beta_inc, beta_max)

        # occasional jitter to escape tiny plateaus
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * 0.02
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()

        # restart when no improvement for a long while
        if steps_without_improve > restart_after:
            a = best_seq.copy()
            # apply a modest perturbation before continuing
            noise = np.random.randn(n) * a.mean() * 0.05
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()
            steps_without_improve = 0
            print(f"[restart] iter={iteration:5d} time={time.time():.1f}s")

    # ------------------------------------------------------------------
    # Return the best sequence found
    # ------------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f}  length = {len(best_seq)}")
    return best_seq.tolist()
