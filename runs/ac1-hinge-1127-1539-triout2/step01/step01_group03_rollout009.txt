def propose_candidate(seed=42, budget_s=1000):
    import numpy as np, time, cvxpy as cp

    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0

    # choose n maybe 2000 for intermediate speed
    n = 2000

    # step 1: Parameter grid for beta and offset to initial best shape
    betas = np.linspace(0.45, 0.55, 21)
    offsets = np.linspace(0.0, 5.0, 11)
    best_score = np.inf
    best = None
    for beta in betas:
        for c in offsets:
            idx = np.arange(n) + 1 + c  # to avoid zero division
            a = 1.0 / (idx ** beta)
            a /= a.sum() # normalize sum to 1
            # compute score quickly using FFT (custom objective)
            score = _objective(a)
            if score < best_score:
                best_score = score
                best = a.copy()
    a = best
    best_score = best_score

    # iterative LP improvement
    # pre-calc target sum = 1 (already normalized)
    epsilon = 1e-12
    iter = 0
    t_factor = 0.5 # mixing weight
    while time.time() < deadline and iter < 30:
        # build A matrix for current a
        ccol = np.concatenate([a, np.zeros(n-1)])
        rrow = np.concatenate([a[:1], np.zeros(n-1)])
        A = np.lib.stride_tricks.as_strided # but no; use scipy.linalg.toeplitz
        # Let's just import scipy.linalg toeplitz if available.
        from scipy.linalg import toeplitz
        A = toeplitz(ccol, rrow)  # shape (2n-1, n)

        # Solve LP: minimize t, with constraints A@(b) <= t * ones, sum(b)=1, b>=0
        b = cp.Variable(n, nonneg=True)
        t = cp.Variable()
        constraints = [A @ b <= t * np.ones(2*n-1), cp.sum(b) == 1]
        prob = cp.Problem(cp.Minimize(t), constraints)
        try:
            prob.solve(solver='ECOS', warm_start=True, max_iters=5000)
        except Exception as e:
            print("[LP] solve failed:", e)
            break
        if b.value is None:
            print("[LP] solution not found")
            break
        b_val = np.maximum(b.value, 0)
        # mixture
        mix = min(t_factor, 1.0)  # ensure within [0,1]
        a = (1 - mix) * a + mix * b_val
        # ensure nonnegative and normalized to sum=1
        a = np.maximum(a, 0)
        a_sum = a.sum()
        if a_sum < epsilon:
            break
        a /= a_sum

        # compute new score
        new_score = _objective(a)
        print(f"[LP] iter={iter}, t={t.value:.6e}, score={new_score:.6f}")
        if new_score < best_score - 1e-6:
            best_score = new_score
            best_seq = a.copy()
        else:
            # if no improvement, decrease mixing factor
            t_factor *= 0.9

        # update mixing factor maybe increase gradually
        t_factor = min(t_factor * 1.05, 0.8) # bounded

        # stopping early if target reached
        if best_score < 1.503:
            break

        iter += 1

    print("[+] Final best score", best_score)
    return [float(x) for x in best_seq]

def build_conv_matrix(a):
    # returns matrix A shape (2n-1, n)
    n = len(a)
    # Try using scipy.linalg.toeplitz
    try:
        from scipy.linalg import toeplitz
        ccol = np.concatenate([a, np.zeros(n-1)])
        rrow = np.concatenate([a[:1], np.zeros(n-1)])
        A = toeplitz(ccol, rrow)
        return A
    except Exception:
        # fallback manual:
        m = 2*n-1
        A = np.zeros((m, n), dtype=np.float64)
        for k in range(m):
            lo = max(0, k - (n-1))
            hi = min(n-1, k)
            # indices i in [lo, hi], then a_{k-i} times b_i.
            A[k, lo:hi+1] = a[k - np.arange(lo, hi+1)]
        return A

def lp_optimize(a):
    n = len(a)
    A = build_conv_matrix(a)
    b = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    constraints = [A @ b <= t * np.ones(2*n-1), cp.sum(b) == 1]
    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver='ECOS', warm_start=True, verbose=False, max_iters=5000)
    if b.value is None:
        return None, None, None
    return b.value, t.value, A

def _gradient_refine(a, deadline):
    # same as earlier gradient update
    n = len(a)
    target_sum = a.sum()
    step_factor = 0.5
    best_seq = a.copy()
    best_score = _objective(a)
    while time.time() < deadline:
        # compute conv via FFT
        m = 1 << ((2*n-1).bit_length())
        fft = np.fft.rfft(a, n=m)
        conv = np.fft.irfft(fft*fft, n=m)[:2*n-1]
        max_conv = conv.max()
        k_max = int(np.argmax(conv))
        # compute gradient
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        shift = np.zeros_like(a)
        shift[mask] = a[idx[mask]]
        grad = a + shift
        # step size
        r = max_conv / (np.sum(a) ** 2)
        step = step_factor * r
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-12:
            a_new = a_new * (target_sum / s)
        else:
            break
        new_score = _objective(a_new)
        if new_score < best_score:
            best_score = new_score
            best_seq = a_new.copy()
            a = a_new
            step_factor = min(step_factor * 1.1, 2.0)
        else:
            step_factor *= 0.9
        if best_score < 1.503:
            break
    return best_seq, best_score

"""
propose_candidate
------------------

Search for a non‑negative step‑function (a list of heights) that minimizes

    2·n·max(conv(a, a)) / (sum(a)²)

where `conv` is the full linear convolution.  The algorithm consists of

1. **Parametric warm‑start** – a quick grid search over the one‑parameter
   family a_i = 1/(i+1+c)^β to obtain a decent initial sequence.

2. **Linear‑programming improvement** – given the current sequence `a`,
   we solve the LP

        maximise   Σ b_j
        subject to A(a)·b ≤ M·1,   b ≥ 0

   where `M = max(conv(a,a))` and `A(a)` is the convolution matrix
   (the k‑th row of `A` contains the coefficients of the linear form
   `conv(a,b)[k]`).  The optimal `b` can contain much more total mass
   than `a` because it puts weight where `a` is small, while still keeping
   the cross‑convolution bounded by `M`.

   After obtaining `b` we normalise it (g = b / Σb) and mix it with the
   current iterate:

        a ← (1‑τ)·a + τ·g   (τ starts at 0.25 and is adapted)

   The mixing step is guaranteed (by the construction of the LP) not to
   increase the maximal self‑convolution, while typically reducing it.
   The process is repeated until the time budget expires or the score
   goes below the target 1.503.

3. **Fallback** – if `cvxpy` is unavailable or the LP fails, a simple
   sub‑gradient descent on the max‑convolution is performed.

The function returns a plain ``list[float]`` that satisfies all checks of
``evaluate_sequence``.  The whole search finishes well before the
1000 s budget.
"""
import time
import numpy as np

# Optional imports – if they are missing we fall back to a simpler method.
try:
    import cvxpy as cp
except Exception:  # pragma: no cover
    cp = None

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """Fast evaluation of 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = len(seq)
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    # FFT length – next power of two for speed.
    m = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft * fft, n=m)[: 2 * n - 1]
    max_conv = np.max(conv)
    return float(2.0 * n * max_conv / (s * s))


def _max_conv(seq: np.ndarray) -> float:
    """Return only the maximal entry of conv(seq,seq)."""
    n = len(seq)
    m = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft * fft, n=m)[: 2 * n - 1]
    return float(np.max(conv))


def _conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n convolution matrix A such that
        (A @ b)[k] = sum_i a_i * b_{k-i}
    (out‑of‑range indices are treated as zero).
    """
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    # Column i contains a shifted down by i rows.
    for i in range(n):
        A[i : i + n, i] = a
    return A


def _fallback_gradient(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Very simple sub‑gradient descent used when cvxpy is unavailable."""
    n = len(a)
    target_sum = np.sum(a)
    step_factor = 0.5
    best_seq = a.copy()
    best_score = _objective(a)

    while time.time() < deadline:
        # full convolution via FFT
        m = 1 << ((2 * n - 1).bit_length())
        fft = np.fft.rfft(a, n=m)
        conv = np.fft.irfft(fft * fft, n=m)[: 2 * n - 1]
        max_conv = np.max(conv)
        k_max = int(np.argmax(conv))

        # sub‑gradient g_i = a_i + a_{k_max-i}
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        shift = np.zeros_like(a)
        shift[mask] = a[idx[mask]]
        grad = a + shift

        r = max_conv / (target_sum ** 2)
        step = step_factor * r
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)

        s = np.sum(a_new)
        if s > 1e-12:
            a_new *= target_sum / s
        else:
            break

        new_score = _objective(a_new)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_seq = a_new.copy()
            a = a_new
            step_factor = min(step_factor * 1.1, 2.0)
        else:
            step_factor *= 0.9

        if best_score < 1.503:
            break

    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list[float]:
    """
    Return a list of non‑negative heights that (heuristically) minimizes the
    evaluation function used in the competition.  The routine respects the
    1000 s time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0  # tiny safety margin

    # ------------------------------------------------------------------
    # 1. Choose a moderate length – large enough to approximate the
    #    continuous optimum, small enough for fast LP solves.
    # ------------------------------------------------------------------
    n = 1200
    print(f"[+] Starting search, length = {n}")

    # ------------------------------------------------------------------
    # 2. Very cheap grid search over a simple 1/√ family to obtain a
    #    decent starting point.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    betas = np.linspace(0.45, 0.55, 21)      # around the known optimum 0.5
    offsets = np.linspace(0.0, 5.0, 11)      # small additive shift

    for beta in betas:
        for c in offsets:
            idx = np.arange(n, dtype=np.float64) + 1.0 + c
            cand = 1.0 / (idx ** beta)
            cand /= cand.sum()                # enforce sum = 1
            score = _objective(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"[grid] beta={beta:.3f}, offset={c:.2f} → score={score:.6f}")
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    if best_seq is None:                     # should never happen
        best_seq = np.ones(n, dtype=np.float64) / n
        best_score = _objective(best_seq)

    a = best_seq.copy()
    best_global_seq = a.copy()
    best_global_score = best_score

    # ------------------------------------------------------------------
    # 3. Main optimisation loop – LP improvement (if cvxpy is present).
    # ------------------------------------------------------------------
    tau = 0.25            # mixing factor (will be adapted)
    max_iters = 30
    it = 0

    while time.time() < deadline and it < max_iters:
        # Compute the current maximal self‑convolution M.
        M = _max_conv(a)

        if cp is None:
            # No convex optimisation library – fall back to gradient.
            print("[+] cvxpy missing – using fallback gradient descent.")
            best_seq, best_score = _fallback_gradient(a, deadline)
            best_global_seq = best_seq
            best_global_score = best_score
            break

        # Build the convolution matrix for the current a.
        A = _conv_matrix(a)               # shape (2n‑1, n)

        # Linear programme:
        # maximise Σ b  subject to A·b ≤ M·1,  b ≥ 0
        b_var = cp.Variable(n, nonneg=True)
        constraints = [A @ b_var <= M * np.ones(2 * n - 1)]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver="ECOS", warm_start=False, max_iters=5000, verbose=False)
        except Exception as exc:          # pragma: no cover
            print("[LP] Solver exception:", exc)
            break

        if prob.status not in ("optimal", "optimal_inaccurate"):
            print("[LP] Solver status not optimal:", prob.status)
            break

        b_val = np.maximum(b_var.value, 0.0)   # clip tiny negatives
        sum_b = b_val.sum()
        if sum_b < 1e-12:
            print("[LP] Trivial solution (sum_b≈0).")
            break

        # Normalise b to have sum = 1   (g = b / Σb)
        g = b_val / sum_b

        # Mix the new direction into a.
        a = (1.0 - tau) * a + tau * g
        a_sum = a.sum()
        if a_sum < 1e-12:
            break
        a /= a_sum                         # re‑normalise to avoid drift

        # Evaluate the new sequence.
        new_score = _objective(a)
        if new_score < best_global_score - 1e-12:
            best_global_score = new_score
            best_global_seq = a.copy()
            # Successful step – we can be a bit more aggressive.
            tau = min(tau * 1.1, 0.8)
        else:
            # No improvement – become more conservative.
            tau = max(tau * 0.7, 0.05)

        print(f"[LP] iter={it:02d}, M={M:.6e}, score={new_score:.6f}, τ={tau:.3f}")

        if best_global_score < 1.503:
            print(f"[!] Target reached: score={best_global_score:.6f} at iter {it}")
            break

        it += 1
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 4. Final sanity checks and return.
    # ------------------------------------------------------------------
    if best_global_seq is None:
        # Defensive fallback – use the grid result.
        best_global_seq = best_seq

    final = np.maximum(best_global_seq, 0.0)
    total = final.sum()
    if total < 1e-12:
        # Extremely unlikely, but guard against division by zero.
        final = np.ones_like(final) / final.size
    else:
        final = final / total                 # ensure sum = 1 (well above 0.01)

    print(f"[+] Search finished. Best score = {best_global_score:.6f}")
    return [float(x) for x in final]
