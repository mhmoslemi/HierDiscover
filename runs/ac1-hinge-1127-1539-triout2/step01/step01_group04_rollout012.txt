"""
Propose a very‑low‑score step‑function.

The procedure consists of three phases:

1. **Exploratory families** – generate several hand‑crafted shapes
   (sqrt‑difference, half‑sine, power‑laws, exponentials and their
   mixtures).  Each candidate is normalised to the same L¹‑norm
   (√(2·n)) and evaluated.  The best of these gives a solid initial
   point.

2. **Smooth‑max Adam optimisation** – the true objective is
   `max_k (a*a)_k`.  To obtain a usable gradient we replace the max
   by a smooth approximation
       Fτ(a) = (1/τ)·log Σ_k exp(τ·(a*a)_k) .
   Its gradient is 2·corr(a,w) where `w` are the soft‑max weights.
   An Adam optimiser (first‑order adaptive step) moves the vector,
   after each step the non‑negativity and L¹–constraint are enforced
   by truncation and scaling.

3. **Random jitter** – every few hundred iterations a tiny Gaussian
   perturbation is added (followed by the same projection).  This
   helps to escape shallow local minima.

All heavy work (convolution, correlation) is performed with FFTs,
so even several ten‑thousands of iterations finish well within the
1000 s budget.  The function returns the best sequence found.
"""

import time
import math
import numpy as np

# Helper routines (all top‑level, no closures)
# ----------------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Self‑convolution a*a using a real FFT of length L.
    Returns the first (2·len(a)‑1) entries."""
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * a.size - 1]


def _smoothmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """Soft‑max weights for the smooth max with temperature tau."""
    m = conv.max()
    # shift to avoid overflow
    shifted = conv - m
    exp_term = np.exp(tau * shifted)
    return exp_term / exp_term.sum()


def _grad_smoothmax(a: np.ndarray, tau: float,
                    L_conv: int, L_corr: int) -> np.ndarray:
    """Gradient of the smooth‑max objective.
    Returns a vector of length len(a)."""
    n = a.size

    # 1) self‑convolution (needed for the soft‑max weights)
    conv = _conv_self(a, L_conv)                # length 2n‑1

    # 2) soft‑max weights
    w = _smoothmax_weights(conv, tau)          # length 2n‑1

    # 3) correlation a ⋆ w  (linear term of the gradient)
    #    correlation via FFT: corr = ifft(FA * conj(FW))
    Fa = np.fft.rfft(a, n=L_corr)
    Fw = np.fft.rfft(w, n=L_corr)
    corr = np.fft.irfft(Fa * np.conj(Fw), n=L_corr)

    #   In the linear correlation the zero‑lag entry sits at index len(w)-1
    start = len(w) - 1
    grad = 2.0 * corr[start:start + n]          # shape (n,)
    return grad


def _score(a: np.ndarray) -> float:
    """Exact evaluation of the supplied objective (no security checks)."""
    s = a.sum()
    if s < 1e-9:
        return float("inf")
    n = a.size
    # convolution via FFT (fast and accurate for our sizes)
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


def _normalise(a: np.ndarray, target_sum: float) -> np.ndarray:
    """Project onto the feasible set: a_i ≥ 0 and Σa_i = target_sum."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        return np.full_like(a, target_sum / a.size)
    return a * (target_sum / s)
