assistantfinal**Solution Overview**

The goal is to minimise  

```
score(a) = 2·n·max(conv(a,a)) / ( Σ a_i )²
```

The score is unchanged by scaling the whole vector, therefore we fix the
ℓ₁‑norm to  

```
Σ a_i = √(2·n)
```

and the problem reduces to minimising the maximal entry of the auto‑convolution.

The algorithm consists of three stages

1. **Rich parametric grid search** –  
   candidates are of the form  

   ```
   a_i = exp(-α·i) / (i+1+c)ᵝ ,   i = 0 … n-1
   ```

   (`β`≈0.5 reproduces the classical 1/√x kernel, the extra
   exponential factor lets the tail decay faster).
   A fine grid over the three parameters (β, c, α) is examined,
   each candidate is normalised and evaluated with a fast FFT based
   implementation of the score.  
   The best vector from this stage is used as a warm start.

2. **Projected sub‑gradient descent** –  
   For the current vector `a` the index `kₘₐₓ` that attains the maximal
   convolution entry is found.  
   The sub‑gradient of `max(conv)` at this point is  

   ```
   g_i = a_i + a_{k_max-i}      (zero when the second term is out of range)
   ```

   A step `a ← a – η·g` is taken, negatives are clipped to zero and the
   vector is re‑normalised to the prescribed ℓ₁‑norm.  
   The step size `η` is adapted during the run; occasional random
   redistributions of a tiny amount of mass help to escape plateaux.

3. **Simple random local search** –  
   After the sub‑gradient phase a short stochastic search (pairwise
   transfer of a small amount of mass) is performed; any improvement is
   kept.

All computations use only NumPy and the FFT, so the whole search fits
comfortably into the 1000 s time budget.  The routine prints progress
information and finally returns the best sequence it has found as a plain
`list[float]`.

"""
Fast search for a low‑score non‑negative step‑function.

The algorithm works in three phases:
1.  A dense grid over the three‑parameter family
        a_i = exp(-α·i) / (i+1+c)^β
    (β ≈ 0.5, α ≥ 0, c ≥ 0) is examined; every candidate is
    ℓ₁‑normalised to √(2·n) and evaluated with a fast FFT based
    objective.
2.  Starting from the best grid point a projected sub‑gradient
    descent on max(conv(a,a)) is performed.  The active index
    k_max defines the sub‑gradient g_i = a_i + a_{k_max-i}.
3.  A short random local search (tiny mass transfers) finishes the
    optimisation.

All operations stay inside the limits imposed by ``evaluate_sequence``.
"""
from __future__ import annotations

import time
from typing import List

import numpy as np

# ----------------------------------------------------------------------
# Helper: fast evaluation of the score (identical to the one used by the
# competition harness).  It works with NumPy arrays and uses FFT for
# speed.
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Return the exact score for *seq* (no clipping applied)."""
    n = len(seq)
    s = np.sum(seq)
    if s < 0.01:
        return float("inf")

    # next power of two for the convolution length 2*n-1
    m = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft * fft, n=m)[: 2 * n - 1]
    max_conv = np.max(conv)

    return float(2.0 * n * max_conv / (s * s))

                if cur < best_score:
                    best_score = cur
                    best_seq = cand.copy()
                    print(f"[grid] β={beta:.3f}, c={c:.2f}, α={alpha:.4f} → score={cur:.6f}")

                if time.time() > deadline:
                    break
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    # If for any reason the grid failed (should not happen) fall back to
    # a uniform sequence.
    if best_seq is None:
        best_seq = np.ones(n, dtype=np.float64) * (target_sum / n)
        best_score = _objective(best_seq)
        print("[!] Grid did not produce a candidate – using uniform vector.")

    # ------------------------------------------------------------------
    # 3. Projected sub‑gradient descent on max(conv(a,a)).
    #    The step size is adapted; occasional random perturbations
    #    avoid stagnation.
    # ------------------------------------------------------------------
    a = best_seq.copy()
    best_global_seq = a.copy()
    best_global_score = best_score

    step_factor = 0.5          # controls the absolute step size
    no_improve = 0
    it = 0

    while time.time() < deadline:
        # ----- convolution and position of current maximum -----
        m_len = 1 << ((2 * n - 1).bit_length())
        fft_a = np.fft.rfft(a, n=m_len)
        conv = np.fft.irfft(fft_a * fft_a, n=m_len)[: 2 * n - 1]
        max_conv = np.max(conv)
        k_max = int(np.argmax(conv))

        # ----- sub‑gradient for the active index -----
        shift_idx = k_max - np.arange(n)
        mask = (shift_idx >= 0) & (shift_idx < n)
        shifted = np.zeros_like(a)
        shifted[mask] = a[shift_idx[mask]]
        grad = a + shifted                # g_i = a_i + a_{k_max-i}

        # ----- step size (scale‑invariant) -----
        r = max_conv / (np.sum(a) ** 2)   # == max_conv / (2·n) after normalisation
        step = step_factor * r

        new_a = a - step * grad
        new_a = np.maximum(new_a, 0.0)

        # ----- renormalise (scaling does not affect the score) -----
        s = np.sum(new_a)
        if s > 1e-12:
            new_a *= target_sum / s
        else:
            break

        new_score = _objective(new_a)

        if new_score < best_global_score - 1e-9:
            best_global_score = new_score
            best_global_seq = new_a.copy()
            a = new_a
            no_improve = 0
            step_factor = min(step_factor * 1.07, 2.0)    # slowly grow step
            if it % 100 == 0:
                print(f"[grad] it={it:04d}, score={new_score:.6f}, step={step:.4e}")
        else:
            no_improve += 1
            # random tiny redistribution every few stalls
            if no_improve % 30 == 0:
                i, j = np.random.randint(n, size=2)
                delta = np.random.rand() * a[i] * 0.03
                a[i] = max(0.0, a[i] - delta)
                a[j] += delta
                a *= target_sum / np.sum(a)
                step_factor *= 0.9
        it += 1

        if best_global_score < 1.503:
            print(f"[!] Desired bound reached (score={best_global_score:.6f}) after {it} iterations.")
            break

    # ------------------------------------------------------------------
    # 4. Short random local search – transfers of a tiny amount of mass.
    # ------------------------------------------------------------------
    a = best_global_seq.copy()
    it_rand = 0
    while time.time() < deadline and it_rand < 5000:
        i, j = np.random.randint(n, size=2)
        # transfer a small fraction of a[i] to a[j]
        delta = a[i] * np.random.rand() * 0.02
        new_a = a.copy()
        new_a[i] = max(0.0, new_a[i] - delta)
        new_a[j] += delta
        new_a *= target_sum / np.sum(new_a)

        new_score = _objective(new_a)
        if new_score < best_global_score:
            best_global_score = new_score
            best_global_seq = new_a.copy()
            a = new_a
            print(f"[rand] it={it_rand:04d}, score={new_score:.6f}")
        it_rand += 1

    print(f"[+] Search finished. Best score = {best_global_score:.6f}")
    # Return a plain Python list – the outer harness will perform the final clipping.
    return [float(x) for x in best_global_seq]
