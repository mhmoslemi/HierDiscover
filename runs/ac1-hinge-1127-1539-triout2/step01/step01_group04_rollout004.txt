"""
Propose a non‑negative step function with a low evaluation score.

The algorithm is a **smooth‑max gradient descent** with Adam‑style
adaptation.  The objective
    score = 2·n·max(b) / (sum(a)²)     with  b = a * a
is scale‑invariant, so we keep the L¹‑norm fixed to √(2·n) and only
minimise the maximal entry of the self‑convolution.

Key ideas
---------
1. Initialise with the classic √‑difference sequence (the best known
   analytic construction).
2. In each iteration:
   * Compute the linear convolution a*a by a single FFT.
   * Replace the hard “max” by a smooth approximation
       f_α(a) = (1/α)·log Σ_j exp(α·b_j) .
     The corresponding gradient is a weighted sum of all shifted copies
     of a; the weights are a soft‑max over the convolution values.
   * Compute the gradient efficiently as a cross‑correlation using FFT.
   * Update a with an Adam step and project back onto the feasible
     simplex (non‑negative, fixed L¹‑norm).
3. The smoothing parameter α is increased slowly (from ≈5 up to ≈200)
   so the method first moves in a well‑behaved direction and later
   focuses on the true maximum.
4. Every few hundred iterations a tiny random jitter is added to help
   escape plateaus, and the best sequence seen so far is always kept.

The whole optimisation runs until the allotted time (≈1000 s) expires
and returns the best sequence discovered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all are top‑level (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Small utility returning the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project a non‑negative vector onto the simplex {x ≥ 0, Σx = target_sum}.
    This implementation simply rescales after truncating negatives,
    which is sufficient for our purposes and cheap.
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # fallback: uniform distribution
        return np.full_like(v, target_sum / v.size, dtype=v.dtype)
    return v * (target_sum / s)


def _smooth_max_weights(b: np.ndarray, alpha: float) -> np.ndarray:
    """
    Return soft‑max weights w_j = exp(α·(b_j‑max(b))) / Σ exp(·).
    The subtraction of max(b) avoids overflow.
    """
    max_b = b.max()
    shifted = b - max_b               # ≤ 0
    # exponentiate; for large negative values the term underflows to 0,
    # which is fine.
    e = np.exp(alpha * shifted)
    # normalise
    return e / e.sum()


def _gradient_via_correlation(a: np.ndarray, w: np.ndarray, L: int) -> np.ndarray:
    """
    Compute   g = 2 * (a ⋆ w)   (cross‑correlation)   on the linear domain.
    Both a (length n) and w (length 2n‑1) are zero‑padded to length L,
    where L ≥ 3·n‑2.
    The result g has length n.
    """
    # FFT of the two real arrays (rfft gives the non‑redundant half)
    A = np.fft.rfft(a, n=L)
    W = np.fft.rfft(w, n=L)

    # correlation = ifft( A * conj(W) )
    corr_full = np.fft.irfft(A * np.conj(W), n=L)

    # The linear correlation starts at index (len(w)‑1) = 2·n‑2
    start = 2 * a.size - 2
    return 2.0 * corr_full[start:start + a.size]


def _self_convolution(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·n‑1) using a single FFT of length L.
    """
    A = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(A * A, n=L)
    return conv_full[: 2 * a.size - 1]

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises the provided
    `evaluate_sequence` function.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the function stops slightly before the limit).

    Returns
    -------
    list[float]
        The best sequence found within the time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Choose the problem size.
    # If a previously discovered sequence exists we reuse its length,
    # otherwise fall back to a reasonably sized default.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        n = 2000

    # Target L1‑norm (makes the evaluation equal to max(b))
    target_sum = math.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # 1) Initialise with the √‑difference sequence (known good start)
    # --------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)
    a = np.sqrt(idx + 1.0) - np.sqrt(idx)           # a_i = √(i+1) – √i
    a *= target_sum / a.sum()                       # normalise to √(2·n)

    # Keep the best sequence seen so far
    best_seq = a.copy()
    best_score = float('inf')
    # (the score equals the maximal convolution entry because of the normalisation)
    # but we also call evaluate_sequence once for safety.
    try:
        best_score = float(evaluate_sequence(best_seq.tolist()))
    except Exception:
        pass
    print(f"Initial score (external eval): {best_score:.6f}")

    # --------------------------------------------------------------
    # Pre‑compute FFT length – it must accommodate both a*a and a⋆w.
    # Length needed: ≥ max(2·n‑1, 3·n‑2).  Use the next power of two.
    # --------------------------------------------------------------
    L = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Adam optimiser state
    # --------------------------------------------------------------
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    base_lr = 0.12                # base learning rate (tuned empirically)

    # --------------------------------------------------------------
    # Smoothing parameter for the soft‑max over the convolution.
    # Starts small (smooth landscape) and grows to focus on the true max.
    # --------------------------------------------------------------
    alpha = 5.0
    max_alpha = 200.0
    alpha_growth = 1.03           # multiplicative growth every few iterations

    # --------------------------------------------------------------
    # Counters for bookkeeping
    # --------------------------------------------------------------
    iteration = 0
    no_improve = 0
    last_improve = time.time()

    while time.time() < deadline:
        # ---- self‑convolution and maximal entry ----
        b = _self_convolution(a, L)            # length 2·n‑1
        max_b = b.max()
        cur_score = float(max_b)               # same as evaluate_sequence for our normalisation

        # ---- keep best so far ----
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            no_improve = 0
            print(f"Iter {iteration:6d} – new best {best_score:.6f}")

        else:
            no_improve += 1

        # ---- soft‑max weights for a smooth approximation of the max ----
        w = _smooth_max_weights(b, alpha)

        # ---- gradient (cross‑correlation) ----
        grad = _gradient_via_correlation(a, w, L)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (iteration + 1))
        v_hat = v / (1.0 - beta2 ** (iteration + 1))

        # Decaying learning rate – helps convergence in later stages
        lr = base_lr / (math.sqrt(iteration + 1) + 1.0)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- projection onto the feasible simplex (non‑negative, fixed sum) ----
        a = _project_onto_simplex(a, target_sum)

        # ---- occasional jitter to escape flat regions ----
        if iteration % 200 == 0 and iteration > 0:
            sigma = target_sum * 5e-4 / math.sqrt(n)   # tiny relative noise
            noise = np.random.randn(n) * sigma
            a = _project_onto_simplex(a + noise, target_sum)

        # ---- increase smoothing (alpha) gradually ----
        if iteration % 50 == 0 and iteration > 0 and alpha < max_alpha:
            alpha = min(max_alpha, alpha * alpha_growth)

        # ---- restart if we have stalled for a long time ----
        if no_improve >= 5000:
            # perturb the current best and continue from there
            perturb = np.random.randn(n) * (target_sum * 1e-3 / math.sqrt(n))
            a = _project_onto_simplex(best_seq + perturb, target_sum)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"Iter {iteration:6d} – restart after stall")

        iteration += 1

    print(f"Finished after {iteration} iterations – best score {best_score:.6f}")
    return best_seq.tolist()
