def cross_corr(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    n = len(a)
    len_w = len(w)   # 2*n -1
    m2 = 1 << ((n + len_w - 1).bit_length())
    fft_a = np.fft.rfft(a, n=m2)
    fft_w = np.fft.rfft(w[::-1], n=m2)   # reversed w for convolution
    conv_full = np.fft.irfft(fft_a * fft_w, n=m2)[:n + len_w -1]
    # g_i = conv_full[ len_w - 1 - np.arange(n) ]  # len_w = 2n-1
    # indices: start = (len_w - 1) - (n-1) = n-1; end = len_w - 1 = 2n-2
    start = len_w - 1 - (n - 1) # = n-1
    # slice indices from start to start + n

import time
import numpy as np
import math

def _objective(seq: np.ndarray) -> float:
    n = seq.shape[0]
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    m = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft * fft, n=m)[: 2 * n - 1]
    max_conv = np.max(conv)
    return float(2.0 * n * max_conv / (s * s))

def _auto_convolution(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fft * fft, n=m)[: 2 * n - 1]
    return conv

def _gradient_cross_correlation(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """Return 2 * sum_j a_j * p_{i+j} for i=0..n-1 (no alpha factor)."""
    n = a.shape[0]
    len_p = p.shape[0]       # = 2*n-1
    p_rev = p[::-1]
    m2 = 1 << ((n + len_p - 1).bit_length())   # >= 3n-2
    fft_a = np.fft.rfft(a, n=m2)
    fft_pr = np.fft.rfft(p_rev, n=m2)
    conv_full = np.fft.irfft(fft_a * fft_pr, n=m2)[: n + len_p - 1]   # 3n-2
    # slice indices n-1 .. 2n-2 (inclusive) -> length n
    mid = conv_full[n-1: 2*n-1]
    # reverse order to obtain g_i = sum_j a_j p_{i+j}
    grad = 2.0 * mid[::-1]
    return grad

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a low‑score non‑negative step function.
    The algorithm consists of three stages:
    1) A cheap grid search on a three‑parameter family
       a_i = 1 / ((i + c)^β + d) (β≈0.5) with n≈5000.
    2) Upscaling the best primitive to a larger dimension (≈20000)
       and running a smoothed–max gradient descent.
       The smoothing is realised by a log‑sum‑exp (controlled by α)
       which gradually turns into the true max‑norm.
       An Adam optimiser together with projection onto the simplex (fixed L1)
       keeps the iterates feasible.
    3) The best sequence discovered during the optimisation is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1. Coarse grid search on a small instance (n_grid = 5000)
    # ------------------------------------------------------------------
    n_grid = 5000
    target_sum_grid = math.sqrt(2.0 * n_grid)

    betas = np.linspace(0.45, 0.55, 11)       # β around the theoretical optimum
    offsets = np.linspace(0.0, 5.0, 11)       # additive shift inside the root
    ds = np.linspace(0.0, 5.0, 11)           # extra denominator offset

    best_params = None
    best_score = np.inf
    best_seq_grid = None

    print("[*] Starting coarse grid search (n={})".format(n_grid))

    for beta in betas:
        for c in offsets:
            for d in ds:
                idx = np.arange(n_grid, dtype=np.float64) + c
                a = 1.0 / (np.power(idx, beta) + d)
                # enforce non‑negativity (already true) and normalise L1
                s = a.sum()
                if s < 0.01:
                    continue
                a *= target_sum_grid / s
                score = _objective(a)
                if score < best_score:
                    best_score = score
                    best_params = (beta, c, d)
                    best_seq_grid = a.copy()
                    print("[grid] β={:.3f}, c={:.2f}, d={:.2f} → score={:.6f}".format(beta, c, d, score))
                if time.time() > deadline:
                    break
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    if best_seq_grid is None:
        # fallback to simple power law
        beta, c, d = 0.5, 0.0, 0.0
        idx = np.arange(n_grid, dtype=np.float64) + c
        best_seq_grid = 1.0 / (np.sqrt(idx) + d)
        best_seq_grid *= target_sum_grid / best_seq_grid.sum()
        best_score = _objective(best_seq_grid)

    print("[*] Grid search finished. Best β={:.3f}, c={:.2f}, d={:.2f}, score={:.6f}".format(*best_params, best_score))

    # ------------------------------------------------------------------
    # 2. Upscale to a larger dimension and start smoothed‑max descent.
    # ------------------------------------------------------------------
    n = 20000                # larger dimension gives more freedom
    target_sum = math.sqrt(2.0 * n)

    beta, c, d = best_params
    idx = np.arange(n, dtype=np.float64) + c
    a = 1.0 / (np.power(idx, beta) + d)
    a *= target_sum / a.sum()

    # initial statistics
    conv_cur = _auto_convolution(a)
    max_conv = conv_cur.max()
    best_global_score = float(2.0 * n * max_conv / (target_sum * target_sum))
    best_global_seq = a.copy()
    print("[*] Starting smoothed‑max optimisation, initial score = {:.6f}".format(best_global_score))

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.05                 # base learning rate
    t = 0

    # smoothing (log‑sum‑exp) schedule
    alpha = 0.5
    alpha_max = 30.0
    alpha_mul = 1.006

    no_improve = 0

    while time.time() < deadline:
        # ----- compute smoothed weights from current convolution -----
        max_conv = conv_cur.max()
        shifted = conv_cur - max_conv          # keep numbers ≤ 0
        w = np.exp(alpha * shifted)            # w_i = exp(alpha*(c_i - max))
        w_sum = w.sum()
        p = w / w_sum                          # probability vector over 2n‑1 entries

        # ----- gradient of the smoothed objective (sub‑gradient of max) -----
        grad_base = _gradient_cross_correlation(a, p)   # already contains factor 2
        grad = alpha * grad_base                        # full gradient

        # ----- Adam update (projected onto the simplex) -----
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.clip(a_new, 0.0, None)

        # renormalise the ℓ₁‑norm (scaling does not affect the score)
        s = a_new.sum()
        if s < 1e-12:
            break
        a_new *= target_sum / s

        # ----- evaluate new point -----
        conv_new = _auto_convolution(a_new)
        new_score = float(2.0 * n * conv_new.max() / (target_sum * target_sum))

        # keep the best found so far
        if new_score < best_global_score:
            best_global_score = new_score
            best_global_seq = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # occasional random jitter to escape flat regions
            if no_improve % 200 == 0:
                i, j = np.random.randint(n, size=2)
                delta = a_new[i] * 0.05 * np.random.rand()
                a_new[i] = max(0.0, a_new[i] - delta)
                a_new[j] += delta
                a_new *= target_sum / a_new.sum()
                # make optimiser forget the previous direction a bit
                m *= 0.5
                v *= 0.5

        # prepare next iteration
        a = a_new
        conv_cur = conv_new

        # increase smoothing strength slowly
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # progress report
        if t % 100 == 0:
            print("[iter {:5d}] score={:.6f} alpha={:.3f}".format(t, new_score, alpha))

        if best_global_score < 1.503:
            print("[!] Desired threshold crossed: {:.6f} after {} iterations".format(best_global_score, t))
            break

    print("[+] Optimisation finished. Best score = {:.6f}".format(best_global_score))
    return [float(v) for v in best_global_seq]

"""
Low‑score step‑function generator.

The search proceeds in three phases:

1. **Coarse grid search** – a three‑parameter family
       a_i = 1 / ((i + c)^β + d)   (β≈0.5) is evaluated on a modest
   size (n≈5 000).  The best parameters are kept.

2. **Upscaling & smoothed‑max descent** – the best primitive is
   up‑scaled to a larger dimension (n≈20 000).  The objective
   ``max(conv(a,a))`` is approximated by a log‑sum‑exp with a
   temperature α that grows during optimisation.  The gradient of the
   smoothed objective is
        ∂F/∂a_i = 2·α·∑_j a_j·p_{i+j},
   where p is the normalised exponential weight vector.
   An Adam optimiser together with projection onto the ℓ₁‑simplex
   (fixed sum) keeps all iterates feasible (non‑negative, constant
   ℓ₁‑norm).

3. **Return the best sequence** – the sequence with the smallest
   evaluation score encountered during the run is returned.

The algorithm respects the time budget (default 1000 s) and the
non‑negativity / clipping rules enforced inside ``evaluate_sequence``.
"""

import time
import math
import numpy as np


def _objective(seq: np.ndarray) -> float:
    """Fast evaluation of the score 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fft = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft * fft, n=m)[: 2 * n - 1]
    max_conv = np.max(conv)
    return float(2.0 * n * max_conv / (s * s))


def _auto_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a via FFT."""
    n = a.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fft * fft, n=m)[: 2 * n - 1]
    return conv


def _gradient_cross_correlation(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Compute 2·∑_j a_j·p_{i+j} for i=0…n‑1 (no α factor).

    This is a cross‑correlation between ``a`` and ``p`` obtained
    efficiently with an FFT‑based convolution of ``a`` and the reversed
    weight vector ``p``.
    """
    n = a.shape[0]
    len_p = p.shape[0]            # = 2·n‑1
    p_rev = p[::-1]
    m2 = 1 << ((n + len_p - 1).bit_length())   # ≥ 3·n‑2
    fft_a = np.fft.rfft(a, n=m2)
    fft_pr = np.fft.rfft(p_rev, n=m2)
    conv_full = np.fft.irfft(fft_a * fft_pr, n=m2)[: n + len_p - 1]  # length 3·n‑2
    # slice indices n‑1 … 2·n‑2 (inclusive) -> length n
    middle = conv_full[n - 1: 2 * n - 1]
    # reverse to obtain the required ordering
    return 2.0 * middle[::-1]

    """

    # ------------------------------------------------------------------

    offsets = np.linspace(0.0, 5.0, 11)   # additive shift inside the power
    ds = np.linspace(0.0, 5.0, 11)       # extra denominator offset

    best_score = np.inf
    best_params = None
    best_seq_grid = None

    print("[*] Starting coarse grid search (n = {})".format(n_grid))

        for c in offsets:
            for d in ds:

                cand = 1.0 / (np.power(idx, beta) + d)

                s = cand.sum()
                if s < 0.01:

                cand *= target_sum_grid / s

                    best_params = (beta, c, d)
                    best_seq_grid = cand.copy()
                    print("[grid] β={:.3f}, c={:.2f}, d={:.2f} → score={:.6f}"
                          .format(beta, c, d, score))

                if time.time() > deadline:
                    break
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    if best_seq_grid is None:          # very unlikely – fallback to a simple power law
        beta, c, d = 0.5, 0.0, 0.0
        idx = np.arange(n_grid, dtype=np.float64) + c
        best_seq_grid = 1.0 / (np.sqrt(idx) + d)
        best_seq_grid *= target_sum_grid / best_seq_grid.sum()
        best_score = _objective(best_seq_grid)
        best_params = (beta, c, d)

    print("[*] Grid search finished – best β={:.3f}, c={:.2f}, d={:.2f}, score={:.6f}"
          .format(*best_params, best_score))

    # ------------------------------------------------------------------
    # 2. Upscale to a larger size and run smoothed‑max gradient descent.
    # ------------------------------------------------------------------
    n = 20000                            # larger dimension for extra flexibility
    target_sum = math.sqrt(2.0 * n)

    beta, c, d = best_params
    idx = np.arange(n, dtype=np.float64) + c
    a = 1.0 / (np.power(idx, beta) + d)
    a *= target_sum / a.sum()

    # initial convolution and score
    conv_cur = _auto_convolution(a)
    best_global_score = float(2.0 * n * conv_cur.max() / (target_sum * target_sum))
    best_global_seq = a.copy()
    print("[*] Starting smoothed‑max optimisation, initial score = {:.6f}"
          .format(best_global_score))

    # Adam optimiser state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.05                     # base learning rate
    t = 0

    # log‑sum‑exp smoothing schedule
    alpha = 0.5
    alpha_max = 30.0
    alpha_mul = 1.006

    no_improve = 0

    while time.time() < deadline:
        # ----- weights for the smoothed max -----
        cur_max = conv_cur.max()
        shifted = conv_cur - cur_max                 # ≤ 0, prevents overflow
        w = np.exp(alpha * shifted)
        w_sum = w.sum()
        p = w / w_sum                               # probability vector over 2·n‑1 entries

        # ----- gradient of the smoothed objective -----
        grad_base = _gradient_cross_correlation(a, p)   # already contains factor 2
        grad = alpha * grad_base                       # full gradient

        # ----- Adam update (projected onto the ℓ₁‑simplex) -----
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.clip(a_new, 0.0, None)

        # renormalise the ℓ₁‑norm (scaling does not affect the score)
        s = a_new.sum()
        if s < 1e-12:
            break
        a_new *= target_sum / s

        # ----- evaluate the new candidate -----
        conv_new = _auto_convolution(a_new)
        new_score = float(2.0 * n * conv_new.max() / (target_sum * target_sum))

        # keep the best sequence seen so far
        if new_score < best_global_score:
            best_global_score = new_score
            best_global_seq = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # occasional random jitter to help escape plateaus
            if no_improve % 200 == 0:
                i, j = np.random.randint(n, size=2)
                delta = a_new[i] * 0.05 * np.random.rand()
                a_new[i] = max(0.0, a_new[i] - delta)
                a_new[j] += delta
                a_new *= target_sum / a_new.sum()
                m *= 0.5
                v *= 0.5

        # prepare next iteration
        a = a_new
        conv_cur = conv_new

        # increase smoothing strength gradually
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # progress report
        if t % 100 == 0:
            print("[iter {:5d}] score={:.6f} α={:.3f}".format(t, new_score, alpha))

        # early exit if the target is already beaten
        if best_global_score < 1.503:
            print("[!] Target threshold crossed: {:.6f} after {} iterations"
                  .format(best_global_score, t))
            break

    print("[+] Optimisation finished. Best score = {:.6f}".format(best_global_score))
    return [float(v) for v in best_global_seq]
