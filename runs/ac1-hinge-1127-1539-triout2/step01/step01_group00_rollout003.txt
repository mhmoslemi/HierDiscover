    n = a.shape[0]
    # pick top indices: using np.argpartition.
    # we can take top m = min(10, number of indices)
    m = 10
    if 2*n-1 <= m:
        idxs = np.arange(2*n-1)
    else:
        # find indices of top m values
        idxs = np.argpartition(conv, -m)[-m:]
    # compute weights: w = np.exp((conv[idxs]-max_val)/temp)
    temp = max_val * 0.1
    w_raw = np.exp((conv[idxs] - max_val) / temp)
    w = w_raw / w_raw.sum()
    # accumulate gradient
    g = np.zeros_like(a)
    for weight, k in zip(w, idxs):
        # compute gradient for this k

        # valid where partner in [0,n)
        idx = np.arange(n)
        partner = k - idx

        valid = (partner >= 0) & (partner < n)

"""
Improved search for a non‑negative step sequence that minimises the
self‑convolution peak.

Algorithm
---------
* Work with the equivalent objective  max_k (a*a)[k]  (the value returned by
  ``evaluate_sequence`` when the L1‑norm is normalised to √(2·n)).
* For a given length *n* initialise a random Dirichlet vector and scale it to
  sum = √(2·n).  If a previously found sequence ``height_sequence_1`` exists it
  is also used as a starting point (with a small jitter).
* Repeatedly:
    - Compute the linear convolution via FFT.
    - Pick the *m* largest convolution entries (default m=10) and form a
      weighted sub‑gradient: the gradient of each selected entry is known
      analytically (2·a[j‑i] except on the diagonal where it is a[i]).
      The weights are a softmax of the selected values (temperature ~5 % of the
      current maximum) so that several “almost‑maximal’’ peaks are driven down
      simultaneously.
    - Perform a multiplicative update ``a ← a·exp(−η·g)`` (guaranteeing
      non‑negativity) and renormalise to keep the L1‑norm fixed.
    - Occasionally add a tiny Gaussian jitter to escape plateaus.
* The inner optimisation runs for a few seconds (or until stagnation); many
  independent restarts with different lengths and random seeds are performed
  within the total time budget.  The best sequence over all restarts is
  returned.
* No external I/O or filesystem access is performed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _conv_full(a: np.ndarray) -> tuple[np.ndarray, float, int]:
    """
    Compute the linear convolution a*a using FFT.
    Returns (full_convolution, max_value, index_of_max).
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()   # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    max_idx = int(np.argmax(conv))
    return conv, float(conv[max_idx]), max_idx


def _combined_gradient(
    a: np.ndarray,
    conv: np.ndarray,
    max_val: float,
    m: int = 10,
    temperature_factor: float = 0.05,
) -> np.ndarray:
    """
    Compute a weighted sub‑gradient using the *m* largest convolution entries.
    The weight of each selected entry is a softmax with temperature
    ``temperature_factor * max_val``.
    """
    n = a.shape[0]
    total_len = conv.shape[0]

    m = min(m, total_len)
    # indices of the m largest convolution entries
    idxs = np.argpartition(conv, -m)[-m:]

    # temperature for the softmax (avoid zero)
    temp = max_val * temperature_factor
    if temp < 1e-12:
        temp = 1e-12

    w_raw = np.exp((conv[idxs] - max_val) / temp)
    w = w_raw / w_raw.sum()

    g = np.zeros_like(a)

    idx = np.arange(n)

    for weight, k in zip(w, idxs):
        partner = k - idx
        valid = (partner >= 0) & (partner < n)

        # off‑diagonal contribution (2*a[partner])
        g[valid] += weight * 2.0 * a[partner[valid]]

        # diagonal correction when k is even and i=j
        if k % 2 == 0:
            d = k // 2
            if 0 <= d < n:
                # we added 2*a[d]; the true derivative is a[d]
                g[d] -= weight * a[d]

    return g


def _initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random non‑negative sequence of length n with
    ∑a = √(2·n).  A Dirichlet(1,…,1) draw is used.
    """
    S = np.sqrt(2.0 * n)
    a = rng.random(n) + 1e-8      # ensure strictly positive to avoid division by zero
    a = a / a.sum() * S
    return a


def _optimize_local(
    a_start: np.ndarray,
    rng: np.random.Generator,
    deadline: float,
    m_peaks: int = 10,
) -> tuple[np.ndarray, float]:
    """
    Local optimisation starting from ``a_start``.
    Returns the best sequence found (within the given time) and its
    maximal convolution value.
    """
    a = a_start.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # initial objective
    _, best_val, _ = _conv_full(a)
    best_seq = a.copy()

    # hyper‑parameters
    eta = 0.5                  # initial step size for multiplicative update
    decay = 0.9995             # exponential decay of η
    jitter_prob = 0.02         # probability of a small random perturbation
    stagnation_limit = 1500    # stop after this many iterations without improvement

    steps_without_improve = 0

    while time.time() < deadline and steps_without_improve < stagnation_limit:
        conv, cur_max, _ = _conv_full(a)

        # weighted gradient from the top peaks
        g = _combined_gradient(a, conv, cur_max, m=m_peaks, temperature_factor=0.05)

        # multiplicative update (guarantees non‑negativity)
        a = a * np.exp(-eta * g)

        # renormalise to keep the L1‑norm fixed
        total = a.sum()
        if total == 0.0:
            a = np.full(n, S / n)
        else:
            a = a * (S / total)

        # evaluate new maximum
        _, new_max, _ = _conv_full(a)

        if new_max < best_val - 1e-12:
            best_val = new_max
            best_seq = a.copy()
            steps_without_improve = 0
        else:
            steps_without_improve += 1

        # step‑size decay
        eta *= decay

        # occasional jitter to jump out of shallow plateaus
        if rng.random() < jitter_prob:
            noise = rng.normal(scale=a.mean() * 0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = a * (S / a.sum())

    return best_seq, best_val

            global_best_val = val
            global_best_seq = seq.copy()
            print(
                f"[global improve] restart={restart_id}, time={time.time():.1f}s, "
                f"len={len(seq)}, max_conv={global_best_val:.6f}"
            )

    # ------------------------------------------------------------------
    # Final reporting
    # ------------------------------------------------------------------
    if global_best_seq is None:
        # fall back to a simple uniform sequence if everything failed
        n = 5000
        global_best_seq = _initial_sequence(n, rng)

    print(
        f"[final] best max convolution = {global_best_val:.6f}, "
        f"length = {len(global_best_seq)}"
    )
    # The returned list is directly usable by ``evaluate_sequence``.
    return global_best_seq.tolist()
