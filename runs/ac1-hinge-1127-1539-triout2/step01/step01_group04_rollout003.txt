"""
search for a non‑negative step function that minimises
evaluate_sequence.

The method combines several ideas:
*   A small pool of deterministic initial shapes (sqrt‑difference,
    simple power‑law, cosine, sine and the previously discovered
    construction) is evaluated and the best one (lowest self‑convolution
    peak) is chosen as a starting point.
*   From this start an Adam‑accelerated sub‑gradient descent is run.
    The sub‑gradient of the infinity‑norm of the self‑convolution
    (active shifts where the maximum is attained) is cheap to compute.
    Adam’s adaptive per‑coordinate steps together with a diminishing
    learning‑rate give a much faster reduction than plain projected
    sub‑gradient descent.
*   Every few hundred iterations a tiny Gaussian jitter is added and the
    sequence is re‑normalised – this helps to escape flat plateaus.
*   All operations (FFT‑based convolution, gradient, projection) are
    O(n log n) and the loop is stopped a little before the given time
    budget, guaranteeing a result is always returned.

The returned list is guaranteed to satisfy the non‑negativity checks of
evaluate_sequence.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – they are defined at top level (no closures)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Compute the (linear) self‑convolution of a real vector using a power‑of‑two
    FFT length L (L ≥ 2·len(a)‑1).  Returns the first 2·len(a)‑1 entries.
    """
    fa = np.fft.rfft(a, n=L)
    # element‑wise product gives the convolution in the frequency domain
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * a.size - 1]


def _add_gradient_for_k(grad: np.ndarray, a: np.ndarray, k: int) -> None:
    """
    Add the sub‑gradient contribution of shift `k` to the gradient array.

    For a fixed shift `k` the derivative of the convolution value
    (a*a)[k] w.r.t. a[i] is 2·a[k‑i] whenever the index is valid.
    The function updates `grad` in‑place.
    """
    n = a.size
    if k < n:
        # indices 0 … k  (the other side of the anti‑diagonal is zero)
        grad[: k + 1] += 2.0 * a[k :: -1]
    else:
        # indices k‑(n‑1) … n‑1
        start = k - n + 1
        length = 2 * n - 1 - k
        grad[start:] += 2.0 * a[: length][::-1]

        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 7390   # length of the best known construction

    # target L1‑norm (the normalisation used in the objective)
    target_sum = math.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Build a small pool of deterministic initial shapes.
    # The best one (lowest self‑convolution peak) will be used.
    # --------------------------------------------------------------
    candidates = []

    # 1) sqrt‑difference (used in the reference solution)
    idx = np.arange(n, dtype=np.float64)
    a_sqrt = np.sqrt(idx + 1.0) - np.sqrt(idx)
    a_sqrt *= target_sum / a_sqrt.sum()
    candidates.append(a_sqrt)

    # 2) power‑law a_i = ((n‑i)/n)^p for several exponents p
    base = (np.arange(n, dtype=np.float64)[::-1] + 1.0) / n   # (n‑i)/n
    for p in np.linspace(0.2, 5.0, 15):
        a = base ** p
        a *= target_sum / a.sum()
        candidates.append(a)

    # 3) cosine shape  a_i = cos(π/2 * i/(n‑1))
    a_cos = np.cos(np.pi * np.arange(n, dtype=np.float64) / (2.0 * (n - 1)))
    a_cos *= target_sum / a_cos.sum()
    candidates.append(a_cos)

    # 4) sine shape (mirrored cosine)
    a_sin = np.sin(np.pi * (np.arange(n, dtype=np.float64)[::-1]) / (2.0 * (n - 1)))
    a_sin *= target_sum / a_sin.sum()
    candidates.append(a_sin)

    # 5) the previously discovered construction, if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        prev_sum = prev.sum()
        if prev_sum > 0.0:
            prev = prev * (target_sum / prev_sum)
            candidates.append(prev)
    except Exception:
        pass

    # Pre‑compute FFT length (next power of two ≥ 2n‑1)
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Pick the best initial candidate according to the self‑convolution peak
    # ------------------------------------------------------------------
    best_start = None
    best_score = float("inf")
    for cand in candidates:
        conv = _conv_fft(cand, L)
        score = float(conv.max())
        if score < best_score:
            best_score = score
            best_start = cand.copy()
    # Fallback safety
    if best_start is None:
        a = np.full(n, target_sum / n, dtype=np.float64)
    else:
        a = best_start

    # Save the best sequence ever seen (initially the start)
    best_seq = a.copy()
    best_score = float(_conv_fft(a, L).max())

    print(f"Initial best score (self‑convolution peak) : {best_score:.6f}")

    # ------------------------------------------------------------------
    # Adam optimiser state
    # ------------------------------------------------------------------
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow = beta1
    beta2_pow = beta2
    iter_count = 0

    # Learning‑rate schedule – start fairly aggressive, then decay slowly
    lr0 = 0.20

    # Jitter configuration
    jitter_interval = 500       # iterations
    jitter_sigma_factor = 1e-3  # relative to target_sum / sqrt(n)

    # Main optimisation loop
    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) Compute self‑convolution and its current maximum.
        # --------------------------------------------------------------
        conv = _conv_fft(a, L)
        max_b = conv.max()
        cur_score = float(max_b)          # equal to evaluate_sequence() value

        # Keep a copy of the best encountered sequence
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"Iter {iter_count:8d} – new best {best_score:.6f}")

        # --------------------------------------------------------------
        # 2) Sub‑gradient of the infinity‑norm (active shifts only)
        # --------------------------------------------------------------
        eps = 1e-12 * max_b
        active_shifts = np.where(conv >= max_b - eps)[0]
        grad = np.zeros_like(a)
        for k in active_shifts:
            _add_gradient_for_k(grad, a, int(k))

        # --------------------------------------------------------------
        # 3) Adam update (projected onto non‑negative orthant and fixed sum)
        # --------------------------------------------------------------
        iter_count += 1
        # bias‑corrected moment estimates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        # decaying base learning‑rate
        lr = lr0 / (1.0 + iter_count * 1e-4)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a -= step

        # Project onto feasible set: a >= 0 and sum = target_sum
        a = np.maximum(a, 0.0)

        s = a.sum()
        if s <= 0.0:
            # degenerate case – fall back to uniform distribution
            a = np.full(n, target_sum / n, dtype=np.float64)
        else:
            a *= (target_sum / s)

        # --------------------------------------------------------------
        # 4) Random jitter every few hundred iterations (helps escape plateaus)
        # --------------------------------------------------------------
        if iter_count % jitter_interval == 0:
            sigma = jitter_sigma_factor * target_sum / math.sqrt(n)
            noise = np.random.randn(n) * sigma
            a = np.maximum(a + noise, 0.0)
            a *= (target_sum / a.sum())

        # --------------------------------------------------------------
        # 5) Update the bias‑correction powers for the next iteration
        # --------------------------------------------------------------
        beta1_pow *= beta1
        beta2_pow *= beta2

    # ------------------------------------------------------------------
    # Return the best sequence discovered.
    # ------------------------------------------------------------------
    print(f"Finished after {iter_count} iterations – best score {best_score:.6f}")
    return best_seq.tolist()
