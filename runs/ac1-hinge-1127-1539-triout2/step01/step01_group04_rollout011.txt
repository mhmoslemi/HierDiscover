def propose_candidate(seed=..., budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1

    # Choose n maybe 3000
    n = len(height_sequence_1) if defined else 3000

    a = np.sqrt(np.arange(n)+1) - np.sqrt(np.arange(n))
    S = math.sqrt(2.0 * n)
    a = a * (S / a.sum())

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print("initial", best_score)

    # precompute FFT length for conv and w calculation (max length 2n-1)
    L = 1 << ((2 * n - 1).bit_length())
    # We'll also need length for correlation with w (size 2n-1 + n - 1 = 3n-2) => L2
    L2 = 1 << ((3 * n - 2).bit_length())

    iteration = 0
    alpha = 5.0
    alpha_max = 100.0
    alpha_inc = 0.02  # increase slowly

    while time.time() < deadline:
        # compute self conv via FFT length L
        fa = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
        M = conv.max()
        # approximate objective
        # compute smooth max
        exp_term = np.exp(alpha * (conv - M))
        sum_exp = exp_term.sum()
        smooth_obj = M + np.log(sum_exp) / alpha  # approximate max
        # track best based on M (actual evaluation)
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"Iter {iteration} new best max {best_score:.6f}")
        # compute weights
        w = exp_term / sum_exp  # shape (2n-1,)
        # compute gradient = 2 * correlation a⋆w
        rev_w = w[::-1]
        # compute correlation via FFT
        fa2 = np.fft.rfft(a, n=L2)
        fw_rev = np.fft.rfft(rev_w, n=L2)
        corr_full = np.fft.irfft(fa2 * fw_rev, n=L2)[: 3 * n - 2]
        # gradient indices for p=0..n-1
        grad = 2.0 * corr_full[(2 * n - 2) - np.arange(n)]

        # step size schedule: lr = c / sqrt(iter+1)
        # Use line search: try decreasing lr until smooth_obj improves or until small
        c = 0.5  # base learning rate
        lr = c / math.sqrt(iteration + 1.0)
        # direction is negative gradient (descend)
        a_new = a - lr * grad

        # project to nonnegative
        a_new = np.maximum(a_new, 0.0)
        # enforce sum S
        s = a_new.sum()
        if s == 0.0:
            a_new = np.full(n, S / n)
        else:
            a_new *= (S / s)

        a = a_new

        # occasionally increase smoothing alpha
        if iteration % 200 == 0 and alpha < alpha_max:
            alpha = min(alpha_max, alpha * 1.2)

        # occasional jitter
        if iteration % 500 == 0 and iteration > 0:
            sigma = S * 1e-3 / math.sqrt(n)
            a += np.random.randn(n) * sigma
            a = np.maximum(a, 0.0)
            a *= (S / a.sum())

        iteration += 1

    # Return best_seq
    print(f"Finished {iteration} iterations, best {best_score}")
    return best_seq.tolist()

"""
Improved optimisation of a non‑negative step function.

The algorithm builds on the sub‑gradient method that already reaches
≈π/2 ≈ 1.5748, but adds a few ingredients that make it capable of
pushing the supremum of the self‑convolution lower:

* **Momentum** – a simple exponential moving average of the sub‑gradient
  (β≈0.9) yields a more stable descent direction.
* **Adaptive line‑search** – the step size is reduced until the maximum
  of the convolution does not increase, guaranteeing monotonic progress
  for the majority of iterations.
* **Dynamic jitter** – every few dozen iterations a tiny normally
  distributed perturbation is added and the sequence re‑normalised.
* **Warm start** – if a previous construction (`height_sequence_1`)
  is available it is used, otherwise the classic √‑difference
  initialization is taken.

All heavy work (self‑convolution, gradient updates) is performed with
FFT‑based convolutions, so thousands of iterations run well within the
1000 s budget even for a few thousand coefficients.  The best sequence
found is returned as a plain list; the surrounding harness will call
`evaluate_sequence` on it for the final score.
"""

import time
import math
import numpy as np

def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution of a real vector using an FFT of length L.
    Returns the first (2·len(a)‑1) entries.
    """
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * a.size - 1]

def _add_gradient_for_k(grad: np.ndarray, a: np.ndarray, k: int) -> None:
    """
    Add the sub‑gradient contribution coming from the active shift k.
    The contribution is 2·a_{k‑i} for all i where the term exists.
    """
    n = a.size
    if k < n:
        # indices 0 … k
        grad[: k + 1] += 2.0 * a[k :: -1]
    else:
        # indices k‑(n‑1) … n‑1
        start = k - n + 1
        length = 2 * n - 1 - k
        grad[start:] += 2.0 * a[: length][::-1]

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) minimises
    the provided ``evaluate_sequence`` function.

    Parameters
    ----------
    seed : int
        Random seed – makes the search deterministic.
    budget_s : int
        Time budget in seconds.

    Returns
    -------
    list[float]
        The best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a small safety margin

    # ------------------------------------------------------------------
    # Choose a length.  Prefer the previously discovered sequence if it
    # exists – it is already close to optimal – otherwise use a modest
    # default (≈ 3000) that keeps each iteration cheap.
    # ------------------------------------------------------------------
    try:
        a0 = np.array(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        n = a0.size
    except Exception:
        n = 3000
        idx = np.arange(n, dtype=np.float64)
        a0 = np.sqrt(idx + 1.0) - np.sqrt(idx)   # a_i = √(i+1) – √i

    # Normalise to the required ℓ¹‑norm √(2 n)
    target_sum = math.sqrt(2.0 * n)
    a = a0 * (target_sum / a0.sum())

    # Keep the best sequence seen so far.
    best_seq = a.copy()
    try:
        best_score = float(evaluate_sequence(best_seq.tolist()))   # type: ignore[name-defined]
    except Exception:
        best_score = float("inf")
    print(f"Initial exact score: {best_score:.6f}")

    # FFT length for the self‑convolution (next power of two ≥ 2n‑1)
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Optimisation loop
    # ------------------------------------------------------------------
    iteration = 0
    momentum = np.zeros_like(a)
    beta = 0.9                # momentum decay
    base_lr = 0.8             # initial step‑size multiplier

    while time.time() < deadline:
        # 1) current self‑convolution and its maximum
        conv = _conv_fft(a, L)
        M = conv.max()

        # Record the best result (objective = max conv when ℓ¹‑norm is fixed)
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"Iter {iteration:6d} – new best max {best_score:.6f}")

        # 2) sub‑gradient coming from all (near‑)active shifts
        eps = 1e-12 * M
        ks = np.where(conv >= M - eps)[0]   # indices where conv is essentially maximal
        grad = np.zeros_like(a)
        for k in ks:
            _add_gradient_for_k(grad, a, k)

        # 3) momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # 4) propose a descent step with a simple back‑tracking line search
        lr = base_lr / math.sqrt(iteration + 1.0)
        accepted = False
        a_candidate = a.copy()

        for _ in range(5):   # at most five reductions of the step size
            step = lr * momentum
            a_candidate = a - step
            a_candidate = np.maximum(a_candidate, 0.0)                # non‑negativity

                a_candidate = np.full(n, target_sum / n, dtype=np.float64)
            else:

            # evaluate the candidate (fast – using FFT)
            conv_cand = _conv_fft(a_candidate, L)
            M_cand = conv_cand.max()

            if M_cand <= M:                                            # monotonic improvement
                accepted = True
                break
            # otherwise shrink the step size and try again
            lr *= 0.5

        if accepted:
            a = a_candidate
        else:
            # No improvement despite the back‑track – keep the old a (but still apply a tiny jitter later)
            pass

        # 5) occasional tiny random jitter to escape flat regions
        if iteration % 80 == 0 and iteration > 0:

            a += np.random.randn(n) * sigma
            a = np.maximum(a, 0.0)
            a *= (target_sum / a.sum())

        iteration += 1

    print(f"Finished after {iteration} iterations – best max {best_score:.6f}")
    return best_seq.tolist()
