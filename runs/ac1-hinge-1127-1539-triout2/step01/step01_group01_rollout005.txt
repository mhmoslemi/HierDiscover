    beta = 0.9
    rel_tol = 0.02  # active set tolerance - start moderate

    iteration = 0
    no_improve = 0
    last_improve_iter = 0
    while time.time() < deadline:
        # compute gradient
        conv = conv_self_fft(a)
        max_val = conv.max()
        act = np.where(conv >= max_val * (1 - rel_tol))[0]
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in act:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2 * a[j[mask]]
        # subgradient direction
        d = -g + np.mean(g)
        # momentum
        direction = beta * direction_prev + (1 - beta) * d
        direction_prev = direction
        # propose new a
        a_new = a + eta * direction
        a_new = np.clip(a_new, 0.0, None)
        a_new = project_to_sum(a_new, target_sum)
        # evaluate
        new_score = evaluate_sequence(a_new.tolist())
        iteration += 1
        if new_score + 1e-12 < best_score:
            best_score = new_score
            best_seq = a_new.copy()
            print(f"Iter ... new best {best_score}")
        if new_score + 1e-12 < current_score:
            # improvement
            a = a_new
            current_score = new_score
            no_improve = 0
            # increase step size a bit
            eta *= 1.02
        else:

            no_improve += 1
            # reject a_new (stay with a)

            eta *= 0.5

"""
Optimiser for the self‑convolution bound (improved).

The optimizer works in three stages:

1. **Extensive parametric sweep** – a large family of symmetric “U‑shapes’’
   (linear, quadratic, cubic, quartic, exponential, and pure‑power shapes)
   is generated.  Each shape is normalised to the required ℓ₁‑norm
   (√(2 n)).  The best candidate from this sweep is used as a starting point.

2. **Projected momentum sub‑gradient descent** – for the current sequence the
   active set of indices where the self‑convolution attains (or almost attains)
   its maximum is identified.  The sub‑gradient of the max‑convolution is
   computed, orthogonalised to the all‑ones vector (so the ℓ₁‑norm stays
   unchanged), combined with a momentum term and a step is taken.
   After the step the vector is clipped, symmetrised (a ← (a+a[::-1])/2)
   and re‑scaled to the target sum.

   Step‑size is increased after a successful move and reduced after a failure,
   while the active‑set tolerance is slowly widened.  This phase runs for the
   bulk of the time budget.

3. **Escaping stagnation** – if no improvement is seen for many iterations
   a small random jitter is injected, and with a low probability a biased
   “edge‑transfer’’ (moving a little mass from the centre to the borders) is
   performed.  The best sequence ever seen is always remembered.

The code never calls ``evaluate_sequence`` inside the search loop – the
objective is computed directly as the maximum entry of the linear convolution
as this is exactly the value returned by ``evaluate_sequence`` once the sum is
normalised.  Only the final sequence is returned as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution of a with itself using a power‑of‑two FFT.
    Returns the full convolution of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """
    Scale a non‑negative vector to have ℓ₁‑norm equal to *target*.
    Zero entries stay zero; if the whole vector is zero a uniform distribution
    is returned.
    """
    s = v.sum()
    if s > 0.0:
        return v * (target / s)
    return np.full_like(v, target / v.shape[0])


def symmetrise(v: np.ndarray) -> np.ndarray:
    """Enforce mirror symmetry: v[i] = v[n‑1‑i]."""
    return (v + v[::-1]) / 2.0


def max_self_conv(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution of *a*."""
    return conv_self_fft(a).max()


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """Move a tiny fraction of mass from a random source index to a random target."""
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0.0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_to_sum(a, target_sum)


def edge_transfer(a: np.ndarray, target_sum: float, scale: float = 0.02) -> np.ndarray:
    """
    Transfer a small amount of mass from a centre index to a border index.
    This encourages “U‑shaped’’ structures.
    """
    n = a.shape[0]
    # centre region (30 %–70 % of the indices)
    centre = np.arange(int(n * 0.30), int(n * 0.70))
    # border region (first/last 15 % of the indices)
    borders = np.concatenate([np.arange(int(n * 0.00), int(n * 0.15)),
                              np.arange(int(n * 0.85), n)])
    if centre.size == 0 or borders.size == 0:
        return a
    i = np.random.choice(centre)
    j = np.random.choice(borders)
    delta = scale * a[i]
    if delta <= 0.0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_to_sum(a, target_sum)


def generate_shape(
    n: int,
    target_sum: float,
    kind: str,
    param: float = 0.0,
    exponent: int = 1,
) -> np.ndarray:
    """
    Produce a symmetric shape of length *n* and normalise it.
    *kind* can be:
      - "offset"  : raw = 1 + param * d**exponent
      - "pure"    : raw = d**exponent               (center = 0)
      - "exp_off" : raw = np.exp(param * d)         (offset = 1)
      - "exp_pure": raw = np.exp(param * d) - 1.0   (center = 0)
    where d = |i - mid| / mid ∈ [0, 1] .
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance in [0,1]

    if kind == "offset":
        raw = 1.0 + param * (d ** exponent)
    elif kind == "pure":
        raw = (d ** exponent)
    elif kind == "exp_off":
        raw = np.exp(param * d)
    elif kind == "exp_pure":
        raw = np.exp(param * d) - 1.0
        raw = np.clip(raw, 0.0, None)
    else:
        raise ValueError(f"unknown shape kind {kind}")
    raw = np.clip(raw, 0.0, None)
    raw = symmetrise(raw)          # ensure perfect symmetry
    return project_to_sum(raw, target_sum)

                best_score = score
                best_seq = cand.copy()
                print(f"[sweep] offset exp={exp} p={p:.2f} → {best_score:.6f}")
            if time.time() >= deadline:
                print("Time exhausted during parametric sweep.")
                return best_seq.tolist()

    # ----- pure power shapes (center exactly zero) -----
    for exp in exponents:
        cand = generate_shape(n, target_sum,
                              kind="pure",
                              exponent=exp)
        score = max_self_conv(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"[sweep] pure exp={exp} → {best_score:.6f}")
        if time.time() >= deadline:
            print("Time exhausted during pure‑power sweep.")
            return best_seq.tolist()

    # ----- exponential families -----
    for p in np.linspace(0.0, 2.5, 26):              # 0 … 2.5
        for kind in ("exp_off", "exp_pure"):
            cand = generate_shape(n, target_sum,
                                  kind=kind,
                                  param=p)
            score = max_self_conv(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"[sweep] {kind} p={p:.2f} → {best_score:.6f}")
            if time.time() >= deadline:
                print("Time exhausted during exponential sweep.")
                return best_seq.tolist()

    # ------------------------------------------------------------------
    # Phase 2 – projected momentum sub‑gradient descent
    # ------------------------------------------------------------------
    a = best_seq.copy()
    cur_score = best_score
    direction_prev = np.zeros_like(a)

    # hyper‑parameters
    eta = 0.02 * (target_sum / n)          # initial step size (~2 % of avg entry)
    beta = 0.85                           # momentum coefficient
    rel_tol = 0.01                        # active‑set tolerance
    iteration = 0
    no_improve = 0

    print(f"[{time.time():.2f}] Starting sub‑gradient descent (n={n})")
    while time.time() < deadline:
        # ----- compute active set and sub‑gradient -----
        conv = conv_self_fft(a)
        max_val = conv.max()
        # indices whose convolution value is within (1‑rel_tol) of the maximum
        act = np.where(conv >= max_val * (1.0 - rel_tol))[0]

        g = np.zeros_like(a)
        idx_arr = np.arange(n, dtype=np.int64)
        for k in act:
            j = k - idx_arr
            mask = (j >= 0) & (j < n)
            # each pair (i, j) contributes 2*a[j] to the gradient of a[i]
            g[mask] += 2.0 * a[j[mask]]

        # direction that preserves the ℓ₁‑norm (sum zero)
        direction = -g + np.mean(g)                     # orthogonal to all‑ones
        # add momentum
        direction = beta * direction_prev + (1.0 - beta) * direction
        direction_prev = direction

        # ----- tentative update -----
        a_new = a + eta * direction
        a_new = np.clip(a_new, 0.0, None)
        a_new = symmetrise(a_new)                       # keep perfect symmetry
        a_new = project_to_sum(a_new, target_sum)

        new_score = max_self_conv(a_new)
        iteration += 1

        # ----- keep the global best -----
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_seq = a_new.copy()
            print(f"Iter {iteration:06d} – new global best {best_score:.6f}")

        # ----- accept or reject the move -----
        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            no_improve = 0
            eta *= 1.03                       # a slight boost after success
        else:
            no_improve += 1
            eta *= 0.65                       # shrink after failure

        # ----- adapt parameters slowly -----
        if iteration % 2500 == 0:
            rel_tol = min(0.12, rel_tol + 0.005)   # broaden active set
            eta = max(eta, target_sum / n * 1e-6)

        # ----- occasional biased edge transfer -----
        if np.random.rand() < 0.03:
            a = edge_transfer(a, target_sum, scale=0.015)
            cur_score = max_self_conv(a)

        # ----- occasional random pair transfer (exploration) -----
        if np.random.rand() < 0.02:
            a = random_pair_transfer(a, target_sum, scale=0.005)
            cur_score = max_self_conv(a)

        # ----- jitter to escape deep plateaus -----
        if no_improve >= 4000:
            jitter = np.random.randn(n) * (target_sum * 0.002 / n)
            a = np.clip(a + jitter, 0.0, None)
            a = symmetrise(a)
            a = project_to_sum(a, target_sum)
            cur_score = max_self_conv(a)
            no_improve = 0
            print(f"Iter {iteration:06d} – jitter injected, score {cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final quick random perturbations (optional, very short)
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Sub‑gradient phase finished; best score {best_score:.6f}")

    # Return the best sequence found as plain Python list
    return best_seq.tolist()
