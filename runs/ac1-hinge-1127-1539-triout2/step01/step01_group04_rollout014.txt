def adam_optimize(a_init, n, L, budget):
    a = a_init.copy()
    target_sum = math.sqrt(2.0 * n)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8
    lr0 = 0.2
    iteration=0
    best_score = float('inf')
    best = None
    while time.time() < deadline:
        conv = _conv_fft(a, L)
        max_b = conv.max()
        # Score = max_b
        if max_b < best_score:
            best_score = max_b
            best = a.copy()
        # determine active ks
        eps_act = 1e-12 * max_b
        ks = np.where(conv >= max_b - eps_act)[0]
        grad = np.zeros_like(a)
        for k in ks:
            _add_gradient_for_k(grad, a, k)
        # average if many ks
        grad = grad / len(ks)
        # Adam update
        iteration += 1
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        # bias corrections
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        lr = lr0 / math.sqrt(iteration)  # decaying
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)
        # projection
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0:
            a = np.full(n, target_sum / n)
        else:
            a *= (target_sum / s)
        # jitter
        if iteration % 200 == 0:
            sigma = target_sum * 5e-4 / math.sqrt(n)
            noise = np.random.randn(n) * sigma
            a = np.maximum(a + noise, 0.0)
            a *= target_sum / a.sum()
        # maybe break if improvement minimal
```

def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full self‑convolution of a real vector a using an FFT of length L."""
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * a.size - 1]
    return conv


def _add_gradient_for_k(grad: np.ndarray, a: np.ndarray, k: int) -> None:
    """
    Add the sub‑gradient contribution of shift k to the gradient accumulator.
    The sub‑gradient of max (a⊛a) w.r.t. a_i for a given active shift k is
    2·a_{k‑i} (when the index is valid).
    """
    n = a.size
    if k < n:
        # indices 0 … k
        grad[: k + 1] += 2.0 * a[k::-1]
    else:
        # indices k‑(n‑1) … n‑1
        start = k - n + 1
        length = 2 * n - 1 - k
        grad[start:] += 2.0 * a[:length][::-1]

    """Return max(a ⊛ a).  The ℓ¹‑norm of a is assumed to be √(2·n)."""
    conv = _conv_fft(a, L)

    while cvxpy_available and lp_iterations < max_lp_iters:
        now = time.time()
        if now + 2.5 > deadline:          # not enough time left for another LP
            break

        # current max‑convolution M
        conv = _conv_fft(a, L)
        M = conv.max()

        # Build the Toeplitz matrix B where (B @ g)[k] = (a ⊛ g)[k]
        rows = 2 * n - 1
        B = np.zeros((rows, n), dtype=np.float64)
        for j in range(n):
            B[j:j + n, j] = a

        # Solve: maximise sum(g)  s.t. B g ≤ M , g ≥ 0
        g = cp.Variable(n, nonneg=True)
        constraints = [B @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=3000)

        if prob.status not in ("optimal", "optimal_inaccurate"):
            # LP failed – stop the LP phase.
            print(f"[LP] iteration {lp_iterations} failed (status={prob.status})")
            break

        g0 = g.value
        if g0 is None:
            print("[LP] solution is None – aborting LP phase")
            break

        sum_g = g0.sum()
        if sum_g <= 0:
            print("[LP] sum(g) <= 0 – aborting LP phase")
            break

        # Scale g to the target ℓ¹‑norm.
        g_scaled = g0 * (target_sum / sum_g)

        # Blend the old and new sequences.
        t = 0.5                     # mixing parameter
        a_new = (1.0 - t) * a + t * g_scaled

        # Projection onto the feasible set (non‑negative & fixed ℓ¹‑norm).
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.full(n, target_sum / n, dtype=np.float64)
        else:
            a_new *= (target_sum / s)

        a = a_new
        lp_iterations += 1

        # Update best record after the LP step.
        cur_score = _max_convolution(a, L)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[LP] iter {lp_iterations:2d} – new best {best_score:.6f}")

    # ------------------------------------------------------------
    # 2) Projected Adam optimisation (sub‑gradient descent).
    # ------------------------------------------------------------
    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    iteration = 0
    while time.time() < deadline:
        # Compute convolution and its maximum.
        conv = _conv_fft(a, L)
        max_b = conv.max()
        if max_b < best_score:
            best_score = max_b
            best_seq = a.copy()
            print(f"[Adam] iter {iteration:6d} – new best {best_score:.6f}")

        # Identify the active shift(s).
        eps_act = 1e-12 * max_b
        ks = np.where(conv >= max_b - eps_act)[0]
        grad = np.zeros_like(a)
        for k in ks:
            _add_gradient_for_k(grad, a, k)
        # Average over active shifts (helps when several are tied).
        grad /= max(1, len(ks))

        # Adam update (step size decays like 1/√iter).
        iteration += 1
        lr = 0.20 / math.sqrt(iteration)          # a slightly larger starter lr
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # Projection onto feasible set: non‑negative & fixed ℓ¹‑norm.
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, target_sum / n, dtype=np.float64)
        else:
            a *= (target_sum / s)

        # Tiny random jitter every 200 iterations – helps escape plateaus.
        if iteration % 200 == 0:
            sigma = target_sum * 5e-4 / math.sqrt(n)
            noise = np.random.randn(n) * sigma
            a = np.maximum(a + noise, 0.0)
            a *= target_sum / a.sum()

    # ------------------------------------------------------------
    # Return the best sequence discovered.
    # ------------------------------------------------------------
    print(f"[done] best score {best_score:.6f} after {iteration} Adam steps "
          f"and {lp_iterations} LP steps.")
    # Convert to plain Python list – the evaluation harness will call
    # ``evaluate_sequence`` on it.
    return best_seq.tolist()
