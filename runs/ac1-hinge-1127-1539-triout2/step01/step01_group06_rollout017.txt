def _project_to_simplex(v, C):
    # algorithm: sort descending,
    # find rho, theta
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - C
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        # all zero?
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

    np.random.seed(seed)

    best_seq = None
    best_score = np.inf

    lengths = [2000, 3000, 4000, 5000, 6000]  # maybe up to 8000 if time

            # local search using gradient descent and random moves.

                best_seq = a.copy()

    # In case best_seq is None, fallback.
    if best_seq is None:

        C = np.sqrt(2*n)
        a = np.ones(n)

    nz = np.where(best_seq > 1e-12)[0]

def _initial_sequence_custom(n, typ, param):
    if typ == 'power':
        p = param
        a = np.arange(1, n+1, dtype=float) ** (-p)
    elif typ == 'exp':
        alpha = param
        a = np.exp(-alpha * np.arange(n))

    else:
        raise
    # re-normalize
    C = np.sqrt(2.0 * n)
    a *= C / a.sum()
    return a

"""
Projected sub‑gradient optimizer for the step‑function bound.

The code searches for a non‑negative vector `a` (length *n*) with
ℓ1‑norm √(2·n).  For such a vector the objective of `evaluate_sequence`
reduces to `max(conv(a,a))`.  We minimise this maximal convolution using
a projected sub‑gradient method combined with occasional random mass‑
transfer moves and a simple annealing acceptance rule.  Several
initialisations (power‑law, exponential decay, sine shape and, if
available, the previously discovered best sequence) are tried for a
few different lengths.  The best sequence found before the time budget
expires is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core numerical helpers (must be top‑level, no closures or lambdas)
# ----------------------------------------------------------------------


def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of power‑of‑two length."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_and_full(a: np.ndarray) -> tuple[float, np.ndarray]:
    """Return (max(conv), full conv) for a non‑negative vector a."""
    conv = _conv_full(a)
    return float(conv.max()), conv


def _max_convolution(a: np.ndarray) -> float:
    """Return only the maximal convolution value (fast shortcut)."""
    conv = _conv_full(a)
    return float(conv.max())


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x≥0, Σx = target_sum}.
    Standard O(n) algorithm (see Duchi et al., 2008).
    """
    # make a copy to avoid modifying the input
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(len(v)) + 1
    # Find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries will be zero after projection
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient_max(a: np.ndarray, conv: np.ndarray, max_val: float) -> np.ndarray:
    """
    Sub‑gradient of the functional max_k Σ_i a_i a_{k-i} at the current
    maximising shifts.  If several shifts tie for the maximum we sum the
    corresponding contributions.
    """
    n = a.size
    # consider a tiny tolerance to gather numerically equal peaks
    tol = 1e-12
    K = np.where(conv >= max_val - tol)[0]
    g = np.zeros_like(a)
    for k in K:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # vector of i indices
        i_vec = np.arange(i_start, i_end + 1)
        # corresponding a‑indices (k-i)
        j_vec = k - i_vec
        g[i_vec] += a[j_vec]
    return g


def _initial_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, renormalised to ℓ1‑norm √(2·n)."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    C = np.sqrt(2.0 * n)
    a *= C / a.sum()
    return a


def _initial_exponential(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ exp(-α·i), renormalised to ℓ1‑norm √(2·n)."""
    a = np.exp(-alpha * np.arange(n, dtype=np.float64))
    C = np.sqrt(2.0 * n)
    a *= C / a.sum()
    return a


def _initial_sine(n: int) -> np.ndarray:
    """a_i ∝ sin(π·(i+1)/(n+1)), renormalised to ℓ1‑norm √(2·n)."""
    a = np.sin(np.pi * (np.arange(1, n + 1, dtype=np.float64) / (n + 1)))
    C = np.sqrt(2.0 * n)
    a *= C / a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisation core
# ----------------------------------------------------------------------


def _optimise_one_start(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with stochastic mass‑transfer moves.
    Returns the best vector (and its max‑convolution) found before `deadline`.
    """
    n = a0.size
    C = np.sqrt(2.0 * n)                     # target ℓ1‑norm
    best_a = a0.copy()
    best_score = _max_convolution(best_a)

    cur_a = a0.copy()
    cur_score = best_score

    start_time = time.time()
    total_budget = max(deadline - start_time, 1e-6)   # avoid division by zero

    it = 0
    # pre‑allocate an array for the gradient to avoid repeated allocations
    while time.time() < deadline:
        it += 1
        remaining = max(deadline - time.time(), 0.0)
        ratio = remaining / total_budget

        # Annealing temperature (tiny, just to allow occasional uphill steps)
        T = 0.005 * ratio + 0.0005

        # Decide which type of move we will propose
        if np.random.rand() < 0.58:
            # --- Projected sub‑gradient step ---------------------------------
            max_val, conv = _max_convolution_and_full(cur_a)
            g = _subgradient_max(cur_a, conv, max_val)

            # Step size proportional to current scale, decaying with time
            step_factor = 0.12 * ratio + 0.01
            max_g = g.max()
            if max_g > 0:
                step = step_factor * C / max_g
                a_candidate = cur_a - step * g
                a_candidate = _project_to_simplex(a_candidate, C)
            else:
                a_candidate = cur_a.copy()
        else:
            # --- Random mass‑transfer move (similar to the original code) ---
            i = np.random.randint(0, n)
            if cur_a[i] <= 1e-14:
                continue
            j = np.random.randint(0, n)
            if i == j:
                continue
            max_frac = 0.14
            frac = np.random.rand() * max_frac
            delta = cur_a[i] * frac
            a_candidate = cur_a.copy()
            a_candidate[i] -= delta
            a_candidate[j] += delta
            # No need to renormalise – the total sum is preserved

        # Evaluate the candidate
        cand_score = _max_convolution(a_candidate)

        # Acceptance rule (always accept improvements, sometimes accept uphill)
        delta_score = cand_score - cur_score
        if delta_score <= 0.0 or np.random.rand() < np.exp(-delta_score / max(T, 1e-12)):
            cur_a = a_candidate
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_candidate.copy()
                print(f"[{time.time():.1f}]  new best score = {best_score:.6f}")

        # Periodic mild perturbation to escape possible stagnation
        if it % 15000 == 0:
            perturb = np.random.randn(n) * (C * 1e-3)
            cur_a = best_a + perturb
            cur_a = np.maximum(cur_a, 0.0)
            cur_a = _project_to_simplex(cur_a, C)
            cur_score = _max_convolution(cur_a)

    return best_a, best_score

# Main entry point required by the competition harness

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function sequence that minimises
    ``evaluate_sequence``.  The routine runs for up to ``budget_s`` seconds,
    trying several initial shapes and lengths; a projected sub‑gradient
    optimiser drives each start towards a low‑maximum convolution.
    The best sequence found is returned as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # small safety margin

    # ------------------------------------------------------------------
    # Prepare a pool of initial candidates (different lengths / shapes)
    # ------------------------------------------------------------------
    lengths = [1800, 2500, 3200, 4000]           # moderate sizes (fits in <0.1 s conv)
    power_exps = [0.44, 0.47, 0.5, 0.53, 0.56]
    exp_alphas = [0.001, 0.002, 0.003, 0.005]
    init_pool = []                               # each entry: (n, vector)

    for n in lengths:
        # power‑law seeds
        for p in power_exps:
            a = _initial_power_law(n, p)
            init_pool.append((n, a))
        # exponential decay seeds
        for a_val in exp_alphas:
            a = _initial_exponential(n, a_val)
            init_pool.append((n, a))
        # sine‑shaped seed
        a = _initial_sine(n)
        init_pool.append((n, a))

    # If the previous best sequence is available in the global namespace,
    # also include it (it often gives a good warm start).
    try:
        global height_sequence_1
        if isinstance(height_sequence_1, (list, np.ndarray)):
            a_prev = np.asarray(height_sequence_1, dtype=np.float64)
            # make sure it respects the required ℓ1‑norm for its length
            n_prev = a_prev.size
            C_prev = np.sqrt(2.0 * n_prev)
            a_prev = a_prev * (C_prev / a_prev.sum())
            init_pool.append((n_prev, a_prev))
    except NameError:
        pass  # no previous sequence, ignore

    best_overall_seq = None
    best_overall_score = np.inf

    # ------------------------------------------------------------------
    # Main optimisation loop – run a short optimisation for each start
    # ------------------------------------------------------------------
    for n, a0 in init_pool:
        if time.time() >= deadline:
            break

        # allocate a modest time slice for this start (≈12 s, less near the end)
        remaining_total = max(deadline - time.time(), 0.0)
        slice_time = min(12.0, remaining_total * 0.4)   # never spend more than 40 % of the rest
        sub_deadline = time.time() + slice_time

        # run the optimiser for this start
        a_opt, score_opt = _optimise_one_start(a0, sub_deadline)

        if score_opt < best_overall_score:
            best_overall_score = score_opt
            best_overall_seq = a_opt.copy()
            print(f"*** Global best improved: n={n}, score={best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never happen) – uniform vector of length 1000
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        n = 1000
        a = np.ones(n, dtype=np.float64)
        a *= np.sqrt(2.0 * n) / a.sum()
        best_overall_seq = a

    # Trim trailing values that are effectively zero (helps a little)
    nz = np.where(best_overall_seq > 1e-12)[0]
    if nz.size:
        best_overall_seq = best_overall_seq[: nz[-1] + 1]

    return best_overall_seq.tolist()
