def propose_candidate(seed=42, budget_s=1000):
    rng = numpy random default_rng(seed)
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_val = np.inf

    # define lengths to try
    lengths = [2000, 4000, 6000, 8000] maybe duplicates.

    for n in lengths:
        # if time low break
        while time.time() < deadline:
            # random parameters for initial shape
            p = rng.uniform(0.4, 0.6)
            alpha = rng.uniform(0, 5)   # shift in power law
            # generate shape a_i = (i+1+alpha)^-p
            indices = np.arange(1, n+1) + alpha
            a = indices** (-p)
            a = a.astype(np.float64)
            # normalize sum to sqrt(2*n)
            a *= np.sqrt(2.0*n) / a.sum()
            # compute initial score
            cur_val = _score(a)   # _score computes ratio, which equals max_conv after scaling
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"init n={n}, p={p:.4f}, α={alpha:.4f}, val={cur_val:.6f}")
            # perform local search
            a, val = _gradient_local_search(a, deadline, rng)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"improved n={n}, val={val:.6f}")
            # optionally break after certain time or few attempts per n
            # if time left > some, continue else break
            if time.time() > deadline:
                break
            # after achieving optimum maybe run another random restart
        # continue to next n

    # final fallback if None
    if best_seq is None:
        best_seq = np.ones(1000)
        best_seq = best_seq * np.sqrt(2.0*best_seq.size) / best_seq.sum()

    # strip trailing zeros
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[:nz[-1]+1]

    return best_seq.tolist()

def _gradient_local_search(a, deadline, rng):
    n = a.size
    best = a.copy()
    best_val = _score(a)
    cur_val = best_val

    T0 = 0.1  # initial temperature
    Tmin = 1e-6
    # schedule: T = T0 * (deadline - time) / (deadline - start)
    start = time.time()
    while time.time() < deadline:
        # compute convolution
        # Possibly compute only max but we need conv anyway
        max_conv = _max_convolution(a)
        # find first index of max (if multiple)
        conv_full = np.fft.irfft(np.fft.rfft(a, n=1<<((2*n-1).bit_length())) * np.fft.rfft(a, n=1<<((2*n-1).bit_length())), n=1<<((2*n-1).bit_length()))
        conv_full = conv_full[:2*n-1]
        m = int(np.argmax(conv_full))
        # compute contributions p_i
        # Determine valid range for i
        # i from max(0, m-(n-1)) to min(n-1, m)
        i_start = max(0, m-(n-1))
        i_end = min(n-1, m)
        # compute product array for candidate i's
        idx = np.arange(i_start, i_end+1)
        j = m - idx
        p_vals = a[idx] * a[j]
        # choose source i = idx[argmax(p_vals)]
        src_idx = idx[np.argmax(p_vals)]
        # choose target idx where p is minimal (maybe zero)
        # Build list of indices with p small
        # but we want j not in same region? Could pick from all indices where a small.
        # Simpler: choose target j randomly from where a is small (< quantile)
        # Let's compute small indices list
        small_idx_candidates = np.where(a < (a.mean()*0.05))[0]  # many small
        if small_idx_candidates.size == 0:
            tgt_idx = rng.integers(0, n)
        else:
            tgt_idx = rng.choice(small_idx_candidates)
        # compute delta
        max_move_frac = 0.2
        delta = a[src_idx] * rng.random() * max_move_frac
        if delta <= 0:
            continue
        a[src_idx] -= delta
        a[tgt_idx] += delta
        # recompute score
        new_val = _score(a)
        # compute temperature schedule
        elapsed = time.time() - start
        total = deadline - start
        T = T0 * max( (total - elapsed) / total, 0.0)
        # acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-8)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = a.copy()
                print(f"[{time.time():.1f}] New best {best_val:.6f}")
        else:
            # revert
            a[tgt_idx] -= delta
            a[src_idx] += delta
        # occasionally random perturbations to escape
        # not needed

    return best, best_val

def _gradient_local_search(a, deadline, rng):

   best = a.copy()
   best_val = _score(a)
   cur_val = best_val

   start = time.time()
   # initial temperature param
   T0 = 0.2
   while time.time() < deadline:
       # compute conv array
       conv = _full_conv(a)
       max_idx = int(np.argmax(conv))
       # compute contributions p_i for indices i in range max_idx-n+1 to max_idx inclusive
       i_start = max(0, max_idx - (a.size - 1))
       i_end = min(a.size - 1, max_idx)
       idx = np.arange(i_start, i_end+1)
       j = max_idx - idx
       p = a[idx] * a[j]    # contributions
       # source: index with max product
       src = idx[np.argmax(p)]
       # target: choose among indices where a is relatively small
       # candidate list: random sample of 10 indices with smallest a
       # compute threshold: keep indices with a < median maybe
       # Let's if there are indices with a < 0.05 * mean, use them; else random
       mean_a = a.mean()
       low_idxs = np.where(a < 0.1 * mean_a)[0]
       if low_idxs.size > 0:
           tgt = rng.choice(low_idxs)
       else:
           tgt = rng.integers(0, a.size)
       # ensure src != tgt
       if src == tgt:
           continue
       # move fraction of source mass
       max_move_frac = 0.15
       delta = a[src] * rng.random() * max_move_frac
       if delta <= 1e-12:
           continue
       a[src] -= delta
       a[tgt] += delta
       # compute new value
       new_val = _score(a)
       # determine temperature
       elapsed = time.time() - start
       total = deadline - start
       T = T0 * max(0.0, (total - elapsed) / total)
       # accept if improved or with prob
       if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-9)):
           cur_val = new_val
           if new_val < best_val:
               best_val = new_val
               best = a.copy()
               print(f"[{time.time():.1f}] New best (grad) = {best_val:.6f}")
       else:
           # revert
           a[tgt] -= delta
           a[src] += delta

   return best, best_val

"""
step‑function optimiser

The optimiser builds a non‑negative sequence with ℓ1‑norm √(2·n) (the objective
is invariant under scaling, so fixing the norm makes the score equal to the
maximum value of the self‑convolution).  The search combines three ideas:

1. **Power‑law initialisation**   a_i ∝ (i+α)^‑p  (α≥0, p≈½) – a good
   approximation of the known continuous optimum.

2. **Linear‑programming refinement** – for a fixed sequence f we solve
   a linear programme
        maximise   Σ b_j
        subject   f * b ≤ M   (pointwise) ,  b ≥ 0
   where M = max(f*f).  The solution b puts as much mass as possible on the
   “cheap’’ coordinates of f.  After scaling b to the required ℓ1‑norm we
   mix f←(1‑t)f+t·b (t≈0.5).  Repeating a few LP steps steadily reduces the
   maximum convolution.

3. **Gradient‑guided stochastic hill‑climbing** – the current worst
   convolution index m is identified, the contribution a_i·a_{m‑i} is
   computed and a small fraction of mass is transferred from the index with
   the largest contribution to an index where a is small.  Moves that
   improve the score are always kept; otherwise they are accepted with a
   temperature‑controlled probability (simulated‑annealing style).

The whole routine repeatedly tries random values of (p,α) for several
different lengths, keeps the best sequence found before the deadline and
returns it as a plain Python list.  If cvxpy is unavailable the optimiser
falls back to the stochastic hill‑climbing phase only.
"""

import time
import numpy as np

# Basic utilities (no closures, no lambdas)

def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using an FFT of length a power of two.
    Returns a 1‑D array of length 2·n‑1.
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution of a."""
    return float(_full_convolution(a).max())


def _score(a: np.ndarray) -> float:
    """
    Compute the competition objective 2·n·max(conv)/sum(a)².
    The value is invariant under scaling; with the ℓ1‑norm fixed to
    √(2·n) the score equals the maximum convolution.
    """
    n = a.size
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _initial_powerlaw(n: int, p: float, offset: float) -> np.ndarray:
    """
    Generate a non‑negative vector of length n following a power‑law:
        a_i ∝ (i+1+offset)^{-p}
    and renormalise it to ℓ1‑norm √(2·n).
    """
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = idx ** (-p)
    a *= np.sqrt(2.0 * n) / a.sum()
    return a


# ----------------------------------------------------------------------
# Linear‑programming refinement (optional – requires cvxpy)
# ----------------------------------------------------------------------


def _lp_refine(f: np.ndarray, S: float, deadline: float) -> tuple[np.ndarray, float]:
    """
    One LP refinement step.
    Solve   maximise Σ b_j   s.t.   f*b ≤ M   (pointwise),   b ≥ 0,
    where M = max(f*f).  The optimiser returns the mixed vector
        f' = (1‑t)·f + t·b̂   (t = 0.5, b̂ = scaled b to ℓ1‑norm S).
    If cvxpy is unavailable or the solve fails, the original f is returned.
    """
    # quick bail‑out if we are out of time
    if time.time() >= deadline:
        return f, _score(f)

    try:
        import cvxpy as cp
    except Exception:                     # cvxpy not installed
        return f, _score(f)

    n = f.size
    M = _max_convolution(f)

    # build the Toeplitz convolution matrix T (shape (2n‑1, n))
    rows = 2 * n - 1
    T = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        T[j:j + n, j] = f                       # column j is f shifted by j

    # linear programme
    b = cp.Variable(n, nonneg=True)
    constraints = [
        cp.matmul(T, b) <= np.full(rows, M, dtype=np.float64)
    ]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)

    # limit the solve time by a modest iteration cap
    try:
        prob.solve(solver=cp.ECOS, max_iters=3000, warm_start=True)
    except Exception:
        # any numerical issue – give up the LP step
        return f, _score(f)

    if b.value is None:
        return f, _score(f)

    b_opt = np.maximum(0.0, np.asarray(b.value, dtype=np.float64))
    sum_b = b_opt.sum()
    if sum_b <= 0.0:
        return f, _score(f)

    # scale to the required ℓ1‑norm and mix
    b_scaled = b_opt * (S / sum_b)
    t = 0.5
    f_new = (1.0 - t) * f + t * b_scaled
    # re‑normalise to guard against round‑off drift
    f_new *= S / f_new.sum()
    return f_new, _score(f_new)


# ----------------------------------------------------------------------
# Gradient‑guided stochastic hill‑climbing (simulated annealing)
# ----------------------------------------------------------------------


def _gradient_local_search(a: np.ndarray,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climbing that always respects the ℓ1‑norm.
    At each step the index of the maximal convolution entry is
    identified; a fraction of mass is moved from the index contributing
    most to this peak to an index where the sequence is small.
    The move is accepted if it improves the score, otherwise with a
    probability decreasing with a simple linear temperature schedule.
    """
    n = a.size
    best = a.copy()
    best_val = _score(a)
    cur_val = best_val

    start = time.time()
    T0 = 0.2                # initial temperature

    while time.time() < deadline:
        # full convolution and the worst index
        conv = _full_convolution(a)
        m = int(np.argmax(conv))

        # indices that actually participate in conv[m]
        i_start = max(0, m - (n - 1))
        i_end = min(n - 1, m)
        idx = np.arange(i_start, i_end + 1)
        j = m - idx
        contrib = a[idx] * a[j]                     # a_i·a_{m-i}

        # source = index with largest contribution
        src = idx[np.argmax(contrib)]

        # target = a small‑value index (randomly chosen among the 10 % smallest)
        mean_a = a.mean()
        low_candidates = np.where(a < 0.1 * mean_a)[0]
        if low_candidates.size == 0:
            tgt = rng.integers(0, n)
        else:
            tgt = rng.choice(low_candidates)

        if src == tgt:
            continue

        # move a random fraction (up to 15 %) of the source mass to the target
        max_frac = 0.15
        delta = a[src] * rng.random() * max_frac
        if delta <= 1e-12:
            continue

        a[src] -= delta
        a[tgt] += delta

        new_val = _score(a)

        # linear temperature schedule
        elapsed = time.time() - start
        total = deadline - start
        T = T0 * max(0.0, (total - elapsed) / total)

        # acceptance rule
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = a.copy()
                print(f"[{time.time():.1f}] new best (grad) = {best_val:.6f}")
        else:
            # revert the move
            a[tgt] -= delta
            a[src] += delta

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point required by the challenge harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function sequence that minimises the supplied
    evaluate_sequence() function.  The routine explores several lengths,
    random power‑law initialisations, a small number of LP refinements
    (when cvxpy is present) and a gradient‑guided stochastic hill‑climbing.
    The best sequence discovered before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    best_seq = None
    best_score = np.inf

    # length candidates – keep them moderate to stay within the LP budget
    lengths = [800, 1200, 1800, 2500]

    for n in lengths:
        if time.time() >= deadline:
            break

        S = np.sqrt(2.0 * n)                 # target ℓ1‑norm

        # a few random starts for each length
        for start_id in range(15):
            if time.time() >= deadline:
                break

            # random power‑law parameters
            p = rng.uniform(0.42, 0.60)       # exponent close to ½
            offset = rng.uniform(0.0, 6.0)    # shift of the index

            a = _initial_powerlaw(n, p, offset)

            cur_val = _score(a)
            if cur_val < best_score:
                best_score = cur_val
                best_seq = a.copy()
                print(f"init n={n}, p={p:.4f}, offset={offset:.2f}, score={cur_val:.6f}")

            # -------------------------------------------------
            # Optional LP refinement (only for modest sizes)
            # -------------------------------------------------
            if n <= 2000 and start_id == 0:   # do it once per length
                a_lp, val_lp = _lp_refine(a, S, deadline)
                if val_lp < best_score:
                    best_score = val_lp
                    best_seq = a_lp.copy()
                    print(f"LP step n={n}, score={val_lp:.6f}")
                a = a_lp                      # continue from the refined vector

                # a short second LP pass can sometimes give further gain
                a_lp2, val_lp2 = _lp_refine(a, S, deadline)
                if val_lp2 < best_score:
                    best_score = val_lp2
                    best_seq = a_lp2.copy()
                    print(f"LP step 2 n={n}, score={val_lp2:.6f}")
                a = a_lp2

            # -------------------------------------------------
            # Gradient‑guided stochastic local optimisation
            # -------------------------------------------------
            sub_deadline = min(deadline, time.time() + 6.0)
            a_opt, val_opt = _gradient_local_search(a, sub_deadline, rng)
            if val_opt < best_score:
                best_score = val_opt
                best_seq = a_opt.copy()
                print(f"local search n={n}, score={val_opt:.6f}")

            # keep the best vector for the next start (helps continuation)
            a = a_opt

    # ------------------------------------------------------------------
    # Fallback – extremely unlikely to be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.ones(1000, dtype=np.float64)
        best_seq *= np.sqrt(2.0 * best_seq.size) / best_seq.sum()

    # strip trailing zeros (just in case)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # return a plain Python list
    return best_seq.tolist()
