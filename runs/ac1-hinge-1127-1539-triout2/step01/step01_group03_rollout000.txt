"""
Propose a low‑score step‑function using a hybrid of
*grid search → LP‑based active‑set improvement → simple gradient descent*.

Algorithm
---------
1. **Initialisation** – Choose a length `n` (default 4000).  Run a tiny
   grid search over the family  a_i = 1/(i + 1 + c)^β  (β≈0.5, c≥0) and
   keep the best candidate (the classic 1/√x construction).

2. **Active‑set linear programme** – For the current sequence `a`,
   compute the full convolution `c = a * a`.  Let `M = max(c)` and collect
   the indices `I` where `c_i` is within 0.1 % of `M`.  For each such
   index the gradient of `c_i` w.r.t. `a` is `g_i = a + a shifted by i`.
   We solve the LP

        maximise   t
        subject to sum(δ) = 0
                   δ ≥ –a               (keeps a+δ non‑negative)
               g_i·δ + t ≤ 0   for i∈I   (forces a decrease of the
                                            active convolutions)

   The variable `δ` is a feasible update direction; we replace `a` by
   `a ← a + α·δ` (α∈{1,0.5,0.25,…}) if the score improves.

3. **Gradient‐descent fallback** – If the LP yields no improvement
   (or after a limited number of LP steps), perform a cheap sub‑gradient
   step on the single most active index:

        a ← a – η·(a + shift_by_kmax)

   followed by renormalisation to the original ℓ₁‑norm.
   This keeps the search moving when the LP becomes stuck.

4. **Loop** – Repeat steps 2–3 until the time budget (≈1000 s) is almost
   exhausted or the target score < 1.503 is reached.  The best sequence
   seen during the run is returned.

The whole routine stays within the allowed time limit, uses only
numpy (for fast FFT‑based convolutions) and cvxpy (ECOS solver) for the
LP, and always returns a list of non‑negative floats.
"""

import time
import numpy as np
import cvxpy as cp


def _objective(seq: np.ndarray) -> float:
    """
    Fast evaluation of the score 2·n·max(conv(seq,seq)) / (sum(seq)²).

    Parameters
    ----------
    seq : np.ndarray
        The candidate array (1‑D, non‑negative).

    Returns
    -------
    float
        The objective value (lower is better).
    """
    n = seq.shape[0]
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    # length for a zero‑padded FFT (next power of two ≥ 2n‑1)
    m = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft * fft, n=m)[: 2 * n - 1]
    max_conv = np.max(conv)
    return float(2.0 * n * max_conv / (s * s))


def propose_candidate(seed: int = 0, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Search for a step‑function with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (the function returns a few seconds before
        the limit to avoid a timeout).

    Returns
    -------
    list[float]
        The best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + max(0.0, budget_s - 2.0)  # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a length.
    #    If a global variable ``height_sequence_1`` exists we reuse its
    #    length, otherwise we pick a default.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(globals()["height_sequence_1"])
        print(f"[+] Using length from existing sequence: n = {n}")
    else:
        n = 4000
        print(f"[+] Default length selected: n = {n}")

    # ------------------------------------------------------------------
    # 2. Grid search over (beta, offset) for the classical 1/√x shape.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    betas = np.linspace(0.44, 0.56, 25)        # a bit wider than before
    offsets = np.linspace(0.0, 6.0, 13)

    for beta in betas:
        for c in offsets:
            idx = np.arange(n, dtype=np.float64) + 1.0 + c
            cand = 1.0 / (idx ** beta)
            score = _objective(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"[grid] beta={beta:.4f}, offset={c:.2f} → score={score:.6f}")
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    # If a pre‑existing sequence is available, check it as well.
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if hs.shape[0] != n:
            # Resize by simple linear interpolation
            hs = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, hs.shape[0]), hs)
        s = _objective(hs)
        if s < best_score:
            best_score = s
            best_seq = hs.copy()
            print(f"[init] height_sequence_1 gives score={s:.6f}")

    # Fallback – should never happen.
    if best_seq is None:
        best_seq = np.ones(n, dtype=np.float64)
        best_score = _objective(best_seq)

    # ------------------------------------------------------------------
    # 3. Main optimisation loop (LP + gradient descent).
    # ------------------------------------------------------------------
    a = best_seq.copy()
    target_sum = np.sum(a)           # invariant – scaling does not change score
    best_global_seq = a.copy()
    best_global_score = best_score

    # parameters governing the loop
    max_lp_steps = 30                # how many LP refinements we attempt
    lp_steps_done = 0
    grad_step_factor = 0.6
    iteration = 0

    while time.time() < deadline:
        # ---- compute full convolution and statistics ----
        m_fft = 1 << ((2 * n - 1).bit_length())
        fft_a = np.fft.rfft(a, n=m_fft)
        conv = np.fft.irfft(fft_a * fft_a, n=m_fft)[: 2 * n - 1]
        max_conv = np.max(conv)
        k_max = int(np.argmax(conv))
        cur_score = float(2.0 * n * max_conv / (target_sum * target_sum))

        # update global best
        if cur_score < best_global_score:
            best_global_score = cur_score
            best_global_seq = a.copy()
            print(f"[iter {iteration:4d}] New best score = {cur_score:.6f}")

        # early exit if target already reached
        if best_global_score < 1.503:
            print(f"[+] Target achieved (score={best_global_score:.6f}) at iter {iteration}")
            break

        # ---- determine active set (indices close to the maximum) ----
        eps = max_conv * 0.001        # 0.1 % tolerance
        active_idx = np.where(conv >= max_conv - eps)[0]

        # keep the active set small for speed
        if active_idx.size > 12:
            # keep the 12 largest values
            order = np.argsort(-conv[active_idx])[:12]
            active_idx = active_idx[order]

        # ---------------------------------------------------------------
        # 3a. Try an LP step if we have not exhausted the LP budget.
        # ---------------------------------------------------------------
        lp_success = False
        if lp_steps_done < max_lp_steps and active_idx.size > 0:
            # Build the gradient matrix G where G[i, :] = a + a shifted by active_idx[i]
            G = np.empty((active_idx.size, n), dtype=np.float64)
            for r, k in enumerate(active_idx):
                idx_shift = k - np.arange(n)
                mask = (idx_shift >= 0) & (idx_shift < n)
                shift = np.zeros_like(a)
                shift[mask] = a[idx_shift[mask]]
                G[r, :] = a + shift

            # LP variables
            delta = cp.Variable(n)
            t = cp.Variable()

            constraints = [
                cp.sum(delta) == 0,
                delta >= -a  # keep a+delta non‑negative
            ]
            # active‑set constraints: G·delta + t ≤ 0
            for r in range(G.shape[0]):
                constraints.append(cp.sum(cp.multiply(G[r, :], delta)) + t <= 0)

            prob = cp.Problem(cp.Maximize(t), constraints)

            try:
                prob.solve(solver=cp.ECOS, max_iters=500, warm_start=True, verbose=False)
            except Exception as exc:
                print(f"[lp] solver exception: {exc}")
                prob = None

            if prob is not None and delta.value is not None and t.value is not None:
                if t.value > 1e-9:          # genuine improvement direction
                    # try full step first
                    a_candidate = a + delta.value
                    a_candidate = np.maximum(a_candidate, 0.0)
                    # renormalise ℓ₁‑norm (does not affect the score)
                    a_candidate *= target_sum / a_candidate.sum()
                    cand_score = _objective(a_candidate)

                    if cand_score < cur_score:
                        a = a_candidate
                        lp_success = True
                        lp_steps_done += 1
                        print(f"[lp {lp_steps_done}] Improved to {cand_score:.6f} (t={t.value:.6f})")
                    else:
                        # try diminishing the step size
                        scale = 0.5
                        while scale > 1e-3:
                            a_test = a + scale * delta.value
                            a_test = np.maximum(a_test, 0.0)
                            a_test *= target_sum / a_test.sum()
                            test_score = _objective(a_test)
                            if test_score < cur_score:
                                a = a_test
                                lp_success = True
                                lp_steps_done += 1
                                print(f"[lp {lp_steps_done}] Scaled ({scale:.2f}) → {test_score:.6f}")
                                break
                            scale *= 0.5
                else:
                    # LP could not find a positive improvement
                    pass

        # ---------------------------------------------------------------
        # 3b. If LP made no progress, fall back to a simple gradient step.
        # ---------------------------------------------------------------
        if not lp_success:
            # sub‑gradient for the most active index only
            idx_shift = k_max - np.arange(n)
            mask = (idx_shift >= 0) & (idx_shift < n)
            shift = np.zeros_like(a)
            shift[mask] = a[idx_shift[mask]]
            grad = a + shift                     # g_i = a_i + a_{k_max-i}

            # step size proportional to the current ratio
            ratio = max_conv / (target_sum ** 2)
            step = grad_step_factor * ratio

            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            a_new_sum = a_new.sum()
            if a_new_sum > 1e-12:
                a_new *= target_sum / a_new_sum
                a = a_new
                grad_step_factor = min(grad_step_factor * 1.02, 2.0)  # slowly increase
                # occasional random jitter to escape plateaus
                if iteration % 50 == 0:
                    i, j = np.random.randint(n, size=2)
                    delta = np.random.rand() * a[i] * 0.03
                    a[i] = max(0.0, a[i] - delta)
                    a[j] += delta
                    a *= target_sum / a.sum()
            else:
                # all zeros – re‑initialise from the current best
                a = best_global_seq.copy()

        iteration += 1

    print(f"[+] Search finished after {iteration} iterations. Best score = {best_global_score:.6f}")
    # Convert to plain Python list (evaluate_sequence will clip if necessary)
    return [float(v) for v in best_global_seq]
