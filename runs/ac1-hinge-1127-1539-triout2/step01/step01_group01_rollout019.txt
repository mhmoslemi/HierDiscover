"""
Simulated‑Annealing / Gradient‑Transfer optimiser.

We keep the sum of the heights fixed to √(2 n) (the normalisation used by
`evaluate_sequence`).  The objective then equals the maximal entry of the
self‑convolution.  Starting from a strongly “U‑shaped’’ distribution we
perform two kinds of moves:

1. **Deterministic gradient transfer** – the active set of convolution
   indices that realise the current maximum is found (optionally with a
   small tolerance).  The sub‑gradient g_i = Σ_{k∈act}2·a_{k‑i} tells how
   much every coefficient contributes to the peak.  We move a small
   fraction of mass from the index with the largest g_i to the index with
   the smallest g_i.  The convolution is updated with an O(n) incremental
   formula (no FFT inside the loop).

2. **Stochastic SA jumps** – occasional random transfers of a tiny amount
   of mass are tried; they are accepted if they improve the objective,
   otherwise with a Metropolis probability that slowly cools.

Both move types preserve the ℓ₁‑norm, keep the vector non‑negative and
avoid the costly `evaluate_sequence` call (the objective is computed
directly from the updated convolution).  The routine runs for at most the
given time budget and returns the best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a power‑of‑two FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """
    Scale a non‑negative vector to have ℓ₁‑norm = target.
    Zero vector → uniform distribution.
    """
    s = v.sum()
    if s > 0.0:
        return v * (target / s)
    return np.full_like(v, target / v.shape[0])


def generate_U_shape(
    n: int,
    target_sum: float,
    kind: str = "exponential",
    param: float = 3.0,
) -> np.ndarray:
    """
    Symmetric “U‑shaped’’ sequence.
    kind ∈ {linear, quadratic, exponential}.
    param ≥ 0 controls the edge‑enhancement.
    Result is ℓ₁‑scaled to target_sum.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    else:
        raise ValueError("unsupported shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_to_sum(raw, target_sum)


# ----------------------------------------------------------------------
# Incremental convolution update for a transfer a[i] → a[j]
# ----------------------------------------------------------------------


def delta_convolution(
    a: np.ndarray, i: int, j: int, delta: float, n: int
) -> np.ndarray:
    """
    Return the change of the full convolution when `delta` is moved from
    index i to index j (i ≠ j).  The returned array has length 2*n‑1.
    """
    L = 2 * n - 1
    d = np.zeros(L, dtype=np.float64)

    # linear term 2·Δ*a  (Δ is -delta at i, +delta at j)
    # contribution –2·delta·a shifted by i
    d[i : i + n] -= 2.0 * delta * a
    # contribution +2·delta·a shifted by j
    d[j : j + n] += 2.0 * delta * a

    # quadratic term Δ*Δ   (only three non‑zero positions)
    ii = 2 * i
    jj = 2 * j
    ij = i + j
    d[ii] += delta * delta
    d[jj] += delta * delta
    d[ij] -= 2.0 * delta * delta
    return d


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence minimising ``evaluate_sequence``.
    The algorithm combines a gradient‑transfer descent with a simulated‑annealing
    jump scheme and respects the 1000 s time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # Choose a length and an initial shape
    # --------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        a = np.asarray(base, dtype=np.float64)
        target_sum = np.sqrt(2.0 * n)
        a = project_to_sum(a, target_sum)
    else:
        # a few different sizes are tried automatically
        n = np.random.randint(8000, 12000)
        target_sum = np.sqrt(2.0 * n)
        # start from a strong exponential U‑shape
        a = generate_U_shape(
            n, target_sum, kind="exponential", param=4.0
        )

    # initial convolution and objective
    b = conv_self_fft(a)
    cur_max = b.max()
    cur_score = 2.0 * n * cur_max / (target_sum ** 2)

    best_a = a.copy()
    best_score = cur_score
    best_max = cur_max

    print(f"[{time.time():.2f}] start, n={n}, initial score={cur_score:.6f}")

    # --------------------------------------------------------------
    # Optimisation parameters
    # --------------------------------------------------------------
    temperature = 1.0               # SA temperature
    step_frac = 0.05                # fraction of source mass moved each gradient step
    rel_tol = 0.0                   # active‑set tolerance (0 ⇒ single max index)
    it = 0
    no_improve = 0

    # --------------------------------------------------------------
    # Main loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        # ----- deterministic gradient transfer --------------------
        # locate active convolution indices (those attaining the current max)
        if rel_tol <= 0.0:
            act = np.array([np.argmax(b)], dtype=np.int64)
        else:
            act = np.where(b >= cur_max * (1.0 - rel_tol))[0]

        # compute sub‑gradient g_i = Σ_{k∈act} 2·a_{k‑i}
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for k in act:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        i_src = int(np.argmax(g))          # location that contributes most
        i_dst = int(np.argmin(g))          # location that contributes least

        # guard against degenerate move
        if i_src != i_dst and a[i_src] > 0.0:
            delta = step_frac * a[i_src]
            # incremental update of convolution
            dB = delta_convolution(a, i_src, i_dst, delta, n)
            new_b = b + dB
            new_max = new_b.max()
            new_score = 2.0 * n * new_max / (target_sum ** 2)

            # greedy accept if improvement, otherwise reject (no SA on this move)
            if new_score < cur_score:
                # accept
                a[i_src] -= delta
                a[i_dst] += delta
                b = new_b
                cur_max = new_max
                cur_score = new_score
                it += 1
                no_improve = 0
                if new_score < best_score - 1e-12:
                    best_score = new_score
                    best_a = a.copy()
                    best_max = new_max
                    print(f"Iter {it:06d} – new best (grad) {best_score:.6f}")
                continue

        # ----- stochastic SA jump ----------------------------------
        # (executed only if the deterministic move did not change the state)
        i, j = np.random.choice(n, 2, replace=False)
        if a[i] <= 0.0:
            continue
        delta = step_frac * a[i]           # same fraction as above
        dB = delta_convolution(a, i, j, delta, n)
        new_b = b + dB
        new_max = new_b.max()
        new_score = 2.0 * n * new_max / (target_sum ** 2)

        # Metropolis acceptance
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temperature, 1e-12))
            accept = np.random.rand() < prob

        if accept:
            a[i] -= delta
            a[j] += delta
            b = new_b
            cur_max = new_max
            cur_score = new_score
            it += 1
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a.copy()
                best_max = new_max
                print(f"Iter {it:06d} – new best (SA) {best_score:.6f}")
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ----- schedule adjustments --------------------------------
        if it % 2000 == 0 and it > 0:
            temperature *= 0.9               # cool down
            step_frac *= 0.9                 # finer moves later
            # re‑project occasionally to avoid drift
            a = project_to_sum(a, target_sum)
            b = conv_self_fft(a)
            cur_max = b.max()
            cur_score = 2.0 * n * cur_max / (target_sum ** 2)

        # gradually broaden the active set (helps later stages)
        if it % 5000 == 0 and rel_tol < 0.05:
            rel_tol += 0.005

        # escape long plateaus by a small random jitter
        if no_improve >= 5000:
            jitter = np.random.randn(n) * (target_sum / (n * 1000.0))
            a = np.clip(a + jitter, 0.0, None)
            a = project_to_sum(a, target_sum)
            b = conv_self_fft(a)
            cur_max = b.max()
            cur_score = 2.0 * n * cur_max / (target_sum ** 2)
            no_improve = 0
            print(f"Iter {it:06d} – jitter reset")

    print(
        f"[{time.time():.2f}] finished, best score {best_score:.6f} (max conv={best_max:.6f})"
    )
    # The returned list is plain Python floats, as required by the harness.
    return best_a.tolist()
