"""
minimise the maximum entry of a self‑convolution (the quantity returned by
`evaluate_sequence`).  The optimiser works in three phases:

1. **Broad parametric sweep** – a collection of simple analytic families
   (symmetric “U” shapes, monotone decreasing/increasing shapes,
   exponential and quadratic variants) are generated, scaled to the
   required ℓ₁‑norm and the best one is kept.

2. **Smooth‑max gradient descent** – the (soft) maximum of the convolution
   is approximated by a log‑sum‑exp.  Its gradient can be written in closed
   form as a convolution and is computed with FFTs.  The update direction
   is projected to the hyper‑plane Σaᵢ = target_sum (by subtracting the
   mean) and a small momentum term is added.  The temperature of the
   soft‑max is decreased slowly, and after each step the vector is clipped
   to non‑negative values and renormalised.

3. **Escapes & fine‑tuning** – occasional random pair‑wise mass transfers
   and jitter are performed when progress stalls.

The whole search respects a 1000 s wall‑clock budget and always returns the
best sequence found.  The implementation uses only NumPy (FFT based
convolutions) and no external optimisation packages.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures / lambdas)
# ----------------------------------------------------------------------


def fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors a and b using an FFT of power‑of‑2 length."""
    n, m = a.shape[0], b.shape[0]
    L = 1 << ((n + m - 1).bit_length())           # next power of two
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)
    return conv[: n + m - 1]


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (length 2·len(a)-1)."""
    return fft_convolve(a, a)


def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector so that its ℓ₁‑norm equals *target*."""
    s = v.sum()
    if s > 0:
        return v * (target / s)
    # fallback – uniform distribution (should never happen)
    return np.full_like(v, target / v.shape[0])


def generate_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
    direction: str = "symmetric",
) -> np.ndarray:
    """
    Produce a deterministic shape of length *n*.

    ``kind``     – {"linear", "quadratic", "exponential"}.
    ``direction`` – {"symmetric", "decreasing", "increasing"}.

    *param* controls the strength of the edge/gradient (≥0).
    The shape is finally scaled so that the ℓ₁‑norm equals *target_sum*.
    """
    idx = np.arange(n, dtype=np.float64)
    if direction == "symmetric":
        # U‑shape centred in the middle
        mid = (n - 1) / 2.0
        d = np.abs(idx - mid) / mid                      # ∈[0,1]
        base = d
    elif direction == "decreasing":
        # high on the left, low on the right
        base = 1.0 - idx / (n - 1)                        # ∈[1,0]
    elif direction == "increasing":
        # low on the left, high on the right
        base = idx / (n - 1)                              # ∈[0,1]
    else:
        raise ValueError("unknown direction")

    if kind == "linear":
        raw = 1.0 + param * base
    elif kind == "quadratic":
        raw = 1.0 + param * base ** 2
    elif kind == "exponential":
        raw = np.exp(param * base)
    else:
        raise ValueError("unknown shape kind")

    raw = np.clip(raw, 0.0, None)
    return project_to_sum(raw, target_sum)


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """Transfer a tiny fraction of mass from a random source index to a random target."""
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0.0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_to_sum(a, target_sum)


def jitter(a: np.ndarray, target_sum: float, sigma: float = 1e-3) -> np.ndarray:
    """Add Gaussian noise (clipped) and renormalise."""
    a = a + np.random.randn(a.shape[0]) * sigma
    a = np.clip(a, 0.0, None)
    return project_to_sum(a, target_sum)

def smooth_max_step(
    a: np.ndarray,
    target_sum: float,
    lr: float,
    tau: float,
    momentum: float,

) -> tuple[np.ndarray, np.ndarray]:
    """
    One iteration of gradient descent on the smoothed max of the convolution.

    Returns the updated vector and the new momentum term.
    """
    n = a.shape[0]

    # 1) convolution a * a
    conv = conv_self_fft(a)                     # length 2n‑1

    # 2) soft‑max weights (log‑sum‑exp trick for stability)
    #    w_k ∝ exp(conv_k / tau)
    # Subtract max(conv) to avoid overflow
    max_c = conv.max()
    exp_arg = (conv - max_c) / tau
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:                           # degenerate case – fall back to hard max
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    # 3) gradient of the smoothed max:
    #    g_i = 2 * Σ_k w_k * a_{k-i}
    #    which is a cross‑correlation of w with a.
    a_rev = a[::-1]
    grad_full = fft_convolve(w, a_rev)         # length 3n‑2
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]   # the slice corresponding to i = 0…n‑1

    # 4) projected direction (preserve the ℓ₁‑norm)
    direction = -(grad - np.mean(grad))

    # 5) momentum + step
    v = momentum * v_prev + direction
    a_new = a + lr * v

    # 6) enforce feasibility
    a_new = np.clip(a_new, 0.0, None)
    a_new = project_to_sum(a_new, target_sum)

    return a_new, v

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Choose a length – reuse a known construction if it exists
    # ------------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
    else:
        # a size that is large enough to give a good bound but still cheap
        n = np.random.randint(7200, 8800)

    target_sum = np.sqrt(2.0 * n)                    # the normalisation used in the paper

    # ------------------------------------------------------------------
    # Phase 1 – extensive parametric sweep (both symmetric and asymmetric)
    # ------------------------------------------------------------------
    best_seq = np.full(n, target_sum / n)            # uniform baseline
    best_score = float("inf")

    # families to explore
    families = [
        ("symmetric", "linear"),
        ("symmetric", "quadratic"),
        ("symmetric", "exponential"),
        ("decreasing", "linear"),
        ("decreasing", "quadratic"),
        ("decreasing", "exponential"),
        ("increasing", "linear"),
        ("increasing", "quadratic"),
        ("increasing", "exponential"),
    ]

    # grid of parameters (coarse – cheap)
    param_grids = {
        "linear": np.linspace(0.0, 5.0, 26),          # 0 … 5 step 0.2
        "quadratic": np.linspace(0.0, 4.0, 21),      # 0 … 4 step 0.2
        "exponential": np.linspace(0.0, 3.0, 16),    # 0 … 3 step 0.2
    }

    print(f"[{time.time():.2f}] Starting parametric sweep (n={n})")
    for direction, kind in families:
        for p in param_grids[kind]:
            cand = generate_shape(n, target_sum, kind=kind, param=p, direction=direction)
            score = evaluate_sequence(cand.tolist())
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(
                    f"  New best {direction}/{kind} (p={p:.2f}) → {best_score:.6f}"
                )
            if time.time() >= deadline:
                print("Time exhausted during sweep.")
                return best_seq.tolist()

    # also try a few random initialisations
    for _ in range(40):
        cand = np.random.rand(n)
        cand = project_to_sum(cand, target_sum)
        score = evaluate_sequence(cand.tolist())
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"  Random init improved → {best_score:.6f}")
        if time.time() >= deadline:
            return best_seq.tolist()

    # ------------------------------------------------------------------
    # Phase 2 – smooth‑max gradient descent with momentum
    # ------------------------------------------------------------------
    a = best_seq.copy()
    v = np.zeros_like(a)                     # momentum buffer

    lr = 0.03 * (target_sum / n)              # step size ≈ 3 % of the average entry
    tau = 0.15 * a.max()                      # initial temperature (fraction of the peak)
    momentum = 0.9

    iteration = 0
    no_improve = 0

    print(f"[{time.time():.2f}] Entering smooth‑max descent (lr={lr:.5f}, tau={tau:.5f})")
    while time.time() < deadline:
        a, v = smooth_max_step(a, target_sum, lr, tau, momentum, v)
        score = evaluate_sequence(a.tolist())
        iteration += 1

        if score + 1e-12 < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"Iter {iteration:06d} – improvement: {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # schedule adaptations
        if iteration % 200 == 0:
            lr *= 0.95                     # slowly decay the step size
            tau *= 0.97                    # cool the temperature
            if lr < 1e-7:
                lr = 1e-7
            if tau < 1e-4:
                tau = 1e-4

        # occasional random pair transfer – useful for escaping plateaus
        if np.random.rand() < 0.03:
            a = random_pair_transfer(a, target_sum, scale=0.008)

        # if we have been stuck for a while, inject jitter
        if no_improve >= 1500:
            a = jitter(a, target_sum, sigma=5e-4)
            v[:] = 0.0                     # reset momentum
            no_improve = 0
            print(f"Iter {iteration:06d} – jitter reset")

    # ------------------------------------------------------------------
    # Phase 3 – final quick random perturbations (optional)
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Descent finished; best score {best_score:.6f}")

    # a very short final polishing loop (if time permits)
    while time.time() < deadline - 0.2:
        a = random_pair_transfer(a, target_sum, scale=0.004)
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"Polish – improvement to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as plain Python list)
    # ------------------------------------------------------------------
    return best_seq.tolist()
