v = np.zeros_like(w)

eps = 1e-8

while time left:
   t += 1
   # compute a from w
   max_w = np.max(w)
   exp_w = np.exp(w - max_w)   # stable
   a = exp_w / np.sum(exp_w)

   internal_max = max_val  # conv_max

def _softmax(w):
    max_w = np.max(w)
    exp_w = np.exp(w - max_w)
    return exp_w / np.sum(exp_w)

def _conv_max(a):
    n = a.shape[0]
    conv_len = 2*n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    idx = int(np.argmax(conv))
    max_val = conv[idx]
    return max_val, idx

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0
    # Choose length based on time budget
    n = 20000
    # ensure n not too large for compute; we can reduce if time low
    # initialize parameters for Adam
    w = rng.normal(loc=0.0, scale=0.01, size=n)   # small random init to break symmetry
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    eta = 0.2  # base learning rate
    t = 0
    best_internal = np.inf
    best_w = None
    best_seq = None
    # initial evaluation
    a = _softmax(w)
    internal_max, k_max = _conv_max(a)
    best_internal = internal_max
    best_w = w.copy()
    best_seq = a.copy()
    print("[init] n={}, internal_max={:.8f}, score≈{:.6f}".format(n, internal_max, 2*n*internal_max))
    # main loop
    while time.time() < deadline:
        t += 1
        a = _softmax(w)
        internal_max, k_max = _conv_max(a)
        # check improvement
        if internal_max < best_internal:
            best_internal = internal_max
            best_w = w.copy()
            best_seq = a.copy()
            print("[improve] t={} time={:.1f}s internal_max={:.8f} score≈{:.6f}".format(t, time.time()- (deadline-budget_s), internal_max, 2*n*internal_max))
        # compute subgradient
        idx = np.arange(n)
        partner = k_max - idx
        valid = (partner >= 0) & (partner < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[partner[valid]]
        # compute moving averages
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        # bias correction
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        # w update
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)
        # occasional jitter to escape stagnation
        if np.random.rand() < 0.005:
            w += rng.normal(scale=0.1, size=n)
        # optionally break if time low
    # final sequence
    final_seq = best_seq if best_seq is not None else _softmax(w)
    final_score = evaluate_sequence(final_seq.tolist())
    print("[final] best internal_max={:.8f} approx score={:.6f} external score={:.6f} length={}".format(best_internal, 2*n*best_internal, final_score, n))
    return final_seq.tolist()

"""
Propose a candidate step‑function (non‑negative heights) that beats the
current 2.0 → 1.5308 bound.

Algorithm
---------
* Work on the simplex by parametrising the heights `a` as a soft‑max of an
  unrestricted vector `w`.  This guarantees non‑negativity and a fixed
  L¹‑norm (= 1) without an explicit projection step.
* The objective is `max_k (a∗a)[k]`.  Its sub‑gradient at the active index
  `k_max` is
        g_i = 2·a_{k_max‑i}   (zero when the partner index is out of range).
* We apply an Adam‑style adaptive exponentiated‑gradient descent on `w`,
  i.e. we update `w ← w – η·m̂/√v̂` where `m̂,v̂` are the bias‑corrected
  first‑ and second‑moment estimates of the sub‑gradient `g`.  The soft‑max
  re‑normalises the vector after every step.
* Occasionally a small random perturbation is added to `w` to escape
  plateaus.
* The whole optimisation runs for at most `budget_s` seconds; the best
  interior solution (minimal `max_k (a∗a)[k]`) is kept.  The external score
  (`evaluate_sequence`) is only called once at the end, avoiding the
  O(n²) cost of the reference implementation.
* The length `n` is chosen automatically (larger when we have a long
  budget).  The implementation works for any `n` that fits comfortably
  into memory (≈ 20 k entries are well within the limits).

The returned list is a valid candidate for the outer grader.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax(w: np.ndarray) -> np.ndarray:
    """Stable soft‑max: returns a probability vector that sums to 1."""
    w_max = np.max(w)
    exp_w = np.exp(w - w_max)          # subtract max for numerical stability
    return exp_w / exp_w.sum()


def _conv_max(a: np.ndarray) -> tuple[float, int]:
    """
    Compute the linear convolution a∗a via FFT and return the maximal entry
    and its index.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def _initial_w(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Initialise the optimisation variables.
    If a global `height_sequence_1` is available we use it (normalised to sum = 1)
    and perturb it slightly; otherwise a small Gaussian vector is used.
    """
    # Start from the existing construction if it exists
    if "height_sequence_1" in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=float)
            if base.size >= n:
                a0 = base[:n].copy()
            else:
                a0 = np.pad(base, (0, n - base.size), constant_values=0.0)
            if a0.sum() > 0:
                a0 = a0 / a0.sum()
                # w = log(a0) + tiny noise
                w = np.log(np.maximum(a0, 1e-12))
                w += rng.normal(scale=0.01, size=n)
                return w
        except Exception:
            pass

    # Fallback: random Gaussian (roughly a Dirichlet draw after soft‑max)
    return rng.normal(scale=0.01, size=n)

# Main optimisation routine
# ----------------------------------------------------------------------

    else:
        n = 8000                      # tighter budget → smaller, faster problem

    # Optional: if a known construction exists and is longer, honour its length
    if "height_sequence_1" in globals():
        try:
            existing_len = len(height_sequence_1)
            if existing_len > n:
                n = existing_len
        except Exception:
            pass

    print(f"[setup] chosen length = {n}")

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    eta = 0.2                         # base step size (tuned empirically)

    # initialise optimisation state
    w = _initial_w(n, rng)
    m = np.zeros_like(w)              # first moment
    v = np.zeros_like(w)              # second moment
    t = 0

    # keep the best interior solution (minimum max‑convolution)
    a = _softmax(w)
    conv_max, k_max = _conv_max(a)
    best_internal = conv_max
    best_w = w.copy()
    best_a = a.copy()

    approx_score = 2.0 * n * conv_max
    print(f"[init] internal_max={conv_max:.8e} approx_score={approx_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        t += 1

        # --- current point -------------------------------------------------
        a = _softmax(w)                     # a sums to 1
        conv_max, k_max = _conv_max(a)

        # store improvement (inner bound)
        if conv_max < best_internal:
            best_internal = conv_max
            best_w = w.copy()
            best_a = a.copy()
            approx_score = 2.0 * n * conv_max
            print(f"[improve] step={t} time={time.time():.1f}s "
                  f"internal_max={conv_max:.8e} approx_score={approx_score:.6f}")

        # --- sub‑gradient of the active convolution entry -----------------
        idx = np.arange(n)
        partner = k_max - idx
        valid = (partner >= 0) & (partner < n)

        g = np.zeros_like(a)
        g[valid] = 2.0 * a[partner[valid]]   # sub‑gradient for max entry

        # --- Adam update on the unconstrained parameters `w` -------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # exponentiated‑gradient step (in log‑space)
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)

        # --- occasional random jitter to escape tiny plateaus ------------
        if rng.random() < 0.001:            # roughly one perturbation per 1000 steps
            w += rng.normal(scale=0.1, size=n)

        # fast exit if we are very close to the deadline
        if time.time() > deadline - 1.0:
            break

    # ------------------------------------------------------------------
    # Finalisation
    # ------------------------------------------------------------------
    # Recover the best interior point found
    final_a = best_a if best_a is not None else _softmax(best_w)

    # Convert to a plain Python list (the external grader expects this)
    candidate = final_a.tolist()

    # One external evaluation for reporting (expensive O(n²) call)
    try:
        final_score = evaluate_sequence(candidate)
        print(f"[final] best internal_max={best_internal:.8e} "
              f"approx_score={2.0 * n * best_internal:.6f} "
              f"external_score={final_score:.6f} length={n}")
    except Exception as exc:
        # In the unlikely event that the external call fails we still return
        # the candidate; the surrounding system will handle the NaN.
        print("[final] external evaluation failed:", exc)

    return candidate
