assistantfinal**Solution Overview**

The score to minimise is  

```
score = 2·n·max(conv(a,a)) / (sum(a)²)                (1)
```

`a` must be non‑negative, the scale of `a` does not change the score,
so we only have to decide the *shape* of the sequence.  
The classic choice `a_i = 1/√(i+1)` gives a score → π/2≈1.571.
A tiny improvement over that value is possible by slightly **tilting**
the curve: the left half of the sequence is kept close to the 1/√i
behaviour, the right half is forced to decay a little faster.
If the two halves meet smoothly the convolution at the centre drops
while the ℓ₁‑norm stays essentially the same – this lowers the
objective below π/2.

The optimiser works in three stages.

1. **Base candidates** – a small grid of the simple power‑law
   (`β≈0.5`) and a few random piece‑wise power‑law shapes.
2. **Polynomial perturbations** – for the best candidate we
   multiply it by a low‑degree polynomial `1+θ₁x+θ₂x²+…` (`x∈[-1,1]`),
   re‑normalise and keep the best perturbation.
3. **Projected sub‑gradient refinement** – the maximal convolution
   index `k*` is identified, the sub‑gradient  
   `g = a + shift` (`shift[i]=a[k*-i]` when the index is valid) is
   computed and a small step in the negative direction is taken.
   The step size follows the current ratio `max(conv)/sum²` and is
   slowly reduced.  The iterate with the lowest score is stored.

All heavy work uses an FFT based convolution (`O(n log n)`) and the
whole search respects the 1000 s time budget.  The final sequence is
returned as a plain list of `float`s – exactly the format required by
`evaluate_sequence`.

```python
"""
Propose a step‑function with a low evaluation score.

The algorithm proceeds in three phases:
  1) Grid / random search over simple power‑law and piece‑wise power‑law
     shapes (a_i = (i+1+offset)^(-beta) or a mixture of two exponents).
  2) Small low‑degree polynomial perturbations of the current best
     candidate (multiplicative factor 1+θ₁x+θ₂x²+… with x∈[-1,1]).
  3) Projected sub‑gradient descent on the true nonsmooth objective,
     using the index of the maximal convolution entry as the active
     constraint.  The best iterate seen during the descent is returned.

All convolutions are evaluated with FFT for speed, the routine stops
well before the allotted time limit and always returns a valid
non‑negative sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast evaluation of the objective (identical to the one used in the
# reference solution, but based on FFT).
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    n = len(seq)
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    # FFT length – next power of two for speed
    m = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft * fft, n=m)[: 2 * n - 1]
    max_conv = np.max(conv)
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Helpers for generating candidate sequences
# ----------------------------------------------------------------------
def _base_power_law(n: int, beta: float, offset: float) -> np.ndarray:
    """a_i = 1 / (i+1+offset)^beta   (i = 0..n-1)"""
    idx = np.arange(n, dtype=np.float64) + 1.0 + offset
    return 1.0 / (idx ** beta)


def _piecewise_power_law(
    n: int,
    beta1: float,
    beta2: float,
    split: float,
    offset: float,
) -> np.ndarray:
    """
    First `split` fraction of the indices uses exponent beta1,
    the remaining part uses exponent beta2, with a continuity factor
    at the split point.
    """
    split_idx = max(1, min(n - 1, int(split * n)))
    idx = np.arange(n, dtype=np.float64) + 1.0 + offset

    seq = np.empty(n, dtype=np.float64)
    seq[:split_idx] = idx[:split_idx] ** (-beta1)

    # continuity factor so that seq[split_idx-1] == seq[split_idx]
    factor = (idx[split_idx] ** (beta2 - beta1))
    seq[split_idx:] = factor * (idx[split_idx:] ** (-beta2))
    return seq


def _polynomial_modification(base: np.ndarray, thetas: np.ndarray) -> np.ndarray:
    """
    Multiply `base` by (1 + Σ θ_k * x^k) where x ∈ [-1,1] is a linear
    map of the index.  Negative values are clipped to zero and the
    result is re‑normalised to the original ℓ₁‑norm.
    """
    n = base.shape[0]
    # centred, scaled coordinate in [-1, 1]
    x = (np.arange(n) - (n - 1) / 2) / ((n - 1) / 2)
    weight = np.ones_like(base)
    for k, th in enumerate(thetas, start=1):
        weight += th * (x ** k)

    cand = base * np.maximum(weight, 0.0)
    s = np.sum(cand)
    if s == 0.0:
        return None
    cand *= np.sum(base) / s
    return cand


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the original non‑smooth objective.
# ----------------------------------------------------------------------
def _refine_subgradient(
    start_seq: np.ndarray,
    max_iter: int,
    deadline: float,
) -> np.ndarray:
    """Return the best sequence seen during a sub‑gradient descent."""
    a = start_seq.copy()
    target_sum = np.sum(a)
    best_seq = a.copy()
    best_score = _objective(a)

    step_factor = 0.7   # initial absolute step size factor
    momentum = np.zeros_like(a)

    n = len(a)
    m = 1 << ((2 * n - 1).bit_length())

    iter_cnt = 0
    while iter_cnt < max_iter and time.time() < deadline:
        # full convolution via FFT
        fft_a = np.fft.rfft(a, n=m)
        conv = np.fft.irfft(fft_a * fft_a, n=m)[: 2 * n - 1]
        k_max = int(np.argmax(conv))
        max_conv = conv[k_max]

        # sub‑gradient of the active index
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        shift = np.zeros_like(a)
        shift[mask] = a[idx[mask]]
        grad = a + shift                     # g_i = a_i + a_{k_max-i}

        # simple Nesterov‑style momentum
        momentum = 0.9 * momentum + grad

        # step size proportional to the current ratio r = max_conv / sum²
        r = max_conv / (target_sum * target_sum)
        step = step_factor * r

        new_a = a - step * momentum
        new_a = np.maximum(new_a, 0.0)

        s = np.sum(new_a)
        if s > 1e-12:
            new_a *= target_sum / s
        else:
            break

        new_score = _objective(new_a)

        if new_score < best_score:
            best_score = new_score
            best_seq = new_a.copy()
            # be a bit more aggressive after an improvement
            step_factor = min(step_factor * 1.05, 2.0)
        else:
            # slowly shrink step factor if we are not improving
            step_factor *= 0.97

        a = new_a
        iter_cnt += 1

        # occasional progress report
        if iter_cnt % 250 == 0:
            print(f"[refine] iter={iter_cnt}, score={new_score:.6f}")

        # early exit if we already beat a strong target
        if best_score < 1.50:
            print(f"[!] Early success: score={best_score:.6f} after {iter_cnt} iters")
            break

    return best_seq


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a coefficient sequence with a low evaluation score.

    The search consists of a cheap global exploration (grid + random
    piece‑wise shapes), a small polynomial perturbation stage and a
    final sub‑gradient refinement.  It always returns a valid list
    before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # tiny safety margin

    # ------------------------------------------------------------------
    # 1. Choose a length – 5000 provides a good trade‑off.
    # ------------------------------------------------------------------
    n = 5000
    print(f"[+] Starting search, length = {n}")

    # ------------------------------------------------------------------
    # 2. Global exploration ------------------------------------------------
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # a) simple power‑law grid (very cheap)
    betas = np.linspace(0.45, 0.55, 21)
    offsets = np.linspace(0.0, 5.0, 11)
    for beta in betas:
        for off in offsets:
            cand = _base_power_law(n, beta, off)
            score = _objective(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"[grid] β={beta:.3f}, off={off:5.2f} → score={score:.6f}")
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    # b) random piece‑wise power‑law shapes
    pw_iters = 2000
    for _ in range(pw_iters):
        if time.time() > deadline:
            break
        beta1 = np.random.uniform(0.45, 0.55)
        beta2 = np.random.uniform(0.55, 0.90)
        split = np.random.uniform(0.25, 0.75)
        off = np.random.uniform(0.0, 4.0)
        cand = _piecewise_power_law(n, beta1, beta2, split, off)
        score = _objective(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"[pw] β1={beta1:.3f}, β2={beta2:.3f}, split={split:.2f}, off={off:.2f}"
                  f" → score={score:.6f}")

    # ------------------------------------------------------------------
    # 3. Low‑degree polynomial perturbations of the current best ----------
    # ------------------------------------------------------------------
    if best_seq is None:                 # fallback – should never happen
        best_seq = np.ones(n, dtype=np.float64) / n
        best_score = _objective(best_seq)

    poly_iters = 4000
    degree = 3
    base = best_seq.copy()
    base_sum = np.sum(base)

    for i in range(poly_iters):
        if time.time() > deadline:
            break
        # amplitude shrinks as we progress
        max_amp = 0.7 * (1 - i / poly_iters)
        thetas = np.random.uniform(-max_amp, max_amp, size=degree)
        cand = _polynomial_modification(base, thetas)
        if cand is None:
            continue
        score = _objective(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"[poly] it={i:04d} → score={score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final sub‑gradient refinement ------------------------------------
    # ------------------------------------------------------------------
    max_refine_it = 5000
    refined = _refine_subgradient(best_seq, max_refine_it, deadline)

    final_score = _objective(refined)
    print(f"[+] Search finished. Best score = {final_score:.6f}")

    # Convert to plain Python list (evaluate_sequence will clip anyway)
    return [float(x) for x in refined]
```
