def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np, math
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # define seq generators
    def seq_powerlaw(n, p, gamma):
        i = np.arange(n, dtype=float)
        a = (np.power(i + gamma + 1.0, p) - np.power(i + gamma, p))
        total = np.power(n + gamma, p) - np.power(gamma, p)
        a = a / total
        return a

    def seq_logdiff(n, gamma):
        i = np.arange(n, dtype=float)
        a = np.log(i + gamma + 1.0) - np.log(i + gamma)
        total = np.log(n + gamma) - np.log(gamma)
        a = a / total
        return a

    def seq_exponential(n, r):
        # a_i = (1-r)*r**i
        a = (1 - r) * (r ** np.arange(n, dtype=float))
        total = 1 - r**n
        a = a / total
        return a

    def eval_score(seq):
        # helper computing evaluation using fast formula
        s = np.sum(seq)
        if s < 1e-12:
            return float('inf')
        b = np.convolve(seq, seq)
        maxb = np.max(b)
        return float(2 * len(seq) * maxb / (s * s))

    best_score = float('inf')
    best_params = None
    best_family = None

    n_small = 5000

    # Power-law family search
    p_vals = np.linspace(0.05, 0.5, 19)  # 0.05,0.075,...,0.5
    gamma_vals = [0,1,2,5,10,20,50,100,200,500,1000,2000,4000]  # ensure <= n_small-1

                best_score = score
                best_params = (p, gamma)

                print(f"[grid power] new best {score:.6f} p={p:.3f} gamma={gamma}")

        if score < best_score:

            best_params = (gamma,)

"""
A fast search for a low‑evaluation non‑negative step function.

The algorithm explores three simple parametric families that are known to give
near‑optimal self‑convolution behaviour:

1. Power‑law differences          a_i = (i+γ+1)^p – (i+γ)^p
2. Log‑difference (p → 0 limit)  a_i = log(i+γ+1) – log(i+γ)
3. Exponential tail               a_i = (1‑r)·r^i

All families are normalised to L1‑norm = 1 (the evaluation is scale‑invariant).

Steps
-----
* **Coarse grid** on a moderate length (n≈5 000) – we sweep p∈[0.05,0.5] and
  many values of the offset γ (up to the size of the vector) as well as the
  parameters of the other families. The best triple (family, parameters, score)
  is recorded.
* **Local hill‑climbing** – starting from the best grid point we perturb the
  parameters (Gaussian steps) on a slightly larger vector (n≈8 000) and keep
  any improvement. Occasionally we also try to switch the whole family.
* **Final construction** – the best parameters are used to build a longer
  sequence (n≈20 000).  A cheap “pairwise‑transfer” stochastic refinement is
  applied directly on the coefficients while preserving the L1‑norm.
* The final list is returned.  The score is printed for reference but the
  function only needs to return the sequence.

The whole search respects the 1000 s time budget.  No file I/O or external
solvers are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions for the three families
# ----------------------------------------------------------------------
def _seq_powerlaw(n: int, p: float, gamma: float) -> np.ndarray:
    """Power‑law difference sequence, L1‑normalised."""
    i = np.arange(n, dtype=float)
    a = np.power(i + gamma + 1.0, p) - np.power(i + gamma, p)
    total = np.power(n + gamma, p) - np.power(gamma, p)
    # guard against division by zero (should not happen for p>0)
    if total == 0:
        total = 1.0
    a /= total
    return a


def _seq_logdiff(n: int, gamma: float) -> np.ndarray:
    """Log‑difference sequence, L1‑normalised (γ≥1)."""
    i = np.arange(n, dtype=float)
    a = np.log(i + gamma + 1.0) - np.log(i + gamma)
    total = np.log(n + gamma) - np.log(gamma)
    if total == 0:
        total = 1.0
    a /= total
    return a


def _seq_exponential(n: int, r: float) -> np.ndarray:
    """Geometric tail (0<r<1), L1‑normalised."""
    i = np.arange(n, dtype=float)
    a = (1.0 - r) * (r ** i)
    total = 1.0 - r ** n
    if total == 0:
        total = 1.0
    a /= total
    return a


def _eval_score(seq: np.ndarray) -> float:
    """
    Fast version of `evaluate_sequence` for our own use:
        2 * n * max(conv(seq,seq)) / (sum(seq)**2)
    All our sequences are non‑negative and far below 1000, so the clipping
    performed inside `evaluate_sequence` does not change the result.
    """
    s = np.sum(seq)
    if s < 1e-12:
        return float('inf')
    conv = np.convolve(seq, seq)
    return float(2.0 * len(seq) * np.max(conv) / (s * s))


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a step‑function sequence whose evaluation (see `evaluate_sequence`)
    is as low as possible.  Returns the list of coefficients.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # 1) Coarse grid search on a moderate length
    # ------------------------------------------------------------------
    n_small = 5000
    best_score = float('inf')
    best_family = None
    best_params = None

    # ---- Power‑law family ------------------------------------------------
    p_vals = np.linspace(0.05, 0.50, 19)                # 0.05 … 0.50
    gamma_vals = [0, 1, 2, 5, 10, 20, 50, 100, 200,
                  500, 1000, 1500, 2000, 2500, 3000, 3500, 4000]  # ≤ n_small‑1
    for p in p_vals:
        for gamma in gamma_vals:
            if gamma >= n_small:
                continue
            seq = _seq_powerlaw(n_small, float(p), float(gamma))
            score = _eval_score(seq)
            if score < best_score:
                best_score = score
                best_family = 'power'
                best_params = (float(p), int(gamma))
                print(f"[grid power] new best {score:.6f} p={p:.3f} γ={gamma}")
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    # ---- Log‑difference family ------------------------------------------
    for gamma in [1, 2, 5, 10, 20, 50, 100, 200,
                  500, 1000, 1500, 2000, 2500, 3000, 3500, 4000]:
        if gamma >= n_small:
            continue
        seq = _seq_logdiff(n_small, float(gamma))
        score = _eval_score(seq)
        if score < best_score:
            best_score = score
            best_family = 'log'
            best_params = (int(gamma),)
            print(f"[grid log] new best {score:.6f} γ={gamma}")
        if time.time() > deadline:
            break

    # ---- Exponential family ---------------------------------------------
    for r in np.linspace(0.70, 0.999, 30):
        seq = _seq_exponential(n_small, float(r))
        score = _eval_score(seq)
        if score < best_score:
            best_score = score
            best_family = 'exp'
            best_params = (float(r),)
            print(f"[grid exp] new best {score:.6f} r={r:.4f}")
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 2) Local hill‑climbing on a slightly larger vector
    # ------------------------------------------------------------------
    n_mid = 8000
    cur_family = best_family
    cur_params = best_params
    cur_score = best_score

    while time.time() < deadline:
        # small Gaussian perturbation of the current parameters
        if cur_family == 'power':
            p, gamma = cur_params
            p_new = np.clip(p + np.random.randn() * 0.02, 0.01, 0.99)
            gamma_new = int(np.clip(gamma + np.random.randn() * 200, 0, n_mid - 1))
            seq = _seq_powerlaw(n_mid, p_new, float(gamma_new))
            score = _eval_score(seq)
            if score < cur_score:
                cur_score = score
                cur_params = (p_new, gamma_new)
                best_family, best_params, best_score = cur_family, cur_params, score
                print(f"[refine power] improved {score:.6f} p={p_new:.3f} γ={gamma_new}")

        elif cur_family == 'log':
            gamma, = cur_params
            gamma_new = int(np.clip(gamma + np.random.randn() * 200, 1, n_mid - 1))
            seq = _seq_logdiff(n_mid, float(gamma_new))
            score = _eval_score(seq)
            if score < cur_score:
                cur_score = score
                cur_params = (gamma_new,)
                best_family, best_params, best_score = cur_family, cur_params, score
                print(f"[refine log] improved {score:.6f} γ={gamma_new}")

        else:  # exponential
            r, = cur_params
            r_new = np.clip(r + np.random.randn() * 0.005, 0.01, 0.9999)
            seq = _seq_exponential(n_mid, r_new)
            score = _eval_score(seq)
            if score < cur_score:
                cur_score = score
                cur_params = (r_new,)
                best_family, best_params, best_score = cur_family, cur_params, score
                print(f"[refine exp] improved {score:.6f} r={r_new:.4f}")

        # occasionally try a completely different family
        if np.random.rand() < 0.01:
            new_family = np.random.choice(['power', 'log', 'exp'])
            if new_family == 'power':

                gamma = np.random.randint(0, n_mid)
                seq = _seq_powerlaw(n_mid, p, float(gamma))
                score = _eval_score(seq)
                if score < cur_score:
                    cur_family, cur_params, cur_score = new_family, (p, gamma), score
                    best_family, best_params, best_score = cur_family, cur_params, score
                    print(f"[switch] power new best {score:.6f} p={p:.3f} γ={gamma}")

            elif new_family == 'log':
                gamma = np.random.randint(1, n_mid)
                seq = _seq_logdiff(n_mid, float(gamma))
                score = _eval_score(seq)
                if score < cur_score:
                    cur_family, cur_params, cur_score = new_family, (gamma,), score
                    best_family, best_params, best_score = cur_family, cur_params, score
                    print(f"[switch] log new best {score:.6f} γ={gamma}")

                r = np.random.uniform(0.70, 0.999)
                seq = _seq_exponential(n_mid, r)
                score = _eval_score(seq)
                if score < cur_score:

                    best_family, best_params, best_score = cur_family, cur_params, score
                    print(f"[switch] exp new best {score:.6f} r={r:.4f}")

        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 3) Build the final (large) sequence and final refinement
    # ------------------------------------------------------------------
    n_big = 20000
    if best_family == 'power':
        p, gamma = best_params
        seq = _seq_powerlaw(n_big, p, float(gamma))
    elif best_family == 'log':
        gamma, = best_params
        seq = _seq_logdiff(n_big, float(gamma))
    else:  # exponential
        r, = best_params
        seq = _seq_exponential(n_big, r)

    cur_score = _eval_score(seq)
    print(f"[final] initial score {cur_score:.6f} (family={best_family}, params={best_params})")

    # ----- very light stochastic fine‑tuning on the coefficients ----------
    #   we keep the L1‑norm exactly 1 by transferring a tiny amount δ
    #   between two randomly chosen positions.
    while time.time() < deadline:
        i, j = np.random.choice(len(seq), 2, replace=False)
        delta = (np.random.rand() - 0.5) * 1e-5   # signed tiny step

        cand = seq.copy()
        cand[i] += delta
        cand[j] -= delta
        score = _eval_score(cand)
        if score < cur_score:
            seq = cand
            cur_score = score
            # occasional progress report
            if np.random.rand() < 0.005:
                print(f"[seq refine] improved to {cur_score:.6f}")

    print(f"[result] final score {cur_score:.6f} (family={best_family})")
    # return a plain Python list as required by the evaluator
    return seq.tolist()
