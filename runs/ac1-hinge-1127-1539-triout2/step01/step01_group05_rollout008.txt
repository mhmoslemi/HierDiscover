def compute_gradient(a, w):
    n = a.shape[0]
    nfft_corr = _next_pow2(3*n-2)
    wf = np.fft.rfft(w, n=nfft_corr)
    af_rev = np.fft.rfft(a[::-1], n=nfft_corr)
    corr_full = np.fft.irfft(wf * af_rev, n=nfft_corr)
    grad = 2.0 * corr_full[n-1:n-1+n]
    return grad
```

def cross_corr(w, a_rev):
    n = a_rev.shape[0]
    nfft = next_pow2(w.shape[0] + a_rev.shape[0] - 1)   # len w = 2n-1, a_rev = n => 3n-2
    wf = np.fft.rfft(w, n=nfft)
    ar = np.fft.rfft(a_rev, n=nfft)
    corr = np.fft.irfft(wf*ar, n=nfft)
    start = n-1
    return 2.0 * corr[start:start+n]
```

"""
Propose a non‑negative step‑function sequence that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The quantity equals the maximum value of the self‑convolution when the
sequence is normalised to  sum(a)=sqrt(2 n).  The algorithm uses a
smoothed‑max gradient descent driven by the Adam optimiser, together with
periodic random jitter and a cheap simulated‑annealing style mass‑transfer
move.  The starting point is either the previously found best construction
(`height_sequence_1`) or the analytical 1/√(i+½) shape.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return the (linear) convolution a * a (length 2·n‑1)."""
    n = a.shape[0]
    nfft = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
    return conv


def _gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max of the convolution w.r.t. a.
    g = 2 * (w ⊛ a_rev) where ⊛ denotes cross‑correlation.
    """
    n = a.shape[0]
    # w has length 2n‑1, a_rev length n  → full correlation length 3n‑2
    nfft = _next_pow2(3 * n - 2)
    wf = np.fft.rfft(w, n=nfft)
    ar = np.fft.rfft(a[::-1], n=nfft)
    corr_full = np.fft.irfft(wf * ar, n=nfft)
    # extract the centre part (indices n‑1 … n‑1+n‑1)
    grad = 2.0 * corr_full[n - 1 : n - 1 + n]
    return grad


def _score(a: np.ndarray) -> float:
    """Fast evaluation of the objective = max(a * a)."""
    conv = _conv_self(a)
    return float(conv.max())


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    Search for a sequence of non‑negative heights with a small
    max‑convolution value.

    Parameters

    seed : int

    budget_s : int

    Returns
    -------
    list[float]
        The best sequence found (as a plain Python list).
    """

    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # safety margin for the harness

    # --------------------------------------------------------------
    # 1. Choose the length and initialise a.
    # --------------------------------------------------------------

    # Normalise the sum to sqrt(2 n) – the objective is scale‑invariant.
    target_sum = np.sqrt(2.0 * n)
    a *= target_sum / a.sum()
    a = np.maximum(a, 0.0)

    # --------------------------------------------------------------
    # 2. Adam optimiser state.
    # --------------------------------------------------------------
    m = np.zeros_like(a)  # first moment
    v = np.zeros_like(a)  # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # --------------------------------------------------------------
    # 3. Hyper‑parameter schedules.
    # --------------------------------------------------------------
    gamma = 1.0          # soft‑max temperature
    gamma_max = 70.0
    lr_initial = 0.12   # base learning rate for Adam
    lr_min = 0.001

    best_a = a.copy()
    best_val = _score(best_a)

    iteration = 0

    # --------------------------------------------------------------
    # 4. Main optimisation loop.
    # --------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # ----- convolution and current score -----
        conv = _conv_self(a)
        cur_val = float(conv.max())

        if cur_val < best_val:

        # Subtract cur_val to keep the exponent bounded.

        m = beta1 * m + (1.0 - beta1) * grad

        gamma = min(gamma * 1.003, gamma_max)

        if iteration % 277 == 0:

            # random direction

            # apply the transfer respecting non‑negativity
            a_i = a[i] - delta
            a_j = a[j] + delta
            if a_i < 0.0:
                # undo the excess
                a_j += a_i  # a_i is negative
                a_i = 0.0
            if a_j < 0.0:
                a_i += a_j
                a_j = 0.0
            a[i] = a_i
            a[j] = a_j
            # re‑normalise (the total mass is unchanged, but rounding may drift)
            a *= target_sum / a.sum()
            # evaluate the perturbed candidate
            val = _score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                elapsed = time.time() - start_time
                print(
                    f"Iter {iteration:6d} (SA): new best {best_val:.6f}  (t={elapsed:.1f}s)"
                )

    # --------------------------------------------------------------
    # 5. Return the best sequence discovered.
    # --------------------------------------------------------------
    return best_a.tolist()
