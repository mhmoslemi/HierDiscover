    np.random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5

    # Determine lengths to try (list)
    lengths = [4000, 5600, 7200, 9000]   # maybe adapt to time.

    best_score = np.inf

    for n in lengths:
        if time.time() > deadline: break

        # we will create a set of seed shapes

        # analytic shape:
        a1 = 1.0/np.sqrt(np.arange(n, dtype=np.float64) + 0.5)
        # another shape (semicircle)
        x = (np.arange(n)+0.5)/n
        a2 = np.sqrt(np.maximum(0, 1 - (2*x-1)**2))
        # third shape (exponential)
        scale = n/5
        a3 = np.exp(-np.arange(n)/scale)
        # fourth shape = reverse of first (increasing)
        a4 = a1[::-1]

        for arr in [a1, a2, a3, a4]:
            init_shapes.append(arr.copy())

        # optional shape from height_sequence_1 truncated or padded:
        if 'height_sequence_1' in globals():

                init_shapes.append(hs[:n].copy())
            else:

                pad[:hs.size] = hs
                init_shapes.append(pad)

        # loop over shapes
        for a_init in init_shapes:
            if time.time() > deadline:
                break
            # normalize sum to sqrt(2n)
            target_sum = np.sqrt(2.0 * n)
            a = a_init.copy()
            if a.sum() == 0:
                a = np.full(n, target_sum / n)
            else:
                a *= target_sum / a.sum()

            # track best for this run
            best_local_a = a.copy()
            best_local_score = _compute_score(a)

            # hyperparameters
            gamma = 2.0
            eta = 0.2
            t = 0.12 # combination factor for f,g updates
            # schedule parameters:
            step = 0
            while time.time() < deadline:
                step += 1

                # Compute convolution and max
                conv = np.convolve(a, a)
                max_conv = conv.max()

                if gamma < 50:

                    w = np.exp(gamma * (conv - max_conv))
                    w /= w.sum()

                    nfft = 1 << ((2*n-1).bit_length())

                    wf = np.fft.rfft(w, n=nfft)
                    af_rev = np.fft.rfft(a[::-1], n=nfft)
                    corr_full = np.fft.irfft(wf * af_rev, n=nfft)
                    grad = 2.0 * corr_full[n-1:n-1+n]
                else:

                    k = np.argmax(conv)
                    # compute gradient
                    grad = np.zeros_like(a)
                    lo = max(0, k - (n-1))

                    idx = np.arange(lo, hi+1)
                    grad[idx] = 2.0 * a[k - idx]

                # renorm

"""
Propose a sequence of non‑negative heights that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm is a hybrid of several ideas:

1. **Multiple start shapes** – analytic 1/√(i+½), a semicircle,
   an exponential decay, a reversed analytic shape and (if available)
   the previously best global construction.  All are renormalised so that
   sum(a)=√(2n).  Trying several seeds helps to escape local minima.

2. **Two‑phase gradient descent**
   * **Early phase** – a soft‑max approximation of the convolution maximum
     (temperature `γ`) gives a smooth gradient (cross‑correlation of the
     soft‑max weights with the reversed vector).  `γ` is slowly increased.
   * **Late phase** – once `γ` is large we switch to the exact sub‑gradient
     of the true maximum (only the indices where the convolution attains
     its maximum contribute).  This provides a sharp descent step.

   The step size `η` is annealed and a simple simulated‑annealing acceptance
   rule permits occasional uphill moves.

3. **Active‑set linear‑programming boost** – every few hundred iterations we
   solve a tiny LP:

        maximise   Σ g_i
        subject to g_i ≥ 0,
                   (f * g)_k ≤ M   for all k where the current convolution
                                   is within a small tolerance of its
                                   maximum M.

   The LP provides a direction `g` that increases the L₁‑mass while keeping
   the convolution bounded.  The solution is rescaled to the required L₁‑norm
   and mixed into the current iterate with a small mixing factor.

4. **Random jitter & restarts** – periodic tiny Gaussian perturbations
   and occasional full restarts from the best known vector help to avoid
   plateaus.

The whole search runs until the allotted time (default 1000 s) is exhausted.
The best sequence found so far is returned as a plain Python list.
"""

import time
import numpy as np
from typing import List

# cvxpy is optional – if unavailable the LP boost is simply skipped.
try:
    import cvxpy as cp
    _HAS_CVXPY = True
except Exception:  # pragma: no cover
    _HAS_CVXPY = False


def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _normalize(a: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale a to have the prescribed L₁ sum, keeping non‑negativity."""
    s = a.sum()
    if s == 0.0:
        return np.full_like(a, target_sum / a.size)
    return a * (target_sum / s)


def _autocorr_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """Return the full (linear) autocorrelation a * a using FFT."""
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * a.size - 1]
    return conv


def _softmax_gradient(a: np.ndarray, conv: np.ndarray, max_c: float,
                      gamma: float, nfft_corr: int) -> np.ndarray:
    """Gradient of the soft‑max approximation of max(conv)."""
    # soft‑max weights
    w = np.exp(gamma * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:                     # safety fallback
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # cross‑correlation w ⊛ a_rev
    wf = np.fft.rfft(w, n=nfft_corr)
    af_rev = np.fft.rfft(a[::-1], n=nfft_corr)
    corr_full = np.fft.irfft(wf * af_rev, n=nfft_corr)

    # the valid part (length n) starts at index n‑1
    n = a.size
    grad = 2.0 * corr_full[n - 1 : n - 1 + n]
    return grad


def _max_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Exact sub‑gradient of the true max of the autocorrelation."""
    n = a.size
    k = int(conv.argmax())                     # index of the maximum
    grad = np.zeros_like(a)

    # indices where the product a_i * a_{k-i} actually contributes
    i_lo = max(0, k - (n - 1))
    i_hi = min(k, n - 1)
    idx = np.arange(i_lo, i_hi + 1, dtype=int)
    # derivative of a_i a_{k-i} w.r.t. a_i is a_{k-i}
    grad[idx] = 2.0 * a[k - idx]
    return grad


def _compute_score(a: np.ndarray) -> float:
    """Fast evaluation of the objective (equivalent to max convolution)."""
    n = a.size
    nfft = _next_pow2(2 * n - 1)
    conv = _autocorr_fft(a, nfft)
    max_c = conv.max()
    # Because sum(a) is normalised to sqrt(2n), the denominator cancels.
    return float(max_c)


def _lp_boost(f: np.ndarray, M: float, target_sum: float,
              active_tol: float = 1e-3) -> np.ndarray:
    """
    Solve a tiny LP that maximises Σ g_i under the constraints
        (f * g)_k ≤ M   for all k with (f*f)_k ≥ M - active_tol·M .
    Returns the scaled vector g (or None on failure).
    """
    if not _HAS_CVXPY:          # pragma: no cover
        return None

    n = f.size
    conv = _autocorr_fft(f, _next_pow2(2 * n - 1))
    active = np.where(conv >= M - active_tol * M)[0]
    if active.size == 0:        # fallback to the single maximal index
        active = np.array([int(conv.argmax())])

    g = cp.Variable(n, nonneg=True)
    constraints = []

    for k in active:
        # Build coefficient vector for (f * g)_k = Σ_i f_i * g_{k-i}
        coeff = np.zeros(n)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        i_range = np.arange(i_low, i_high + 1, dtype=int)
        coeff[k - i_range] = f[i_range]
        constraints.append(cp.sum(cp.multiply(coeff, g)) <= M)

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    except Exception:            # pragma: no cover
        try:
            prob.solve(solver=cp.SCS, max_iters=5000)
        except Exception:
            return None

    if g.value is None:
        return None

    g0 = np.maximum(np.array(g.value, dtype=float), 0.0)
    s = g0.sum()
    if s == 0.0:
        return None
    # Scale to meet the required L₁ norm.
    return g0 * (target_sum / s)

    """
    Search for a sequence of non‑negative heights that minimises the
    evaluation function.  The routine follows the hybrid strategy described
    in the module docstring and returns the best sequence found before the
    time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # 1. Choose a set of problem sizes (different lengths give diversified
    #    search spaces).  All lengths are chosen large enough to be competitive
    #    but still allow many iterations within the time budget.
    # ------------------------------------------------------------------
    lengths = [4000, 5600, 7200, 9000]
    # If a previous global construction exists, use its length as an extra size.
    if 'height_sequence_1' in globals():
        try:
            lengths.append(len(height_sequence_1))   # type: ignore[name-defined]
        except Exception:
            pass

    best_global_score = np.inf

    # ------------------------------------------------------------------
    # 2. Loop over chosen lengths and several initial shapes.
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break

        target_sum = np.sqrt(2.0 * n)
        nfft_conv = _next_pow2(2 * n - 1)
        nfft_corr = _next_pow2(3 * n - 2)

        # ---- generate a few deterministic start shapes -----------------
        start_shapes = []

        # (a) analytic 1 / sqrt(i+0.5)
        a1 = 1.0 / np.sqrt(np.arange(n, dtype=np.float64) + 0.5)

        start_shapes.append(a2)

        # (c) exponential decay
        scale = n / 5.0
        a3 = np.exp(-np.arange(n, dtype=np.float64) / scale)
        start_shapes.append(a3)

        # (d) reversed analytic shape (increasing)
        start_shapes.append(a1[::-1].copy())

        # (e) previous best construction, if it exists

                if prev.size >= n:
                    start_shapes.append(prev[:n].copy())
                else:
                    pad = np.zeros(n, dtype=np.float64)
                    pad[:prev.size] = prev
                    start_shapes.append(pad)
            except Exception:
                pass

        # ------------------------------------------------------------------
        # 3. Run an optimisation loop for each start shape.
        # ------------------------------------------------------------------
        for init_a in start_shapes:
            if time.time() > deadline:
                break

            # Normalise initial vector.
            a = _normalize(init_a.astype(np.float64), target_sum)

            # Track the best found for this run.
            best_local_a = a.copy()
            best_local_score = _compute_score(a)

            # Hyper‑parameters.
            gamma = 5.0          # soft‑max temperature
            gamma_max = 200.0

            eta_min = 0.001
            mix_factor = 0.12    # mixing weight for LP‑boosted update
            iteration = 0

            while time.time() < deadline:
                iteration += 1

                # ---------------------------------------------------------
                # 3.1 Compute autocorrelation and its maximum.
                # ---------------------------------------------------------
                conv = _autocorr_fft(a, nfft_conv)
                max_c = conv.max()

                # ---------------------------------------------------------
                # 3.2 Gradient (soft‑max while gamma is moderate,
                #            exact sub‑gradient once gamma is high).
                # ---------------------------------------------------------
                if gamma < 60.0:
                    grad = _softmax_gradient(a, conv, max_c, gamma, nfft_corr)
                else:
                    grad = _max_subgradient(a, conv)

                # ---------------------------------------------------------
                # 3.3 Gradient descent step + projection + renormalisation.
                # ---------------------------------------------------------
                a_new = a - eta * grad

                # 3.5 Evaluate the candidate.
                # ---------------------------------------------------------
                score_new = _compute_score(a_new)

                # ---------------------------------------------------------

                accept = False

                else:
