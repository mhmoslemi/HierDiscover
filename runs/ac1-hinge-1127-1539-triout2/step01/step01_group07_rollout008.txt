Implement projection to simplex: the algorithm "project onto simplex" which ensures sum=1 and all entries nonnegative.

Pseudo steps:

```
def proj_simplex(v):
    # map v (numpy array) to simplex (sum=1, nonneg)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

We apply after each gradient update.

"""
Propose a step‑function sequence with a very low evaluation value.

Algorithm
---------
1. **Coarse parametric search** – on a moderate length (≈8000) we sweep the
   power‑law exponent `p` (≈0.36‒0.44) and an offset `γ` (≈0‒400).  The sequence
   a_i = (i+γ+1)^p-(i+γ)^p is normalised to L1‑norm 1.  The best (p,γ) pair
   is kept as a warm start.

2. **Large construction** – using the best parameters we build a much longer
   vector (default n≈20000).  This already improves the bound compared to the
   previous 5675‑element construction.

3. **Projected sub‑gradient descent** – the objective after normalisation
   reduces to 2·n·max(conv(a,a)).  We repeatedly:
        * compute the self‑convolution (FFT based) and locate all indices
          where the maximum is attained,
        * obtain a sub‑gradient of the max‑norm: for a max‑index set K the
          gradient is 2·∑_{k∈K} a_{k‑i} (i.e. convolution of the mask with the
          reversed vector),
        * take a small step opposite to the gradient,
        * project back onto the simplex (non‑negative, sum = 1).

   The step size decays as 1/√iteration.  Whenever an improvement is found
   the best sequence is updated.  Random perturbations are injected after a
   long period without improvement to escape plateaus.

All heavy convolution work uses `scipy.signal.fftconvolve` which is
near‑linear in the length, allowing thousands of iterations well within the
1000 s budget.

The returned list is a valid non‑negative sequence; the official evaluator
will compute the exact score.
"""

import time
import math
import numpy as np
from scipy.signal import fftconvolve


def _generate_sequence(n: int, p: float, gamma: float) -> np.ndarray:
    """Return a normalised non‑negative sequence of length n.
    a_i = (i+γ+1)^p - (i+γ)^p , then scaled to L1‑norm 1."""
    i = np.arange(n, dtype=float)
    a = np.power(i + gamma + 1.0, p) - np.power(i + gamma, p)
    total = np.power(n + gamma, p) - np.power(gamma, p)
    a /= total
    # numerical safety – force exact sum = 1
    a = a / a.sum()
    return a


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  Returns the best list found before the time budget
    expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1. Coarse grid search on medium size vector to locate good (p,γ)
    # ------------------------------------------------------------------
    n_grid = 8000
    best_score = float("inf")
    best_params = (0.4, 10.0)

    # ranges inspired by the theoretical optimum p≈0.405
    p_vals = np.linspace(0.36, 0.44, 41)          # step ≈0.002
    gamma_vals = np.arange(0, 401, 20)           # 0 … 400 step 20

    for p in p_vals:
        for gamma in gamma_vals:
            if time.time() > deadline:
                break
            seq = _generate_sequence(n_grid, p, float(gamma))
            # fast evaluation (sum = 1 → denominator = 1)
            conv = fftconvolve(seq, seq, mode="full")
            score = 2.0 * n_grid * conv.max()
            if score < best_score:
                best_score = score
                best_params = (p, float(gamma))
                print(
                    f"[grid] new best: score={score:.6f}  (p={p:.4f}, γ={gamma})"
                )
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 2. Build a larger construction from the best parameters
    # ------------------------------------------------------------------
    n_big = 20000                     # typical size; enlarged if time permits
    p_best, gamma_best = best_params
    seq = _generate_sequence(n_big, p_best, gamma_best)

    # Ensure exact L1‑normalisation (important for the analytical formula)
    seq = seq / seq.sum()
    n = n_big
    conv = fftconvolve(seq, seq, mode="full")
    best_score = 2.0 * n * conv.max()
    best_seq = seq.copy()
    print(
        f"[init] n={n}, p={p_best:.4f}, γ={gamma_best:.1f} → score={best_score:.6f}"
    )

    # ------------------------------------------------------------------
    # 3. Projected sub‑gradient descent on max(conv(seq,seq))
    # ------------------------------------------------------------------
    iteration = 0
    step0 = 0.20                     # initial step size
    no_improve = 0                  # counter for stagnation detection

    while time.time() < deadline:
        iteration += 1

        # current convolution and maximal value
        conv = fftconvolve(seq, seq, mode="full")
        max_val = conv.max()
        max_inds = np.flatnonzero(conv == max_val)

        # sub‑gradient of the ∞‑norm (average over all maximal indices)
        a_rev = seq[::-1]
        mask = np.zeros(2 * n - 1, dtype=float)
        mask[max_inds] = 1.0 / len(max_inds)

        # gradient = 2 * (mask * a_rev) evaluated at the centre slice
        grad_full = fftconvolve(mask, a_rev, mode="full")
        grad = 2.0 * grad_full[(n - 1) : (n - 1) + n]

        # diminishing step size
        step = step0 / math.sqrt(iteration)

        # take a step opposite to the gradient and re‑project onto simplex
        cand = seq - step * grad
        cand = _project_onto_simplex(cand)

        # evaluate candidate
        conv_cand = fftconvolve(cand, cand, mode="full")
        cand_score = 2.0 * n * conv_cand.max()

        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_seq = cand.copy()
            seq = cand                     # continue from the improved point
            no_improve = 0
            print(f"[iter {iteration:05d}] improved score → {best_score:.6f}")
        else:
            no_improve += 1
            # occasional stochastic shake‑up to escape flat regions
            if no_improve % 1500 == 0:
                noise = np.random.randn(n) * 1e-5
                seq = _project_onto_simplex(seq + noise)
                no_improve = 0

        # periodic status report
        if iteration % 1000 == 0:
            print(
                f"[status] iter={iteration}, best_score={best_score:.6f}, step={step:.5f}"
            )

    print(f"[final] best score after optimisation = {best_score:.6f}")
    return best_seq.tolist()
