def solve_lp_for_g(f, M):
   n = len(f)
   # 2n-1 constraints for conv
   rows = []
   cols = []
   data = []
   for k in range(2*n-1):
       # i ranges from max(0, k-(n-1)) to min(n-1, k)
       i_min = max(0, k-(n-1))
       i_max = min(n-1, k)
       for i in range(i_min, i_max+1):
           j = k - i  # g_j index
           rows.append(k)
           cols.append(j)
           data.append(f[i])
   A = sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
   # A @ g <= M, i.e., A @ g <= M (vector)
   b = np.full(2*n-1, M)
   # objective c = -1 for each variable (maximize sum)
   c = -np.ones(n)
   # bounds: (0, None) for each g
   res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
   # check success
   return res

def build_conv_matrix(f):
    n = len(f)
    # build rows for each k (0..2n-2)
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        for i in range(i_start, i_end+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    A = sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return A

def propose_candidate(seed=123, budget_s=1000):
    import time, numpy as np, math, scipy.sparse as sp, scipy.optimize
    # time limit
    deadline = time.time() + budget_s - 0.5

    # initial n
    n = 2000
    # initial parameters: use p=0.4, gamma=10 maybe
    p = 0.4
    gamma = 10.0
    seq = _generate_sequence(n, p, gamma) # returns normalized L1=1 vector (np.ndarray)
    best_seq = seq.copy()
    best_score = evaluate_sequence(seq.tolist())
    print(f"[init] n={n}, p={p}, gamma={gamma}, score={best_score:.6f}")

    while time.time() < deadline:
        # Build convolution matrix of current f
        A = sp.diags([np.full(n, fi) for fi in seq], offsets=np.arange(n), shape=(2*n-1, n), format='csr')
        # Append column for -M
        M_col = -np.ones((2*n-1, 1))
        A_ub = sp.hstack([A, M_col], format='csr')
        b_ub = np.zeros(2*n-1)

        # Equality constraint sum(g)=1
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        # Objective
        c = np.zeros(n+1)
        c[-1] = 1.0   # minimize M

        # bounds
        bounds = [(0, None)] * (n+1)

        # Solve
        res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if not res.success:
            print("[lp] solve failed:", res.message)
            break

        g = res.x[:n]
        M_opt = res.x[-1]  # minimal possible max convolution with f

        # Compute true self-convolution max for g
        conv_g = np.convolve(g, g)
        max_g = np.max(conv_g)

        # Compute evaluation for g
        score_g = evaluate_sequence(g.tolist())
        print(f"[lp] M_opt={M_opt:.6e}, max_g={max_g:.6e}, score={score_g:.6f}")

        # Accept if improvement
        if score_g + 1e-12 < best_score:
            best_score = score_g
            best_seq = g.copy()

            print("[lp] no improvement")
            break

    # Optionally, after terminating early, we can do a fine-tuning random moves on best_seq.

"""
Propose a step function via a coarse grid search followed by a linear‑programming
refinement.

The candidate family is a power‑law difference
    a_i = ((i + γ + 1)**p - (i + γ)**p)    (p∈(0,1), γ≥0)

which, after ℓ₁–normalisation, approximates the density
f(x) = p·x^{p‑1} on ℝ₊.  For a fixed length n the evaluation score is

    score = 2·n·max( f * f )      (because we keep sum(f)=1)

The best (p,γ) from a quick grid (n≈1500) is taken as a starting point.
Then we repeatedly solve a tiny linear programme:

    minimise   M
    subject to   (f * g)_k ≤ M   for all k
                 Σ_j g_j = 1
                 g_j ≥ 0

The optimal g spreads the mass of f as evenly as possible while keeping the
maximum convolution ≤ M.  Mixing the old vector with the new one
    f ← (1‑t)·f + t·g   (t starts at 0.5 and shrinks) reduces the objective.
A few stochastic “mass‑transfer’’ tweaks are finally applied.

All work respects the 1000 s budget; the best sequence found is returned as a
plain list of floats.
"""

import time
import numpy as np
import math
import random
import scipy.sparse as sp
from scipy.optimize import linprog


def _generate_sequence(n: int, p: float, gamma: float) -> np.ndarray:
    """Power‑law difference sequence, ℓ₁‑normalised to 1."""
    i = np.arange(n, dtype=float)
    a = (np.power(i + gamma + 1.0, p) - np.power(i + gamma, p))
    total = np.power(n + gamma, p) - np.power(gamma, p)
    a /= total
    return a


def _conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Return the (2·n‑1) × n sparse matrix A such that A @ g = f * g
    (full convolution, zero‑padded outside the support).
    """
    n = f.size
    # each diagonal k (offset k) contains the constant f[k]
    diags_data = [np.full(n, fk, dtype=float) for fk in f]
    offsets = np.arange(n, dtype=int)
    A = sp.diags(diags_data, offsets, shape=(2 * n - 1, n), format='csr')
    return A


def _lp_step(f: np.ndarray, t: float) -> np.ndarray | None:
    """
    Perform one LP‑based improvement step.
    Returns the mixed vector (1‑t)·f + t·g (ℓ₁‑normalised) or None on failure.
    """
    n = f.size
    # build convolution matrix for the current f
    A = _conv_matrix(f)                     # shape (2n‑1) × n
    # column for the variable M (with coefficient –1)
    Mcol = -sp.csr_matrix(np.ones((2 * n - 1, 1), dtype=float))
    A_ub = sp.hstack([A, Mcol], format='csr')
    b_ub = np.zeros(2 * n - 1, dtype=float)

    # equality: Σ g = 1
    A_eq = np.zeros((1, n + 1), dtype=float)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=float)

    # objective: minimise M
    c = np.zeros(n + 1, dtype=float)
    c[-1] = 1.0

    bounds = [(0.0, None)] * (n + 1)   # g_j ≥ 0, M ≥ 0

    res = linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        A_eq=A_eq,
        b_eq=b_eq,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "quiet": True},
    )

    if not res.success:
        return None

    g = np.maximum(res.x[:n], 0.0)
    # mix and renormalise to maintain ℓ₁ = 1
    new_f = (1.0 - t) * f + t * g
    new_f /= new_f.sum()
    return new_f


def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list[float]:
    """
    Search for a step‑function with evaluation < 1.503.
    The routine performs:
      1. a modest grid search over (p,γ);
      2. builds a longer candidate from the best parameters;
      3. iteratively improves it with a tiny LP‑based smoothing step;
      4. finishes with a light stochastic refinement.
    """
    random.seed(seed)
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ----------------------------------------------------------------------
    # 1. Coarse grid search (fast, short vectors)
    # ----------------------------------------------------------------------
    n_small = 1500
    best_score = float("inf")
    best_params = (0.4, 10.0)   # (p, gamma) placeholder

    p_vals = np.arange(0.30, 0.55 + 1e-12, 0.005)   # 0.30 … 0.55
    gamma_vals = np.arange(0, 201, 1)               # 0 … 200

    print("[search] starting grid exploration")
    for p in p_vals:
        for gamma in gamma_vals:
            if time.time() > deadline:
                break
            seq = _generate_sequence(n_small, float(p), float(gamma))
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_params = (float(p), float(gamma))
                print(f"[grid] new best: {score:.6f} (p={p:.3f}, γ={gamma})")
        if time.time() > deadline:
            break

    # ----------------------------------------------------------------------
    # 2. Build a longer candidate with the best parameters
    # ----------------------------------------------------------------------
    n_big = 5000
    p_best, gamma_best = best_params
    seq = _generate_sequence(n_big, p_best, gamma_best)
    # make sure the vector sums to exactly 1 (tiny rounding errors may appear)
    seq /= seq.sum()
    best_score = evaluate_sequence(seq.tolist())
    print(f"[init] big sequence (n={n_big}) p={p_best:.3f} γ={gamma_best:.1f} → {best_score:.6f}")

    # ----------------------------------------------------------------------
    # 3. LP‑based smoothing iterations
    # ----------------------------------------------------------------------
    t_mix = 0.5          # start with a fairly aggressive mix
    iteration = 0
    while time.time() < deadline and iteration < 20:
        iteration += 1
        new_seq = _lp_step(seq, t_mix)
        if new_seq is None:
            print("[lp] LP step failed – terminating LP loop")
            break
        new_score = evaluate_sequence(new_seq.tolist())
        print(f"[lp] iter {iteration:02d} → score {new_score:.6f} (t={t_mix:.3f})")
        if new_score + 1e-12 < best_score:
            best_score = new_score
            seq = new_seq
            t_mix = min(0.9, t_mix * 1.2)   # slowly increase mixing if we’re improving
            print(f"[lp]  improvement – keep new sequence, t now {t_mix:.3f}")
        else:
            # no improvement – shrink the mixing factor and try again
            t_mix *= 0.5
            print(f"[lp]  no improvement – reducing t to {t_mix:.3f}")
            if t_mix < 1e-4:
                print("[lp]  t too small – exiting LP loop")
                break

    # ----------------------------------------------------------------------
    # 4. Very light stochastic refinement (mass‑transfer moves)
    # ----------------------------------------------------------------------
    seq = seq.copy()
    n = len(seq)
    while time.time() < deadline:
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 1e-4   # tiny signed transfer
        if seq[i] + delta < 0 or seq[j] - delta < 0:
            continue
        cand = seq.copy()
        cand[i] += delta
        cand[j] -= delta
        # renormalise (the transfer keeps the ℓ₁‑sum unchanged)
        cand /= cand.sum()
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < best_score:
            best_score = cand_score
            seq = cand
            print(f"[refine] improved score → {best_score:.6f}")

    print(f"[final] best score achieved: {best_score:.6f} (length {len(seq)})")
    return seq.tolist()
