```
def compute_gradient(a, gamma, nfft_conv, nfft_corr):
    # compute conv for softmax
    af = np.fft.rfft(a, n=nfft_conv); conv = np.fft.irfft(af * af, n=nfft_conv)[:2*n-1]; maxc = conv.max()
    w = np.exp(gamma * (conv - maxc))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0/w.size)
        w_sum = 1.0
    w /= w_sum
    # correlation
    wf = np.fft.rfft(w, n=nfft_corr)
    af_rev = np.fft.rfft(a[::-1], n=nfft_corr)
    corr_full = np.fft.irfft(wf * af_rev, n=nfft_corr)
    grad = 2.0 * corr_full[n-1 : n-1 + n]
    return grad, maxc
```

```python
"""
Propose a sequence that (hopefully) beats the previous 1.575 bound.

Algorithm
---------
* Use a fast FFT‑based evaluation of the objective.
* Initialise the search from several simple analytic families
  (power law, cosine‑shaped) and from the best known construction
  (`height_sequence_1` when available).
* For each initialisation run a short Adam optimiser that
  minimises a smooth approximation of the maximum of the auto‑
  convolution.  The smooth maximum is built with a soft‑max
  (temperature `γ`).  `γ` is increased exponentially so that the
  optimiser gradually focuses on the true peak.
* The Adam optimiser keeps the sum of the coefficients fixed
  (`∑a = √(2n)`) by a simple renormalisation after every step and
  projects onto the non‑negative orthant.
* The best sequence seen over the whole time budget is returned.
"""

import numpy as np
import time
import math

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _fast_score(a: np.ndarray) -> float:
    """Return the objective 2 n max(conv(a,a)) / (sum(a)^2) using FFT."""
    n = a.shape[0]
    nfft = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    return float(2.0 * n * max_c / (s * s))

def _softmax_gradient(a: np.ndarray, gamma: float,
                      nfft_conv: int, nfft_corr: int):
    """
    Compute the gradient of the smooth maximum of a*a w.r.t. a.
    Returns (grad, max_conv_value).
    """
    n = a.shape[0]
    # convolution a*a
    af = np.fft.rfft(a, n=nfft_conv)
    conv = np.fft.irfft(af * af, n=nfft_conv)[: 2 * n - 1]
    max_c = conv.max()
    # soft‑max weights over the convolution entries
    w = np.exp(gamma * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
        w_sum = 1.0
    w /= w_sum
    # correlation w ⊛ a_rev
    wf = np.fft.rfft(w, n=nfft_corr)
    af_rev = np.fft.rfft(a[::-1], n=nfft_corr)
    corr_full = np.fft.irfft(wf * af_rev, n=nfft_corr)
    grad = 2.0 * corr_full[n - 1 : n - 1 + n]   # length n
    return grad, max_c

def _normalise(a: np.ndarray, target_sum: float) -> np.ndarray:
    """Project onto non‑negative orthant and renormalise to target_sum."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        # fallback to uniform vector
        a = np.full_like(a, target_sum / a.shape[0])
    else:
        a *= target_sum / s
    return a

        use_existing = False

    nfft_conv = _next_pow2(2 * n - 1)
    nfft_corr = _next_pow2(3 * n - 2)

    # Helper to build a random initial shape
    # ------------------------------------------------------------------
    def random_initial():
        """Return a random admissible initial vector."""
        # Choose a family at random

import numpy as np
import time
import math

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _fast_score(a: np.ndarray) -> float:
    """Fast evaluation of 2·n·max(conv(a,a)) / (sum(a)**2) using FFT."""
    n = a.shape[0]
    nfft = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    return float(2.0 * n * max_c / (s * s))

def _softmax_gradient(a: np.ndarray, gamma: float,
                      nfft_conv: int, nfft_corr: int):
    """
    Compute the gradient of the smooth maximum of a*a with respect to a.
    Returns (gradient, max_conv_value).
    """

    max_c = conv.max()

    wf = np.fft.rfft(w, n=nfft_corr)

    corr_full = np.fft.irfft(wf * af_rev, n=nfft_corr)
```
