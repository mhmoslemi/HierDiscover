import numpy as np, time, math

def simplex_projection(v):
    v = np.asarray(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v)+1)
    # find rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback
        return np.maximum(v, 0)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def generate_sequence(p, sigma, lam, max_len=2000, eps=1e-8):
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p) * np.exp(-lam * i)
    mask = a > eps
    return a[mask]

# Helper utilities

def _eval(seq):
    """Safely call the external `evaluate_sequence`."""
    try:
        v = evaluate_sequence(seq)          # provided by the platform
        if np.isfinite(v):
            return float(v)
    except Exception:
        pass
    return np.inf


def _simplex_projection(v):
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D vector")
    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    # Find the pivot
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _generate_sequence(p, sigma, lam, max_len=3000, eps=1e-12):
    """
    Produce a truncated sequence a_i = (i+σ)^(-p) · exp(-λ·i).
    Truncation stops when values drop below `eps` or when `max_len` is
    reached.  The returned array is *not* normalised – the evaluation
    function is scale‑invariant.
    """
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p) * np.exp(-lam * i)
    mask = a > eps
    return a[mask]


def _subgradient_max_conv(x):
    """
    Sub‑gradient of the ℓ∞‑norm of the self‑convolution at the current
    vector `x`.  The gradient for a single maximising index k is
        g_i = x_{k-i}  (zero when the index is out of bounds).
    When several indices share the maximal value, the sub‑gradient is the
    sum of the individual contributions.
    """
    conv = np.convolve(x, x)
    max_val = conv.max()
    # Numerical tolerance for “active” constraints
    tol = max_val * 1e-12 + 1e-15
    k_idxs = np.where(conv >= max_val - tol)[0]

    n = x.shape[0]
    g = np.zeros_like(x)
    for k in k_idxs:
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        idx = np.arange(i_min, i_max + 1)
        jdx = k - idx
        # derivative of (x * x)_k w.r.t. x_i is 2·x_{k-i}; the factor 2
        # will be absorbed into the step‑size schedule.
        g[idx] += x[jdx]
    return g


# ----------------------------------------------------------------------
# Stage 1 – random parametric sweep
# ----------------------------------------------------------------------


def _parametric_sweep(time_budget):
    """Explore random (p,σ,λ) triples and keep the best sequence."""
    deadline = time.time() + time_budget
    best_seq = None
    best_score = np.inf
    # Fixed limits for the random search
    max_len = 3000
    eps = 1e-12
    trials = 0

    while time.time() < deadline and trials < 5000:
        trials += 1
        # Sample parameters
        p = np.random.uniform(0.30, 0.70)                     # exponent
        sigma = 10 ** np.random.uniform(-4, 2)               # shift
        lam = np.random.uniform(0.0, 0.002)                  # exponential tail
        seq = _generate_sequence(p, sigma, lam, max_len, eps)

        if seq.size < 5:          # absurdly short – ignore
            continue

        score = _eval(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.tolist()
            print(f"[sweep] trial {trials:04d} – p={p:.4f} σ={sigma:.4e} λ={lam:.6f} "
                  f"len={len(seq)} score={score:.6f}")
            if best_score < 1.503:        # early exit if target already met
                break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Stage 2 – mirror descent with momentum & occasional perturbations
# ----------------------------------------------------------------------


def _mirror_descent_momentum(x, best_score, deadline):
    """
    Exponentiated‑gradient descent on the ℓ∞‑norm of x∗x.
    `x` must already lie on the simplex.  Returns the best sequence found.
    """
    n = x.shape[0]
    momentum = np.zeros_like(x)
    beta = 0.9                     # momentum decay
    iteration = 0
    best_seq = x.copy()
    cur_best = best_score

    while time.time() < deadline:
        iteration += 1
        eta = 0.50 / np.sqrt(iteration)       # step‑size schedule

        # sub‑gradient at the current maximiser(s)
        g = _subgradient_max_conv(x)

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g

        # exponentiated (multiplicative) update
        y = x * np.exp(-eta * momentum)

        # projection onto the simplex
        x = _simplex_projection(y)

        # ------------------- housekeeping -------------------
        # prune harmless trailing near‑zero entries (reduces n → faster conv)
        if x[-1] < 1e-12 and x.shape[0] > 10:
            cut = np.where(x > 1e-12)[0]
            if cut.size:
                last = cut[-1] + 1
                x = x[:last]
                x = x / x.sum()          # re‑normalise after truncation

        # occasional random jitter to escape plateaus
        if np.random.rand() < 0.02:
            i = np.random.randint(len(x))
            delta = (np.random.rand() - 0.5) * 0.02 * x[i]
            x[i] = max(x[i] + delta, 0.0)
            x = x / x.sum()

        # evaluate the current candidate
        cur_score = _eval(x.tolist())
        if cur_score < cur_best:
            cur_best = cur_score
            best_seq = x.copy()
            print(f"[md] it {iteration:06d} – new best {cur_best:.6f} (len={len(x)})")
            if cur_best < 1.503:          # target reached – stop early
                break

        # if no improvement for a long while, inject a modest random
        # perturbation around the current best
        if iteration % 5000 == 0 and cur_best > 1.5:
            pert = best_seq.copy()
            for _ in range(3):
                i = np.random.randint(len(pert))
                j = np.random.randint(len(pert))
                eps = 0.02 * pert[i]
                delta = (np.random.rand() - 0.5) * eps
                pert[i] = max(pert[i] + delta, 0.0)
                pert[j] = max(pert[j] - delta, 0.0)
            pert = pert / pert.sum()
            pert_score = _eval(pert.tolist())
            if pert_score < cur_best:
                x = pert
                cur_best = pert_score
                best_seq = pert.copy()
                print(f"[md] it {iteration:06d} – perturb improved to {cur_best:.6f}")

    return best_seq.tolist(), cur_best


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0):
    """
    Search for a non‑negative step‑function (list of heights) that minimizes
    `evaluate_sequence`.  The routine respects the 1000 s time limit and
    returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – cheap random parametric sweep
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    best_seq, best_score = _parametric_sweep(remaining)

    # If the sweep failed to produce a candidate, fall back to a uniform vector
    if best_seq is None:
        n_fallback = 500
        best_seq = np.ones(n_fallback, dtype=np.float64)
        best_seq = best_seq / best_seq.sum()
        best_score = _eval(best_seq.tolist())
        print("[fallback] uniform sequence – score =", best_score)

    # ------------------------------------------------------------------
    # Phase 2 – mirror descent with momentum
    # ------------------------------------------------------------------
    x = np.asarray(best_seq, dtype=np.float64)
    if x.sum() == 0.0:
        x = np.abs(np.random.randn(len(x)))
    x = x / x.sum()                # ensure we start on the simplex

    remaining = deadline - time.time()
    final_seq, final_score = _mirror_descent_momentum(x, best_score, deadline)

    print("=== final best ===")
    print("score :", final_score)
    print("len   :", len(final_seq))
    return final_seq
