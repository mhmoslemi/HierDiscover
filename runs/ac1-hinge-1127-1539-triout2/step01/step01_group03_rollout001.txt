"""
Exponential decay construction.

The continuous kernel 1/√x yields a constant auto‑convolution and a score
≈π/2≈1.57.  A much simpler discrete family – a geometric (exponential)
sequence a_i = exp(-ε·i) – can make the maximal convolution stay close to
1 while the ℓ₁‑norm grows like n/ε.  Choosing ε = log(n)/n (i.e. ε·n = log n)
balances these effects and gives

    score ≈ 2·n·1 / ( (n/log n)² ) = 2·(log n)² / n,

which for n≈5 000 already yields ≈0.03, far below the target 1.503.
The function searches a few lengths and a small multiplicative window around
the optimal ε, picks the best candidate and returns it as a plain list of
floats.
"""

import time
import math
import numpy as np

def _objective(seq: np.ndarray) -> float:
    """Fast evaluation of the objective (same formula as evaluate_sequence)."""
    n = len(seq)
    s = np.sum(seq)
    if s < 0.01:
        return float("inf")
    # FFT‑based full convolution (real‑to‑real)
    m = 1 << ((2 * n - 1).bit_length())      # next power of two
    fft = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft * fft, n=m)[: 2 * n - 1]
    return float(2.0 * n * np.max(conv) / (s * s))


def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Generate a low‑score step‑function.

    The method builds exponential sequences a_i = exp(-ε·i) with
    ε≈log(n)/n.  Several lengths and a tiny ε‑scale grid are tried;
    the best sequence (according to the fast _objective) is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1. Lengths to try – larger n gives a lower theoretical bound,
    #    but we keep them modest to stay well within the time budget.
    # ------------------------------------------------------------------
    lengths = [2000, 5000, 10000, 20000]

    # ------------------------------------------------------------------
    # 2. Multiplicative factors around the “optimal” ε = log(n)/n.
    # ------------------------------------------------------------------
    factors = [0.8, 0.9, 1.0, 1.1, 1.2]

    best_seq = None
    best_score = float("inf")
    best_info = None

    for n in lengths:
        # pre‑compute the array of indices once
        idx = np.arange(n, dtype=np.float64)

        # optimal ε for this n (theoretical)
        eps_opt = math.log(n) / n

        for f in factors:
            eps = eps_opt * f
            seq = np.exp(-eps * idx)          # a_i = exp(-ε·i)

            # Fast score – the same quantity that evaluate_sequence will use.
            score = _objective(seq)

            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_info = (n, eps, f)
                print(f"[candidate] n={n}, factor={f:.2f}, ε={eps:.6g} → "
                      f"score={score:.6f}")

            # Respect the time budget
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 3. (Optional) very cheap local refinement: a single Newton‑like
    #    step on ε for the currently best length.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n, eps, _ = best_info
        # take a small step on ε using a finite‑difference estimate of the
        # derivative of the objective.
        step = eps * 0.05
        for _ in range(8):
            eps_up = eps + step
            eps_dn = max(eps - step, 1e-12)

            seq_up = np.exp(-eps_up * np.arange(n, dtype=np.float64))
            seq_dn = np.exp(-eps_dn * np.arange(n, dtype=np.float64))

            s_up = _objective(seq_up)
            s_dn = _objective(seq_dn)

            # gradient approx
            grad = (s_up - s_dn) / (2 * step)

            # move opposite to gradient
            eps_new = eps - 0.5 * grad * step
            eps_new = max(eps_new, 1e-12)

            seq_new = np.exp(-eps_new * np.arange(n, dtype=np.float64))
            s_new = _objective(seq_new)

            if s_new < best_score:
                best_score = s_new
                best_seq = seq_new.copy()
                eps = eps_new
                print(f"[refine] n={n}, ε={eps:.6g} → score={s_new:.6f}")

            # shrink step size for the next iteration
            step *= 0.7

            if time.time() > deadline:
                break

    # ------------------------------------------------------------------
    # 4. Return the best sequence found as a plain Python list.
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback – uniform sequence (score 2).  Should never happen.
        best_seq = np.ones(5000, dtype=np.float64)
        best_score = _objective(best_seq)
        print("[fallback] using uniform sequence, score=2.0")

    print(f"[final] best score = {best_score:.6f} (length {len(best_seq)})")
    # Convert to plain list of Python floats for the evaluator.
    return [float(x) for x in best_seq]
