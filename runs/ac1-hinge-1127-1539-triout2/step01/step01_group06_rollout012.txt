"""
Simulated‑annealing / greedy hill‑climbing for the step‑function bound.

The objective after the ℓ1‑normalisation (∑a = √(2·n)) equals the maximum
entry of the auto‑convolution a*a.  The search therefore tries to minimise
that maximum by repeatedly moving a small amount of mass from the index that
contributes most to the current peak towards an index that carries little
mass.  The move size shrinks as time runs out; occasional uphill moves are
accepted according to a linear cooling schedule (simulated annealing).  Several
vector lengths and power‑law initialisations are tried; the best sequence found
within the time budget is returned.

All helper functions are top‑level (no closures, no lambdas) and the code runs
with a single CPU.  The routine respects the 1000 s budget and returns a plain
Python list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _conv_full(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using an FFT of length a power of two.
    Returns the (2·n‑1)‑long array of convolution values.
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the auto‑convolution a*a."""
    return float(_conv_full(a).max())


def _score(a: np.ndarray) -> float:
    """
    Evaluation function used by the competition.  The vector a must be
    non‑negative and its ℓ1 norm is expected to be √(2·n); the score then
    equals the maximum convolution entry.
    """
    n = a.size
    s = a.sum()
    if s < 0.01:                     # safety – should never happen
        return np.inf
    return float(2.0 * n * _max_convolution(a) / (s * s))


def _initial_sequence(p: float, n: int) -> np.ndarray:
    """
    Deterministic power‑law start: a_i ∝ (i+1)^{-p}, renormalised to
    ∑a = √(2·n).  This guarantees a valid start for any exponent p>0.
    """
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= np.sqrt(2.0 * n) / a.sum()
    return a


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------


def _anneal_one_start(a: np.ndarray,
                      deadline: float,
                      start_time: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing / greedy hill‑climbing starting from array ``a``.
    The total ℓ1‑norm is kept constant; moves transfer a fraction of the
    mass from the index that most contributes to the current peak of the
    convolution to the smallest entry of the vector.  The routine returns the
    best sequence (and its score) discovered before ``deadline``.
    """
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # current state
    cur_val = _max_convolution(a)
    best_a = a.copy()
    best_val = cur_val

    # cooling schedule parameters
    init_temp = cur_val * 0.2                # temperature at t=0
    max_frac = 0.20                          # max fraction of source mass moved
    min_frac = 0.003                         # min fraction near the end

    while time.time() < deadline:
        # ------------------------------------------------------------------
        # 1. analyse the current peak of the convolution
        # ------------------------------------------------------------------
        conv = _conv_full(a)
        max_val = float(conv.max())
        k_max = int(conv.argmax())

        # indices that participate in the maximal sum
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)
        # contribution of each i to the peak: a_i * a_{k_max-i}
        contrib = a[idx] * a[k_max - idx]

        # source = index with the largest contribution to the peak
        src = idx[int(np.argmax(contrib))]
        if a[src] <= 1e-15:                      # nothing to move
            continue

        # target = index with the smallest current value (avoid src)
        tgt = int(np.argmin(a))
        if tgt == src:
            # pick the second smallest if the smallest coincides with src
            sorted_idx = np.argsort(a)
            for cand in sorted_idx:
                if cand != src:
                    tgt = cand
                    break

        # ------------------------------------------------------------------
        # 2. propose a mass‑transfer move
        # ------------------------------------------------------------------
        # move size shrinks linearly with elapsed time
        elapsed = time.time() - start_time
        progress = elapsed / (deadline - start_time)          # in [0,1]
        cur_frac = max_frac * (1.0 - progress) + min_frac * progress
        delta = a[src] * cur_frac
        if delta <= 0.0:
            continue

        a[src] -= delta
        a[tgt] += delta

        # re‑normalise tiny drift (numerical safety)
        s = a.sum()
        if abs(s - target_sum) > 1e-8:
            a *= target_sum / s

        # ------------------------------------------------------------------
        # 3. accept / reject
        # ------------------------------------------------------------------
        new_val = _max_convolution(a)

        if new_val < cur_val:                 # strict improvement
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}]  new best value = {best_val:.6f}")
        else:
            # simulated annealing acceptance
            temp = init_temp * (1.0 - progress)
            accept_prob = np.exp(-(new_val - cur_val) / (temp + 1e-12))
            if np.random.rand() < accept_prob:
                cur_val = new_val
            else:
                # revert the move
                a[tgt] -= delta
                a[src] += delta
                # (sum is unchanged, no need to renormalise)

    return best_a, best_val


# ----------------------------------------------------------------------
# Public entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function sequence that minimises ``evaluate_sequence``.
    The routine uses several random restarts, each driven by a greedy
    simulated‑annealing improvement phase.  The best sequence encountered before
    the time budget expires is returned as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # keep the overall best candidate
    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 0. optional warm start from a previously supplied solution
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0:
                cand = np.maximum(0.0, cand)               # enforce non‑negativity
                # normalise to the required ℓ1‑norm
                cand *= np.sqrt(2.0 * cand.size) / cand.sum()
                val = _score(cand)
                if val < best_score:
                    best_score = val
                    best_seq = cand.copy()
                    print(f"Warm‑start from height_sequence_1 → score {val:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1. configure the search space (different lengths & power‑law exponents)
    # ------------------------------------------------------------------
    lengths = [800, 1200, 1800, 2500]                 # all comfortably below 10 000
    exponents = [0.30, 0.35, 0.40, 0.45, 0.50,
                 0.55, 0.60, 0.65, 0.70]            # wide sweep around 0.5

    # simple time budgeting: allocate a few seconds per (n,p) pair,
    # but always respect the global deadline.
    for n in lengths:
        for p in exponents:
            if time.time() >= deadline:
                break

            # ------------------------------------------------------------------
            # 2. initialise a power‑law vector and evaluate it
            # ------------------------------------------------------------------
            a = _initial_sequence(p, n)
            cur_score = _score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"Init (n={n}, p={p:.2f}) → score {cur_score:.6f}")

            # ------------------------------------------------------------------
            # 3. run a focused annealing phase for a short but significant chunk
            # ------------------------------------------------------------------
            # we reserve at most 12 s for each start, but never exceed the global budget
            remaining = deadline - time.time()
            chunk = min(12.0, remaining * 0.4)          # keep a pool for later starts
            sub_deadline = time.time() + chunk

            a_opt, opt_score = _anneal_one_start(a, sub_deadline, start_time)

            if opt_score < best_score:
                best_score = opt_score
                best_seq = a_opt.copy()
                print(f"After annealing (n={n}, p={p:.2f}) best = {opt_score:.6f}")

    # ------------------------------------------------------------------
    # 4. fallback – should never trigger
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.ones(n, dtype=np.float64)
        best_seq *= np.sqrt(2.0 * n) / n

    # trim trailing (near‑zero) entries – they do not affect the score
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # convert to plain list as required by the outer harness
    return best_seq.tolist()
