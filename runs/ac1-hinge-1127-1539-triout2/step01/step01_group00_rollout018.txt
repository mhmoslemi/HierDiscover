def _gradient_update(a, k, eta):
    n = a.shape[0]
    idx = np.arange(n)
    partner = k - idx
    valid = (partner>=0)&(partner<n)
    g = np.zeros_like(a)
    g[valid] = 2*a[partner[valid]]
    if k%2==0:
        d=k//2
        if 0 <= d < n:
            g[d] = a[d]   # set diag weight single
    a_new = a * np.exp(-eta * g)
    # renorm
    S = np.sqrt(2.0 * n)
    a_new *= S / a_new.sum()
    return a_new

def _initial_sequence_n(n, rng):
    # try use existing if available with some probability
    if 'height_sequence_1' in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=float)
            if base.size >= n:
                a = base[:n].copy()
            else:
                a = np.pad(base, (0, n - base.size), constant_values=0.0)
            # Normalize
            a = a / a.sum() * np.sqrt(2.0 * n)
            # optionally jitter a little to avoid being stuck
            if rng.random() < 0.5:
                # add small random noise
                noise = rng.random(n) * a.mean() * 0.01
                a = np.clip(a + noise, 0, None)
                a = a / a.sum() * np.sqrt(2.0 * n)
            return a
        except Exception:
            pass
    # else random
    alpha = rng.choice([0.2, 0.5, 1.0, 2.0])
    a = rng.gamma(shape=alpha, scale=1.0, size=n)
    a = a / a.sum() * np.sqrt(2.0 * n)
    return a

def _optimize_length(n, rng, deadline, best_seq, best_score):
    # run multiple restarts
    # step size schedule
    init_eta = 1.0
    decay = 0.9995
    jitter_prob = 0.02
    trial = 0
    while time.time() < deadline:
        # create initial a:
        a = _initial_sequence_n(n, rng)
        # compute current max
        _, cur_max, _ = _conv_max(a)
        if cur_max < best_score:
            best_score = cur_max
            best_seq = a.copy()
            print(f"[init] new best from init cur_max={cur_max:.6f}")

        eta = init_eta
        # inner loop
        iters_without_improve = 0
        while time.time() < deadline:
            conv, cur_max, k = _conv_max(a)
            if cur_max < best_score:
                best_score = cur_max
                best_seq = a.copy()
                print(f"[improve] n={n} time={time.time():.1f}s score={best_score:.6f}")
                iters_without_improve = 0
            else:
                iters_without_improve += 1
                # If many iterations without improvement, break
                if iters_without_improve >= 5000:
                    break

            # gradient update
            a = _gradient_update(a, k, eta)

            # step size decay
            eta = eta * decay

            # jitter
            if rng.random() < jitter_prob:
                i, j = rng.integers(0, n, size=2)
                if i != j:
                    delta = a[i] * 0.02 * rng.random()
                    a[i] -= delta
                    a[j] += delta
                    a = np.clip(a, 0.0, None)
                    a *= np.sqrt(2.0 * n) / a.sum()
        # after inner termination, continue next restart
        trial += 1
        # maybe reduce jitter or change start
    return best_seq, best_score

"""
Large‑scale projected multiplicative gradient descent.

The score returned by ``evaluate_sequence`` equals the maximal entry of the
self‑convolution when the L¹‑norm is fixed to √(2 n).  Hence we can work with
the internal fast FFT based max‑convolution as the true objective and avoid the
expensive ``evaluate_sequence`` calls.

The algorithm:
* Choose a relatively large length (up to a few ·10⁴).  The theoretical lower
  bound (the average convolution value) tends to 1 as n grows, so larger n
  gives more room for improvement.
* Initialise the sequence with a random Dirichlet‑style draw (Gamma
  distribution).  If the global variable ``height_sequence_1`` exists we also
  try a copy of it.
* Repeatedly:
    – compute the full linear convolution by FFT,
    – locate the index k where the maximum occurs,
    – form the sub‑gradient of that entry (g_j = 2·a_{k−j}, with the diagonal
      term halved),
    – take a multiplicative update a ← a·exp(−η·g) (keeps a≥0),
    – renormalise to keep Σa = √(2 n).
  The step size η starts relatively large and decays slowly.
* Occasionally inject a tiny random “jitter’’ (move a small amount of mass
  between two positions) to escape plateaus.
* Whenever the current maximal convolution value improves on the best
  seen so far we store the sequence.  No call to ``evaluate_sequence`` is
  required during the search – the internal max already equals the external
  score (the protection steps in ``evaluate_sequence`` do not affect our
  non‑negative, modest‑sized vectors).
* The routine runs for the allotted time (default 1000 s) and finally
  returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def _conv_max(a: np.ndarray) -> tuple[np.ndarray, float, int]:
    """
    Compute the linear convolution a*a using an FFT, returning the full
    convolution array, its maximum value and the index of that maximum.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    # next power of two for efficient FFT
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    max_idx = int(np.argmax(conv))
    return conv, float(conv[max_idx]), max_idx


def _gradient_update(a: np.ndarray, k_max: int, eta: float) -> np.ndarray:
    """
    Perform one multiplicative gradient step aimed at reducing the current
    maximal self‑convolution entry (located at index ``k_max``).
    """
    n = a.shape[0]
    idx = np.arange(n)
    partner = k_max - idx
    valid = (partner >= 0) & (partner < n)

    # sub‑gradient of the entry at k_max
    g = np.zeros_like(a)
    g[valid] = 2.0 * a[partner[valid]]

    # diagonal case (i == j) when k_max is even
    if (k_max & 1) == 0:            # k_max % 2 == 0
        d = k_max // 2
        if 0 <= d < n:
            g[d] = a[d]              # only a[d], not doubled

    # multiplicative update – guarantees non‑negativity
    a_new = a * np.exp(-eta * g)

    # renormalise to keep the L1 norm fixed (√(2·n))
    S = np.sqrt(2.0 * n)
    total = a_new.sum()
    if total == 0.0:                # safety, should not happen
        a_new = np.full(n, S / n)
    else:
        a_new *= (S / total)

    # avoid exact zeros that could freeze coordinates
    eps = 1e-15
    a_new = np.maximum(a_new, eps)
    return a_new


def _initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce an initial non‑negative sequence of length n, normalised so that
    Σa = √(2·n).  With 50 % chance we start from the optional global variable
    ``height_sequence_1``; otherwise we draw from a Gamma distribution with a
    randomly chosen shape (0.2, 0.5, 1.0 or 2.0) to obtain a variety of
    sparsities.
    """
    S = np.sqrt(2.0 * n)

    # try to reuse the known construction
    if 'height_sequence_1' in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=float)
            if base.size >= n:
                a = base[:n].copy()
            else:
                a = np.pad(base, (0, n - base.size), constant_values=0.0)
            # occasional random perturbation of the reused vector
            if rng.random() < 0.5:
                noise = rng.random(n) * a.mean() * 0.01
                a = np.clip(a + noise, 0.0, None)
        except Exception:
            a = None
    else:
        a = None

    if a is None:
        # Gamma draw – shape parameter chosen uniformly from a short list
        shape = rng.choice([0.2, 0.5, 1.0, 2.0])
        a = rng.gamma(shape=shape, scale=1.0, size=n)

    # normalise to the required L1 norm
    a = a / a.sum() * S
    # ensure no exact zero (helps the multiplicative update)
    a = np.maximum(a, 1e-15)
    return a


def _optimize_length(
    n: int,
    rng: np.random.Generator,
    deadline: float,
    best_seq: np.ndarray | None,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Run the projected multiplicative gradient descent for a given length ``n``.
    Multiple restarts are performed until the time limit is reached.
    The function returns the best sequence discovered and its maximal
    convolution value.
    """
    S = np.sqrt(2.0 * n)
    # schedule parameters – these work well for a wide range of n
    init_eta = 5.0          # initial step size (larger for longer vectors)
    decay = 0.9995          # geometric decay each iteration
    jitter_prob = 0.02      # probability of a random mass‑shift per iteration
    max_no_improve = 5000   # abort a restart after this many stagnant steps

    while time.time() < deadline:
        # ----------- initialise a new trial ----------
        a = _initial_sequence(n, rng)

        # initial best update (no external evaluation needed)
        _, cur_max, _ = _conv_max(a)
        if cur_max < best_score:
            best_score = cur_max
            best_seq = a.copy()
            print(f"[init] n={n} score={best_score:.6f}")

        eta = init_eta
        stagnant = 0

        # -------------- main optimisation loop --------------
        while time.time() < deadline:
            conv, cur_max, k_max = _conv_max(a)

            # store improvement
            if cur_max < best_score:
                best_score = cur_max
                best_seq = a.copy()
                stagnant = 0
                print(f"[improve] n={n} time={time.time():.1f}s score={best_score:.6f}")
            else:
                stagnant += 1
                if stagnant >= max_no_improve:
                    break

            # gradient step
            a = _gradient_update(a, k_max, eta)

            # step‑size decay
            eta *= decay

            # occasional jitter to shake out of flat regions
            if rng.random() < jitter_prob:
                i, j = rng.integers(0, n, size=2)
                if i != j:
                    # transfer a small fraction of mass from i to j
                    delta = a[i] * 0.02 * rng.random()
                    a[i] -= delta
                    a[j] += delta
                    a = np.clip(a, 0.0, None)
                    a *= S / a.sum()
                    a = np.maximum(a, 1e-15)

        # after a restart we try another one (different random seed)
        # unless we are already out of time.
    return best_seq, best_score


# ----------------------------------------------------------------------
# Required entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) yields a lower
    ``evaluate_sequence`` score than the current best bound of 2.0.
    The routine runs for at most ``budget_s`` seconds, exploiting a fast
    projected multiplicative gradient descent on long sequences.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    best_seq = None
    best_score = np.inf

    # try a few increasing lengths – larger n gives a lower theoretical floor.
    # The list is ordered from largest to smallest; the loop stops when the
    # time budget is exhausted.
    lengths = [40000, 30000, 20000, 15000, 12000, 8000, 5641]

    for n in lengths:
        if time.time() >= deadline:
            break
        best_seq, best_score = _optimize_length(
            n, rng, deadline, best_seq, best_score
        )
        # early exit if we have already passed the target bound
        if best_score < 1.503:
            print(f"[target reached] n={n} score={best_score:.6f}")
            break

    # Fallback – should never happen, but guarantees a return value.
    if best_seq is None:
        n = 1
        best_seq = np.array([np.sqrt(2.0 * n)], dtype=float)

    print(f"[final] best score = {best_score:.6f} length = {len(best_seq)}")
    return best_seq.tolist()
