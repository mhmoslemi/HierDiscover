def solve_lp_for_g(f, M):
    n = len(f)
    rows = 2*n - 1
    # Build A_ub matrix
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = f
    b = np.full(rows, M, dtype=np.float64)
    # objective maximize sum(g) => minimize -sum(g)
    c = -np.ones(n, dtype=np.float64)
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n,
                                 method='highs', options={'presolve': True})
    if not res.success:
        return None
    return res.x
```

"""
Propose a step‑function (non‑negative sequence) that minimizes
`evaluate_sequence`.  

The algorithm works in three stages:

1. **Parametric sweep** – try two simple families of sequences:
   * a symmetric “Beta‑shape’’ a_i ∝ (i+σ)^(-α)*(n‑1‑i+σ)^(-α)
   * a one‑sided power‑law a_i ∝ (i+σ)^(-p) (truncated when tiny).
   The best candidate of modest length (≈ 800) is kept.

2. **Linear‑program (Frank–Wolfe) refinement** – for the current
   distribution `f` (∑f = 1) we solve  
   maximise   ∑g_j               subject to   g ≥ 0,
               (f * g)_k ≤ M  ∀k,      where M = max(f*f).  
   The solution `g` is rescaled to a probability vector and mixed into
   `f` with a decreasing step size `t`.  Repeating this a few dozen
   times drives down the maximal auto‑convolution value.

3. **Mirror‑descent polishing** – the refined distribution is
   up‑sampled to a longer vector (≈ 2500–3000 elements) and a
   multiplicative mirror‑descent update is applied many thousands of
   times (exponential update + projection onto the simplex).  Random
   jitter is added occasionally to escape plateaus.

All stages respect the global 1000 s budget.  The best sequence seen
throughout the run is returned.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper: generate a truncated power‑law sequence
def _generate_powerlaw(p: float, sigma: float,
                       max_len: int = 5000, eps: float = 1e-8) -> np.ndarray:
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p)
    a = a[a > eps]                         # truncate tiny tail
    return a

# Helper: symmetric “Beta‑shape’’ sequence
def _beta_shape(alpha: float, sigma: float, n: int) -> np.ndarray:
    # symmetric product (i+σ)^(-α)*(n-1-i+σ)^(-α)
    i = np.arange(n, dtype=np.float64)
    left = i + sigma
    right = (n - 1 - i) + sigma
    a = (left ** (-alpha)) * (right ** (-alpha))
    # avoid infinities if sigma is too small
    a = np.where(np.isfinite(a), a, 0.0)
    return a

# ------------------------------------------------------------
# Mirror‑descent step (multiplicative update + simplex projection)
def _mirror_descent_step(x: np.ndarray, eta: float) -> np.ndarray:
    n = x.shape[0]
    conv = np.convolve(x, x)               # full convolution, length 2n‑1
    k = int(conv.argmax())                 # index of the current maximum
    # sub‑gradient of (max conv) at this point:
    g = np.zeros_like(x)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    idx = np.arange(i_min, i_max + 1)
    jdx = k - idx
    g[idx] = x[jdx]                        # factor 2 omitted – absorbed in η
    # exponential update
    y = x * np.exp(-eta * g)
    if y.sum() == 0.0:                     # pathological case – re‑initialise randomly
        y = np.abs(np.random.randn(n))
    y = y / y.sum()
    return y

# ------------------------------------------------------------
# Linear‑program for the Frank–Wolfe type improvement
# Returns the optimal g (may be un‑normalised).  If scipy is missing
# the function returns None.
def _solve_lp_for_g(f: np.ndarray, M: float):
    try:
        from scipy.optimize import linprog
    except Exception:                     # scipy not available
        return None
    n = f.shape[0]
    rows = 2 * n - 1
    # Build matrix A such that (f * g)_k = Σ_i f_i * g_{k-i}
    # For column j (variable g_j) the coefficient in row k equals f_{k-j}
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f                     # copies f into a diagonal band
    b = np.full(rows, M, dtype=np.float64)    # upper bound for every convolution entry
    # maximise Σ g_j  ↔ minimise −Σ g_j
    c = -np.ones(n, dtype=np.float64)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True})
    if not res.success:
        return None
    return res.x
