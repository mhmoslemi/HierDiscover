"""
propose_candidate
-----------------
A two–phase search for a non‑negative step function that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm is deliberately different from the previous mirror‑descent‑only
approach:

1. **Parametric sweep** –  we generate sequences belonging to the family
   a_i = (i + σ)^{-p} (σ > 0, p ∈ (0,1)).  Both the exponent *p*, the shift
   *σ* and the length *L* are explored on a coarse grid and by random
   sampling.  Because the objective is scale‑invariant, the raw values are
   used directly – no explicit normalisation is required.

2. **Refinement** –  starting from the best candidate found,
   an entropic mirror‑descent loop is executed.  The sub‑gradient is taken
   from *all* convolution entries that are within a small tolerance of the
   current maximum (instead of a single index).  This yields a smoother
   descent direction.  The step size η decays as 1/√t and occasional
   random jitter is added to escape plateaus.

The whole procedure respects the 1000 s limit (a 5 s safety margin) and
prints progress information.  The function finally returns the list of
heights that achieved the smallest evaluation score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top‑level (no closures, no lambdas)
# ----------------------------------------------------------------------


def _generate_powerlaw_fixed(p: float, sigma: float, length: int) -> np.ndarray:
    """
    Return a deterministic power‑law sequence
        a_i = (i + sigma) ** (-p)   for i = 0,…,length‑1 .
    No normalisation is performed; the homogeneity of the objective makes
    the absolute scale irrelevant.
    """
    i = np.arange(length, dtype=np.float64)
    a = (i + sigma) ** (-p)
    return a


def _mirror_descent_step_multi(x: np.ndarray, eta: float, tol: float = 1e-3
                               ) -> np.ndarray:
    """
    One mirror‑descent iteration using *all* near‑maximising convolution indices.
    x must be a probability vector (non‑negative, sum = 1).
    The sub‑gradient for the ℓ∞‑norm at the current point is the sum of the
    gradients of all entries whose value lies within ``tol`` of the maximum.
    """
    n = x.shape[0]
    conv = np.convolve(x, x)                # length 2n‑1
    max_val = conv.max()
    # indices whose value is close to the maximum
    idxs = np.where(conv >= max_val - tol)[0]

    # build the sub‑gradient: g_i = sum_{k∈idxs, i in support} x_{k-i}
    g = np.zeros_like(x)
    for k in idxs:
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        idx = np.arange(i_min, i_max + 1, dtype=int)
        jdx = k - idx
        g[idx] += x[jdx]                     # factor 2 omitted – absorbed in η

    # exponential (multiplicative) update, then project onto the simplex
    y = x * np.exp(-eta * g)
    if y.sum() == 0.0:
        y = np.abs(np.random.randn(n))
    y /= y.sum()
    return y


def _evaluate(seq: np.ndarray) -> float:
    """Thin wrapper around the provided ``evaluate_sequence``."""
    try:
        return evaluate_sequence(seq.tolist())
    except Exception:
        return np.inf

    """

    deadline = time.time() + budget_s - 5.0   # safety margin

    best_score = float('inf')
    best_seq = None

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    # coarse grid for the exponent
