from scipy.optimize import linprog
import scipy.sparse as sp

def solve_linear_program(f, M):
    n=len(f)
    rows = 2*n-1
    data=[]
    row_ind=[]
    col_ind=[]
    for i, fi in enumerate(f):
        if fi==0: continue
        start = i
        end = i + n  # exclusive
        # rows start..end-1
        # for row k, col i gets coefficient fi
        # vector of length n for rows: row indices: start..end-1
        # we can extend slices
        row_idx = np.arange(start, end, dtype=int)
        data.extend([fi]*len(row_idx))
        row_ind.extend(row_idx.tolist())
        col_ind.extend([i]*len(row_idx))
    A_ub = sp.csr_matrix((data, (row_ind, col_ind)), shape=(rows, n))
    bounds = [(0, None)] * n
    # objective: maximize sum(g) → linprog minimizes c^T x; so set c = -1
    c = -np.ones(n)
    res = linprog(c, A_ub=A_ub, b_ub=np.full(rows, M), bounds=bounds, method='highs')
    if not res.success:
        return None
    g = res.x
    return g

    # Use coarse grid to get best.

    best_score = np.inf

        for c in offsets:
            idx = np.arange(n0, dtype=np.float64) + 1.0 + c
            f = 1.0 / (idx ** beta)
            f = f * (target_sum0 / np.sum(f))
            score = _objective(f)
            if score < best_score:
                best_score = score
                best_f = f.copy()
    # If height_sequence_1 available, consider using it truncated/scale
    if 'height_sequence_1' in globals():
        hs = np.array(height_sequence_1, dtype=np.float64)
        if len(hs) >= n0:
            f_h = hs[:n0].copy()
            f_h = f_h * (target_sum0 / np.sum(f_h))
            score = _objective(f_h)
            if score < best_score:
                best_score = score
                best_f = f_h.copy()

    def solve_lp(f_vec, M_val):

        rows = 2 * n - 1

        # column i contributes f_i to rows i .. i+n-1

        cols_idx = []

                continue

            end = i + n
            length = n
            # Actually rows from i to i+n-1 inclusive
            row_range = np.arange(start, end, dtype=np.int64)
            data.extend([fi] * len(row_range))
            rows_idx.extend(row_range.tolist())
            cols_idx.extend([i] * len(row_range))

"""
Propose a low‑score step function.

The algorithm consists of several stages:

1. **Coarse power‑law start** – a short sequence (n≈800) built from
   a_i = 1/(i+1+c)^β and normalised to √(2n).  A small grid over β and c
   gives a good initialization.

2. **Linear‑programming refinement** – repeatedly solve the LP
      maximise   Σ g_i
      subject to f*g ≤ M   (convolution linear in g) and g_i ≥ 0,
   where M = max(f*f).  The solution g is rescaled to the same ℓ₁‑norm
   and mixed with the current f.  This mimics the “active‑set’’ method
   described in the problem statement and quickly lowers the worst
   convolution value.

3. **Upscaling** – the refined short vector is interpolated to a longer
   length (n≈5000).  The ℓ₁‑norm is again normalised to √(2n).

4. **Active‑set gradient descent** – using the *top‑k* largest
   convolution entries (instead of a single maximum) we compute a
   sub‑gradient and take a scaled step, re‑normalising after each step.

5. **Simulated‑annealing fine‑tuning** – random mass transfers between
   two positions are accepted if they improve the objective or with a
   probability decreasing with a temperature schedule.  This helps to
   escape small plateaux.

All heavy work is performed with a fast FFT‑based objective, while the
final sequence is returned as a plain list of Python floats, ready for
``evaluate_sequence``.  The procedure respects the time budget
(`budget_s`) and aborts gracefully when the deadline approaches.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog


def _objective(seq: np.ndarray) -> float:
    """Fast evaluation of 2·n·max(conv(a,a)) / (sum(a)²) using FFT."""
    n = len(seq)
    s = np.sum(seq)
    if s < 0.01:
        return float("inf")
    m = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft * fft, n=m)[: 2 * n - 1]
    return float(2.0 * n * np.max(conv) / (s * s))


def _solve_lp(f_vec: np.ndarray, M_val: float) -> np.ndarray | None:
    """
    Solve the LP:
        maximise   Σ g_i
        subject to (f * g)_k ≤ M_val   for all k
                   g_i ≥ 0
    Returns the optimal g (as a dense ndarray) or None on failure.
    """
    n = len(f_vec)
    rows = 2 * n - 1

    # Build the sparse matrix A where A[k,i] = f_i if i contributes to (f*g)_k
    data = []
    row_ind = []
    col_ind = []
    for i, fi in enumerate(f_vec):
        if fi == 0.0:
            continue
        # rows i … i+n‑1 receive coefficient fi from column i
        start = i
        end = i + n                     # exclusive
        row_range = np.arange(start, end, dtype=np.int64)
        data.extend([fi] * len(row_range))
        row_ind.extend(row_range.tolist())
        col_ind.extend([i] * len(row_range))

    A = sp.csr_matrix((data, (row_ind, col_ind)), shape=(rows, n))
    b = np.full(rows, M_val, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)   # linprog minimises, so -c = maximise Σg

    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if res.success:
        return np.array(res.x, dtype=np.float64)
    return None


def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises the
    evaluation score.  The function respects the provided time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1. Short power‑law start (n0 ≈ 800)
    # ------------------------------------------------------------------
    n0 = 800
    target_sum0 = math.sqrt(2.0 * n0)

    betas = np.linspace(0.45, 0.55, 11)
    offsets = np.linspace(0.0, 4.0, 9)

    best_f = None
    best_score = float("inf")
    for beta in betas:
        for c in offsets:
            if time.time() > deadline:
                break
            idx = np.arange(n0, dtype=np.float64) + 1.0 + c
            f = 1.0 / (idx ** beta)
            f = f * (target_sum0 / np.sum(f))
            score = _objective(f)
            if score < best_score:
                best_score = score
                best_f = f.copy()
                print(f"[grid] beta={beta:.4f}, c={c:.2f} → score={score:.6f}")

    # optionally start from the provided height_sequence_1
    height_sequence_1 = globals().get("height_sequence_1")
    if height_sequence_1 is not None:
        hs = np.array(height_sequence_1, dtype=np.float64)
        if hs.size >= n0:
            cand = hs[:n0].copy()
            cand = cand * (target_sum0 / np.sum(cand))
            s = _objective(cand)
            if s < best_score:
                best_score = s
                best_f = cand.copy()
                print(f"[init] height_sequence_1 used, score={s:.6f}")

    f = best_f.copy()

    # ------------------------------------------------------------------
    # 2. Linear‑programming refinement (few iterations)
    # ------------------------------------------------------------------
    mix = 0.2
    max_lp_iters = 25
    lp_iter = 0
    while time.time() < deadline and lp_iter < max_lp_iters:
        # current max convolution of f with itself
        m = 1 << ((2 * n0 - 1).bit_length())
        fft_f = np.fft.rfft(f, n=m)
        conv_f = np.fft.irfft(fft_f * fft_f, n=m)[: 2 * n0 - 1]
        M = np.max(conv_f)

        g = _solve_lp(f, M)
        if g is None:
            break
        # renormalise g to the same ℓ₁‑norm
        g = np.maximum(g, 0.0)
        if np.sum(g) > 0:
            g = g * (target_sum0 / np.sum(g))
        else:
            g = np.full_like(g, target_sum0 / len(g))

        # convex combination update
        f = (1.0 - mix) * f + mix * g
        lp_iter += 1
        mix = max(mix * 0.9, 0.02)   # slowly reduce mixing factor

        if lp_iter % 5 == 0:
            print(f"[LP] iter={lp_iter}, score={_objective(f):.6f}")

    # ------------------------------------------------------------------
    # 3. Upscale to a longer sequence (n_big ≈ 5000)
    # ------------------------------------------------------------------
    n_big = 5000
    xs_small = np.linspace(0.0, 1.0, n0)
    xs_big = np.linspace(0.0, 1.0, n_big)
    a = np.interp(xs_big, xs_small, f)          # linear interpolation
    target_sum_big = math.sqrt(2.0 * n_big)
    a = a * (target_sum_big / np.sum(a))
    a = np.maximum(a, 0.0)

    # ------------------------------------------------------------------
    # 4. Gradient descent with a small active set (top‑k conv entries)
    # ------------------------------------------------------------------
    cur = a.copy()
    cur_sum = np.sum(cur)
    best_seq = cur.copy()
    best_score = _objective(best_seq)

    step_factor = 0.8
    no_improve = 0
    it_grad = 0
    top_k = 12                     # number of convolution entries that drive the gradient

    while time.time() < deadline:
        m = 1 << ((2 * n_big - 1).bit_length())
        fft_cur = np.fft.rfft(cur, n=m)
        conv_cur = np.fft.irfft(fft_cur * fft_cur, n=m)[: 2 * n_big - 1]

        max_conv = np.max(conv_cur)

        # pick the indices of the top‑k values (may contain duplicates for equal values)
        if top_k < conv_cur.size:
            idx_top = np.argpartition(conv_cur, -top_k)[-top_k:]
        else:
            idx_top = np.arange(conv_cur.size)

        # sub‑gradient summed over the selected indices
        grad = np.zeros_like(cur)
        for k in idx_top:
            shift = k - np.arange(n_big)
            mask = (shift >= 0) & (shift < n_big)
            grad[mask] += cur[shift[mask]]
        grad *= 2.0                     # d(max_conv)/da_i = 2·a_{k‑i}

        # step size proportional to the current ratio
        r = max_conv / (cur_sum ** 2)
        step = step_factor * r

        new_cur = cur - step * grad
        new_cur = np.maximum(new_cur, 0.0)

        s = np.sum(new_cur)
        if s < 1e-12:
            break
        new_cur = new_cur * (cur_sum / s)   # renormalise ℓ₁‑norm

        new_score = _objective(new_cur)
        if new_score < best_score:
            best_score = new_score
            best_seq = new_cur.copy()
            cur = new_cur
            no_improve = 0
            step_factor = min(step_factor * 1.07, 3.0)
            print(f"[grad] it={it_grad}, score={new_score:.6f}")
        else:
            # no improvement – keep moving but shrink the step
            cur = new_cur
            no_improve += 1
            step_factor = max(step_factor * 0.93, 0.01)
            if no_improve % 35 == 0:
                # occasional random redistribution to escape plateaux
                i, j = np.random.randint(n_big, size=2)
                delta = np.random.rand() * cur[i] * 0.07
                cur[i] = max(0.0, cur[i] - delta)
                cur[j] += delta
                cur = cur * (cur_sum / np.sum(cur))

        it_grad += 1
        if best_score < 1.502:
            print(f"[!Target] reached in gradient stage: {best_score:.6f}")
            break

    # ------------------------------------------------------------------
    # 5. Simulated annealing fine‑tuning
    # ------------------------------------------------------------------
    cur_seq = best_seq.copy()
    cur_score = best_score
    best_seq_sa = cur_seq.copy()
    best_score_sa = cur_score

    it_sa = 0
    T0 = 0.01 * cur_score
    Tmin = 1e-6

    while time.time() < deadline:
        # geometric temperature decay
        T = max(T0 * (0.9995 ** it_sa), Tmin)

        i, j = np.random.randint(n_big, size=2)
        delta = np.random.rand() * cur_seq[i] * 0.05
        if delta > cur_seq[i]:
            delta = cur_seq[i]

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta   # total sum unchanged

        new_score = _objective(new_seq)
        dE = new_score - cur_score

        if dE < 0 or np.random.rand() < math.exp(-dE / T):
            cur_seq = new_seq
            cur_score = new_score
            if new_score < best_score_sa:
                best_score_sa = new_score
                best_seq_sa = new_seq.copy()
                print(f"[SA] it={it_sa}, best={new_score:.6f}")

        it_sa += 1
        if best_score_sa < 1.502:
            print(f"[!Target] SA reached {best_score_sa:.6f} at it={it_sa}")
            break

    print(f"[+] Search finished. Best score = {best_score_sa:.6f}")
    return [float(x) for x in best_seq_sa]
```
