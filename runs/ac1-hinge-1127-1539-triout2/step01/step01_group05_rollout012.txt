    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The strategy consists of two stages.

1️⃣  **Parameter sweep**  
    We use the analytically motivated family  

        a_i(α,δ) = (i + δ) ** (-α) ,   i = 0 … n‑1 ,

    For a grid of (α,δ) values we normalise the vector so that
    sum(a)=sqrt(2·n) (the evaluation does not depend on the absolute scale)
    and keep the best candidate found.

    This simple family already reaches scores well below the target

2️⃣  **Local refinement** (optional, runs while time permits)

    applied.  Small random jitter is injected periodically to escape plateaus.
    The best sequence encountered during this phase is returned.

# Helper utilities (all top‑level, no closures)

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _compute_score(a: np.ndarray) -> float:
    """
    Fast version of `evaluate_sequence` – the security wrapper is omitted.
    The function assumes `a` is a 1‑D float array of non‑negative numbers.
    """
    n = a.shape[0]
    nfft = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))


def _make_candidate(alpha: float, delta: float, n: int, target_sum: float) -> np.ndarray:
    """
    Build the vector a_i = (i+delta)^(-alpha) and renormalise it so that
    sum(a) = target_sum.
    """
    i = np.arange(n, dtype=np.float64)
    raw = (i + delta) ** (-alpha)
    # prevent division by zero if delta==0 (not used here)
    scale = target_sum / raw.sum()
    return raw * scale


def _gradient_at_max(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv(a,a)) at the index k_max.
    g_i = 2 * ( a_{k_max-i}  (if inside) + a_{i-k_max} (if inside) )
    """
    n = a.shape[0]
    i = np.arange(n, dtype=np.int64)

    idx1 = k_max - i                     # k - i
    idx2 = i - k_max                     # i - k

    mask1 = (idx1 >= 0) & (idx1 < n)
    mask2 = (idx2 >= 0) & (idx2 < n)

    grad = np.zeros_like(a)
    grad[i[mask1]] += a[idx1[mask1]]
    grad[i[mask2]] += a[idx2[mask2]]
    grad *= 2.0
    return grad

        n = len(height_sequence_1)               # type: ignore[name-defined]
    except Exception:
        n = 5641                                   # the length that gave the last bound

    target_sum = math.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # 1️⃣  Grid search over (α,δ)
    # --------------------------------------------------------------
    best_a = None
    best_score = float("inf")
    best_params = None

    # Coarse grid – wide enough to include the known good region.
    alphas = np.linspace(0.45, 0.70, 26)          # step ≈0.01
    deltas = np.linspace(0.5, 5.0, 26)           # step ≈0.18

    print(f"Starting grid search (n={n}) – {len(alphas)*len(deltas)} candidates")
    for alpha in alphas:
        for delta in deltas:
            # time‑out check
            if time.time() >= deadline:
                break

            a = _make_candidate(alpha, delta, n, target_sum)
            score = _compute_score(a)

            if score < best_score:
                best_score = score
                best_a = a.copy()
                best_params = (alpha, delta)
                elapsed = time.time() - start_time
                print(f"  New grid best: score={best_score:.6f} (α={alpha:.3f}, δ={delta:.3f}) "
                      f"elapsed={elapsed:.1f}s")

        if time.time() >= deadline:
            break

    if best_a is None:          # should never happen, but be safe
        raise RuntimeError("No valid candidate was generated in the grid search.")

    # --------------------------------------------------------------
    # 2️⃣  Local refinement – gradient descent on the true max
    # --------------------------------------------------------------
    # Use the remaining time for a short but effective refinement.
    time_left = deadline - time.time()
    if time_left > 30.0:
        print("Starting local refinement phase")
        a = best_a.copy()
        nfft = _next_pow2(2 * n - 1)

        # simple momentum variables
        momentum = 0.9
        v = np.zeros_like(a)

        # step‑size schedule
        eta = 0.12
        eta_min = 0.001

        iteration = 0
        while time.time() < deadline:
            iteration += 1

            # ----- convolution and location of the maximal entry -----
            af = np.fft.rfft(a, n=nfft)
            conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
            k_max = int(conv.argmax())
            max_conv = conv[k_max]

            # ----- gradient of the (non‑smooth) max  -----
            grad = _gradient_at_max(a, k_max)

            # ----- momentum update & projection -----
            v = momentum * v - eta * grad
            a_new = a + v
            a_new = np.maximum(a_new, 0.0)                # non‑negativity

            # renormalise to keep the sum fixed (scaling does not affect the score)
            s = a_new.sum()
            if s == 0.0:                                 # very unlikely
                a_new = np.full_like(a_new, target_sum / n)
            else:
                a_new *= target_sum / s

            # ----- occasional jitter to avoid stagnation -----
            if iteration % 97 == 0:
                jitter = 1e-5 * np.random.randn(n)
                a_new = np.maximum(a_new + jitter, 0.0)
                a_new *= target_sum / a_new.sum()

            # ----- evaluate -----
            score_new = _compute_score(a_new)

            # ----- accept/reject -----
            improve = score_new < best_score
            if improve:
                a = a_new
                best_score = score_new
                best_a = a_new.copy()
                elapsed = time.time() - start_time
                print(f"Iter {iteration:5d}: improved score={best_score:.6f} "
                      f"(elapsed {elapsed:.1f}s)")
            else:
                # simple back‑tracking: if the step was not good, discard momentum
                v[:] = 0.0

            # ----- adapt step size -----
            eta = max(eta * 0.998, eta_min)

            # optional early stop if the improvement becomes tiny
            if iteration % 500 == 0:
                if eta < 0.002:
                    break

    # --------------------------------------------------------------
    # Return the best sequence found as a plain Python list.
    # --------------------------------------------------------------
    print(f"Finished. Best score = {best_score:.6f} (params α={best_params[0]:.3f}, "
          f"δ={best_params[1]:.3f})")
    return best_a.tolist()
