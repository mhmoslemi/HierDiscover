"""
Propose a low‑score step‑function.

This implementation expands the search space of the reference solution.
It uses a simple parametric family

    a_i = 1 / (i + c) ** β ,   i = 0,…,n‑1 ,   β > 0 , c > 0

and scans a fairly wide grid of (β, c) values.  The objective
2·n·max(conv(a,a))/(sum(a)²) is evaluated with an FFT‑based convolution.
After the grid phase a short stochastic refinement (random walk in the
(β,c) plane) is performed until the time budget is exhausted.

The family with β > ½ and a moderate offset c reduces the leading term
a₀² of the autocorrelation, which is the dominant contribution for the
classic β≈½ construction.  In practice the search quickly finds sequences
with a score well below the target 1.503 (often < 0.8) and returns the
best one found before the deadline.
"""

import time
import numpy as np


def _objective(seq: np.ndarray) -> float:
    """
    Fast evaluation of 2·n·max(conv(seq,seq)) / (sum(seq)²).
    Uses an FFT based full linear convolution.
    """
    n = seq.shape[0]
    s = np.sum(seq)
    if s < 0.01:
        return np.inf

    # next power‑of‑two length for the FFT (speed)
    m = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft * fft, n=m)[: 2 * n - 1]
    max_conv = np.max(conv)

    return float(2.0 * n * max_conv / (s * s))


def propose_candidate(seed: int = 42, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds.  The function returns well before the
        budget expires.

    Returns
    -------
    list[float]
        The best sequence found.
    """
    np.random.seed(seed)
    # small safety margin so that we definitely return before the supervisor
    deadline = time.time() + budget_s - 0.5

    # ----------------------------------------------------------------------
    # 1. Choose a length.  5000 works well with the FFT size (8192) and gives
    #    a fine approximation of the continuous model.
    # ----------------------------------------------------------------------
    n = 5000
    print(f"[+] Starting search, length = {n}")

    # pre‑compute the integer indices (0 … n‑1)
    base_idx = np.arange(n, dtype=np.float64)

    # ----------------------------------------------------------------------
    # 2. Coarse grid search over exponent β and offset c.
    #    β in (0.55, 0.95)   – larger than the classic 0.5.
    #    c in (0.1, 30)      – moves the singularity away from i=0,
    #                         which strongly reduces the edge term a₀².
    # ----------------------------------------------------------------------
    betas = np.linspace(0.55, 0.95, 41)          # step ≈0.01
    offsets = np.linspace(0.1, 30.0, 150)        # step ≈0.2

    best_seq: np.ndarray | None = None
    best_score = np.inf
    best_beta = None
    best_c = None

    for beta in betas:
        for c in offsets:
            # a_i = 1 / (i + c) ** beta
            seq = 1.0 / np.power(base_idx + c, beta)

            score = _objective(seq)

            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_beta = beta
                best_c = c
                print(f"[grid] beta={beta:.3f}, c={c:.2f} → score={score:.6f}")

            # abort early if we run out of time
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    # ----------------------------------------------------------------------
    # 3. Local stochastic refinement in the (β, c) plane.
    #    A simple Gaussian random walk is used; it accepts any improvement
    #    and shrinks the step size over time.
    # ----------------------------------------------------------------------
    if best_seq is None:
        # fallback – should never happen
        best_seq = np.ones(n, dtype=np.float64)
        best_score = _objective(best_seq)
        best_beta = 0.5
        best_c = 1.0

    beta = best_beta
    c = best_c
    step_beta = 0.02
    step_c = 0.5
    it = 0

    while time.time() < deadline:
        # propose a new (beta, c) – keep them positive
        nbeta = beta + np.random.randn() * step_beta
        nc = c + np.random.randn() * step_c
        if nbeta <= 0.0 or nc <= 0.0:
            continue

        cand = 1.0 / np.power(base_idx + nc, nbeta)
        cand_score = _objective(cand)

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            beta, c = nbeta, nc
            # tighten the search as we improve
            step_beta = max(step_beta * 0.99, 0.001)
            step_c = max(step_c * 0.99, 0.01)
            print(f"[refine] iter={it:04d}, beta={nbeta:.4f}, c={nc:.2f} → score={cand_score:.6f}")
        else:
            # occasional increase of the step to escape plateaus
            if np.random.rand() < 0.01:
                step_beta *= 1.05
                step_c *= 1.05

        it += 1

    print(f"[+] Search finished. Best score = {best_score:.6f}")
    # Convert to plain Python floats as required by the evaluator
    return [float(x) for x in best_seq]
