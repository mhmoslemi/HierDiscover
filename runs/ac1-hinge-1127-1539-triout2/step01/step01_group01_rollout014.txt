"""
propose_candidate – annealed soft‑max gradient descent for the self‑convolution
bound.

The algorithm works in three stages:

1. **Initial sweep** – a collection of simple “U‑shaped’’ families (linear,
   quadratic, exponential and a pure power law) are tried.  The best
   candidate of this sweep (or the provided global `height_sequence_1`,
   if it exists) is taken as the starting point.

2. **Annealed soft‑max gradient descent** – we minimise the smooth approximation
   `log‑sum‑exp_β (a*a)` of the maximal self‑convolution.
   For a given β the gradient is  
       ∇ = 2·β·(a ⊛ p)   (⊛ = cross‑correlation, p = soft‑max weights),
   which is obtained efficiently with a single FFT.  β is increased
   gradually (annealing) so that the optimisation first moves in a
   well‑behaved smooth landscape and finally focuses on the true maximum.
   After each step we clip negative entries and re‑scale to the required
   ℓ₁‑norm (`√(2n)`), thus keeping the objective exactly the one used by
   `evaluate_sequence`.

3. **Escaping plateaus** – occasional random mass transfers, small Gaussian
   jitter and a simple back‑tracking line‑search (step‑size adaptation) are
   performed to avoid getting stuck.

The whole search respects the 1000 s budget and always returns the best
sequence found.  All numbers are non‑negative and the ℓ₁‑norm is kept
positive, satisfying the security checks in `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using a common FFT length L_fft.
    Returns the full convolution of length 2*len(a)-1.
    """
    fa = np.fft.rfft(a, n=L_fft)
    conv_full = np.fft.irfft(fa * fa, n=L_fft)
    n = a.shape[0]
    return conv_full[: 2 * n - 1]


def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """
    Scale a non‑negative vector to have ℓ₁‑norm `target`.
    Zero entries stay zero; if the whole vector is zero a uniform
    distribution is returned.
    """
    s = v.sum()
    if s > 0.0:
        return v * (target / s)
    # fallback – uniform distribution
    return np.full_like(v, target / v.shape[0])


def generate_U_shape(
    n: int, target_sum: float, kind: str = "linear", param: float = 0.0
) -> np.ndarray:
    """
    Produce a symmetric “U‑shaped’’ sequence of length n.
    kind ∈ {'linear','quadratic','exponential'}.
    `param` ≥ 0 controls the edge‐enhancement strength.
    The result is scaled to the prescribed ℓ₁‑norm.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid  # ∈[0,1], distance from centre

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_to_sum(raw, target_sum)


def generate_power_shape(
    n: int, target_sum: float, exponent: float = 1.0, epsilon: float = 1e-6
) -> np.ndarray:
    """
    Produce a symmetric shape where the edge amplitude follows a power law.
    a_i ∝ ε + (|i‑mid|/mid) ** exponent.
    The result is scaled to `target_sum`.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid
    raw = epsilon + d ** exponent
    raw = np.clip(raw, 0.0, None)
    return project_to_sum(raw, target_sum)


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a tiny fraction of mass from one random index to another.
    The total ℓ₁‑norm is preserved by a final projection.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0.0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_to_sum(a, target_sum)


def softmax_weights(x: np.ndarray, beta: float) -> np.ndarray:
    """
    Stable soft‑max with temperature 1/β.
    Returns a probability vector p_i = exp(β·x_i)/Σ_j exp(β·x_j).
    """
    max_x = x.max()
    exps = np.exp(beta * (x - max_x))
    s = exps.sum()
    if s == 0.0:
        # degenerate case – fall back to uniform distribution
        return np.full_like(x, 1.0 / x.shape[0])
    return exps / s


def gradient_logsumexp(
    a: np.ndarray, conv: np.ndarray, beta: float, L_fft: int
) -> np.ndarray:
    """
    Gradient of  log‑sum‑exp_β (a*a)  with respect to a.
    Implements the formula
        ∇ = 2·β·(a ⊛ p)
    where p are the soft‑max weights of the convolution entries.
    The cross‑correlation is computed via a single FFT.
    """
    n = a.shape[0]
    # soft‑max weights
    p = softmax_weights(conv, beta)
    # reverse p for the cross‑correlation
    p_rev = p[::-1]
    # FFT of a (computed already once outside for speed, but we redo here
    # because a changes each iteration – the cost is negligible compared to
    # the whole loop).
    fa = np.fft.rfft(a, n=L_fft)
    fp_rev = np.fft.rfft(p_rev, n=L_fft)
    # cross‑correlation a ⊛ p = conv(a, p_rev) evaluated at indices
    # (2·n‑2)−j, j=0..n‑1
    corr_full = np.fft.ifft(fa * fp_rev, n=L_fft).real
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * beta * corr_full[idx]
    return grad


def compute_score(a: np.ndarray, target_sum: float) -> float:
    """
    Exact objective value used by `evaluate_sequence`.
    Assumes a is non‑negative and its ℓ₁‑norm ≈ target_sum.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_b = conv.max()
    return float(2.0 * n * max_b / (target_sum ** 2))


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises the
    `evaluate_sequence` bound.  The method uses an initial parametric sweep
    followed by an annealed soft‑max gradient descent with occasional
    random jumps.  It runs for at most *budget_s* seconds and returns the
    best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    # --------------------------------------------------------------
    # Choose a length.  Preserve the length of a known construction if
    # one is already present; otherwise pick a size that gives a good
    # trade‑off between quality and speed.
    # --------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        init_seq = np.array(base, dtype=np.float64)
    else:
        n = np.random.randint(7000, 10000)  # a “large but manageable’’ size
        init_seq = None

    target_sum = np.sqrt(2.0 * n)           # normalisation used by the paper

    # Pre‑compute the FFT length that can hold the longest linear convolution
    # needed (a * a_rev) has length 3·n‑2.
    L_fft = 1 << ((3 * n - 2).bit_length())

    # --------------------------------------------------------------
    # Phase 1 – cheap parametric sweep over simple families
    # --------------------------------------------------------------
    best_seq = np.full(n, target_sum / n, dtype=np.float64)  # uniform fallback
    best_score = compute_score(best_seq, target_sum)

    print(f"[{time.time():.2f}] Starting parametric sweep (n={n})")

    shape_kinds = ("linear", "quadratic", "exponential")
    param_grids = {
        "linear": np.linspace(0.0, 5.0, 51),          # 0 … 5 in steps of 0.1
        "quadratic": np.linspace(0.0, 4.0, 41),      # 0 … 4
        "exponential": np.linspace(0.0, 3.0, 31),    # 0 … 3
    }

    for kind in shape_kinds:
        for p in param_grids[kind]:
            cand = generate_U_shape(n, target_sum, kind=kind, param=p)
            score = compute_score(cand, target_sum)
            if score < best_score:
                best_score, best_seq = score, cand.copy()
                print(f"  New best from {kind} sweep (p={p:.2f}): {best_score:.6f}")

            if time.time() >= deadline:
                print("Time exhausted during sweep.")
                return best_seq.tolist()

    # also try the pure power law family
    for exp in np.linspace(0.5, 5.0, 46):
        cand = generate_power_shape(n, target_sum, exponent=exp)
        score = compute_score(cand, target_sum)
        if score < best_score:
            best_score, best_seq = score, cand.copy()
            print(f"  New best from power sweep (exp={exp:.2f}): {best_score:.6f}")
        if time.time() >= deadline:
            print("Time exhausted during power sweep.")
            return best_seq.tolist()

    # if a historical construction is available and better, use it
    if init_seq is not None:
        score = compute_score(init_seq, target_sum)
        if score < best_score:
            best_score, best_seq = score, init_seq.copy()
            print(f"  Using provided height_sequence_1: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – annealed soft‑max gradient descent
    # --------------------------------------------------------------
    a = best_seq.copy()
    cur_score = best_score

    # learning‑rate (absolute step size).  We express it as a factor of the
    # average entry `target_sum / n`.
    avg_entry = target_sum / n
    lr = avg_entry * 0.5                     # initial step size
    min_lr = avg_entry * 1e-5

    beta = 0.5                               # start temperature (smooth)
    beta_max = 50.0
    beta_mul = 1.12                          # multiplicative annealing factor

    iteration = 0
    no_improve = 0

    print(f"[{time.time():.2f}] Entering annealed gradient descent (lr={lr:.6f}, β={beta:.2f})")

    while time.time() < deadline:
        iteration += 1

        # ----- compute self‑convolution (full) and its gradient -----
        conv = conv_self_fft(a, L_fft)           # length 2·n‑1
        grad = gradient_logsumexp(a, conv, beta, L_fft)

        # ----- trial step (simple back‑tracking line search) -----
        cand = a - lr * grad
        cand = np.clip(cand, 0.0, None)
        cand = project_to_sum(cand, target_sum)

        cand_score = compute_score(cand, target_sum)

        if cand_score + 1e-12 < cur_score:
            # successful step
            a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"Iter {iteration:06d} – improved to {best_score:.6f}")
            no_improve = 0
            lr = min(lr * 1.02, avg_entry)        # slowly grow (but keep reasonable)
        else:
            # reject – shrink step size
            lr = max(lr * 0.5, min_lr)
            no_improve += 1

        # ----- anneal β (make the approximation sharper) -----
        if iteration % 300 == 0 and beta < beta_max:
            beta = min(beta_max, beta * beta_mul)

        # ----- occasional random perturbations to escape plateaus -----
        if np.random.rand() < 0.015:
            a = random_pair_transfer(a, target_sum, scale=0.008)
            cur_score = compute_score(a, target_sum)   # cheap re‑evaluate

        # ----- jitter after prolonged stagnation -----
        if no_improve >= 2500:
            jitter = np.random.randn(n) * avg_entry * 0.002
            a = np.clip(a + jitter, 0.0, None)
            a = project_to_sum(a, target_sum)
            cur_score = compute_score(a, target_sum)
            no_improve = 0
            print(f"Iter {iteration:06d} – jitter applied (lr={lr:.6f}, β={beta:.2f})")

        # safety check for the time budget
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # Phase 3 – final brief random search (optional)
    # --------------------------------------------------------------
    print(f"[{time.time():.2f}] Gradient phase finished; best score {best_score:.6f}")

    # Perform a few cheap random pair transfers in the remaining time
    while time.time() < deadline:
        i, j = np.random.choice(n, 2, replace=False)
        # transfer a small fraction from i to j
        delta = 0.005 * a[i]
        if delta > 0.0:
            a[i] -= delta
            a[j] += delta
            a = project_to_sum(a, target_sum)
            s = compute_score(a, target_sum)
            if s < best_score:
                best_score, best_seq = s, a.copy()
                print(f"  Random swap improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # Return the best sequence (plain Python list)
    # --------------------------------------------------------------
    return best_seq.tolist()
