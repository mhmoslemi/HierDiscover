def annealed_gradient_descent(a, target_sum, deadline):
   best_score = evaluate_sequence(a.tolist())
   best_seq = a.copy()
   n = len(a)
   # initial conv
   conv = conv_self_fft(a)
   max_conv = conv.max()
   tau = max_conv * 0.5 if max_conv>0 else 1.0
   eta = target_sum/n * 0.01  # smaller step
   momentum = np.zeros_like(a)
   beta = 0.9
   iteration = 0
   while time.time() < deadline:
        conv = conv_self_fft(a)
        max_conv = conv.max()
        # update temperature tau
        tau = max(tau * 0.995, 1e-6)   # decays
        # compute softmax weights
        # stable
        t = conv / tau
        t_max = np.max(t)
        w = np.exp(t - t_max)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1/w.shape[0])   # uniform
        else:
            w = w / w_sum
        # compute gradient via correlation
        corr = np.correlate(a, w, mode='full')
        start = len(w) - 1
        g = 2.0 * corr[start:start + n]
        # orthogonalize gradient
        direction = -g + np.mean(g)
        # momentum update
        momentum = beta * momentum + (1 - beta) * direction
        a_new = a + eta * momentum
        a_new = np.clip(a_new, 0.0, None)
        a_new = project_to_sum(a_new, target_sum)
        # evaluate
        score = evaluate_sequence(a_new.tolist())
        iteration += 1
        # if improvement

Propose a step‑function (non‑negative sequence) that minimises the
self‑convolution bound used in `evaluate_sequence`.

Algorithm
---------

1. **Initialisation** – If a previously found sequence `height_sequence_1`
   is available, keep its length and rescale it to the normalisation
   Σa = √(2 n).  Otherwise pick a large length (≈8 000) and start from
   a uniform distribution.

2. **Parametric sweep** – Try a few symmetric “U‑shaped’’ families
   (linear, quadratic, exponential) with several strengths.  Keep the
   best candidate.

3. **Projected sub‑gradient phase** – Repeatedly move the current
   vector opposite to the sub‑gradient of the maximal self‑convolution
   (computed only at the active indices).  The step is orthogonal to the
   all‑ones direction so the ℓ₁‑norm stays fixed; after the step the
   vector is clipped, symmetrised and projected back onto the prescribed
   sum.  Occasionally perform tiny random pair‑wise mass transfers and
   inject jitter to escape plateaus.

4. **Annealed soft‑max gradient phase** – Replace the hard max by a
   smooth approximation `τ · log ∑exp(conv/τ)`.  The gradient of this
   surrogate is cheap to obtain via a correlation computed with FFTs.
   Decrease the temperature `τ` geometrically; use a momentum term
   (exponential moving average) to stabilise the updates.  After each
   update enforce symmetry, clip negatives, and re‑project onto the
   required ℓ₁‑norm.  Keep the best sequence seen throughout.

The whole search respects the supplied time budget (≈1000 s) and always
returns the best sequence found.  All helper functions are top‑level,
use only NumPy and the provided `evaluate_sequence`, and avoid closures
or lambdas.
"""

import time
import numpy as np

# Helper utilities

def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution of a with itself using a power‑of‑two FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def correlate_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Full linear correlation of a with w (both real) using FFTs."""
    n = a.shape[0]
    m = w.shape[0]
    L = 1 << ((n + m - 1).bit_length())
    Fa = np.fft.rfft(a, n=L)
    Fw = np.fft.rfft(w, n=L)
    corr_full = np.fft.irfft(Fa * np.conj(Fw), n=L)[: n + m - 1]
    return corr_full


def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector to have ℓ₁‑norm equal to *target*."""
    s = v.sum()
    if s > 0.0:
        return v * (target / s)
    # fallback – uniform distribution
    return np.full_like(v, target / v.shape[0])


def enforce_symmetry(v: np.ndarray) -> np.ndarray:
    """Make the vector symmetric (v[i] = v[-i‑1]) by averaging."""
    return (v + v[::-1]) * 0.5


def generate_U_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
) -> np.ndarray:
    """
    Produce a symmetric “U‑shaped’’ sequence of length *n*.
    `kind` ∈ {linear, quadratic, exponential}.
    `param` ≥ 0 controls the edge‑enhancement strength.
    The result is rescaled to have ℓ₁‑norm = *target_sum*.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid  # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    else:
        raise ValueError("unknown shape kind")

    raw = np.clip(raw, 0.0, None)
    return project_to_sum(raw, target_sum)


def subgradient_step(
    a: np.ndarray,
    target_sum: float,
    eta: float,
    rel_tol: float = 0.0,
) -> np.ndarray:
    """
    One projected sub‑gradient step aimed at decreasing the maximal
    entry of the self‑convolution.
    *rel_tol* enlarges the active set: all convolution entries ≥ (1‑rel_tol)·max.
    """
    n = a.shape[0]
    conv = conv_self_fft(a)
    max_val = conv.max()
    if rel_tol <= 0.0:
        act = np.array([np.argmax(conv)], dtype=np.int64)
    else:
        act = np.where(conv >= max_val * (1.0 - rel_tol))[0]

    g = np.zeros_like(a)
    idx = np.arange(n, dtype=np.int64)
    for k in act:
        j = k - idx
        mask = (j >= 0) & (j < n)
        g[mask] += 2.0 * a[j[mask]]

    # orthogonal to the all‑ones direction (preserve sum)
    direction = -g + np.mean(g)
    a_new = a + eta * direction
    a_new = np.clip(a_new, 0.0, None)
    a_new = enforce_symmetry(a_new)
    a_new = project_to_sum(a_new, target_sum)
    return a_new


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """Transfer a small mass from one random index to another."""
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0.0:
        return a
    a[i] -= delta
    a[j] += delta
    a = enforce_symmetry(a)
    return project_to_sum(a, target_sum)

            a = enforce_symmetry(a)
            a = project_to_sum(a, target_sum)
            no_improve = 0
            print(f"Iter {iteration:06d} – jitter applied")

        if time.time() >= deadline:
            print("Time exhausted during sub‑gradient phase.")
            break

    # --------- Phase 3: annealed soft‑max gradient descent --------------------
    a = best_seq.copy()
    # initialise temperature as a fraction of the current max convolution
    conv = conv_self_fft(a)
    tau = max(conv.max() * 0.5, 1e-6)
    eta = 0.01 * (target_sum / n)   # a bit smaller step for the smoother phase
    momentum = np.zeros_like(a)
    beta = 0.9                      # momentum decay
    iter_ann = 0

    print(f"[{time.time():.2f}] Starting annealed gradient phase (tau={tau:.6f})")
    while time.time() < deadline:
        # FFT of current a (same length for both convolutions and correlation)
        n = a.shape[0]
        L = 1 << ((3 * n - 2).bit_length())          # enough for correlation
        Fa = np.fft.rfft(a, n=L)

        # ---------- self‑convolution (for evaluation and temperature) ----------
        conv_full = np.fft.irfft(Fa * Fa, n=L)[: 2 * n - 1]
        max_conv = conv_full.max()

        # ---------- temperature schedule ----------
        tau = max(tau * 0.995, max_conv * 1e-5)      # decay, never below tiny fraction

        # ---------- soft‑max weights ----------
        t = conv_full / tau
        t_max = np.max(t)
        exp_t = np.exp(t - t_max)                   # stable exp
        w = exp_t / exp_t.sum()                     # shape (2n‑1,)

        # ---------- gradient via correlation ----------
        Fw = np.fft.rfft(w, n=L)
        corr_full = np.fft.irfft(Fa * np.conj(Fw), n=L)[: n + (2 * n - 1) - 1]
        start = (2 * n - 1) - 1                      # = 2n‑2
        g = 2.0 * corr_full[start:start + n]

        # ---------- projected gradient step ----------
        direction = -g + np.mean(g)                  # orthogonal to all‑ones
        momentum = beta * momentum + (1.0 - beta) * direction
        a_new = a + eta * momentum
        a_new = np.clip(a_new, 0.0, None)
        a_new = enforce_symmetry(a_new)
        a_new = project_to_sum(a_new, target_sum)

        # ---------- occasional stochastic perturbations ----------
        if np.random.rand() < 0.015:
            a_new = random_pair_transfer(a_new, target_sum, scale=0.003)

        # ---------- evaluate ----------
        score = evaluate_sequence(a_new.tolist())
        iter_ann += 1

        if score + 1e-12 < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(f"Annealed iter {iter_ann:06d} – improved to {best_score:.6f} (tau={tau:.6f})")

        # keep the current iterate for the next step
        a = a_new

        # occasional jitter to keep exploration alive
        if iter_ann % 2500 == 0:
            jitter = np.random.randn(n) * 0.0004
            a = np.clip(a + jitter, 0.0, None)
            a = enforce_symmetry(a)
            a = project_to_sum(a, target_sum)

        # safety check for time
        if time.time() >= deadline:
            print("Time exhausted during annealed phase.")
            break

    print(f"[{time.time():.2f}] Search finished – best score {best_score:.6f}")
    return best_seq.tolist()
