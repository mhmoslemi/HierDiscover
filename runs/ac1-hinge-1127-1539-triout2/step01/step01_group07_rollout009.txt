def fast_score(seq):
    a = np.asarray(seq, dtype=np.float64)
    n = a.shape[0]
    # zero pad to length 2*n - 1, but using FFT power of 2 for speed
    size = 1 << ((2*n-1).bit_length())
    # rfft size
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa*fa, n=size)[:2*n-1]
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2 * n * max_b / (sum_a**2)
```

def fast_score(seq):
    a = np.asarray(seq, dtype=np.float64)
    n = a.shape[0]
    # compute convolution via FFT
    size = 1 << ((2*n - 1).bit_length())
    # use real FFT
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2*n-1]
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))
```

"""
search for a low‑evaluation non‑negative step function.

The algorithm consists of three phases:

1. **Coarse grid search** over the two‑parameter family
       a_i = ((i + γ + 1)^p – (i + γ)^p)      (p∈(0,1), γ≥0)
   with a moderate length (≈20 000).  The pair (p,γ) giving the smallest
   fast evaluation (2 n·max(conv)/∑a²) is remembered.

2. **Fine‑grained refinement** of (p,γ) in a neighbourhood of the best
   pair and a final construction on a larger length (≈50 000).  The
   sequence is normalised to unit L¹‑norm.

3. **Simple stochastic hill‑climbing** on the simplex:
   tiny amounts of mass are moved from a currently large coefficient to a
   small one (or a random pair).  After each move the fast score is
   recomputed (via an FFT‑based convolution).  Improvements are kept.
   The loop runs until the time budget (≈1000 s) is exhausted.

The returned list is ready for the official `evaluate_sequence` routine.
"""

import time
import numpy as np


def _generate_sequence(n: int, p: float, gamma: float) -> np.ndarray:
    """Return a non‑negative vector of length *n* defined by the
    power‑law difference family and normalised to L¹ = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.power(i + gamma + 1.0, p) - np.power(i + gamma, p)
    total = np.power(n + gamma, p) - np.power(gamma, p)
    # Guard against degenerate denominator (should not happen for the
    # ranges we explore).
    if total == 0.0:
        a[:] = 0.0
        return a
    a /= total
    # Numerical noise may create tiny negatives – clip them.
    np.clip(a, 0.0, 1000.0, out=a)
    return a


def _fast_score(seq: np.ndarray) -> float:
    """Fast counterpart of `evaluate_sequence`.  Uses an FFT based
    convolution; the expression is 2 n·max(b)/∑a² (with n = len(seq))."""
    a = np.asarray(seq, dtype=np.float64)
    n = a.shape[0]
    # Size of FFT – next power of two ≥ 2 n‑1.
    fft_len = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list:
    """
    Produce a list of non‑negative heights with a low evaluation score.
    The search runs for at most *budget_s* seconds.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # a small safety margin

    # -----------------------------------------------------------------
    # 1. Coarse grid search on a medium sized vector
    # -----------------------------------------------------------------
    n_grid = 20000
    best_score = float("inf")
    best_p = None
    best_gamma = None

    p_vals = np.linspace(0.30, 0.55, 26)          # step ≈0.01
    gamma_vals = np.arange(0, 201, 5)            # 0 … 200

    for p in p_vals:
        for gamma in gamma_vals:
            seq = _generate_sequence(n_grid, p, float(gamma))
            val = _fast_score(seq)
            if val < best_score:
                best_score = val
                best_p = p
                best_gamma = float(gamma)
                print(f"[grid] new best {val:.6f} (p={p:.4f}, γ={gamma})")
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    # -----------------------------------------------------------------
    # 2. Fine neighbourhood search (tiny local optimisation of parameters)
    # -----------------------------------------------------------------
    if best_p is not None and time.time() < deadline:
        p0, g0 = best_p, best_gamma
        # explore a narrow box around the current best
        for dp in (-0.008, -0.004, 0.0, 0.004, 0.008):
            p = max(0.01, min(0.99, p0 + dp))
            for dg in range(-20, 21, 4):
                gamma = max(0.0, g0 + dg)
                seq = _generate_sequence(n_grid, p, float(gamma))
                val = _fast_score(seq)
                if val < best_score:
                    best_score = val
                    best_p = p
                    best_gamma = float(gamma)
                    print(f"[refine] new best {val:.6f} (p={p:.4f}, γ={gamma})")
                if time.time() > deadline:
                    break
            if time.time() > deadline:
                break

    # -----------------------------------------------------------------
    # 3. Build a longer final sequence with the best parameters
    # -----------------------------------------------------------------
    n_final = 50000
    # If the remaining time is too small, fall back to a shorter length.
    if time.time() + 2.0 > deadline:
        n_final = 30000

    seq = _generate_sequence(n_final, best_p, best_gamma)
    seq = np.clip(seq, 0.0, 1000.0)

    # Normalise explicitly (tiny numerical drift may have occurred)
    s = seq.sum()
    if s == 0.0:

    print(f"[init] length={n_final}, p={best_p:.4f}, γ={best_gamma:.1f}, "

        print(f"[final] external evaluation = {ext:.6f}")
    except Exception:
        pass

    return best_seq.tolist()
