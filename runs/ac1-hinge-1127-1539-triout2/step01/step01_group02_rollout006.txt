def project_simplex(v):
    # v can have negative entries
    # ensure non-negative sum to 1
    # standard algorithm:
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def project_to_simplex(v):
    # ensure sum(v) = 1 and v >= 0
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        # all negative, return uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def gradient_step(x, beta, eta, m=None, v=None, t=None):
    n = x.shape[0]
    conv = np.convolve(x, x)  # length 2n-1
    w = np.exp(beta * conv)
    w_sum = w.sum()
    p = w / w_sum
    # cross-correlation
    revx = x[::-1]
    corr = np.convolve(p, revx)  # length 3n-2
    offset = n - 1
    g = 2 * beta * corr[offset: offset + n]
    # Adam update if desired
    if m is None:
        # simple gradient descent
        y = x - eta * g
    else:
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        y = x - eta * m_hat / (np.sqrt(v_hat) + 1e-8)
    # project
    y = project_to_simplex(y)
    return y, m, v

"""
propose_candidate
=================

Search for a non‑negative step‑function (a list of heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2),

where ``conv`` is the ordinary discrete convolution.  The objective is
scale‑invariant, therefore we work with probability vectors
(x_i ≥ 0, Σ x_i = 1).

The algorithm proceeds in three stages:

1. **Generate a diverse pool of initial candidates** – uniform,
   power‑law families, random Dirichlet draws and (if available) the
   previously found best construction.  All candidates are normalised
   to the simplex and quickly evaluated.

2. **Smooth‑max optimisation** – we minimise the smooth approximation

        f_β(x) = (1/β)·log Σ_i exp(β·(x*x)_i)

   of the ℓ∞‑norm of the convolution.  The gradient of `f_β` is
   inexpensive to compute via a pair of convolutions, and we perform
   projected (mirror) gradient descent with a step size η = η₀/β.
   The smoothing parameter β is increased geometrically (β←β·γ),
   which drives the optimisation towards the true max‑norm.
   The best vector found during this phase is kept.

3. **Optional mirror‑descent polishing** – a few iterations of the
   sub‑gradient method used in the reference implementation are run
   to fine‑tune the final candidate.

The whole procedure respects a global time budget (default 1000 s) and
returns the best sequence discovered.
"""

import time

def project_to_simplex(v):
    """
    Euclidean projection of a vector onto the probability simplex:
        { w | w_i ≥ 0, Σ w_i = 1 }.
    Implements the algorithm of Duchi et al. (2008).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive → return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def upscale_sequence(seq, new_len):
    """
    Linearly interpolate an existing sequence to a longer length
    while preserving the total mass (the returned vector sums to 1).
    """
    old_len = len(seq)
    if old_len == new_len:
        return seq.copy()
    old_x = np.linspace(0, 1, old_len, dtype=np.float64)
    new_x = np.linspace(0, 1, new_len, dtype=np.float64)
    up = np.interp(new_x, old_x, seq)
    up = np.maximum(up, 0.0)
    up /= up.sum()
    return up


def _mirror_descent_step(x, eta):
    """
    One iteration of the sub‑gradient (mirror‑descent) step used in the
    reference code.  ``x`` must be a probability vector.
    """
    n = x.shape[0]
    conv = np.convolve(x, x)          # length 2n‑1
    k = int(conv.argmax())            # index of the maximal entry
    # sub‑gradient g_i = x_{k-i} (zero out‑of‑range entries)
    g = np.zeros_like(x)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    idx = np.arange(i_min, i_max + 1, dtype=int)
    jdx = k - idx
    g[idx] = x[jdx]                   # factor 2 omitted (absorbed in η)
    y = x * np.exp(-eta * g)
    if y.sum() == 0.0:
        y = np.abs(np.random.randn(n))
    y = project_to_simplex(y)
    return y


def _smooth_max_gradient_step(x, beta, eta):
    """
    One projected gradient step on the smooth max objective
    f_β(x) = (1/β)·log Σ_i exp(β·(x*x)_i).

    Returns the updated probability vector.
    """
    n = x.shape[0]

    # ----- compute soft‑max weights ----------
    conv_xx = np.convolve(x, x)               # length 2n‑1
    z = beta * conv_xx
    z_max = np.max(z)                         # for numerical stability
    w_unnorm = np.exp(z - z_max)
    w = w_unnorm / w_unnorm.sum()             # Σ w = 1

    # ----- gradient of f_β (see analysis) ----
    revx = x[::-1]
    corr = np.convolve(w, revx)               # length 3n‑2
    offset = n - 1
    grad = 2.0 * beta * corr[offset: offset + n]   # ∇f_β

    # ----- simple (scaled) gradient step -----
    y = x - eta * grad
    y = project_to_simplex(y)
    return y
