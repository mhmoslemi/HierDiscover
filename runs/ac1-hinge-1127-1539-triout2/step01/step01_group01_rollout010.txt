"""
Optimiser for the self‑convolution bound.

The algorithm proceeds in three stages:

1. **Extensive parametric sweep** – we generate many symmetric “U‑shaped’’
   families (linear, quadratic, cubic, quartic, exponential, cosine)
   with a wide range of strength parameters.  The best candidate from this
   sweep is used as a starting point.

2. **Projected sub‑gradient descent with line‑search and momentum** – a
   sub‑gradient of the max‑convolution is computed on the currently
   active indices (those attaining the maximum, optionally widened by a
   tolerance).  The descent direction is made orthogonal to the all‑ones
   vector so that the ℓ₁‑norm (the total sum) stays fixed; after each step
   we clip negatives and project back onto the simplex of the required
   sum.  A simple back‑tracking line‑search over a few step‑size factors
   yields larger improvements per iteration.

3. **Stochastic mass‑transfer annealing** – occasional random transfers
   of a small fraction of mass between two positions are performed.
   The new configuration is accepted if it improves the objective or,
   with a temperature‑dependent probability, even if it is worse.
   This helps to escape plateaus.

The whole search respects the time budget (≈1000 s) and always returns
the best sequence discovered.  All numbers are non‑negative, the ℓ₁‑norm
is kept equal to `sqrt(2 n)`, and the final list is returned as plain
Python floats.
"""

import time
import numpy as np

def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using an FFT of power‑of‑two length."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())   # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]

def compute_max_conv(a: np.ndarray) -> float:
    """Return the maximal entry of the (linear) self‑convolution of a."""
    return float(conv_self_fft(a).max())

def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector to have ℓ₁‑norm equal to *target*."""
    s = v.sum()
    if s > 0.0:
        return v * (target / s)
    # fallback – uniform distribution
    return np.full_like(v, target / v.shape[0])

def generate_U_shape(n: int, target_sum: float,
                    kind: str = "linear",
                    param: float = 0.0,
                    power: int = 1) -> np.ndarray:
    """Symmetric “U‑shaped’’ sequence of length *n*.
       *kind* can be linear, quadratic, cubic, quartic, exponential or cosine.
       For the polynomial families the exponent is given by *power*."""
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid          # normalised distance in [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * (d ** 2)
    elif kind == "cubic":
        raw = 1.0 + param * (d ** 3)
    elif kind == "quartic":
        raw = 1.0 + param * (d ** 4)
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "cosine":
        raw = 1.0 + param * (1.0 - np.cos(np.pi * d))
    else:
        raise ValueError(f"unknown shape kind {kind}")

    raw = np.clip(raw, 0.0, None)
    return project_to_sum(raw, target_sum)

def subgradient_step_line(a: np.ndarray,
                          target_sum: float,
                          eta: float,
                          rel_tol: float = 0.0) -> tuple[np.ndarray, float]:
    """
    One projected sub‑gradient step with a tiny back‑tracking line‑search.
    Returns the new vector and its max‑convolution value.
    """
    n = a.shape[0]
    conv = conv_self_fft(a)
    max_val = conv.max()
    if rel_tol <= 0.0:
        act = np.array([np.argmax(conv)], dtype=np.int64)
    else:
        act = np.where(conv >= max_val * (1.0 - rel_tol))[0]

    # sub‑gradient for the active set
    g = np.zeros_like(a)
    idx = np.arange(n, dtype=np.int64)
    for k in act:
        j = k - idx
        mask = (j >= 0) & (j < n)
        g[mask] += 2.0 * a[j[mask]]

    # direction orthogonal to the all‑ones vector (preserve ℓ₁‑norm)
    direction = -g + np.mean(g)

    # try a few step sizes and keep the best candidate
    best_cand = a
    best_score = compute_max_conv(a)
    for factor in (1.0, 0.5, 0.25, 0.125):
        cand = a + eta * factor * direction
        cand = np.clip(cand, 0.0, None)
        cand = project_to_sum(cand, target_sum)
        sc = compute_max_conv(cand)
        if sc < best_score:
            best_score = sc
            best_cand = cand
    return best_cand, best_score

def random_pair_transfer(a: np.ndarray,
                         target_sum: float,
                         scale: float = 0.01) -> np.ndarray:
    """
    Transfer a small fraction *scale* of the mass from a random index to
    another one, then re‑project onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0.0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_to_sum(a, target_sum)

"""
Optimiser for the self‑convolution bound.

The algorithm proceeds in three stages:

1. **Extensive parametric sweep** – many symmetric “U‑shaped’’ families
   (linear, quadratic, cubic, quartic, exponential, cosine) are generated
   with a wide range of strength parameters.  The best candidate from this
   sweep seeds the subsequent optimisation.

2. **Projected sub‑gradient descent with line‑search** – for the current
   vector the indices where the self‑convolution attains its maximum
   (optionally widened by a tolerance) are identified.  The sub‑gradient
   of the max‑convolution is assembled, made orthogonal to the all‑ones
   vector (so the ℓ₁‑norm stays constant), and a step is taken.  A tiny
   back‑tracking line‑search over a few step‑size factors selects the most
   beneficial move.  After every step negatives are clipped and the vector
   is projected back onto the simplex of the required sum.

3. **Stochastic mass‑transfer annealing** – occasional random transfers
   of a small fraction of mass between two positions are performed.
   The new configuration is accepted if it improves the objective or,
   with a temperature‑dependent probability, even if it is worse.
   This helps to escape plateaus.

All numbers remain non‑negative and the sum is kept equal to
`sqrt(2 n)`, which is exactly the normalisation used by `evaluate_sequence`.
The routine respects the 1000 s time budget and always returns the best
sequence found.
"""

import time
import numpy as np


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a power‑of‑two FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())   # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def compute_max_conv(a: np.ndarray) -> float:
    """Return the maximal entry of the self‑convolution of *a*."""
    return float(conv_self_fft(a).max())


def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector to have ℓ₁‑norm *target*."""
    s = v.sum()
    if s > 0.0:
        return v * (target / s)
    # fallback – uniform distribution
    return np.full_like(v, target / v.shape[0])


def generate_U_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
) -> np.ndarray:
    """
    Produce a symmetric “U‑shaped’’ sequence of length *n*.
    Supported *kind*: linear, quadratic, cubic, quartic, exponential, cosine.
    ``param`` controls the strength of the edge‑enhancement (≥0).
    The returned vector is projected onto the ℓ₁‑simplex of size *target_sum*.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "cubic":
        raw = 1.0 + param * d ** 3
    elif kind == "quartic":
        raw = 1.0 + param * d ** 4
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "cosine":
        raw = 1.0 + param * (1.0 - np.cos(np.pi * d))
    else:
        raise ValueError(f"unknown shape kind {kind}")

    raw = np.clip(raw, 0.0, None)
    return project_to_sum(raw, target_sum)


def subgradient_step_line(
    a: np.ndarray,
    target_sum: float,
    eta: float,
    rel_tol: float = 0.0,
) -> tuple[np.ndarray, float]:
    """
    One projected sub‑gradient step with a tiny line‑search.
    Returns the new vector and its maximal self‑convolution value.
    """
    n = a.shape[0]
    conv = conv_self_fft(a)
    max_val = conv.max()

    if rel_tol <= 0.0:
        act = np.array([np.argmax(conv)], dtype=np.int64)
    else:
        act = np.where(conv >= max_val * (1.0 - rel_tol))[0]

    # sub‑gradient of the max‑convolution for the active set
    g = np.zeros_like(a)
    idx = np.arange(n, dtype=np.int64)
    for k in act:
        j = k - idx
        mask = (j >= 0) & (j < n)
        g[mask] += 2.0 * a[j[mask]]

    # make the direction orthogonal to the all‑ones vector (preserve sum)
    direction = -g + np.mean(g)

    # try a few step‑size factors and keep the best result
    best_cand = a
    best_score = compute_max_conv(a)
    for factor in (1.0, 0.5, 0.25, 0.125):
        cand = a + eta * factor * direction
        cand = np.clip(cand, 0.0, None)
        cand = project_to_sum(cand, target_sum)
        sc = compute_max_conv(cand)
        if sc < best_score:
            best_score = sc
            best_cand = cand
    return best_cand, best_score


def random_pair_transfer(
    a: np.ndarray,
    target_sum: float,
    scale: float = 0.01,
) -> np.ndarray:
    """
    Transfer a small amount of mass from one random index to another,
    then re‑project onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0.0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_to_sum(a, target_sum)
