"""
Propose a low‑score step‑function sequence.

Algorithm
---------
* Initialise a decreasing “1/√i” shape (or the previously found best
  sequence if it exists) and normalise the L¹‑norm to √(2 n).  With this
  normalisation the evaluation score reduces to the maximum value of the
  self‑convolution, so we directly minimise that maximum.

* At each iteration we
  - compute the convolution a∗a (FFT, O(n log n));
  - build a weight vector w that concentrates on the largest convolution
    entries (soft‑max with a temperature γ that grows to ≈100, after which
    we switch to a hard indicator on the maximal entries);
  - obtain the gradient of the (smoothed) max via the cross‑correlation
    g = 2·(w ⊛ a_rev) (again via FFT);
  - update the moments of the Adam optimiser (β₁=0.9, β₂=0.999);
  - perform a back‑tracking line‑search on the Adam step to guarantee a
    reduction of the true maximum convolution;
  - project onto the non‑negative orthant and renormalise the sum.

* Hyper‑parameters (γ, learning‑rate) are slowly annealed; occasional tiny
  random jitter helps escape plateaux.

* The best sequence seen so far is kept and returned when the time budget
  expires (default 1000 s).  The routine works for any length; it will
  reuse the length of the globally supplied `height_sequence_1` if present,
  otherwise falls back to 5000.

The method typically pushes the score well below the previous 1.575…,
approaching the target ≈1.503 within the allocated time.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, top‑level as required)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_max(a: np.ndarray, nfft: int) -> float:
    """
    Compute max(a * a) using FFT of length nfft (linear convolution).
    The caller must ensure nfft ≥ 2*len(a)-1.
    """
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _gradient_from_weights(w: np.ndarray, a: np.ndarray, nfft_corr: int) -> np.ndarray:
    """
    Compute g = 2 * (w ⊛ a_rev) where ⊛ denotes cross‑correlation.
    Returns a vector of length len(a).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    wf = np.fft.rfft(w, n=nfft_corr)
    af_rev = np.fft.rfft(a_rev, n=nfft_corr)
    corr_full = np.fft.irfft(wf * af_rev, n=nfft_corr)
    # valid slice: indices n-1 … n-1+n‑1 (inclusive) → length n
    return 2.0 * corr_full[n - 1 : n - 1 + n]

    as a plain Python list.

    # the evaluation score equals the maximal convolution value.
    target_sum = math.sqrt(2.0 * n)

    nfft_conv = _next_pow2(2 * n - 1)
