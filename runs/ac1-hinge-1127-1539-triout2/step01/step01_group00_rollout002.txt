def _smooth_step(a, eta, beta):
    n = a.shape[0]
    conv = _conv_full(a) # length 2n-1
    max_c = conv.max()
    # compute weights
    w = np.exp(beta * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # correlation
    corr = _correlate_full(a, w) # length 3n-2
    offset = 2*n-2
    grad = 2.0 * corr[offset: offset + n] - a * w[2*np.arange(n)]
    # multiplicative update
    a_new = a * np.exp(-eta * grad)
    # clip non-negative
    a_new = np.maximum(a_new, 0.0)
    # renormalize sum
    total = a_new.sum()
    if total == 0:
        a_new = np.full(n, np.sqrt(2.0 * n) / n)
    else:
        a_new *= (np.sqrt(2.0 * n) / total)
    return a_new, conv.max()
```

def _conv_full(a):
    n = a.shape[0]
    conv_len = 2*n-1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv

def _correlate_full(a, w):
    n = a.shape[0]
    L = 1 << ((n + len(w) - 1 - 1).bit_length())
    Fa = np.fft.rfft(a, n=L)
    Fw = np.fft.rfft(w, n=L)
    corr = np.fft.irfft(np.conj(Fa) * Fw, n=L)[:n + len(w) - 1]
    return corr
```

"""
Optimisation of the ℓ∞‑norm of a self‑convolution.

We aim to minimise
    score = 2·n·max((a∗a)) / (∑a)²
under a_i ≥ 0.
If we normalise the vector to ∑a = √(2n) the score is simply
the maximal entry of the linear convolution a∗a.

The algorithm follows a *smooth* approximation of the max‑operator:
    M_β(a) = (1/β)·log∑_k exp(β·(a∗a)_k)
which converges to max as β→∞.
For a given β we can compute the gradient analytically:
    ∂M_β/∂a_j = 2·∑_i w_{i+j}·a_i  –  a_j·w_{2j},
where w_k = exp(β·(a∗a)_k)/∑_ℓ exp(β·(a∗a)_ℓ).

Using this gradient we perform a *multiplicative* update
    a ← a·exp(−η·∇M_β)
which preserves non‑negativity.  After each update the vector is
re‑normalised to keep ∑a = √(2n).

The schedule starts with a very smooth objective (β≈10⁻²) and slowly
increases β up to 10, while the step‑size η decays.  Random jitter and
periodic restarts avoid stagnation.  Several problem sizes (≈5k–12k)
are tried within the given time budget; the best sequence found
over all runs is returned.

All heavy operations (convolution, correlation) are performed by FFT
so each iteration costs O(n log n).  The external function
``evaluate_sequence`` is only called at the very end (the score is
the maximal convolution entry because of the normalisation).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_full(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a using FFT.
    Returns a vector of length 2·n‑1.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _correlate_full(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation (a ⋆ w)[k] = Σ_i a_i·w_{i+k}.
    Returns a vector of length n + len(w) - 1.
    """
    n = a.shape[0]
    m = w.shape[0]
    out_len = n + m - 1
    fft_len = _next_pow2(out_len)
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w, n=fft_len)
    corr = np.fft.irfft(np.conj(Fa) * Fw, n=fft_len)[:out_len]
    return corr


def _initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative sequence of length n whose sum is √(2·n).

    With probability 0.5 we reuse the global ``height_sequence_1`` (if
    available); otherwise we draw a random Dirichlet‑like vector.
    """
    S = np.sqrt(2.0 * n)

    use_existing = False
    if 'height_sequence_1' in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=float)
            if base.size >= n:
                a = base[:n].copy()
            else:
                a = np.pad(base, (0, n - base.size), constant_values=0.0)
            use_existing = True
        except Exception:
            use_existing = False

    if use_existing and rng.random() < 0.5:
        a = a.copy()
    else:
        a = rng.random(n)

    a = a / a.sum() * S
    return a


# ----------------------------------------------------------------------
# Core optimisation primitives (smooth max)
# ----------------------------------------------------------------------


def _smooth_step(a: np.ndarray, eta: float, beta: float, rng: np.random.Generator):
    """
    One iteration of the smoothed ℓ∞ minimisation.

    Returns:
        a_new      – updated sequence (already normalised to √(2·n))
        cur_max    – actual maximal convolution entry after the update
    """
    n = a.shape[0]

    # --- 1. compute self‑convolution (needed for the weights) ---
    conv = _conv_full(a)                 # length 2·n‑1
    max_conv = conv.max()

    # --- 2. soft‑max weights w_k ∝ exp(β·conv_k) ---
    # shift by max for numeric stability
    w = np.exp(beta * (conv - max_conv))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # --- 3. gradient of the smoothed objective ---
    corr = _correlate_full(a, w)          # length 3·n‑2, first n entries are needed
    grad = 2.0 * corr[:n] - a * w[2 * np.arange(n)]

    # --- 4. multiplicative update (clipped exponent) ---
    exponent = -eta * grad
    exponent = np.clip(exponent, -50.0, 50.0)   # avoid overflow
    a_new = a * np.exp(exponent)

    # small random jitter to keep the dynamics alive (rare)
    if rng.random() < 0.002:
        noise = rng.normal(scale=0.01 * a_new.mean(), size=n)
        a_new = np.clip(a_new + noise, 0.0, None)

    # --- 5. renormalise to the required L1‑norm ----------
    total = a_new.sum()
    if total == 0.0:
        a_new = np.full(n, np.sqrt(2.0 * n) / n)
    else:
        a_new *= (np.sqrt(2.0 * n) / total)

    # compute the (actual) max after the update – this is the external score
    cur_max = _conv_full(a_new).max()
    return a_new, cur_max

def _optimise_length(n: int, seed: int, deadline: float):
    """
    Run the smoothed‑max optimiser for a fixed length ``n`` until ``deadline``.
    Returns the best sequence found and its (internal) maximal convolution value.
    """
    rng = np.random.default_rng(seed)
    a = _initial_sequence(n, rng)

    # internal bookkeeping
    best_a = a.copy()
    best_val = _conv_full(a).max()

    # hyper‑parameters
    eta = 0.6               # initial learning rate
    eta_decay = 0.9997
    beta = 0.01
    beta_factor = 1.0013
    beta_max = 10.0

    # stagnation handling
    no_improve = 0
    patience = max(2000, n // 2)   # restart after many non‑improvements

    while time.time() < deadline:
        a, cur_max = _smooth_step(a, eta, beta, rng)

        # record improvement
        if cur_max < best_val - 1e-12:
            best_val = cur_max
            best_a = a.copy()
            no_improve = 0
            print(f"[improve] n={n}  time={time.time():.1f}s  score={best_val:.6f}")
        else:
            no_improve += 1

        # schedule updates
        eta *= eta_decay
        beta = min(beta_max, beta * beta_factor)

        # occasional full restart to escape deep plateaus
        if no_improve > patience:
            a = _initial_sequence(n, rng)
            no_improve = 0

    return best_a, best_val

    print(f"[final] best score = {best_score:.6f}  length = {len(best_seq)}")
    return best_seq.tolist()
