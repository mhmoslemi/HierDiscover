from scipy.optimize import linprog

def solve_lp(f, M_val):
    n = len(f)
    # Build convolution matrix
    # using Toeplitz
    # We'll allocate 2n-1 x n
    A = np.zeros((2*n-1, n))
    for i in range(n):
        A[i:i+n, i] = f
    # So conv = A @ b
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M_val), bounds=[(0, None)]*n, method='highs', options={'presolve': True})
    if not res.success:
        # fallback maybe quick solution
        return None
    return res.x

"""
Propose a low‑score step function.

Algorithm
---------
1. **Initial candidate** – a short grid search over the family
   a_i = 1/(i+1+c)^β (β≈0.5, c≥0) for a moderate length `n`
   (default 500).  The best grid point is kept as a starting vector.

2. **Linear‑programming improvement** – following the idea from the
   paper, we repeatedly solve a linear program

        maximize   Σ b_j
        subject to b_j ≥ 0
                   (f * b)_k ≤ M   for every convolution index k,

   where `f` is the current vector and `M = max(f*f)`.  The LP
   increases the ℓ₁‑mass while keeping the maximal auto‑convolution
   bounded.  The solution `b` is rescaled to the original ℓ₁‑norm,
   mixed with `f` (mixing factor `t`), and the process repeats.
   For `n≈500` the LP has ≈1000 variables and ≤1000 constraints and
   solves in a fraction of a second with SciPy’s HiGHS solver.

3. **Fine‑grained gradient descent** – after the LP phase we polish
   the result with a simple sub‑gradient scheme (the same as the
   reference code) that directly reduces the maximal convolution
   entry while re‑normalising the sum.

All steps respect the security checks of `evaluate_sequence`; the
returned list consists of non‑negative floats and the whole search
fits comfortably into the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast evaluation of the objective (same idea as the reference)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Return the evaluation score (2·n·max(conv(seq,seq)) / sum(seq)²)."""
    n = len(seq)
    s = np.sum(seq)
    if s < 0.01:
        return np.inf

    # next power of two for a fast FFT‑based convolution
    m = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft * fft, n=m)[: 2 * n - 1]
    max_conv = np.max(conv)
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Linear program: maximise ℓ₁‑norm of b under the convolution bound.
# Uses SciPy's HiGHS (fallback to CVXPY if SciPy is unavailable).
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog

    def _solve_lp(f: np.ndarray, M: float) -> np.ndarray | None:
        """Solve  max Σ b  s.t. b≥0 and (f*b)_k ≤ M for all k."""
        n = len(f)
        # Build the convolution matrix A (shape (2n‑1, n)):
        #   (f*b)[k] = Σ_i f[i] * b[k-i]  (zero outside range)
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for i in range(n):
            A[i : i + n, i] = f
        c = -np.ones(n, dtype=np.float64)            # maximise Σb ⇔ minimise –Σb
        ub = np.full(2 * n - 1, M, dtype=np.float64)  # upper bound for each conv entry
        bounds = [(0.0, None)] * n
        res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            return None
        return res.x

except Exception:  # pragma: no cover
    # SciPy not available – fall back to CVXPY (very small problems only)
    import cvxpy as cp

    def _solve_lp(f: np.ndarray, M: float) -> np.ndarray | None:
        n = len(f)
        b = cp.Variable(n, nonneg=True)
        constraints = []
        # build the same convolution matrix via explicit sums (fast enough for tiny n)
        for k in range(2 * n - 1):
            expr = cp.sum(cp.multiply(f, cp.hstack([b[k - i] if 0 <= k - i < n else 0
                                                    for i in range(n)])))
            constraints.append(expr <= M)
        prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
        prob.solve(solver=cp.ECOS, warm_start=True)
        if prob.status not in ("optimal", "optimal_inaccurate"):
            return None
        return np.asarray(b.value, dtype=np.float64)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    evaluate_sequence().  Returns a plain list[float].
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0  # safety margin

    # --------------------------------------------------------------
    # 1. Choose a modest length – large enough to exhibit the 1/√
    #    behaviour but small enough for the LP (default 500).
    # --------------------------------------------------------------
    n = 500
    print(f"[+] Starting search, length = {n}")

    # --------------------------------------------------------------
    # 2. Grid search over the classic family a_i = 1/(i+1+c)^β.
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    betas = np.linspace(0.44, 0.56, 25)      # a little broader than before
    offsets = np.linspace(0.0, 10.0, 21)    # up to 10 to explore edge effects

    for beta in betas:
        for c in offsets:
            idx = np.arange(n, dtype=np.float64) + 1.0 + c
            cand = 1.0 / (idx ** beta)
            score = _objective(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"[grid] β={beta:.3f}, c={c:5.2f} → score={score:.6f}")
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    # fallback – should never happen
    if best_seq is None:
        best_seq = np.ones(n, dtype=np.float64)
        best_score = _objective(best_seq)

    # --------------------------------------------------------------
    # 3. Linear‑programming refinement (alternating LP, as described
    #    in the reference approach).
    # --------------------------------------------------------------
    f = best_seq.copy()
    sum_f = np.sum(f)            # we keep this ℓ₁‑norm fixed
    t_mix = 0.25                 # initial mixing proportion
    iter_lp = 0

    while time.time() < deadline and iter_lp < 30:
        # current maximal auto‑convolution value
        conv_f = np.convolve(f, f)
        M = np.max(conv_f)

        # solve the LP: keep the current bound M, maximise the ℓ₁‑mass
        b = _solve_lp(f, M)
        if b is None:
            print("[LP] solver failed – terminating LP phase.")
            break

        sum_b = np.sum(b)
        # If the LP cannot increase the ℓ₁‑mass we stop the LP loop.
        if sum_b <= sum_f * (1.0 + 1e-9):
            print(f"[LP] no improvement after iteration {iter_lp}.")
            break

        # rescale b to have the same ℓ₁‑norm as the current vector
        g = b * (sum_f / sum_b)

        # convex combination and re‑normalisation (protect against tiny drift)
        f_new = (1.0 - t_mix) * f + t_mix * g
        f_new = f_new * (sum_f / np.sum(f_new))

        new_score = _objective(f_new)
        print(f"[LP] iter={iter_lp:2d}, t={t_mix:.3f}, score={new_score:.6f}")

        if new_score < best_score:
            best_score = new_score
            best_seq = f_new.copy()
            print(f"      ** new best (LP) {best_score:.6f}")

        f = f_new
        iter_lp += 1
        # slowly decay the mixing factor – large steps at the beginning,
        # fine adjustments later.
        t_mix = max(0.02, t_mix * 0.85)

        if time.time() > deadline:
            break

    # --------------------------------------------------------------
    # 4. Sub‑gradient gradient descent (identical to the reference
    #    implementation) for a final polish.
    # --------------------------------------------------------------
    a = best_seq.copy()
    target_sum = np.sum(a)
    step_factor = 0.4
    no_improve = 0
    iter_gd = 0
    best_global_seq = a.copy()
    best_global_score = best_score

    while time.time() < deadline:
        # full convolution – needed to locate the active index
        m = 1 << ((2 * n - 1).bit_length())
        fft_a = np.fft.rfft(a, n=m)
        conv = np.fft.irfft(fft_a * fft_a, n=m)[: 2 * n - 1]
        max_conv = np.max(conv)
        k_max = int(np.argmax(conv))

        # sub‑gradient for that active shift
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        shift = np.zeros_like(a)
        shift[mask] = a[idx[mask]]
        grad = a + shift                         # g_i = a_i + a_{k_max-i}

        # step size – proportional to the current ratio r = max_conv / sum²
        r = max_conv / (np.sum(a) ** 2)
        step = step_factor * r

        new_a = a - step * grad
        new_a = np.maximum(new_a, 0.0)

        # renormalise the ℓ₁‑norm (size does not affect the score)
        s = np.sum(new_a)
        if s > 1e-12:
            new_a *= target_sum / s
        else:
            break

        new_score = _objective(new_a)

        if new_score < best_global_score:
            best_global_score = new_score
            best_global_seq = new_a.copy()
            a = new_a
            no_improve = 0
            step_factor = min(step_factor * 1.05, 2.0)
            if iter_gd % 100 == 0:
                print(f"[grad] iter={iter_gd:4d}, score={new_score:.6f}")
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                # occasional random jitter to escape plateaus
                i, j = np.random.randint(n, size=2)
                delta = np.random.rand() * a[i] * 0.07
                a[i] = max(0.0, a[i] - delta)
                a[j] += delta
                a *= target_sum / np.sum(a)
                step_factor *= 0.9
        iter_gd += 1

        if best_global_score < 1.503:
            print(f"[!] Target beaten early: score={best_global_score:.6f}")
            break

        if time.time() > deadline:
            break

    print(f"[+] Search finished. Best score = {best_global_score:.6f}")
    return [float(x) for x in best_global_seq]
