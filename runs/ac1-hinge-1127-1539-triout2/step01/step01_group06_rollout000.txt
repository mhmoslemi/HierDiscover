"""
Hybrid optimiser for the step‑function convolution bound.

The objective after normalising the ℓ¹‑norm to sqrt(2·n) is simply
    max(conv(a, a)).
We first perform a cheap stochastic hill‑climbing (mass‑transfer)
starting from a power‑law profile (or a supplied “height_sequence_1”).
After a short warm‑up we improve the current shape by solving a linear
programme:

    maximise   Σ g_i
    subject to g_i ≥ 0
               conv(f, g) ≤ M   (M = max(conv(f, f)))

The LP yields a vector g with the same convolution bound but maximal
ℓ¹‑mass; we rescale it to the required ℓ¹‑norm and move the current
solution a slightly towards g.  Repeating this step (with a decreasing
step size) rapidly drives the maximum convolution down.  Occasional
short local‑search refinements are applied after each LP step.

The whole routine stops when the allotted wall‑time expires and returns
the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional linear‑programming support (SciPy)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    _LP_AVAILABLE = True
except Exception:               # pragma: no cover
    _LP_AVAILABLE = False

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray) -> float:
    """Maximum value of the linear convolution a * a (real convolution)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """Exact objective used by evaluate_sequence (fast version)."""
    n = a.size
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:                      # matches safety check of evaluate_sequence
        return np.inf
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Initial profile generator
# ----------------------------------------------------------------------
def _initial_sequence(p: float, n: int) -> np.ndarray:
    """
    Power‑law start a_i ∝ (i+1)^{-p}, then normalised to ℓ¹‑norm √(2·n).
    """
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= np.sqrt(2.0 * n) / a.sum()
    return a


# ----------------------------------------------------------------------
# Simple hill‑climbing that preserves the total sum
# ----------------------------------------------------------------------
def _local_search(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Mass‑transfer hill‑climbing, keeping Σa constant."""
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    cur_val = best_val

    max_frac = 0.12               # at most 12 % of source mass moved
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        frac = rng.random() * max_frac
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        new_val = _score(a)

        if new_val < cur_val:               # strict improvement
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
        else:
            # revert
            a[j] -= delta
            a[i] += delta

    return best_a, best_val


# ----------------------------------------------------------------------
# Build the convolution matrix for the LP: rows = (2n‑1), cols = n
# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return dense matrix A such that A @ g = conv(f, g)."""
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # column i consists of f shifted by i positions
    for i in range(n):
        A[i:i + n, i] = f
    return A


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises evaluate_sequence().
    The routine runs for at most ``budget_s`` seconds and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2   # tiny safety margin

    # ------------------------------------------------------------------
    # Choose a working length.  If a previously discovered sequence is
    # available we adopt its length; otherwise we fall back to 2000.
    # ------------------------------------------------------------------
    global height_sequence_1          # may be set by the harness
    if 'height_sequence_1' in globals() and isinstance(height_sequence_1, list):
        init_seq = np.array(height_sequence_1, dtype=np.float64)
        n = init_seq.size
        # Ensure the normalisation (the score is homogeneous, so we can
        # safely re‑scale to √(2·n)).
        init_seq *= np.sqrt(2.0 * n) / init_seq.sum()
        best_a = init_seq.copy()
        best_score = _score(best_a)
        print(f"[init] Using supplied sequence (n={n}) score={best_score:.6f}")
    else:
        # No supplied sequence – start from a power law.
        n = 2000
        # a few exponents to try; keep the best.
        exponents = [0.45, 0.48, 0.5, 0.52, 0.55]
        best_a = None
        best_score = np.inf
        for p in exponents:
            if time.time() >= deadline:
                break
            a0 = _initial_sequence(p, n)
            a_opt, val = _local_search(a0.copy(), time.time() + 2.5)
            if val < best_score:
                best_score = val
                best_a = a_opt.copy()
                print(f"[init] p={p:.2f} → score={val:.6f}")

        # sanity fallback
        if best_a is None:
            best_a = _initial_sequence(0.5, n)
            best_score = _score(best_a)

    # ------------------------------------------------------------------
    # Main optimisation loop (LP‑guided refinement + occasional local search)
    # ------------------------------------------------------------------
    t = 0.20                     # initial mixing coefficient
    iteration = 0

    while True:
        now = time.time()
        if now + 1.0 >= deadline:          # guarantee we have time for the LP
            break

        iteration += 1
        f = best_a
        n = f.size
        M = _max_convolution(f)            # current max convolution

        # ------------------------------------------------------------------
        # Linear programme: maximise Σ g_i  subject to  g ≥ 0 ,  conv(f,g) ≤ M
        # ------------------------------------------------------------------
        if not _LP_AVAILABLE:
            # If SciPy is not present we skip the LP part and go straight to
            # a short local search.
            print("[lp] SciPy not available – skipping LP step.")
            a_ref, ref_score = _local_search(best_a.copy(), now + 2.0)
            if ref_score < best_score:
                best_score, best_a = ref_score, a_ref.copy()
                print(f"[refine] Improved to {best_score:.6f} (no LP)")
            continue

        A = _build_conv_matrix(f)          # (2n‑1) × n dense matrix
        rows = A.shape[0]

        c = -np.ones(n)                    # maximise Σ g_i  -> minimise -Σ g_i
        b_ub = np.full(rows, M, dtype=np.float64)
        bounds = [(0.0, None)] * n

        # Solve with HiGHS (fast interior‑point method)
        try:
            lp_res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                             method='highs', options={'presolve': True})
        except Exception as e:              # pragma: no cover
            print(f"[lp] Exception: {e}")
            break

        if not lp_res.success:
            print(f"[lp] Failed (iter {iteration}): {lp_res.message}")
            break

        g0 = lp_res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0.0:
            print("[lp] Zero solution – aborting.")
            break

        # Rescale g0 to the required ℓ¹‑norm √(2·n)
        target_L1 = np.sqrt(2.0 * n)
        g = g0 * (target_L1 / sum_g0)

        # Convex combination of f and g
        a_new = (1.0 - t) * f + t * g
        a_new = np.maximum(a_new, 0.0)          # numerical safety
        new_score = _score(a_new)

        print(f"[iter {iteration:02d}] t={t:.3f}  score={new_score:.6f}")

        # Keep the best ever seen
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            print(f"  *** new global best: {best_score:.6f}")

        # Reduce the mixing proportion gradually (helps convergence)
        t = max(0.05, t * 0.93)

        # --------------------------------------------------------------
        # Short local‑search refinement after each LP step (if time permits)
        # --------------------------------------------------------------
        if time.time() + 2.0 < deadline:
            a_ref, ref_score = _local_search(best_a.copy(),
                                             time.time() + 2.0)
            if ref_score < best_score:
                best_score, best_a = ref_score, a_ref.copy()
                print(f"[refine] post‑LP improvement → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries and return a plain list.
    # ------------------------------------------------------------------
    nz = np.where(best_a > 1e-12)[0]
    if nz.size:
        best_a = best_a[: nz[-1] + 1]
    else:
        best_a = best_a[:1]   # at least one element

    return best_a.tolist()
