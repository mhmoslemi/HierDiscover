def conv_self_fft(a):
    n = a.shape[0]
    m = 2*n - 1
    L = 1 << ((m).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[:m]

def conv_fft(a,b):
    m = a.shape[0] + b.shape[0] - 1

    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)

"""
A mirror‑descent / Adam optimiser for the self‑convolution bound.

The idea is to work in the simplex of non‑negative vectors with a fixed
ℓ₁‑norm (the normalisation used by `evaluate_sequence`).  A vector `a`
is represented as `a = S * softmax(l)` where `S = sqrt(2·n)` and `l`
are free logits.  The loss is a smooth approximation of the maximum
convolution entry:
        L(l) = (1/α)·log( Σ_k exp(α·(a*a)_k) )
which tends to the true objective `max_k (a*a)_k` as `α→∞`.  Its
gradient can be written in closed form and evaluated efficiently with
FFT‑based convolutions.  Adam is used to update the logits, while the
logits are kept symmetric (mirrored) after each step – the optimal
solutions are known to be (approximately) palindromic.  After the main
mirror‑descent phase a short projected sub‑gradient descent (as in the
reference implementation) is run to polish the result.

The optimiser respects the 1000 s time budget and always returns the
best sequence found.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution of a with itself (full length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    L = next_pow2(m)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return np.real(conv[:m])


def conv_fft(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Full linear convolution of u and v using FFT."""
    m = u.shape[0] + v.shape[0] - 1
    L = next_pow2(m)
    fu = np.fft.rfft(u, n=L)
    fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fu * fv, n=L)
    return np.real(conv[:m])


def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector to have ℓ₁‑norm = target."""
    s = v.sum()
    if s > 0.0:
        return v * (target / s)
    # fallback – uniform
    return np.full_like(v, target / v.shape[0])


def generate_U_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
) -> np.ndarray:
    """
    Produce a symmetric “U‑shaped’’ sequence.
    kind ∈ {'linear', 'quadratic', 'exponential'}.
    param ≥ 0 controls the edge‑enhancement strength.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid  # normalised distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_to_sum(raw, target_sum)


def subgradient_step(
    a: np.ndarray,
    target_sum: float,
    eta: float,
    rel_tol: float = 0.0,
) -> np.ndarray:
    """
    One projected sub‑gradient step reducing max self‑convolution.
    The active set contains indices whose convolution value is at least
    (1‑rel_tol)·max.
    """
    n = a.shape[0]
    conv = conv_self_fft(a)
    max_val = conv.max()
    if rel_tol <= 0.0:
        act = np.array([np.argmax(conv)], dtype=np.int64)
    else:
        act = np.where(conv >= max_val * (1.0 - rel_tol))[0]

    g = np.zeros_like(a)
    idx = np.arange(n, dtype=np.int64)
    for k in act:
        j = k - idx
        mask = (j >= 0) & (j < n)
        # contribution 2·a_{k‑i}
        g[mask] += 2.0 * a[j[mask]]

    # direction orthogonal to the all‑ones vector (preserve sum)
    direction = -g + np.mean(g)
    a_new = a + eta * direction
    a_new = np.clip(a_new, 0.0, None)
    a_new = project_to_sum(a_new, target_sum)
    return a_new


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a tiny fraction of mass from one random index to another.
    The total ℓ₁‑norm is preserved.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0.0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_to_sum(a, target_sum)

# Main optimiser entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    `evaluate_sequence`.  The function runs for at most *budget_s* seconds
    and returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Determine problem size
    # ------------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        init_seq = np.asarray(base, dtype=np.float64)
    else:
        # a large but manageable size
        n = np.random.randint(7500, 9000)
        init_seq = None

    target_sum = np.sqrt(2.0 * n)   # the normalisation used by the benchmark

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep (optional)
    # ------------------------------------------------------------------
    best_seq = np.full(n, target_sum / n)
    best_score = float("inf")

    # try a few simple U‑shapes; this is fast and gives a decent start
    kinds = ("linear", "quadratic", "exponential")
    grids = {
        "linear": np.linspace(0.0, 4.0, 21),
        "quadratic": np.linspace(0.0, 3.0, 21),
        "exponential": np.linspace(0.0, 2.0, 21),
    }
    for kind in kinds:
        for p in grids[kind]:
            cand = generate_U_shape(n, target_sum, kind=kind, param=p)
            score = np.max(conv_self_fft(cand))   # equals evaluate_sequence (sum normalised)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
        if time.time() >= deadline:
            return best_seq.tolist()

    # If a previous construction is available, use it as a stronger start
    if init_seq is not None:
        init_seq = project_to_sum(init_seq, target_sum)
        # evaluate it
        init_score = np.max(conv_self_fft(init_seq))
        if init_score < best_score:
            best_score = init_score
            best_seq = init_seq.copy()

    # ------------------------------------------------------------------
    # Phase 1 – mirror‑descent / Adam on a smooth max‑approximation
    # ------------------------------------------------------------------
    # Represent the vector via logits: a = S * softmax(logits)
    eps = 1e-12
    p = best_seq / target_sum
    logits = np.log(p + eps)

    # enforce symmetry (palindromic) already
    logits = (logits + logits[::-1]) / 2.0

    # Adam state
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.15                         # learning rate for logits
    iter_no = 0

    # schedule for the smooth‑max parameter α
    alpha = 1.0
    alpha_max = 200.0
    alpha_inc = (alpha_max - alpha) / 8000.0   # reach α_max after ~8000 iters

    print(f"[{time.time():.2f}] Starting mirror‑descent, n={n}")

    while time.time() < deadline:
        # --- compute current a -------------------------------------------------
        max_log = np.max(logits)
        exp_logits = np.exp(logits - max_log)
        a = target_sum * exp_logits / np.sum(exp_logits)

        # symmetry enforcement (keeps search in the palindromic subspace)
        a = (a + a[::-1]) / 2.0
        # re‑compute logits that correspond to the symmetrised a
        logits = np.log(a / target_sum + eps)

        # --- self‑convolution --------------------------------------------------
        conv = conv_self_fft(a)
        max_conv = conv.max()
        # keep global best
        if max_conv < best_score:
            best_score = max_conv
            best_seq = a.copy()
            print(f"Iter {iter_no:06d} – new best {best_score:.6f}")

        # --- smooth max (log‑sum‑exp) -----------------------------------------
        max_c = max_conv
        # normalized to avoid overflow
        tmp = alpha * (conv - max_c)
        exp_tmp = np.exp(tmp)
        w = exp_tmp / np.sum(exp_tmp)           # weights, sum to 1

        # --- gradient w.r.t. a -------------------------------------------------
        a_rev = a[::-1]
        grad_full = conv_fft(w, a_rev)          # length 3·n‑2
        grad_a = 2.0 * grad_full[(n - 1):(n - 1 + n)]

        # --- gradient w.r.t. logits (mirror descent) ---------------------------
        weighted_mean = np.dot(grad_a, a) / target_sum
        grad_logits = a * (grad_a - weighted_mean)

        # --- Adam update -------------------------------------------------------
        iter_no += 1
        m = beta1 * m + (1 - beta1) * grad_logits
        v = beta2 * v + (1 - beta2) * (grad_logits ** 2)
        m_hat = m / (1 - beta1 ** iter_no)
        v_hat = v / (1 - beta2 ** iter_no)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # enforce symmetry again (logits mirror)
        logits = (logits + logits[::-1]) / 2.0

        # --- schedule updates --------------------------------------------------
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha + alpha_inc)

        # occasional jitter to escape flat regions
        if np.random.rand() < 0.004:
            jitter = np.random.randn(n) * 0.001 * target_sum
            logits += jitter * 0.02

        # decay learning rate very slowly
        if iter_no % 4000 == 0:
            lr *= 0.85

    # ------------------------------------------------------------------
    # Phase 2 – polish with projected sub‑gradient descent (as in baseline)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    eta = 0.015 * (target_sum / n)          # small step size
    rel_tol = 0.0
    no_improve = 0
    iter2 = 0

    print(f"[{time.time():.2f}] Mirror‑descent finished, best {best_score:.6f}")
    while time.time() < deadline:
        a = subgradient_step(a, target_sum, eta, rel_tol=rel_tol)
        cur = np.max(conv_self_fft(a))
        iter2 += 1
        if cur + 1e-12 < best_score:
            best_score = cur
            best_seq = a.copy()
            print(f"Sub‑grad iter {iter2:06d} – improved to {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # adapt parameters slowly
        if iter2 % 500 == 0:
            eta *= 0.9
            eta = max(eta, 1e-6)
            rel_tol = min(0.06, rel_tol + 0.005)

        # occasional random pair transfer
        if np.random.rand() < 0.02:
            a = random_pair_transfer(a, target_sum, scale=0.005)

        # reset after long stagnation
        if no_improve >= 3000:
            jitter = np.random.randn(n) * 0.001
            a = np.clip(a + jitter, 0.0, None)
            a = project_to_sum(a, target_sum)
            no_improve = 0

    print(f"[{time.time():.2f}] Optimisation finished. Best score {best_score:.6f}")

    return best_seq.tolist()
