"""
Guided stochastic optimizer for the step‑function bound.

The objective to minimise is
    2·n·max(conv(a,a)) / (sum(a)²)

Because the objective is homogeneous we can fix the ℓ¹‑norm to
    sum(a) = √(2·n)
which makes the objective equal to the maximum entry of the auto‑convolution.
The optimiser starts from a family of power‑law profiles a_i ∝ (i+1)^{-p}
and then performs a *gradient‑guided* hill‑climb:

* For the current vector a we locate the index k where the convolution attains
  its maximum.
* The contribution of entry a_i to that maximum is w_i = a_{k‑i}
  (zero when the index is out of range).  Moving a small amount δ
  from i to j changes the peak by ≈‑2·δ·w_i (or ‑δ·a_i when i=k/2).
* Hence we repeatedly transfer a fraction of mass from a source with large
  w_i·a_i (large contribution to the peak) to a sink with very small w_j.
* Every accepted move strictly decreases the peak; occasional uphill moves
  are allowed via a tiny simulated‑annealing temperature to escape plateaux.
* The search is run for many short time slices on a range of lengths n and
  exponents p; the best sequence found overall is returned.
* A short final random‑move refinement (the previous simple local search)
  cleans up any remaining tiny improvements.

The whole routine respects the 1000 s time budget, prints progress, and
returns a plain list of non‑negative floats suitable for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Compute the full linear convolution a*a using an FFT of length a power of two.
    Returns the real-valued convolution array of length 2·len(a)‑1.
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """
    Return only the maximum value of a*a (much cheaper than the full array).
    """
    conv = _full_convolution(a)
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """
    Evaluation function used by the competition.
    The vector a must be non‑negative and non‑empty.
    """
    n = a.size
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    # With the normalisation sum = √(2·n) the denominator cancels, so we
    # can simply return the maximum convolution.
    return max_b


def _initial_sequence(p: float, n: int) -> np.ndarray:
    """
    Produce a deterministic start sequence a_i ∝ (i+1)^{-p}
    and renormalise it to the required ℓ¹‑norm √(2·n).
    """
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= np.sqrt(2.0 * n) / a.sum()
    return a


def _guided_search(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Gradient‑guided hill‑climbing that preserves the ℓ¹‑norm.
    At each step the index k of the maximal convolution is identified.
    Mass is transferred from a source with large contribution to that peak
    to a sink that hardly contributes.
    Only moves that lower the peak are kept (uphill moves are allowed
    with a tiny temperature for escaping flat regions).
    Returns the best vector seen and its score.
    """
    n = a.size
    rng = np.random.default_rng()
    best_a = a.copy()
    best_val = _score(a)
    cur_val = best_val

    # algorithmic parameters
    max_frac = 0.14          # maximal fraction of a[source] that may be moved
    temperature = 5e-5       # tiny simulated‑annealing temperature
    it = 0

    while time.time() < deadline:
        # full convolution to locate the current peak
        conv = _full_convolution(a)
        M = float(conv.max())
        k = int(conv.argmax())

        # w[i] = a[k-i] for those i where the index exists, otherwise 0
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.intp)
        w = np.zeros(n, dtype=np.float64)
        w[i_vals] = a[k - i_vals]

        # contribution of each entry to the peak (a_i * a_{k-i})
        contrib = w * a

        # -------- choose source (large contribution) ----------
        if np.any(contrib > 0):
            # take the 80‑th percentile as a cutoff for “large” contributions
            threshold = np.percentile(contrib[contrib > 0], 80)
            src_candidates = np.where(contrib >= threshold)[0]
            src = rng.choice(src_candidates)
        else:
            # fallback: any non‑zero entry
            nz = np.where(a > 1e-12)[0]
            if nz.size == 0:
                break
            src = rng.choice(nz)

        # -------- choose sink (tiny contribution) ---------------
        w_min = w.min()
        sink_candidates = np.where(w <= w_min + 1e-12)[0]
        sink_candidates = sink_candidates[sink_candidates != src]
        if sink_candidates.size == 0:
            # any position different from src
            sink = rng.integers(0, n)
            while sink == src:
                sink = rng.integers(0, n)
        else:
            sink = rng.choice(sink_candidates)

        # -------- propose a transfer ----------------------------
        delta = a[src] * (rng.random() * max_frac)
        if delta <= 1e-14:
            continue

        a[src] -= delta
        a[sink] += delta

        new_M = _max_convolution(a)

        if new_M < cur_val:
            # strictly improving move
            cur_val = new_M
            if new_M < best_val:
                best_val = new_M
                best_a = a.copy()
                print(f"[{time.time():.1f}] guided improved score = {best_val:.6f}")
        else:
            # possibly accept uphill move (tiny temperature)
            if temperature > 0.0 and rng.random() < np.exp((cur_val - new_M) / temperature):
                cur_val = new_M
            else:
                # reject – revert
                a[sink] -= delta
                a[src] += delta

        # decay parameters slowly
        it += 1
        if it % 800 == 0:
            max_frac = max(0.03, max_frac * 0.9)

    return best_a, best_val


def _random_hill_climb(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    A very simple stochastic hill‑climb (the method that was used originally).
    It is kept as a final polishing step.
    """
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    cur_val = best_val
    max_frac = 0.12
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        frac = rng.random() * max_frac
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        new_val = _score(a)

        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}] random hill‑climb improved score = {best_val:.6f}")
        else:
            a[j] -= delta
            a[i] += delta

    return best_a, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function sequence that minimises `evaluate_sequence`.
    The routine spends the allotted seconds performing a guided stochastic
    optimisation that starts from several power‑law profiles and iteratively
    reduces the peak of the auto‑convolution.
    The best sequence found within the time budget is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0    # a small safety margin

    best_seq = None
    best_score = np.inf

    # a selection of problem sizes – larger n gives a finer discretisation
    lengths = [800, 1500, 2500, 3500]
    exponents = [0.42, 0.45, 0.48, 0.5, 0.53, 0.56]

    # ------------------------------------------------------------------
    # Phase 1 – explore many initial shapes and run a short guided search
    # ------------------------------------------------------------------
    for n in lengths:
        for p in exponents:
            if time.time() >= deadline:
                break

            a = _initial_sequence(p, n)
            cur_score = _score(a)

            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] start (n={n}, p={p}) score = {cur_score:.6f}")

            # allocate a modest slice of time for the guided optimisation
            sub_deadline = min(deadline, time.time() + 7.0)
            a_opt, opt_score = _guided_search(a, sub_deadline)

            if opt_score < best_score:
                best_score = opt_score
                best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] after guided (n={n}, p={p}) best = {opt_score:.6f}")

        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Phase 2 – final polishing with the original random hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        sub_deadline = deadline
        a_opt, opt_score = _random_hill_climb(best_seq.copy(), sub_deadline)
        if opt_score < best_score:
            best_score = opt_score
            best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] final random hill‑climb best = {opt_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.ones(n, dtype=np.float64)
        best_seq *= np.sqrt(2.0 * n) / best_seq.sum()

    # Trim trailing zeros for neatness (the outer harness does not require it)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Return as a plain Python list
    return best_seq.tolist()
