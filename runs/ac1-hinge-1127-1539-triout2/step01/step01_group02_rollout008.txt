def smoothed_max_gradient(x, beta):
    conv = np.convolve(x, x)  # or use FFT for bigger n
    # shift by subtracting max for numeric stability
    s = beta * conv
    w = np.exp(s - np.max(s))
    w = w / np.sum(w)
    # compute gradient (2) * convolution of w with reversed x
    x_rev = x[::-1]
    g_full = np.convolve(w, x_rev)  # length 3n-2
    # extract middle part
    n = len(x)
    grad = 2 * g_full[n-1: n-1 + n]
    return grad, np.max(conv)   # maybe also return actual max

def project_onto_simplex(v):

    v = v - np.max(v)  # for numerical stability?
    # Actually Duchi's algorithm: sort descending.
    u = np.sort(v)[::-1]

    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]

   candidate we compute the objective *in‑house* using an FFT based
   convolution (fast even for several‑thousand entries).  The best
   candidate from this phase becomes the starting point for the next
   stage.

2. **Smooth‑gradient mirror descent** – treat the objective
   `F(x)=2·n·max(conv(x,x))` (with `∑x=1`).  The non‑differentiable
   `max` is approximated by a soft‑max with temperature `β`.  The
   resulting gradient can be written as a convolution and is obtained
   efficiently with FFTs.  An exponentiated‑gradient (mirror descent)
   update `x_i ← x_i·exp(−η·∇F_i)` is performed, followed by a
   re‑normalisation onto the probability simplex.  The temperature
   `β` is slowly increased so that the optimisation first sees a
   smoothed landscape and then focuses on the true maximum.

3. **Periodical pruning & jitter** – to reduce the factor `n` we
   occasionally trim a low‑mass tail of the vector (all entries below a
   tiny threshold are discarded) and re‑normalise.  Small random
   perturbations are injected from time‑to‑time to escape plateaus.

The whole search respects the global 1000 s budget.  The best sequence
found is returned as a plain Python list; its official score is also
printed for reference.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def conv_fft(x: np.ndarray) -> np.ndarray:
    """Full self‑convolution using a power‑of‑two FFT (real‑valued)."""
    n = x.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    X = np.fft.rfft(x, n=size)
    conv_full = np.fft.irfft(X * X, n=size)
    return conv_full[: 2 * n - 1]


def compute_score(x: np.ndarray) -> float:
    """Return the exact objective value (2·n·max(conv) / sum²)."""
    s = np.sum(x)
    if s < 1e-12:
        return float("inf")
    conv = conv_fft(x)
    max_b = np.max(conv)
    n = x.shape[0]
    return float(2.0 * n * max_b / (s * s))


def generate_powerlaw(p: float, sigma: float,
                      max_len: int = 8000, eps: float = 1e-12) -> np.ndarray:
    """a_i = (i+σ)^{-p}, truncated when values fall below eps."""
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p)
    mask = a > eps
    return a[mask]


def generate_exponential(lam: float,
                         max_len: int = 8000, eps: float = 1e-12) -> np.ndarray:
    """a_i = exp(-λ·i), truncated when values fall below eps."""
    i = np.arange(max_len, dtype=np.float64)
    a = np.exp(-lam * i)
    mask = a > eps
    return a[mask]


def generate_mixed(p: float, sigma: float, lam: float,
                   max_len: int = 8000, eps: float = 1e-12) -> np.ndarray:
    """a_i = (i+σ)^{-p}·exp(-λ·i)"""
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p) * np.exp(-lam * i)
    mask = a > eps
    return a[mask]


def generate_dirichlet(n: int, alpha: float) -> np.ndarray:
    """Sample from a Dirichlet(alpha,…,alpha) and normalise to sum 1."""
    g = np.random.gamma(alpha, 1.0, size=n)
    if np.sum(g) == 0.0:
        g = np.abs(np.random.randn(n))
    return g / np.sum(g)


def softmax_gradient(x: np.ndarray, beta: float):
    """
    Compute the gradient of the smoothed max objective:
        S(x) = (1/β)·log Σ_i exp(β·conv_i).
    Returns (grad, max_conv) where grad has the same shape as x.
    """
    n = x.shape[0]
    conv = conv_fft(x)                     # length 2n‑1
    # soft‑max weights (avoid overflow)
    s = beta * conv
    s_max = np.max(s)
    w = np.exp(s - s_max)
    w_sum = np.sum(w)
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.shape[0])
    else:
        w /= w_sum

    # gradient = 2·(w * x_rev) reduced to the central n entries
    x_rev = x[::-1]
    size = 1 << ((len(w) + n - 1).bit_length())
    W = np.fft.rfft(w, n=size)
    Xr = np.fft.rfft(x_rev, n=size)
    conv_wx = np.fft.irfft(W * Xr, n=size)
    grad = 2.0 * conv_wx[(n - 1) : (n - 1 + n)]
    return grad, np.max(conv)


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def prune_tail(x: np.ndarray, thr: float = 1e-10) -> np.ndarray:
    """Remove a trailing zero‑mass tail (entries ≤ thr)."""
    idx = np.nonzero(x > thr)[0]
    if idx.size == 0:
        return x
    last = idx[-1] + 1
    return x[:last]


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises `evaluate_sequence`.  The routine respects a 1000 s wall‑time
    budget and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # Phase 1 – broad parametric sweep (cheap, many candidates)
    # ------------------------------------------------------------------
    max_len = 8000
    eps = 1e-12

    # ----- power‑law families (p between 0.20 and 0.60) -----
    p_vals = np.concatenate((
        np.linspace(0.20, 0.45, 9),
        np.linspace(0.45, 0.60, 7)
    ))
    sigma_vals = np.logspace(-3, 2, 25)   # 1e‑3 … 1e2

    print("=== Phase 1a : power‑law sweep ===")
    for p in p_vals:
        for sigma in sigma_vals:
            if time.time() >= deadline:
                break
            seq = generate_powerlaw(p, sigma, max_len=max_len, eps=eps)
            # normalise to sum 1 (objective is scale‑invariant)
            seq = seq / np.sum(seq)
            score = compute_score(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"new best | p={p:.3f} σ={sigma:.4g} n={len(seq)} "
                      f"score={best_score:.6f}")
            # early exit if we cross the target bound
            if best_score < 1.503:
                break
        if best_score < 1.503 or time.time() >= deadline:
            break

    # ----- exponential families -----
    if best_score >= 1.503 and time.time() < deadline:
        lam_vals = np.logspace(-4, -0.5, 20)   # 1e‑4 … 0.3
        print("=== Phase 1b : exponential sweep ===")
        for lam in lam_vals:
            if time.time() >= deadline:
                break
            seq = generate_exponential(lam, max_len=max_len, eps=eps)
            seq = seq / np.sum(seq)
            score = compute_score(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"new best | λ={lam:.5g} n={len(seq)} score={best_score:.6f}")

    # ----- mixed families -----
    if best_score >= 1.503 and time.time() < deadline:
        print("=== Phase 1c : mixed (power·exp) sweep ===")
        for p in [0.25, 0.30, 0.35, 0.40]:
            for sigma in [0.01, 0.1, 1.0]:
                for lam in [0.001, 0.01, 0.1]:
                    if time.time() >= deadline:
                        break
                    seq = generate_mixed(p, sigma, lam, max_len=max_len, eps=eps)
                    seq = seq / np.sum(seq)
                    score = compute_score(seq)
                    if score < best_score:
                        best_score = score
                        best_seq = seq.copy()
                        print(f"new best | p={p:.2f} σ={sigma:.2e} λ={lam:.2e} "
                              f"n={len(seq)} score={best_score:.6f}")

    # ----- random Dirichlet vectors -----
    if best_score >= 1.503 and time.time() < deadline:
        print("=== Phase 1d : random Dirichlet sampling ===")
        for n in [400, 800, 1600, 3200, 6400]:
            for alpha in [0.1, 0.3, 1.0, 3.0]:
                for _ in range(5):
                    if time.time() >= deadline:
                        break
                    seq = generate_dirichlet(n, alpha)
                    score = compute_score(seq)
                    if score < best_score:
                        best_score = score
                        best_seq = seq.copy()
                        print(f"new best | Dirichlet n={n} α={alpha:.2f} "
                              f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # In case the sweep failed (should not happen), fall back to a uniform
    # ------------------------------------------------------------------
    if best_seq is None:
        n0 = 1000
        best_seq = np.ones(n0, dtype=np.float64)
        best_seq /= n0
        best_score = compute_score(best_seq)
        print("fallback to uniform vector, score =", best_score)

    # ------------------------------------------------------------------
    # Phase 2 – smooth‑gradient mirror descent refinement
    # ------------------------------------------------------------------
    x = best_seq.copy()             # already on the simplex
    n = x.shape[0]

    iteration = 0
    beta = 8.0                       # start with a modest smoothing
    beta_max = 5000.0
    base_eta = 0.20

    print("=== Phase 2 : smooth‑gradient mirror descent ===")
    while time.time() < deadline:
        iteration += 1

        # step‑size schedule (decreasing)
        eta = base_eta / math.sqrt(iteration + 1)

        # increase temperature slowly so that we focus more on the true max
        beta = min(beta_max, beta * 1.001 + 0.0005)

        # compute gradient of the smoothed max
        grad, cur_max = softmax_gradient(x, beta)

        # exponentiated‑gradient (mirror descent) update
        y = x * np.exp(-eta * grad)

        # re‑normalise onto the simplex
        if np.sum(y) == 0.0:
            y = np.abs(np.random.randn(len(x)))
        x = y / np.sum(y)

        # occasional tiny random jitter to avoid stagnation
        if np.random.rand() < 0.02:
            idx = np.random.randint(len(x))
            delta = (np.random.rand() - 0.5) * 0.04 * x[idx]
            x[idx] = max(0.0, x[idx] + delta)
            x /= np.sum(x)

        # every 200 iterations try to prune a low‑mass tail
        if iteration % 200 == 0:
            x_pruned = prune_tail(x, thr=1e-12)
            if x_pruned.shape[0] != x.shape[0]:
                x_pruned /= np.sum(x_pruned)
                # evaluate the pruned version (exact objective)
                prune_score = compute_score(x_pruned)
                if prune_score < best_score:
                    best_score = prune_score
                    best_seq = x_pruned.copy()
                    print(f"iteration {iteration:5d} – tail prune → "
                          f"len={len(x_pruned)} score={best_score:.6f}")
                x = x_pruned

        # evaluate (exact) every 10 iterations
        if iteration % 10 == 0:
            cur_score = compute_score(x)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = x.copy()
                print(f"iteration {iteration:5d} → improved score = "
                      f"{best_score:.6f}")

        # early stop if target already achieved
        if best_score < 1.503:
            print("Target bound reached – stopping early.")
            break

    # ------------------------------------------------------------------
    # Final verification with the official evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
    except Exception:
        final_score = float("inf")
    print("final best score :", final_score, "length :", len(final_seq))
    return final_seq
