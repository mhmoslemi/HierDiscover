def project_onto_simplex(v):
    # ensure v is a numpy array
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
softmax_descent.py

A heuristic optimizer for the step‑function bound.
The algorithm proceeds in two stages:

1. **Initial parametric sweep** – generate power‑law candidates
   a_i = (i + σ)^{-p} (fixed length n) for a grid of p≈0.5 and
   σ∈[1e‑4,1], normalise them to sum 1 and keep the best one.

2. **Softmax‑weighted mirror descent** – minimise the
   ℓ_∞‑norm of the autocorrelation c = x∗x.  At each iteration we
   compute the full convolution, turn it into a probability vector
   w by a softmax with a temperature parameter α (the approximation
   of the max), and take a multiplicative (exponential) update
   x_i ← x_i·exp(−η·g_i) with the gradient
       g = 2·correlate(w, x, mode='valid')
   (this is the exact gradient of the smooth max).  The step size η
   decays as 1/√t and α is slowly increased so that the smooth
   approximation becomes sharper.  After every update we re‑normalise
   x onto the probability simplex.  Random jitter and occasional
   random restarts help to escape plateaus.

The objective is homogeneous, therefore we keep ∑x_i = 1; the
evaluation score is simply 2·n·max(x∗x).  The routine respects a
global time budget (default 1000 s) and returns the best sequence
found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection only defined for 1‑D arrays.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    # Find the last index where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – return uniform vector
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _powerlaw_sequence(p: float, sigma: float, n: int) -> np.ndarray:
    """
    Return a length‑n power‑law vector a_i = (i+σ)^{-p},
    normalised to sum 1.
    """
    i = np.arange(n, dtype=np.float64)
    a = (i + sigma) ** (-p)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum
    return a


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Return softmax‑scaled, normalised weights for a convolution vector."""
    # shift by the maximum to avoid overflow
    shift = np.max(conv)
    ex = np.exp(alpha * (conv - shift))
    s = ex.sum()
    if s == 0.0:
        # all entries underflowed – fall back to uniform
        return np.full_like(conv, 1.0 / conv.size)
    return ex / s


def _smooth_max_score(x: np.ndarray, conv: np.ndarray) -> float:
    """Given a normalised vector x (∑x=1) and its convolution, compute the
    evaluate_sequence score 2·n·max(conv).
    """
    n = x.size
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises `evaluate_sequence`.  The routine respects the given
    time budget and returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – cheap parametric sweep (fixed length)
    # ------------------------------------------------------------------
    n = 1500                               # manageable size, good resolution
    best_seq = None
    best_score = float('inf')
    print("=== Phase 1 : parametric sweep ===")

    # grids around the theoretical exponent 0.5
    p_vals = np.linspace(0.48, 0.52, 9)    # 9 values
    sigma_vals = np.logspace(-4, 0, 20)    # from 1e‑4 to 1

    for p in p_vals:
        for sigma in sigma_vals:
            if time.time() >= deadline:
                break
            seq = _powerlaw_sequence(p, sigma, n)
            conv = np.convolve(seq, seq)
            score = _smooth_max_score(seq, conv)
            if score < best_score:
                best_score = float(score)
                best_seq = seq.copy()
                print(f'new best | p={p:.4f} sigma={sigma:.4e} score={best_score:.6f}')
            # early break if we already beat a reasonable target
            if best_score < 1.55:
                break
        if best_score < 1.55 or time.time() >= deadline:
            break

    # Fallback to uniform if sweep failed for any reason
    if best_seq is None:
        print("Parametric sweep failed – falling back to uniform.")
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _smooth_max_score(best_seq, np.convolve(best_seq, best_seq))

    # ------------------------------------------------------------------
    # Phase 2 – softmax‑weighted mirror descent
    # ------------------------------------------------------------------
    print("\n=== Phase 2 : smooth mirror descent ===")
    x = best_seq.astype(np.float64)          # keep on the simplex (∑x=1)
    iteration = 0
    lr0 = 0.30                               # base learning rate
    alpha = 1.0                              # start with a gentle softmax
    alpha_max = 200.0
    no_improve = 0
    patience = 30000                         # restart threshold

    while time.time() < deadline:
        iteration += 1

        # -- step‑size and temperature schedules --
        eta = lr0 / (np.sqrt(iteration) + 1.0)
        if iteration % 2000 == 0 and alpha < alpha_max:
            alpha = min(alpha_max, alpha * 1.25)

        # -- convolution and smooth max weights --
        conv = np.convolve(x, x)                     # length 2n‑1
        w = _softmax_weights(conv, alpha)           # same length, sums to 1

        # -- gradient of the smooth max (valid correlation) --
        #  g_i = 2 * Σ_k w_k * x_{k-i}   (indices out of range give 0)
        g = 2.0 * np.correlate(w, x, mode='valid')   # shape (n,)

        # -- multiplicative (exponential) update --
        # keep positivity automatically
        mult = np.exp(-eta * g)
        # guard against overflow/underflow
        mult = np.clip(mult, 1e-30, 1e30)
        y = x * mult
        # renormalise onto the simplex (simple L1‑normalisation works)
        if y.sum() == 0.0:
            # re‑initialise if we hit the zero vector
            y = np.abs(np.random.randn(n))
        x = y / y.sum()

        # occasional small random jitter to avoid stagnation
        if np.random.rand() < 0.015:
            idx = np.random.randint(n)
            delta = (np.random.rand() - 0.5) * 0.02 * x[idx]
            x[idx] = max(0.0, x[idx] + delta)
            x = x / x.sum()

        # -- evaluate current candidate (fast, no extra convolution) --
        cur_score = _smooth_max_score(x, conv)

        if cur_score < best_score:
            best_score = float(cur_score)
            best_seq = x.copy()
            no_improve = 0
            print(f'iteration {iteration:6d} | score={best_score:.6f} '
                  f'| α={alpha:.2f} | η={eta:.5f}')
            if best_score < 1.503:
                print('Target bound reached – terminating early.')
                break
        else:
            no_improve += 1
            # random restart if we have been stuck for a long time
            if no_improve >= patience:
                print('No improvement for many iterations – restarting.')
                # initialise from a fresh Dirichlet draw
                rnd = np.random.rand(n)
                x = rnd / rnd.sum()
                no_improve = 0

    print('final best score:', best_score, 'length:', len(best_seq))
    return best_seq.tolist()
