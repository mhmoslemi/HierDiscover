def _fast_evaluate(seq):
    seq = np.asarray(seq, dtype=np.float64)
    n = len(seq)
    sum_a = seq.sum()
    if sum_a < 0.01: # improbable
        return float('inf')
    # convolution via FFT with size pow2 >= 2*n-1
    size = 1 << ((2*n-1).bit_length())
    A = np.fft.rfft(seq, n=size)
    B_full = np.fft.irfft(A*A, n=size)[:2*n-1]
    max_b = B_full.max()
    return 2 * n * max_b / (sum_a**2)

"""
Search for a low‑evaluation step‑function.

The algorithm works in three stages:

1. **Coarse grid search** on a short vector (n≈2000).  The family
   a_i = ((i+γ+1)^p – (i+γ)^p) normalized to L1‑norm 1 is evaluated
   by a fast FFT‑based convolution, giving a cheap estimate of the
   `evaluate_sequence` objective.

2. **Fine search on longer vectors** (n≈12000‑18000).  Using the best
   exponent found in stage 1 we explore a small neighbourhood of the
   parameters (p,γ) and keep the best‑scoring candidate.

3. **Projected stochastic hill‑climbing**.  Starting from the best
   candidate we repeatedly move a tiny amount of mass from one index
   to another while preserving the L1‑norm; each move is accepted only
   if the fast FFT‑based objective improves.  The loop runs until the
   time budget expires.

The final sequence (still normalized to L1‑norm 1) is returned as a
plain list of floats.  The routine never produces negative entries,
so the security checks in `evaluate_sequence` are satisfied.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------
def _powerlaw_sequence(n: int, p: float, gamma: float) -> np.ndarray:
    """
    Return a non‑negative sequence of length n belonging to the
    power‑law family a_i = (i+γ+1)^p – (i+γ)^p, normalised to L1 = 1.
    """
    i = np.arange(n, dtype=np.float64)
    a = np.power(i + gamma + 1.0, p) - np.power(i + gamma, p)
    total = np.power(n + gamma, p) - np.power(gamma, p)
    # protect against division by zero – should never happen for p>0
    if total == 0.0:
        total = 1.0
    a /= total
    return a


def _fast_evaluate(seq: np.ndarray) -> float:
    """
    Compute the same quantity that ``evaluate_sequence`` returns but
    using an FFT based convolution (O(n log n)).  The result is only
    an estimate – the exact value is obtained later with the official
    ``evaluate_sequence`` call.
    """
    seq = np.asarray(seq, dtype=np.float64)
    n = seq.shape[0]
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    # size for linear convolution (next power of two)
    size = 1 << ((2 * n - 1).bit_length())
    ft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(ft * ft, n=size)[: 2 * n - 1]
    max_b = conv.max()
    return 2.0 * n * max_b / (sum_a * sum_a)

    best sequence discovered earlier.
    """
    return globals().get("height_sequence_1", [])

The algorithm proceeds in three phases:

1. **Coarse grid** on a short vector (n≈2000).  The family
   a_i = (i+γ+1)^p – (i+γ)^p, normalised to L1=1, is examined.
   Convolution is evaluated with an FFT‑based routine (O(n log n))
   so that thousands of candidates can be screened quickly.

2. **Fine search on longer vectors** (n≈12 k–18 k).  Using the best
   exponent from phase 1 we explore a small neighbourhood of (p,γ)
   for each length, again with the fast estimator, and keep the best
   candidate.

3. **Projected stochastic hill‑climbing**.  Starting from the best
   candidate we repeatedly move a tiny amount of mass from one index
   to another while preserving the L1‑norm.  Each move is accepted
   only if the fast estimator improves.  The loop runs until the time
   budget is exhausted.

def _powerlaw_sequence(n: int, p: float, gamma: float) -> np.ndarray:
    """
    Generate a non‑negative sequence belonging to the power‑law family

        a_i = (i+γ+1)^p – (i+γ)^p ,    i = 0 … n‑1

    and normalise it to L1‑norm 1.
    """
    i = np.arange(n, dtype=np.float64)
    a = np.power(i + gamma + 1.0, p) - np.power(i + gamma, p)
    total = np.power(n + gamma, p) - np.power(gamma, p)
    if total == 0.0:            # should never happen for p>0
        total = 1.0
    a /= total
    return a


def _fast_evaluate(seq: np.ndarray) -> float:
    """
    Cheap approximation of ``evaluate_sequence`` using an FFT based
    convolution (linear convolution via zero‑padding).  Returns the same
    formula 2 n·max(b)/(sum(a)^2) but does not apply the security
    clamps – our sequences never need them.

    seq = np.asarray(seq, dtype=np.float64)
    n = seq.shape[0]
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")

    ft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(ft * ft, n=size)[: 2 * n - 1]
    max_b = conv.max()
    return 2.0 * n * max_b / (sum_a * sum_a)
