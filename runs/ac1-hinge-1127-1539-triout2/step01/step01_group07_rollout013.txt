def conv_max(a):
    n = len(a)

Pseudo: def project_to_simplex(v, s=1.0):

    n = len(v)

    u = np.sort(v)[::-1]

    cssv = np.cumsum(u) - s

    ind = np.arange(n) + 1

    cond = u - cssv / ind > 0

    rho = ind[cond][-1]

    theta = cssv[cond][-1] / rho

    w = np.maximum(v - theta, 0)

    return w

"""
Propose a step‑function with a very low evaluation score.

The algorithm works in three stages:

1. **Coarse grid search** (small n) on the family
        a_i = (i+γ+1)^p – (i+γ)^p
   to obtain a good start (p≈0.4, γ≈10).

2. **Local refinement of the two parameters** on a larger vector (n≈20000)
   using a simple hill‑climbing / coordinate‑descent scheme.  The
   convolution and the evaluation are performed with an FFT‑based
   convolution (O(n log n)) to keep the runtime low.

3. **Projected gradient descent on the full vector**.
   For the current maximal convolution entry k* we compute the
   sub‑gradient g_i = 2·a_{k*‑i} (zero outside the support) and take
   a step a←a‑η·g, projecting back onto the probability simplex
   (∑a_i=1, a_i≥0).  The step size η is adapted dynamically and the
   loop stops when the time budget is exhausted.

The final sequence (list of floats) is returned; its quality is printed
as a sanity check.  No file I/O is performed and the code respects the
`evaluate_sequence` API that will be provided in the execution
environment.
"""

import time
import numpy as np

# ──────────────────────────────────────────────────────────────────────────────
# Helper functions (top‑level, no closures)
# ──────────────────────────────────────────────────────────────────────────────

def conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a using a power‑of‑two FFT length.
    Returns the full 2·n‑1 length array.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    return conv


def fast_evaluate(seq: np.ndarray) -> float:
    """
    Same objective as `evaluate_sequence` but much faster
    (uses FFT for the convolution).  The sequence is assumed to be
    non‑negative; sum may be <0.01 → return inf.
    """
    n = seq.shape[0]
    conv = conv_fft(seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x | x>=0, sum(x)=s}.
    Implements the algorithm of Wang & Carreira‑Perpiñán.
    """
    if np.sum(v) == s and np.all(v >= 0):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero after projection
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def generate_powerlaw_seq(n: int, p: float, gamma: float) -> np.ndarray:
    """
    a_i = (i+γ+1)^p – (i+γ)^p   (i = 0 … n‑1)
    Normalised so that sum_i a_i = 1.
    """
    i = np.arange(n, dtype=float)
    a = (np.power(i + gamma + 1.0, p) - np.power(i + gamma, p))
    total = np.power(n + gamma, p) - np.power(gamma, p)
    # Guard against division by zero (should not happen for valid params)
    if total == 0:
        total = 1.0
    a /= total
    # Numerical safety
    a = np.maximum(a, 0.0)
    return a

                p_new = p_cur + dp
                g_new = gamma_cur + dg
                if not (0.01 < p_new < 0.99 and g_new >= 0.0):
                    continue
                cand_seq = generate_powerlaw_seq(n_big, p_new, g_new)
                cand_val = fast_evaluate(cand_seq)
                if cand_val < best_val:
                    best_val = cand_val
                    p_cur, gamma_cur = p_new, g_new
                    improved = True
                    print(f"[local] improved to {cand_val:.6f} (p={p_cur:.5f}, γ={gamma_cur:.2f})")
        if not improved:
            # shrink steps – when they become tiny we stop
            step_p *= 0.5
            step_g *= 0.5
            if step_p < 1e-5:
                break

    # -------------------------------------------------------------------------
    # 3. Build the final candidate and run projected gradient descent
    # -------------------------------------------------------------------------
    a = generate_powerlaw_seq(n_big, p_cur, gamma_cur).copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_val = fast_evaluate(a)
    print(f"[final‑init] (p={p_cur:.5f}, γ={gamma_cur:.2f}) score={best_val:.6f}")

    # projected gradient descent on the simplex
    eta = 0.2                     # initial step size
    no_imp = 0                    # counter for stagnation

    while time.time() < deadline:
        # 1) compute convolution and the index of the current maximum
        conv = conv_fft(a)
        k_max = int(np.argmax(conv))

        # 2) sub‑gradient of the max‑entry (only the active index matters)
        g = np.zeros_like(a)
        rev = k_max - np.arange(n)
        mask = (rev >= 0) & (rev < n)
        g[mask] = 2.0 * a[rev[mask]]

        # 3) take a step and project back onto the probability simplex
        cand = a - eta * g
        cand = project_onto_simplex(cand, s=1.0)

        cand_val = fast_evaluate(cand)
        if cand_val < best_val - 1e-9:          # strict improvement
            best_val = cand_val
            a = cand
            best_seq = cand
            no_imp = 0
            eta = min(eta * 1.1, 1.0)           # try a slightly larger step
            print(f"[gd] improved to {best_val:.6f}  (η={eta:.4f})")
        else:
            # no improvement – reduce step size
            eta *= 0.5
            no_imp += 1
            if eta < 1e-8 or no_imp > 30:
                # step size too small or stuck → break out of GD
                break

        # occasional random tiny transfer to escape shallow plateaus
        if np.random.rand() < 0.02:
            i, j = np.random.choice(n, 2, replace=False)
            delta = (np.random.rand() - 0.5) * 0.01
            if a[i] + delta >= 0.0 and a[j] - delta >= 0.0:
                rnd = a.copy()
                rnd[i] += delta
                rnd[j] -= delta
                rnd = project_onto_simplex(rnd, s=1.0)
                rnd_val = fast_evaluate(rnd)
                if rnd_val < best_val:
                    best_val = rnd_val
                    a = rnd
                    best_seq = rnd
                    print(f"[rand] improved via random move to {best_val:.6f}")

    # -------------------------------------------------------------------------
    # 4. Final verification with the official evaluator (optional)
    # -------------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[result] final evaluated score = {final_score:.6f}")

    return best_seq.tolist()
