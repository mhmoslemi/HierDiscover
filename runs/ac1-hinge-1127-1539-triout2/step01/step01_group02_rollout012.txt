def fast_score(seq):
    a = np.asarray(seq, dtype=np.float64)
    n = a.shape[0]
    sum_a = a.sum()
    if sum_a < 0.01: return np.inf
    # compute convolution via FFT
    m = 2*n - 1
    # Next power of 2 for speed:
    size = 1 << (m-1).bit_length()
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)[:m]
    max_b = conv.max()
    score = 2.0 * n * max_b / (sum_a**2)
    return float(score)

def fast_score(seq):
    a = np.asarray(seq, dtype=np.float64)
    n = a.shape[0]
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    m = 2*n - 1
    size = 1 << (m-1).bit_length()
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)[:m]
    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_a*sum_a))

def _generate_powerlaw(p, sigma, max_len=2000, eps=1e-10):
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p)
    a = a[a > eps]
    return a

def _generate_exponential(lam, max_len=2000, eps=1e-10):
    i = np.arange(max_len, dtype=np.float64)
    a = np.exp(-lam * i)
    a = a[a > eps]
    return a

def _solve_lp_step(a, M, target_sum, tau=0.5):
    # a: numpy array current sequence, sum = target_sum
    n = a.shape[0]
    L = 2 * n - 1
    # Build convolution matrix F (L x n)
    # each column j is a placed at rows j:j+n
    F = np.zeros((L, n), dtype=np.float64)
    for j in range(n):
        F[j:j+n, j] = a
    g = cp.Variable(n, nonneg=True)
    # constraints: F @ g <= M
    cons = [F @ g <= float(M)]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), cons)
    # warm start optional
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, feastol=1e-8, abstol=1e-8, reltol=1e-8)

"""
Search for a non‑negative step function (list of heights) that minimizes
`evaluate_sequence`.

The algorithm has three stages:

1. **Broad parametric sweep** – power‑law and exponential families are
   generated for several lengths, exponents and scale parameters.  The best
   candidate (according to our own fast‑implementation of the score) is kept.

2. **Alternating linear‑programming refinement** – for the current best
   sequence `a` (normalised to unit ℓ1‑norm) we solve
       maximise   1ᵀ g        (g ≥ 0)
       subject to a * g ≤ M,
   where `M` is the current maximal convolution value.  The optimal `g`
   is rescaled to the same ℓ1‑norm as `a`, mixed with `a`, and the process is
   repeated a few times.  This is exactly the LP‑step described in the
   statement and is very effective at decreasing the peak autocorrelation.

3. **Mirror‑descent polishing** – a short multiplicative‑update loop
   (sub‑gradient descent on the ℓ∞‑norm of the convolution) cleans up any
   remaining slack.

All steps respect the 1000 s global time budget.  The final sequence is
returned as a plain Python list; its scale is irrelevant because the
objective is homogeneous.
"""

import time
import math
import numpy as np
import random

# cvxpy is optional – if unavailable we simply skip the LP phase.
try:
    import cvxpy as cp
    _CVXPY_OK = True
except Exception:                # pragma: no cover
    _CVXPY_OK = False


def _fast_score(seq: list[float]) -> float:
    """
    Compute the same objective as ``evaluate_sequence`` but using FFT, which
    is much faster for long sequences.
    """
    a = np.asarray(seq, dtype=np.float64)
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:
        return float("inf")
    m = 2 * n - 1
    # size = next power of two for fast FFT
    size = 1 << (m - 1).bit_length()
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:m]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


def _generate_powerlaw(p: float, sigma: float,
                       max_len: int = 2000,
                       eps: float = 1e-10) -> np.ndarray:
    """Truncated power‑law a_i = (i+σ)^(-p)."""
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p)
    return a[a > eps]


def _generate_exponential(lam: float,
                          max_len: int = 2000,
                          eps: float = 1e-10) -> np.ndarray:
    """Truncated exponential a_i = exp(-λ·i)."""
    i = np.arange(max_len, dtype=np.float64)
    a = np.exp(-lam * i)
    return a[a > eps]


def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix `F` such that (F @ g) == f * g   (full convolution).
    Shape of the result is (2*len(f)-1, len(f)).
    """
    n = f.shape[0]
    L = 2 * n - 1
    F = np.zeros((L, n), dtype=np.float64)
    for j in range(n):
        F[j:j + n, j] = f
    return F


def _lp_refine_step(a: np.ndarray,
                    target_sum: float,
                    tau: float = 0.5) -> np.ndarray | None:
    """
    One iteration of the LP‑based refinement:

    * compute M = max(a*a)
    * solve maximise 1ᵀ g  s.t. a*g ≤ M , g≥0
    * rescale g to have the same ℓ1‑norm as ``a``
    * return (1‑τ)·a + τ·g_scaled   (renormalised to target_sum).
    """
    n = a.shape[0]
    M = np.max(np.convolve(a, a))
    F = _conv_matrix(a)                     # (2n‑1) × n

    g = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(g)),
                      [F @ g <= float(M)])
    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                   max_iters=5000, feastol=1e-8,
                   abstol=1e-8, reltol=1e-8)
    except Exception:                       # pragma: no cover
        return None

    if g.value is None:
        return None

    g_vec = np.maximum(g.value, 0.0)
    if g_vec.sum() == 0.0:
        return None

    # rescale to the prescribed ℓ1‑norm
    g_scaled = g_vec * (target_sum / g_vec.sum())
    a_new = (1.0 - tau) * a + tau * g_scaled
    # renormalise just in case of tiny numerical drift
    a_new = a_new * (target_sum / a_new.sum())
    return a_new


def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,

    """
    Return a list of non‑negative heights that (hopefully) yields a score

    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # tiny safety margin

    best_score = float("inf")
    best_seq: list[float] | None = None

    except Exception:

    print("=== Phase 1 : parametric sweep ===")
    # We explore a few reasonable lengths; longer ones are slower.

    for max_len in length_candidates:

    # -----------------------------------------------------------------

    print("=== Phase 3 : mirror‑descent polishing ===")
    x = np.array(best_seq, dtype=np.float64)

        eta = 0.25 / math.sqrt(it)          # slowly decreasing stepsize
        conv = np.convolve(x, x)
        k = int(conv.argmax())
        n = x.shape[0]
        # sub‑gradient of the ℓ∞‑norm at the maximal index
        g = np.zeros_like(x)

            if best_score < 1.503:
                print("Target reached – stopping early.")
                break
        x = y

    print("final best score:", best_score, "length:", len(best_seq))
    return best_seq
