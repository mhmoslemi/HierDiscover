"""
Propose a step‑function height sequence that minimizes the self‑convolution
bound (the value returned by `evaluate_sequence`).  The code runs for at most
the supplied time budget (≈1000 s).  It combines three ideas:

1. **Soft‑max smoothing of the L∞ objective** – the maximum entry of the
   self‑convolution is approximated by  ε·log ∑ exp(b_k/ε).  Its gradient is a
   weighted sum of the individual convolution gradients, where the weights are
   a soft‑max over the current convolution values.  For large ε the landscape
   is smooth, for tiny ε it approaches the true sub‑gradient.

2. **Projected (momentum) gradient descent** – after each gradient step the
   vector is clipped to non‑negative values and projected back onto the
   ℓ₁‑simplex of radius √(2 n), which enforces the normalisation used by the
   evaluator.

3. **Escaping plateaus** – occasional random pairwise mass transfers,
   jitter and a simple restart‑on‑stagnation mechanism keep the search
   from getting trapped.

All heavy work (convolutions and gradients) is performed with FFTs, so each
iteration costs only a few × 10⁻⁴ s for n≈8000, allowing many thousands of
updates inside the time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution of a with itself using an FFT of power‑of‑two size.
    Returns the full convolution (length 2*len(a)-1).
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    Fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Fa * Fa, n=L)
    return conv[: 2 * n - 1]


def conv_linear_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors a and b using FFT.
    Returns the full convolution (length len(a)+len(b)-1).
    """
    n, m = a.shape[0], b.shape[0]
    L = 1 << ((n + m - 1).bit_length())          # next power of two
    Fa = np.fft.rfft(a, n=L)
    Fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Fa * Fb, n=L)
    return conv[: n + m - 1]


def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """
    Scale a non‑negative vector to have the prescribed ℓ₁‑norm.
    Zero entries stay zero; if the whole vector is zero a uniform distribution
    is returned.
    """
    s = v.sum()
    if s > 0.0:
        return v * (target / s)
    # fallback – uniform distribution
    return np.full_like(v, target / v.shape[0])


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a tiny amount of mass from one randomly chosen index to another,
    preserving the total ℓ₁‑norm (up to rounding error).
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]          # amount to move from i to j
    if delta <= 0.0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_to_sum(a, target_sum)


def gradient_softmax_max(a: np.ndarray, eps: float):
    """
    Compute the smooth approximation of max(b) where b = a*a (linear conv).
    Returns (max_b, grad) where grad = ∂/∂a  (∑_k p_k b_k) and
    p_k = soft‑max(b_k/eps) (∑_k p_k = 1).

    The gradient formula:
        g_j = 2 * Σ_i a_i p_{i+j}  –  p_{2j} * a_j
    is obtained by differentiating the weighted sum of the convolution entries.
    """
    n = a.shape[0]

    # 1) self‑convolution and max entry
    b = conv_self_fft(a)
    max_b = b.max()

    # 2) soft‑max weights (numerically stable)
    shift = max_b
    w = np.exp((b - shift) / eps)
    w_sum = w.sum()
    if w_sum == 0.0:                 # degenerate case (should not happen)
        p = np.full_like(w, 1.0 / w.size)
    else:
        p = w / w_sum                # normalized probabilities, Σ p = 1

    # 3) convolution a * p_rev  (full length = n + (2n-1) - 1 = 3n-2)
    p_rev = p[::-1]
    conv = conv_linear_fft(a, p_rev)          # length 3n-2

    # index pattern: g_j = 2 * conv[(2n-2)-j] – p_{2j} * a_j
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * conv[idx] - p[2 * np.arange(n)] * a
    return max_b, grad


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    `evaluate_sequence`.  The algorithm runs for at most *budget_s*
    seconds and returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Choose a length – reuse the known construction if available
    # ------------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        a = np.array(base, dtype=np.float64)
    else:
        n = np.random.randint(7500, 9000)           # manageable large size
        a = np.full(n, 1.0, dtype=np.float64)

    target_sum = np.sqrt(2.0 * n)                    # the normalisation used in the paper
    a = np.clip(a, 0.0, None)
    a = project_to_sum(a, target_sum)

    # ------------------------------------------------------------------
    # Initial evaluation
    # ------------------------------------------------------------------
    b0 = conv_self_fft(a)
    best_score = b0.max()

    print(f"[{time.time():.2f}] start n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    lr = 0.02 * (target_sum / n)          # initial learning rate (≈2 % of avg entry)
    eps = best_score * 0.5                # initial smoothing parameter

        a_next = a - lr * momentum

        a_next = np.clip(a_next, 0.0, None)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
