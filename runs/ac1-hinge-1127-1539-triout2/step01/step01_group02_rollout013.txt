import math

def _generate_powerlaw(p, sigma, max_len=2000, eps=1e-10):
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p)
    return a[a > eps]

def _build_conv_matrix(f):
    n = f.shape[0]
    rows = 2*n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = f[i]
    return A

def _lp_max_sum(f, M):
    n = f.shape[0]
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = f[i]
    # Scale constraints to avoid huge numbers:
    A_ub = A / M
    b_ub = np.ones(2*n-1)
    c = -np.ones(n)   # minimize -sum(b) -> maximize sum(b)
    bounds = [(0, None) for _ in range(n)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        # fallback: return current f (feasible)
        return f.copy()
    b = res.x

def propose_candidate(seed=0, budget_s=1000.0):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0

    best_seq = None
    best_score = float('inf')

    # try to use global height_sequence_1
    try:
        init = np.array(height_sequence_1, dtype=np.float64)
        if init.size > 0:
            seq = init.tolist()
            score = evaluate_sequence(seq)
            if np.isfinite(score):
                best_score = float(score)
                best_seq = seq
                print(f'start from global seq score {best_score}')
    except Exception:
        init = None

    # Phase 1: param sweep
    max_len = 2000
    eps = 1e-9
    p_vals = np.linspace(0.35, 0.65, 13) # broader
    sigma_vals = np.logspace(-4, 0, 25)  # 1e-4 to 1
    for p in p_vals:
        for sigma in sigma_vals:
            if time.time() >= deadline:
                break
            arr = _generate_powerlaw(p, sigma, max_len, eps)
            if arr.size == 0:
                continue
            seq = arr.tolist()
            score = evaluate_sequence(seq)
            if np.isfinite(score) and score < best_score:
                best_score = float(score)
                best_seq = seq
                print(f'new param best p={p:.3f} sigma={sigma:.5f} len={len(seq)} score={best_score:.6f}')
                # if already good, break loops earlier?
                if best_score < 1.503:
                    break
        if best_score < 1.503 or time.time() >= deadline:
            break

    if best_seq is None:
        # fallback uniform
        n_fallback = 500
        best_seq = [1.0] * n_fallback
        best_score = evaluate_sequence(best_seq)
        print('fallback uniform', best_score)

    # Convert to float array normalized to sum=1
    f = np.array(best_seq, dtype=np.float64)
    f = f / f.sum()   # sum to 1

    # Phase 2: iterative linear-program + mixing
    iteration = 0
    t_initial = 0.4
    while time.time() < deadline:
        iteration += 1
        # compute current corv and score
        conv_f = np.convolve(f, f)
        M = np.max(conv_f)
        # solve LP to get b
        b = _lp_max_sum(f, M)
        s = b.sum()
        if s <= 0:
            # fallback: break
            break
        g = b / s
        # mixing parameter decays
        t = t_initial / np.sqrt(iteration)
        f_new = (1 - t) * f + t * g
        # re‑normalise
        f_new = f_new / f_new.sum()
        # evaluate candidate (list)
        cur_score = evaluate_sequence(f_new.tolist())
        if np.isfinite(cur_score) and cur_score < best_score:
            best_score = float(cur_score)
            best_seq = f_new.tolist()
            print(f'iter {iteration:5d} score={best_score:.6f} n={len(best_seq)}')

                break

        f = f_new

        if np.random.rand() < 0.02:

    print('final best score', best_score, 'len', len(best_seq))

def _lp_max_sum(f, M):
    n = f.shape[0]
    rows = 2*n - 1
    A = np.empty((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = f[i]
    # We can normalise constraints to avoid big numbers
    A_ub = A / M
    b_ub = np.ones(rows)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',options={'presolve': True})
    if res.success:
        return res.x
    else:
        # fallback: just return current f normalized to sum S? (should be feasible)
        return np.copy(f)

"""Propose a candidate step function using a power‑law seed and a linear‑program
refinement.  The algorithm first sweeps a few power‑law families to obtain a
reasonable starting distribution, then repeatedly solves a small linear program
that maximises the total mass of a new distribution under the same ‑∞‑convolution
bound as the current one.  A convex combination of the old and the new
distribution is taken (with a decreasing step‑size), re‑normalised, and the
score is evaluated.  The process continues until the time budget expires or
the target bound 1.503 is reached.
"""
import time
import numpy as np

# Try to import a linear‑programming solver; fall back to a simple
# mirror‑descent method if it is unavailable.
try:
    from scipy.optimize import linprog
    _SCIPY_OK = True
except Exception:                     # pragma: no cover
    _SCIPY_OK = False

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _generate_powerlaw(p: float, sigma: float, max_len: int = 2000,
                       eps: float = 1e-9) -> np.ndarray:
    """
    Produce a truncated power‑law sequence a_i = (i+σ)^{‑p}.
    Elements smaller than `eps` are discarded; the result is not normalised.
    """
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p)
    return a[a > eps]


def _score_of(seq: np.ndarray) -> float:
    """
    Compute the evaluation score exactly as evaluate_sequence would,
    assuming all entries are non‑negative and ≤1000.
    """
    if seq.ndim != 1 or seq.size == 0:
        return float('inf')
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _build_toeplitz_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix A such that (A @ b) equals the full
    convolution f*b.  Entry A[k,i] = f[k‑i] when 0≤k‑i<n, otherwise 0.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # Each column i contributes f[i] on rows i … i+n‑1
    for i in range(n):
        A[i:i + n, i] = f[i]
    return A


def _lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve  max 1ᵀb   s.t.   A·b ≤ M·1 ,   b ≥ 0
    where A is the convolution matrix of the current vector ``f``.
    The function returns the optimal b (may be the zero vector if the LP
    fails – the caller should treat that case as a fallback).
    """
    n = f.shape[0]
    A = _build_toeplitz_matrix(f)
    # Normalise the constraints to keep numbers of order 1
    A_ub = A / M
    b_ub = np.ones(A.shape[0], dtype=np.float64)

    # linprog solves a minimisation problem; we minimise –∑b.
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None) for _ in range(n)]

    # ``highs`` is the default modern interior‑point/simplex hybrid.
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if res.success:
        return res.x
    # In the unlikely event of failure we simply return a copy of f,
    # which is always feasible (f*f ≤ M implies f satisfies the constraints).
    return f.copy()


def _mirror_descent_step(x: np.ndarray, eta: float) -> np.ndarray:
    """
    One iteration of the exponential‑mirror‑descent scheme used in the
    original baseline.  It is kept as a cheap fallback when the LP solver
    is unavailable.
    """
    n = x.shape[0]
    conv = np.convolve(x, x)
    k = int(conv.argmax())
    g = np.zeros_like(x)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    idx = np.arange(i_min, i_max + 1)
    jdx = k - idx
    g[idx] = x[jdx]
    y = x * np.exp(-eta * g)
    if y.sum() == 0.0:
        y = np.abs(np.random.randn(n))
    y /= y.sum()
    return y
