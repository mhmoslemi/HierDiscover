def smooth_max(conv, gamma):
    max_val = np.max(conv)
    # shift to avoid overflow:
    exp_vals = np.exp(gamma * (conv - max_val))
    sum_exp = exp_vals.sum()
    f = (np.log(sum_exp) / gamma) + max_val
    w = exp_vals / sum_exp  # weighted probabilities
    return f, w
```

def adam_update(a, grad, m, v, t, lr, beta1, beta2, eps):
    m = beta1*m + (1-beta1)*grad
    v = beta2*v + (1-beta2)*(grad**2)
    m_hat = m/(1 - beta1**t)
    v_hat = v/(1 - beta2**t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    # project nonneg and renormalize sum
    a_new = np.maximum(a_new, 0)
    total = a_new.sum()
    if total == 0:
        a_new = np.full_like(a, S / len(a))
    else:
        a_new *= (S / total)
    return a_new, m, v

def _mult_grad_desc(a, deadline, best_seq, best_score):
    n = len(a)
    S = np.sqrt(2.0 * n)
    eta = 0.3
    decay = 0.9995
    jitter_prob = 0.01

    while time.time() < deadline:
        # compute conv and max index
        conv, _, k_max = _conv_max(a)  # we need conv_max function returns conv, max, idx
        # compute gradient g
        idx = np.arange(n)
        partner = k_max - idx
        valid = (partner >= 0) & (partner < n)

        g = np.zeros_like(a)
        g[valid] = 2.0 * a[partner[valid]]
        if k_max % 2 == 0:
            d = k_max // 2
            if 0 <= d < n:
                g[d] = a[d]

        # multiplicative update
        a = a * np.exp(-eta * g)

        # renorm
        total = a.sum()
        if total == 0:
            a = np.full(n, S / n)
        else:
            a = a * (S / total)

        # evaluate improvement
        cur_max = conv.max()
        if cur_max < best_score - 1e-12:
            ext = evaluate_sequence(a.tolist())
            if ext < best_score:
                best_score = ext
                best_seq = a.copy()
                print(f"[mult improve] time={time.time():.1f}s score={best_score:.6f}")

        eta *= decay

        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * 0.01
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()

    return best_seq, best_score

"""
Propose a step‑function sequence with a low self‑convolution maximum.

The optimizer works in two phases (both stay within the 1000 s budget).

1. **Smooth‑max Adam phase**  
   * Replace the hard max in `max_k (a*a)[k]` by a smooth log‑sum‑exp
     approximation with a temperature γ.  
   * The gradient of the smooth objective can be written in closed form as  
       ∇ = 2·corr_w(a) – w[2·i]·a_i,
     where w are the soft‑max weights and `corr_w(a)` is the correlation of w
     with the current vector a.  Both the correlation and the self‑convolution
     are computed by FFTs (O(n log n)).  
   * Adam updates keep the variables non‑negative and a fixed L1‑norm
     (√(2 n)), which is exactly the normalisation used by `evaluate_sequence`.

2. **Multiplicative‑gradient phase** (the method from the reference
   implementation) fine‑tunes the solution once γ is large, i.e. when the
   smooth approximation is close to the true max.

Both phases keep the best sequence seen so far and stop early if the target
bound (1.503) is reached.  The final sequence is returned as a plain Python
list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper routines (no closures)
# ----------------------------------------------------------------------


def _conv_full(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2*n‑1) via FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_max(a: np.ndarray) -> tuple[np.ndarray, float, int]:
    """
    Return (full convolution, maximum value, index of the maximum).
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    max_idx = int(np.argmax(conv))
    return conv, float(conv[max_idx]), max_idx


def _smooth_max_and_weights(conv: np.ndarray, gamma: float) -> tuple[float, np.ndarray]:
    """
    Smooth approximation of max(conv) = (1/γ)·log Σ exp(γ·conv_i).
    Returns the smooth value and the corresponding soft‑max weights.
    """
    max_val = np.max(conv)
    # shift to avoid overflow
    shifted = conv - max_val
    exp_vals = np.exp(gamma * shifted)
    sum_exp = exp_vals.sum()
    smooth_val = max_val + np.log(sum_exp) / gamma
    weights = exp_vals / sum_exp
    return smooth_val, weights


def _grad_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth max w.r.t. a.
    Implements  ∇_i = 2·Σ_j w_{i+j}·a_j – w_{2i}·a_i.
    All operations are performed with FFT‑based convolutions (O(n log n)).
    """
    n = a.shape[0]
    # correlation w ⋆ a (length n) = (w * a_rev)[n‑1 : n‑1+n]
    a_rev = a[::-1]
    conv_len = len(w) + n - 1                # length of linear convolution w * a_rev
    fft_len = 1 << (conv_len - 1).bit_length()
    W = np.fft.rfft(w, n=fft_len)
    A = np.fft.rfft(a_rev, n=fft_len)
    full = np.fft.irfft(W * A, n=fft_len)[:conv_len]
    start = n - 1
    c = full[start:start + n]                # correlation values c_i

    idx = np.arange(n)
    diag = w[2 * idx] * a                     # w_{2i}·a_i
    grad = 2.0 * c - diag
    return grad


def _initial_sequence(n: int, seed: int) -> np.ndarray:
    """Create a non‑negative vector of length n with L1‑norm √(2 n)."""
    rng = np.random.default_rng(seed)
    S = math.sqrt(2.0 * n)

    if 'height_sequence_1' in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=float)
            if base.size >= n:
                a = base[:n].copy()
            else:
                a = np.pad(base, (0, n - base.size), constant_values=0.0)
        except Exception:
            a = rng.random(n)
    else:
        a = rng.random(n)

    a = a / a.sum() * S
    return a

# Phase 1 – Adam on smooth max
# ----------------------------------------------------------------------


def _adam_phase(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Adam optimisation on the smooth‑max surrogate."""
    n = a.shape[0]
    S = math.sqrt(2.0 * n)

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[adam init] n={n} score={best_score:.6f}")

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.2

    beta1_pow, beta2_pow = 1.0, 1.0
    t = 0

    # schedule for the temperature γ
    gamma = 5.0
    gamma_max = 200.0
    gamma_factor = 1.03

    while time.time() < deadline:
        t += 1
        beta1_pow *= beta1
        beta2_pow *= beta2

        # current convolution a*a
        conv = _conv_full(a)

        # smooth max and soft‑max weights
        _, w = _smooth_max_and_weights(conv, gamma)

        # gradient of the smooth objective
        grad = _grad_from_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the feasible set (non‑negative + L1‑norm)
        a = np.maximum(a, 0.0)
        tot = a.sum()
        if tot == 0.0:
            a = np.full_like(a, S / n)
        else:
            a *= (S / tot)

        # check for improvement (using the true max, cheap to evaluate)
        cur_max = conv.max()
        if cur_max + 1e-12 < best_score:
            ext = evaluate_sequence(a.tolist())
            if ext < best_score:
                best_score = ext
                best_seq = a.copy()
                print(f"[adam improve] iter={t} time={time.time():.1f}s "
                      f"score={best_score:.6f} γ={gamma:.2f}")

        # schedule updates
        gamma = min(gamma * gamma_factor, gamma_max)
        if t % 200 == 0:
            lr *= 0.9

        # occasional random jitter
        if np.random.rand() < 0.005:
            noise = np.random.randn(n) * a.mean() * 0.01
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()

        # early exit if we already beat the target
        if best_score <= 1.503:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Phase 2 – classical multiplicative gradient descent
# ----------------------------------------------------------------------


def _mult_phase(a: np.ndarray, deadline: float,
                best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Fine‑tune the solution using the original multiplicative update."""
    n = a.shape[0]
    S = math.sqrt(2.0 * n)

    eta = 0.3
    decay = 0.9995
    jitter_prob = 0.01

    while time.time() < deadline:
        # convolution and the index of its maximum
        conv, cur_max, k_max = _conv_max(a)

        # sub‑gradient of the maximal entry
        idx = np.arange(n)
        partner = k_max - idx
        valid = (partner >= 0) & (partner < n)

        g = np.zeros_like(a)
        g[valid] = 2.0 * a[partner[valid]]
        if k_max % 2 == 0:
            d = k_max // 2
            if 0 <= d < n:
                g[d] = a[d]                     # diagonal contribution

        # multiplicative update (keeps non‑negativity)
        a = a * np.exp(-eta * g)

        # re‑normalise
        tot = a.sum()
        if tot == 0.0:
            a = np.full(n, S / n)
        else:
            a *= (S / tot)

        # report improvement
        if cur_max + 1e-12 < best_score:
            ext = evaluate_sequence(a.tolist())
            if ext < best_score:
                best_score = ext
                best_seq = a.copy()
                print(f"[mult improve] time={time.time():.1f}s score={best_score:.6f}")

        # step‑size decay and occasional jitter
        eta *= decay
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * 0.01
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()

        if best_score <= 1.503:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative step heights whose evaluation score
    (max self‑convolution) is as low as possible within ``budget_s`` seconds.
    """
    np.random.seed(seed)
    # small safety margin
    deadline = time.time() + budget_s - 5.0

    # Choose a length – use the known construction length if available,
    # otherwise a moderately large size that still runs fast.
    n = 8000
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)

    # ---- initial vector -------------------------------------------------
    a0 = _initial_sequence(n, seed)

    # ---- Phase 1: Adam on smooth max (≈80 % of the time) ----------------
    phase1_end = time.time() + (budget_s - 5) * 0.8
    best_seq, best_score = _adam_phase(a0, phase1_end)

    # ---- Phase 2: multiplicative gradient descent (remaining time) -----
    if time.time() < deadline:
        best_seq, best_score = _mult_phase(best_seq, deadline,
                                           best_seq, best_score)

    print(f"[final] best score = {best_score:.6f}  length = {len(best_seq)}")
    return best_seq.tolist()
