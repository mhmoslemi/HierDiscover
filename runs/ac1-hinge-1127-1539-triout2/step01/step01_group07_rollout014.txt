Pseudo:

def project_simplex(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    # Sort v decreasing
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # Find rho = max{i | u_i - cssv_i / i > 0}
    condition = u - cssv / ind > 0
    rho = np.where(condition)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def project_simplex(v):
    """Project vector v onto the probability simplex {x>=0, sum=1}."""
    # implementation by Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        tau = 0
    else:
        rho = rho[-1]
        tau = cssv[rho] / (rho + 1)
    w = np.maximum(v - tau, 0)
    return w

"""
Propose a step‐function sequence with a low value of the provided
evaluation function.

The algorithm works in three stages:

1. **Parameter sweep** – The family
       a_i = ((i+γ+1)^p – (i+γ)^p)   (i = 0 … n−1)
   is generated for many values of the exponent `p` (0<p<1) and the
   offset `γ`.  The sequence is normalised to L1‑norm = 1, so the
   evaluation reduces to `2·n·max(conv(a,a))`.  A coarse grid (p≈0.15‑0.5,
   γ≈0‑400) is searched on a relatively long vector (n=20000).  The best
   parameters are kept.

2. **Local refinement** – Around the best parameters a finer grid
   (small steps in `p` and `γ`) is examined to possibly improve the
   objective further.

3. **Stochastic fine‑tuning** – Starting from the best sequence a
   simple hill‑climbing moves a tiny amount of mass from one index to
   another while keeping the L1‑norm fixed.  Each move is accepted only
   if it reduces the score.  This stage continues until the time budget
   (≈1000 s) is exhausted.

All operations respect the constraints of `evaluate_sequence`
(non‑negative, bounded values).  The function finally returns the best
list of heights found.
"""

import time
import numpy as np

def _generate_sequence(n: int, p: float, gamma: float) -> np.ndarray:
    """
    Return a length‑n non‑negative sequence belonging to the
    (p,γ) power‑law family, L1‑normalised to 1.
    """
    i = np.arange(n, dtype=float)
    a = (np.power(i + gamma + 1.0, p) - np.power(i + gamma, p))
    total = np.power(n + gamma, p) - np.power(gamma, p)
    # avoid division by zero – total > 0 for p>0
    a /= total
    return a


def _score_sequence(seq: np.ndarray, n: int) -> float:
    """
    Compute the evaluation value 2·n·max(conv(seq,seq)).
    The L1‑norm of `seq` is assumed to be 1 (as produced by
    _generate_sequence), so the denominator of `evaluate_sequence`
    equals 1 and we can compute the score directly.
    """
    conv = np.convolve(seq, seq)         # length 2·n−1
    max_b = np.max(conv)
    return 2.0 * n * max_b


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) whose
    evaluation value is as small as possible.  The routine respects the
    1000 s time limit, prints progress, and returns the best list it
    finds.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1. Coarse grid search (large n, modest grid)
    # ------------------------------------------------------------------
    n = 20000                     # length of the final construction
    best_score = float('inf')
    best_params = (0.4, 0.0)      # (p, gamma) placeholder

    # p in (0.15, 0.5) – step 0.01  (≈36 values)
    p_vals = np.linspace(0.15, 0.50, 36)
    # gamma in [0,400] – step 20   (21 values)
    gamma_vals = np.arange(0, 401, 20)

    print("[search] starting coarse grid search (n = %d)" % n)

    for p in p_vals:
        for gamma in gamma_vals:
            seq = _generate_sequence(n, float(p), float(gamma))
            score = _score_sequence(seq, n)

            if score < best_score:
                best_score = score
                best_params = (float(p), float(gamma))
                print(f"[grid] new best: score={score:.6f}, p={p:.4f}, γ={gamma}")

            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 2. Fine local grid around the current best parameters
    # ------------------------------------------------------------------
    if time.time() < deadline:
        p0, g0 = best_params
        # finer p‑step = 0.005, γ‑step = 2, +- neighbourhood
        p_fine = np.arange(max(0.01, p0 - 0.03), p0 + 0.031, 0.005)
        g_fine = np.arange(max(0, int(g0) - 30), int(g0) + 31, 2)

        print("[search] starting fine grid refinement")
        for p in p_fine:
            for gamma in g_fine:
                seq = _generate_sequence(n, float(p), float(gamma))
                score = _score_sequence(seq, n)

                    print(f"[fine] improved: score={score:.6f}, p={p:.4f}, γ={gamma}")

                    break
            if time.time() > deadline:
                break

    # ------------------------------------------------------------------
    p_best, gamma_best = best_params
    seq = _generate_sequence(n, p_best, gamma_best)
    best_score = _score_sequence(seq, n)
    print(f"[final] best parameters -> p={p_best:.5f}, γ={gamma_best:.5f}")
    print(f"[final] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Light stochastic refinement (pairwise mass transfers)
    # ------------------------------------------------------------------
    best_seq = seq.copy()
    while time.time() < deadline:
        # pick two distinct indices and move a tiny amount of mass
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 1e-3   # signed step, magnitude ~10⁻³
        if seq[i] + delta < 0 or seq[j] - delta < 0:
            continue
        cand = seq.copy()
        cand[i] += delta
        cand[j] -= delta
        # sum is unchanged, negativity already checked
        cand_score = _score_sequence(cand, n)

        if cand_score < best_score:
            best_score = cand_score
            seq = cand
            best_seq = cand.copy()
            print(f"[refine] improved score -> {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Return the best sequence (as a plain Python list)
    # ------------------------------------------------------------------
    print(f"[result] best score achieved: {best_score:.6f} (length {n})")
    return best_seq.tolist()
