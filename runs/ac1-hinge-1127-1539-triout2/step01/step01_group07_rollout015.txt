def fast_evaluation(seq):
    n = len(seq)
    S = np.sum(seq)
    if S < 1e-9:
        return np.inf
    # compute convolution via FFT
    # pad to next power of two for speed
    m = 1 << ((2*n - 1).bit_length())
    # Use real FFT
    a = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(a * a, n=m)[:2*n-1]
    max_b = np.max(conv)
    return float(2 * n * max_b / (S * S))
```

def generate_combined(n, p, gamma, beta):
    i = np.arange(n, dtype=float)
    diff = (i + gamma + 1) ** p - (i + gamma) ** p
    a = diff * np.exp(-beta * i)
    total = np.sum(a)
    a /= total
    return a

"""
Propose a low‑evaluation step function.

The algorithm proceeds in three stages:

1. **Fast random search**  
   – Three parametric families are used:
        * power‑law differences (the previous best family)    
        * the same family with an extra tiny exponential decay
        * a Beta‑distribution shape (a·x^a·(1−x)^b)
   – For each candidate a short sequence (n≈5 000) is generated.
   – The evaluation 2·n·max(conv)/sum² is computed with an FFT‑based
     convolution (exact up to floating‑point noise) – this is
     orders of magnitude faster than the O(n²) `np.convolve` used by
     `evaluate_sequence`.
   – The best parameters (family + its numeric parameters) are kept.

2. **Upscaling**  
   – Using the best parameters a longer sequence (n≈15 000) is built.
     The asymptotic value of the evaluation for the power‑law family is
     2 p² B(p,p)≈1.35 for p≈0.4, therefore a larger n pushes the
     actual value closer to this theoretical optimum.

3. **Light hill‑climbing refinement** (still using the fast FFT
   evaluator) – tiny mass transfers between random positions are tried
   while time permits.  The sum is preserved, all entries stay
   non‑negative, and any improvement is kept.

The final sequence (as a plain Python list) is returned; the external
grader will call `evaluate_sequence` on it to obtain the official score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper generators for the different families
# ----------------------------------------------------------------------


def generate_powerdiff(n: int, p: float, gamma: float) -> np.ndarray:
    """Power‑law differences: a_i = (i+γ+1)^p - (i+γ)^p, L1‑normalised."""
    i = np.arange(n, dtype=np.float64)
    a = (np.power(i + gamma + 1.0, p) - np.power(i + gamma, p))
    total = (n + gamma) ** p - gamma ** p          # telescoping sum – exact
    if total == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    a /= total
    return a


def generate_combined(n: int, p: float, gamma: float, beta: float) -> np.ndarray:
    """
    Same as `generate_powerdiff` but multiplied by a tiny exponential decay
    exp(-beta·i).  The extra factor gives a little extra flexibility.
    """
    i = np.arange(n, dtype=np.float64)
    diff = (np.power(i + gamma + 1.0, p) - np.power(i + gamma, p))
    a = diff * np.exp(-beta * i)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    a /= total
    return a


def generate_beta(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑distribution shape on the discrete grid, L1‑normalised."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a_vals = np.power(x, a) * np.power(1.0 - x, b)
    total = a_vals.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    a_vals /= total
    return a_vals


# ----------------------------------------------------------------------
# Fast evaluator – uses FFT instead of the O(n²) np.convolve
# ----------------------------------------------------------------------


def _fast_evaluate(seq: np.ndarray) -> float:
    n = len(seq)
    total = seq.sum()
    if total < 1e-12:
        return float("inf")
    # pad to the next power of two for a very fast FFT
    size = 1 << ((2 * n - 1).bit_length())
    ft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(ft * ft, n=size)[: 2 * n - 1]
    max_b = np.max(conv)
    return float(2.0 * n * max_b / (total * total))

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a step‑function with evaluation < 1.503.

    The function respects the 1000 s wall‑clock budget and returns a
    list of non‑negative floats (the heights of the step function).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    # ------------------------------------------------------------------
    # 0.  Start from the supplied construction if it exists
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    best_info = None

    if "height_sequence_1" in globals():
        try:
            seq0 = np.asarray(height_sequence_1, dtype=np.float64)
            score0 = _fast_evaluate(seq0)
            best_seq = seq0.tolist()
            best_score = score0
            best_info = ("init", {})
            print(f"[init] baseline fast score = {score0:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1. Random search on a moderate size (n≈5 000)
    # ------------------------------------------------------------------
    n_small = 5000
    families = ["powerdiff", "combined", "beta"]
    it = 0
    while time.time() < deadline:
        family = np.random.choice(families)
        if family == "powerdiff":
            p = np.random.uniform(0.35, 0.55)          # focus around the known optimum
            gamma = np.random.uniform(0.0, 120.0)
            seq = generate_powerdiff(n_small, p, gamma)
            info = {"p": p, "gamma": gamma}
        elif family == "combined":
            p = np.random.uniform(0.35, 0.55)
            gamma = np.random.uniform(0.0, 120.0)
            beta = np.random.uniform(0.0, 0.02)        # tiny exponential twist
            seq = generate_combined(n_small, p, gamma, beta)
            info = {"p": p, "gamma": gamma, "beta": beta}
        else:  # beta family
            a_par = np.random.uniform(0.2, 4.0)
            b_par = np.random.uniform(0.2, 4.0)
            seq = generate_beta(n_small, a_par, b_par)
            info = {"a": a_par, "b": b_par}

        score = _fast_evaluate(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.tolist()
            best_info = (family, info)
            print(f"[search] new best {score:.6f} (family={family})")

        it += 1
        if it >= 20000:      # a generous cap; the time check will usually finish earlier
            break

    # ------------------------------------------------------------------
    # 2. Build a larger sequence (n≈15 000) using the best parameters
    # ------------------------------------------------------------------
    n_big = 15000
    if best_info is None:
        # fall back to the classic construction
        seq_big = generate_powerdiff(n_big, 0.40, 10.0)
    else:
        family, params = best_info
        if family == "powerdiff":
            seq_big = generate_powerdiff(n_big, params["p"], params["gamma"])
        elif family == "combined":
            seq_big = generate_combined(
                n_big, params["p"], params["gamma"], params.get("beta", 0.0)
            )
        else:  # beta
            seq_big = generate_beta(n_big, params["a"], params["b"])

    seq = seq_big.copy()
    best_big_score = _fast_evaluate(seq)
    print(f"[final] initial big fast score = {best_big_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Light hill‑climbing refinement (still with the fast evaluator)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        i, j = np.random.randint(n_big, size=2)
        if i == j:
            continue
        # typical magnitude of entries is ~1/n, use a step an order of magnitude larger
        delta = (np.random.rand() - 0.5) * 1e-4
        if seq[i] + delta < 0 or seq[j] - delta < 0:
            continue
        cand = seq.copy()
        cand[i] += delta
        cand[j] -= delta
        cand_score = _fast_evaluate(cand)
        if cand_score < best_big_score:
            best_big_score = cand_score
            seq = cand
            print(f"[refine] improved big fast score = {best_big_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Report the true score once (optional, does not affect the return)
    # ------------------------------------------------------------------
    try:
        true_score = evaluate_sequence(seq.tolist())
        print(f"[result] true evaluation by evaluate_sequence = {true_score:.6f}")
    except Exception:
        pass

    return seq.tolist()
