import time
import numpy as np

def generate_sequence(p, gamma, n):
    i = np.arange(n, dtype=np.float64)
    a = (np.power(i + gamma + 1.0, p) - np.power(i + gamma, p))
    total = np.power(n + gamma, p) - np.power(gamma, p)
    a = a / total
    return a

def fast_evaluation(seq):
    n = len(seq)
    # Compute convolution via FFT (real-to-real)
    n_fft = 1 << ((2*n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n_fft)
    conv = np.fft.irfft(fft_seq * fft_seq, n_fft)
    max_b = conv[:2*n-1].max()
    return 2 * n * max_b

"""
A fast two‑stage optimizer for the step‑function evaluation problem.

Stage 1 – a cheap grid / random search on the two‑parameter family

        a_i = (i + γ + 1)^p – (i + γ)^p ,   i = 0 … n‑1

    The sequence is normalised so that Σ a_i = 1, hence the objective
    reduces to   2·n·max((a∗a)).   Convolution is computed with FFT
    (O(n log n)), which makes it possible to explore thousands of
    candidates quickly.

Stage 2 – build a longer final sequence with the best (p,γ) and
    perform a tiny hill‑climbing on the actual coefficients.  The
    final length (≈20 000) is small enough that the reference
    `evaluate_sequence` can finish comfortably, but large enough that
    the evaluation approaches the theoretical limit (≈1.35 for p≈0.4).

All numbers are kept non‑negative and the L1‑norm stays equal to one,
so the function respects the security checks in `evaluate_sequence`.
The routine stops before the supplied time budget expires.
"""

import time
import numpy as np

def _generate_sequence(p: float, gamma: float, n: int) -> np.ndarray:
    """
    Return the normalised non‑negative sequence of length *n* defined by
    the power‑law difference formula.
    """
    i = np.arange(n, dtype=np.float64)
    a = (np.power(i + gamma + 1.0, p) - np.power(i + gamma, p))
    total = np.power(n + gamma, p) - np.power(gamma, p)   # telescoping sum
    a /= total                                            # Σ a_i = 1
    return a


def _fast_evaluation(seq: np.ndarray) -> float:
    """
    Fast evaluation of the objective for a sequence whose sum is 1.
    Returns 2·n·max((seq∗seq)) using an FFT‑based convolution.
    """
    n = seq.shape[0]
    # next power of two that accommodates the full convolution
    n_fft = 1 << ((2 * n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n_fft)
    conv = np.fft.irfft(fft_seq * fft_seq, n_fft)
    max_b = conv[: 2 * n - 1].max()
    return 2.0 * n * max_b


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a step‑function with evaluation < 1.503.

    The algorithm proceeds in three phases:
    1) coarse grid search on a medium sized sequence,
    2) stochastic refinement of the (p,γ) parameters,
    3) construction of a longer final sequence with a tiny mass‑transfer
       hill‑climbing on the actual coefficients.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0           # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – coarse grid search (n = 15000)
    # ------------------------------------------------------------------
    n_search = 15000
    best_score = float("inf")
    best_p = None
    best_beta = None                                 # gamma = beta * n

    # reasonable ranges for the two parameters
    p_vals = np.linspace(0.25, 0.45, 41)              # 0.25 … 0.45
    beta_vals = np.linspace(0.002, 0.10, 25)          # gamma as a fraction of n

    for p in p_vals:
        for beta in beta_vals:
            gamma = beta * n_search
            seq = _generate_sequence(p, gamma, n_search)
            score = _fast_evaluation(seq)
            if score < best_score:
                best_score = score
                best_p = p
                best_beta = beta
                print(f"[grid] new best={score:.6f} (p={p:.4f}, beta={beta:.4f})")
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # Phase 2 – stochastic refinement of (p, beta)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # propose a small random move in the (p,beta) plane
        p_prop = best_p + np.random.randn() * 0.003
        beta_prop = best_beta + np.random.randn() * 0.004

        # keep inside sensible bounds
        if not (0.20 < p_prop < 0.55):
            continue
        if not (0.0 < beta_prop < 0.20):
            continue

        gamma = beta_prop * n_search
        seq = _generate_sequence(p_prop, gamma, n_search)
        score = _fast_evaluation(seq)

        if score < best_score:
            best_score = score
            best_p = p_prop
            best_beta = beta_prop
            print(f"[refine] new best={score:.6f} (p={p_prop:.4f}, beta={beta_prop:.4f})")

    # ------------------------------------------------------------------
    # Phase 3 – build final (larger) sequence and fine‑tune it
    # ------------------------------------------------------------------
    n_final = 20000                                 # still fast enough for the reference evaluator
    gamma_final = best_beta * n_final
    seq_final = _generate_sequence(best_p, gamma_final, n_final)

    # small hill‑climbing on the actual coefficients (transfer mass between two indices)
    best_seq = seq_final.copy()
    best_seq_score = _fast_evaluation(best_seq)

    while time.time() < deadline:
        i, j = np.random.choice(n_final, 2, replace=False)

        if best_seq[i] + delta < 0 or best_seq[j] - delta < 0:

        cand_score = _fast_evaluation(cand)

            best_seq = cand
            print(f"[final‑refine] improved to {cand_score:.6f}")

    # ------------------------------------------------------------------

        print(f"[official] final score = {official:.6f}")
    except Exception:
        pass
