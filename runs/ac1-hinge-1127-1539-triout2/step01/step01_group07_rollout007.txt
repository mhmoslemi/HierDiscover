"""
Search for a step function with low evaluation.

We explore a 1‑parameter power‑law family a_i = (i+γ+1)^p-(i+γ)^p,
normalised to L1=1.  A coarse grid on a modest length (n≈4000) is
evaluated with a fast FFT‑based surrogate of `evaluate_sequence`.
The best (p,γ) pair is refined randomly, then a longer vector (n≈15000)
is built and a simple mass‑transfer hill‑climb is performed.
The final sequence is returned; the true objective is computed with the
provided `evaluate_sequence` before returning.
"""

import time
import math
import random
import numpy as np

def _generate_sequence(n: int, p: float, gamma: float) -> np.ndarray:
    """
    Generate a non‑negative step function of length n.
    The coefficients are differences of the power law x↦(x+γ)^p,
    normalised so that the L1‑norm equals 1.
    """
    i = np.arange(n, dtype=np.float64)
    a = np.power(i + gamma + 1.0, p) - np.power(i + gamma, p)
    total = np.power(n + gamma, p) - np.power(gamma, p)
    # Avoid division by zero – when gamma is zero and p is close to 0,
    # total may be extremely small; in that case fall back to uniform.
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    # Clamp to the bounds used by evaluate_sequence
    a = np.clip(a, 0.0, 1000.0)
    return a


def _fast_evaluate(seq: np.ndarray) -> float:
    """
    Fast surrogate for `evaluate_sequence`.
    Computes 2*n*max(conv(seq,seq))/(sum(seq)^2) using FFT.
    """
    seq = np.asarray(seq, dtype=np.float64)
    n = seq.shape[0]
    # Length of the linear convolution
    L = 2 * n - 1
    # FFT of the zero‑padded sequence
    fft_seq = np.fft.fft(seq, n=L)
    conv = np.fft.ifft(fft_seq * fft_seq).real
    max_b = conv.max()
    sum_a = seq.sum()

"""
Search for a low‑evaluation step‑function.

The algorithm works in four phases:

1.  **Coarse grid search** on a moderate length (≈4000).  
    The candidate family is a power‑law difference
        a_i = (i+γ+1)^p – (i+γ)^p,
    normalised so that the L¹‑norm equals 1.
    A fast FFT‑based surrogate of `evaluate_sequence`
    (2·n·max(conv)/sum²) is used to rank the candidates.

2.  **Random refinement** of the two parameters (p, γ) while the fast
    surrogate still guides the search.

3.  **Construction of a longer vector** (≈15000) with the best (p,γ).
    The true objective `evaluate_sequence` is computed once for logging.

4.  **Simple hill‑climbing on the coefficients**: a tiny amount of mass
    is moved from an early (large) index to a late (small) index.
    The fast surrogate discards non‑promising moves, while an
    improvement is verified with the official `evaluate_sequence`.
    The best sequence found is returned.
"""

import time
import random

def _generate_sequence(n: int, p: float, gamma: float) -> np.ndarray:
    """
    Produce a non‑negative step function of length ``n``.
    a_i = (i+γ+1)^p - (i+γ)^p, then normalise so that Σa_i = 1.
    The result is clipped to the bounds used by ``evaluate_sequence``.
    """
    i = np.arange(n, dtype=np.float64)
    a = np.power(i + gamma + 1.0, p) - np.power(i + gamma, p)
    total = np.power(n + gamma, p) - np.power(gamma, p)
    # Guard against a zero total (should not happen for 0<p<1)
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    a = np.clip(a, 0.0, 1000.0)
    return a
