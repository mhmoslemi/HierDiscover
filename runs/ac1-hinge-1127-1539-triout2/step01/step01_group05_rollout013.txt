            a = np.concatenate([a, np.zeros(n - cut, dtype=np.float64)])

        # Normalise to the target sum.
        s = a.sum()
        if s == 0.0:
            continue
        a *= target_sum / s

        sc = _compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.time() - start_time
            print(
                f"[Phase 1] New best {best_score:.6f} (p={p:.4f},β={beta:.4f},cut={cut_ratio:.3f}) "
                f"elapsed {elapsed:.1f}s"
            )

    # If the random search failed to produce a candidate (unlikely), fall back to the analytic start.
    if best_a is None:
        a = 1.0 / np.sqrt(np.arange(n, dtype=np.float64) + 0.5)
        a *= target_sum / a.sum()
        best_a = a.copy()
        best_score = _compute_score(best_a)

    # --------------------------------------------------------------
    # Phase 2 – Adam optimisation using a soft‑max approximation.
    # --------------------------------------------------------------
    a = best_a.copy()
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.12                     # initial learning rate
    lr_min = 0.001

    # FFT sizes that stay constant throughout the run.
    nfft_conv = _next_pow2(2 * n - 1)   # for a * a
    nfft_corr = _next_pow2(3 * n - 2)   # for w ⊛ rev(a)

    gamma = 5.0
    gamma_max = 250.0
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # ----------------------------------------------------------
        # 1) Convolution a * a (real FFT, keep only the first 2n‑1 values)
        # ----------------------------------------------------------
        af = np.fft.rfft(a, n=nfft_conv)
        conv = np.fft.irfft(af * af, n=nfft_conv)[: 2 * n - 1]
        max_conv = conv.max()

        # ----------------------------------------------------------
        # 2) Build the weights w.
        #    When gamma is large we switch to a hard max (one‑hot) to get the
        #    exact sub‑gradient of the true max.
        # ----------------------------------------------------------
        if gamma >= 150.0:
            # Hard max: a single one‑hot entry at the current arg‑max.
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            # Soft‑max (log‑sum‑exp) weights.
            w = np.exp(gamma * (conv - max_conv))   # max weight will be 1
            w_sum = w.sum()
            if w_sum == 0.0:                       # numerical safety
                w.fill(1.0 / w.size)
                w_sum = 1.0
            w /= w_sum

        # ----------------------------------------------------------
        # 3) Gradient g = 2 * ( w ⊛ a_rev )
        # ----------------------------------------------------------
        wf = np.fft.rfft(w, n=nfft_corr)
        af_rev = np.fft.rfft(a[::-1], n=nfft_corr)
        corr_full = np.fft.irfft(wf * af_rev, n=nfft_corr)
        # Valid part starts at index n‑1 and has length n.
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]

        # ----------------------------------------------------------
        # 4) Adam update (projected onto the feasible set)
        # ----------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Projection onto the non‑negative orthant.
        a_new = np.maximum(a_new, 0.0)

        # Renormalise to keep Σa fixed (the objective is scale‑invariant).
        s = a_new.sum()
        if s == 0.0:
            a_new = np.full_like(a_new, target_sum / n)
        else:
            a_new *= target_sum / s

        # ----------------------------------------------------------
        # 5) Evaluate and keep the best ever seen.
        # ----------------------------------------------------------
        cur_score = _compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            elapsed = time.time() - start_time
            print(
                f"[Iter {iteration:6d}] New best {best_score:.6f} "
                f"(elapsed {elapsed:.1f}s, γ={gamma:.1f}, lr={lr:.4f})"
            )

        # Accept the step (the algorithm is purely descent with projection).
        a = a_new

        # ----------------------------------------------------------
        # 6) Hyper‑parameter schedules
        # ----------------------------------------------------------
        # Slowly increase the soft‑max temperature to focus on the true max.
        gamma = min(gamma * 1.004, gamma_max)

        # Decay learning rate, but keep a modest floor.
        lr = max(lr * 0.998, lr_min)

        # Periodically inject a tiny random jitter to avoid being stuck.
        if iteration % 97 == 0:
            jitter = 1e-4 * np.random.randn(n)
            a = np.maximum(a + jitter, 0.0)
            a_sum = a.sum()
            if a_sum == 0.0:
                a = np.full_like(a, target_sum / n)
            else:
                a *= target_sum / a_sum

        # Occasionally restart from the current best with a fresh momentum
        # (helps when the optimiser gets trapped in a flat region).
        if iteration % 613 == 0:
            a = best_a.copy()
            m.fill(0.0)
            v.fill(0.0)

    # --------------------------------------------------------------
    # Return the best sequence found (as a plain Python list).
    # --------------------------------------------------------------
    return best_a.tolist()
```

This implementation respects all constraints (no filesystem/network access, single‑CPU usage, no closures, clear top‑level helpers) and is designed to explore a richer family of candidate shapes before fine‑tuning them with a modern optimisation routine.
