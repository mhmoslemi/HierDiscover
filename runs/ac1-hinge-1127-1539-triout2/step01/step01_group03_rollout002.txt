def toeplitz_conv_matrix(f):
    n = len(f)
    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        i_min = max(0, k - (n-1))
        i_max = min(k, n-1)
        for i in range(i_min, i_max + 1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    return scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

import cvxpy as cp

def solve_lp(f, M):
    n = len(f)
    T = toeplitz_conv_matrix(f)  # shape (2n-1, n)
    g = cp.Variable(n)
    constraints = [g >= 0,
                   T @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    if g.value is None:
        # fallback to simple heuristic: return f
        return f.copy()
    return np.array(g.value).flatten()

def solve_lp_max_sum(f, M):
    n = len(f)
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        i_min = max(0, k-(n-1))
        i_max = min(k, n-1)
        # f_i values for i from i_min to i_max
        # g index = k - i
        # iterate i and compute j = k - i
        for i in range(i_min, i_max+1):
            j = k - i
            rows.append(k)
            cols.append(j)

    T = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

    return np.array(g.value).flatten()

"""
search_step_sequence.py

A multi‑stage optimizer that tries to minimise
    2·n·max(conv(a,a)) / (sum(a)²)

The strategy is:
1.  Random / grid search over the simple power‑law family
    a_i = 1/(i+1+c)^β   (β≈0.5, c≥0) to obtain a decent starting shape.
2.  A short projected gradient descent that reduces the current maximal
    convolution entry.
3.  An LP‑based mixing step (if CVXPY is available).  For a fixed f we solve

        maximise   Σ g_j
        subject to T_f·g ≤ M ,   g ≥ 0

    where T_f is the linear‑convolution matrix (conv(f,g) = T_f·g) and
    M = max(conv(f,f)).  The solution g has a larger ℓ₁‑norm while still
    respecting the same max‑convolution bound.  After rescaling g to the
    same ℓ₁‑norm as f we replace f by a convex combination (1‑t)·f + t·g.
    This strictly reduces the maximal convolution value in practice.
4.  A final fine‑grained gradient descent to polish the result.

All operations keep the ℓ₁‑norm fixed to √(2·n) (the normalisation used by
the official evaluator).  The algorithm respects the 1000 s time budget
and prints progress information.  It returns a list[float] that can be
directly fed to ``evaluate_sequence``.
"""

import time
import numpy as np
import scipy.sparse

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """
    Fast evaluation of the objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)
    using an FFT based convolution.  The function is scale‑invariant, i.e.
    the value equals max(conv(seq,seq)) when sum(seq)=√(2·n).
    """
    n = len(seq)
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    # length of the FFT (next power of two for speed)
    m = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft * fft, n=m)[: 2 * n - 1]
    max_conv = np.max(conv)
    # after the normalisation used later the factor 2·n/(s**2) is exactly 1
    return float(2.0 * n * max_conv / (s * s))


def _full_convolution(seq: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of seq with itself."""
    n = len(seq)
    m = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft * fft, n=m)[: 2 * n - 1]
    return conv


def _build_toeplitz_matrix(f: np.ndarray) -> scipy.sparse.csr_matrix:
    """
    Build the (2n‑1)×n sparse matrix T such that conv(f,g) = T·g
    for any non‑negative g of length n.
    """
    n = len(f)
    rows = []
    cols = []
    data = []
    # Fill row by row – each row corresponds to a convolution index k
    for k in range(2 * n - 1):
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        # i runs over indices of f that participate in conv at position k
        for i in range(i_min, i_max + 1):
            j = k - i          # column index for g
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    T = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return T


def _solve_lp_max_sum(f: np.ndarray, M: float):
    """
    Solve  maximise Σ g_j   subject to  T_f·g ≤ M ,  g ≥ 0.
    Returns the optimal g as a dense NumPy array, or None on failure.
    """
    try:
        import cvxpy as cp
    except Exception:          # CVXPY not available – give up the LP step
        return None

    n = len(f)
    T = _build_toeplitz_matrix(f)

    g = cp.Variable(n, nonneg=True)
    constraints = [T @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    # ECOS is usually the fastest open‑source solver for this size
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000,
                   abstol=1e-8, reltol=1e-8, feastol=1e-8)
    except Exception:
        return None

    if g.value is None:
        return None
    return np.array(g.value).reshape(-1)


def _project_onto_l1_ball(x: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale x so that its ℓ₁‑norm becomes target_sum (preserving non‑negativity)."""
    s = np.sum(x)
    if s <= 0:
        # avoid division by zero – return a uniform tiny vector
        return np.full_like(x, target_sum / len(x))
    return x * (target_sum / s)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Return a low‑score step‑function sequence.
    The function respects the 1000 s wall‑clock budget and never returns an
    empty/invalid list.
    """
    np.random.seed(seed)
    deadline = time.time() + max(0.0, budget_s - 5.0)   # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a problem size.
    #    n ≈ 1500 is a good trade‑off: large enough for a fine shape,
    #    small enough for the LP‐step to stay fast.
    # ------------------------------------------------------------------
    n = 1500
    target_sum = np.sqrt(2.0 * n)

    print("[+] Starting search, length =", n)

    # ------------------------------------------------------------------
    # 2. Initialisation – random/grid search over the (β,c) power‑law family.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # a) coarse deterministic grid
    betas = np.linspace(0.45, 0.55, 21)          # around the theoretical optimum 0.5
    offsets = np.linspace(0.0, 8.0, 17)          # a broader range than before

    for beta in betas:
        for c in offsets:
            idx = np.arange(n, dtype=np.float64) + 1.0 + c
            cand = 1.0 / (idx ** beta)
            cand = _project_onto_l1_ball(cand, target_sum)
            sc = _objective(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[grid] β={beta:.3f}, c={c:.2f} → score={sc:.6f}")
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    # b) additional random sampling (helps to escape the coarse grid)
    trials = 200
    while time.time() < deadline and trials > 0:
        beta = np.random.uniform(0.42, 0.58)
        c = np.random.uniform(0.0, 20.0)
        idx = np.arange(n, dtype=np.float64) + 1.0 + c
        cand = 1.0 / (idx ** beta)
        cand = _project_onto_l1_ball(cand, target_sum)
        sc = _objective(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[rand] β={beta:.3f}, c={c:.2f} → score={sc:.6f}")
        trials -= 1

    if best_seq is None:
        # fallback – uniform sequence (never happens in practice)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _objective(best_seq)

    # ------------------------------------------------------------------
    # 3. First short gradient‑descent refinement (few hundred iterations)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    step_factor = 0.4
    no_improve = 0
    best_global_seq = a.copy()
    best_global_score = best_score
    iter_cnt = 0

    while time.time() < deadline and iter_cnt < 400:
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))
        max_conv = conv[k_max]

        # sub‑gradient of the max‑entry (g_i = a_i + a_{k_max-i})
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        shift = np.zeros_like(a)
        shift[mask] = a[idx[mask]]
        grad = a + shift

        # step size proportional to the current ratio
        r = max_conv / (np.sum(a) ** 2)
        step = step_factor * r
        new_a = a - step * grad
        new_a = np.maximum(new_a, 0.0)
        new_a = _project_onto_l1_ball(new_a, target_sum)

        new_score = _objective(new_a)
        if new_score < best_global_score - 1e-8:
            best_global_score = new_score
            best_global_seq = new_a.copy()
            a = new_a
            no_improve = 0
            step_factor = min(step_factor * 1.05, 1.0)
            if iter_cnt % 50 == 0:
                print(f"[grad‑pre] iter={iter_cnt}, score={new_score:.6f}")
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                # tiny random shake to escape possible plateaus
                i, j = np.random.randint(n, size=2)
                delta = np.random.rand() * a[i] * 0.03
                a[i] = max(0.0, a[i] - delta)
                a[j] += delta
                a = _project_onto_l1_ball(a, target_sum)
                step_factor *= 0.9
        iter_cnt += 1

    print(f"[+] After first gradient phase: best score = {best_global_score:.6f}")

    # ------------------------------------------------------------------
    # 4. LP‑based mixing step (if CVXPY is available)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        try:
            import cvxpy  # just to check availability
            have_cvx = True
        except Exception:
            have_cvx = False

    if have_cvx:
        f = best_global_seq.copy()
        lp_iters = 0
        while time.time() < deadline and lp_iters < 12:
            conv_f = _full_convolution(f)
            M = np.max(conv_f)
            g = _solve_lp_max_sum(f, M)
            if g is None:
                print("[lp] LP solver failed – aborting LP phase.")
                break

            # Rescale g to the same ℓ₁‑norm as f
            g = _project_onto_l1_ball(g, target_sum)

            # Convex combination – start with a modest mixing factor
            t = 0.25
            mixed = (1.0 - t) * f + t * g
            mixed = _project_onto_l1_ball(mixed, target_sum)

            mixed_score = _objective(mixed)
            print(f"[lp] iter={lp_iters}, before={_objective(f):.6f}, after={mixed_score:.6f}")

            if mixed_score < best_global_score - 1e-8:
                best_global_score = mixed_score
                best_global_seq = mixed.copy()
                f = mixed
                # gradually increase mixing weight if progress is good
                t = min(t + 0.1, 0.9)
            else:
                # no improvement – reduce mixing aggressiveness
                t *= 0.5
                if t < 1e-3:
                    break
            lp_iters += 1

        print(f"[+] After LP phase: best score = {best_global_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final fine‑grained gradient descent (hundreds of iterations)
    # ------------------------------------------------------------------
    a = best_global_seq.copy()
    step_factor = 0.2
    no_improve = 0
    iter_cnt = 0
    while time.time() < deadline and iter_cnt < 800:
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))
        max_conv = conv[k_max]

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        shift = np.zeros_like(a)
        shift[mask] = a[idx[mask]]
        grad = a + shift

        r = max_conv / (np.sum(a) ** 2)
        step = step_factor * r
        new_a = a - step * grad
        new_a = np.maximum(new_a, 0.0)
        new_a = _project_onto_l1_ball(new_a, target_sum)

        new_score = _objective(new_a)
        if new_score < best_global_score - 1e-9:
            best_global_score = new_score
            best_global_seq = new_a.copy()
            a = new_a
            no_improve = 0
            step_factor = min(step_factor * 1.07, 1.0)
            if iter_cnt % 100 == 0:
                print(f"[grad‑fin] iter={iter_cnt}, score={new_score:.6f}")
        else:
            no_improve += 1
            if no_improve % 25 == 0:
                i, j = np.random.randint(n, size=2)
                delta = np.random.rand() * a[i] * 0.04
                a[i] = max(0.0, a[i] - delta)
                a[j] += delta
                a = _project_onto_l1_ball(a, target_sum)
                step_factor *= 0.85
        iter_cnt += 1

        # early exit if we already beat the target benchmark
        if best_global_score < 1.503:
            print(f"[!] Target beaten: score={best_global_score:.6f} after {iter_cnt} final‑grad iterations.")
            break

    print(f"[+] Search finished. Best final score = {best_global_score:.6f}")
    # Convert to plain Python list for the external evaluator.
    return [float(x) for x in best_global_seq]
