def simplex_proj(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - 1) / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

Now gradient step:

def gradient_step(x, beta, eta):
    conv = np.convolve(x, x)
    # compute softmax weights
    exp_beta_conv = np.exp(beta * conv - np.max(beta * conv))  # stability
    w = exp_beta_conv / np.sum(exp_beta_conv)
    # convolution w*x
    conv_wx = np.convolve(w, x)
    g = 2 * conv_wx[len(x)-1:len(x)-1+len(x)]
    # descent
    y = x - eta * g
    # projected onto simplex
    y = simplex_projection(y)
    return y, conv.max()  # maybe also compute actual max conv for info

"""
Propose a step‑function (non‑negative list) that minimizes
evaluate_sequence(seq).  The algorithm works in two phases:

1️⃣  A fast parametric sweep over several families (power‑law,
    pure exponential, stretched exponential) for a few different
    lengths.  Each candidate is normalised to unit L1‑norm, because
    the objective is scale‑invariant.  The best candidate so far is
    kept.

2️⃣  A continuous optimisation that directly reduces the
    ℓ∞‑norm of the autocorrelation using a smooth approximation.
    For a current probability vector x we compute
        w = softmax(β·(x*x))                (β grows during the run)
    and the gradient of the soft‑max loss:
        g = 2·conv(w, x) (central slice)
    An Adam step on the simplex (projection after the update) is
    performed.  The loss and the true objective are evaluated only
    occasionally, and small random perturbations are injected to
    escape plateaus.  The loop runs until the time budget (≈1000 s)
    is exhausted or the target bound 1.503 is reached.

The final best sequence (as a Python list) is returned.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector v onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Uses the O(n log n) algorithm of
    Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    # If the vector is already non‑negative and sums to 1, return it.
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v.copy()
    # Shift to make the problem a projection onto the simplex.
    u = np.sort(v)[::-1]                     # sorted descending
    cssv = np.cumsum(u)
    rho_arr = np.where(u - (cssv - 1) / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = v - theta
    w = np.maximum(w, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:          # numerical safety
        w = np.full_like(v, 1.0 / n)
    else:
        w /= w_sum
    return w


def softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max on beta·conv (returns a probability vector)."""
    z = beta * conv
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    return exp_z / exp_z.sum()


def grad_softmax_conv(x: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max loss L = logsumexp(β·(x*x))/β.
    The derivative is 2·conv(w, x) restricted to the central slice.
    """
    n = x.shape[0]
    # full convolution of w (length 2n‑1) with x (length n)
    conv_wx = np.convolve(w, x)
    # central part corresponds to indices n‑1 … n‑1+n‑1
    return 2.0 * conv_wx[n - 1 : n - 1 + n]


def score_from_vector(x: np.ndarray) -> float:
    """
    Fast internal score: 2·n·max(conv(x,x)).
    This equals evaluate_sequence when Σx = 1.
    """
    n = x.shape[0]
    conv = np.convolve(x, x)
    return 2.0 * n * np.max(conv)

# Generation of parametric families (all vectors are L1‑normalised)

def generate_powerlaw(p: float, sigma: float, n: int) -> np.ndarray:
    a = (np.arange(n, dtype=np.float64) + sigma) ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def generate_exponential(lam: float, n: int) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def generate_stretched(p_exp: float, sigma: float, n: int) -> np.ndarray:
    a = np.exp(- (np.arange(n, dtype=np.float64) / sigma) ** p_exp)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)
