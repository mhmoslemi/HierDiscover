"""
Optimiser to lower the maximum of the self‑convolution (hence the
evaluate_sequence value).  It first scans a few parametric “U‑shaped’’
families to get a decent start, then performs exponentiated‑gradient
descent on the probability vector p (the normalised coefficients) with
projection onto the simplex.  The active set of the convolution’s
maximal entries is expanded gradually and occasional random transfers
or jitter are applied to escape plateaus.  The best sequence found
before the time budget expires is returned.
"""

import time
import numpy as np

# Helper functions ---------------------------------------------------------

def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution of a with itself using an FFT of power‑of‑two size.
    Returns the full convolution of length 2*len(a)-1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())   # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]

def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector so that its ℓ₁‑norm equals *target*."""
    s = v.sum()
    if s > 0:
        return v * (target / s)
    return np.full_like(v, target / v.shape[0])

def generate_U_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
) -> np.ndarray:
    """Generate a symmetric “U‑shaped’’ probability vector (scaled to
    *target_sum*) using the indicated functional form."""
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "cosine":
        raw = 1.0 + param * (1.0 - np.cos(np.pi * d))
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_to_sum(raw, target_sum)

def compute_gradient(p: np.ndarray, active: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max‑convolution w.r.t. the probability vector *p*
    for the given active set of convolution indices."""
    n = p.shape[0]
    g = np.zeros_like(p)
    for k in active:
        # i runs over all indices that can appear in a pair that sums to k
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        jdx = k - idx
        # each ordered pair contributes 2·p_j to the derivative
        g[idx] += 2.0 * p[jdx]
    return g

def eg_update(p: np.ndarray, g: np.ndarray, eta: float) -> np.ndarray:
    """Exponentiated‑gradient update on the simplex.
    The step is taken orthogonal to the all‑ones vector by centering *g*."""
    # centre the gradient to keep the sum invariant after the multiplicative step
    g_centered = g - np.mean(g)
    # work in log‑space to stay in the simplex automatically
    logp = np.log(p + 1e-300)
    logp_new = logp - eta * g_centered
    p_new = np.exp(logp_new)
    # re‑normalise to a probability vector
    s = p_new.sum()
    if s == 0 or not np.isfinite(s):
        # fallback – uniform distribution
        p_new = np.full_like(p, 1.0 / p.shape[0])
    else:
        p_new = p_new / s
    # enforce a tiny floor to avoid exact zeros (helps the log later)
    eps = 1e-12
    p_new = np.maximum(p_new, eps)
    p_new = p_new / p_new.sum()
    return p_new

def random_pair_transfer(p: np.ndarray, scale: float = 0.01) -> np.ndarray:
    """Move a small fraction of mass from a random index to another."""
    n = p.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * p[i]
    if delta <= 0.0:
        return p
    p[i] -= delta
    p[j] += delta
    p = p / p.sum()
    # keep a tiny floor
    eps = 1e-12
    p = np.maximum(p, eps)
    return p / p.sum()

def apply_jitter(p: np.ndarray, scale: float = 0.001) -> np.ndarray:
    """Add a small Gaussian perturbation and re‑project onto the simplex."""
    noise = np.random.randn(p.shape[0]) * scale
    p = p + noise
    p = np.clip(p, 0.0, None)
    if p.sum() == 0.0:
        p = np.full_like(p, 1.0 / p.shape[0])
    else:
        p = p / p.sum()
    eps = 1e-12
    p = np.maximum(p, eps)
    return p / p.sum()

# Main optimisation --------------------------------------------------------

def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """Search for a non‑negative height sequence that minimises
    ``evaluate_sequence``.  The routine respects the supplied time budget
    and always returns the best sequence found."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # keep a small safety margin

    # --------------------------------------------------------------------
    # Choose a length – reuse a known construction if available
    # --------------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
    else:
        n = np.random.randint(7500, 9500)   # a large but still fast size

    target_sum = np.sqrt(2.0 * n)           # the normalisation used in the paper

    # --------------------------------------------------------------------
    # Phase 1 – inexpensive parametric sweep to obtain a good start
    # --------------------------------------------------------------------
    best_p = np.full(n, 1.0 / n)            # uniform probability vector
    best_metric = float("inf")
    shape_kinds = ("linear", "quadratic", "exponential", "cosine")
    param_grids = {
        "linear": np.linspace(0.0, 5.0, 51),          # 0 … 5
        "quadratic": np.linspace(0.0, 4.0, 41),
        "exponential": np.linspace(0.0, 3.0, 31),
        "cosine": np.linspace(0.0, 3.0, 31),
    }

    print(f"[{time.time():.2f}] Parametric sweep, n={n}")
    for kind in shape_kinds:
        for p in param_grids[kind]:
            a = generate_U_shape(n, target_sum, kind=kind, param=p)
            # fast metric = 2·n·max(conv(p)), but we reuse evaluate_sequence for safety
            score = evaluate_sequence(a.tolist())
            if score < best_metric:
                best_metric = score
                best_p = a / target_sum
                print(f"  New best {kind} p={p:.3f} → {best_metric:.6f}")
            if time.time() >= deadline:
                print("Time exhausted during parametric sweep.")
                return (best_p * target_sum).tolist()

    # --------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient descent on the simplex
    # --------------------------------------------------------------------
    p = best_p.copy()
    eta = 0.5                 # initial learning rate in log‑space
    rel_tol = 0.0             # start with a strict active set
    no_improve = 0
    iteration = 0

    print(f"[{time.time():.2f}] Starting EG descent, initial metric={best_metric:.6f}")

    while time.time() < deadline:
        # compute convolution of the current probability vector
        conv = conv_self_fft(p)
        max_val = conv.max()
        metric = 2.0 * n * max_val   # this equals evaluate_sequence (up to clipping)
        iteration += 1

        if metric + 1e-12 < best_metric:
            best_metric = metric
            best_p = p.copy()
            print(f"Iter {iteration:06d} – improved metric {best_metric:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # active set (indices attaining (almost) the maximum)
        if rel_tol <= 0.0:
            active = np.array([np.argmax(conv)], dtype=np.int64)

"""
Optimiser for the self‑convolution bound (the value returned by
``evaluate_sequence``).  The algorithm works in three stages:

1. **Parametric sweep** – a few simple “U‑shaped’’ families
   (linear, quadratic, exponential, cosine) are tried.  The best
   candidate from this coarse search is kept as a starting point.

2. **Exponentiated‑gradient descent on the simplex** – we work with the
   normalised probability vector `p = a / sum(a)`.  For the current
   vector the self‑convolution is computed, the indices attaining the
   (almost) maximum are collected, a sub‑gradient of the max‑convolution
   is formed and an exponentiated‑gradient (mirror‑descent) step is taken.
   The step is orthogonal to the all‑ones direction, hence the ℓ₁‑norm
   stays fixed after re‑normalisation.  The learning rate is gradually
   decayed and the active‑set tolerance is slowly increased.

3. **Escaping plateaus** – occasional random mass transfers and a small
   Gaussian jitter are applied when progress stalls.

The whole search respects the 1000 s budget and always returns the
best sequence found.  All numbers are non‑negative, the sum is kept
positive and the length stays constant, satisfying the requirements of
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution of *a* with itself via a power‑of‑two FFT.
    Returns the full convolution of length ``2*len(a)-1``."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())   # next power of two ≥ 2n‑1
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector so that its ℓ₁‑norm equals ``target``."""
    s = v.sum()
    if s > 0.0:
        return v * (target / s)
    # fallback – uniform distribution
    return np.full_like(v, target / v.shape[0])


def generate_U_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
) -> np.ndarray:
    """Produce a symmetric “U‑shaped’’ sequence of length ``n``.
    ``kind`` ∈ {linear, quadratic, exponential, cosine}.
    ``param`` ≥ 0 controls the edge‑enhancement strength.
    The result is scaled to have ℓ₁‑norm ``target_sum``."""
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "cosine":
        raw = 1.0 + param * (1.0 - np.cos(np.pi * d))
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_to_sum(raw, target_sum)


def compute_gradient(p: np.ndarray, active: np.ndarray) -> np.ndarray:
    """Sub‑gradient of ``max(conv(p,p))`` w.r.t. the probability vector ``p``.
    ``active`` contains the convolution indices that are (almost) maximal."""
    n = p.shape[0]
    g = np.zeros_like(p)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        jdx = k - idx
        # derivative contribution = 2·p_j for each ordered pair (i,j)
        g[idx] += 2.0 * p[jdx]
    return g


def eg_update(p: np.ndarray, g: np.ndarray, eta: float) -> np.ndarray:
    """Exponentiated‑gradient (mirror‑descent) update on the simplex.
    The gradient is centred so that the update preserves the total sum."""
    # centre the gradient (orthogonal to the all‑ones vector)
    g_centered = g - np.mean(g)

    # work in log‑space – ensures positivity automatically
    logp = np.log(p + 1e-300)
    logp_new = logp - eta * g_centered
    p_new = np.exp(logp_new)

    # renormalise to a probability vector
    s = p_new.sum()
    if s == 0.0 or not np.isfinite(s):
        p_new = np.full_like(p, 1.0 / p.shape[0])
    else:
        p_new = p_new / s

    # avoid exact zeros (helps future log‑operations)
    eps = 1e-12
    p_new = np.maximum(p_new, eps)
    return p_new / p_new.sum()


def random_pair_transfer(p: np.ndarray, scale: float = 0.01) -> np.ndarray:
    """Transfer a tiny fraction of mass from one random index to another."""
    n = p.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * p[i]
    if delta <= 0.0:
        return p
    p[i] -= delta
    p[j] += delta
    p = p / p.sum()
    eps = 1e-12
    p = np.maximum(p, eps)
    return p / p.sum()


def apply_jitter(p: np.ndarray, scale: float = 0.001) -> np.ndarray:
    """Add a small Gaussian perturbation and project back onto the simplex."""
    noise = np.random.randn(p.shape[0]) * scale
    p = p + noise
    p = np.clip(p, 0.0, None)
    if p.sum() == 0.0:
        p = np.full_like(p, 1.0 / p.shape[0])
    else:
        p = p / p.sum()
    eps = 1e-12
    p = np.maximum(p, eps)
    return p / p.sum()


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence``.  The routine runs for at most ``budget_s``
    seconds and returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose a length – reuse a known construction if available
    # ------------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
    else:
        n = np.random.randint(7500, 9500)   # large but still fast

    target_sum = np.sqrt(2.0 * n)           # normalisation used in the paper

    # ------------------------------------------------------------------
    # Phase 1 – cheap parametric sweep for a decent initial point
    # ------------------------------------------------------------------
    best_p = np.full(n, 1.0 / n)            # uniform probability vector
    best_metric = float("inf")

    shape_kinds = ("linear", "quadratic", "exponential", "cosine")
    param_grids = {
        "linear": np.linspace(0.0, 5.0, 51),
        "quadratic": np.linspace(0.0, 4.0, 41),
        "exponential": np.linspace(0.0, 3.0, 31),
        "cosine": np.linspace(0.0, 3.0, 31),
    }

    print(f"[{time.time():.2f}] Parametric sweep, n={n}")
    for kind in shape_kinds:
        for p in param_grids[kind]:
            a = generate_U_shape(n, target_sum, kind=kind, param=p)
            score = evaluate_sequence(a.tolist())        # safe reference evaluation
            if score < best_metric:
                best_metric = score
                best_p = a / target_sum
                print(f"  New best {kind} (p={p:.3f}) → {best_metric:.6f}")
            if time.time() >= deadline:
                print("Time exhausted during sweep.")
                return (best_p * target_sum).tolist()

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient descent on the simplex
    # ------------------------------------------------------------------
    p = best_p.copy()
    eta = 0.8                                 # initial log‑step size
    rel_tol = 0.01                            # start with a modest active set
    no_improve = 0
    iteration = 0

    print(f"[{time.time():.2f}] Starting EG descent, metric={best_metric:.6f}")

    while time.time() < deadline:
        conv = conv_self_fft(p)               # self‑convolution of the probability vector
        max_val = conv.max()
        metric = 2.0 * n * max_val           # equals ``evaluate_sequence`` for the normalised sum
        iteration += 1

        if metric + 1e-12 < best_metric:
            best_metric = metric
            best_p = p.copy()
            print(f"Iter {iteration:06d} – improved metric {best_metric:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # ---- active set -------------------------------------------------
        if rel_tol <= 0.0:

        else:
            active = np.where(conv >= max_val * (1.0 - rel_tol))[0]

        # ---- gradient and update ----------------------------------------
        g = compute_gradient(p, active)
        p = eg_update(p, g, eta)

        # ---- occasional random transfer ---------------------------------
        if np.random.rand() < 0.02:
            p = random_pair_transfer(p, scale=0.005)

        # ---- schedule: decay eta, enlarge active set --------------------
        if iteration % 400 == 0:
            eta *= 0.85
            if eta < 1e-3:
                eta = 1e-3
            rel_tol = min(0.07, rel_tol + 0.008)

        # ---- stagnation handling -----------------------------------------
        if no_improve >= 2500:
            p = apply_jitter(p, scale=0.0015)
            no_improve = 0
            print(f"Iter {iteration:06d} – jitter applied")

    # ------------------------------------------------------------------
    # Phase 3 – final short random perturbation pass (optional)
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] EG phase finished; best metric {best_metric:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence (rescaled to the required ℓ₁‑norm)
    # ------------------------------------------------------------------
    best_a = best_p * target_sum
    return best_a.tolist()
