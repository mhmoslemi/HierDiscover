def _lp_minimize_max(f, target_sum):
    n = len(f)
    rows = np.zeros((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        lo = max(0, k - (n-1))
        hi = min(k, n-1)
        i = np.arange(lo, hi+1)
        rows[k, i] = f[k - i]
    # A_ub
    A_ub = np.hstack([rows, -np.ones((2*n-1, 1), dtype=np.float64)])
    b_ub = np.zeros(2*n-1)
    # Equality sum constraint
    A_eq = np.hstack([np.ones((1, n), dtype=np.float64), np.zeros((1, 1), dtype=np.float64)])
    b_eq = np.array([target_sum])
    c = np.zeros(n+1, dtype=np.float64)
    c[-1] = 1.0
    bounds = [(0.0, None)]*n + [(0.0, None)]
    try:
        from scipy.optimize import linprog
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception as e:
        print("[LP] Exception", e)
        return None
    if not res.success:
        print("[LP] failed", res.message)
        return None
    x = res.x
    b = x[:n]
    M = x[-1]
    return b, M

"""
Hybrid search for a non‑negative step function with minimal auto‑convolution peak.

Key ideas
---------
* Build a diverse pool of initial candidates (power‑law, exponential, random,
  and the previous best if available) and keep the best one.
* Run a few short annealed gradient‑descent runs to improve the candidate.
* Apply simple orderings (alternating, even‑spread) that are known to lower
  the convolution peak for a given multiset of values.
* Down‑scale the sequence (≈1000‑1500 elements) and solve a *linear program*
  that **directly minimizes the maximal convolution value** `M` under the
  fixed ℓ¹‑norm constraint.  This replaces the earlier random‑objective LP.
  The LP is solved with SciPy’s HiGHS backend.
* Replace the current sequence by the LP solution (it already respects the
  ℓ¹‑norm) and repeat a few LP iterations until no further improvement.
* Upscale to a larger length, optionally reorder again, and finish with a
  short sub‑gradient descent followed by a deterministic mass‑transfer
  refinement.
* Trim trailing near‑zero entries and return the final list.

The whole routine respects the provided time budget (`budget_s` seconds) and
logs intermediate scores.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)² (for Σa=√(2n) this equals maxconv)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# ℓ¹‑simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised to the required ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex (Dirichlet‑like)."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """
    Spread the large values uniformly by placing them on a cyclic sequence
    with a step coprime to the length.
    """
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]          # descending
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Soft‑max gradient (used by annealed GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent_v2_ext(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      max_beta: float = 1e5,
                                      step_factor: float = 0.7,
                                      gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Momentum projected GD on the soft‑max approximation of max‑conv."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD_ext it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that uses the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]
        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val

# Deterministic mass‑transfer refinement (simulated annealing)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy mass moves with a decaying Metropolis acceptance rule."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear program that directly minimizes the convolution peak
# ----------------------------------------------------------------------


def _lp_minimize_max(f: np.ndarray, target_sum: float,
                     rng: np.random.Generator):
    """
    Solve a LP that minimizes M = max_k (f * b)_k  subject to
        f * b <= M,
        b >= 0,
        Σ b = target_sum.
    Returns (b, M) on success, else (None, None).
    """
    n = f.shape[0]

    # Build the (2n‑1) × n convolution matrix A where row k contains
    # the coefficients of b in (f * b)_k .
    rows = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        rows[k, idx] = f[k - idx]

    # Inequalities: A b - M <= 0
    A_ub = np.hstack([rows, -np.ones((2 * n - 1, 1), dtype=np.float64)])
    b_ub = np.zeros(2 * n - 1, dtype=np.float64)

    # Equality: sum(b) = target_sum
    A_eq = np.hstack([np.ones((1, n), dtype=np.float64), np.zeros((1, 1), dtype=np.float64)])
    b_eq = np.array([target_sum], dtype=np.float64)

    # Objective: minimise M
    c = np.zeros(n + 1, dtype=np.float64)
    c[-1] = 1.0

    bounds = [(0.0, None)] * n + [(0.0, None)]

    try:
        from scipy.optimize import linprog
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
    except Exception as e:
        print(f"[LP] exception: {e}")
        return None, None

    if not res.success:
        print(f"[LP] failure: {res.message}")
        return None, None

    x = res.x
    b = x[:n]
    M = x[-1]
    # Numerical safety: re‑project onto the simplex (should already be satisfied)
    b = _project_to_simplex(b, target_sum)
    return b, M

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid search for a non‑negative step function with a low evaluation score.
    The routine follows:
        1) Diverse seeding + short annealed GD.
        2) Reordering to lower the convolution peak.
        3) Down‑scale, then iteratively solve a LP that *directly* minimises
           the maximal convolution value.
        4) Upscale, final sub‑gradient polishing and deterministic mass‑transfer.
    All steps respect the ℓ¹‑simplex constraint Σa = √(2·n).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size for the exploratory phase
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n_expl = 12000
    elif budget_s >= 400:
        n_expl = 9000
    else:
        n_expl = 6000
    target_expl = np.sqrt(2.0 * n_expl)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Build a pool of seeds
    # ------------------------------------------------------------------
    init_pool = []

    # power‑law families
    for p in np.linspace(0.2, 1.5, 13):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))

    # exponential families
    for lam in np.linspace(0.001, 0.025, 8):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))

    # random points on the simplex
    for _ in range(12):
        init_pool.append(_random_simplex(n_expl, target_expl, rng))

    # previous best (if any)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_expl))

    # small jittered variations of everything we have so far
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.02, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    # ------------------------------------------------------------------
    # Pick the best seed (by the true objective)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    for seq in init_pool:
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
    print(f"[{time.time():.2f}] Initial best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Short annealed‑gradient runs to move off local plateaus
    # ------------------------------------------------------------------
    while time.time() < deadline - 30:
        # jitter around current best
        seed = best_seq + rng.normal(scale=target_expl * 0.005, size=n_expl)
        seed = np.maximum(seed, 0.0)
        seed = _project_to_simplex(seed, target_expl)

        # run a very short GD (≈8‑12 s)
        gd_deadline = time.time() + min(12.0, deadline - time.time() - 15.0)
        if gd_deadline <= time.time():
            break
        a_opt, val_opt = _annealed_gradient_descent_v2_ext(
            seed, gd_deadline, target_expl,
            max_beta=3e5, step_factor=0.7, gamma=0.99,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] GD new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Simple reorderings (they are cheap and often beneficial)
    # ------------------------------------------------------------------
    ordering_candidates = [
        best_seq,
        _alternating_order(best_seq),
        _even_spread_order(best_seq),
    ]
    best_seq = min(ordering_candidates, key=_score)
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] After ordering score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Down‑scale for the LP stage
    # ------------------------------------------------------------------
    n_lp = min(1500, max(800, n_expl // 6))
    target_lp = np.sqrt(2.0 * n_lp)
    seq_lp = _resize_sequence(best_seq, n_lp)
    seq_lp = _project_to_simplex(seq_lp, target_lp)

    # ------------------------------------------------------------------
    # LP refinement: minimise the maximal convolution value directly
    # ------------------------------------------------------------------
    lp_iter = 0
    while time.time() < deadline - 20 and lp_iter < 12:
        cur_M = _max_convolution_fft(seq_lp)
        b_opt, M_opt = _lp_minimize_max(seq_lp, target_lp, rng)
        if b_opt is None:
            break
        # Accept if we really improved the peak
        if M_opt < cur_M - 1e-8:
            seq_lp = b_opt
            seq_lp = _project_to_simplex(seq_lp, target_lp)
            cur_score = _score(seq_lp)
            print(f"[{time.time():.2f}] LP iter {lp_iter} M={M_opt:.6f} score={cur_score:.6f}")
        else:
            print(f"[{time.time():.2f}] LP iter {lp_iter} no improvement (M={M_opt:.6f})")
            break
        lp_iter += 1

    # ------------------------------------------------------------------
    # Optional brief GD polishing on the small problem
    # ------------------------------------------------------------------
    if time.time() < deadline - 12:
        gd_dead = time.time() + min(8.0, deadline - time.time() - 10.0)
        a_opt, val_opt = _annealed_gradient_descent_v2_ext(
            seq_lp, gd_dead, target_lp,
            max_beta=1e5, step_factor=0.6, gamma=0.98,
        )
        if val_opt < _score(seq_lp):
            seq_lp = a_opt.copy()
            print(f"[{time.time():.2f}] Post‑LP GD score {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Upscale to a larger length for the final polishing stages
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 80:
        n_final = 18000
    elif remaining > 40:
        n_final = 14000
    else:
        n_final = max(n_lp, n_expl)
    target_final = np.sqrt(2.0 * n_final)
    seq_big = _resize_sequence(seq_lp, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # ------------------------------------------------------------------
    # Reordering after up‑scaling
    # ------------------------------------------------------------------
    cand = [
        seq_big,
        _alternating_order(seq_big),
        _even_spread_order(seq_big),
    ]
    seq_big = min(cand, key=_max_convolution_fft)
    best_score = _score(seq_big)
    print(f"[{time.time():.2f}] Upscaled start score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Sub‑gradient polishing on the full‑size problem
    # ------------------------------------------------------------------
    if time.time() < deadline - 6:
        sub_dead = deadline - 4
        a_sub, val_sub = _subgradient_descent(
            seq_big, sub_dead, target_final,
            step_factor=0.45, gamma=0.995,
        )
        if val_sub < best_score:
            best_score = val_sub
            seq_big = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑grad best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final deterministic mass‑transfer pass
    # ------------------------------------------------------------------
    if time.time() < deadline - 2:
        mt_dead = time.time() + min(5.0, deadline - time.time() - 1.0)
        a_mt, val_mt = _deterministic_mass_transfer(
            seq_big, mt_dead, target_final,
            start_temp=0.25, end_temp=1e-5, low_frac=0.05,
        )
        if val_mt < best_score:
            best_score = val_mt
            seq_big = a_mt.copy()
            print(f"[{time.time():.2f}] Final MT best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # ------------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Optional final evaluation (purely for logging)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # store for possible reuse in the next invocation
    globals()["height_sequence_1"] = final_seq
    return final_seq
