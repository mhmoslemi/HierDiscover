def _pairwise_opt_transfer_contrib(a0, deadline, target_sum, donor_frac=0.02, receiver_frac=0.02, min_improve=1e-7):

    a = a0.copy()
    n = a.shape[0]
    best_score = _score(a)
    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k = int(conv.argmax())
        # compute contributions
        # compute partner array using vectorized indexing
        idx = np.arange(n)
        k_minus_idx = k - idx
        valid = (k_minus_idx >= 0) & (k_minus_idx < n)
        partner = np.zeros_like(a)
        partner[valid] = a[k_minus_idx[valid]]
        contributions = a * partner

        nd = max(1, int(donor_frac * n))
        nr = max(1, int(receiver_frac * n))
        donor_idx = np.argsort(-contributions)[:nd]
        receiver_idx = np.argsort(contributions)[:nr]

        # compute padded base
        pad_len = n - 1
        base = np.concatenate([np.zeros(pad_len, dtype=np.float64), a, np.zeros(pad_len, dtype=np.float64)])

        # precompute shifted views
        needed = np.unique(np.concatenate((donor_idx, receiver_idx)))
        shifted = {}
        for i in needed:
            shifted[i] = base[i:i + 2*n - 1]

        best_new_max = max_val
        best_pair = None
        best_delta = 0.0

        for i in donor_idx:
            shift_i = shifted[i]
            a_i = a[i]
            if a_i <= 0:
                continue
            for j in receiver_idx:
                if i == j:
                    continue
                shift_j = shifted[j]
                d = 2.0 * (shift_j - shift_i)
                d_k = d[k]
                if d_k >= 0:
                    continue
                # compute feasible delta
                mask = d > d_k
                if np.any(mask):
                    denom = d[mask] - d_k  # positive
                    numer = max_val - conv[mask]
                    # need delta <= numer / denom for each mask index
                    feasible = numer / denom
                    delta_max = min(feasible.min(), a_i)  # donor available
                else:
                    delta_max = a_i
                if delta_max <= 1e-12:
                    continue
                new_max = max_val + d_k * delta_max
                if new_max < best_new_max - min_improve:
                    best_new_max = new_max
                    best_pair = (i, j)
                    best_delta = delta_max
        if best_pair is None:
            break
        i, j = best_pair
        a[i] -= best_delta
        a[j] += best_delta
        # project occasionally
        if int(time.time() * 10) % 30 == 0:
            a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            # log
            print(f"[{time.time():.2f}] Pairwise contrib move i={i} j={j:.0f} Δ={best_delta:.3e} new_score={best_score:.6f}")

    return a, best_score

def _simulated_annealing(a0, deadline, target_sum, temp_start=0.01, temp_end=1e-6, steps=5000):
    a = a0.copy()
    cur_score = _score(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    total_steps = steps
    # compute schedule: exponential cooling
    for step in range(total_steps):
        if time.time() >= deadline:
            break
        # temperature linearly decreasing in log scale
        t = temp_start * ((temp_end/temp_start) ** (step/total_steps))
        # pick donor from top fraction 0.02, receiver from bottom fraction 0.02
        k_top = max(1, int(0.02 * n))
        k_low = max(1, int(0.02 * n))
        idx_desc = np.argsort(-a)
        idx_asc = np.argsort(a)
        donor = idx_desc[rng.integers(k_top)]
        receiver = idx_asc[rng.integers(k_low)]
        if donor == receiver:
            continue
        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0:
            continue
        a[donor] -= delta
        a[receiver] += delta
        # sum unchanged => maintain target sum
        # compute new score
        new_score = _score(a)
        diff = new_score - cur_score
        if diff < 0 or rng.random() < np.exp(-diff / max(t, 1e-12)):
            # accept
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.2f}] SA accepted new best {best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
    return best_seq, best_score

"""
Advanced search for a non‑negative step function with a low
self‑convolution peak.

Workflow
--------
0. Choose a problem size n (≈12‑20k) according to the time budget.
1. Generate a diverse pool of seed sequences (power‑law, tilted,
   linear, Gaussian, random jitter) and keep the best one.
2. Adam‑style projected gradient descent on a soft‑max approximation of
   the maximum convolution (several restarts, slowly increasing the
   soft‑max temperature β).
3. Exact deterministic mass‑transfer (the “DMT2” refinement) to push
   the peak down.
4. Pairwise optimal mass transfer – donors are chosen by the magnitude
   of their contribution to the current peak, receivers by the smallest
   contribution.  The analytically maximal admissible δ is moved.
5. Simulated‑annealing style random mass transfers to escape possible
   local minima.
6. Final high‑β Adam polishing.
7. Trim trailing near‑zero entries and return the list.

All steps respect the given time budget and use only NumPy (no
external solvers).  The returned sequence is guaranteed to be
non‑negative and its sum is normalised to √(2 n), so the score equals
the maximum entry of the convolution.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Objective (score) – with Σa = √(2 n) the denominator equals 2 n,
# therefore the score equals the maximal convolution entry.
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex Σx = s, x ≥ 0."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families of seed sequences
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float,
                            target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _linear_sequence(n: int, target_sum: float) -> np.ndarray:
    """Decreasing linear shape that hits zero roughly halfway."""
    m = n // 2
    raw = np.arange(m, 0, -1, dtype=np.float64)
    a = np.concatenate([raw, np.zeros(n - m, dtype=np.float64)])
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, target_sum: float) -> np.ndarray:
    sigma = n / 6.0
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((x - n / 2) / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max approximation of the max (used by Adam‑GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max of the convolution w.r.t. a."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # convolution of a_rev with p gives the gradient (times 2)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on the soft‑max objective
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Perform several short Adam runs starting from jittered versions of `init`."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=2e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (DMT2)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """Greedy move of a small amount of mass from a donor to a receiver
    while guaranteeing that the current peak of the convolution stays the
    maximum entry.  Repeated until the gap to the second‑largest entry is
    below `tol`. """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second highest entry
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = int(idx_max[0])        # pick one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]        # a_{k-i}

        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # Receiver: smallest partner (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.7, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Global random mass‑transfer (exploratory)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """Very cheap random moves (donor from top fraction, receiver from
    bottom fraction) with immediate projection onto the simplex."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random move new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Pairwise optimal mass‑transfer using contributions to the peak
# ----------------------------------------------------------------------


def _pairwise_opt_transfer_contrib(a0: np.ndarray,
                                   deadline: float,
                                   target_sum: float,
                                   donor_frac: float = 0.02,
                                   receiver_frac: float = 0.02,
                                   min_improve: float = 1e-7) -> tuple[np.ndarray, float]:
    """Greedy pairwise transfer.  Donors are selected by their
    contribution a_i·a_{k-i} to the current maximal convolution entry,
    receivers by the smallest contribution.  The analytically largest
    feasible δ is moved."""
    a = a0.copy()
    n = a.shape[0]
    best_score = _score(a)

    pad_len = n - 1

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k = int(conv.argmax())

        # contribution of each coefficient to the peak at k
        idx = np.arange(n)
        k_minus_idx = k - idx
        valid = (k_minus_idx >= 0) & (k_minus_idx < n)
        partner = np.zeros_like(a)
        partner[valid] = a[k_minus_idx[valid]]
        contributions = a * partner

        nd = max(1, int(donor_frac * n))
        nr = max(1, int(receiver_frac * n))
        donor_idx = np.argsort(-contributions)[:nd]
        receiver_idx = np.argsort(contributions)[:nr]

        # padded version of a for fast shifted views
        base = np.concatenate([np.zeros(pad_len, dtype=np.float64),
                               a,
                               np.zeros(pad_len, dtype=np.float64)])

        needed = np.unique(np.concatenate((donor_idx, receiver_idx)))
        shifted = {i: base[i:i + 2 * n - 1] for i in needed}

        best_new_max = max_val
        best_pair = None
        best_delta = 0.0

        for i in donor_idx:
            shift_i = shifted[i]
            a_i = a[i]
            if a_i <= 0.0:
                continue
            for j in receiver_idx:
                if i == j:
                    continue
                shift_j = shifted[j]

                d = 2.0 * (shift_j - shift_i)        # change of conv per unit δ
                d_k = d[k]
                if d_k >= 0.0:                       # would raise the peak – not useful
                    continue

                mask = d > d_k
                if np.any(mask):
                    denom = d[mask] - d_k               # always >0
                    numer = max_val - conv[mask]
                    feasible = numer / denom
                    delta_max = min(feasible.min(), a_i)
                else:
                    delta_max = a_i

                if delta_max <= 1e-12:
                    continue

                new_max = max_val + d_k * delta_max
                if new_max < best_new_max - min_improve:
                    best_new_max = new_max
                    best_pair = (i, j)
                    best_delta = delta_max

        if best_pair is None:
            break

        i, j = best_pair
        a[i] -= best_delta
        a[j] += best_delta

        # occasional re‑projection to keep Σa exact
        if int(time.time() * 10) % 50 == 0:
            a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            print(f"[{time.time():.2f}] Pairwise i={i} j={j} Δ={best_delta:.3e} new_score={best_score:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing style random mass transfer
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         steps: int = 8000,
                         temp_start: float = 0.01,
                         temp_end: float = 1e-6) -> tuple[np.ndarray, float]:
    """Random moves accepted with a Metropolis criterion.  Moves keep the
    ℓ₁‑norm fixed, so only a projection is needed when the move is
    rejected (re‑add the removed mass)."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    for step in range(steps):
        if time.time() >= deadline:
            break

        # exponential cooling schedule
        t = temp_start * ((temp_end / temp_start) ** (step / steps))

        # pick donor / receiver from tails of the distribution
        top_k = max(1, int(0.02 * n))
        low_k = max(1, int(0.02 * n))

        donor = np.argsort(-a)[rng.integers(top_k)]
        receiver = np.argsort(a)[rng.integers(low_k)]
        if donor == receiver:
            continue

        delta = rng.random() * a[donor] * 0.1
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta

        new_score = _score(a)
        diff = new_score - best_score
        if diff < 0 or rng.random() < np.exp(-diff / max(t, 1e-12)):
            # accept
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA new best={best_score:.6f}")
        else:
            # reject – restore
            a[donor] += delta
            a[receiver] -= delta

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """Search for a non‑negative step function that minimises the maximum
    self‑convolution.  The routine respects the provided time budget and
    returns the best sequence found as a plain Python list."""
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size – larger n gives more freedom but costs a bit more time
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 14000
    else:
        n = 10000

    target_sum = np.sqrt(2.0 * n)   # Σa = √(2 n)  →  score = max convolution

    # ------------------------------------------------------------------
    # Phase 0 – generate diverse seeds and keep the best
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    # power‑law family
    for p in np.linspace(0.25, 0.90, 27):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # exponentially tilted power‑law
    q_grid = np.linspace(0.0, 6.0, 15)
    for p in np.linspace(0.25, 0.90, 15):
        for q in q_grid:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # linear decreasing shape
    seq = _linear_sequence(n, target_sum)
    val = _score(seq)
    if val < best_score:
        best_score, best_seq = val, seq.copy()

    # Gaussian bump
    seq = _gaussian_sequence(n, target_sum)
    val = _score(seq)
    if val < best_score:
        best_score, best_seq = val, seq.copy()

    # incorporate a previously found construction if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score, best_seq = val, resized.copy()

    # jittered population around current best
    for _ in range(6):
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _score(cand)
        if val < best_score:
            best_score, best_seq = val, cand.copy()

    print(f"[{time.time():.2f}] After seeding best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam GD with multiple restarts
    # ------------------------------------------------------------------
    gd_deadline = time.time() + (deadline - time.time()) * 0.30
    while time.time() < gd_deadline - 5.0:
        start_vec = best_seq + rng.normal(scale=target_sum * 0.01, size=n)
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _gradient_descent_with_restarts(start_vec,
                                                         target_sum,
                                                         gd_deadline)
        if val_opt < best_score - 1e-9:
            best_score, best_seq = val_opt, a_opt.copy()
            print(f"[{time.time():.2f}] ADAM restart improved best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Deterministic mass‑transfer (core refinement)
    # ------------------------------------------------------------------
    while time.time() < deadline - 8.0:
        a_dm, val_dm = _deterministic_mass_transfer_exact2(best_seq,
                                                          deadline,
                                                          target_sum,
                                                          tol=1e-12)
        if val_dm < best_score - 1e-9:
            best_score, best_seq = val_dm, a_dm.copy()
            print(f"[{time.time():.2f}] DMT2 improved best_score={best_score:.6f}")

        # one cheap random move to possibly escape a plateau
        a_rand, val_rand = _global_random_mass_transfer(best_seq,
                                                        deadline,
                                                        target_sum)
        if val_rand < best_score - 1e-9:
            best_score, best_seq = val_rand, a_rand.copy()
            print(f"[{time.time():.2f}] Random move improved best_score={best_score:.6f}")

        if time.time() > deadline - 5.0:
            break

    # ------------------------------------------------------------------
    # Phase C – Pairwise optimal mass transfer (contribution based)
    # ------------------------------------------------------------------
    if time.time() < deadline - 4.0:
        a_pair, val_pair = _pairwise_opt_transfer_contrib(best_seq,
                                                          deadline,
                                                          target_sum,
                                                          donor_frac=0.02,
                                                          receiver_frac=0.02)
        if val_pair < best_score:
            best_score, best_seq = val_pair, a_pair.copy()
            print(f"[{time.time():.2f}] Pairwise contrib improved best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase D – Simulated annealing random walk
    # ------------------------------------------------------------------
    if time.time() < deadline - 3.0:
        a_sa, val_sa = _simulated_annealing(best_seq,
                                            deadline,
                                            target_sum,
                                            steps=8000)
        if val_sa < best_score:
            best_score, best_seq = val_sa, a_sa.copy()
            print(f"[{time.time():.2f}] SA improved best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase E – Final high‑β Adam polishing
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                deadline,
                                                target_sum,
                                                max_beta=1e7,
                                                step_factor=0.6,
                                                gamma=0.995)
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing – trim trailing near‑zero entries
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final evaluation (harmless if evaluate_sequence is not defined)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
