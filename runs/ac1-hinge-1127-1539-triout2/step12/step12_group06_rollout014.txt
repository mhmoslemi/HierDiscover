"""
Hybrid search for a non‑negative step function minimizing the max‑autocorrelation.
The pipeline:

1.  Build a diverse pool of seed shapes (power‑law, exponential,
    stretched‑exponential, sinusoidal, random Dirichlet, linear).
2.  Perform many short projected sub‑gradient (max‑conv) runs on a modest
    length (≈4000) to discover promising shapes.
3.  Upscale the best shape to a larger length (≈12000‑15000) and apply
    ordering heuristics (alternating placement) that tend to flatten the
    convolution peak.
4.  Refine the enlarged vector with longer sub‑gradient descent,
    high‑β annealed soft‑max gradient descent, a targeted permutation
    optimisation, and a deterministic mass‑transfer pass.
5.  Finish with a simple value‑annealing (jitter) loop.
6.  Trim trailing zeros and return the best sequence found.
The code respects the 1000 s budget, dynamically allocating time to each
stage and constantly records the best objective (which equals the peak
convolution entry once the sum is normalised to √(2n)).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Euclidean projection onto a simplex Σx_i = s, x_i ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative vectors with sum s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators (all return a vector with sum == target_sum)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _stretched_exponential_sequence(n: int, lam: float, q: float,
                                    target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * (i ** q))
    a *= target_sum / a.sum()
    return a


def _sinusoidal_sequence(n: int, power: float, target_sum: float) -> np.ndarray:
    # sin(pi * (i+1)/(n+1)) is positive for i=0..n-1
    a = np.sin(np.pi * (np.arange(1, n + 1, dtype=np.float64) / (n + 1))) ** power
    a *= target_sum / a.sum()
    return a


def _linear_decreasing_sequence(n: int, target_sum: float) -> np.ndarray:
    a = np.linspace(1.0, 0.0, n, dtype=np.float64)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Ordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place largest to left, 2nd largest to right, 3rd to next left …."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    to_left = True
    for v in sorted_vals:
        if to_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        to_left = not to_left
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


# ----------------------------------------------------------------------
# Scoring – the objective (max convolution entry after normalisation)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Return the objective value (equivalent to 2·n·max(b)/(sum(a)²))."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Exact sub‑gradient of the peak convolution entry
# ----------------------------------------------------------------------


def _max_conv_and_subgrad(a: np.ndarray):
    """
    Returns (conv, max_val, kmax, subgrad) where subgrad is the gradient
    of the peak entry of the convolution w.r.t. a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    kmax = int(np.argmax(conv))
    max_val = conv[kmax]

    grad = np.zeros_like(a)
    lo = max(0, kmax - (n - 1))
    hi = min(kmax, n - 1)
    idx = np.arange(lo, hi + 1)
    partner = a[kmax - idx]
    grad[lo:hi + 1] = 2.0 * partner
    return conv, max_val, kmax, grad


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on max(conv(a,a))
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.6,
                         gamma: float = 0.95) -> tuple[np.ndarray, float]:
    """Momentum projected sub‑gradient descent on the max‑convolution peak."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0
    momentum = np.zeros_like(a)

    while time.time() < deadline:
        _, _, _, grad = _max_conv_and_subgrad(a)

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            lr = step_factor * target_sum / norm
            a -= lr * momentum
        else:
            a -= step_factor * target_sum * grad / (np.linalg.norm(grad) + 1e-12)

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] SubGD it={it:5d}  score={best_val:.6f}")

        it += 1

    return best, best_val


# ----------------------------------------------------------------------
# Soft‑max (smooth max) gradient descent with annealing of β
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of β‑soft‑max of the convolution (returns grad and full conv)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * (a_rev) ⋆ p
    a_rev = a[::-1]
    grad = 2.0 * np.fft.irfft(np.fft.rfft(a_rev, n=size) *
                             np.fft.rfft(p, n=size), n=size)[: 2 * n - 1]
    grad = grad[n - 1: n - 1 + n]  # central n entries
    return grad, b


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e6,
                               step_factor: float = 0.5,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Momentum projected gradient descent on the soft‑max approximation."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # jitter
        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Annealed it={it:5d} β={beta:6.1f}  score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (greedy donor→receiver)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.4,
                                end_temp: float = 1e-5,
                                low_frac: float = 0.04) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]          # contribution to the peak
        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0.0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d}  new best={best_score:.6f}")
        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2500 == 0 and it > 0:
            # occasional jitter to keep moving
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted permutation search (swap indices that affect the peak)
# ----------------------------------------------------------------------


def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.15,
                                 end_temp: float = 1e-5,
                                 top_frac: float = 0.04,
                                 low_frac: float = 0.04) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)

        if donor == receiver:
            continue

        a[donor], a[receiver] = a[receiver], a[donor]
        new_score = _score(a)

        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] PermSearch new best={best_score:.6f}")

        if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / max(temp, 1e-12)):
            # keep swap
            pass
        else:
            # revert
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with a low benchmark score.
    The function respects the budget and stores the best vector globally.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # 1. Choose a moderate length for rapid exploration
    # ------------------------------------------------------------------
    n_mid = 4000
    target_mid = np.sqrt(2.0 * n_mid)

    # ------------------------------------------------------------------
    # 2. Build a diverse pool of seed sequences
    # ------------------------------------------------------------------
    seeds = []

    # random Dirichlet‑like
    for _ in range(20):
        a = rng.random(n_mid)
        a = a * target_mid / a.sum()
        seeds.append(a)

    # power‑law
    for p in np.linspace(0.2, 1.5, 14):
        seeds.append(_power_law_sequence(n_mid, float(p), target_mid))

    # exponential
    for lam in np.linspace(0.0005, 0.03, 15):
        seeds.append(_exponential_sequence(n_mid, float(lam), target_mid))

    # stretched exponential
    for lam in [0.001, 0.005, 0.012]:
        for q in [0.5, 1.0, 1.5, 2.0]:
            seeds.append(_stretched_exponential_sequence(n_mid, float(lam),
                                                         float(q), target_mid))

    # sinusoidal shapes
    for power in [0.5, 1.0, 2.0, 3.0]:
        seeds.append(_sinusoidal_sequence(n_mid, float(power), target_mid))

    # linear decreasing
    seeds.append(_linear_decreasing_sequence(n_mid, target_mid))

    # previous best (if any) – resized to current length
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        seeds.append(_resize_sequence(prev, n_mid))

    # jittered copies of every seed to increase diversity
    jitter_scale = target_mid * 0.008
    for seq in list(seeds):
        j = seq + rng.normal(scale=jitter_scale, size=n_mid)
        j = np.maximum(j, 0.0)
        j = _project_to_simplex(j, target_mid)
        seeds.append(j)

    # ------------------------------------------------------------------
    # 3. Exploration: many short sub‑gradient runs on the moderate size
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # reserve time for the later intensive phases
    reserve = 250.0  # seconds we keep for up‑scaling & polishing

    while time.time() < deadline - reserve:
        seed = seeds[rng.integers(len(seeds))]
        run_dead = time.time() + 12.0   # 12 s per short run
        if run_dead > deadline - reserve:
            break
        seq, val = _subgradient_descent(seed, run_dead, target_mid,
                                        step_factor=0.6, gamma=0.95)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Exploration new best={best_score:.6f}")

    # If nothing succeeded, fall back to a simple power‑law shape
    if best_seq is None:
        best_seq = _power_law_sequence(n_mid, 0.7, target_mid)
        best_score = _score(best_seq)

    # ------------------------------------------------------------------
    # 4. Upscale to a larger length (more degrees of freedom)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 200.0:
        n_final = 15000
    elif remaining > 120.0:
        n_final = 12000
    else:
        n_final = n_mid

    target_final = np.sqrt(2.0 * n_final)
    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # try a few simple deterministic reorderings and keep the best
    candidates = [seq_big,
                  _alternating_order(seq_big),
                  _reversed_alternating(seq_big),
                  _alternating_order(_reversed_alternating(seq_big))]
    best_seq = seq_big
    best_score = _score(seq_big)
    for cand in candidates[1:]:
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
    print(f"[{time.time():.2f}] Upscaled n={n_final} start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Long sub‑gradient refinement on the enlarged vector
    # ------------------------------------------------------------------
    if time.time() < deadline - 130.0:
        sub_dead = time.time() + min(70.0, deadline - time.time() - 120.0)
        seq, sc = _subgradient_descent(best_seq, sub_dead, target_final,
                                       step_factor=0.6, gamma=0.95)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.2f}] SubGD (big) new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. High‑β annealed soft‑max gradient descent (fine polishing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 100.0:
        anneal_dead = time.time() + min(35.0, deadline - time.time() - 80.0)
        seq, sc = _annealed_gradient_descent(best_seq, anneal_dead,
                                             target_final,
                                             max_beta=5e6,
                                             step_factor=0.5,
                                             gamma=0.99)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.2f}] Annealed GD new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Targeted permutation search (re‑ordering only)
    # ------------------------------------------------------------------
    if time.time() < deadline - 70.0:
        perm_dead = time.time() + min(25.0, deadline - time.time() - 50.0)
        seq, sc = _permutation_targeted_search(best_seq, perm_dead,
                                                start_temp=0.2,
                                                end_temp=1e-5,
                                                top_frac=0.05,
                                                low_frac=0.05)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.2f}] Permutation search new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Deterministic mass‑transfer refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 40.0:
        mt_dead = time.time() + min(20.0, deadline - time.time() - 30.0)
        seq, sc = _deterministic_mass_transfer(best_seq, mt_dead,
                                               target_final,
                                               start_temp=0.4,
                                               end_temp=1e-5,
                                               low_frac=0.04)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.2f}] Mass‑transfer new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 9. Final small‑scale jitter annealing on the values (optional)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        a = best_seq.copy()
        cur_score = _score(a)

        start_temp = 0.25
        end_temp = 1e-4
        start = time.time()
        while time.time() < deadline - 5.0:
            elapsed = time.time() - start
            tau = elapsed / (deadline - start - 5.0)
            temp = start_temp * (end_temp / start_temp) ** tau

            jitter = rng.normal(scale=target_final * 0.001 * temp, size=a.shape)
            a_prop = a + jitter
            a_prop = np.maximum(a_prop, 0.0)
            a_prop = _project_to_simplex(a_prop, target_final)

            s = _score(a_prop)
            if s < cur_score or rng.random() < np.exp(-(s - cur_score) / max(temp, 1e-12)):
                a = a_prop
                cur_score = s
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[{time.time():.2f}] Jitter anneal new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 10. Trim trailing zeros for compactness
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # 11. Log final evaluation (optional)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # store for possible reuse in later calls
    globals()["height_sequence_1"] = final_seq
    return final_seq
