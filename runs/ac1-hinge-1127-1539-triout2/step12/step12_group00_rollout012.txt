def _permute_anneal(seq, deadline):
    a = seq.copy()
    cur_val = _max_convolution_fft(a)
    best_seq = a.copy()
    best_val = cur_val
    rng = np.random.default_rng()
    n = len(a)
    # Use temperature schedule: start_temp = cur_val*0.2? not important.
    start_temp = cur_val * 0.2
    end_temp = cur_val * 0.001
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time # we need start_time param
        tau = (elapsed) / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau
        # Random swap indices
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j:
            continue
        # Swap values
        b = a.copy()
        b[i], b[j] = b[j], b[i]
        new_val = _max_convolution_fft(b)
        # Accept if better or with probability
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / max(temp, 1e-12)):
            a = b
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = b.copy()
                print(f"[{time.time():.2f}] PermAnneal best={best_val:.6f} it={it}")
        it += 1
    return best_seq, best_val

"""
Hybrid optimizer with aggressive LP‑alternation, permutation annealing and
simplex‑projected mass‑transfer.  The goal is to minimise the peak of the
self‑convolution (equivalently the evaluation score).  The algorithm proceeds
in stages:

1.  Build a diverse pool of decaying sequences (power‑law, exponential,
    random simplex) and optionally reuse the previous best construction.
2.  Perform many short annealed‑gradient runs to obtain a good initial
    candidate.
3.  Refine with a deterministic mass‑transfer (Metropolis‑style) move set.
4.  Down‑scale to a moderate size (≈1500–2000) and run several iterations of
    a linear‑programming alternation: solve a LP that maximises a random linear
    functional under the current max‑convolution bound, mix the LP solution
    with the current vector and optionally polish with a few GD steps.
5.  Apply a simulated‑annealing permutation search on the down‑scaled vector
    (swapping entries) to find a favourable ordering.
6.  Upscale to a large size (≈12000–16000), try a few deterministic reorderings,
    then run a long sub‑gradient descent on the full problem.
7.  Final mass‑transfer pass and trimming of trailing zeros.

All updates respect non‑negativity and the ℓ¹‑simplex constraint
(∑a = √(2 n)).  The routine returns a list of floats ready for
`evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Objective (max‑convolution) for a simplex‑normalised vector
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)² (for Σa=√(2n) this equals maxconv)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised to the required ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex (Dirichlet‑like)."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """
    Spread the large values uniformly by placing them on a cyclic sequence
    with a step coprime to the length.
    """
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]          # descending
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Soft‑max gradient (used by annealed GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent_v2_ext(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      max_beta: float = 1e5,
                                      step_factor: float = 0.7,
                                      gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Momentum projected GD on the soft‑max approximation."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD_ext it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

# Sub‑gradient descent on the exact max‑convolution

def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that uses the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]
        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val

def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy mass moves with a decaying Metropolis acceptance rule."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            it += 1
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

# Linear‑programming alternating refinement (small dimensionality)
# ----------------------------------------------------------------------


def _lp_optimize(f: np.ndarray, M: float, target_sum: float,
                 rng: np.random.Generator) -> np.ndarray | None:
    """
    Solve a LP that maximises a random linear functional of `b`
    subject to `f * b <= M` and `b >= 0`.  The result is scaled to
    have ℓ¹‑norm = target_sum.  Returns None on failure.
    """
    n = f.shape[0]
    rows = []
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=np.int64)
        row = np.zeros(n, dtype=np.float64)
        row[i] = f[k - i]
        rows.append(row)
    A = np.vstack(rows)               # (2n‑1) × n
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Random objective – maximise c·b  ⇔ minimise –c·b
    c = -rng.random(n)

    bounds = [(0.0, None)] * n

    try:
        from scipy.optimize import linprog
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception as e:
        print(f"[{time.time():.2f}] LP exception: {e}")
        return None

    if not res.success:
        print(f"[{time.time():.2f}] LP failed: {res.message}")
        return None

    b = res.x
    total = b.sum()
    if total <= 0:
        return None
    b *= target_sum / total
    return b

def _permute_anneal(seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing over permutations of `seq` (swap moves) to reduce
    the peak of the self‑convolution.  Returns the best ordering found.
    """
    a = seq.copy()
    cur_val = _max_convolution_fft(a)
    best_seq = a.copy()
    best_val = cur_val
    rng = np.random.default_rng()
    n = a.shape[0]
    start_time = time.time()
    # exponential cooling from 20% of the current value down to 0.1%
    start_temp = cur_val * 0.2
    end_temp = cur_val * 0.001

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        b = a.copy()
        b[i], b[j] = b[j], b[i]
        new_val = _max_convolution_fft(b)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / max(temp, 1e-12)):
            a = b
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = b.copy()
                print(f"[{time.time():.2f}] PermAnneal it={it:5d} best={best_val:.6f}")

        it += 1

    return best_seq, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid search for a non‑negative step function with a low evaluation score.
    The routine follows the phases described in the module docstring.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Exploration problem size
    # ------------------------------------------------------------------
    if budget_s >= 900:
        n_expl = 12000
    elif budget_s >= 600:
        n_expl = 10000
    elif budget_s >= 400:
        n_expl = 8000
    else:
        n_expl = 6000
    target_expl = math.sqrt(2.0 * n_expl)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Build a pool of diverse seeds
    # ------------------------------------------------------------------
    init_pool = []

    # power‑law seeds
    for p in np.linspace(0.1, 1.5, 11):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))
    # exponential seeds
    for lam in np.linspace(0.0005, 0.05, 8):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))
    # random simplex seeds
    for _ in range(10):
        init_pool.append(_random_simplex(n_expl, target_expl, rng))

    # previous best (if any)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_expl))

    # jittered variants to enlarge the pool
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.02, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    # ------------------------------------------------------------------
    # Exploration: many short annealed‑gradient runs
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    while time.time() < deadline - 20.0:
        if best_seq is not None and rng.random() < 0.6:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.005, size=n_expl)
            seed += jitter
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)

        run_dead = min(time.time() + 12.0, deadline - 18.0)
        if run_dead <= time.time():
            break

        a_opt, val_opt = _annealed_gradient_descent_v2_ext(
            seed, run_dead, target_expl,
            max_beta=5e5, step_factor=0.6, gamma=0.99,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Light deterministic refinement before LP stage
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        mt_dead = time.time() + (deadline - time.time()) * 0.1
        a_ref, val_ref = _deterministic_mass_transfer(
            best_seq, mt_dead, target_expl,
            start_temp=0.4, end_temp=1e-4, low_frac=0.06,
        )
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] After MT refinement best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Down‑scale for LP / permutation stage
    # ------------------------------------------------------------------
    n_small = min(2000, max(800, n_expl // 5))
    target_small = math.sqrt(2.0 * n_small)
    seq_small = _resize_sequence(best_seq, n_small)
    seq_small = _project_to_simplex(seq_small, target_small)

    # ------------------------------------------------------------------
    # Linear‑programming alternation with occasional GD polishing
    # ------------------------------------------------------------------
    lp_iter = 0
    while time.time() < deadline - 30.0 and lp_iter < 12:
        M = _max_convolution_fft(seq_small)
        b_opt = _lp_optimize(seq_small, M, target_small, rng)
        if b_opt is None:
            break
        tau = 0.2
        seq_small = (1 - tau) * seq_small + tau * b_opt
        seq_small = _project_to_simplex(seq_small, target_small)
        cur_sc = _score(seq_small)
        print(f"[{time.time():.2f}] LP iter={lp_iter} score={cur_sc:.6f}")

        # brief GD polishing (≈4 s)
        gd_dead = time.time() + 4.0
        if gd_dead < deadline - 10.0:
            a_gd, val_gd = _annealed_gradient_descent_v2_ext(
                seq_small, gd_dead, target_small,
                max_beta=1e5, step_factor=0.5, gamma=0.98,
            )
            if val_gd < _score(seq_small):
                seq_small = a_gd.copy()
        lp_iter += 1

    # ------------------------------------------------------------------
    # Permutation annealing on the modest problem
    # ------------------------------------------------------------------
    if time.time() < deadline - 15.0:
        perm_dead = time.time() + min(12.0, deadline - time.time() - 5.0)
        seq_perm, perm_val = _permute_anneal(seq_small, perm_dead)
        if perm_val < _score(seq_small):
            seq_small = seq_perm.copy()
            print(f"[{time.time():.2f}] Permutation anneal best={perm_val:.6f}")

    # ------------------------------------------------------------------
    # Upscale to a larger size for final polishing
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 120.0:
        n_final = 16000
    elif remaining > 60.0:
        n_final = 12000
    else:
        n_final = n_expl
    target_final = math.sqrt(2.0 * n_final)
    seq_big = _resize_sequence(seq_small, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # ------------------------------------------------------------------
    # Simple reorderings to reduce the convolution peak
    # ------------------------------------------------------------------
    candidates = [seq_big,
                  _alternating_order(seq_big),
                  _even_spread_order(seq_big)]
    seq_big = min(candidates, key=_max_convolution_fft)
    best_score = _score(seq_big)
    print(f"[{time.time():.2f}] After ordering length={len(seq_big)} start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Sub‑gradient polishing on the large problem
    # ------------------------------------------------------------------
    if time.time() < deadline - 8.0:
        sub_dead = deadline - 5.0
        a_sub, val_sub = _subgradient_descent(seq_big, sub_dead, target_final,
                                              step_factor=0.45, gamma=0.995)
        if val_sub < best_score:
            best_score = val_sub
            seq_big = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑grad final best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final deterministic mass‑transfer pass
    # ------------------------------------------------------------------
    if time.time() < deadline - 3.0:
        mt_dead = time.time() + min(5.0, deadline - time.time() - 1.0)
        a_mt, val_mt = _deterministic_mass_transfer(seq_big, mt_dead, target_final,
                                                    start_temp=0.25, end_temp=1e-5, low_frac=0.05)
        if val_mt < best_score:
            best_score = val_mt
            seq_big = a_mt.copy()
            print(f"[{time.time():.2f}] Final MT best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # ------------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Optional final evaluation (purely for logging)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose the result for the next round
    globals()["height_sequence_1"] = final_seq
    return final_seq
