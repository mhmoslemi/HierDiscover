def adam_descent(a0, target_sum, deadline):

    a = a0 * (target_sum / a0.sum())
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_a = a.copy()
    best_val = _score(a)
    beta_temp = 1.0

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta_temp)

        t += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = 0.01 / (np.sqrt(v_hat) + eps)   # step size adjust
        a = a - step * m_hat
        # enforce nonneg by projection onto simplex
        a = _project_to_simplex(a, target_sum)  # but ensure nonneg
        # optionally after each step compute validation value
        if t % 1000 == 0:
            val = _score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"Adam step {t} beta {beta_temp} score {best_val}")

        # maybe increase beta_temp gradually (exponential schedule)
        if t % 5000 == 0:
            beta_temp = min(beta_temp*1.5, 65536)

        # If time left, continue

    return best_a, best_val

"""
Optimization of a non‑negative step function to minimise
    2·n·max(a*a) / (sum(a)^2)

The search consists of three phases:

1. **Adam on a smooth surrogate** – we minimise a soft‑max
   approximation of the maximal self‑convolution.  After each
   gradient step the vector is projected onto the simplex
   {x≥0, Σx = √(2n)}.

2. **Projected sub‑gradient descent** – using the true max‑convolution
   (the active peak) we take a sub‑gradient step and project back
   onto the simplex.

3. **Simulated‑annealing mass transfer** – a long annealing run that
   moves mass from indices that contribute heavily to the current peak
   to indices with low height.  The temperature starts high (≈1.5·current
   score) and decays with a slow power‑law; move sizes are not tied to
   the temperature so the algorithm can explore more aggressively.

The routine respects the global time‑budget (default 1000 s) and
stores the best sequence in the global variable ``height_sequence_1``
for warm‑starting subsequent runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Small helper – next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution a*a using an FFT of power‑of‑2 size.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Objective  2·n·max(a*a) / (sum(a)²).
    When Σa = √(2n) this equals max(a*a).
    """
    n = a.shape[0]
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of the max‑convolution.
    Returns (grad, full_convolution) where grad has the same length as a.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)

    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    # stable soft‑max
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    grad_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(grad_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase 1 – Adam optimisation on soft‑max surrogate
# ----------------------------------------------------------------------


def adam_optimize(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Adam optimiser on the smooth surrogate maxβ(a*a) = (1/β) log Σ exp(β·b).
    The iterate is projected onto the simplex after each step.
    """
    a = a0 * (target_sum / a0.sum())
    n = a.shape[0]

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # soft‑max temperature schedule
    beta = 1.0
    max_beta = 65536.0

    best_a = a.copy()
    best_val = _score(a)

    start = time.time()
    it = 0
    while time.time() < deadline:
        # gradient of smooth max
        g, _ = _softmax_gradient(a, beta)

        it += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # base learning rate – scaled with target_sum
        lr = 0.05 * target_sum
        step = lr / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_to_simplex(a, target_sum)

        # evaluate
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam iter={it:,} β={beta:.1f} score={best_val:.6f}")

        # increase β slowly
        if it % 2000 == 0:
            beta = min(beta * 1.5, max_beta)

        # occasional jitter to escape plateaus
        if it % 2500 == 0:
            rng = np.random.default_rng()
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

    return best_a, best_val

def subgradient_optimize(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Projected sub‑gradient descent on the exact peak of the self‑convolution.
    The step size decays linearly from 2.0 to 0.05 over the allotted time.
    """
    a = a0 * (target_sum / a0.sum())
    n = a.shape[0]

    best_a = a.copy()
    best_val = _score(a)

    start = time.time()
    it = 0
    while time.time() < deadline:
        # compute convolution and the index of the current peak
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # sub‑gradient of the active peak
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        # linear decay of the step size
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        eta = 2.0 * (1.0 - frac) + 0.05 * frac   # from 2.0 → 0.05

        a = a - eta * g
        a = _project_to_simplex(a, target_sum)

        # evaluate
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad iter={it:,} η={eta:.4f} score={best_val:.6f}")

        # jitter every few hundred iterations
        if it % 500 == 0:
            rng = np.random.default_rng()
            jitter = rng.normal(scale=target_sum * 0.0003, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val

def anneal_optimize(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Long‑run annealing that repeatedly moves a small amount of mass
    from a “heavy” index (large contribution to the current peak) to a
    “light” index (small height).  Temperature starts at ≈1.5·score and
    decays with a slow power‑law; move size is independent of temperature
    (up to a generous bound) so the chain can make decisive jumps.
    """
    a = a0.copy()
    n = a.shape[0]

    b = _conv_fft(a)               # exact convolution a*a
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start

    # initial temperature (a bit higher than in the previous implementation)
    T0 = cur_val * 1.5

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        # slow power‑law cooling (exponent < 1 -> slower early cooling)
        temperature = max(0.0, T0 * (1.0 - frac ** 0.3))

        # current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # donor – proportional to contribution to the peak
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – favour small heights (inverse weighting)
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # allow fairly large transfers (up to 12 % of donor or 2 % of total)
        max_delta = min(a[donor] * 0.12, target_sum * 0.02)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # ---- exact O(n) update of the convolution -----------------------
        b_new = b.copy()
        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²) and cross term
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        # Metropolis acceptance based on the *current* energy
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved best={best_val:.6f}")

        # periodic full recomputation to avoid numerical drift
        if it % 5000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val

    start = time.time()
    deadline = start + budget_s - 0.5        # safety margin

    # ------------------------------------------------------------------
    # Problem size – larger n gives finer granularity but costs time.
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Warm‑start (if a previous construction exists) or random init.
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
    else:
        # Dirichlet‑like random start (α=0.5 gives a varied shape)
        init = rng.gamma(0.5, 1.0, size=n)
        init = init.astype(np.float64)
    # small perturbation to avoid perfect symmetry
    init += rng.normal(scale=init.mean() * 0.001, size=n)
    init = _project_to_simplex(init, target_sum)

    # ------------------------------------------------------------------
    # Phase 1 – Adam on the smooth surrogate (≈15 % of the budget)
    # ------------------------------------------------------------------
    phase1_deadline = time.time() + 0.15 * (deadline - time.time())
    a1, val1 = adam_optimize(init, target_sum, phase1_deadline)
    best_seq, best_score = a1, val1
    print(f"[{time.time():.2f}] After Adam phase best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent (≈15 % of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase2_deadline = time.time() + 0.15 * (deadline - time.time())
        a2, val2 = subgradient_optimize(best_seq, target_sum, phase2_deadline)
        if val2 < best_score:
            best_seq, best_score = a2, val2
        print(f"[{time.time():.2f}] After sub‑gradient phase best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Long simulated‑annealing refinement (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a3, val3 = anneal_optimize(best_seq, target_sum, deadline)
        if val3 < best_score:
            best_seq, best_score = a3, val3
        print(f"[{time.time():.2f}] After annealing best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final clean‑up: cut trailing zeros, store as global warm‑start.
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # final evaluation (just for logging – the caller only needs the list)
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
