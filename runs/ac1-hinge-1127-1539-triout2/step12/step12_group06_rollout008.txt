"""
Hybrid optimizer with optional LP refinement.

The algorithm proceeds in several stages:

1. Build a diverse pool of seed sequences (power‑law, exponential, random Dirichlet,
   linear, previous best) of moderate length.
2. Perform quick projected sub‑gradient descent on the exact max‑convolution
   for each seed, keeping the best.
3. (Optional) Refine the best seed with a small linear‑programming step:
   given f, solve max Σ g_j s.t. f∗g ≤ max(f∗f).  This is the classical
   “alternating LP” improvement.  The LP is solved on a down‑scaled version
   (≈2000 entries) to keep memory low, then the result is up‑sampled.
4. Upscale the current best to a larger length (≈15000–20000) and run a
   longer sub‑gradient phase, a high‑β soft‑max gradient descent, a
   simulated‑annealing permutation search and a deterministic mass‑transfer
   pass.  The remaining time is split adaptively.
5. Trim trailing zeros and return the sequence.

All vectors are projected onto the simplex {x_i ≥ 0, Σ x_i = √(2·n)} after each
update, so the evaluation metric reduces to the peak of the autocorrelation.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·n‑1) with FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


# ----------------------------------------------------------------------
# Scoring (objective = max convolution entry, because sum = sqrt(2n))
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Return max convolution value (the evaluation objective)."""
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(max_b)


# ----------------------------------------------------------------------
# Exact max‑convolution sub‑gradient
# ----------------------------------------------------------------------


def _max_conv_and_subgrad(a: np.ndarray):
    """Return (conv, max_val, kmax, gradient of peak w.r.t. a)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    kmax = int(np.argmax(conv))
    max_val = conv[kmax]

    grad = np.zeros_like(a)
    lo = max(0, kmax - (n - 1))
    hi = min(kmax, n - 1)
    idx = np.arange(lo, hi + 1)
    partner = a[kmax - idx]
    grad[lo:hi + 1] = 2.0 * partner
    return conv, max_val, kmax, grad


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0
    momentum = np.zeros_like(a)

    while time.time() < deadline:
        _, _, _, grad = _max_conv_and_subgrad(a)

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_factor * target_sum / norm
            a -= lr * momentum
        else:
            a -= step_factor * target_sum * grad / (np.linalg.norm(grad) + 1e-12)

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] SubGD it={it:5d} score={best_val:.6f}")

        it += 1
    return best, best_val


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) gradient descent
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    grad = 2.0 * np.fft.irfft(np.fft.rfft(a_rev, n=size) *
                               np.fft.rfft(p, n=size), n=size)[: 2 * n - 1]
    grad = grad[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 1e6,
                               step_factor: float = 0.5,
                               gamma: float = 0.995) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SoftGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1
    return best_a, best_val

def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]
        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score

def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.1,
                                 end_temp: float = 1e-4,
                                 top_frac: float = 0.02,
                                 low_frac: float = 0.02) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)

        if donor == receiver:
            continue

        a[donor], a[receiver] = a[receiver], a[donor]

        new_score = _score(a)

        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Perm new best={best_score:.6f}")

        if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / max(temp, 1e-12)):
            pass
        else:
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score


# ----------------------------------------------------------------------
# Optional LP refinement on a down‑scaled problem
# ----------------------------------------------------------------------


def _lp_refine_small(f_seq: np.ndarray,
                     target_sum: float,
                     max_iter: int = 2,
                     combine_frac: float = 0.3,
                     deadline: float = 0.0):
    """Refine f_seq using a linear program on a reduced size (≈2000)."""
    try:
        from scipy.optimize import linprog
    except Exception:  # SciPy not available
        return f_seq

    n = f_seq.shape[0]
    # Build constraints: A_ub @ g <= M
    M = _max_convolution_fft(f_seq)

    # Build dense A_ub (2n-1 rows, n cols)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        start = i
        end = i + n
        A[start:end, i] = f_seq[i]

    b = np.full(rows, M, dtype=np.float64)
    # Objective: maximize sum(g) ==> minimize -sum(g)
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0, None)] * n

    cur_f = f_seq.copy()
    cur_target = target_sum

    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Solve LP
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': max(1.0, deadline - time.time())})
        if not res.success:
            print(f"[{time.time():.2f}] LP iteration {it} failed: {res.message}")
            break
        g0 = res.x
        s0 = g0.sum()
        if s0 < 1e-12:
            break
        # Scale to correct sum
        g = cur_target * g0 / s0
        # Convex combination
        cur_f = (1 - combine_frac) * cur_f + combine_frac * g
        cur_f = _project_to_simplex(cur_f, cur_target)
        # Update M for next iteration
        M = _max_convolution_fft(cur_f)
        # Re‑build A with the new f (cheap loop)
        for i in range(n):
            start = i
            end = i + n
            A[start:end, i] = cur_f[i]
        print(f"[{time.time():.2f}] LP refinement iter={it} max={M:.6f}")
    return cur_f

    """

    deadline = start_time + budget_s - 0.5  # safety margin

    # Helper for remaining time
    # ------------------------------------------------------------------
    def time_left():
        return deadline - time.time()

    # ------------------------------------------------------------------
    # Choose moderate exploration length
    # ------------------------------------------------------------------
    if budget_s >= 900:
        n_expl = 10000
    elif budget_s >= 700:
        n_expl = 8000
    elif budget_s >= 500:
        n_expl = 6000
    else:
        n_expl = 4000
    target_expl = math.sqrt(2.0 * n_expl)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Build a diverse pool of seed sequences
    # ------------------------------------------------------------------
    init_pool = []

    # power‑law branch
    for p in np.linspace(0.30, 0.80, 9):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))

    # exponential branch
    for lam in np.linspace(0.001, 0.02, 8):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))

    # random Dirichlet‑like points
    for _ in range(8):
        a = rng.random(n_expl)
        a *= target_expl / a.sum()
        init_pool.append(a)

    # linear decreasing shape
    lin = np.linspace(1.0, 0.0, n_expl)
    lin *= target_expl / lin.sum()
    init_pool.append(lin)

    # previous best if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_expl))

    # jittered copies for extra diversity
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.008, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    # ------------------------------------------------------------------
    # Quick sub‑gradient exploration on moderate length
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    # Reserve roughly 25 % of the budget for later refined phases
    exploration_deadline = deadline - max(30.0, 0.25 * budget_s)

    while time.time() < exploration_deadline:
        seed = rng.choice(init_pool)
        # allocate ~10 s for each short run (or less if time runs out)
        run_dead = min(time.time() + 10.0, deadline - 5.0)
        if run_dead <= time.time():
            break
        a_opt, val_opt = _subgradient_descent(seed, run_dead,
                                              target_expl,
                                              step_factor=0.5,
                                              gamma=0.99)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best={best_val:.6f}")

    if best_seq is None:
        # fallback to a generic power‑law if nothing succeeded
        best_seq = _power_law_sequence(n_expl, 0.55, target_expl)

    # ------------------------------------------------------------------
    # Optional LP refinement on a down‑scaled version (if time permits)
    # ------------------------------------------------------------------
    if time_left() > 20.0:
        # down‑scale to ~2000 entries for LP tractability
        n_lp = min(2000, max(1500, n_expl // 4))
        seq_lp = _resize_sequence(best_seq, n_lp)
        target_lp = math.sqrt(2.0 * n_lp)
        seq_lp = _project_to_simplex(seq_lp, target_lp)
        lp_deadline = time.time() + min(12.0, time_left() - 8.0)
        seq_lp = _lp_refine_small(seq_lp, target_lp,
                                  max_iter=3,
                                  combine_frac=0.35,
                                  deadline=lp_deadline)
        # upscale back to current length and keep if better
        seq_up = _resize_sequence(seq_lp, n_expl)
        seq_up = _project_to_simplex(seq_up, target_expl)
        val_up = _score(seq_up)
        if val_up < best_val:
            best_val = val_up
            best_seq = seq_up.copy()
            print(f"[{time.time():.2f}] LP‑refine improved best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Upscale to a larger length for final refinement
    # ------------------------------------------------------------------
    remaining = time_left()
    if remaining > 30.0 and budget_s >= 800:
        n_final = 18000
    elif remaining > 20.0 and budget_s >= 600:
        n_final = 15000
    else:
        n_final = n_expl

    target_final = math.sqrt(2.0 * n_final)
    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # Try a couple of deterministic re‑orderings
    candidates = [seq_big,
                  _alternating_order(seq_big),
                  _reversed_alternating(seq_big)]
    best_seq = seq_big
    best_val = _score(seq_big)
    for cand in candidates[1:]:
        sc = _score(cand)
        if sc < best_val:
            best_val = sc
            best_seq = cand.copy()
    print(f"[{time.time():.2f}] Upscaled to n={n_final}, start score={best_val:.6f}")

    # ------------------------------------------------------------------
    # Longer sub‑gradient descent on the enlarged vector
    # ------------------------------------------------------------------
    if time_left() > 15.0:
        sub_dead = time.time() + min(12.0, time_left() - 5.0)
        a_opt, val_opt = _subgradient_descent(best_seq, sub_dead,
                                              target_final,
                                              step_factor=0.5,
                                              gamma=0.99)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] SubGD (big) improved best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Soft‑max gradient descent with high β
    # ------------------------------------------------------------------
    if time_left() > 12.0:
        final_dead = deadline - 4.0
        a_opt, val_opt = _annealed_gradient_descent(best_seq, final_dead,
                                                    target_final,
                                                    max_beta=3e6,
                                                    step_factor=0.5,
                                                    gamma=0.995)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Soft‑max final best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Targeted permutation search (only re‑ordering)
    # ------------------------------------------------------------------
    if time_left() > 6.0:
        perm_dead = time.time() + min(8.0, time_left() - 2.0)
        a_opt, val_opt = _permutation_targeted_search(best_seq, perm_dead,
                                                      start_temp=0.2,
                                                      end_temp=1e-5,
                                                      top_frac=0.04,
                                                      low_frac=0.04)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Perm‑search improved best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Final deterministic mass‑transfer pass
    # ------------------------------------------------------------------
    if time_left() > 3.0:
        mt_dead = time.time() + min(5.0, time_left() - 1.0)
        a_opt, val_opt = _deterministic_mass_transfer(best_seq, mt_dead,
                                                      target_final,
                                                      start_temp=0.3,
                                                      end_temp=1e-5,
                                                      low_frac=0.05)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Mass‑transfer final best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing zeros for compactness
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Logging of final evaluation (optional)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    globals()["height_sequence_1"] = final_seq
    return final_seq
