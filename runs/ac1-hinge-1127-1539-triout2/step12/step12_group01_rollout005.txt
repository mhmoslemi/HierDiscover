def softmax_mirror_descent(a0, target_sum, deadline, best_a, best_val):
    a = a0.copy()
    beta = 0.5
    max_beta = 1e5
    eta_base = 0.05
    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        grad, conv = _softmax_gradient(a, beta)
        # step size adapt
        eta = eta_base / np.sqrt(it+1) / (beta+1)
        a = a * np.exp(-eta * grad)
        # ensure sum
        a = a * (target_sum / a.sum())
        # jitter occasional
        if it % 500 == 0:
            a += rng.normal(scale=target_sum*0.001, size=a.shape)
            a = np.maximum(a, 0.0)
            a = a * (target_sum / a.sum())
        # possibly increase beta gradually
        if it % 1000 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)
        # evaluate
        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Softmax descent it={it} β={beta:.1f} score={best_val:.6f}")
        it += 1
    return best_a, best_val

def subgradient_descent2(a0, target_sum, deadline, best_a, best_val):
    a = a0.copy()
    start = time.time()
    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))
        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi+1)
        # compute subgradient
        g = np.zeros_like(a)
        # For each i in idx, derivative w.r.t a_i is 2*a_{k_max - i}
        g[idx] = 2.0 * a[k_max - idx]
        # step size schedule: decreasing linear
        elapsed = time.time() - start
        total = deadline - start
        eta = 1.5 * (1 - elapsed/total) + 0.05
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())
        if it % 400 == 0:
            # jitter
            a += rng.normal(scale=target_sum*0.0005, size=a.shape)
            a = np.maximum(a, 0.0)
            a = a * (target_sum / a.sum())
        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad2 it={it} score={best_val:.6f}")

    return best_a, best_val

def aggressive_anneal(a0, target_sum, deadline, best_a, best_val):
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_val = _score(a)
    best_val_local = best_val
    best_local = best_a.copy()
    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.8
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1 - np.sqrt(frac)))
        # find peak
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1)
        # contributions
        contrib = a[idx] * a[k_max - idx]
        donor = idx[np.argmax(contrib)]
        # receiver: smallest a
        receiver = int(np.argmin(a))
        if donor == receiver:
            # pick second smallest
            sorted_idxs = np.argsort(a)
            if sorted_idxs[0] == donor:
                receiver = int(sorted_idxs[1])
            else:
                receiver = int(sorted_idxs[0])
        max_delta = min(a[donor] * 0.07, target_sum * 0.001)  # 7% donor, 0.1% total mass
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta * (temperature / T0 if T0 > 0 else 0.0)
        if delta <= 0:
            continue
        # update convolution
        b_new = b.copy()
        # remove donor
        b_new[donor:donor + n] -= 2.0 * delta * a
        # add receiver
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic terms
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta
        new_val = float(b_new.max())
        # acceptance
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True
        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val_local:
                best_val_local = new_val
                best_local = a.copy()
                print(f"[{time.time():.2f}] AggAnneal improved best={best_val_local:.6f}")

"""
Optimisation of a non‑negative step sequence (benchmark #1)

The score = 2·n·max(a*a) / (∑a)² = max(a*a) when the sum is fixed to √(2n).
We therefore minimise the peak of the self‑convolution while keeping the
ℓ₁‑norm fixed.

The search consists of four phases (all respect the global time budget):

1. **Initialisation** – a few random perturbations of a decreasing
   power‑law (or the previous best if it exists) and keep the best one.

2. **Soft‑max mirror descent** – minimise a smooth approximation of the
   max‑convolution using an exponentiated‑gradient step.  The smoothing
   parameter β is increased exponentially, turning the surrogate into the
   true max.

3. **Multiplicative sub‑gradient descent** on the exact peak.  The
   sub‑gradient of the active convolution entry is applied with a
   decreasing step‑size.

4. **Deterministic aggressive annealing** – repeatedly move a small
   amount of mass from the index that contributes most to the current
   peak to the globally smallest index.  The move is accepted with a
   Metropolis rule; the temperature follows a √–schedule and the move
   size scales with the temperature.

All phases keep the ℓ₁‑norm exactly equal to √(2 n) (or renormalise
periodically to combat numerical drift).  The best sequence found is
returned and stored in the global variable ``height_sequence_1`` for
future warm‑starts.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (∑a)² (the benchmark score)."""
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:                     # should never happen (protected by caller)
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of the max‑convolution.
    Returns (grad, full_convolution) where grad has length n.
    The surrogate is  log∑_k exp(beta·b_k) where b = a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()                     # soft‑max weights over entries of b

    # grad = 2·a * (p convolved with reversed a)
    a_rev = a[::-1]
    grad_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(grad_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]    # keep central n entries
    return grad, b


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)

def _initial_candidate(n: int, target_sum: float):
    rng = np.random.default_rng()
    # base power‑law (exponent tuned a little)
    base = (np.arange(1, n + 1, dtype=np.float64) ** (-0.58))
    base = base * (target_sum / base.sum())
    best_a = base
    best_val = _score(base)

    # a few random perturbations
    for _ in range(4):
        pert = base + rng.normal(scale=target_sum * 0.001, size=n)
        pert = np.maximum(pert, 0.0)
        pert = pert * (target_sum / pert.sum())
        val = _score(pert)
        if val < best_val:
            best_val = val
            best_a = pert
    return best_a, best_val

# Phase 1 – soft‑max mirror descent
# ----------------------------------------------------------------------


def _softmax_mirror_descent(a0: np.ndarray, target_sum: float,
                            deadline: float, best_a: np.ndarray,
                            best_val: float):
    a = a0.copy()
    beta = 0.5
    max_beta = 1e5
    eta_base = 0.03
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        grad, _ = _softmax_gradient(a, beta)

        # diminishing step size, also shrinks when β grows
        eta = eta_base / np.sqrt(it + 1) / (beta + 1.0)
        a = a * np.exp(-eta * grad)
        a = a * (target_sum / a.sum())

        # occasional jitter to help escape plateaus
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # increase smoothing parameter gradually
        if it % 800 == 0 and beta < max_beta:
            beta = min(beta * 1.8, max_beta)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SoftmaxMD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        it += 1
    return best_a, best_val

# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, target_sum: float,
                         deadline: float, best_a: np.ndarray,
                         best_val: float):
    a = a0.copy()
    start = time.time()
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))
        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # exact sub‑gradient of the active entry
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        # linear schedule for the step size
        elapsed = time.time() - start
        total = deadline - start
        eta = 1.5 * (1.0 - elapsed / total) + 0.05

        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        # jitter every few hundred steps
        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.0004, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1
    return best_a, best_val

def _aggressive_anneal(a0: np.ndarray, target_sum: float,
                       deadline: float, best_a: np.ndarray,
                       best_val: float):
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    cur_val = float(b.max())
    best_val_local = best_val
    best_local = best_a.copy()
    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.80                 # initial temperature (≈80 % of current score)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        # non‑linear cooling – stay hot a little longer
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        # current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contributions of each index to that peak
        contrib = a[idx] * a[k_max - idx]

        # donor = index with the *largest* contribution to the peak
        donor = idx[np.argmax(contrib)]

        # receiver = globally smallest entry (helps push mass to the tails)
        receiver = int(np.argmin(a))
        if donor == receiver:                               # avoid trivial move
            # pick the second smallest element
            sorted_idx = np.argsort(a)
            receiver = int(sorted_idx[1]) if sorted_idx[0] == donor else int(sorted_idx[0])

        # size of the transfer – proportional to donor mass, capped globally
        max_delta = min(a[donor] * 0.2, target_sum * 0.005)   # ≤20 % of donor, ≤0.5 % of total mass
        if max_delta <= 0.0:
            continue
        # scale with temperature (large moves while hot, tiny when cold)
        delta = rng.random() * max_delta * (temperature / T0 if T0 > 0 else 0.0)
        if delta <= 0.0:
            continue

        # ----- exact O(n) update of the convolution -----------------
        b_new = b.copy()
        # linear part: remove donor, add receiver
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic self‑terms
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        # cross‑term between donor and receiver
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta
        # -----------------------------------------------------------

        new_val = float(b_new.max())

        # Metropolis acceptance based on *current* energy
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val_local:
                best_val_local = new_val
                best_local = a.copy()
                print(f"[{time.time():.2f}] Anneal improved best={best_val_local:.6f}")

        # periodic full recomputation to keep numerical drift negligible
        if it % 5000 == 0:
            a = a * (target_sum / a.sum())          # renormalise (sum stays virtually constant)
            b = _conv_fft(a)
            cur_val = float(b.max())
            if cur_val < best_val_local:
                best_val_local = cur_val
                best_local = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved best={best_val_local:.6f}")

        it += 1
    return best_local, best_val_local

        # re‑normalise the imported warm‑start
        init = init * (target_sum / init.sum())
        a, cur_best_val = _initial_candidate(n, target_sum)   # we still explore a few perturbations
        # keep the warm‑start if it is better
        warm_val = _score(init)
        if warm_val < cur_best_val:
            a = init.copy()
            cur_best_val = warm_val
    best_a = a.copy()
    best_val = cur_best_val
    print(f"[{time.time():.2f}] Initial best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – soft‑max mirror descent (≈12 % of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase_deadline = time.time() + 0.12 * (deadline - time.time())
        best_a, best_val = _softmax_mirror_descent(best_a, target_sum,
                                                   phase_deadline, best_a, best_val)
        print(f"[{time.time():.2f}] After soft‑max MD best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – sub‑gradient descent on the exact max‑convolution (≈12 %)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase_deadline = time.time() + 0.12 * (deadline - time.time())
        best_a, best_val = _subgradient_descent(best_a, target_sum,
                                                phase_deadline, best_a, best_val)
        print(f"[{time.time():.2f}] After sub‑gradient descent best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – aggressive deterministic annealing (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _aggressive_anneal(best_a, target_sum,
                                              deadline, best_a, best_val)
        print(f"[{time.time():.2f}] After annealing best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: trim trailing zeros, store for the next run
    # ------------------------------------------------------------------
    arr = np.array(best_a, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # store for warm‑starts in future runs
    globals()["height_sequence_1"] = final_seq

    # final evaluation (for logging only)
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    return final_seq
