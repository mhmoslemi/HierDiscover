def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    start = time.time()
    deadline = start + budget_s - 0.5

    # Define n_target based on budget
    # maybe choose n_target = 30000 if budget >= 900 else 20000 etc.
    if budget_s >= 900:
        n_target = 30000
    elif budget_s >= 600:
        n_target = 25000
    else:
        n_target = 20000

    target_sum = math.sqrt(2.0 * n_target)

    rng = np.random.default_rng()

    # Build seeds
    seeds = []

    # power law seeds
    for p in np.linspace(0.3, 0.9, 5):
        seeds.append(_power_law_sequence(n_target, float(p), target_sum))

    # exponential seeds
    for lam in np.linspace(0.001, 0.02, 5):
        seeds.append(_exponential_sequence(n_target, float(lam), target_sum))

    # random Dirichlet seeds
    for _ in range(5):
        a = rng.random(n_target)
        a = np.maximum(a, 0.0)
        a *= target_sum / a.sum()
        seeds.append(a)

    # include previous best if available (resized)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        prev_resized = _resize_sequence(prev, n_target)
        prev_resized = _project_to_simplex(prev_resized, target_sum)
        seeds.append(prev_resized)

    # jitter each seed a bit to increase diversity
    jittered = []
    for seq in seeds:
        jitter = rng.normal(scale=target_sum * 0.005, size=n_target)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        jittered.append(seq_j)
    seeds.extend(jittered)

    # Phase 1: quick subgradient to find promising seed within first part of time
    # Use time slices: allocate early portion (e.g., 20% of budget) for exploring seeds
    explore_deadline = start + min(0.2 * budget_s, 250)  # ensure at least some time but not too much

    best_seq = None
    best_score = np.inf

    # Explore seeds
    for seed in seeds:
        now = time.time()
        if now >= explore_deadline:
            break
        # allocate up to e.g., 10 sec per seed, but not exceed explore_deadline.
        run_dead = min(explore_deadline, now + 10.0)
        if run_dead <= now:
            break
        a_opt, val_opt = _subgradient_descent(seed, run_dead, target_sum,
                                              step_factor=0.8, gamma=0.95)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best: {best_score:.6f}")

    # If no best found, just take random seed
    if best_seq is None:
        best_seq = _power_law_sequence(n_target, 0.55, target_sum)
        best_score = _score(best_seq)

    # Phase 2: long subgradient refinement on best until we have limited time left (before final annealing)
    # Let's reserve 200 seconds for final annealing phases.
    remaining = deadline - time.time()
    if remaining > 250.0:
        # allocate most to subgradient: 200 seconds
        sub_dead = time.time() + min(200.0, remaining - 150.0)  # leave at least 150 sec for annealing & mass transfer
        a_opt, val_opt = _subgradient_descent(best_seq, sub_dead, target_sum,
                                              step_factor=0.5, gamma=0.99)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Refine subGD best: {best_score:.6f}")

    # Phase 3: simulated annealing on swaps (permutation)
    remaining = deadline - time.time()
    if remaining > 120.0:
        perm_dead = time.time() + min(100.0, remaining - 20.0)
        a_opt, val_opt = _permutation_targeted_search(best_seq, perm_dead,
                                                      start_temp=0.2,
                                                      end_temp=1e-5,
                                                      top_frac=0.05,
                                                      low_frac=0.05)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Permutation best: {best_score:.6f}")

    # Phase 4: mass-transfer annealing
    remaining = deadline - time.time()
    if remaining > 30.0:
        mt_dead = time.time() + min(30.0, remaining - 5.0)
        a_opt, val_opt = _deterministic_mass_transfer(best_seq, mt_dead,
                                                      target_sum,
                                                      start_temp=0.3,
                                                      end_temp=1e-5,
                                                      low_frac=0.05)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] MassTransfer final best: {best_score:.6f}")

    # Trim zeros and final evaluation
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-13)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Record global
    globals()["height_sequence_1"] = final_seq
    # Optional evaluate
    try:
        val = evaluate_sequence(final_seq)  # as imported
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as e:
        print(f"Evaluation error: {e}")

    return final_seq

"""
Candidate proposer for the non‑negative step‑function problem.

Algorithm outline
-----------------
1. Build a diverse pool of seed vectors (power‑law, exponential,
   Dirichlet‑like, previous best).  All seeds are projected onto the
   simplex Σa_i = √(2·n) so that the objective 2·n·max(conv)/Σa² is
   invariant to scaling.
2. A first exploration phase runs a short projected sub‑gradient
   descent (exact max‑convolution gradient) on each seed.  The best
   result is kept.
3. The best seed is up‑scaled to a large length (≈30 000) which gives
   many degrees of freedom for flattening the self‑convolution.
4. A long sub‑gradient descent refines the shape.
5. Two stochastic polishing phases are applied:
   • permutation‑annealing (swap large/small entries to reduce the peak);
   • deterministic mass‑transfer annealing (move a small amount of mass
     from a “donor’’ position to a “receiver’’).
6. Simple post‑processing: trim trailing zeros and store the result
   globally for possible reuse.

The time budget (default 1000 s) is split automatically; the code
leaves a small safety margin before the deadline.  The implementation
uses only NumPy and the supplied `evaluate_sequence` function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) shape, renormalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """exp(-lam·i) shape, renormalised to target_sum."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → left, 2nd largest → right, 3rd → next left, …."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


# ----------------------------------------------------------------------
# Scoring utility (objective = 2·n·max(conv) / Σa²)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Return the objective value for vector a."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Exact max‑convolution sub‑gradient
# ----------------------------------------------------------------------


def _max_conv_and_subgrad(a: np.ndarray):
    """
    Returns (conv, max_val, k_max, subgrad) where subgrad is the gradient
    of the peak entry of the convolution w.r.t. a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    kmax = int(np.argmax(conv))
    max_val = conv[kmax]

    grad = np.zeros_like(a)
    lo = max(0, kmax - (n - 1))
    hi = min(kmax, n - 1)
    idx = np.arange(lo, hi + 1)
    partner = a[kmax - idx]
    grad[lo:hi + 1] = 2.0 * partner
    return conv, max_val, kmax, grad


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on max(conv)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Momentum projected sub‑gradient descent on max(conv(a,a)).
    Runs until `deadline` (unix time).  Returns best vector seen and its score.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0
    momentum = np.zeros_like(a)

    while time.time() < deadline:
        _, _, _, grad = _max_conv_and_subgrad(a)

        # momentum update
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_factor * target_sum / norm
            a -= lr * momentum
        else:
            a -= step_factor * target_sum * grad / (np.linalg.norm(grad) + 1e-12)

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to prevent stagnation
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] SubGD it={it:5d}  score={best_val:.6f}")

        it += 1

    return best, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (simulated‑annealing style)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy donor→receiver moves with an exponential temperature schedule."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # locate the current peak
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to the peak
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted permutation search (simulated annealing on index order)
# ----------------------------------------------------------------------


def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.1,
                                 end_temp: float = 1e-4,
                                 top_frac: float = 0.02,
                                 low_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """Swap indices that heavily contribute to the peak; Metropolis acceptance."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)

        if donor == receiver:
            continue

        a[donor], a[receiver] = a[receiver], a[donor]

        new_score = _score(a)

        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] PermSearch new best={best_score:.6f}")

        if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / max(temp, 1e-12)):
            # keep swap
            pass
        else:
            # revert swap
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score

# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with a low evaluation score.
    The routine respects the 1000 s budget, stores the best vector in the
    global variable `height_sequence_1` and returns it as a plain list.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Choose a target length based on the available budget.
    # Larger n gives more freedom to flatten the convolution.
    # ------------------------------------------------------------------
    if budget_s >= 900:
        n_target = 30000
    elif budget_s >= 600:
        n_target = 25000
    else:
        n_target = 20000
    target_sum = math.sqrt(2.0 * n_target)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Build a diverse pool of seed sequences.
    # ------------------------------------------------------------------
    seeds = []

    # power‑law family
    for p in np.linspace(0.3, 0.9, 5):
        seeds.append(_power_law_sequence(n_target, float(p), target_sum))

    # exponential family
    for lam in np.linspace(0.001, 0.02, 5):
        seeds.append(_exponential_sequence(n_target, float(lam), target_sum))

    # random Dirichlet‑like points
    for _ in range(5):
        a = rng.random(n_target)
        a *= target_sum / a.sum()
        seeds.append(a)

    # previous best (if available)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        prev_resized = _resize_sequence(prev, n_target)
        prev_resized = _project_to_simplex(prev_resized, target_sum)
        seeds.append(prev_resized)

    # add jittered copies for extra diversity
    jittered = []
    for seq in seeds:
        jitter = rng.normal(scale=target_sum * 0.005, size=n_target)
        seq_j = np.maximum(seq + jitter, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        jittered.append(seq_j)
    seeds.extend(jittered)

    # ------------------------------------------------------------------
    # Phase 1: quick exploration on each seed (short sub‑gradient runs).
    # ------------------------------------------------------------------
    explore_deadline = start_time + min(0.2 * budget_s, 250.0)  # at most 250 s
    best_seq = None
    best_score = np.inf

    for seed in seeds:
        now = time.time()
        if now >= explore_deadline:
            break
        run_dead = min(explore_deadline, now + 10.0)   # ≤10 s per seed
        if run_dead <= now:
            break
        a_opt, val_opt = _subgradient_descent(seed, run_dead, target_sum,
                                              step_factor=0.8, gamma=0.95)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best = {best_score:.6f}")

    if best_seq is None:
        # fallback to a generic power‑law seed
        best_seq = _power_law_sequence(n_target, 0.55, target_sum)
        best_score = _score(best_seq)

    # ------------------------------------------------------------------
    # Phase 2: long sub‑gradient refinement on the current best.
    # Reserve time for later annealing phases.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 250.0:
        # keep ~150 s for permutation & mass‑transfer passes
        refine_dead = time.time() + min(200.0, remaining - 150.0)
        a_opt, val_opt = _subgradient_descent(best_seq, refine_dead,
                                              target_sum,
                                              step_factor=0.5, gamma=0.99)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Refine sub‑GD best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3: targeted permutation annealing (swap large / small entries).
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 130.0:
        perm_dead = time.time() + min(100.0, remaining - 30.0)
        a_opt, val_opt = _permutation_targeted_search(best_seq, perm_dead,
                                                      start_temp=0.2,
                                                      end_temp=1e-5,
                                                      top_frac=0.05,
                                                      low_frac=0.05)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Permutation best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4: deterministic mass‑transfer (annealed) refinement.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        mt_dead = time.time() + min(30.0, remaining - 5.0)
        a_opt, val_opt = _deterministic_mass_transfer(best_seq, mt_dead,
                                                      target_sum,
                                                      start_temp=0.3,
                                                      end_temp=1e-5,
                                                      low_frac=0.05)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Mass‑transfer final best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5: a final re‑ordering (alternating) – cheap but sometimes
    #          reduces the peak further.
    # ------------------------------------------------------------------
    alt = _alternating_order(best_seq)
    alt_score = _score(alt)
    if alt_score < best_score:
        best_score = alt_score
        best_seq = alt
        print(f"[{time.time():.2f}] Alternating order improved score to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing zeros for compactness.
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-13)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Optional final evaluation (for logging only).
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as exc:
        print(f"Evaluation error: {exc}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
