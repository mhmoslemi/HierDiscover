def _permutation_search_explore(a0, deadline, start_temp=0.5, end_temp=1e-5, top_frac=0.05, low_frac=0.05):
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    n = a.shape[0]
    start_time = time.time()
    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        kmax_candidates = np.flatnonzero(conv == max_val)
        kmax = rng.choice(kmax_candidates)
        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]
        # decide whether to use targeted swap or random swap
        if rng.random() < 0.7:
            top_k = max(2, int(top_frac * n))
            low_k = max(2, int(low_frac * n))
            donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
            receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]
            donor = rng.choice(donor_candidates)
            receiver = rng.choice(receiver_candidates)
        else:
            donor = rng.integers(n)
            receiver = rng.integers(n)
            while receiver == donor:
                receiver = rng.integers(n)
        a[donor], a[receiver] = a[receiver], a[donor]
        new_score = _score(a)
        elapsed = time.time() - start_time

"""
Non‑negative step‑function optimiser.

The optimiser combines several complementary ideas:

1.  A very diverse initial population (power‑law, exponential,
    symmetric‑exponential, Gaussian, linear, random, alternating, and
    jittered copies of each).  Every seed is projected onto the ℓ¹‑simplex
    Σa = √(2 n).

2.  Short projected‑gradient runs using a smooth soft‑max approximation of
    the peak of the auto‑convolution.  The gradient is obtained by FFT,
    and a momentum update together with an exponential schedule for the
    soft‑max temperature β provides fast descent.

3.  A deterministic “mass‑transfer’’ refinement that moves a tiny amount
    of mass from indices that contribute most to the current peak to
    indices that contribute least.

4.  Upscaling to a larger length (up to ~15000) by linear interpolation,
    re‑projecting onto the simplex and trying a few deterministic
    reorderings (alternating order, reversed alternating).

5.  A high‑β gradient polishing phase (longer now than in the previous
    submission).

6.  An extended simulated‑annealing on the *order* of the values.
    The move set is a mix of targeted swaps (high‑contributor ↔
    low‑contributor) and occasional completely random swaps.

7.  A final tiny mass‑transfer pass and trimming of trailing zeros.

The whole pipeline respects the global time budget (default 1000 s) and
always returns the best sequence found so far.  All vectors are kept on
the ℓ¹‑simplex, so the evaluation value equals the maximum entry of the
auto‑convolution and can be compared directly to the target 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)² (with Σa=√(2n) this equals max)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection of v onto the simplex {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _symmetric_exponential(n: int, lam: float, target_sum: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-lam * np.abs(idx - centre))
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-((idx - centre) ** 2) / (2.0 * (sigma * n) ** 2))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Projected gradient descent on the soft‑max objective
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.5,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid flat regions
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        # increase β exponentially
        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (small moves)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-4,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # index of the current peak
        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]
        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Permutation search (simulated annealing on index order)
# ----------------------------------------------------------------------


def _permutation_search_explore(a0: np.ndarray,
                                deadline: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-5,
                                top_frac: float = 0.05,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Simulated annealing that mostly swaps a high‑contributor index with a
    low‑contributor one but sometimes performs a completely random swap.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    n = a.shape[0]
    start_time = time.time()

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        kmax_candidates = np.flatnonzero(conv == max_val)
        kmax = rng.choice(kmax_candidates)

        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        # 70 % targeted, 30 % random
        if rng.random() < 0.7:
            top_k = max(2, int(top_frac * n))
            low_k = max(2, int(low_frac * n))
            donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
            receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]
            donor = rng.choice(donor_candidates)
            receiver = rng.choice(receiver_candidates)
        else:
            donor = rng.integers(n)
            receiver = rng.integers(n)
            while receiver == donor:
                receiver = rng.integers(n)

        if donor == receiver:
            continue

        a[donor], a[receiver] = a[receiver], a[donor]
        new_score = _score(a)

        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12))

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] PermExplore new best={best_score:.6f}")
        else:
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑score non‑negative step‑function.
    The routine runs three main phases:
        1. Exploration with many short gradient‑descent runs from a diverse pool.
        2. Upscaling to a larger problem size followed by a long high‑β polishing
           and an extensive permutation‑annealing.
        3. Final tiny mass‑transfer pass and trimming.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # tiny safety margin

    # ------------------------------------------------------------------
    # 1. Exploration size
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n_expl = 8000
    elif budget_s >= 600:
        n_expl = 6000
    else:
        n_expl = 5000
    target_expl = np.sqrt(2.0 * n_expl)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # 2. Build a diverse pool of initial candidates
    # ------------------------------------------------------------------
    init_pool: list[np.ndarray] = []

    # power‑law family
    for p in np.linspace(0.30, 0.80, 9):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))

    # exponential (monotone) family
    for lam in np.linspace(0.001, 0.020, 6):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))

    # symmetric exponential (peak in the centre)
    for lam in np.linspace(0.001, 0.020, 5):
        init_pool.append(_symmetric_exponential(n_expl, float(lam), target_expl))

    # Gaussian‑like families
    for sigma in np.linspace(0.08, 0.30, 4):
        init_pool.append(_gaussian_sequence(n_expl, float(sigma), target_expl))

    # linear decreasing shape
    lin = np.linspace(1.0, 0.0, n_expl)
    lin *= target_expl / lin.sum()
    init_pool.append(lin)

    # random Dirichlet‑like points
    for _ in range(5):
        a = rng.random(n_expl)
        a *= target_expl / a.sum()
        init_pool.append(a)

    # alternating high‑low ordering of a power‑law seed
    init_pool.append(_alternating_order(_power_law_sequence(n_expl, 0.55, target_expl)))

    # reuse previous best if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_expl))

    # jittered copies of everything – increase diversity
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.008, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    # ------------------------------------------------------------------
    # 3. Exploration: many short gradient‑descent runs
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # keep a modest buffer for the later, more expensive phases
    buffer_for_later = 90.0

    while time.time() < deadline - buffer_for_later:
        # pick a seed
        if best_seq is not None and rng.random() < 0.55:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.004, size=n_expl)
            seed = seed + jitter
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = init_pool[rng.integers(len(init_pool))]

        # run a bounded GD instance (≈20 s each)
        run_dead = min(time.time() + 20.0, deadline - buffer_for_later + 1.0)
        if run_dead <= time.time():
            break

        a_opt, val_opt = _annealed_gradient_descent(
            seed,
            run_dead,
            target_expl,
            max_beta=5e5,
            step_factor=0.7,
            gamma=0.99,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Brief deterministic mass‑transfer refinement (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - buffer_for_later + 10.0:
        mt_dead = time.time() + min(12.0,
                                    deadline - time.time() - buffer_for_later + 10.0)
        a_ref, val_ref = _deterministic_mass_transfer(
            best_seq,
            mt_dead,
            target_expl,
            start_temp=0.5,
            end_temp=1e-4,
            low_frac=0.05,
        )
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] After MT refinement best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Upscaling to a larger problem size
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.55, target_expl)

    remaining = deadline - time.time()
    # decide final length based on remaining time
    if remaining > 200.0 and budget_s >= 800:
        n_final = 15000
    elif remaining > 120.0:
        n_final = 12000
    elif remaining > 70.0:
        n_final = 10000
    else:
        n_final = n_expl

    target_final = np.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # try a few deterministic reorderings before polishing
    candidates = [seq_big,
                  _alternating_order(seq_big),
                  _reversed_alternating(seq_big)]
    best_seq = seq_big
    best_score = _score(seq_big)
    for cand in candidates[1:]:
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand
    print(f"[{time.time():.2f}] Upscaled n={n_final}, start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final polishing – high‑β gradient descent (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        pol_dead = deadline - 20.0   # keep ~20 s for later steps
        a_pol, val_pol = _annealed_gradient_descent(
            best_seq,
            pol_dead,
            target_final,
            max_beta=2e6,
            step_factor=0.5,
            gamma=0.995,
        )
        if val_pol < best_score:
            best_score = val_pol
            best_seq = a_pol.copy()
            print(f"[{time.time():.2f}] Final polishing best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Extended permutation annealing (uses most of the remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 8.0:
        perm_dead = deadline - 0.5
        a_perm, val_perm = _permutation_search_explore(
            best_seq,
            perm_dead,
            start_temp=0.5,
            end_temp=1e-5,
            top_frac=0.05,
            low_frac=0.05,
        )
        if val_perm < best_score:
            best_score = val_perm
            best_seq = a_perm.copy()
            print(f"[{time.time():.2f}] After permutation search best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Tiny deterministic mass‑transfer pass (final polishing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        mt_dead = time.time() + min(5.0, deadline - time.time() - 0.4)
        a_mt2, val_mt2 = _deterministic_mass_transfer(
            best_seq,
            mt_dead,
            target_final,
            start_temp=0.3,
            end_temp=1e-5,
            low_frac=0.05,
        )
        if val_mt2 < best_score:
            best_score = val_mt2
            best_seq = a_mt2.copy()
            print(f"[{time.time():.2f}] Final MT best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 9. Trim trailing zeros for compactness
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # 10. Log final score (optional)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    globals()["height_sequence_1"] = final_seq
    return final_seq
