"""
Step‑function optimiser for the benchmark score.

The algorithm follows a multi‑stage scheme:

1. **Exploration** – generate a diverse pool of seeds (power‑law,
   exponential, random Dirichlet, linear) and run several short projected
   gradient descents on a moderate size (≈ 8000).  The best candidate is
   kept.

2. **Upscaling** – interpolate the best candidate to a larger length
   (≈ 12000‑15000) and renormalise it to the ℓ¹‑simplex (∑a = √(2 n)).

3. **High‑β polishing** – a long annealed gradient descent on the
   smooth‑max (soft‑max) approximation pushes the maximum of the
   autocorrelation down.

4. **Peak‑focused mass transfer** – deterministic moves that take mass
   from indices contributing most to the current peak and give it to
   low‑contributing indices; accepted with a simple simulated‑annealing
   schedule.

5. **Targeted permutation search** – simulated‑annealing swaps a
   “donor’’ index with a large product contribution with a “receiver’’
   having a small contribution, gradually flattening the autocorrelation
   peak.

6. **Final polishing** – another short soft‑max gradient descent and a
   tiny mass‑transfer pass.

All steps respect the non‑negativity and ℓ¹‑simplex constraints via a
Euclidean projection.  The routine respects the global 1000 s budget and
stores the resulting sequence in ``height_sequence_1`` for reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Benchmark score for a sequence that already satisfies
    ∑a = √(2 n).  In that case the objective reduces to the peak of the
    autocorrelation.
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    # The denominator 2 n / (∑a)² equals 1 because we normalise the sum.
    return float(max_b)


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the ℓ¹‑simplex of radius s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _linear_decreasing_sequence(n: int, target_sum: float) -> np.ndarray:
    a = np.linspace(1.0, 0.0, n, dtype=np.float64)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation from old length to new_len."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place the biggest, then the smallest, then the second biggest, …"""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution (smooth approximation)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Soft‑max gradient of max(convolution) w.r.t. a."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Projected gradient descent on the soft‑max objective
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.5,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Momentum‑gradient descent on the smooth‑max approximation.
    The learning rate is scaled with the current gradient norm.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Peak‑focused deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _mass_transfer_peak(a0: np.ndarray,
                       deadline: float,
                       target_sum: float,
                       start_temp: float = 0.5,
                       end_temp: float = 1e-5,
                       low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Moves a little mass from indices with a large product contribution to
    the current peak toward indices with a small contribution.
    Uses a simple simulated‑annealing acceptance rule.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start = time.time()

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        n = a.shape[0]
        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)

        # product contributions to the current peak
        contrib = np.zeros_like(a)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        donor = np.argmax(contrib)
        low_cnt = max(1, int(low_frac * n))
        receiver = rng.choice(np.argpartition(contrib, low_cnt - 1)[:low_cnt])

        if donor == receiver:
            continue

        # transfer a random fraction of the donor's mass
        max_delta = a[donor] * 0.5
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        new_score = _score(a)

        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT peak it={donor}->{receiver} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted permutation simulated annealing
# ----------------------------------------------------------------------


def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-5,
                                 top_frac: float = 0.05,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Swaps a high‑contributing index (with respect to the current peak)
    with a low‑contributing one.  Temperature decays slowly.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    n = a.shape[0]
    start = time.time()

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        kmax_candidates = np.flatnonzero(conv == max_val)
        kmax = rng.choice(kmax_candidates)

        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor = rng.choice(np.argpartition(-contrib, top_k - 1)[:top_k])
        receiver = rng.choice(np.argpartition(contrib, low_k - 1)[:low_k])

        if donor == receiver:
            continue

        # swap values
        a[donor], a[receiver] = a[receiver], a[donor]

        new_score = _score(a)

        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] Perm search new best={best_score:.6f}")
        else:
            # revert swap
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:

    Find a non‑negative step‑function with a low benchmark score.
    The procedure adapts its time allocation depending on the remaining
    budget and progressively refines the candidate.
    """

    deadline = start_time + budget_s - 0.5   # small safety margin

    # 1. Choose an exploration size
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n_expl = 8000
    elif budget_s >= 500:
        n_expl = 6000
    else:
        n_expl = 4000

    # 2. Build a diverse pool of seeds
    # ------------------------------------------------------------------
    init_pool = []

    for lam in np.linspace(0.001, 0.020, 8):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))

    # random Dirichlet‑like points
    for _ in range(6):
        a = rng.random(n_expl)
        a = _project_to_simplex(a, target_expl)
        init_pool.append(a)

    # linear decreasing
    init_pool.append(_linear_decreasing_sequence(n_expl, target_expl))

    # previous best, if any
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_expl))

    # jittered copies for extra diversity
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.008, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    # ------------------------------------------------------------------
    # 3. Exploration: many short gradient‑descent runs
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    while time.time() < deadline - 8.0:  # keep a few seconds for later stages
        # seed selection
        if best_seq is not None and rng.random() < 0.55:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.004, size=n_expl)
            seed = seed + jitter
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)

        # short run (≈15 s)
        run_deadline = min(time.time() + 15.0, deadline - 6.0)
        if run_deadline <= time.time():
            break

        a_opt, val_opt = _annealed_gradient_descent(
            seed,
            run_deadline,
            target_expl,
            max_beta=5e5,
            step_factor=0.6,
            gamma=0.99,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Brief deterministic mass‑transfer refinement (optional)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 6.0:
        mt_dead = time.time() + (deadline - time.time()) * 0.12
        a_ref, val_ref = _mass_transfer_peak(
            best_seq,
            mt_dead,
            target_expl,
            start_temp=0.5,
            end_temp=1e-4,
            low_frac=0.05,
        )
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] After MT refinement best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Upscaling to a larger problem size
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.55, target_expl)

    remaining = deadline - time.time()
    if remaining > 150.0 and budget_s >= 800:
        n_final = 15000
    elif remaining > 80.0:
        n_final = 12000
    else:
        n_final = n_expl

    target_final = np.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # try a few simple reorderings
    candidates = [seq_big,
                  _alternating_order(seq_big),
                  seq_big[::-1]]
    best_seq = seq_big
    best_score = _score(seq_big)
    for cand in candidates[1:]:
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand
    print(f"[{time.time():.2f}] Upscaled length={n_final} start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Long high‑β gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 12.0:
        gd_dead = min(deadline - 6.0, time.time() + max(30.0, (deadline - time.time()) * 0.4))
        a_gd, val_gd = _annealed_gradient_descent(
            best_seq,
            gd_dead,
            target_final,
            max_beta=2e6,
            step_factor=0.45,
            gamma=0.995,
        )
        if val_gd < best_score:
            best_score = val_gd
            best_seq = a_gd.copy()
            print(f"[{time.time():.2f}] High‑β GD best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Peak‑focused mass‑transfer refinement (medium duration)
    # ------------------------------------------------------------------
    if time.time() < deadline - 8.0:
        mt2_dead = time.time() + max(5.0, (deadline - time.time()) * 0.15)
        a_mt2, val_mt2 = _mass_transfer_peak(
            best_seq,
            mt2_dead,
            target_final,
            start_temp=0.5,
            end_temp=1e-5,
            low_frac=0.05,
        )
        if val_mt2 < best_score:
            best_score = val_mt2
            best_seq = a_mt2.copy()
            print(f"[{time.time():.2f}] MT2 best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Extended targeted permutation search (uses most remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 4.0:
        perm_dead = deadline - 0.5
        a_perm, val_perm = _permutation_targeted_search(
            best_seq,
            perm_dead,
            start_temp=0.5,
            end_temp=1e-5,
            top_frac=0.05,
            low_frac=0.05,
        )
        if val_perm < best_score:
            best_score = val_perm
            best_seq = a_perm.copy()
            print(f"[{time.time():.2f}] After permutation search best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 9. Final short soft‑max gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        final_gd_dead = min(deadline - 0.5, time.time() + 5.0)
        a_final, val_final = _annealed_gradient_descent(
            best_seq,
            final_gd_dead,
            target_final,
            max_beta=3e6,
            step_factor=0.35,
            gamma=0.997,
        )
        if val_final < best_score:
            best_score = val_final
            best_seq = a_final.copy()
            print(f"[{time.time():.2f}] Final GD best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 10. Tiny deterministic mass‑transfer pass (final polishing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        mt3_dead = time.time() + min(3.0, deadline - time.time() - 0.4)
        a_mt3, val_mt3 = _mass_transfer_peak(
            best_seq,
            mt3_dead,
            target_final,
            start_temp=0.3,
            end_temp=1e-5,
            low_frac=0.05,
        )
        if val_mt3 < best_score:
            best_score = val_mt3
            best_seq = a_mt3.copy()
            print(f"[{time.time():.2f}] Post‑polish MT3 best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 11. Trim trailing near‑zero entries for compactness
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.flatnonzero(arr > 1e-12)
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # 12. Optional final evaluation log
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for reuse in subsequent calls
    globals()["height_sequence_1"] = final_seq
    return final_seq
