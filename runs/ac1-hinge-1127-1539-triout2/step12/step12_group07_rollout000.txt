"""
Hybrid optimizer for the step‑function challenge.

The routine runs five adaptive phases:

1. **Exploration** – generate a diverse pool of random seeds
   (power‑law, exponential, uniform simplex) and run short
   annealed‑gradient descent runs on a modest size (≈4–5 k).

2. **Refinement on the small scale** – a quick mass‑transfer
   simulated‑annealing pass.

3. **Upscaling** – linearly interpolate the best small‑scale solution
   to a larger length (≈15–25 k) and project onto the ℓ¹‑simplex.

4. **Ordering & local improvement** – try several deterministic
   reorderings (alternating, even‑spread, modular spreads with random
   coprime steps), a short random‑permutation search, and a
   hill‑climbing swap phase that accepts only improvements.

5. **Large‑scale polishing** – projected sub‑gradient descent followed
   by a final mass‑transfer pass.

All phases respect non‑negativity and the ℓ¹‑norm constraint.
The best sequence found before the deadline is returned.
"""
import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (linear convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray, target_sum: float) -> float:
    """Objective 2·n·max(a*a)/(∑a)² (for ∑a=target_sum)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), then renormalised to L1 norm = target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """
    Spread the large values uniformly by placing them on a cyclic sequence
    with a step coprime to the length (≈√n).
    """
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


def _modular_spread_order(seq: np.ndarray, step: int) -> np.ndarray:
    """Place values in decreasing order using a fixed modular step."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    out = np.empty_like(seq)
    pos = 0
    for v in sorted_vals:
        out[pos] = v
        pos = (pos + step) % n
    return out


# ----------------------------------------------------------------------
# Annealed gradient descent (soft‑max approximation)
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.9,
                               gamma: float = 0.98) -> tuple[np.ndarray, float]:
    """Momentum projected GD on the soft‑max approximation of the max‑conv."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a, target_sum)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_b = np.max(b)
        exps = np.exp(beta * (b - max_b))
        p = exps / np.sum(exps)

        a_rev = a[::-1]
        Ar_fft = np.fft.rfft(a_rev, n=size)
        Pr_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a, target_sum)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AGD it={it:5d} β={beta:7.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent using the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a, target_sum)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        # indices attaining the maximum (within tiny tolerance)
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]

        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a, target_sum)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SGD it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy mass‑transfer with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.5,
                                      end_temp: float = 1e-4,
                                      low_frac: float = 0.06) -> tuple[np.ndarray, float]:
    """
    Greedy mass moves from a “donor’’ (large contribution to the peak)
    to a “receiver’’ (small contribution) with a Metropolis acceptance rule.
    """
    a = a0.copy()
    cur_score = _score(a, target_sum)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a, target_sum)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Random permutation search (fast heuristic)
# ----------------------------------------------------------------------


def _permutation_search(base_seq: np.ndarray,
                        trials: int,
                        deadline: float) -> tuple[np.ndarray, float]:
    """Try random reorderings of `base_seq` and keep the best one."""
    best_seq = base_seq.copy()
    best_val = _max_convolution_fft(best_seq)
    rng = np.random.default_rng()
    n = base_seq.shape[0]

    for _ in range(trials):
        if time.time() >= deadline:
            break
        perm = rng.permutation(base_seq)
        val = _max_convolution_fft(perm)
        if val < best_val:
            best_val = val
            best_seq = perm.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Hill‑climbing swap improvement (purely permutation based)
# ----------------------------------------------------------------------


def _swap_hill_climb(seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Randomly swap two positions; keep the swap only if it strictly reduces
    the max convolution. Stops when time runs out.
    """
    a = seq.copy()
    best_val = _max_convolution_fft(a)
    n = a.shape[0]
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]
        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
            # keep swap
        else:
            # revert swap
            a[i], a[j] = a[j], a[i]
    return a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine follows the multi‑phase schedule described in the module
    docstring and never exceeds `budget_s` seconds.
    """
    start_time = time.time()
    # safety margin of a few tenths of a second
    deadline = start_time + budget_s - 0.4
    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # Phase 1 – rapid exploration on a modest size
    # --------------------------------------------------------------
    # Choose a modest length that fits many gradient runs.
    n_small = 5000 if budget_s >= 800 else (4000 if budget_s >= 500 else 3000)
    target_small = math.sqrt(2.0 * n_small)

    # Build a diverse pool of seeds
    seeds = []

    for p in np.linspace(0.2, 2.0, 12):
        seeds.append(_power_law_sequence(n_small, float(p), target_small))
    for lam in np.linspace(0.0005, 0.03, 12):
        seeds.append(_exponential_sequence(n_small, float(lam), target_small))
    for _ in range(12):
        seeds.append(_random_simplex(n_small, target_small, rng))
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        seeds.append(_resize_sequence(prev, n_small))

    # Jitter copies to increase diversity
    for seq in list(seeds):
        jitter = rng.normal(scale=target_small * 0.02, size=n_small)
        seq_j = np.maximum(seq + jitter, 0.0)
        seq_j = _project_to_simplex(seq_j, target_small)
        seeds.append(seq_j)

    best_seq = None
    best_val = np.inf

    # Allocate roughly 30 % of the total budget to this phase
    phase1_deadline = min(deadline - 30.0, start_time + budget_s * 0.30)

    while time.time() < phase1_deadline:
        seed = rng.choice(seeds)

        # each annealed run gets a few seconds
        run_deadline = min(time.time() + 6.0, phase1_deadline)
        a_opt, val_opt = _annealed_gradient_descent(
            seed, run_deadline, target_small,
            max_beta=8e5, step_factor=0.9, gamma=0.98)

        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Phase 1 new best={best_val:.6f}")

    if best_seq is None:
        # fallback: random simplex
        best_seq = _random_simplex(n_small, target_small, rng)

    # --------------------------------------------------------------
    # Phase 2 – short mass‑transfer refinement on the small scale
    # --------------------------------------------------------------
    if time.time() < deadline - 20.0:
        mt_deadline = min(deadline - 10.0, time.time() + 10.0)
        a_mt, val_mt = _mass_transfer_simulated_annealing(
            best_seq, mt_deadline, target_small,
            start_temp=0.5, end_temp=1e-5, low_frac=0.07)
        if val_mt < best_val:
            best_val = val_mt
            best_seq = a_mt.copy()
            print(f"[{time.time():.2f}] Phase 2 MT refined best={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – upscale to a larger problem size
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 200:
        n_large = 25000
    elif remaining > 120:
        n_large = 20000
    elif remaining > 80:
        n_large = 15000
    else:
        n_large = max(n_small, int(remaining * 150))  # very rough heuristic

    target_large = math.sqrt(2.0 * n_large)

    seq_large = _resize_sequence(best_seq, n_large)
    seq_large = _project_to_simplex(seq_large, target_large)

    # Try several deterministic reorderings
    candidates = [
        seq_large,
        _alternating_order(seq_large),
        _even_spread_order(seq_large)
    ]

    # modular spread with several random coprime steps
    steps = set()
    while len(steps) < 10:
        step = rng.integers(1, n_large)
        if math.gcd(step, n_large) == 1:
            steps.add(int(step))
    for step in steps:
        candidates.append(_modular_spread_order(seq_large, step))

    seq_large = min(candidates, key=_max_convolution_fft)
    print(f"[{time.time():.2f}] Upscaled to {n_large} elements, max={_max_convolution_fft(seq_large):.6f}")

    # --------------------------------------------------------------
    # Phase 4 – permutation/random search
    # --------------------------------------------------------------
    if time.time() < deadline - 10.0:
        perm_deadline = time.time() + min(8.0, deadline - time.time() - 5.0)
        trials = 60
        seq_large, _ = _permutation_search(seq_large, trials, perm_deadline)
        print(f"[{time.time():.2f}] After permutation search max={_max_convolution_fft(seq_large):.6f}")

    # --------------------------------------------------------------
    # Phase 5 – hill‑climbing swaps
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        swap_deadline = time.time() + min(10.0, deadline - time.time() - 2.0)
        seq_large, best_swap_val = _swap_hill_climb(seq_large, swap_deadline)
        print(f"[{time.time():.2f}] Swap hill‑climb best max={best_swap_val:.6f}")

    # --------------------------------------------------------------
    # Phase 6 – large‑scale sub‑gradient descent
    # --------------------------------------------------------------
    if time.time() < deadline - 3.0:
        sub_deadline = deadline - 2.0
        a_sub, val_sub = _subgradient_descent(
            seq_large, sub_deadline, target_large,
            step_factor=0.45, gamma=0.995)
        if val_sub < best_val:
            best_val = val_sub
            seq_large = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑grad final best={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 7 – final mass‑transfer pass (time‑permitting)
    # --------------------------------------------------------------
    if time.time() < deadline - 1.0:
        mt_deadline = deadline - 0.5
        a_mt, val_mt = _mass_transfer_simulated_annealing(
            seq_large, mt_deadline, target_large,
            start_temp=0.3, end_temp=1e-5, low_frac=0.06)
        if val_mt < best_val:
            best_val = val_mt
            seq_large = a_mt.copy()
            print(f"[{time.time():.2f}] Final MT best={best_val:.6f}")

    # --------------------------------------------------------------
    # Trim trailing near‑zero entries and return
    # --------------------------------------------------------------
    arr = np.asarray(seq_large, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional logging of the final evaluator value
    try:
        ev = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {ev:.6f}")
    except Exception:
        pass

    # expose the sequence for possible later runs
    globals()["height_sequence_2"] = final_seq
    return final_seq
