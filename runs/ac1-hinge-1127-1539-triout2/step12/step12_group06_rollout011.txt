def _simulated_annealing(a, target_sum, deadline):
    n = len(a)
    cur = a.copy()
    cur_score = _score(cur)
    best = cur.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    # parameters
    initial_temp = cur_score * 0.05
    final_temp = cur_score * 0.001
    total_iter_est = (deadline - time.time()) * 5000  # guess 5000 iterations per sec
    # We'll compute temperature based on elapsed fraction
    iter_count = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tf = elapsed / (deadline - start_time)
        temp = initial_temp * ((final_temp / initial_temp) ** tf)  # geometric cooling

        # choose move
        if rng.random() < 0.3:  # swap
            i, j = rng.choice(n, size=2, replace=False)
            if cur[i] == cur[j]:
                continue
            # swap
            cur[i], cur[j] = cur[j], cur[i]
            new_score = _score(cur)
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
                print(f"[{time.time():.2f}] SA swap best={best_score:.6f}")
            # acceptance
            if new_score <= cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
                cur_score = new_score
            else:
                # revert
                cur[i], cur[j] = cur[j], cur[i]
        else:
            # mass transfer
            donor = rng.choice(n)
            if cur[donor] <= 1e-14:
                continue
            receiver = rng.choice(n)
            if donor == receiver:
                continue
            # max delta = cur[donor] * 0.5 maybe
            max_delta = cur[donor] * 0.5
            # also scale with temperature
            max_delta = max_delta * (1 - tf) + (target_sum * 0.001) * tf
            delta = rng.random() * max_delta
            cur[donor] -= delta
            cur[receiver] += delta
            # keep nonneg
            # not needed because donor more than delta
            new_score = _score(cur)
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
                print(f"[{time.time():.2f}] SA mass best={best_score:.6f}")
            if new_score <= cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
                cur_score = new_score
            else:
                # revert
                cur[donor] += delta
                cur[receiver] -= delta
        iter_count += 1

    return best, best_score

Since (f*g)[k] = sum_i f[i] * g[k - i] (by convolution). It's linear in g.

Thus it's a linear program with n variables g_i and (2n-1) inequality constraints: for each k, sum_{i} f_i * g_{k - i} <= M. def _simulated_annealing(a0: np.ndarray, target_sum: float, deadline: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _score(a)
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    n = a.shape[0]
    start = time.time()
    it = 0
    while time.time() < deadline:
        # temperature schedule
        frac = (time.time() - start) / (deadline - start)
        temp = cur_score * (0.5 ** frac)  # geometric cooling from current score down to small
        # choose move type
        if rng.random() < 0.35:
            # swap
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            new_score = _score(a)
            accept = new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if accept:
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best = a.copy()
                    print(f"[{time.time():.2f}] SA swap best={best_score:.6f}")
            else:
                a[i], a[j] = a[j], a[i]  # revert
        else:
            # mass transfer
            donor = rng.integers(n)
            if a[donor] <= 1e-14:
                continue
            receiver = rng.integers(n)
            if donor == receiver:
                continue
            max_delta = a[donor] * 0.5
            # scale delta by temperature fraction to encourage larger jumps early
            max_delta = max_delta * (1 - frac) + (target_sum * 0.001) * frac
            delta = rng.random() * max_delta
            a[donor] -= delta
            a[receiver] += delta
            new_score = _score(a)
            accept = new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if accept:
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best = a.copy()
                    print(f"[{time.time():.2f}] SA mass best={best_score:.6f}")
                # occasionally renormalize
                if it % 2000 == 0:
                    a = _project_to_simplex(a, target_sum)
            else:
                # revert
                a[donor] += delta
                a[receiver] -= delta
        it += 1
    return best, best_score

"""
Hybrid optimiser for the non‑negative step‑function problem.

Strategy
--------
1. Build a diverse pool of seed vectors (power‑law, exponential,
   random Dirichlet, linear, previous best) of a modest length.
2. Quick projected sub‑gradient runs on the pool to obtain a decent
   candidate.
3. Upscale the best candidate to a larger length (12‑18 k entries)
   and project onto the simplex {x≥0, Σx = √(2 n)}.
4. Long runs of projected sub‑gradient descent on the exact max‑convolution
   (the non‑smooth objective) to drive the peak down.
5. High‑β soft‑max (annealed) gradient descent for fine polishing.
6. Targeted permutation SA that swaps indices contributing heavily to the
   current peak.
7. Custom simulated‑annealing that moves mass between positions while
   preserving the sum, again guided by the current peak.
8. A short deterministic mass‑transfer pass (greedy “donor→receiver”).
9. Trim trailing zeros and return the best sequence found.

All phases respect the global deadline (default 1000 s).  The final
sequence is stored in the global variable ``height_sequence_1`` for
possible reuse by subsequent runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators (used for the seed pool)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) shape, renormalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """exp(-lam·i) shape, renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation from old length to new_len."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → left, 2nd largest → right, 3rd → next left, …."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


# ----------------------------------------------------------------------
# Scoring routine (objective = max convolution entry)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Objective value: 2·n·max(conv)/sum(a)^2."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Exact max‑convolution sub‑gradient
# ----------------------------------------------------------------------


def _max_conv_and_subgrad(a: np.ndarray):
    """Return (conv, max_val, k_max, subgradient of the peak)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    kmax = int(np.argmax(conv))
    max_val = conv[kmax]

    grad = np.zeros_like(a)
    lo = max(0, kmax - (n - 1))
    hi = min(kmax, n - 1)
    idx = np.arange(lo, hi + 1)
    partner = a[kmax - idx]
    grad[lo:hi + 1] = 2.0 * partner
    return conv, max_val, kmax, grad


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Momentum projected sub‑gradient descent for the non‑smooth max‑convolution.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    while time.time() < deadline:
        _, _, _, grad = _max_conv_and_subgrad(a)

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_factor * target_sum / norm
            a -= lr * momentum
        else:
            a -= step_factor * target_sum * grad / (np.linalg.norm(grad) + 1e-12)

        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if rng.random() < 0.02:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] SubGD  score={best_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Soft‑max (annealed) gradient descent
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(conv(a,a))."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    grad = 2.0 * np.fft.irfft(np.fft.rfft(a_rev, n=size) *
                               np.fft.rfft(p, n=size), n=size)[: 2 * n - 1]
    grad = grad[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 1e6,
                               step_factor: float = 0.5,
                               gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max surrogate,
    slowly increasing the temperature parameter β.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if rng.random() < 0.02:
            jitter = rng.normal(scale=step_base / (beta + 1.0) * 0.001,
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Anneal β={beta:.1f}  score={best_val:.6f}")

        if beta < max_beta and rng.random() < 0.5:
            beta = min(beta * 2.0, max_beta)

    return best_a, best_val


# ----------------------------------------------------------------------
# Targeted permutation search (Metropolis swaps)
# ----------------------------------------------------------------------


def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.1,
                                 end_temp: float = 1e-4,
                                 top_frac: float = 0.02,
                                 low_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """Swap indices that heavily contribute to the current peak."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donors = np.argpartition(-contrib, top_k - 1)[:top_k]
        receivers = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donors)
        receiver = rng.choice(receivers)

        if donor == receiver:
            continue

        a[donor], a[receiver] = a[receiver], a[donor]
        new_score = _score(a)

        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** frac

        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] PermSwap new best={best_score:.6f}")

        if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / max(temp, 1e-12)):
            # keep swap
            pass
        else:
            # revert
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (greedy donor→receiver)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy annealed donor→receiver moves."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start = time.time()
    conv = _conv_fft(a)
    n = a.shape[0]

    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]          # contribution weight to the peak
        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional jitter to keep dynamics alive
        if rng.random() < 0.01:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Custom simulated annealing moving mass between entries + swaps
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Simple SA that preserves the total sum (mass moves + swaps)."""
    a = a0.copy()
    cur_score = _score(a)
    best = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    n = a.shape[0]
    start = time.time()
    it = 0

    while time.time() < deadline:
        # geometric temperature schedule based on elapsed fraction
        frac = (time.time() - start) / (deadline - start)
        temp = max(cur_score * (0.5 ** frac), 1e-12)

        if rng.random() < 0.35:
            # ---- Swap move ----
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            new_score = _score(a)
            accept = new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temp)
            if accept:
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best = a.copy()
                    print(f"[{time.time():.2f}] SA swap best={best_score:.6f}")
            else:
                a[i], a[j] = a[j], a[i]          # revert
        else:
            # ---- Mass‑transfer move ----
            donor = rng.integers(n)
            if a[donor] <= 1e-14:
                continue
            receiver = rng.integers(n)
            if donor == receiver:
                continue

            max_delta = a[donor] * 0.5
            # early iterations allow larger jumps
            max_delta = max_delta * (1.0 - frac) + (target_sum * 0.001) * frac
            delta = rng.random() * max_delta

            a[donor] -= delta
            a[receiver] += delta
            new_score = _score(a)
            accept = new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temp)
            if accept:
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best = a.copy()
                    print(f"[{time.time():.2f}] SA mass best={best_score:.6f}")
                # occasional re‑projection to avoid drift
                if it % 2500 == 0:
                    a = _project_to_simplex(a, target_sum)
            else:
                # revert
                a[donor] += delta
                a[receiver] -= delta

        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with a low benchmark score.
    The function respects the 1000 s budget and stores the best vector
    in the global variable ``height_sequence_1`` for possible reuse.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 1.  Choose a modest initial length for rapid exploration
    # --------------------------------------------------------------
    n_expl = 8000 if budget_s >= 800 else 6000
    target_expl = np.sqrt(2.0 * n_expl)

    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # 2.  Build a diverse pool of seed sequences
    # --------------------------------------------------------------
    init_pool = []

    for p in [0.45, 0.55, 0.65]:
        init_pool.append(_power_law_sequence(n_expl, p, target_expl))

    for lam in [0.001, 0.004, 0.008, 0.012]:
        init_pool.append(_exponential_sequence(n_expl, lam, target_expl))

    for _ in range(5):
        a = rng.random(n_expl)
        a = a / a.sum() * target_expl
        init_pool.append(a)

    lin = np.linspace(1.0, 0.0, n_expl)
    lin = lin / lin.sum() * target_expl
    init_pool.append(lin)

    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_expl))

    # jittered copies of everything already in the pool
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.008, size=n_expl)
        cand = np.maximum(seq + jitter, 0.0)
        cand = _project_to_simplex(cand, target_expl)
        init_pool.append(cand)

    # --------------------------------------------------------------
    # 3.  Quick sub‑gradient runs on the moderate size (exploration)
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    while time.time() < deadline - 70:   # reserve ~70 s for the heavy phases
        seed = rng.choice(init_pool)
        run_dead = time.time() + 6.0      # very fast 6‑s runs
        a_opt, val_opt = _subgradient_descent(seed, run_dead, target_expl,
                                              step_factor=0.6, gamma=0.97)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration best={best_score:.6f}")

    # --------------------------------------------------------------
    # 4.  Upscale to a larger length (more degrees of freedom)
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.55, target_expl)

    remaining = deadline - time.time()
    if remaining > 300:
        n_final = 18000
    elif remaining > 150:
        n_final = 15000
    else:
        n_final = 12000

    target_final = np.sqrt(2.0 * n_final)
    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # try a few deterministic reorderings
    cand_list = [seq_big,
                 _alternating_order(seq_big),
                 _reversed_alternating(seq_big)]
    best_seq = seq_big
    best_score = _score(seq_big)
    for c in cand_list[1:]:
        sc = _score(c)
        if sc < best_score:
            best_score = sc
            best_seq = c.copy()
    print(f"[{time.time():.2f}] Upscaled n={n_final} start score={best_score:.6f}")

    # --------------------------------------------------------------
    # 5.  Longer sub‑gradient descent on the enlarged vector
    # --------------------------------------------------------------
    if time.time() < deadline - 150:
        sub_dead = time.time() + min(80.0, deadline - time.time() - 40.0)
        a_opt, val_opt = _subgradient_descent(best_seq, sub_dead,
                                              target_final,
                                              step_factor=0.5, gamma=0.99)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] SubGD (big) best={best_score:.6f}")

    # --------------------------------------------------------------
    # 6.  Annealed soft‑max gradient descent (high β)
    # --------------------------------------------------------------
    if time.time() < deadline - 100:
        anneal_dead = time.time() + min(70.0, deadline - time.time() - 30.0)
        a_opt, val_opt = _annealed_gradient_descent(best_seq, anneal_dead,
                                                    target_final,
                                                    max_beta=5e6,
                                                    step_factor=0.5,
                                                    gamma=0.995)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Annealed GD best={best_score:.6f}")

    # --------------------------------------------------------------
    # 7.  Targeted permutation search
    # --------------------------------------------------------------
    if time.time() < deadline - 60:
        perm_dead = time.time() + min(50.0, deadline - time.time() - 20.0)
        a_opt, val_opt = _permutation_targeted_search(best_seq, perm_dead,
                                                      start_temp=0.2,
                                                      end_temp=1e-5,
                                                      top_frac=0.03,
                                                      low_frac=0.03)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Permutation best={best_score:.6f}")

    # --------------------------------------------------------------
    # 8.  Custom simulated‑annealing (mass moves + swaps)
    # --------------------------------------------------------------
    if time.time() < deadline - 30:
        sa_dead = time.time() + min(30.0, deadline - time.time() - 10.0)
        a_opt, val_opt = _simulated_annealing(best_seq, target_final, sa_dead)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] SA mass‑transfer best={best_score:.6f}")

    # --------------------------------------------------------------
    # 9.  Final deterministic mass‑transfer pass (greedy)
    # --------------------------------------------------------------
    if time.time() < deadline - 10:
        mt_dead = time.time() + min(10.0, deadline - time.time() - 2.0)
        a_opt, val_opt = _deterministic_mass_transfer(best_seq, mt_dead,
                                                      target_final,
                                                      start_temp=0.3,
                                                      end_temp=1e-5,
                                                      low_frac=0.05)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Final MT best={best_score:.6f}")

    # --------------------------------------------------------------
    # 10. Trim trailing zeros for compactness
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # --------------------------------------------------------------
    # 11. Optional final evaluation logging
    # --------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    globals()["height_sequence_1"] = final_seq
    return final_seq
