"""
A compact optimiser for the “max‑convolution” benchmark.

The objective (when the ℓ¹‑norm is √(2n)) equals the peak entry of the
auto‑convolution a*a.  The algorithm works in three phases:

1. **Exploration** – many short runs of a *sub‑gradient* method that
   directly minimizes the current peak (the true max, not a smooth
   surrogate).  The iterates are projected onto the simplex
   Σa=√(2n), a≥0.  A diverse pool of initial shapes (power‑law,
   exponential, random) is used and jittered between runs.

2. **Refinement** – a short deterministic “mass‑transfer” local search
   that moves a tiny amount of mass from the most contributing index
   of the current peak to a weak one.  It is a simple simulated‑annealing
   on the same objective.

3. **Permutation search** – a long simulated‑annealing that swaps two
   positions.  The swap candidates are chosen from the tails of the
   contribution distribution to the current peak, which is the most
   effective neighbourhood for reducing the peak.

The routine respects the time budget (default 1000 s) and returns the
best sequence it found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())     # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Simplex projection (ℓ¹‑simplex with fixed sum)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Scoring (peak of the auto‑convolution)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Objective value – the peak of a*a (sum is √(2n) ⇒ score = max)."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to change length, preserving shape."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Sub‑gradient descent for max‑convolution (direct optimisation)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step0: float = 1.0,
                         decay: float = 0.995,
                         top_m: int = 3) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    The gradient for the peak at index k is 2*a shifted by k.
    For robustness we average the gradients of the `top_m` largest
    convolution entries.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    it = 0
    while time.time() < deadline:
        # full convolution (needed for peak indices)
        conv = _conv_fft(a)

        # indices of the `top_m` peaks (allow ties)
        idx_peaks = np.argpartition(-conv, range(top_m))[:top_m]

        grad = np.zeros_like(a)
        n = a.shape[0]

        for k in idx_peaks:
            # range of i that actually participates in b[k] = Σ a[i] a[k-i]
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i_range = np.arange(lo, hi + 1, dtype=np.int64)
            partner = k - i_range
            grad[i_range] += 2.0 * a[partner]

        # average because we summed over top_m entries
        grad /= top_m

        # step size schedule
        lr = step0 * (decay ** it)
        if lr < 1e-12:
            lr = 1e-12

        a -= lr * grad
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} lr={lr:.2e} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (small moves)
# ----------------------------------------------------------------------


def _mass_transfer_refinement(a0: np.ndarray,
                             deadline: float,
                             target_sum: float,
                             start_temp: float = 0.5,
                             end_temp: float = 1e-4,
                             low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic transfers of mass from a donor (high contribution)
    to a receiver (low contribution) with simulated‑annealing acceptance.
    """
    a = a0.copy()
    cur_val = _score(a)
    best_a = a.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    start = time.time()
    n = a.shape[0]

    while time.time() < deadline:
        # temperature decreasing (exponential schedule)
        tau = (time.time() - start) / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        conv = _conv_fft(a)
        kmax = int(conv.argmax())

        # donors/receivers based on contribution to the current peak
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        w = a[idx] + a[kmax - idx]   # contribution of each participating index

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        receivers = idx[np.argpartition(w, low_cnt - 1)[:low_cnt]]
        receiver = rng.choice(receivers)

        if donor == receiver:
            continue

        # transfer a random fraction of the donor's mass
        max_delta = a[donor] * 0.5
        delta = rng.random() * max_delta

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        new_val = _score(a)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(temp, 1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_val:.6f}")

        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_val


# ----------------------------------------------------------------------
# Targeted permutation search (simulated annealing on index order)
# ----------------------------------------------------------------------


def _permutation_search(a0: np.ndarray,
                        deadline: float,
                        start_temp: float = 0.5,
                        end_temp: float = 1e-5,
                        top_frac: float = 0.05,
                        low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing that swaps a high‑contributing index with a
    low‑contributing one (contributions are defined w.r.t. the current
    peak of the convolution).
    """
    a = a0.copy()
    cur_val = _score(a)
    best_a = a.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    n = a.shape[0]
    start = time.time()

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        # pick one of the (possibly many) peaks uniformly
        kmax_candidates = np.flatnonzero(conv == max_val)
        kmax = rng.choice(kmax_candidates)

        # contribution of each index to that peak
        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)

        if donor == receiver:
            continue

        # swap the two entries
        a[donor], a[receiver] = a[receiver], a[donor]

        new_val = _score(a)

        # temperature schedule (slow exponential decay)
        tau = (time.time() - start) / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(temp, 1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] PermSwap new best={best_val:.6f}")
        else:
            # revert
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low peak convolution.
    The routine respects the time budget and returns the best sequence it
    found (as a plain list of floats).  The global name ``height_sequence_1``
    is updated for possible reuse in later runs.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    # ------------------------------------------------------------------
    # Choose a problem size (larger size gives more flexibility)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 500:
        n = 10000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)
    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Build a pool of diverse initial candidates
    # ------------------------------------------------------------------
    init_pool = []

    # power‑law shapes (exponent sweep)
    for p in np.linspace(0.3, 0.9, 9):
        init_pool.append(_power_law_sequence(n, float(p), target_sum))

    # exponential shapes (rate sweep)
    for lam in np.linspace(0.001, 0.012, 7):
        init_pool.append(_exponential_sequence(n, float(lam), target_sum))

    # pure random points on the simplex
    for _ in range(6):
        init_pool.append(_random_simplex(n, target_sum, rng))

    # slightly jittered copies of everything to increase diversity
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_sum * 0.006, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        init_pool.append(seq_j)

    # reuse the best sequence from a previous run if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n))

    # ------------------------------------------------------------------
    # Phase 1 – many short sub‑gradient runs (exploration)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    # allocate most of the time to exploration; leave ~30 % for later phases
    exploration_deadline = deadline - max(30.0, 0.25 * (deadline - start_time))

    while time.time() < exploration_deadline:
        # pick a seed: either jittered best or a random pool member
        if best_seq is not None and rng.random() < 0.55:
            seed = best_seq.copy()
            seed += rng.normal(scale=target_sum * 0.004, size=n)
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_sum)
        else:
            seed = rng.choice(init_pool)

        # run a bounded sub‑gradient descent (≈12‑15 s)
        run_dead = min(time.time() + 13.0, exploration_deadline)
        a_opt, val_opt = _subgradient_descent(
            seed,
            run_dead,
            target_sum,
            step0=target_sum * 0.2,
            decay=0.996,
            top_m=3,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – deterministic mass‑transfer refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 20.0:
        mt_dead = time.time() + min(7.0, (deadline - time.time()) * 0.10)
        a_ref, val_ref = _mass_transfer_refinement(
            best_seq,
            mt_dead,
            target_sum,
            start_temp=0.6,
            end_temp=1e-4,
            low_frac=0.05,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] Post‑MT best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – extended permutation search (uses virtually all remaining time)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 3.0:
        perm_dead = deadline - 0.5
        a_perm, val_perm = _permutation_search(
            best_seq,
            perm_dead,
            start_temp=0.6,
            end_temp=1e-5,
            top_frac=0.04,
            low_frac=0.04,
        )
        if val_perm < best_val:
            best_val = val_perm
            best_seq = a_perm.copy()
            print(f"[{time.time():.2f}] Final permutation best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – tiny mass‑transfer polishing (very short)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1.0:
        mt2_dead = time.time() + min(4.0, deadline - time.time() - 0.4)
        a_mt2, val_mt2 = _mass_transfer_refinement(
            best_seq,
            mt2_dead,
            target_sum,
            start_temp=0.3,
            end_temp=1e-5,
            low_frac=0.04,
        )
        if val_mt2 < best_val:
            best_val = val_mt2
            best_seq = a_mt2.copy()
            print(f"[{time.time():.2f}] Tiny MT best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing zeros for compactness
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.flatnonzero(arr > 1e-12)
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for possible later reuse
    globals()["height_sequence_1"] = final_seq
    return final_seq
