"""
Improved step‑function optimizer for the maximal self‑convolution problem.

The strategy builds on the previous four‑stage pipeline but adds
* a richer parametric pre‑search (including edge‑heavy power‑laws,
  inverse‑product families and a two‑parameter beta‑distribution),
* a larger problem size (up to 30 000 points when the time budget
  permits it),
* more aggressive Adam restarts and a longer sub‑gradient phase,
* a simple simulated‑annealing style random mass‑transfer that runs
  in the remaining budget after deterministic DMT.

All steps respect the simplex constraint Σa = √(2 n) and keep the
variables non‑negative.  The final sequence is trimmed of trailing
near‑zero entries before being returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution of a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for the pre‑search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _inverse_product_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Heavy at the edges: a_i ∝ ((i+1)*(n-i))^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1.0) * (n - i)) ** (-p)
    a *= target_sum / a.sum()
    return a


def _beta_distribution_sequence(n: int, alpha: float, beta: float, target_sum: float) -> np.ndarray:
    """Beta‑type shape on a discrete grid: (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    a = np.where(np.isfinite(a), a, 0.0)
    a *= target_sum / a.sum()
    return a


def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gd(a0: np.ndarray,
             deadline: float,
             target_sum: float,
             max_beta: float = 2e5,
             step_factor: float = 0.8,
             gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """Adam on the soft‑max approximation of max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc = 250

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow plateaus
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a = _project_to_simplex(a + jitter, target_sum)

        cur = _max_convolution(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _adam_restarts(start_vec: np.ndarray,
                   target_sum: float,
                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several short Adam runs from jittered restarts."""
    best = start_vec.copy()
    best_val = _max_convolution(best)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.015, size=best.shape)
        s = best + jitter
        s = np.maximum(s, 0.0)
        s = _project_to_simplex(s, target_sum)

        a, val = _adam_gd(s, time.time() + run_budget, target_sum,
                           max_beta=2e5, step_factor=0.85, gamma=0.995)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] Adam restart improved best={best_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (true max)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         max_iter: int = 5000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_val = _max_convolution(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.flatnonzero(conv >= max_val - 1e-12)

        # second‑largest value
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second = conv2.max()
        delta_target = max_val - second
        if delta_target < 1e-14:
            break

        k = idx_max[rng.integers(len(idx_max))]
        n = a.shape[0]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k - idx]

        norm2 = np.dot(g, g)
        if norm2 == 0.0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = _project_to_simplex(a + jitter, target_sum)
            it += 1
            continue

        # Polyak‑type step
        step = 0.9 * delta_target / norm2
        step = min(step, target_sum * 0.4 / (np.linalg.norm(g) + 1e-12))

        # backtracking
        for _ in range(6):
            cand = _project_to_simplex(a - step * g, target_sum)
            cand_val = _max_convolution(cand)
            if cand_val < best_val - 1e-12:
                a = cand
                best_a = cand.copy()
                best_val = cand_val
                print(f"[{time.time():.2f}] Subgrad it={it:5d} step={step:.3e} score={best_val:.6f}")
                break
            step *= 0.5
        else:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = _project_to_simplex(a + jitter, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass transfer (greedy reduction of the peak)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 tol: float = 1e-12) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _max_convolution(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second‑largest value
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second = conv2.max()
        if max_val - second < tol:
            break

        # pick first maximal index (any works)
        k = idx_max[0]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]                     # a_{k-i}
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        # donor: maximise a_i * partner_i
        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]
        partner_donor = partner[donor_local]

        # receiver: minimise partner (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second) / denom
        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur = _max_convolution(a)
        if cur < best_val - 1e-12:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] DMT it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val

# Random mass‑transfer (simulated‑annealing style)
# ----------------------------------------------------------------------


def _random_mass_transfer(a0: np.ndarray,
                         deadline: float,
                         target_sum: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_val = _max_convolution(a)

    # acceptance temperature schedule (fast cooling)
    T0 = target_sum * 0.05
    half_time = (deadline - time.time()) / 2.0

    while time.time() < deadline:
        # pick donor (large a_i) and receiver (small a_i)
        k_top = max(1, int(0.02 * n))
        donor = np.argsort(-a)[rng.integers(k_top)]

        k_low = max(1, int(0.02 * n))
        receiver = np.argsort(a)[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.2
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur = _max_convolution(a)
        # accept if improvement, otherwise accept with small probability
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] Random move new best={best_val:.6f}")
        else:
            # simulated‑annealing acceptance
            elapsed = time.time() - (deadline - (deadline - start_time))
            T = T0 * max(0.0, 1.0 - elapsed / half_time)
            if T > 0 and rng.random() < np.exp(-(cur - best_val) / T):
                best_val = cur
                best_a = a.copy()
                print(f"[{time.time():.2f}] Random move accepted (T={T:.4f})")

    return best_a, best_val

        n = 20000
    else:
        n = 12000
    target_sum = np.sqrt(2.0 * n)

    # -------------------- Phase 0: parametric sweep --------------------
    best_a = None
    best_val = np.inf
    rng = np.random.default_rng()

    # (a) decreasing power‑law
    for p in np.linspace(0.2, 1.2, 21):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _max_convolution(seq)
        if val < best_val:
            best_val, best_a = val, seq.copy()

    # (b) increasing edge‑heavy power‑law (symmetric)
    for p in np.linspace(0.2, 1.2, 21):
        # shape a_i ∝ (i+1)^p + (n-i)^p  (symmetric)
        i = np.arange(n, dtype=np.float64)
        a = ((i + 1) ** p) + ((n - i) ** p)
        a *= target_sum / a.sum()
        val = _max_convolution(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    # (c) inverse‑product family (very edge‑heavy)
    for p in np.linspace(0.4, 1.8, 15):
        seq = _inverse_product_sequence(n, float(p), target_sum)
        val = _max_convolution(seq)
        if val < best_val:
            best_val, best_a = val, seq.copy()

    # (d) beta‑distribution shape (α,β)
    alphas = np.linspace(0.6, 3.0, 7)
    betas = np.linspace(0.6, 3.0, 7)
    for α in alphas:
        for β in betas:
            seq = _beta_distribution_sequence(n, float(α), float(β), target_sum)
            val = _max_convolution(seq)
            if val < best_val:
                best_val, best_a = val, seq.copy()

    # (e) power‑law with exponential tail
    for p in np.linspace(0.2, 1.0, 11):
        for q in np.linspace(0.0, 5.0, 9):
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _max_convolution(seq)
            if val < best_val:
                best_val, best_a = val, seq.copy()

    # (f) simple sine shape (always positive)
    seq = _sin_sequence(n, target_sum)
    val = _max_convolution(seq)
    if val < best_val:
        best_val, best_a = val, seq.copy()

    # (g) previous best if available (resized)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _max_convolution(resized)
        if val < best_val:
            best_val, best_a = val, resized.copy()

    # (h) small jittered population around current best
    for _ in range(6):
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        cand = best_a + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _max_convolution(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()

    print(f"[{time.time():.2f}] After parametric sweep best={best_val:.6f}")

    # -------------------- Phase A: Adam GD with restarts --------------------
    gd_deadline = time.time() + (deadline - time.time()) * 0.30
    best_a, best_val = _adam_restarts(best_a, target_sum, gd_deadline)

    # -------------------- Phase B: Exact sub‑gradient descent --------------------
    sub_deadline = time.time() + (deadline - time.time()) * 0.25
    a_sg, val_sg = _subgradient_descent(best_a, target_sum, sub_deadline)
    if val_sg < best_val:
        best_val, best_a = val_sg, a_sg.copy()
        print(f"[{time.time():.2f}] Sub‑gradient improved best={best_val:.6f}")

    # -------------------- Phase C: Deterministic mass‑transfer --------------------
    dmt_deadline = time.time() + (deadline - time.time()) * 0.30
    a_dmt, val_dmt = _deterministic_mass_transfer(best_a, dmt_deadline, target_sum)
    if val_dmt < best_val:
        best_val, best_a = val_dmt, a_dmt.copy()
        print(f"[{time.time():.2f}] Deterministic DMT improved best={best_val:.6f}")

    # -------------------- Phase D: Random annealing style --------------------
    rand_deadline = time.time() + (deadline - time.time()) * 0.15
    a_rand, val_rand = _random_mass_transfer(best_a, rand_deadline, target_sum)
    if val_rand < best_val:
        best_val, best_a = val_rand, a_rand.copy()
        print(f"[{time.time():.2f}] Random transfer improved best={best_val:.6f}")

    # -------------------- Phase E: Final high‑β Adam polish --------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gd(best_a,
                                   deadline,
                                   target_sum,
                                   max_beta=1e6,
                                   step_factor=0.6,
                                   gamma=0.995)
        if val_fin < best_val:
            best_val, best_a = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best={best_val:.6f}")

    # -------------------- Post‑processing --------------------
    arr = np.array(best_a, dtype=float)
    # trim trailing near‑zero entries
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional log of the official evaluator
    try:
        eval_val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Evaluated final score = {eval_val:.6f}")
    except Exception:
        pass

    return final_seq
