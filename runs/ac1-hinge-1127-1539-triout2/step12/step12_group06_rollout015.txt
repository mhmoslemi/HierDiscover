def _lp_optimize_f(f, M):
    n = len(f)
    # Build sparse convolution matrix

    except Exception:
        # fallback
        return None
    if g.value is None:
        return None
    return np.array(g.value).flatten()

"""
Hybrid optimiser for the non‑negative step‑function problem.

The new version adds a *convex‑program refinement* (linear programming) layer
that works on a modest length (≈2000‑2500).  For a fixed sum
S=√(2·n) we iteratively:

1.  Compute the current peak M = max (a⋆a).
2.  Solve   max Σ g_j   s.t.  g≥0  and  (a⋆g)_k ≤ M  for all k.
    This is a linear program because (a⋆g)_k is linear in g.
3.  Scale g to the required sum S and take a convex combination
    a←(1‑t)·a + t·g (t≈0.5).  The peak never increases and typically shrinks.
4.  After a few LP‑iterations the vector is up‑scaled (linear interpolation)
    to a much larger size (8000‑15000) and polished with the fast
    sub‑gradient / soft‑max gradient descent, a targeted permutation annealing
    and a deterministic mass‑transfer pass.

The LP stage supplies a much better “shape” for the step function,
while the subsequent gradient phases exploit the extra degrees of freedom
of the larger vector.  The whole pipeline respects the 1000 s budget and
stores the best sequence in the global variable `height_sequence_1`.
"""

import time
import numpy as np

# Optional heavy imports – they may be unavailable in some environments.
try:
    import cvxpy as cp                 # Linear programming
    import scipy.sparse as sp          # Sparse convolution matrix
except Exception:  # pragma: no cover
    cp = None
    sp = None

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT (real-to-complex)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x≥0, Σx=s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


def _score(a: np.ndarray) -> float:
    """Objective = max convolution entry (sum is √(2·n))."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation from old length to new_len."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → left, 2nd largest → right, 3rd → next left, …."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


# ----------------------------------------------------------------------
# Sub‑gradient / gradient descents (same as the previous version)
# ----------------------------------------------------------------------


def _max_conv_and_subgrad(a: np.ndarray):
    """Return (conv, max_val, kmax, subgrad) for the peak entry."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    kmax = int(np.argmax(conv))
    max_val = conv[kmax]

    grad = np.zeros_like(a)
    lo = max(0, kmax - (n - 1))
    hi = min(kmax, n - 1)
    idx = np.arange(lo, hi + 1)
    partner = a[kmax - idx]
    grad[lo:hi + 1] = 2.0 * partner
    return conv, max_val, kmax, grad


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Momentum projected sub‑gradient descent on max(conv(a,a))."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0
    momentum = np.zeros_like(a)

    while time.time() < deadline:
        _, _, _, grad = _max_conv_and_subgrad(a)

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        lr = step_factor * target_sum / (norm + 1e-12)
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # jitter occasionally
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] SubGD it={it:5d}  score={best_val:.6f}")

        it += 1

    return best, best_val


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of smooth soft‑max approximation of max(conv)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    grad = 2.0 * np.fft.irfft(np.fft.rfft(a_rev, n=size) *
                               np.fft.rfft(p, n=size), n=size)[: 2 * n - 1]
    grad = grad[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 1e6,
                               step_factor: float = 0.5,
                               gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """Momentum projected gradient descent on the soft‑max approximation."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy donor→receiver moves with exponential temperature schedule."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]          # contribution weight to the peak
        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.1,
                                 end_temp: float = 1e-4,
                                 top_frac: float = 0.02,
                                 low_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """Swap indices that heavily contribute to the peak; Metropolis acceptance."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)

        if donor == receiver:
            continue

        a[donor], a[receiver] = a[receiver], a[donor]

        new_score = _score(a)

        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] PermSearch new best={best_score:.6f}")

        if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / max(temp, 1e-12)):
            # keep swap
            pass
        else:
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑program (LP) refinement utilities
# ----------------------------------------------------------------------


def _build_convolution_matrix(f: np.ndarray):
    """
    Build a sparse matrix C of shape (2n‑1, n) such that C @ g = f⋆g.
    Row k contains the coefficients f_{k‑j} for the product with g_j.
    """
    n = f.shape[0]
    # each offset d (0 … n‑1) contributes a full diagonal of length n with value f[d]
    diags = [np.full(n, f[d]) for d in range(n)]
    offsets = list(range(n))               # row – col = d
    C = sp.diags(diags, offsets, shape=(2 * n - 1, n), format="csr")
    return C


def _lp_optimize_f(f: np.ndarray, M: float):
    """Solve  max Σ g_j   s.t. g≥0 and (f⋆g)_k ≤ M (for all k)."""
    if cp is None or sp is None:
        return None
    n = f.shape[0]
    C = _build_convolution_matrix(f)
    g = cp.Variable(n, nonneg=True)
    constraints = [C @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000,
                   abstol=1e-7, reltol=1e-6, feastol=1e-6, verbose=False)
    except Exception:                     # pragma: no cover
        return None
    if g.value is None:
        return None
    return np.array(g.value).flatten()


def _lp_iterative_refine(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         max_iter: int = 10) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve the LP, take a convex combination with the current vector,
    and keep the best (lowest peak).  Stops when time runs out.
    """
    a = a0.copy()
    best_a = a.copy()
    best_M = np.max(_conv_fft(a))

    for it in range(max_iter):
        if time.time() > deadline:
            break

        M = np.max(_conv_fft(a))
        g_raw = _lp_optimize_f(a, M)
        if g_raw is None:
            break
        sum_g = g_raw.sum()
        if sum_g < 1e-12:
            break

        # scale to the required sum
        g = g_raw * (target_sum / sum_g)

        # convex blend (t≈0.5).  If it does not improve, try a smaller step.
        for t in (0.5, 0.25, 0.125):
            a_new = (1 - t) * a + t * g
            M_new = np.max(_conv_fft(a_new))
            if M_new < M:
                a = a_new
                if M_new < best_M:
                    best_M = M_new
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] LP iter {it} t={t:.3f} M={best_M:.6f}")
                break
        else:
            # no improvement possible – exit early
            break

    return best_a, best_M

# Main entry point required by the competition harness

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with a low benchmark score.
    The function respects the 1000 s budget and stores the best vector in
    the global variable `height_sequence_1` for possible reuse.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    # ------------------------------------------------------------------
    # 1. Choose a modest size for the LP stage (≈2000‑2500)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n_small = 2500
    elif budget_s >= 500:
        n_small = 2000
    else:
        n_small = 1500
    target_small = np.sqrt(2.0 * n_small)

    # ------------------------------------------------------------------
    # 2. Build an initial seed of length n_small
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        seed = np.array(globals()["height_sequence_1"], dtype=float)
        seed = _resize_sequence(seed, n_small)
    else:
        seed = _power_law_sequence(n_small, 0.55, target_small)

    seed = _project_to_simplex(seed, target_small)

    # ------------------------------------------------------------------
    # 3. LP refinement (if we have cvxpy and enough time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    lp_time_budget = min(200.0, max(30.0, remaining - 400.0))   # leave enough for later phases
    if cp is not None and sp is not None and lp_time_budget > 30.0:
        lp_deadline = time.time() + lp_time_budget
        best_small, _ = _lp_iterative_refine(seed, target_small, lp_deadline, max_iter=8)
        print(f"[{time.time():.2f}] LP stage finished, score={_score(best_small):.6f}")
    else:
        best_small = seed

    # ------------------------------------------------------------------
    # 4. Upscale to a larger length (more degrees of freedom)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    # decide final length based on remaining time
    if remaining > 350.0:
        n_big = 15000
    elif remaining > 250.0:
        n_big = 12000
    elif remaining > 120.0:
        n_big = 10000
    else:
        n_big = 8000

    target_big = np.sqrt(2.0 * n_big)
    a = _resize_sequence(best_small, n_big)
    a = _project_to_simplex(a, target_big)

    # try a few deterministic orderings
    candidates = [a, _alternating_order(a), _reversed_alternating(a)]
    best_a = a
    best_val = _score(a)
    for cand in candidates[1:]:
        sc = _score(cand)
        if sc < best_val:
            best_val = sc
            best_a = cand
    print(f"[{time.time():.2f}] Upscaled to n={n_big}, start score={best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Long sub‑gradient descent on the enlarged vector
    # ------------------------------------------------------------------
    if time.time() < deadline - 120.0:
        sub_dead = deadline - 60.0
        a_opt, val_opt = _subgradient_descent(best_a, sub_dead,
                                              target_big,
                                              step_factor=0.5,
                                              gamma=0.99)
        if val_opt < best_val:
            best_val = val_opt
            best_a = a_opt.copy()
            print(f"[{time.time():.2f}] Sub‑GD finished, best={best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Soft‑max gradient descent (high β) for fine polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 60.0:
        gd_dead = deadline - 15.0
        a_opt, val_opt = _annealed_gradient_descent(best_a, gd_dead,
                                                    target_big,
                                                    max_beta=2e6,
                                                    step_factor=0.5,
                                                    gamma=0.995)
        if val_opt < best_val:
            best_val = val_opt
            best_a = a_opt.copy()
            print(f"[{time.time():.2f}] Soft‑max GD best={best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Targeted permutation search (only re‑ordering)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        perm_dead = deadline - 10.0
        a_opt, val_opt = _permutation_targeted_search(best_a, perm_dead,
                                                      start_temp=0.15,
                                                      end_temp=1e-5,
                                                      top_frac=0.04,
                                                      low_frac=0.04)
        if val_opt < best_val:
            best_val = val_opt
            best_a = a_opt.copy()
            print(f"[{time.time():.2f}] Permutation search best={best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Tiny deterministic mass‑transfer pass (if any time left)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        mt_dead = deadline - 2.0
        a_opt, val_opt = _deterministic_mass_transfer(best_a, mt_dead,
                                                      target_big,
                                                      start_temp=0.3,
                                                      end_temp=1e-5,
                                                      low_frac=0.05)
        if val_opt < best_val:
            best_val = val_opt
            best_a = a_opt.copy()
            print(f"[{time.time():.2f}] Mass‑transfer final best={best_val:.6f}")

    # ------------------------------------------------------------------
    # 9. Trim trailing zeros for compactness
    # ------------------------------------------------------------------
    arr = np.array(best_a, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # 10. Optional final evaluation logging
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)      # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    globals()["height_sequence_1"] = final_seq
    return final_seq
