    deadline = start + budget_s - 0.2

    # For initial large n, maybe 8000 if budget >= 800 else 4000.
    n = 8000 if budget_s>=800 else 4000
    target_sum = np.sqrt(2.0 * n)

    # Build initial candidates: power law, exponential, random, previous sequence if available.
    # Select best among a few after a quick gradient descent for short time to get good base.

    # Run annealed gradient descent for about half of time.
    gd_deadline = start + (budget_s * 0.45)
    a_best, best_score = _annealed_gradient_descent(...)

    perm_deadline = deadline - 5
    if time.time() < perm_deadline:
        a_best, best_score = _perm_anneal(a_best, perm_deadline)

    final_gd_deadline = deadline - 1

def _permutation_annealing(seq, deadline, rng):
    # seq: numpy array, sum=target_sum
    n = seq.shape[0]
    # optional: sort values decreasing to start with
    # but we may start with given seq
    cur_seq = seq.copy()
    cur_max = _max_convolution_fft(cur_seq)
    best_seq = cur_seq.copy()
    best_max = cur_max

    start_temp = 0.1
    end_temp = 1e-6
    total_time = deadline - time.time()
    it = 0
    while time.time() < deadline:
        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose swap
        new_seq = cur_seq.copy()
        new_seq[i], new_seq[j] = new_seq[j], new_seq[i]
        new_max = _max_convolution_fft(new_seq)

        delta = new_max - cur_max
        if delta < 0:
            accept = True
        else:
            # temperature schedule
            elapsed = time.time() - (deadline - total_time)
            tau = elapsed / total_time
            temp = start_temp * (end_temp / start_temp) ** tau
            accept = rng.random() < np.exp(-delta / (temp + 1e-12))

        if accept:
            cur_seq = new_seq
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_seq = cur_seq.copy()
                print(f"[{time.time():.2f}] PermAnneal iter={it} new best={best_max:.6f}")
        it += 1
    return best_seq, best_max

"""
Hybrid optimizer for the step‑function max‑convolution problem.

The pipeline:
1. Small‑scale shape search (power‑law / exponential seeds → softmax GD,
   a few LP alternations) to obtain a good ℓ¹‑profile.
2. Upscale the profile to a larger length (linear interpolation) and
   re‑project onto the ℓ¹‑simplex.
3. Local permutation annealing: repeatedly swap two entries and accept
   moves that lower (or, with a temperature schedule, occasionally raise)
   the peak of the convolution.
4. Optional final fine‑tuning with soft‑max GD and a deterministic
   mass‑transfer pass.
All steps respect non‑negativity and the ℓ¹‑norm Σa = √(2n), so the
objective returned by `evaluate_sequence` equals the maximal convolution
value.  The code aims to push this value below the previous best ~1.5068.
"""

import time
import math

import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective for a sequence with sum = √(2n) (simply max‑convolution)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple families of seed sequences
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised to have ℓ¹‑norm = target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform random point on the ℓ¹‑simplex."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution (used by annealed GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max_{k} (a*a)[k].
    Returns gradient vector of length n and the raw convolution array.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Annealed projected gradient descent on the soft‑max surrogate
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 1e5,
                               step_factor: float = 0.7,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the smoothed max‑convolution.
    Parameters:
        a0          – initial vector (will be projected to ℓ¹‑simplex)
        deadline    – absolute time after which the routine must stop
        target_sum  – required ℓ¹‑norm (= √(2·n))
        max_beta    – maximum soft‑max temperature (larger → tighter max)
        step_factor – controls base step size (proportional to target_sum)
        gamma       – momentum decay factor
    Returns a tuple (best_vector, best_score).
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        # slowly raise soft‑max temperature
        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement (small dimensionality)
# ----------------------------------------------------------------------


def _lp_optimize(f: np.ndarray, M: float, target_sum: float,
                 rng: np.random.Generator) -> np.ndarray | None:
    """
    Solve a LP that maximises a random linear functional of b
    subject to f * b ≤ M and b ≥ 0.
    Returns the scaled solution (ℓ¹‑norm = target_sum) or None on failure.
    """
    n = f.shape[0]
    rows = []
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=np.int64)
        row = np.zeros(n, dtype=np.float64)
        row[i] = f[k - i]
        rows.append(row)
    A = np.vstack(rows)                 # (2n‑1) × n
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Random objective – maximise c·b  ⇔ minimise –c·b
    c = -rng.random(n)

    bounds = [(0.0, None)] * n

    try:
        from scipy.optimize import linprog
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception as e:
        print(f"[{time.time():.2f}] LP exception: {e}")
        return None

    if not res.success:
        print(f"[{time.time():.2f}] LP failed: {res.message}")
        return None

    b = res.x
    total = b.sum()
    if total <= 0:
        return None
    b *= target_sum / total
    return b


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (Metropolis style)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy mass transfers with a decaying Metropolis acceptance temperature.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            it += 1
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Permutation annealing (swap‑based local search on the ordering)
# ----------------------------------------------------------------------


def _permutation_annealing(seq: np.ndarray,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style local search that swaps two entries of `seq`.
    The ℓ¹‑norm is preserved; only the ordering changes.
    Returns the best sequence seen and its (max‑convolution) score.
    """
    n = seq.shape[0]
    cur_seq = seq.copy()
    cur_max = _max_convolution_fft(cur_seq)
    best_seq = cur_seq.copy()
    best_max = cur_max

    start_temp = 0.07
    end_temp = 1e-6
    total_time = deadline - time.time()
    it = 0

    while time.time() < deadline:
        if total_time <= 0:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        new_seq = cur_seq.copy()
        new_seq[i], new_seq[j] = new_seq[j], new_seq[i]
        new_max = _max_convolution_fft(new_seq)

        delta = new_max - cur_max
        if delta < 0:
            accept = True
        else:
            elapsed = time.time() - (deadline - total_time)
            tau = elapsed / total_time
            temp = start_temp * (end_temp / start_temp) ** tau
            accept = rng.random() < np.exp(-delta / (temp + 1e-12))

        if accept:
            cur_seq = new_seq
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_seq = cur_seq.copy()
                print(f"[{time.time():.2f}] PermAnneal it={it:5d} new best={best_max:.6f}")

        it += 1

    return best_seq, best_max


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with a low evaluation score.
    The algorithm combines a small‑scale shape optimisation (GD + LP),
    up‑sampling, permutation annealing and a final polishing stage.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.3          # safety margin

    # ------------------------------------------------------------------
    # Problem sizes
    # ------------------------------------------------------------------
    # Small problem for cheap LP refinements
    n_small = 1200
    target_small = np.sqrt(2.0 * n_small)

    # Large problem used for the final answer
    if budget_s >= 900:
        n_big = 16000
    elif budget_s >= 600:
        n_big = 12000
    else:
        n_big = 8000
    target_big = np.sqrt(2.0 * n_big)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Build a modest pool of seed vectors for the small problem
    # ------------------------------------------------------------------
    seed_pool = []

    for p in np.linspace(0.3, 1.0, 8):
        seed_pool.append(_power_law_sequence(n_small, float(p), target_small))
    for lam in np.linspace(0.001, 0.015, 5):
        seed_pool.append(_exponential_sequence(n_small, float(lam), target_small))
    for _ in range(5):
        seed_pool.append(_random_simplex(n_small, target_small, rng))

    # Bring in the previous best sequence if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        prev_small = _resize_sequence(prev, n_small)
        prev_small = _project_to_simplex(prev_small, target_small)
        seed_pool.append(prev_small)

    # ------------------------------------------------------------------
    # Quick exploration on the small problem: soft‑max GD from each seed
    # ------------------------------------------------------------------
    best_small = None
    best_small_score = np.inf

    for seed in seed_pool:
        if time.time() >= deadline - 30.0:
            break
        run_dead = min(time.time() + 5.0, deadline - 25.0)
        if run_dead <= time.time():
            break
        a_opt, val_opt = _annealed_gradient_descent(seed, run_dead,
                                                    target_small,
                                                    max_beta=5e5,
                                                    step_factor=0.8,
                                                    gamma=0.99)
        if val_opt < best_small_score:
            best_small_score = val_opt
            best_small = a_opt.copy()
            print(f"[{time.time():.2f}] Small‑GD new best={best_small_score:.6f}")

    # fallback if nothing succeeded
    if best_small is None:
        best_small = _random_simplex(n_small, target_small, rng)

    # ------------------------------------------------------------------
    # Alternate LP refinement and short GD passes on the small problem
    # ------------------------------------------------------------------
    while time.time() < deadline - 30.0:
        M = _max_convolution_fft(best_small)
        b_opt = _lp_optimize(best_small, M, target_small, rng)
        if b_opt is None:
            break
        # mix a little of the LP solution to keep stability
        t_mix = 0.25
        best_small = (1 - t_mix) * best_small + t_mix * b_opt
        best_small = _project_to_simplex(best_small, target_small)

        # a short GD polishing
        gd_dead = min(time.time() + 8.0, deadline - 25.0)
        if gd_dead <= time.time():
            break
        best_small, val_opt = _annealed_gradient_descent(best_small, gd_dead,
                                                         target_small,
                                                         max_beta=5e5,
                                                         step_factor=0.6,
                                                         gamma=0.995)
        if val_opt < best_small_score:
            best_small_score = val_opt
            print(f"[{time.time():.2f}] Small‑LP+GD improved to {best_small_score:.6f}")

    # ------------------------------------------------------------------
    # Upscale to the large dimension and re‑project
    # ------------------------------------------------------------------
    seq_big = _resize_sequence(best_small, n_big)
    seq_big = _project_to_simplex(seq_big, target_big)

    # ------------------------------------------------------------------
    # Optional brief deterministic mass‑transfer before permutation phase
    # ------------------------------------------------------------------
    if time.time() < deadline - 12.0:
        mt_dead = min(time.time() + 6.0, deadline - 12.0)
        seq_big, mt_score = _deterministic_mass_transfer(seq_big, mt_dead,
                                                         target_big,
                                                         start_temp=0.4,
                                                         end_temp=1e-5,
                                                         low_frac=0.06)
        print(f"[{time.time():.2f}] Pre‑perm DMT score={mt_score:.6f}")

    # ------------------------------------------------------------------
    # Permutation annealing (swap‑based local search on ordering)
    # ------------------------------------------------------------------
    if time.time() < deadline - 6.0:
        perm_dead = deadline - 5.0
        seq_big, perm_score = _permutation_annealing(seq_big, perm_dead, rng)
        print(f"[{time.time():.2f}] Permutation phase best={perm_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing: short soft‑max GD followed by a quick DMT pass
    # ------------------------------------------------------------------
    if time.time() < deadline - 4.0:
        fin_dead = min(time.time() + 5.0, deadline - 4.0)
        seq_big, fin_score = _annealed_gradient_descent(seq_big, fin_dead,
                                                        target_big,
                                                        max_beta=1e7,
                                                        step_factor=0.25,
                                                        gamma=0.999)
        print(f"[{time.time():.2f}] Final GD score={fin_score:.6f}")

    if time.time() < deadline - 2.0:
        mt2_dead = min(time.time() + 4.0, deadline - 2.0)
        seq_big, mt2_score = _deterministic_mass_transfer(seq_big, mt2_dead,
                                                          target_big,
                                                          start_temp=0.2,
                                                          end_temp=1e-6,
                                                          low_frac=0.05)
        print(f"[{time.time():.2f}] Final DMT best={mt2_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # ------------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    globals()["height_sequence_1"] = final_seq

    # ------------------------------------------------------------------
    # Optional final evaluation (only for logging; ignored by harness)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
