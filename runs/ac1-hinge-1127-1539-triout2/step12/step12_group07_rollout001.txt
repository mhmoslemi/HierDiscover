    target0 = math.sqrt(2.0*n0)

    best_seq=None; best_max=np.inf

    # families:
    for p in np.linspace(0.2,3.0,25):
        seq=_power_law_sequence(n0,p,target0)
        m=_max_convolution_fft(seq)
        if m<best_max:

    for lam in np.linspace(0.0002,0.05,25):
        seq=_exponential_sequence(n0,lam,target0)
        m=_max_convolution_fft(seq)
        if m<best_max: best_seq=seq; best_max=m

    # logistic: a_i = 1/(1+exp((i-mid)/scale)). Parameter scale & shift.
    mid = n0/2
    for scale in np.linspace(0.5, n0*0.5, 10):
        x=np.arange(n0)
        seq=1.0/(1+np.exp((x-mid)/scale))
        seq=seq/seq.sum()*target0
        m=_max_convolution_fft(seq)
        if m<best_max: best_seq=seq; best_max=m

    # random simplex points:
    rng = np.random.default_rng()
    for _ in range(20):
        seq = rng.random(n0)
        seq = seq/seq.sum()*target0
        m=_max_convolution_fft(seq)
        if m<best_max: best_seq=seq; best_max=m

    # cross test with previous global sequence if exists
    if "height_sequence_1" in globals():

        seq = _resize_sequence(prev,n0)

    # iteratively improve via small jitter and evaluate

Hybrid optimiser – shape‑parameter sweep + multi‑scale local search.

The optimiser works in several stages:

1. **Coarse shape search** (short sequences, ≈4‑5 k).  
   Families: power‑law, exponential, logistic (sigmoid) and a few random
   simplex points.  Each candidate is projected onto the ℓ¹‑simplex of the
   required norm and its maximal self‑convolution is measured via FFT.
   The best shape is kept.

2. **Upscaling** – the best short shape is linearly interpolated to a
   larger length (15‑40 k depending on remaining time) and re‑projected.

3. **Deterministic re‑ordering** – three simple orderings
   (original, alternating, even‐spread) are tried and the one with the
   smallest convolution peak is retained.

4. **Random permutation search** – many random shuffles of the values
   are evaluated quickly; the best ordering is kept.

5. **Projected sub‑gradient descent** – exact sub‑gradient steps on the
   true max‑convolution are performed with momentum.

6. **Greedy mass‑transfer simulated annealing** – mass is moved from
   indices that contribute heavily to the current peak toward indices that
   contribute little, with a Metropolis acceptance rule.

All steps respect the non‑negativity and ℓ¹‑simplex constraints.
The final sequence is trimmed of trailing near‑zero entries and stored
in the global `height_sequence_1` for reuse.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (full linear convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex Σx_i = s, x_i ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _logistic_sequence(n: int, scale: float, target_sum: float) -> np.ndarray:
    """Sigmoid shape: 1/(1+exp((i-mid)/scale))."""
    x = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp((x - mid) / scale))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform random point on ℓ¹‑simplex (Dirichlet‑like)."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, …."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread large values uniformly using a step coprime to n."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent with momentum."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)  # objective = max conv after normalisation

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        # indices attaining the maximum (allow small tolerance)
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]

        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            # derivative of a_i a_{k-i} w.r.t a_i is a_{k-i}
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0004, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy mass‑transfer simulated annealing
# ----------------------------------------------------------------------


def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.5,
                                      end_temp: float = 1e-5,
                                      low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Move mass from indices heavily involved in the current peak to
    low‑contributing indices.  Acceptance follows a Metropolis rule.
    """
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    # initial convolution
    conv = _conv_fft(a)

    while time.time() < deadline:
        # temperature schedule (exponential decay)
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # current peak
        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to the peak (product a_i * a_{k-i})
        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        # propose moving a fraction of the donor's mass
        max_delta = min(a[donor] * 0.9, a[receiver] * 0.9)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        # tentative move
        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        # recompute convolution (full recompute – still cheap)
        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        # Metropolis acceptance
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

    return best_a, best_score

# Random permutation search (fast heuristic)

def _permutation_search(base_seq: np.ndarray,
                        trials: int,
                        deadline: float) -> tuple[np.ndarray, float]:
    """Search over random shuffles of `base_seq`."""
    best_seq = base_seq.copy()
    best_val = _max_convolution_fft(best_seq)
    rng = np.random.default_rng()
    n = base_seq.shape[0]

    for _ in range(trials):
        if time.time() >= deadline:
            break
        perm = rng.permutation(base_seq)
        val = _max_convolution_fft(perm)
        if val < best_val:
            best_val = val
            best_seq = perm.copy()
    return best_seq, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function minimising the benchmark.
    The routine follows a multi‑scale pipeline:
      1) coarse shape sweep on a short sequence,
      2) up‑scale,
      3) deterministic reorderings,
      4) random permutation search,
      5) projected sub‑gradient descent,
      6) greedy mass‑transfer simulated annealing.
    The final sequence is stored in the global `height_sequence_1`.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Phase 1: coarse shape sweep (short length)
    # ------------------------------------------------------------------
    n0 = 5000 if budget_s >= 800 else (4000 if budget_s >= 500 else 3000)
    target0 = math.sqrt(2.0 * n0)

    best_seq = None
    best_max = np.inf

    # Power‑law family
    for p in np.linspace(0.2, 3.0, 26):
        seq = _power_law_sequence(n0, float(p), target0)
        m = _max_convolution_fft(seq)
        if m < best_max:
            best_max, best_seq = m, seq

    # Exponential family
    for lam in np.linspace(0.0002, 0.05, 26):
        seq = _exponential_sequence(n0, float(lam), target0)
        m = _max_convolution_fft(seq)
        if m < best_max:
            best_max, best_seq = m, seq

    # Logistic (sigmoid) family
    for scale in np.linspace(0.5, max(5.0, n0 * 0.5), 20):
        seq = _logistic_sequence(n0, float(scale), target0)
        m = _max_convolution_fft(seq)
        if m < best_max:
            best_max, best_seq = m, seq

    # Random simplex points
    for _ in range(30):
        seq = _random_simplex(n0, target0, rng)
        m = _max_convolution_fft(seq)
        if m < best_max:
            best_max, best_seq = m, seq

    # Include previous global sequence if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        seq = _resize_sequence(prev, n0)
        seq = _project_to_simplex(seq, target0)
        m = _max_convolution_fft(seq)
        if m < best_max:
            best_max, best_seq = m, seq

    # Light jitter‑based refinement while we have a few seconds spare
    while time.time() < deadline - 30.0:
        jitter = rng.normal(scale=target0 * 0.02, size=n0)
        seq = best_seq + jitter
        seq = np.maximum(seq, 0.0)
        seq = _project_to_simplex(seq, target0)
        m = _max_convolution_fft(seq)
        if m < best_max:
            best_max, best_seq = m, seq

    # ------------------------------------------------------------------
    # Phase 2: upscale to a larger length (depends on remaining time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 300:
        n_final = 40000
    elif remaining > 150:
        n_final = 30000
    elif remaining > 80:
        n_final = 20000
    else:
        n_final = n0

    target_final = math.sqrt(2.0 * n_final)
    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # ------------------------------------------------------------------
    # Phase 3: deterministic re‑orderings
    # ------------------------------------------------------------------
    candidates = [
        seq_big,
        _alternating_order(seq_big),
        _even_spread_order(seq_big)
    ]
    seq_big = min(candidates, key=_max_convolution_fft)
    cur_max = _max_convolution_fft(seq_big)
    print(f"[{time.time():.2f}] Upscaled to {n_final} elems, initial max={cur_max:.6f}")

    # ------------------------------------------------------------------
    # Phase 4: random permutation search
    # ------------------------------------------------------------------
    perm_deadline = time.time() + min(20.0, remaining * 0.12)
    if perm_deadline > deadline - 10.0:
        perm_deadline = deadline - 10.0
    seq_big, perm_max = _permutation_search(seq_big, trials=200, deadline=perm_deadline)
    if perm_max < cur_max:
        cur_max = perm_max
        print(f"[{time.time():.2f}] After perm search max={cur_max:.6f}")

    # ------------------------------------------------------------------
    # Phase 5: sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        sub_deadline = deadline - 30.0
        a_sub, val_sub = _subgradient_descent(seq_big, sub_deadline, target_final,

        if val_sub < cur_max:
            cur_max = val_sub
            seq_big = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑grad refined max={cur_max:.6f}")

    # ------------------------------------------------------------------
    # Phase 6: greedy mass‑transfer simulated annealing
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:

        a_mt, val_mt = _mass_transfer_simulated_annealing(seq_big, mt_deadline,
                                                          target_final,
                                                          start_temp=0.6,
                                                          end_temp=1e-5,
                                                          low_frac=0.07)
        if val_mt < cur_max:
            cur_max = val_mt
            seq_big = a_mt.copy()
            print(f"[{time.time():.2f}] Final mass‑transfer max={cur_max:.6f}")

        pass

    return final_seq
