def propose_candidate(budget_s=1000, **kwargs):
    start = time.time()
    deadline = start + budget_s - 0.5

    # Problem size selection (n)
    n = 12000 if budget_s>800 else 8000
    target_sum = sqrt(2*n)

    # Initialize many seeds
    pool = []
    for p in np.linspace(0.2, 1.0, 9):
        pool.append(_power_law_sequence(n, p, target_sum))
    for lam in np.linspace(0.001, 0.02, 6):
        pool.append(_exponential_sequence(n, lam, target_sum))
    for _ in range(4):
        pool.append(_random_simplex(n, target_sum, np.random.default_rng()))

    # Include previous best from global variable if exists
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        pool.append(_resize_sequence(prev, n))

    # Evaluate each seed with short gradient descent (annealed) for e.g. 8 sec each
    best_seq = None
    best_score = np.inf

    rng = np.random.default_rng()
    i = 0
    while time.time() < deadline - 30 and i < len(pool):
        seed = pool[i]
        # optionally jitter a bit
        seed = _project_to_simplex(seed, target_sum)
        # run gradient
        run_dead = time.time() + min(10.0, deadline - time.time() - 20.0)
        a_opt, val_opt = _annealed_gradient_descent_v2_ext(seed, run_dead, target_sum,
                                                          max_beta=3e5, step_factor=0.6, gamma=0.99)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"New best after GD {best_score}")
        i += 1

    # If we still have time, refine best_seq via ordering local search
    if best_seq is not None and time.time() < deadline - 10:
        # get sorted descending values
        vals = np.sort(best_seq)[::-1]
        # initial order: even spread
        cur_seq = _even_spread_order(vals)
        cur_score = _max_convolution_fft(cur_seq)
        best_order_seq = cur_seq.copy()
        best_order_score = cur_score

        # simulated annealing swaps
        T0 = cur_score * 0.1
        T_end = cur_score * 0.001
        total_iters = 20000
        it = 0
        while time.time() < deadline - 5 and it < total_iters:
            # linear cooling
            T = T0 + (T_end - T0) * (it / total_iters)
            # propose swap
            i1, i2 = rng.integers(0, n, size=2)
            if i1 == i2:
                continue
            # swap
            cur_seq[i1], cur_seq[i2] = cur_seq[i2], cur_seq[i1]
            new_score = _max_convolution_fft(cur_seq)
            delta = new_score - cur_score
            accept = delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12))
            if accept:
                cur_score = new_score
                if new_score < best_order_score:
                    best_order_score = new_score
                    best_order_seq = cur_seq.copy()
                    print(f"Swap iteration {it}, new best order score={best_order_score:.6f}")
            else:
                # revert swap
                cur_seq[i1], cur_seq[i2] = cur_seq[i2], cur_seq[i1]
            it += 1

        # after annealing, if best_order_score improves
        if best_order_score < best_score:
            best_score = best_order_score
            best_seq = best_order_seq.copy()
            print(f"Best order score after anneal {best_score}")

    # final polishing via subgradient descent
    if best_seq is not None and time.time() < deadline - 2:
        sub_dead = deadline - 1.5
        a_sub, val_sub = _subgradient_descent(best_seq, sub_dead, target_sum,
                                              step_factor=0.4, gamma=0.998)
        if val_sub < best_score:
            best_score = val_sub
            best_seq = a_sub.copy()
            print(f"After subgrad final best {best_score}")

    # Trim zero trailing
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[:nz[-1]+1]
    final_seq = arr.tolist()
    globals()['height_sequence_1'] = final_seq
    # optionally evaluate
    try:
        eval_score = evaluate_sequence(final_seq)
        print("Final evaluate:", eval_score)
    except Exception:
        pass

    return final_seq

"""
Hybrid optimizer with an additional permutation‐annealing stage.

Goal
----
Search a non‑negative sequence a of length n (n≈1.2·10⁴) that minimises

    2·n·max(a∗a) / ( Σ a )²                      (1)

(the `evaluate_sequence` routine implements exactly this).

Method
------
1. **Problem size** – pick a moderate n (≈12000 for the full budget,
   ≈8000 if the time budget is smaller).

2. **Diverse seeds** – generate power‑law, exponential and random
   simplex points and normalise them to the ℓ¹–norm `sqrt(2·n)`.

3. **Soft‑max annealed GD** – minimise a smooth approximation of the
   max‑convolution (`β` grows up to 5·10⁵) while staying on the simplex.
   Momentum (Nesterov/Adam‑like) and occasional jitter avoid stagnation.

4. **Deterministic mass‑transfer** – a short Metropolis‑style greedy
   redistribution that reduces the dominant convolution peak.

5. **Permutation annealing** – keep the multiset of values fixed,
   repeatedly swap two positions (simulated‑annealing schedule) and
   accept improvements or uphill moves with probability exp(−Δ/T).
   This spreads large coefficients more uniformly and often lowers
   `max(a∗a)` beyond the simple alternating/even‑spread orders.

6. **Small‑scale LP refinement** – shrink the best candidate to ≈1300
   entries, solve a linear programme (maximise a random linear functional
   under the current convolution bound) and mix the LP solution back
   (`t≈0.25`).  A few iterations tighten the bound.

7. **Upscaling & final polishing** – interpolate back to the original
   length, run a few iterations of projected sub‑gradient descent on the
   exact `max` and finish with another short mass‑transfer phase.

All updates respect non‑negativity and the ℓ¹‑simplex.
The routine returns the best sequence found before the deadline.

The algorithm is deliberately modular: every helper is a top‑level
function (no closures or lambdas) and can be unit‑tested separately.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = rfft(a, n=size)
    conv = irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = rfft(a, n=size)
    conv = irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective (1) –  a is already on the ℓ¹‑simplex."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto {x≥0, Σx = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(‑p) then renormalised."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(‑lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread large values uniformly using a coprime step."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Soft‑max gradient (used by annealed GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(a∗a)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = rfft(a, n=size)
    b = irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = rfft(a_rev, n=size)
    Pr_fft = rfft(p, n=size)
    conv_rev = irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent_ext(a0: np.ndarray,
                                   deadline: float,
                                   target_sum: float,
                                   max_beta: float = 5e5,
                                   step_factor: float = 0.6,
                                   gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max surrogate.
    Returns the best point found before `deadline`.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow plateaus
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.45,
                         gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent using the exact `max`."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]
        grad = np.zeros_like(a)
        for k in max_idxs:
            lo = max(0, k - (a.shape[0] - 1))
            hi = min(k, a.shape[0] - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (Metropolis style)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy mass moves with a decaying temperature."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # index where convolution attains (near) maximum
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            it += 1
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming alternation (small problem)
# ----------------------------------------------------------------------


def _lp_optimize(f: np.ndarray, M: float, target_sum: float,
                 rng: np.random.Generator) -> np.ndarray | None:
    """
    Solve a LP that maximises a random linear functional of b
    subject to `f * b ≤ M` and `b ≥ 0`. Result is rescaled to ℓ¹‑norm = target_sum.
    Returns None on failure.
    """
    n = f.shape[0]
    rows = []
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=np.int64)
        row = np.zeros(n, dtype=np.float64)
        row[i] = f[k - i]
        rows.append(row)
    A = np.vstack(rows)                     # (2n‑1) × n
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Random objective – maximise c·b  ⇔ minimise –c·b
    c = -rng.random(n)

    bounds = [(0.0, None)] * n

    try:
        from scipy.optimize import linprog
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception as e:
        print(f"[{time.time():.2f}] LP exception: {e}")
        return None

    if not res.success:
        print(f"[{time.time():.2f}] LP failed: {res.message}")
        return None

    b = res.x
    total = b.sum()
    if total <= 0:
        return None
    b *= target_sum / total
    return b


# ----------------------------------------------------------------------
# Permutation annealing (the novel ingredient)
# ----------------------------------------------------------------------


def _perm_anneal(vals: np.ndarray,
                 deadline: float,
                 start_temp_factor: float = 0.15,
                 end_temp_factor: float = 0.001,
                 max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simulated annealing on permutations of `vals` (the multiset of coefficients).
    Returns the best ordering found before `deadline`.
    """
    n = vals.shape[0]
    # initial ordering – even spread (already spread out large values)
    cur_seq = _even_spread_order(vals.copy())
    cur_score = _max_convolution_fft(cur_seq)

    best_seq = cur_seq.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    # temperature schedule (linear in iteration count)
    T0 = cur_score * start_temp_factor
    T1 = cur_score * end_temp_factor

    it = 0
    while time.time() < deadline and it < max_iters:
        # linear cooling
        T = T0 + (T1 - T0) * (it / max_iters)

        i1, i2 = rng.integers(0, n, size=2)
        if i1 == i2:
            it += 1
            continue

        # propose swap
        cur_seq[i1], cur_seq[i2] = cur_seq[i2], cur_seq[i1]
        new_score = _max_convolution_fft(cur_seq)
        delta = new_score - cur_score

        accept = delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-12))
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
                print(f"[{time.time():.2f}] SA it={it:6d} best={best_score:.6f}")
        else:
            # revert swap
            cur_seq[i1], cur_seq[i2] = cur_seq[i2], cur_seq[i1]

        it += 1

    return best_seq, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:

    # ------------------------------------------------------------------

        if val_ref < best_score:

    # ------------------------------------------------------------------
    # Permutation annealing (the main novelty)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 40:
        # Keep the multiset of coefficients, only reorder.

                                            start_temp_factor=0.12,
                                            end_temp_factor=0.001,

        if perm_score < best_score:
            best_score = perm_score
            best_seq = seq_perm.copy()
            print(f"[{time.time():.2f}] Perm‑anneal improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Small‑scale LP alternation (optional – only if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30:
        n_lp = min(1500, max(800, n // 8))
        target_lp = math.sqrt(2.0 * n_lp)
        seq_lp = _resize_sequence(best_seq, n_lp)
        seq_lp = _project_to_simplex(seq_lp, target_lp)

        max_lp_iters = 4
        lp_iter = 0
        while lp_iter < max_lp_iters and time.time() < deadline - 20:
            M = _max_convolution_fft(seq_lp)
            b_opt = _lp_optimize(seq_lp, M, target_lp, rng)
            if b_opt is None:
                break
            t = 0.25
            seq_lp = (1 - t) * seq_lp + t * b_opt
            seq_lp = _project_to_simplex(seq_lp, target_lp)
            cur_score = _score(seq_lp)
            print(f"[{time.time():.2f}] LP iter={lp_iter} score={cur_score:.6f}")
            lp_iter += 1

        # Upscale back to the original size
        seq_big = _resize_sequence(seq_lp, n)
        seq_big = _project_to_simplex(seq_big, target_sum)
    else:
        # Direct up‑scaling if LP stage cannot be afforded
        seq_big = _resize_sequence(best_seq, n)
        seq_big = _project_to_simplex(seq_big, target_sum)

    # ------------------------------------------------------------------
    # Simple deterministic reorderings (fallback) and best‑of‑many
    # ------------------------------------------------------------------
    candidates = [
        seq_big,
        _alternating_order(seq_big),
        _even_spread_order(seq_big),
    ]
    seq_big = min(candidates, key=_max_convolution_fft)
    best_score = _score(seq_big)
    print(f"[{time.time():.2f}] After up‑scale length={len(seq_big)} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Sub‑gradient polishing on the full‑size problem
    # ------------------------------------------------------------------
    if time.time() < deadline - 10:
        sub_dead = deadline - 6.0
        a_sub, val_sub = _subgradient_descent(
            seq_big, sub_dead, target_sum,
            step_factor=0.42, gamma=0.996,
        )
        if val_sub < best_score:
            best_score = val_sub
            seq_big = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑grad polished best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final deterministic mass‑transfer pass
    # ------------------------------------------------------------------
    if time.time() < deadline - 2:
        mt_dead = time.time() + min(6.0, deadline - time.time() - 1.0)
        a_mt, val_mt = _deterministic_mass_transfer(
            seq_big, mt_dead, target_sum,
            start_temp=0.25, end_temp=1e-5, low_frac=0.05,
        )
        if val_mt < best_score:
            best_score = val_mt
            seq_big = a_mt.copy()
            print(f"[{time.time():.2f}] Final MT best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # ------------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Optional final evaluation (purely for logging)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for the next round (if any)
    globals()["height_sequence_1"] = final_seq
    return final_seq
