def propose_candidate(budget_s: int = 1000, **kwargs):
    import time, math, numpy as np, random

    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # set n
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 400:
        n = 12000
    else:
        n = 8000
    target_sum = math.sqrt(2.0 * n)

    # param families search function
    def evaluate(seq):
        return _score(seq)   # uses _score defined in module

    best_seq = None
    best_score = float('inf')
    # 1) power law grid
    for p in np.arange(0.15, 3.01, 0.05):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = evaluate(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()
    # 2) exp power law
    for p in np.arange(0.15, 2.51, 0.1):
        for q in np.arange(0.0, 20.001, 0.5):
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = evaluate(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()
    # 3) broken power law
    for p1 in np.arange(0.15, 2.01, 0.2):
        for p2 in np.arange(0.5, 4.01, 0.3):
            for alpha in np.arange(0.05, 0.96, 0.1):
                seq = _broken_power_law_sequence(n, float(p1), float(p2), float(alpha), target_sum)
                val = evaluate(seq)
                if val < best_score:
                    best_score, best_seq = val, seq.copy()
    print(f"[{time.time():.2f}] After param grid: best_score={best_score:.6f}")

    # Phase A: Adam GD with high beta
    # allocate about 15% time
    remaining = deadline - time.time()
    if remaining > 10:
        adam_deadline = time.time() + 0.15 * remaining
        a_adam, val_adam = _adam_gradient_descent(best_seq, adam_deadline, target_sum, max_beta=5e5, step_factor=0.7, gamma=0.99)
        if val_adam < best_score:
            best_score, best_seq = val_adam, a_adam.copy()
            print(f"[{time.time():.2f}] Adam improved: {best_score:.6f}")

    # Phase B: improved guided mass transfer
    # allocate about 30% time
    remaining = deadline - time.time()
    if remaining > 10:
        mt_deadline = time.time() + 0.30 * remaining
        a_mt, val_mt = _guided_mass_transfer_improved(best_seq, mt_deadline, target_sum)
        if val_mt < best_score:
            best_score, best_seq = val_mt, a_mt.copy()

    # Phase C: quick random local moves

    if remaining > 5:

def _guided_mass_transfer_improved(a0, deadline, target_sum, tol=1e-12):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        # get indices near max
        mask_max = conv >= max_val - tol
        max_idxs = np.where(mask_max)[0]
        # if many max, pick random one
        k = rng.choice(max_idxs) if max_idxs.size > 0 else np.argmax(conv)
        # compute gradient for this k
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(a)
        # contribution 2*a_{k-idx} for each idx, except middle may half, but ignore
        g[idx] = 2.0 * a[k - idx]
        # adjust to preserve sum
        sum_g = g.sum()
        d = -g + sum_g / n * np.ones_like(a)
        # step constraints
        neg = d < 0
        if np.any(neg):
            t_nonneg = np.min(a[neg] / -d[neg])
        else:
            t_nonneg = np.inf
        # compute delta for this direction at shift k
        delta = 2.0 * _conv_fft_pair(a, d)
        delta_k = delta[k]
        if delta_k >= 0:
            # direction not reducing peak; try another max idx if any
            # try other max indices up to few attempts
            reduced = False
            for alt_k in max_idxs:
                lo = max(0, alt_k - (n - 1))
                hi = min(alt_k, n - 1)
                idx = np.arange(lo, hi + 1, dtype=np.int64)
                g_alt = np.zeros_like(a)
                g_alt[idx] = 2.0 * a[alt_k - idx]
                sum_g_alt = g_alt.sum()
                d_alt = -g_alt + sum_g_alt / n * np.ones_like(a)
                neg_alt = d_alt < 0
                t_nonneg_alt = np.min(a[neg_alt] / -d_alt[neg_alt]) if np.any(neg_alt) else np.inf
                delta_alt = 2.0 * _conv_fft_pair(a, d_alt)
                delta_alt_k = delta_alt[alt_k]
                if delta_alt_k < 0:
                    # select this direction
                    d = d_alt
                    delta_k = delta_alt_k
                    t_nonneg = t_nonneg_alt
                    reduced = True
                    break
            if not reduced:
                # no reducing direction found; add jitter
                jitter = rng.normal(scale=target_sum*0.001, size=n)
                a = np.maximum(a + jitter, 0.0)
                a = _project_to_simplex(a, target_sum)
                continue
        # compute second max excluding entries within tol of max
        mask_notmax = conv < max_val - tol
        second_max = conv[mask_notmax].max() if np.any(mask_notmax) else 0.0
        # step to bring max down to second max
        if max_val - second_max <= 0:
            t_to_second = np.inf
        else:
            t_to_second = (max_val - second_max) / (-delta_k)
        # safe step
        t = min(t_nonneg, t_to_second) * 0.99
        if t <= 0:
            # jitter to escape
            jitter = rng.normal(scale=target_sum*0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)
            continue
        a = np.maximum(a + t * d, 0.0)
        a = _project_to_simplex(a, target_sum)
        cur_score = _score(a)
        if cur_score < best_score - 1e-12:

"""
Improved step‑function optimizer for the max‑autocorrelation benchmark.

Key ideas
---------
1. **Broader parametric pre‑search** – dense scanning of
   power‑law, exponential‑power‑law and two‑segment broken‑power‑law
   families on a fairly large problem size (n≈15k).  This quickly yields
   a high‑quality starting shape.

2. **Projected Adam on a soft‑max approximation** – a few short Adam
   runs with an exponentially increasing temperature (beta) drive the
   solution towards a true max‑convolution optimum.

3. **Exact single‑peak guided mass‑transfer** – instead of averaging
   the sub‑gradient over all maximal indices we use the gradient of a
   *single* maximal entry (chosen adaptively).  An exact line‑search
   moves mass from “peak‑inflating’’ positions towards “peak‑relieving’’
   positions while preserving the simplex constraint.  Occasional
   jitter helps escape plateaus.

4. **Lightweight random local moves** and a final high‑beta Adam polish.

All phases respect the wall‑clock budget (≈1000 s) and the simplex
constraint Σ a = √(2 n).  The routine returns the best sequence found,
trimming trailing near‑zero values.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) using real FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft_pair(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution x*y (full length 2·n‑1) using real FFT."""
    n = x.size
    size = 1 << ((2 * n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Benchmark score (equal to max convolution when Σa = √(2n)).
    """
    n = a.shape[0]
    max_b = _max_conv(a)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (dense pre‑search)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _broken_power_law_sequence(
    n: int, p1: float, p2: float, alpha: float, target_sum: float
) -> np.ndarray:
    K = max(1, int(alpha * n))
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.empty_like(idx)
    a[:K] = idx[:K] ** (-p1)
    a[K:] = idx[K:] ** (-p2)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max approximation)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(a*a)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = conv_full.max()
    exps = np.exp(beta * (conv_full - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, conv_full


def _adam_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 5e5,
    step_factor: float = 0.8,
    gamma: float = 0.999,
) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max approximation of the max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact *single‑peak* guided mass‑transfer (Phase B – refined)
# ----------------------------------------------------------------------


def _guided_mass_transfer_improved(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the current maximal convolution entry using the
    exact sub‑gradient of a *single* maximal index (chosen adaptively).
    When the chosen direction cannot reduce the peak we try other
    maximal indices or inject a small random jitter.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        mask_max = conv >= max_val - tol
        max_idxs = np.where(mask_max)[0]

        # choose a candidate maximal index (random among them)
        k = rng.choice(max_idxs) if max_idxs.size else np.argmax(conv)

        # gradient for this single index k
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k - idx]

        # direction preserving Σa
        sum_g = g.sum()
        d = -g + sum_g / n * np.ones_like(a)

        # maximal admissible step before any component becomes negative
        neg = d < 0
        if np.any(neg):
            t_nonneg = np.min(a[neg] / -d[neg])
        else:
            t_nonneg = np.inf

        # linearised change of the convolution in direction d
        delta = 2.0 * _conv_fft_pair(a, d)   # length 2n‑1
        delta_k = delta[k]

        # If this direction does NOT reduce the peak, try alternative maxima
        if delta_k >= 0:
            improved = False
            for alt_k in max_idxs:
                lo = max(0, alt_k - (n - 1))
                hi = min(alt_k, n - 1)
                idx = np.arange(lo, hi + 1, dtype=np.int64)
                g_alt = np.zeros_like(a)
                g_alt[idx] = 2.0 * a[alt_k - idx]
                sum_g_alt = g_alt.sum()
                d_alt = -g_alt + sum_g_alt / n * np.ones_like(a)
                neg_alt = d_alt < 0
                t_nonneg_alt = np.min(a[neg_alt] / -d_alt[neg_alt]) if np.any(neg_alt) else np.inf
                delta_alt = 2.0 * _conv_fft_pair(a, d_alt)
                delta_alt_k = delta_alt[alt_k]
                if delta_alt_k < 0:
                    d = d_alt
                    delta_k = delta_alt_k
                    t_nonneg = t_nonneg_alt
                    improved = True
                    break
            if not improved:
                # stall – inject a tiny jitter and continue
                jitter = rng.normal(scale=target_sum * 0.0005, size=n)
                a = np.maximum(a + jitter, 0.0)
                a = _project_to_simplex(a, target_sum)
                continue

        # second largest entry (excluding ties with the current max)
        mask_notmax = conv < max_val - tol
        second_max = conv[mask_notmax].max() if np.any(mask_notmax) else 0.0

        # step needed to drive current max down to the second max
        if max_val - second_max > 0:
            t_to_second = (max_val - second_max) / (-delta_k)
        else:
            t_to_second = np.inf

        # safe step (slightly smaller to stay away from the boundary)
        t = min(t_nonneg, t_to_second) * 0.99
        if t <= 0:
            # again stall – jitter
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)
            continue

        # apply the step
        a = np.maximum(a + t * d, 0.0)
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] GuidedMT it={it} score={best_score:.6f}")

        # occasional jitter to escape tiny plateaus
        if it % 300 == 0 and rng.random() < 0.07:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Light global random exploration (optional)
# ----------------------------------------------------------------------


def _global_random_move(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing‑style mass moves."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    while time.time() < deadline - 0.5:
        # pick a donor among the top 2 % and a receiver among the bottom 2 %
        top_k = max(1, int(0.02 * n))
        low_k = max(1, int(0.02 * n))
        donor = np.argsort(-a)[rng.integers(top_k)]
        receiver = np.argsort(a)[rng.integers(low_k)]
        if donor == receiver:
            continue
        delta = rng.random() * a[donor] * 0.03   # move up to 3 % of donor mass
        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] RandomMove new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimizing the benchmark score.
    The routine respects the supplied wall‑clock budget and returns the best
    sequence found (trailing near‑zero entries are trimmed).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # small safety margin

    # ---------------------------------------------------------------
    # Choose problem size – larger n gives a finer discretisation.
    # ---------------------------------------------------------------
    if budget_s >= 900:
        n = 18000
    elif budget_s >= 600:
        n = 15000
    elif budget_s >= 400:
        n = 12000
    else:
        n = 8000

    target_sum = math.sqrt(2.0 * n)          # Σa is fixed to this value

    # ---------------------------------------------------------------
    # Phase 0 – dense parametric pre‑search (coarse but wide)
    # ---------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    rng = np.random.default_rng()

    # Power‑law family
    for p in np.arange(0.15, 3.01, 0.05):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # Exponential‑power‑law family
    for p in np.arange(0.15, 2.51, 0.1):
        for q in np.arange(0.0, 20.001, 0.5):
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # Two‑segment broken power‑law
    for p1 in np.arange(0.15, 2.01, 0.2):
        for p2 in np.arange(0.5, 4.01, 0.3):
            for alpha in np.arange(0.05, 0.96, 0.1):
                seq = _broken_power_law_sequence(
                    n, float(p1), float(p2), float(alpha), target_sum
                )
                val = _score(seq)
                if val < best_score:
                    best_score, best_seq = val, seq.copy()

    # Warm‑start from a previously discovered construction, if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            # linear interpolation to the new size
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score, best_seq = val, resized.copy()

    print(f"[{time.time():.2f}] After dense pre‑search: best_score={best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase A – short Adam runs (exploratory restarts)
    # ---------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10:
        ada_deadline = time.time() + 0.15 * remaining
        while time.time() < ada_deadline - 2.0:
            jitter = rng.normal(scale=target_sum * 0.015, size=n)
            start_vec = best_seq + jitter
            start_vec = np.maximum(start_vec, 0.0)
            start_vec = _project_to_simplex(start_vec, target_sum)

            a_opt, val_opt = _adam_gradient_descent(
                start_vec,

                gamma=0.99,
            )
            if val_opt < best_score:

                print(f"[{time.time():.2f}] Adam restart improved best = {best_score:.6f}")

    # ---------------------------------------------------------------

    while time.time() < deadline - 8.0:
        a_ref, val_ref = _guided_mass_transfer_improved(
            best_seq, deadline, target_sum

        if val_rand < best_score:

            print(f"[{time.time():.2f}] RandomMove improved best = {best_score:.6f}")

        )

            print(f"[{time.time():.2f}] Final polish improved best = {best_score:.6f}")

    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated benchmark score = {val:.6f}")
    except Exception:
        pass

    return final_seq
