"""
Hybrid search for low‐autocorrelation step functions.

The algorithm combines several complementary phases:
  1.  Exploration – many short annealed gradient descents on a
      medium‑size problem (n≈8000) to discover good candidates.
  2.  Light deterministic mass‑transfer (MT) to improve the peak.
  3.  LP‑alternating refinement on a down‑scaled instance (n≈1500).
  4.  Upscaling to a large length (≈15000‑20000) and a quick re‑ordering.
  5.  Sub‑gradient polishing on the full problem.
  6.  A final alternating loop of short sub‑gradient + MT passes,
      using the remaining time budget.
  7.  Optional random hill‑climbing fine tuning.

All updates keep the ℓ¹‑norm equal to √(2 n), which makes the
objective equal to the maximal entry of the self‑convolution.
The best sequence found at any moment is retained and finally
returned (trailing near‑zeros are trimmed).  The function
`propose_candidate` is the entry point used by the competition harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)².  For Σa = √(2n) this equals maxconv."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection of v onto {x ≥ 0, Σx = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)⁻ᵖ  then renormalised."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(−lam·i)  then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform random point on the ℓ¹‑simplex."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread large values uniformly by a coprime step."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]          # descending
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Soft‑max gradient (used by annealed GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the smooth max (soft‑max) of the self‑convolution."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.7,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max approximation.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent using the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        # indices where the max is attained (with tiny tolerance)
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]

        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (Metropolis style)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy mass moves with a decaying Metropolis acceptance rule.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)
    n = a.shape[0]

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # index of the current convolution peak
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # product contributions for each pair (i, kmax-i)
        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]

        # pick a receiver among low‑contributing pairs
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        if max_delta <= 0:
            continue

        delta = rng.random() * max_delta
        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT new best={best_score:.6f}")
        else:
            # reject: rollback
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional jitter to increase exploration
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming alternating refinement (small problem)
# ----------------------------------------------------------------------


def _lp_optimize(f: np.ndarray,
                 M: float,
                 target_sum: float,
                 rng: np.random.Generator) -> np.ndarray | None:
    """
    Solve a LP that maximises a random linear functional of `b`
    subject to `f * b <= M` and `b >= 0`.  The result is scaled to
    have ℓ¹‑norm = target_sum.
    """
    n = f.shape[0]
    rows = []
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=np.int64)
        row = np.zeros(n, dtype=np.float64)
        row[i] = f[k - i]
        rows.append(row)
    A = np.vstack(rows)                # (2n‑1) × n
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Random objective – maximise c·b  ⇔ minimise –c·b
    c = -rng.random(n)

    bounds = [(0.0, None)] * n

    try:
        from scipy.optimize import linprog
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception as e:
        print(f"[{time.time():.2f}] LP exception: {e}")
        return None

    if not res.success:
        print(f"[{time.time():.2f}] LP failed: {res.message}")
        return None

    b = res.x
    total = b.sum()
    if total <= 0:
        return None
    b *= target_sum / total
    return b


# ----------------------------------------------------------------------
# Random hill‑climbing fine tuner
# ----------------------------------------------------------------------


def _hill_climb(a0: np.ndarray,
                target_sum: float,
                deadline: float,
                rng: np.random.Generator,
                max_frac: float = 0.1) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing: repeatedly move a random fraction of mass
    from a donor to a receiver, keeping Σa fixed.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _score(a)
    n = a.shape[0]

    while time.time() < deadline:
        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(n)
        if i == j:
            continue
        delta = rng.random() * min(max_frac * a[i], a[i])
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        # projection not needed – sum is preserved and non‑negativity holds
        val = _score(a2)
        if val < best_val:
            best_val = val
            best_a = a2.copy()
            a = a2
            # occasional logging
            if int(best_val * 100000) % 1000 == 0:
                print(f"[{time.time():.2f}] HillClimb new best={best_val:.6f}")
        # else reject
    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid optimizer for the step‑function bound.
    Returns a list of non‑negative heights.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin
    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # 1) Exploration on a moderate size problem
    # --------------------------------------------------------------
    if budget_s >= 800:
        n_expl = 8000
    elif budget_s >= 400:
        n_expl = 6000
    else:
        n_expl = 4000
    target_expl = np.sqrt(2.0 * n_expl)

    # Diverse seeds
    init_pool = []

    for p in np.linspace(0.2, 1.0, 9):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))
    for lam in np.linspace(0.001, 0.02, 6):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))
    for _ in range(8):
        init_pool.append(_random_simplex(n_expl, target_expl, rng))

    # Warm‑start from previous best if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_expl))

    # Add jittered variants of the pool
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.02, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    best_seq = None
    best_score = np.inf

    # Short annealed GD runs – keep the best found
    while time.time() < deadline - 12.0:
        # Choose a seed (biased toward current best)
        if best_seq is not None and rng.random() < 0.6:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.005, size=n_expl)
            seed += jitter
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)

        run_dead = min(time.time() + 18.0, deadline - 10.0)
        if run_dead <= time.time():
            break

        a_opt, val_opt = _annealed_gradient_descent(seed, run_dead, target_expl,
                                                    max_beta=5e5,
                                                    step_factor=0.7,
                                                    gamma=0.99)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best={best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Light deterministic mass‑transfer refinement
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        mt_dead = time.time() + (deadline - time.time()) * 0.12
        a_ref, val_ref = _deterministic_mass_transfer(best_seq, mt_dead,
                                                      target_expl,
                                                      start_temp=0.4,
                                                      end_temp=1e-4,
                                                      low_frac=0.06)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] After MT refinement best={best_score:.6f}")

    # --------------------------------------------------------------
    # 3) LP alternating refinement on a down‑scaled problem
    # --------------------------------------------------------------
    if time.time() < deadline - 30.0:
        n_lp = min(1500, max(800, n_expl // 5))
        target_lp = np.sqrt(2.0 * n_lp)
        seq_lp = _resize_sequence(best_seq, n_lp)
        seq_lp = _project_to_simplex(seq_lp, target_lp)

        max_lp_iters = 8
        lp_iter = 0
        while lp_iter < max_lp_iters and time.time() < deadline - 12.0:
            M = _max_convolution_fft(seq_lp)
            b_opt = _lp_optimize(seq_lp, M, target_lp, rng)
            if b_opt is None:
                break
            t = 0.22   # moderate mixing
            seq_lp = (1 - t) * seq_lp + t * b_opt
            seq_lp = _project_to_simplex(seq_lp, target_lp)
            cur_score = _score(seq_lp)
            print(f"[{time.time():.2f}] LP iter={lp_iter} score={cur_score:.6f}")
            lp_iter += 1

        # Brief GD polishing on the small problem
        if time.time() < deadline - 8.0:
            gd_dead = time.time() + min(9.0, deadline - time.time() - 6.0)
            a_opt, val_opt = _annealed_gradient_descent(seq_lp, gd_dead,
                                                        target_lp,
                                                        max_beta=2e5,
                                                        step_factor=0.6,
                                                        gamma=0.98)
            if val_opt < _score(seq_lp):
                seq_lp = a_opt.copy()
                print(f"[{time.time():.2f}] Post‑LP GD score={val_opt:.6f}")

        # Upscale to a larger length for final polishing
        if deadline - time.time() > 80.0:
            n_final = 16000
        elif deadline - time.time() > 45.0:
            n_final = 12000
        else:
            n_final = n_expl
        target_final = np.sqrt(2.0 * n_final)
        seq_big = _resize_sequence(seq_lp, n_final)
        seq_big = _project_to_simplex(seq_big, target_final)
    else:
        # Not enough time: upscale directly
        if deadline - time.time() > 80.0:
            n_final = 16000
        elif deadline - time.time() > 45.0:
            n_final = 12000
        else:
            n_final = n_expl
        target_final = np.sqrt(2.0 * n_final)
        seq_big = _resize_sequence(best_seq, n_final)
        seq_big = _project_to_simplex(seq_big, target_final)

    # --------------------------------------------------------------
    # 4) Simple re‑orderings before large‑scale polishing
    # --------------------------------------------------------------
    candidates = [seq_big,
                  _alternating_order(seq_big),
                  _even_spread_order(seq_big)]
    seq_big = min(candidates, key=_max_convolution_fft)
    best_score = _score(seq_big)
    print(f"[{time.time():.2f}] Upscaled length={len(seq_big)} start score={best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Sub‑gradient polishing on the large problem
    # --------------------------------------------------------------
    if time.time() < deadline - 6.0:
        sub_dead = deadline - 4.0
        a_sub, val_sub = _subgradient_descent(seq_big, sub_dead,
                                              target_final,
                                              step_factor=0.45,
                                              gamma=0.995)
        if val_sub < best_score:
            best_score = val_sub
            seq_big = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑grad polished best={best_score:.6f}")

    # --------------------------------------------------------------
    # 6) Alternating MT ↔ sub‑gradient passes for the remaining time
    # --------------------------------------------------------------
    while time.time() < deadline - 4.0:
        remaining = deadline - time.time()
        # sub‑gradient short burst
        sub_dead = time.time() + min(4.0, remaining * 0.4)
        a_sub, val_sub = _subgradient_descent(seq_big, sub_dead,
                                              target_final,
                                              step_factor=0.4,
                                              gamma=0.99)
        if val_sub < best_score:
            best_score = val_sub
            seq_big = a_sub.copy()
            print(f"[{time.time():.2f}] Alt Sub‑grad best={best_score:.6f}")

        # deterministic mass transfer short burst
        remaining = deadline - time.time()
        if remaining <= 2.0:
            break
        mt_dead = time.time() + min(4.0, remaining * 0.4)
        a_mt, val_mt = _deterministic_mass_transfer(seq_big, mt_dead,
                                                    target_final,
                                                    start_temp=0.25,
                                                    end_temp=1e-5,
                                                    low_frac=0.05)
        if val_mt < best_score:
            best_score = val_mt
            seq_big = a_mt.copy()
            print(f"[{time.time():.2f}] Alt MT best={best_score:.6f}")

    # --------------------------------------------------------------
    # 7) Random hill‑climbing fine tuner (last few seconds)
    # --------------------------------------------------------------
    if time.time() < deadline - 1.0:
        hc_dead = deadline - 0.5
        a_hc, val_hc = _hill_climb(seq_big, target_final, hc_dead, rng,
                                   max_frac=0.08)
        if val_hc < best_score:
            best_score = val_hc
            seq_big = a_hc.copy()
            print(f"[{time.time():.2f}] Hill‑climb best={best_score:.6f}")

    # --------------------------------------------------------------
    # 8) Trim trailing near‑zeros and finalize
    # --------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final evaluation for logging (no effect on the result)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    globals()["height_sequence_1"] = final_seq
    return final_seq
