    a = _project_to_simplex(a, target_sum)

"""
Sub‑gradient optimizer for the step‑function benchmark.

The algorithm proceeds in three phases:

1. **Exploration** – many short runs of a projected sub‑gradient descent
   on the exact max‑convolution objective (rather than the soft‑max
   surrogate).  The method fixes the ℓ¹‑norm (∑a = √(2 n)) by a simplex
   projection after each step and uses a simple √t diminishing step‑size.
   A diverse pool of seeds (power‑law, exponential, random, previous
   best) is employed.

2. **Permutation annealing** – the best candidate from phase 1 is further
   refined by a simulated‑annealing that swaps indices with high/low
   contribution to the current peak of the convolution.

3. **Tiny mass‑transfer polish** – a deterministic local move that shifts a
   small amount of mass from the strongest donor to the weakest receiver,
   removing trailing zeros for compactness.

The search adapts the problem size `n` to the available time (up to
≈ 15000 entries) and always returns a non‑negative list of floats that
satisfies the required ℓ¹‑normalisation.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (∑a)² (with ∑a = √(2n) this equals max)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution objective
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on  max(conv) with simplex projection."""
    a = _project_to_simplex(a0.copy(), target_sum)
    n = a.shape[0]
    # initial score
    conv = _conv_fft(a)
    max_b = conv.max()
    best_val = 2.0 * n * max_b / (target_sum * target_sum)
    best_a = a.copy()

    rng = np.random.default_rng()
    step0 = target_sum * 0.02            # initial step magnitude
    it = 0

    while time.time() < deadline:
        # convolution and current peak
        conv = _conv_fft(a)
        max_b = conv.max()
        # pick a random maximiser (break ties randomly)
        kmax = int(rng.choice(np.flatnonzero(conv == max_b)))

        # sub‑gradient of the peak entry
        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[partner[mask]]

        # diminishing step size
        step = step0 / np.sqrt(it + 1.0)

        # gradient step + projection
        a = a - step * grad
        a = _project_to_simplex(a, target_sum)

        # occasional small jitter to escape flat regions
        if it % 800 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        # evaluate new score (reuse convolution if cheap)
        conv = _conv_fft(a)
        max_b = conv.max()
        val = 2.0 * n * max_b / (target_sum * target_sum)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (small moves)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-4,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    cur_score = 2.0 * n * _max_convolution_fft(a) / (target_sum * target_sum)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = 2.0 * n * _max_convolution_fft(a) / (target_sum * target_sum)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = 2.0 * n * _max_convolution_fft(a) / (target_sum * target_sum)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted permutation search (simulated annealing on index order)
# ----------------------------------------------------------------------


def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-5,
                                 top_frac: float = 0.05,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Simulated annealing that swaps indices with high and low contribution
    to the current peak of the convolution.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = 2.0 * n * _max_convolution_fft(a) / (np.sum(a) ** 2)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        kmax_candidates = np.flatnonzero(conv == max_val)
        kmax = rng.choice(kmax_candidates)

        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)

        if donor == receiver:
            continue

        a[donor], a[receiver] = a[receiver], a[donor]

        new_score = 2.0 * n * _max_convolution_fft(a) / (np.sum(a) ** 2)

        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12))

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] PermSearch new best={best_score:.6f}")
        else:
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine performs:
      1. Exploration via many short projected sub‑gradient runs.
      2. A permutation‑only simulated annealing using the remaining time.
      3. A final tiny mass‑transfer polish and trimming of trailing zeros.
    The result respects the required ℓ¹‑normalisation.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5           # tiny safety margin

    # ------------------------------------------------------------------
    # Choose problem size based on the available budget
    # ------------------------------------------------------------------
    if budget_s >= 850:
        n = 15000
    elif budget_s >= 600:
        n = 12000
    else:
        n = 8000
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Build a diverse pool of initial candidates
    # ------------------------------------------------------------------
    init_pool: list[np.ndarray] = []

    # power‑law family
    for p in np.linspace(0.35, 0.75, 9):
        init_pool.append(_power_law_sequence(n, float(p), target_sum))

    # exponential family
    for lam in np.linspace(0.0015, 0.015, 7):
        init_pool.append(_exponential_sequence(n, float(lam), target_sum))

    # random Dirichlet‑like points
    for _ in range(6):
        a = rng.random(n)
        a *= target_sum / a.sum()
        init_pool.append(a)

    # linear decreasing shape
    lin = np.linspace(1.0, 0.0, n)
    lin *= target_sum / lin.sum()
    init_pool.append(lin)

    # re‑use previous best if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        prev_resized = _resize_sequence(prev, n)
        prev_resized = _project_to_simplex(prev_resized, target_sum)
        init_pool.append(prev_resized)
        init_pool.append(_alternating_order(prev_resized))
        init_pool.append(_reversed_alternating(prev_resized))

    # ------------------------------------------------------------------
    # Exploration: many short sub‑gradient runs
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    while time.time() < deadline - 30.0:   # reserve a few seconds for later phases
        # seed selection
        if best_seq is not None and rng.random() < 0.55:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_sum * 0.005, size=n)
            seed = seed + jitter
            seed = _project_to_simplex(seed, target_sum)
        else:
            seed = rng.choice(init_pool)

        # run a bounded sub‑gradient instance (~12 s each)
        run_dead = min(time.time() + 12.0, deadline - 20.0)
        if run_dead <= time.time():
            break

        a_opt, val_opt = _subgradient_descent(seed, run_dead, target_sum)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # If no candidate was found (unlikely), fall back to a simple shape
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _power_law_sequence(n, 0.55, target_sum)
        best_score = _score(best_seq)

    # ------------------------------------------------------------------
    # Permutation‑only simulated annealing (uses all remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        perm_dead = deadline - 0.5
        a_perm, val_perm = _permutation_targeted_search(
            best_seq,
            perm_dead,
            start_temp=0.5,
            end_temp=1e-5,
            top_frac=0.05,
            low_frac=0.05,
        )
        if val_perm < best_score:
            best_score = val_perm
            best_seq = a_perm.copy()
            print(f"[{time.time():.2f}] After permutation search best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final tiny deterministic mass‑transfer polish
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        mt_dead = time.time() + min(4.0, deadline - time.time() - 0.4)
        a_mt, val_mt = _deterministic_mass_transfer(
            best_seq,
            mt_dead,
            target_sum,
            start_temp=0.3,
            end_temp=1e-5,
            low_frac=0.05,
        )
        if val_mt < best_score:
            best_score = val_mt
            best_seq = a_mt.copy()
            print(f"[{time.time():.2f}] Final mass‑transfer best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing zeros for compactness
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Log final score (optional)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for possible later reuse
    globals()["height_sequence_1"] = final_seq
    return final_seq
