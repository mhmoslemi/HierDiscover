def propose_candidate(budget_s=1000):
    start = time.time()
    deadline = start + budget_s - 0.5

    # choose n based on budget (max possible)
    if budget_s >= 900:
        n = 36000
    elif budget_s >= 600:
        n = 24000
    else:
        n = 18000

    target_sum = np.sqrt(2.0*n)

    # Initialize sequence
    if "height_sequence_1" in globals():
        init = np.array(globals()["height_sequence_1"], dtype=np.float64)
        init = _resize_sequence(init, n)
    else:
        # start with power-law decreasing shape
        x = np.arange(1, n+1, dtype=np.float64)
        init = x**(-0.55)
    # normalize
    init *= target_sum/init.sum()

    # Stage 1: exponential descent (softmax)
    exp_deadline = time.time() + 0.10*(deadline-time.time())
    a, best_val = exponential_descent(init, target_sum, exp_deadline)
    best_seq = a
    best_score = best_val
    print(f"After exponential descent: {best_score:.6f}")

    # Stage 2: Adam-like exponentiated subgradient descent
    sub_deadline = time.time() + 0.30*(deadline-time.time())
    a, val = exp_subgradient_descent(best_seq, target_sum, sub_deadline)
    if val < best_score:
        best_seq, best_score = a, val
    print(f"After subgradient descent: {best_score:.6f}")

    # Stage 3: annealing
    anneal_deadline = deadline
    a, val = anneal_mass_transfer(best_seq, target_sum, anneal_deadline)
    if val < best_score:
        best_seq, best_score = a, val
    print(f"After annealing: {best_score:.6f}")

    # Clean up trailing zeros
    arr = best_seq
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[:nz[-1]+1]
    final_seq = arr.tolist()
    # evaluate final
    try:
        final_val = evaluate_sequence(final_seq)
        print(f"Final evaluated score = {final_val:.6f}")
    except Exception as e:
        print("Evaluation error:", e)
    globals()["height_sequence_1"] = final_seq
    return final_seq

def subgradient_exp_descent(a0, target_sum, deadline, top_k=3):
    a = a0.copy()
    a *= target_sum/a.sum()
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        b = _conv_fft(a)
        # get top_k indices
        if top_k == 1:
            idxs = [np.argmax(b)]
        else:
            idxs = np.argpartition(b, -top_k)[-top_k:]
        g = np.zeros_like(a)
        n = a.shape[0]
        for k in idxs:
            lo = max(0, k-(n-1))
            hi = min(k, n-1)
            idx = np.arange(lo, hi+1)
            g[idx] += 2.0 * a[k - idx]
        # eta schedule: start high, decrease linearly
        elapsed = time.time() - start
        total = deadline - start
        eta = 0.5 * (1 - elapsed/total) + 0.02
        a = a * np.exp(-eta * g)
        a = a * (target_sum/a.sum())
        # jitter occasional
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum*1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum/a.sum())
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubExp it={it} score={best_val:.6f}")
        it+=1
    return best_a, best_val

def _conv_fft(a):
    n = a.size
    size = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa*fa, n=size)[:2*n-1]
    return conv

"""
Propose a step‑function sequence that minimises `evaluate_sequence`.

The algorithm works in three phases within the given time budget.

1. **Exponential (mirror) descent on a soft‑max surrogate** – a fast
   multiplicative update that reduces the peak of the self‑convolution
   while keeping the L¹‑norm fixed.

2. **Exponentiated sub‑gradient descent on the true max‑convolution** –
   the gradient is built from the K largest convolution entries (default
   K=3).  A decaying step‑size together with occasional Gaussian jitter
   and light smoothing helps escape plateaus.

3. **Simulated‑annealing mass‑transfer refinement** – local transfers
   of a tiny amount of mass from a donor (chosen proportionally to its
   contribution to the current peak) to a receiver (chosen inversely to
   its height).  The temperature follows a √‑schedule, and rare global
   perturbations are allowed to explore more distant basins.

All updates respect non‑negativity and the required sum
`Σ a_i = sqrt(2·n)`.  The best sequence seen is returned (trailing
zeros are removed).  If a previous run stored a sequence in the global
variable `height_sequence_1`, it is used as a warm‑start and resized to
the current problem size.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution / scoring utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  For Σa=√(2n) this equals max(a*a)."""
    n = a.shape[0]
    b = _conv_fft(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of the max‑convolution.
    Returns (grad, full_convolution) where grad has the same length as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    grad_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(grad_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)

# Phase A – exponential (mirror) descent on a soft‑max surrogate
# ----------------------------------------------------------------------


def exponential_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Mirror descent on the soft‑max surrogate of the max‑convolution.
    Returns the best sequence seen and its score.
    """
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        # occasional jitter
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # occasional light smoothing
        if it % 500 == 0:
            kernel = np.array([0.25, 0.5, 0.25])
            a = np.convolve(a, kernel, mode='same')

            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()

        if it % 400 == 0 and beta < max_beta:

        it += 1

    return best_a, best_val

def subgradient_exp_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    top_k: int = 3,
):
    """
    Exponentiated sub‑gradient descent using the top‑k convolution peaks.
    Returns the best sequence seen and its score.
    """
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0
    n = a.shape[0]

    while time.time() < deadline:
        b = _conv_fft(a)

        # pick the K largest entries of the convolution
        if top_k == 1:
            peak_idxs = [int(np.argmax(b))]
        else:
            peak_idxs = np.argpartition(b, -top_k)[-top_k:]

        # accumulate sub‑gradient contributions
        g = np.zeros_like(a)
        for k in peak_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            g[idx] += 2.0 * a[k - idx]

        # linear decay of the step‑size (from ≈0.5 to 0.02)
        elapsed = time.time() - start

        eta = 0.5 * (1.0 - frac) + 0.02

        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        # jitter and light smoothing to keep exploration alive
        if it % 500 == 0:

            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

            a = np.convolve(a, kernel, mode='same')
            a = np.maximum(a, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGradExp it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val

def anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Simulated annealing that moves tiny amounts of mass from a “bad’’ donor
    to a “good’’ receiver.  The temperature follows a sqrt‑schedule,
    occasional global perturbations are tried, and light smoothing is
    applied periodically.
    Returns the best sequence seen and its score.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.85  # initial temperature (≈80 % of current score)

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        # --------------------------------------------------------------
        # Occasionally try a global perturbation (large random jump)
        # --------------------------------------------------------------
        if rng.random() < 0.001:
            perturb = rng.normal(scale=target_sum * 0.02, size=n)
            a_new = a + perturb
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new * (target_sum / a_new.sum())
            new_val = _score(a_new)
            accept = False
            if new_val <= cur_val:
                accept = True
            elif temperature > 0.0:
                prob = np.exp(-(new_val - cur_val) / temperature)
                if rng.random() < prob:
                    accept = True
            if accept:
                a = a_new
                b = _conv_fft(a)
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best_a = a.copy()

        # Local mass‑transfer move

        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            it += 1
            continue

        if max_delta <= 0.0:
            it += 1
            continue

            it += 1
            continue

        b_new = b.copy()

        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta

            b_new[donor + receiver] -= 2.0 * delta * delta

        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved best={best_val:.6f}")

        # periodic full recomputation (prevents drift)
        if it % 5000 == 0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved best={best_val:.6f}")

        # occasional smoothing to keep the shape regular
        if it % 2000 == 0:
            kernel = np.array([0.25, 0.5, 0.25])
            a = np.convolve(a, kernel, mode='same')
            a = np.maximum(a, 0.0)
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = _score(a)

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence that attains a low
    `evaluate_sequence` score within the given time budget (seconds).

    Returns the best sequence found as a plain Python list.
    """
    start = time.time()
    deadline = start + budget_s - 0.5     # safety margin against overruns

    # --------------------------------------------------------------
    # Choose a problem size – longer budgets allow a finer discretisation.
    # --------------------------------------------------------------
    if budget_s >= 900:
        n = 36000
    elif budget_s >= 600:
        n = 24000
    else:
        n = 18000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Warm‑start from a previously discovered construction if available.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = np.array(globals()["height_sequence_1"], dtype=np.float64)
        init = _resize_sequence(init, n)
    else:
        # simple decreasing power‑law (empirically good)
        x = np.arange(1, n + 1, dtype=np.float64)
        init = x ** (-0.55)

    init = init * (target_sum / init.sum())

    # --------------------------------------------------------------
    # Phase A – exponential descent (≈12 % of the remaining time)
    # --------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    if remaining > 0.0:
        exp_deadline = now + 0.12 * remaining
        a_exp, val_exp = exponential_descent(init, target_sum, exp_deadline)
        best_seq, best_score = a_exp, val_exp
        print(f"[{time.time():.2f}] After exponential descent best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – exponentiated sub‑gradient descent (≈35 % of remaining)
    # --------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    if remaining > 0.0:
        sub_deadline = now + 0.35 * remaining
        a_sub, val_sub = subgradient_exp_descent(best_seq, target_sum, sub_deadline, top_k=3)
        if val_sub < best_score:
            best_seq, best_score = a_sub, val_sub
        print(f"[{time.time():.2f}] After sub‑gradient descent best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase C – annealing (use all still available time)
    # --------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    if remaining > 0.0:
        a_ann, val_ann = anneal_mass_transfer(best_seq, target_sum, deadline)
        if val_ann < best_score:
            best_seq, best_score = a_ann, val_ann
        print(f"[{time.time():.2f}] After annealing best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final clean‑up – remove trailing zeros and evaluate once more.
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for warm‑starts of subsequent runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
