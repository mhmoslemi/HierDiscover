"""
Improved optimiser for the non‑negative step‑function benchmark.

Strategy
--------
1. Choose a fairly large length `n` (≈12‑15k) – a longer vector gives more
   freedom to push the auto‑convolution peak down.
2. Build a diverse pool of seed vectors:
   * power‑law, exponential, linear, half‑zero (mass only on a prefix)
   * random Dirichlet‑like draws
   * the previous global best (`height_sequence_1`) if available
   * jittered copies of every seed
3. Run cheap projected sub‑gradient descent (exact max‑convolution gradient)
   on many seeds for a few seconds each; keep the best.
4. Optionally solve a small convex instance (n≈120) with CVXPY, then up‑scale
   the result – this gives a near‑optimal ‘shape’ we can polish later.
5. Refine the current best with a longer sub‑gradient run.
6. Apply a high‑β soft‑max gradient descent (smooth approximation of the max).
7. Perform a targeted permutation simulated‑annealing that swaps indices
   contributing heavily to the current peak.
8. Finish with a deterministic mass‑transfer pass.
9. Trim trailing zeros and return the list.

All optimisation phases respect the global deadline (≈1000 s).  The code
contains no file I/O, uses only NumPy (+optional CVXPY) and respects the
required signature `propose_candidate`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x|x_i≥0, Σx_i=s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _half_zero_power_sequence(
    n: int, p: float, target_sum: float, fraction: float = 0.5
) -> np.ndarray:
    """Mass on the first `fraction` of the indices, power‑law decay."""
    m = int(np.floor(fraction * n))
    a = np.zeros(n, dtype=np.float64)
    a[:m] = (np.arange(1, m + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


# ----------------------------------------------------------------------
# Scoring utility (objective = 2·n·max(conv)/sum²)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Exact max‑convolution sub‑gradient
# ----------------------------------------------------------------------


def _max_conv_and_subgrad(a: np.ndarray):
    """
    Returns (conv, max_val, k_max, subgrad) where subgrad is the gradient
    of the peak entry of the convolution w.r.t. a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    kmax = int(np.argmax(conv))
    max_val = conv[kmax]

    grad = np.zeros_like(a)
    lo = max(0, kmax - (n - 1))
    hi = min(kmax, n - 1)
    idx = np.arange(lo, hi + 1)
    partner = a[kmax - idx]
    grad[lo:hi + 1] = 2.0 * partner
    return conv, max_val, kmax, grad


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    step_factor: float = 0.5,
    gamma: float = 0.99,
) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0
    momentum = np.zeros_like(a)

    while time.time() < deadline:
        _, _, _, grad = _max_conv_and_subgrad(a)

        # momentum
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_factor * target_sum / norm
            a -= lr * momentum
        else:
            a -= step_factor * target_sum * grad / (np.linalg.norm(grad) + 1e-12)

        a = _project_to_simplex(a, target_sum)

        # jitter every so often
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] Sub‑GD it={it:5d}  score={best_val:.6f}")

        it += 1

    return best, best_val


# ----------------------------------------------------------------------
# Soft‑max gradient descent (annealed)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    grad = 2.0 * np.fft.irfft(
        np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size), n=size
    )[: 2 * n - 1]
    grad = grad[n - 1 : n - 1 + n]
    return grad, b


def _annealed_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 1e6,
    step_factor: float = 0.5,
    gamma: float = 0.995,
) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Soft‑GD it={it:5d} β={beta:.1f} score={best_val:.6f}")

        # increase β
        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

# Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    start_temp: float = 0.5,
    end_temp: float = 1e-4,
    low_frac: float = 0.05,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]  # contribution weight to the peak
        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted permutation search (simulated annealing)
# ----------------------------------------------------------------------


def _permutation_targeted_search(
    a0: np.ndarray,
    deadline: float,
    start_temp: float = 0.1,
    end_temp: float = 1e-4,
    top_frac: float = 0.02,
    low_frac: float = 0.02,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)

        if donor == receiver:
            continue

        a[donor], a[receiver] = a[receiver], a[donor]

        new_score = _score(a)

        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] PermSearch new best={best_score:.6f}")

        # Metropolis acceptance
        if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / max(temp, 1e-12)):
            pass
        else:
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score

# (Optional) Small‑scale convex optimisation using CVXPY

def _convex_small_opt(n_small: int = 120) -> np.ndarray | None:
    """Return a near‑optimal shape for a tiny problem via CVXPY, or None."""
    try:
        import cvxpy as cp
    except Exception:
        return None

    target = np.sqrt(2.0 * n_small)
    a = cp.Variable(n_small, nonneg=True)
    M = cp.Variable(nonneg=True)

    constraints = [cp.sum(a) == target]

    # Build quadratic matrices for each convolution index
    Qs = []
    for k in range(2 * n_small - 1):
        Q = np.zeros((n_small, n_small))
        lo = max(0, k - (n_small - 1))
        hi = min(k, n_small - 1)
        for i in range(lo, hi + 1):
            j = k - i
            Q[i, j] = 1.0
        Qs.append(Q)

    constraints += [cp.quad_form(a, Q) <= M for Q in Qs]

    prob = cp.Problem(cp.Minimize(M), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=4000, verbose=False)
    except Exception:
        return None

    if a.value is None:
        return None

    a_np = np.array(a.value).flatten()
    a_np = np.maximum(a_np, 0.0)
    a_np = a_np / a_np.sum() * target
    return a_np

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with a low benchmark score.
    The routine respects the 1000 s budget and stores the best vector in the
    global variable `height_sequence_1` for possible reuse.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # tiny safety margin

    # ------------------------------------------------------------------
    # 1. Choose a working length based on the available time budget
    # ------------------------------------------------------------------
    if budget_s >= 900:
        n = 15000
    elif budget_s >= 700:
        n = 12000
    else:
        n = 8000
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # 2. Build a diverse pool of seed sequences
    # ------------------------------------------------------------------
    init_pool: list[np.ndarray] = []

    # power‑law family
    for p in [0.2, 0.35, 0.55, 0.75, 1.0]:
        init_pool.append(_power_law_sequence(n, p, target_sum))

    # exponential family
    for lam in [0.001, 0.003, 0.006, 0.012]:
        init_pool.append(_exponential_sequence(n, lam, target_sum))

    # half‑zero power‑law (mass only on a prefix)
    for frac in [0.33, 0.5]:
        for p in [0.5, 0.9, 1.4]:
            init_pool.append(_half_zero_power_sequence(n, p, target_sum, fraction=frac))

    # linear decreasing shape
    lin = np.linspace(1.0, 0.0, n)
    lin *= target_sum / lin.sum()
    init_pool.append(lin)

    # random Dirichlet‑like draws
    for _ in range(6):
        a = rng.random(n)
        a *= target_sum / a.sum()
        init_pool.append(a)

    # previous global best, if any
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n))

    # jittered copies for extra diversity
    jittered = []
    for seq in init_pool:
        jitter = rng.normal(scale=target_sum * 0.008, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        jittered.append(seq_j)
    init_pool.extend(jittered)

    # optional convex shape from a tiny problem
    small_opt = _convex_small_opt(120)
    if small_opt is not None:
        big_from_small = _resize_sequence(small_opt, n)
        big_from_small = _project_to_simplex(big_from_small, target_sum)
        init_pool.append(big_from_small)

    # ------------------------------------------------------------------
    # 3. Quick sub‑gradient runs on many seeds
    # ------------------------------------------------------------------
    best_seq: np.ndarray | None = None
    best_val = np.inf
    it_seed = 0

    while time.time() < deadline - 40.0:  # reserve ~40 s for final phases
        seed = rng.choice(init_pool)
        run_dead = min(deadline - 5.0, time.time() + 10.0)  # ≈10 s per run
        if run_dead <= time.time():
            break
        cand_seq, cand_val = _subgradient_descent(
            seed, run_dead, target_sum, step_factor=0.4, gamma=0.99
        )
        if cand_val < best_val:
            best_val = cand_val
            best_seq = cand_seq.copy()
            print(
                f"[{time.time():.2f}] Exploration it={it_seed} new best={best_val:.6f}"
            )
        it_seed += 1

    # If nothing succeeded (should not happen), fall back to a simple power‑law
    if best_seq is None:
        best_seq = _power_law_sequence(n, 0.55, target_sum)

    # ------------------------------------------------------------------
    # 4. Longer refinement on the current best
    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0:
        refine_dead = deadline - 10.0
        cand_seq, cand_val = _subgradient_descent(
            best_seq, refine_dead, target_sum, step_factor=0.35, gamma=0.995
        )
        if cand_val < best_val:
            best_val = cand_val
            best_seq = cand_seq.copy()
            print(f"[{time.time():.2f}] Refine sub‑GD best={best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. High‑β soft‑max gradient descent (smooth max)
    # ------------------------------------------------------------------
    if time.time() < deadline - 12.0:
        soft_dead = deadline - 4.0
        cand_seq, cand_val = _annealed_gradient_descent(
            best_seq,
            soft_dead,
            target_sum,
            max_beta=2e6,
            step_factor=0.35,
            gamma=0.998,
        )
        if cand_val < best_val:
            best_val = cand_val
            best_seq = cand_seq.copy()
            print(f"[{time.time():.2f}] Soft‑max final best={best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Targeted permutation simulated annealing (re‑ordering only)
    # ------------------------------------------------------------------
    if time.time() < deadline - 6.0:
        perm_dead = deadline - 2.0
        cand_seq, cand_val = _permutation_targeted_search(
            best_seq,
            perm_dead,
            start_temp=0.2,
            end_temp=1e-5,
            top_frac=0.08,
            low_frac=0.08,
        )
        if cand_val < best_val:
            best_val = cand_val
            best_seq = cand_seq.copy()
            print(f"[{time.time():.2f}] PermSearch final best={best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Tiny deterministic mass‑transfer polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        mt_dead = deadline - 1.0
        cand_seq, cand_val = _deterministic_mass_transfer(
            best_seq,
            mt_dead,
            target_sum,
            start_temp=0.3,
            end_temp=1e-5,
            low_frac=0.05,
        )
        if cand_val < best_val:
            best_val = cand_val
            best_seq = cand_seq.copy()
            print(f"[{time.time():.2f}] MT final best={best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Trim trailing zeros for compactness
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # 9. Log final evaluation (optional)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    globals()["height_sequence_1"] = final_seq
    return final_seq
