def _simulated_annealing(a0, target_sum, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    cur_score = best_score
    rng = np.random.default_rng()
    n = a.shape[0]
    T0 = best_score * 0.05  # initial temperature relative to objective
    T = T0
    iteration = 0
    # For speed, pre-allocate type?
    while time.time() < deadline:
        # Choose donor and receiver
        # Sample donor from top k fraction or just max
        # Heuristic: donor from top 5%:
        top_frac = 0.02
        k_top = max(1, int(top_frac*n))
        # Get random indices for donor selection among top heavy:
        # Instead of sorting each iteration, maintain list later? Might compute sorted only occasionally.
        if iteration % 2000 == 0:
            sorted_desc = np.argsort(-a)  # descending
            sorted_asc = np.argsort(a)    # ascending
        donor = sorted_desc[rng.integers(k_top)]
        receiver = sorted_asc[rng.integers(k_top)]
        if donor == receiver:
            continue
        max_delta = a[donor]  # cannot exceed
        if max_delta <= 0:
            continue
        # Choose delta as a random fraction of donor mass, e.g., up to 30% of donor
        delta = max_delta * rng.random() * 0.3
        # ensure not too small (skip negligible)
        if delta < 1e-12:
            continue
        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        # No need to project because sum unchanged and both non-negative.
        # Compute new score
        new_score = _score(a_new)
        delta_obj = new_score - cur_score
        accept = False
        if delta_obj <= 0:
            accept = True
        else:
            # Metropolis
            prob = np.exp(-delta_obj / max(T, 1e-12))

        iteration += 1
        if iteration % 5000 == 0:

"""
Step‑function optimizer for the max‑self‑convolution objective.

The algorithm proceeds in several stages within the allotted time budget:

1. **Parametric sweep** – evaluates simple families (pure power‑law,
   power‑law with exponential tail, sinusoidal) on a grid of parameters.
   The best candidate is retained.

2. **Deterministic mass‑transfer** – a greedy reduction of the current
   convolution peak (the “exact DMT2” routine from the reference code).

3. **Simulated annealing** – stochastic moves that transfer a small amount
   of mass from a high‑value index to a low‑value index.  Acceptance follows
   a Metropolis rule with a slowly cooling temperature, allowing the search
   to escape local minima.

4. **Final refinement** – a short run of exact mass‑transfer followed by
   an Adam‑style projected gradient descent on a soft‑max approximation of
   the max‑convolution (high β) to polish the result.

All intermediate candidates are projected onto the simplex
{a_i ≥ 0, Σ a_i = √(2 n)}.  The returned list is trimmed of trailing
near‑zero entries.  The routine respects the time budget (default 1000 s)
and prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective value – the maximal self‑convolution entry."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for the quick pre‑search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    """A smooth sine‑shaped candidate (always positive)."""
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (core refinement)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the maximal convolution entry.
    Moves a carefully chosen amount of mass from a “donor’’ index (large
    partner value) to a “receiver’’ index (small partner value) so that the
    current peak drops to the second‑largest value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second‑largest entry (ignore all current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # pick one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k‑i}
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        # donor = index with largest partner value among viable donors
        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # receiver = index (different from donor) with smallest partner value
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom

        # move a safe fraction of the donor mass
        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing on the simplex
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic mass‑transfer with Metropolis acceptance.  Each move transfers
    a random amount of mass from a high‑value index to a low‑value index.
    Temperature is exponentially cooled.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    cur_score = best_score

    n = a.shape[0]
    rng = np.random.default_rng()

    # initial temperature (fraction of the current objective)
    T = max(best_score * 0.05, 1e-6)
    it = 0
    top_frac = 0.02                      # donor / receiver selected from top/bottom 2 %
    top_k = max(1, int(top_frac * n))
    # pre‑compute sorted orders periodically
    sorted_desc = np.argsort(-a)
    sorted_asc = np.argsort(a)

    while time.time() < deadline:
        # refresh sorted lists occasionally (cost O(n log n))
        if it % 2000 == 0:
            sorted_desc = np.argsort(-a)
            sorted_asc = np.argsort(a)

        donor = sorted_desc[rng.integers(top_k)]
        receiver = sorted_asc[rng.integers(top_k)]
        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor]
        if max_delta <= 0.0:
            it += 1
            continue

        # Random fraction up to 30 % of donor mass
        delta = max_delta * rng.random() * 0.3
        if delta < 1e-12:
            it += 1
            continue

        a_candidate = a.copy()
        a_candidate[donor] -= delta
        a_candidate[receiver] += delta

        # both entries stay non‑negative, sum unchanged → already feasible
        new_score = _score(a_candidate)
        delta_obj = new_score - cur_score

        accept = False
        if delta_obj <= 0.0:
            accept = True
        else:
            prob = np.exp(-delta_obj / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_candidate
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA improved best={best_score:.6f} it={it}")

        # cooling
        it += 1
        if it % 5000 == 0:
            T *= 0.9
            if T < 1e-7:
                T = 1e-7

    return best_a, best_score


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on a soft‑max surrogate
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(b)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max smoothed objective."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimizing the maximal
    self‑convolution.  The function respects the time budget and returns
    the best list of heights found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # --- problem size ---------------------------------------------------
    # Pick n based on available time (larger n gives more freedom)
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000
    target_sum = np.sqrt(2.0 * n)   # Σ a = √(2 n) makes the objective = max_conv

    # ------------------------------------------------------------------
    # Phase 0 – fast parametric pre‑search
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    # grids for power‑law exponent and exponential tail strength
    p_grid = np.linspace(0.20, 1.20, 31)   # broader range than before
    q_grid = np.linspace(0.0, 10.0, 31)

    # (a) pure power‑law
    for p in p_grid:
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # (b) power‑law + exponential tail
    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # (c) simple sine shape
    seq = _sin_sequence(n, target_sum)
    val = _score(seq)
    if val < best_score:
        best_score = val
        best_seq = seq.copy()

    # (d) previous best if available (resize if needed)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score = val
            best_seq = resized.copy()

    # (e) small jittered population around the current best
    for _ in range(8):
        jitter = rng.normal(scale=target_sum * 0.005, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _score(cand)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()

    print(f"[{time.time():.2f}] After parametric sweep best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – deterministic mass‑transfer (quick refinement)
    # ------------------------------------------------------------------
    mt_deadline = time.time() + (deadline - time.time()) * 0.15
    best_seq, best_score = _deterministic_mass_transfer_exact2(best_seq,
                                                               mt_deadline,
                                                               target_sum)

    print(f"[{time.time():.2f}] After first DMT best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Simulated annealing
    # ------------------------------------------------------------------
    sa_deadline = time.time() + (deadline - time.time()) * 0.45
    best_seq, best_score = _simulated_annealing(best_seq,
                                                target_sum,
                                                sa_deadline)

    print(f"[{time.time():.2f}] After SA best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – second deterministic mass‑transfer (tightening)
    # ------------------------------------------------------------------
    mt2_deadline = time.time() + (deadline - time.time()) * 0.20
    best_seq, best_score = _deterministic_mass_transfer_exact2(best_seq,
                                                               mt2_deadline,
                                                               target_sum)

    print(f"[{time.time():.2f}] After second DMT best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final Adam polishing with a very high β
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        final_deadline = deadline
        best_seq, best_score = _adam_gradient_descent(best_seq,
                                                      final_deadline,
                                                      target_sum,
                                                      max_beta=1e6,
                                                      step_factor=0.6,
                                                      gamma=0.995)
        print(f"[{time.time():.2f}] Final Adam polish best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return list
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final evaluation (harmless if evaluate_sequence is unavailable)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
