def _lp_max_sum(a, M, target_sum, top_k=20):

    n = a.shape[0]

    # compute conv for a*a to get top_k peaks

    conv = _conv_fft(a)  # shape 2n-1

    idx_sorted = np.argsort(-conv)  # descending

    idx_constraints = idx_sorted[:top_k]

    # Build A matrix shape (top_k, n)

    A = np.zeros((top_k, n), dtype=np.float64)

    for row_idx, i in enumerate(idx_constraints):

        lo = max(0, i - (n - 1))

        hi = min(i, n - 1)

        # indices j from lo..hi such that a_{i-j} correspond

        j = np.arange(lo, hi+1, dtype=np.int64)

        A[row_idx, j] = a[i - j]

    # bounds for variables: g_j >=0, no upper bound

    bounds = [(0, None)] * n

    # constraint: A @ g <= M

    # Solve maximize sum(g) = -c^T g where c = -1

    c = -np.ones(n)  # minimize - sum(g)

    res = linprog(c, A_ub=A, b_ub=np.full(top_k, M), bounds=bounds, method='highs', options={'presolve':True})

    if not res.success:

        return None

    g = res.x

    # ensure sum positive

    if g.sum() <= 1e-12:

        return None

    # scale to target sum exactly (since we need sum = target_sum)

    g = g * (target_sum / g.sum())

    return g

def _conv_fft_two(a,b):
    n = a.size

    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: 2 * n - 1]
    return conv

"""
Propose a step‑function with a minimal maximal self‑convolution.

The strategy improves on the reference implementation by adding a
deterministic linear‑programming refinement step.  For the current
sequence `a` we solve

    maximise   Σ g_j
    subject to  (a * g)_i ≤ M   for the indices i where the
                self‑convolution a*a attains its maximum M,
                g_j ≥ 0.

The solution `g` is rescaled to the required ℓ¹‑norm and combined with
`a`.  For a given pair (a,g) the best convex mixing weight `t∈[0,1]`
is found by a cheap 1‑D line search on the exact maximum of the
quadratic convex combination

    (1−t)²·(a*a) + 2t(1−t)·(a*g) + t²·(g*g).

The method is embedded in a time‑budgeted pipeline:

 * a fast parametric sweep (power‑law, exponential‑tail, sine, beta)
   gives a good initial guess;
 * a short Adam‑style soft‑max gradient descent refines the guess;
 * the LP‑refinement loop (the novel part) iteratively reduces the
   peak;
 * a final high‑β Adam polishing finishes the search.

All sequences are projected onto the simplex
    {x≥0, Σx = √(2 n)}
so that the evaluation function reduces to the maximal convolution
entry.  The routine respects the 1000 s time limit and returns the best
sequence found.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft_two(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (fast pre‑search)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0))
    a *= target_sum / a.sum()
    return a


def _beta_sequence(n: int, alpha: float, beta: float, target_sum: float) -> np.ndarray:
    """Beta‑distribution‑shaped candidate (always non‑negative)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    a = (np.maximum(x, eps) ** (alpha - 1)) * (np.maximum(1.0 - x, eps) ** (beta - 1))
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma_frac: float, target_sum: float) -> np.ndarray:
    """Gaussian centred in the middle, σ expressed as a fraction of n."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    rev_fft = np.fft.rfft(a_rev, n=size)
    p_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(rev_fft * p_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# 1‑D line search for the optimal convex mixing weight
# ----------------------------------------------------------------------


def _optimal_mix_weight(a2: np.ndarray, g2: np.ndarray, ag: np.ndarray,
                        max_iter: int = 30) -> float:
    """Find t∈[0,1] minimising max_i ((1‑t)² a2 + 2t(1‑t) ag + t² g2)."""
    lo, hi = 0.0, 1.0
    for _ in range(max_iter):
        t1 = lo + (hi - lo) / 3.0
        t2 = hi - (hi - lo) / 3.0

        c1 = (1.0 - t1) ** 2 * a2 + 2.0 * t1 * (1.0 - t1) * ag + t1 ** 2 * g2
        c2 = (1.0 - t2) ** 2 * a2 + 2.0 * t2 * (1.0 - t2) * ag + t2 ** 2 * g2

        m1 = c1.max()
        m2 = c2.max()
        if m1 < m2:
            hi = t2
        else:
            lo = t1
    return (lo + hi) / 2.0


# ----------------------------------------------------------------------
# Linear‑programming refinement (core improvement)
# ----------------------------------------------------------------------


def _lp_refinement(a0: np.ndarray,
                   target_sum: float,
                   deadline: float,
                   top_k: int = 20) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the maximal convolution entry using a small LP.
    For the current peak indices we maximise the total mass of a donor
    sequence g under the constraint (a * g)_i ≤ M.
    The best convex mixing factor t is obtained by a cheap 1‑D line
    search on the exact quadratic expression.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_fft(a)
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _conv_fft(a)
        M = conv.max()
        # select the top‑k indices (with largest convolution values)
        if top_k >= conv.size:
            idx = np.arange(conv.size)
        else:
            idx = np.argpartition(-conv, top_k - 1)[:top_k]

        n = a.size
        K = len(idx)
        # build the constraint matrix A (K × n)
        A = np.zeros((K, n), dtype=np.float64)
        for row, i in enumerate(idx):
            lo = max(0, i - (n - 1))
            hi = min(i, n - 1)
            j = np.arange(lo, hi + 1, dtype=np.int64)
            A[row, j] = a[i - j]

        # linear programme: maximise Σ g_j  ⇔ minimise –Σ g_j
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(K, M, dtype=np.float64)
        bounds = [(0.0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            print(f"[{time.time():.2f}] LP failed, terminating refinement.")
            break

        g = res.x
        if g.sum() < 1e-12:
            print(f"[{time.time():.2f}] LP returned zero mass, stop.")
            break

        # rescale to the required ℓ¹‑norm
        g = g * (target_sum / g.sum())

        # compute the three convolutions needed for the line search
        a2 = conv                     # a * a (already computed)
        g2 = _conv_fft(g)             # g * g
        ag = _conv_fft_two(a, g)      # a * g

        t_opt = _optimal_mix_weight(a2, g2, ag, max_iter=25)

        a_new = (1.0 - t_opt) * a + t_opt * g
        a_new = _project_to_simplex(a_new, target_sum)

        new_score = _max_convolution_fft(a_new)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new               # continue from the improved point
            print(f"[{time.time():.2f}] LP refinement improved score={best_score:.6f} (t={t_opt:.4f})")
            continue

        # No improvement – try a tiny random jitter to escape possible stall
        jitter = rng.normal(scale=target_sum * 0.0005, size=n)
        cand = _project_to_simplex(a + jitter, target_sum)
        cand_score = _max_convolution_fft(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
            print(f"[{time.time():.2f}] Random jitter after LP improved score={best_score:.6f}")
            continue

        # If neither the LP move nor the jitter improves the score we quit.
        break

    return best_a, best_score
