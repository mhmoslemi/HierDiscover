"""
A multi‑phase optimizer for the maximal self‑convolution problem.

1. Fast parametric sweep (power‑law, sin, etc.).
2. Random exploration of richer analytic families (offset power‑law,
   power‑law + exponential tail, Beta‑shaped) on the full size.
3. Adam‑style projected gradient descent on a soft‑max approximation,
   with several restarts.
4. Exact projected sub‑gradient descent on the true max‑convolution.
5. Deterministic “mass‑transfer” greedy reduction of the current peak.
6. Optional random mass–transfer to escape plateaus.
7. Final high‑β polishing.

All phases respect a single‑CPU 1000 s budget.  The best sequence found
is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Scoring (objective is exactly the maximal convolution entry)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Return the objective value (max‑convolution entry)."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families of candidate shapes
# ----------------------------------------------------------------------


def _power_law_offset_seq(n: int, p: float, alpha: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i + α)^{‑p}."""
    i = np.arange(n, dtype=np.float64)
    base = (i + alpha + 1e-12) ** (-p)
    base *= target_sum / base.sum()
    return base


def _exp_power_law_seq(
    n: int, p: float, alpha: float, q: float, r: float, target_sum: float
) -> np.ndarray:
    """a_i ∝ (i+α)^{‑p}·exp(‑q·(i/(n‑1))^r)."""
    i = np.arange(n, dtype=np.float64)
    x = i / max(1.0, n - 1)
    base = (i + alpha + 1e-12) ** (-p) * np.exp(-q * (x ** r))
    base *= target_sum / base.sum()
    return base


def _beta_seq(n: int, p1: float, p2: float, target_sum: float) -> np.ndarray:
    """a_i ∝ x^{p1}·(1‑x)^{p2} with x∈(0,1)."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n  # stay away from exact 0/1
    base = (x ** p1) * ((1.0 - x) ** p2)
    base *= target_sum / base.sum()
    return base


# ----------------------------------------------------------------------
# Randomised parametric exploration on the full dimension
# ----------------------------------------------------------------------


def _random_param_search(
    n: int, target_sum: float, deadline: float
) -> tuple[np.ndarray, float]:
    """Sample many analytic families and keep the best sequence."""
    rng = np.random.default_rng()
    # start from the uniform distribution (a safe baseline)
    best_seq = np.full(n, target_sum / n, dtype=np.float64)
    best_val = _score(best_seq)

    while time.time() < deadline:
        family = rng.integers(3)  # 0 = offset PL, 1 = exp‑PL, 2 = Beta
        if family == 0:
            p = rng.uniform(0.1, 2.0)
            alpha = rng.uniform(0.0, 5.0)
            seq = _power_law_offset_seq(n, p, alpha, target_sum)

        elif family == 1:
            p = rng.uniform(0.1, 2.0)
            alpha = rng.uniform(0.0, 5.0)
            q = rng.uniform(0.0, 8.0)
            r = rng.uniform(0.5, 3.0)
            seq = _exp_power_law_seq(n, p, alpha, q, r, target_sum)

        else:  # Beta family
            p1 = rng.uniform(0.0, 5.0)
            p2 = rng.uniform(0.0, 5.0)
            seq = _beta_seq(n, p1, p2, target_sum)

        val = _score(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(
                f"[{time.time():.2f}] Random family={family} val={best_val:.6f}"
            )
    return best_seq, best_val


# ----------------------------------------------------------------------
# Soft‑max based gradient (used by Adam)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(b) w.r.t. a."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]  # slice of length n
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 2e5,
    step_factor: float = 0.8,
    gamma: float = 0.999,
) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} score={best_val:.6f}"
            )

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(
    init: np.ndarray, target_sum: float, deadline: float
) -> tuple[np.ndarray, float]:
    """Run several short Adam restarts from jittered starts."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(
            start,
            time.time() + run_budget,
            target_sum,
            max_beta=2e5,
            step_factor=0.8,
            gamma=0.995,
        )
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] Adam restart improved best={best_val:.6f}"
            )

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------


def _subgradient_descent(
    a0: np.ndarray, target_sum: float, deadline: float, max_iter: int = 5000
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.flatnonzero(conv >= max_val - 1e-12)

        # second‑largest value (ignore current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()
        delta_target = max_val - second_max

        if delta_target < 1e-14:
            break

        k = idx_max[rng.integers(len(idx_max))]
        n = a.shape[0]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k - idx]

        norm_g2 = np.dot(g, g)
        if norm_g2 == 0.0:
            # degenerate – apply tiny random move
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = _project_to_simplex(a + jitter, target_sum)
            it += 1
            continue

        step = 0.9 * delta_target / norm_g2
        step = min(step, target_sum * 0.4 / (np.linalg.norm(g) + 1e-12))

        for _ in range(6):
            candidate = _project_to_simplex(a - step * g, target_sum)
            cand_score = _score(candidate)
            if cand_score < best_score - 1e-12:
                a = candidate
                best_a = candidate.copy()
                best_score = cand_score
                print(
                    f"[{time.time():.2f}] Subgrad it={it:5d} step={step:.3e} score={best_score:.6f}"
                )
                break
            step *= 0.5
        else:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = _project_to_simplex(a + jitter, target_sum)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (core refinement)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Greedy reduction of the maximal convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest value (ignore all maximal indices)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]  # take the first maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]  # a_{k‑i}
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # receiver – smallest partner value (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}"
            )

        it += 1

    return best_a, best_score

def _global_random_mass_transfer(

    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        donor = np.argsort(-a)[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        receiver = np.argsort(a)[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] Global random new best={best_score:.6f}"
            )

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal max‑convolution.
    The function respects the given time budget and returns a Python list.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n gives more freedom but costs more)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 16000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000
    target_sum = np.sqrt(2.0 * n)  # Σa = √(2 n)

    # --------------------------------------------------------------
    # Initial best candidate (uniform or previous global variable)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            candidate = np.interp(new_x, old_x, prev)
        else:
            candidate = prev.copy()
        candidate = _project_to_simplex(candidate, target_sum)
    else:
        candidate = np.full(n, target_sum / n, dtype=np.float64)

    best_seq = candidate.copy()
    best_score = _score(best_seq)

    # --------------------------------------------------------------
    # Phase 0 – fast deterministic families (power‑law, sin, etc.)
    # --------------------------------------------------------------
    p_grid = np.linspace(0.25, 0.90, 31)
    q_grid = np.linspace(0.0, 6.0, 31)

    # pure power‑law
    for p in p_grid:
        seq = _power_law_offset_seq(n, float(p), alpha=1.0, target_sum=target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # power‑law + exponential tail
    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_seq(
                n, float(p), alpha=1.0, q=float(q), r=1.0, target_sum=target_sum
            )
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # simple sine shape
    x = np.arange(n, dtype=np.float64) + 0.5
    sin_seq = np.sin(np.pi * x / (n + 1.0))
    sin_seq *= target_sum / sin_seq.sum()
    val = _score(sin_seq)
    if val < best_score:
        best_score, best_seq = val, sin_seq.copy()

    print(f"[{time.time():.2f}] After deterministic sweep best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 0b – random analytic‑family exploration (full size)
    # --------------------------------------------------------------
    search_deadline = time.time() + (deadline - time.time()) * 0.20
    seq_rand, val_rand = _random_param_search(n, target_sum, search_deadline)
    if val_rand < best_score:
        best_score, best_seq = val_rand, seq_rand.copy()
        print(f"[{time.time():.2f}] Random family search improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase A – Adam‑style gradient descent with restarts
    # --------------------------------------------------------------
    gd_deadline = time.time() + (deadline - time.time()) * 0.30
    while time.time() < gd_deadline - 5.0:
        # small jittered restart around current best
        rng = np.random.default_rng()
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _gradient_descent_with_restarts(
            start_vec, target_sum, gd_deadline
        )
        if val_opt < best_score:
            best_score, best_seq = val_opt, a_opt.copy()
            print(f"[{time.time():.2f}] Adam restart improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – Exact projected sub‑gradient descent
    # --------------------------------------------------------------
    subgrad_deadline = time.time() + (deadline - time.time()) * 0.20
    a_sg, val_sg = _subgradient_descent(best_seq, target_sum, subgrad_deadline)
    if val_sg < best_score:
        best_score, best_seq = val_sg, a_sg.copy()
        print(f"[{time.time():.2f}] Sub‑gradient descent improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase C – Deterministic mass‑transfer + occasional random moves
    # --------------------------------------------------------------
    while time.time() < deadline - 8.0:
        # deterministic greedy refinement
        a_ref, val_ref = _deterministic_mass_transfer_exact2(
            best_seq, deadline, target_sum
        )
        if val_ref < best_score:
            best_score, best_seq = val_ref, a_ref.copy()
            print(f"[{time.time():.2f}] DMT refined best to {best_score:.6f}")

        # short global random phase to escape possible plateaus
        a_rand, val_rand = _global_random_mass_transfer(
            best_seq, deadline, target_sum
        )
        if val_rand < best_score:
            best_score, best_seq = val_rand, a_rand.copy()
            print(f"[{time.time():.2f}] Random move improved best to {best_score:.6f}")

        if time.time() > deadline - 2.0:
            break

    # --------------------------------------------------------------
    # Phase D – Final polishing with an even larger soft‑max β
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(
            best_seq,
            deadline,
            target_sum,
            max_beta=1e6,
            step_factor=0.6,
            gamma=0.995,
        )
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best={best_score:.6f}")

    # --------------------------------------------------------------
    # Post‑processing – trim trailing near‑zero entries
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final evaluation (harmless if evaluate_sequence is not present)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
