    best_max = cur_max

    total_time = deadline - time.time()

    start_time = time.time()

        # linear cooling from t0 to 0.001
        T = t0 * max(0.0, 1.0 - elapsed / total_time)
        # choose donor, receiver
        # donor weighted by a (prefer large)
        if a.sum() == 0:
            break
        probs = a / a.sum()
        i = rng.choice(n, p=probs)
        # receiver weighted by inverse, but ensure not i
        # use square of (max - a) as weight
        inv = (target_sum - a) + 1e-12
        inv_probs = inv / inv.sum()
        j = rng.choice(n, p=inv_probs)

"""
Hybrid optimizer with a final simulated‑annealing mass‑transfer phase.

The algorithm proceeds in the following stages:
1) Seed generation – power‑law, exponential, random and the previous best.
2) Exploration – short annealed‑gradient runs on a moderately large problem.
3) Light deterministic mass‑transfer polishing.
4) Down‑scale → LP alternating refinement (more iterations than the baseline).
5) Upscale to a large length.
6) Sub‑gradient polishing on the large problem.
7) **Simulated‑annealing mass‑transfer** – incremental O(n) updates of the
   convolution vector, accepting moves with a temperature schedule.
8) Final reorderings, trimming and return.
All steps keep the ℓ¹‑norm fixed to √(2 n) and enforce non‑negativity.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)² (for Σa=√(2n) this equals maxconv)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised to the required ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex (Dirichlet‑like)."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread the large values uniformly by placing them on a cyclic sequence."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]          # descending
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Soft‑max gradient (for annealed GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent_v2_ext(a0: np.ndarray,
                                     deadline: float,
                                     target_sum: float,
                                     max_beta: float = 1e5,
                                     step_factor: float = 0.7,
                                     gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Momentum projected GD on the soft‑max approximation."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD_ext it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.45,
                         gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that uses the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]
        grad = np.zeros_like(a)
        for k in max_idxs:
            lo = max(0, k - (a.shape[0] - 1))
            hi = min(k, a.shape[0] - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (as used before)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy mass moves with a decaying Metropolis acceptance rule."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            it += 1
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming alternating refinement (small dimensionality)
# ----------------------------------------------------------------------


def _lp_optimize(f: np.ndarray, M: float, target_sum: float,
                 rng: np.random.Generator) -> np.ndarray | None:
    """
    Solve a LP that maximises a random linear functional of `b`
    subject to `f * b <= M` and `b >= 0`.  The result is scaled to
    have ℓ¹‑norm = target_sum.  Returns None on failure.
    """
    n = f.shape[0]
    rows = []
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=np.int64)
        row = np.zeros(n, dtype=np.float64)
        row[i] = f[k - i]
        rows.append(row)
    A = np.vstack(rows)               # (2n‑1) × n
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Random objective – maximise c·b  ⇔ minimise –c·b
    c = -rng.random(n)

    bounds = [(0.0, None)] * n

    try:
        from scipy.optimize import linprog
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception as e:
        print(f"[{time.time():.2f}] LP exception: {e}")
        return None

    if not res.success:
        print(f"[{time.time():.2f}] LP failed: {res.message}")
        return None

    b = res.x
    total = b.sum()
    if total <= 0:
        return None
    b *= target_sum / total
    return b


# ----------------------------------------------------------------------
# Simulated‑annealing mass‑transfer (exact convolution updates)
# ----------------------------------------------------------------------


def _simulated_annealing_mass_transfer(a0: np.ndarray,
                                      conv0: np.ndarray,
                                      deadline: float,
                                      target_sum: float) -> tuple[np.ndarray, float]:
    """
    Incremental SA that moves a small amount of mass between two indices,
    updating the convolution exactly (including quadratic corrections).
    Returns the best sequence seen and its max‑convolution value.
    """
    a = a0.copy()
    conv = conv0.copy()
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)
    it = 0

    # Temperature will linearly decay from 0.1*cur_max to 0.
    T0 = 0.1 * cur_max

    n = a.shape[0]

    while time.time() < deadline:
        # linear cooling
        elapsed = time.time() - start_time
        T = max(0.0, T0 * (1.0 - elapsed / total_time))

        # pick donor (large) and receiver (small)
        if a.sum() == 0:
            break
        probs = a / a.sum()
        i = rng.choice(n, p=probs)

        inv_weights = (target_sum - a) + 1e-12
        probs2 = inv_weights / inv_weights.sum()
        j = rng.choice(n, p=probs2)

        if i == j:
            continue

        # delta bounded to a fraction of the donor and of total mass
        max_delta = min(a[i] * 0.5, target_sum * 0.02)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # linear part of the convolution change
        delta_conv = np.zeros_like(conv)
        delta_conv[i:i + n] -= 2.0 * delta * a
        delta_conv[j:j + n] += 2.0 * delta * a

        # quadratic corrections (self‑terms and cross‑term)
        delta_conv[2 * i] += delta * delta
        delta_conv[2 * j] += delta * delta
        if i != j:
            delta_conv[i + j] -= 2.0 * delta * delta

        new_conv = conv + delta_conv
        new_max = new_conv.max()

        accept = False
        if new_max < cur_max:
            accept = True
        elif T > 0:
            prob = np.exp(-(new_max - cur_max) / T)
            accept = rng.random() < prob

        if accept:
            a[i] -= delta
            a[j] += delta
            conv = new_conv
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it} new best={best_max:.6f}")

        it += 1

        # every few thousand steps recompute from scratch to kill numeric drift
        if it % 5000 == 0:
            conv = _conv_fft(a)
            cur_max = conv.max()
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()

    return best_a, best_max

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:

    # ------------------------------------------------------------------
    # Phase 2 – Light deterministic mass‑transfer refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 8.0:

        )
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] After MT refinement best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – LP alternating refinement on a down‑scaled problem
    # ------------------------------------------------------------------

        n_lp = min(1500, max(800, n_expl // 5))

        seq_lp = _project_to_simplex(seq_lp, target_lp)

        lp_iter = 0
        while lp_iter < max_lp_iters and time.time() < deadline - 12.0:
            M = _max_convolution_fft(seq_lp)
            b_opt = _lp_optimize(seq_lp, M, target_lp, rng)
            if b_opt is None:
                break
            t = 0.25
            seq_lp = (1 - t) * seq_lp + t * b_opt
            seq_lp = _project_to_simplex(seq_lp, target_lp)
            cur_score = _score(seq_lp)
            print(f"[{time.time():.2f}] LP iter={lp_iter} score={cur_score:.6f}")
            lp_iter += 1

        # short GD polishing on the small problem
        if time.time() < deadline - 8.0:
            gd_dead = time.time() + min(10.0, deadline - time.time() - 6.0)
            a_opt, val_opt = _annealed_gradient_descent_v2_ext(
                seq_lp, gd_dead, target_lp,
                max_beta=1e5, step_factor=0.6, gamma=0.98,
            )
            if val_opt < _score(seq_lp):
                seq_lp = a_opt.copy()
                print(f"[{time.time():.2f}] Post‑LP GD score={val_opt:.6f}")

        # ------------------------------------------------------------------
        # Upscale to a larger size for the final polishing phases
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 90.0:

        elif remaining > 45.0:
            n_final = 12000

            n_final = n_expl

    else:

        if remaining > 90.0:
            n_final = 16000
        elif remaining > 45.0:
            n_final = 12000
        else:
            n_final = n_expl
        target_final = np.sqrt(2.0 * n_final)
        seq_big = _resize_sequence(best_seq, n_final)
        seq_big = _project_to_simplex(seq_big, target_final)

    # ------------------------------------------------------------------
    # Simple reorderings to reduce the convolution peak
    # ------------------------------------------------------------------
    candidates = [seq_big,
                  _alternating_order(seq_big),
                  _even_spread_order(seq_big)]
    seq_big = min(candidates, key=_max_convolution_fft)
    best_score = _score(seq_big)
    print(f"[{time.time():.2f}] Upscaled length={len(seq_big)} start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Sub‑gradient polishing on the large problem
    # ------------------------------------------------------------------
    if time.time() < deadline - 8.0:
        sub_dead = deadline - 4.0
        a_sub, val_sub = _subgradient_descent(
            seq_big, sub_dead, target_final,
            step_factor=0.45, gamma=0.995,
        )
        if val_sub < best_score:
            best_score = val_sub
            seq_big = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑grad polished best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – Simulated‑annealing mass‑transfer refinement (exact conv)
    # ------------------------------------------------------------------
    if time.time() < deadline - 6.0:
        conv_big = _conv_fft(seq_big)
        sa_dead = deadline - 2.0
        a_sa, max_sa = _simulated_annealing_mass_transfer(
            seq_big, conv_big, sa_dead, target_final
        )
        if max_sa < best_score:
            best_score = max_sa
            seq_big = a_sa.copy()
            print(f"[{time.time():.2f}] SA after polishing best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final deterministic mass‑transfer pass (short)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        mt_dead = time.time() + min(6.0, deadline - time.time() - 1.0)
        a_mt, val_mt = _deterministic_mass_transfer(
            seq_big, mt_dead, target_final,
            start_temp=0.25, end_temp=1e-5, low_frac=0.05,
        )
        if val_mt < best_score:
            best_score = val_mt
            seq_big = a_mt.copy()
            print(f"[{time.time():.2f}] Final MT best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # ------------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Optional final evaluation (purely for logging)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for the next round
    globals()["height_sequence_1"] = final_seq
    return final_seq
