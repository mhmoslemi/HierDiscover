import cvxpy as cp

def solve_lp_convolution(f, M):
    n = f.shape[0]
    g = cp.Variable(n)
    # Build matrix A
    rows = np.arange(2*n-1)[:, None]
    cols = np.arange(n)[None, :]
    diff = rows - cols
    A = np.where((diff >=0) & (diff < n), f[diff], 0.0)
    # A shape (2n-1, n)
    constraints = [g >= 0, A @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    if g.value is None:
        # fallback: return zeros
        return np.zeros_like(f)
    return np.maximum(g.value, 0.0)

"""
Improved step‑function optimiser.

The algorithm consists of three stages:

1. **Coarse shape optimisation on a small problem** –  
   Starting from a simple power‑law shape we perform a few
   iterations of the classic alternating LP scheme:
   given a current sequence f, compute M = max(f∗f) and solve the
   linear program   maximise ∑g   s.t.   f∗g ≤ M , g ≥ 0.
   The solution g is mixed with f (f←(1‑t)f + t g) with a
   decreasing step size t.  A small n (≈1200) keeps each LP
   cheap (a few seconds) while still capturing the essential
   shape.

2. **Upscaling & fine‑grained polishing** –  
   The best small‑scale shape is interpolated to a larger length
   (≈8000–12000).  A short projected‑gradient descent on a
   soft‑max approximation of the max‑convolution refines the
   shape, followed by a long permutation‑only simulated annealing
   that swaps indices with high/low contribution to the current
   peak.

3. **Final mass‑transfer pass & trimming** –  
   A deterministic tiny mass‑transfer step removes any remaining
   spikes and the trailing zeros are stripped for compactness.

The whole pipeline respects the 1000 s budget, leaving a small
buffer before returning the final sequence.
"""

import time
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)² (with Σa=√(2n) this reduces to max)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to length new_len."""
    old_len = seq.shape[0]
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place the largest, then the smallest, then 2nd largest, …"""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Projected gradient descent on the soft‑max objective
# (Momentum version)
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.5,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (small moves)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-4,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

# Targeted permutation search (simulated annealing on index order)
# ----------------------------------------------------------------------


def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-5,
                                 top_frac: float = 0.05,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    n = a.shape[0]
    start_time = time.time()

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        kmax_candidates = np.flatnonzero(conv == max_val)
        kmax = rng.choice(kmax_candidates)

        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)

        if donor == receiver:
            continue

        a[donor], a[receiver] = a[receiver], a[donor]

        new_score = _score(a)

        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12))

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] PermSearch new best={best_score:.6f}")
        else:
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming sub‑routine (alternating minimisation)
# ----------------------------------------------------------------------


def _conv_matrix_from_f(f: np.ndarray) -> np.ndarray:
    """Dense matrix A where (A @ g)[k] = (f ∗ g)[k]."""
    n = f.shape[0]
    rows = np.arange(2 * n - 1)[:, None]          # (2n‑1) × 1
    cols = np.arange(n)[None, :]                 # 1 × n
    diff = rows - cols                           # broadcasting, shape (2n‑1, n)
    mask = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    A[mask] = f[diff[mask]]
    return A


def _lp_max_sum_given_bound(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   maximise ∑ g   s.t.   f∗g ≤ M , g ≥ 0 .
    Returns the (non‑negative) solution g.
    """
    n = f.shape[0]
    A = _conv_matrix_from_f(f)                # (2n‑1) × n
    g = cp.Variable(n, nonneg=True)

    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    # ECOS works well for moderate dense problems
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, abstol=1e-9, reltol=1e-9)
    except Exception as e:
        print(f"LP solve exception: {e}")
        return np.zeros_like(f)

    if g.value is None:
        # fallback – return zeros (will be ignored by the caller)
        return np.zeros_like(f)
    return np.maximum(g.value, 0.0)

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The pipeline is:

      1. Small‑scale alternating LP (shape discovery)
      2. Upscale to a larger length
      3. Short soft‑max gradient descent
      4. Long permutation‑only simulated annealing
      5. Final tiny mass‑transfer pass
      6. Trim trailing zeros and return.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # 1. Small‑scale alternating LP stage
    # ------------------------------------------------------------------
    # Choose a modest size – cheap LPs but still expressive.
    if budget_s >= 800:
        n_small = 1200
    elif budget_s >= 500:
        n_small = 900
    else:
        n_small = 600
    target_small = np.sqrt(2.0 * n_small)

    # initialise with a simple power‑law shape
    f = _power_law_sequence(n_small, p=0.55, target_sum=target_small)

    print(f"[{time.time():.2f}] Starting LP stage, n={n_small}")

    # alternating minimisation parameters
    max_lp_iters = 6
    t = 0.25                       # mixing step size
    for it in range(max_lp_iters):
        if time.time() > deadline - 30:      # keep time for later phases
            break

        M = _max_convolution_fft(f)
        g = _lp_max_sum_given_bound(f, M)

        # simple convex combination
        f = (1 - t) * f + t * g
        # keep numbers non‑negative and within a reasonable range
        f = np.maximum(f, 0.0)

        cur_score = _score(f)
        print(f"[{time.time():.2f}] LP‑iter {it+1}/{max_lp_iters}, M={M:.6f}, score={cur_score:.6f}")

        # gradually shrink the mixing coefficient
        t = max(0.05, t * 0.85)

    # ------------------------------------------------------------------
    # 2. Upscale to a larger length for fine optimisation
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 180 and budget_s >= 800:
        n_big = 12000
    elif remaining > 80:
        n_big = 10000
    else:
        n_big = 8000

    target_big = np.sqrt(2.0 * n_big)
    seq_big = _resize_sequence(f, n_big)

    # Normalise to the canonical ℓ¹‑norm (purely for numeric stability)
    seq_big = seq_big * (target_big / seq_big.sum())
    print(f"[{time.time():.2f}] Upscaled to n={n_big}, initial score={_score(seq_big):.6f}")

    # ------------------------------------------------------------------
    # 3. Short high‑β gradient descent (soft‑max)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30:
        gd_deadline = min(deadline - 10, time.time() + 20)
        seq_big, best_val = _annealed_gradient_descent(
            seq_big,
            gd_deadline,
            target_big,
            max_beta=8e5,
            step_factor=0.5,
            gamma=0.99,
        )
        print(f"[{time.time():.2f}] Gradient descent refined score={best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Extended permutation‑only simulated annealing
    # ------------------------------------------------------------------
    if time.time() < deadline - 6:
        perm_deadline = deadline - 1.0
        seq_big, perm_val = _permutation_targeted_search(
            seq_big,
            perm_deadline,
            start_temp=0.5,
            end_temp=1e-5,
            top_frac=0.05,
            low_frac=0.05,
        )
        print(f"[{time.time():.2f}] Permutation search best score={perm_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Tiny deterministic mass‑transfer refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 1:
        mt_deadline = time.time() + min(4.0, deadline - time.time() - 0.5)
        seq_big, mt_val = _deterministic_mass_transfer(
            seq_big,
            mt_deadline,
            target_big,
            start_temp=0.3,
            end_temp=1e-5,
            low_frac=0.05,
        )
        print(f"[{time.time():.2f}] Final mass‑transfer best={mt_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Trim trailing zeros and return
    # ------------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # Log final score (optional)
    try:
        final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_score:.6f}")
    except Exception:
        pass

    globals()["height_sequence_1"] = final_seq
    return final_seq
