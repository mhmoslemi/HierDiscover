"""
Propose a step‑function with a low maximal self‑convolution.

Algorithm (≈ 1000 s budget)

1. **Resolution choice** – pick a fairly large n (≈ 2·10⁴) while staying
   within the time budget.  The objective after normalising Σa = √(2n)
   is exactly max(a*a), i.e. the largest entry of the linear convolution.

2. **Parametric sweep** – generate a bunch of simple families (power‑law,
   power‑law + exp‑tail, sine, cosine, Gaussian) and keep the best.

3. **Coarse optimisation** – repeatedly run Adam‑style projected GD
   (soft‑max approximation of the max) from jittered restarts.
   The schedule for the soft‑max temperature β is aggressive:
   start small, double every few hundred iterations and finish at
   β≈2·10⁶.

4. **Exact sub‑gradient descent** – move mass using the true sub‑gradient
   of the current maximal convolution entry, with a back‑tracking line
   search.

5. **Deterministic mass‑transfer (DMT)** – for the current peak index k,
   move mass from a donor index (large a_{k‑i}) to a receiver (small
   a_{k‑j}) by the analytically required amount that would bring the
   peak down to the second‑largest value.

6. **Final polishing** – a last Adam run with a very high β and a small
   step‑size.

All steps respect the global time budget, use FFT‑based convolutions and
project onto the simplex {a ≥ 0, Σa = √(2n)} after each update.
Trailing near‑zero entries are stripped before returning.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Scoring (the objective reduces to the maximal convolution entry)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Return the evaluation score – the maximum of a*a."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s."""
    # Implementation of the algorithm from
    # https://doi.org/10.2139/ssrn.3392820 (simple O(n log n) version)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for the fast pre‑search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ i^{‑p}."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(
    n: int, p: float, q: float, target_sum: float
) -> np.ndarray:
    """a_i ∝ i^{‑p}·exp(‑q·i/n)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _sine_sequence(n: int, target_sum: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+½)/(n+1))."""
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0))
    a *= target_sum / a.sum()
    return a


def _cosine_window_sequence(n: int, target_sum: float) -> np.ndarray:
    """a_i ∝ 0.5·(1‑cos(2π i/(n‑1))) – a raised‑cosine (Hann) window."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(‑(i‑mid)²/(2σ²))."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = np.exp(-((i - mid) ** 2) / (2.0 * sigma ** 2))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max based gradient (used by Adam)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max smoothed max‑convolution."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    # soft‑max weights
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]  # slice of length n
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 2e6,
    step_factor: float = 0.7,
    gamma: float = 0.995,
) -> tuple[np.ndarray, float]:
    """Adam with a soft‑max approximation of the max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 250

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow plateaus
        if it % 300 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:7.1f} score={best_val:.6f}")

        if (it % beta_inc_iter == 0) and (beta < max_beta):
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _adam_restarts(
    start_vec: np.ndarray,
    target_sum: float,
    deadline: float,
    max_beta: float = 2e6,
) -> tuple[np.ndarray, float]:
    """Run several Adam restarts (jittered starts) within the time budget."""
    best_a = start_vec.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 2.0:
        # allocate a modest chunk of time for each restart
        remaining = deadline - time.time()
        chunk = min(30.0, remaining * 0.3)
        if chunk < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        a0 = best_a + jitter
        a0 = np.maximum(a0, 0.0)
        a0 = _project_to_simplex(a0, target_sum)

        a_opt, val_opt = _adam_gradient_descent(
            a0,
            time.time() + chunk,
            target_sum,
            max_beta=max_beta,
            step_factor=0.7,
            gamma=0.995,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_a = a_opt.copy()
            print(f"[{time.time():.2f}] Adam restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iter: int = 20000,
) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on max(a*a) with back‑tracking line search."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        max_val = conv.max()
        # indices attaining the maximum (within a tolerance)
        idx_max = np.flatnonzero(conv >= max_val - 1e-12)

        # second largest value (ignore current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()
        delta_target = max_val - second_max
        if delta_target < 1e-14:
            break

        # pick a random maximizing index
        k = idx_max[rng.integers(len(idx_max))]

        n = a.shape[0]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # sub‑gradient for this k
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k - idx]

        norm_g2 = np.dot(g, g)
        if norm_g2 == 0.0:
            # degenerate – add tiny jitter and continue
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = _project_to_simplex(a + jitter, target_sum)
            it += 1
            continue

        # step size that would close the current gap (with safety factor)
        step = 0.9 * delta_target / norm_g2
        step = min(step, target_sum * 0.4 / (np.linalg.norm(g) + 1e-12))

        # simple back‑tracking line search
        for _ in range(6):
            a_cand = _project_to_simplex(a - step * g, target_sum)
            cand_score = _score(a_cand)
            if cand_score < best_score - 1e-12:
                a = a_cand
                best_a = a_cand.copy()
                best_score = cand_score
                print(
                    f"[{time.time():.2f}] Subgrad it={it:5d} step={step:.3e} score={best_score:.6f}"
                )
                break
            step *= 0.5
        else:
            # no improvement – add jitter
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = _project_to_simplex(a + jitter, target_sum)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (greedy reduction of the peak)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Greedy reduction of the maximal convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest value
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()
        if max_val - second_max < tol:
            break

        # pick the first maximal index (any works)
        k = idx_max[0]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]          # a_{k‑i}
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        # donor = index with largest partner value
        donor_local = np.argmax(partner * donor_mask)
        donor = idx[donor_local]

        # receiver = index with smallest partner value (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom

        # move a safe fraction of the donor's mass
        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] DMT it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Random global mass‑transfer (exploratory moves)
# ----------------------------------------------------------------------


def _global_random_transfer(
    a0: np.ndarray, deadline: float, target_sum: float
) -> tuple[np.ndarray, float]:
    """Tiny random moves to jump out of shallow local minima."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 0.5:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal max‑convolution.
    Returns the best sequence found (plain Python list of floats).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Choose problem size – larger n gives more degrees of freedom.
    # ------------------------------------------------------------------
    if budget_s >= 900:
        n = 20000
    elif budget_s >= 600:
        n = 16000
    elif budget_s >= 400:
        n = 12000
    else:
        n = 8000
    target_sum = np.sqrt(2.0 * n)  # Σa is fixed to this value

    # ------------------------------------------------------------------
    # Phase 0 – dense parametric pre‑search
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    # --- grids ---
    p_grid = np.linspace(0.25, 0.90, 30)
    q_grid = np.linspace(0.0, 8.0, 31)

    # (a) pure power‑law
    for p in p_grid:
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # (b) power‑law + exponential tail
    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # (c) sine, cosine, Gaussian
    for maker in (_sine_sequence, _cosine_window_sequence):
        seq = maker(n, target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # Gaussian with a few sigma values
    for sigma in (n / 6.0, n / 4.0, n / 8.0):
        seq = _gaussian_sequence(n, sigma, target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # (d) previous best if available (resized)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score = val
            best_seq = resized.copy()

    # (e) jittered local population around current best
    for _ in range(10):
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _score(cand)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()

    print(f"[{time.time():.2f}] Pre‑search best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam‑style GD with many restarts (≈30 % of budget)
    # ------------------------------------------------------------------
    phase_a_deadline = time.time() + (deadline - time.time()) * 0.30
    while time.time() < phase_a_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.015, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _adam_restarts(
            start_vec, target_sum, phase_a_deadline, max_beta=2e6
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Adam restart improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Exact projected sub‑gradient descent (≈25 % of budget)
    # ------------------------------------------------------------------
    phase_b_deadline = time.time() + (deadline - time.time()) * 0.25
    a_sg, val_sg = _subgradient_descent(
        best_seq, target_sum, phase_b_deadline, max_iter=20000
    )
    if val_sg < best_score:
        best_score = val_sg
        best_seq = a_sg.copy()
        print(f"[{time.time():.2f}] Subgradient descent improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase C – Deterministic mass‑transfer + occasional random jumps
    # ------------------------------------------------------------------
    while time.time() < deadline - 8.0:
        # deterministic reduction
        a_ref, val_ref = _deterministic_mass_transfer(
            best_seq, deadline, target_sum, tol=1e-13
        )
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] DMT improved best={best_score:.6f}")

        # short random perturbation phase
        a_rand, val_rand = _global_random_transfer(
            best_seq, deadline, target_sum
        )
        if val_rand < best_score:
            best_score = val_rand
            best_seq = a_rand.copy()
            print(f"[{time.time():.2f}] Random jump improved best={best_score:.6f}")

        if time.time() > deadline - 4.0:
            break

    # ------------------------------------------------------------------
    # Phase D – Final polishing with a higher β
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(
            best_seq,
            deadline,
            target_sum,
            max_beta=5e6,
            step_factor=0.5,
            gamma=0.995,
        )
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing – trim trailing near‑zero entries
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final evaluation (harmless if evaluate_sequence is absent)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
