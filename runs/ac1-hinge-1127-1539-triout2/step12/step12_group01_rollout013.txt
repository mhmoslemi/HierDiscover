"""
Improved heuristic for minimising the benchmark score of a non‑negative
step function.

The algorithm works in three stages:

1. **Initialisation** – generate several diverse starting sequences
   (power‑law, random exponential, warm‑start if available) and keep the
   best one according to the internal score (identical to the benchmark
   objective when Σa = √(2 n)).

2. **Multi‑restart annealing** – the bulk of the time budget is spent in
   a temperature‑controlled simulated‑annealing loop that repeatedly
   transfers a small amount of mass from a “donor’’ index to a “receiver’’
   index.  The donor is chosen proportionally to its contribution to the
   current maximal convolution entry, the receiver is favoured at low
   heights.  The temperature schedule follows a √‑decay, the step size
   scales with the temperature, and the convolution is updated in‑place
   (O(n) per move).  Periodic full recomputation of the convolution
   prevents drift.

3. **Fine‑grained sub‑gradient descent** – a short run of true
   sub‑gradient descent on the max‑convolution refines the final result.

The routine respects the supplied time budget (default 1000 s) and returns
the best sequence found as a plain list of floats.  The global variable
`height_sequence_1` is updated for warm‑starting future runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (full linear self‑convolution)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray, target_sum: float) -> float:
    """
    Objective 2·n·max(a*a) / (target_sum²).
    The optimisation constraint is Σa = target_sum.
    """
    n = a.shape[0]
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (target_sum * target_sum))


def _normalise_sum(a: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale a to have the required L1‑norm."""
    s = a.sum()
    if s == 0:
        return a
    return a * (target_sum / s)


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Simple multiplicative sub‑gradient descent on the max‑convolution."""
    a = a0.copy()
    a = _normalise_sum(a, target_sum)
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a, target_sum)

    rng = np.random.default_rng()
    start = time.time()

    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        # indices that contribute to this entry
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]          # sub‑gradient of the active peak

        # linear schedule for the step‑size
        elapsed = time.time() - start
        total = deadline - start
        eta = 2.0 * (1.0 - elapsed / total) + 0.2 * (elapsed / total)

        a = a * np.exp(-eta * g)
        a = _normalise_sum(a, target_sum)

        val = _score(a, target_sum)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={int(elapsed)} score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing with mass‑transfer moves
# ----------------------------------------------------------------------


def _anneal_mass_transfer(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         init_temp_factor: float = 1.5,
                         donor_frac: float = 0.07,
                         max_transfer_frac: float = 0.02) -> (np.ndarray, float):
    """
    Simulated annealing that moves a little mass from a donor index to a
    receiver index.  The donor is chosen proportionally to its contribution
    to the current maximal convolution entry, the receiver inversely to its
    height.  The temperature follows a sqrt‑decay, the step size scales with
    the temperature, and the convolution is updated in‑place.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                       # exact convolution a*a
    cur_val = _score(a, target_sum)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * init_temp_factor          # initial temperature

    it = 0
    while time.time() < deadline:
        # ---- temperature schedule ----
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))   # sqrt‑cooling

        # ---- locate the current convolution peak ----
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # ---- donor selection (focus on the peak) ----
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.choice(idx)

        # ---- receiver selection (prefer low heights) ----
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            it += 1
            continue

        # ---- step size (scaled by temperature) ----
        max_delta = min(a[donor] * donor_frac, target_sum * max_transfer_frac)
        if max_delta <= 0:
            it += 1
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            it += 1
            continue

        # ---- in‑place convolution update (linear‑time) ----
        # keep a copy of the old slice of a for the update
        a_old = a.copy()

        # linear part of the change
        b[donor:donor + n] -= 2.0 * delta * a_old
        b[receiver:receiver + n] += 2.0 * delta * a_old

        # quadratic part (Δ² terms)
        b[2 * donor] += delta * delta
        b[2 * receiver] += delta * delta
        if donor != receiver:
            b[donor + receiver] -= 2.0 * delta * delta

        # ---- compute new score ----
        new_val = float(2.0 * n * b.max() / (target_sum * target_sum))

        # ---- Metropolis acceptance (based on current energy) ----
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal it={it:07d} best={best_val:.6f}")
        else:
            # revert convolution update
            b[donor:donor + n] += 2.0 * delta * a_old
            b[receiver:receiver + n] -= 2.0 * delta * a_old
            b[2 * donor] -= delta * delta
            b[2 * receiver] -= delta * delta
            if donor != receiver:
                b[donor + receiver] += 2.0 * delta * delta
            # a stays unchanged

        # ---- occasional global jitter to escape plateaus ----
        if it % 2000 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _normalise_sum(a, target_sum)
            b = _conv_fft(a)
            cur_val = _score(a, target_sum)

        # ---- periodic full recomputation to avoid drift ----
        if it % 8000 == 0 and it > 0:
            a = _normalise_sum(a, target_sum)
            b = _conv_fft(a)
            cur_val = _score(a, target_sum)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute it={it} best={best_val:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.3  # safety margin

    # --------------------------------------------------------------
    # Choose problem size – a compromise between granularity and speed.
    # --------------------------------------------------------------
    if budget_s >= 900:
        n = 14000
    elif budget_s >= 600:
        n = 10000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Build a pool of diverse initial sequences.
    # --------------------------------------------------------------
    init_seqs = []

    # 1) warm‑start if previously stored
    if "height_sequence_1" in globals():
        seq = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        # linear interpolation to the new length
        old_len = seq.shape[0]
        if old_len != n:
            old_x = np.linspace(0.0, 1.0, old_len, endpoint=False)
            new_x = np.linspace(0.0, 1.0, n, endpoint=False)
            seq = np.interp(new_x, old_x, seq)
        init_seqs.append(seq)

    # 2) deterministic power‑law fall‑off (baseline)
    power_seq = np.arange(1, n + 1, dtype=np.float64) ** (-0.55)
    init_seqs.append(power_seq)

    # 3) random exponential start
    rng = np.random.default_rng()
    random_seq = rng.exponential(scale=1.0, size=n)
    init_seqs.append(random_seq)

    # --------------------------------------------------------------
    # Normalise and keep the best of the pool (using the internal score).
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    for seq in init_seqs:
        seq = _normalise_sum(seq, target_sum)
        sc = _score(seq, target_sum)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
    print(f"[{time.time():.2f}] Initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Multi‑restart annealing – split the remaining budget roughly equally.
    # --------------------------------------------------------------
    remaining_time = deadline - time.time()
    restarts = 4
    for r in range(restarts):
        if remaining_time < 30.0:   # not enough time for another full run
            break
        # allocate a slice (fair share) of the remaining budget
        slice_deadline = time.time() + remaining_time / (restarts - r)
        # jitter the start point a little to explore a different neighbourhood
        jitter = rng.normal(scale=target_sum * 0.0007, size=n)
        start_a = np.maximum(best_seq + jitter, 0.0)
        start_a = _normalise_sum(start_a, target_sum)

        print(f"[{time.time():.2f}] Restart {r + 1}/{restarts}, allocated "
              f"{slice_deadline - time.time():.1f}s")
        cand_seq, cand_score = _anneal_mass_transfer(start_a,
                                                     target_sum,
                                                     slice_deadline)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time():.2f}] New global best = {best_score:.6f}")

        remaining_time = deadline - time.time()

    # --------------------------------------------------------------
    # Short sub‑gradient fine‑tuning with whatever time is left.
    # --------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = deadline
        cand_seq, cand_score = _subgradient_descent(best_seq,
                                                    target_sum,
                                                    sub_deadline)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time():.2f}] Post‑anneal sub‑grad best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Clean‑up: truncate trailing near‑zero entries and convert to list.
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # --------------------------------------------------------------
    # Final evaluation via the provided benchmark function.
    # --------------------------------------------------------------
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for warm‑starting future runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
