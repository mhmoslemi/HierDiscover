"""
Hybrid optimization for the min‑max self‑convolution problem.

The routine consists of three stages:

1. **Coarse parametric search** – a tiny‐n (≈4000) power‑law with
   exponential tilt is explored by simulated annealing on the three
   continuous parameters (exponent `p`, decay `q` and optional offset).
   The best parameters are up‑scaled to the final problem size.

2. **Refinement on the full size** – starting from the best shape,
   a short Adam‑style projected gradient descent on the smooth soft‑max
   surrogate is performed (multiple restarts, increasing temperature).

3. **Deterministic & pairwise mass‑transfer** – two greedy procedures
   that move mass from large to small coefficients while guaranteeing
   a strict decrease of the peak convolution value.  Finally a high‑β
   Adam polish is applied.

All steps respect the global time budget (default 1000 s) and the final
sequence is projected onto the simplex Σa = √(2 n) and trimmed of
trailing near‑zero entries before returning.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective value returned by evaluate_sequence when Σa = √(2 n)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))   # equals max_b when Σa = √(2 n)


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex Σx = s, x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (power‑law with optional exponential tilt and offset)
# ----------------------------------------------------------------------


def _parametric_seq(n: int, p: float, q: float, offset: float, target_sum: float) -> np.ndarray:
    """Generate a non‑negative sequence a_i ∝ (i+1+offset)^(-p)·exp(-q·i/n) and
    project it onto the simplex Σa = target_sum."""
    i = np.arange(n, dtype=np.float64)
    base = (i + 1.0 + offset)
    # avoid non‑positive bases (offset > -1 guarantees positivity)
    base = np.maximum(base, 1e-12)
    a = base ** (-p) * np.exp(-q * i / n)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.zeros_like(a)
    else:
        a = a * (target_sum / a_sum)
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam‑style GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(conv(a,a))."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max surrogate."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(start_vec: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run a few short Adam restarts from a jittered start."""
    best_a = start_vec.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=2e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (DMT2)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest value (ignore current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # pick one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k-i}

        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # pick a receiver with minimal partner value (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.7, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Pairwise optimal mass transfer (new refinement)
# ----------------------------------------------------------------------


def _pairwise_optimal_transfer(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               donor_frac: float = 0.02,
                               receiver_frac: float = 0.02,
                               min_improve: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Greedy pairwise mass transfer.  For each iteration we pick the current peak
    of the convolution and move mass from a large coefficient (donor) to a small
    one (receiver).  The amount moved is the largest delta that keeps the peak
    index maximal.  The best donor‑receiver pair is applied.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = _score(a)

    pad_len = n - 1

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k = int(np.argmax(conv))

        nd = max(1, int(donor_frac * n))
        nr = max(1, int(receiver_frac * n))
        donor_idx = np.argsort(-a)[:nd]
        receiver_idx = np.argsort(a)[:nr]

        base = np.concatenate([np.zeros(pad_len, dtype=np.float64), a,
                               np.zeros(pad_len, dtype=np.float64)])

        # cache shifted views for the candidate indices (lightweight views)
        shifted = {}
        for idx in np.unique(np.concatenate((donor_idx, receiver_idx))):
            shifted[idx] = base[idx: idx + 2 * n - 1]

        best_new_max = max_val
        best_pair = None
        best_delta = 0.0

        for i in donor_idx:
            shift_i = shifted[i]
            a_i = a[i]
            if a_i <= 0.0:
                continue
            for j in receiver_idx:
                if i == j:
                    continue
                shift_j = shifted[j]

                d = 2.0 * (shift_j - shift_i)          # effect of moving mass from i to j
                d_k = d[k]
                if d_k >= 0.0:
                    continue    # would not lower the peak

                # constraints from other positions that would become new maxima
                mask = d > d_k
                if np.any(mask):
                    denom = d[mask] - d_k
                    numer = max_val - conv[mask]
                    feasible = numer / denom
                    delta_max = min(feasible.min(), a_i)
                else:
                    delta_max = a_i

                if delta_max <= 1e-12:
                    continue

                new_max = max_val + d_k * delta_max
                if new_max < best_new_max - min_improve:
                    best_new_max = new_max
                    best_pair = (i, j)
                    best_delta = delta_max

        if best_pair is None:
            break

        i, j = best_pair
        a[i] -= best_delta
        a[j] += best_delta

        # occasional reprojection to eliminate numerical drift
        if int(time.time() * 10) % 50 == 0:
            a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            print(f"[{time.time():.2f}] Pairwise i={i} j={j} Δ={best_delta:.3e} new_score={best_score:.6f}")

    return a, best_score

# Global random mass transfer (exploratory moves)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random new best={best_score:.6f}")

    return best_a, best_score

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

        off_new = np.clip(offset_cur + rng.normal(scale=sigma_off), -0.9, 0.5)

        val_small = _score(seq_small)

                best_val = val_large
                best_seq = seq_large.copy()
                print(f"[{time.time():.2f}] SA improved score={best_val:.6f} p={p_cur:.4f} q={q_cur:.4f} off={offset_cur:.4f}")

        # cooling / sigma decay
        temp *= 0.9995
        sigma_p *= 0.9997
        sigma_q *= 0.9997
        sigma_off *= 0.9997

    # ------------------------------------------------------------------
    # Phase A – Adam‑style projected gradient descent (with restarts)
    # ------------------------------------------------------------------
    # Allocate roughly 30 % of the remaining budget for GD restarts
    gd_deadline = time.time() + 0.30 * (deadline - time.time())
    while time.time() < gd_deadline - 5.0:
        jitter = rng.normal(scale=target_sum_large * 0.01, size=n_large)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum_large)

        a_opt, val_opt = _gradient_descent_with_restarts(start_vec,
                                                         target_sum_large,
                                                         gd_deadline)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Deterministic mass‑transfer (core refinement)
    # ------------------------------------------------------------------
    while time.time() < deadline - 12.0:
        a_ref, val_ref = _deterministic_mass_transfer_exact2(best_seq,
                                                             deadline,
                                                             target_sum_large)
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] Exact DMT2 improved best={best_val:.6f}")

        # occasional random perturbation to escape plateaus
        a_rand, val_rand = _global_random_mass_transfer(best_seq,
                                                        deadline,
                                                        target_sum_large)
        if val_rand < best_val:
            best_val = val_rand
            best_seq = a_rand.copy()
            print(f"[{time.time():.2f}] Global random improved best={best_val:.6f}")

        if time.time() > deadline - 5.0:
            break

    # ------------------------------------------------------------------
    # Phase C – Pairwise optimal mass transfer (more aggressive)
    # ------------------------------------------------------------------
    if time.time() < deadline - 4.0:
        a_pair, val_pair = _pairwise_optimal_transfer(best_seq,
                                                      deadline,
                                                      target_sum_large,
                                                      donor_frac=0.03,
                                                      receiver_frac=0.03)
        if val_pair < best_val:
            best_val = val_pair
            best_seq = a_pair.copy()
            print(f"[{time.time():.2f}] Pairwise transfer improved best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase D – Final high‑β Adam polishing
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                deadline,
                                                target_sum_large,
                                                max_beta=1e6,
                                                step_factor=0.6,
                                                gamma=0.995)
        if val_fin < best_val:
            best_val = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries and return a plain Python list
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final evaluation (harmless if evaluate_sequence is not present)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
