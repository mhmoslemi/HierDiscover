"""
Improved step‑function optimiser.

The optimiser follows the same three‑stage pipeline (exploration →
up‑scaling → polishing) but with two key extensions:
  * the exponential family is explored over a much finer range,
    including very slowly decaying sequences (λ down to 1e‑5).  These
    shapes are known to push the peak of the self‑convolution below
    the 2.0 barrier.
  * the final gradient‑descent (“soft‑max”) phase is given a longer
    budget and a higher β‑limit (2 × 10⁷) to sharpen the max‑approximation,
    yielding a tighter optimum.

All other components (FFT‑based convolution, projection onto the
ℓ¹‑simplex, deterministic mass‑transfer refinement and a long
permutation‑only annealing) are reused from the previous state‑of‑the‑art
implementation.  The routine respects the 1000 s budget, returning the
best sequence found before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a)/(∑a)² (same as evaluate_sequence when ∑a=√(2n))."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation resizing."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest‑smallest interleaving."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    l, r = 0, n - 1
    left = True
    for v in sorted_vals:
        if left:
            out[l] = v
            l += 1
        else:
            out[r] = v
            r -= 1
        left = not left
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Projected gradient descent on the soft‑max objective
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 2e7,
                               step_factor: float = 0.5,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Momentum GD on the soft‑max surrogate of max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to shake out of flat spots
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        # increase β exponentially (with a hard cap)
        if it % 300 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (small stochastic moves)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-4,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / max(deadline - start_time, 1e-12)
        temp = start_temp * (end_temp / start_temp) ** tau

        # locate the current peak index
        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]
        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted permutation simulated annealing
# ----------------------------------------------------------------------


def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-5,
                                 top_frac: float = 0.05,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Swap high‑contributing with low‑contributing indices to lower the peak."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    n = a.shape[0]
    start_time = time.time()

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        # pick a random location among all peaks to avoid deterministic bias
        peak_positions = np.flatnonzero(conv == max_val)
        kmax = rng.choice(peak_positions)

        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)

        if donor == receiver:
            continue

        a[donor], a[receiver] = a[receiver], a[donor]

        new_score = _score(a)

        elapsed = time.time() - start_time
        tau = elapsed / max(deadline - start_time, 1e-12)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] PermSearch new best={best_score:.6f}")
        else:
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score

# Main entry point required by the competition harness

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine performs:
      1. Exploration (many short gradient runs from a diverse initial pool)
      2. A brief deterministic mass‑transfer refinement
      3. Upscaling to a larger length
      4. A long high‑β gradient polishing
      5. Extended permutation‑only annealing (remaining time)
      6. Final tiny mass‑transfer pass and trimming.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # keep a tiny safety margin

    # ------------------------------------------------------------------
    # 1. Choose a working size for the exploratory phase
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n_expl = 12000
    elif budget_s >= 500:
        n_expl = 10000
    else:
        n_expl = 8000

    target_expl = np.sqrt(2.0 * n_expl)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # 2. Build a diverse pool of initial candidates
    # ------------------------------------------------------------------
    init_pool: list[np.ndarray] = []

    # Power‑law family (broad range)
    for p in np.linspace(0.2, 1.5, 13):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))

    # Exponential family – very fine λ grid (including tiny decay rates)
    for lam in np.logspace(-5, -1, 15):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))

    # Random Dirichlet‑like points
    for _ in range(10):
        a = rng.random(n_expl)
        a *= target_expl / a.sum()
        init_pool.append(a)

    # Linear decreasing shape (classic baseline)
    lin = np.linspace(1.0, 0.0, n_expl)
    lin *= target_expl / lin.sum()
    init_pool.append(lin)

    # Re‑use the previous best if it exists
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_expl))

    # Jittered copies to increase diversity
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.008, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    # ------------------------------------------------------------------
    # 3. Exploration: many short gradient‑descent runs
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # allocate roughly 60 % of the total budget to the exploration stage
    expl_deadline = start_time + 0.60 * budget_s

    while time.time() < min(deadline - 12.0, expl_deadline):
        # pick a seed – either a fresh pool element or a jittered version of the current best
        if best_seq is not None and rng.random() < 0.55:
            seed = best_seq.copy()
            seed += rng.normal(scale=target_expl * 0.004, size=n_expl)
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)

        # each GD run gets at most 12 s (or whatever is left before the exploration deadline)
        run_dead = min(time.time() + 12.0, expl_deadline, deadline - 6.0)
        if run_dead <= time.time():
            break

        a_opt, val_opt = _annealed_gradient_descent(
            seed,
            run_dead,
            target_expl,
            max_beta=2e7,
            step_factor=0.6,
            gamma=0.995,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Brief deterministic mass‑transfer refinement (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 8.0:
        mt_dead = time.time() + (deadline - time.time()) * 0.12
        a_ref, val_ref = _deterministic_mass_transfer(
            best_seq,
            mt_dead,
            target_expl,
            start_temp=0.5,
            end_temp=1e-4,
            low_frac=0.05,
        )
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] After MT refinement best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Upscaling to a larger problem size (more degrees of freedom)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.8, target_expl)

    remaining = deadline - time.time()
    # decide final length based on remaining time
    if remaining > 80.0 and budget_s >= 800:
        n_final = 20000
    elif remaining > 40.0:
        n_final = 15000
    else:
        n_final = n_expl

    target_final = np.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # test a couple of deterministic reorderings
    candidates = [seq_big,
                  _alternating_order(seq_big),
                  _reversed_alternating(seq_big)]
    best_seq = seq_big
    best_score = _score(seq_big)
    for cand in candidates[1:]:
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand
    print(f"[{time.time():.2f}] Upscaled length={n_final} start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final polishing – high‑β gradient descent (major time consumer)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        a_final, val_final = _annealed_gradient_descent(
            best_seq,
            deadline - 5.0,
            target_final,
            max_beta=2e7,
            step_factor=0.5,
            gamma=0.997,
        )
        if val_final < best_score:
            best_score = val_final
            best_seq = a_final.copy()
            print(f"[{time.time():.2f}] Final polish best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Extended targeted permutation search (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        perm_dead = deadline - 0.5   # tiny buffer for final steps
        a_perm, val_perm = _permutation_targeted_search(
            best_seq,
            perm_dead,
            start_temp=0.4,
            end_temp=1e-6,
            top_frac=0.04,
            low_frac=0.04,
        )
        if val_perm < best_score:
            best_score = val_perm
            best_seq = a_perm.copy()
            print(f"[{time.time():.2f}] After permutation search best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Tiny deterministic mass‑transfer pass (final polishing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        mt_dead = time.time() + min(5.0, deadline - time.time() - 0.4)
        a_mt2, val_mt2 = _deterministic_mass_transfer(
            best_seq,
            mt_dead,
            target_final,
            start_temp=0.3,
            end_temp=1e-5,
            low_frac=0.05,
        )
        if val_mt2 < best_score:
            best_score = val_mt2
            best_seq = a_mt2.copy()
            print(f"[{time.time():.2f}] Post‑polish MT best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 9. Trim trailing zeros for compactness
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # 10. Log final score (optional)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for possible later reuse (next round of optimisation)
    globals()["height_sequence_1"] = final_seq
    return final_seq
