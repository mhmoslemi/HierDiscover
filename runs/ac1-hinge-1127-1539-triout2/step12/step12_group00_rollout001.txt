def differential_evolution(n, target_sum, time_deadline, pop_size=80, F=0.5, CR=0.9, seed=None):

    rng = np.random.default_rng(seed)

    # initialize pop: list of np arrays
    pop = []

    # use from random simplex
    for _ in range(pop_size):
        a = rng.random(n)
        a = a / a.sum() * target_sum
        pop.append(a)

    # incorporate some deterministic seeds
    for p in np.linspace(0.2, 1.0, 5):
        pop.append(_power_law_sequence(n, p, target_sum))
    for lam in np.linspace(0.001, 0.02, 3):
        pop.append(_exponential_sequence(n, lam, target_sum))
    # ensure length maybe > pop_size, cut if needed.

    # compute scores array
    scores = np.array([_score(v) for v in pop])

    # record best
    best_idx = int(np.argmin(scores))
    best = pop[best_idx].copy()
    best_score = scores[best_idx]

    # keep best mutation at each generation
    generation = 0
    while time.time() < time_deadline:
        for i in range(len(pop)):
            # ensure time safety maybe at start of iteration break if time out.

            # mutation: sample distinct indices
            idxs = list(range(len(pop)))
            idxs.remove(i)
            a,b,c = rng.choice(idxs, size=3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            # clip and project
            mutant = np.maximum(mutant, 0.0)
            mutant = _project_to_simplex(mutant, target_sum)

            # crossover
            cross = np.empty_like(mutant)
            j_rand = rng.integers(n)
            for j in range(n):
                if rng.random() < CR or j == j_rand:
                    cross[j] = mutant[j]
                else:
                    cross[j] = pop[i][j]
            # project again
            cross = _project_to_simplex(cross, target_sum)

            # evaluate
            score_cross = _score(cross)

            # selection
            if score_cross < scores[i]:
                pop[i] = cross
                scores[i] = score_cross

                if score_cross < best_score:
                    best_score = score_cross
                    best = cross.copy()
                    print(f"[{time.time():.2f}] DE gen={generation} new best={best_score:.6f}")

        generation += 1

        # optional adaptation: adjust F or CR based on success ratio.
        # also random restart for worst individuals.
        # If many individuals stagnated, we replace them with random.
        # Let's do: after every 50 generations, identify indices with scores > median and reinitialize them random.
        if generation % 30 == 0:
            median_score = np.median(scores)
            for i in range(len(pop)):
                if scores[i] > median_score and rng.random() < 0.3:
                    # random new
                    a = rng.random(n)
                    a = a / a.sum() * target_sum
                    pop[i] = a
                    scores[i] = _score(a)

    return best, best_score

"""
Hybrid DE → Annealed GD → Mass‑Transfer optimizer.

The pipeline works on three layers:
1. **Differential Evolution (DE)** on a moderate size (≈2000).  DE is
   well‑suited for non‑convex, box‑constrained problems and quickly finds
   a good “shape” for the step function.
2. The best DE solution is **up‑scaled** to the final length (≈12‑16k)
   and refined by **annealed soft‑max gradient descent** – the same
   smooth approximation used in the reference implementation, but with
   a more aggressive temperature schedule.
3. A short **deterministic mass‑transfer** (Metropolis‑style) pass
   finishes the polishing.  Finally a few cheap reorderings (alternating,
   even‑spread and a few random permutations) are tested and the best
   one is returned.

All vectors are kept on the ℓ¹‑simplex (sum = sqrt(2·n)) and clipped to
[0,1000] by the scoring routine, so the optimizer never produces illegal
values.  The code respects the 1000 s time budget and prints progress
messages.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT based, fast for many calls)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a (linear convolution) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (Σa)².
    For Σa = √(2n) this equals max convolution.
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – ensures Σa = target and a_i ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = target}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories (used for seeding the DE population)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Simple reorderings that tend to lower the convolution peak
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, …."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread large values uniformly using a step coprime to the length."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Gradient descent on the smooth soft‑max approximation of the max‑conv
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float) -> Tuple[np.ndarray, np.ndarray]:
    """Gradient of the soft‑max of the auto‑convolution."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # convolution of reversed a with p (gives gradient of Σ p_k * (a*a)_k)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.6,
                               gamma: float = 0.98) -> Tuple[np.ndarray, float]:
    """Projected momentum GD on the soft‑max surrogate."""
    rng = np.random.default_rng()
    a = _project_to_simplex(a0.copy(), target_sum)

    best = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur = _score(a)
        if cur < best_val:
            best_val = cur
            best = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 300 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (Metropolis‑style)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.4,
                                end_temp: float = 1e-5,
                                low_frac: float = 0.07) -> Tuple[np.ndarray, float]:
    """Greedy mass moves with a decaying temperature."""
    rng = np.random.default_rng()
    a = a0.copy()
    cur_score = _score(a)
    best = a.copy()
    best_score = cur_score
    start_time = time.time()

    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        # linear cooling schedule
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # locate the current maximal convolution index
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contributions a[i] * a[kmax-i]
        prod = a[idx] * a[kmax - idx]

        # donor = index with the biggest product (most responsible for the peak)
        donor = idx[np.argmax(prod)]

        # receiver = a low‑contributing index (chosen among the worst few)
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            it += 1
            continue

        # move a small amount of mass from donor to receiver
        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Differential Evolution on a modest size (≈2000)
# ----------------------------------------------------------------------


def _de_optimize(n: int,
                 target_sum: float,
                 deadline: float,
                 pop_size: int = 80,
                 F: float = 0.6,
                 CR: float = 0.9) -> Tuple[np.ndarray, float]:
    """
    Simple DE with ℓ¹‑simplex projection.
    Returns the best vector found and its score.
    """
    rng = np.random.default_rng()
    # ---- initialise population -------------------------------------------------
    pop = []

    # a few deterministic seeds (power‑law / exponential)
    for p in np.linspace(0.2, 1.0, 4):
        pop.append(_power_law_sequence(n, float(p), target_sum))
    for lam in np.linspace(0.001, 0.02, 3):
        pop.append(_exponential_sequence(n, float(lam), target_sum))

    # random simplex points
    while len(pop) < pop_size:
        a = _random_simplex(n, target_sum, rng)
        pop.append(a)

    pop = [np.array(v, copy=False) for v in pop]
    scores = np.array([_score(v) for v in pop], dtype=np.float64)

    best_idx = int(np.argmin(scores))
    best_vec = pop[best_idx].copy()
    best_score = scores[best_idx]

    generation = 0
    while time.time() < deadline:
        for i in range(len(pop)):
            # make sure we still have time for the evaluation
            if time.time() >= deadline:
                break

            # ----- mutation ----------------------------------------------------
            idxs = list(range(len(pop)))
            idxs.remove(i)
            a_idx, b_idx, c_idx = rng.choice(idxs, size=3, replace=False)
            mutant = pop[a_idx] + F * (pop[b_idx] - pop[c_idx])
            mutant = np.maximum(mutant, 0.0)
            mutant = _project_to_simplex(mutant, target_sum)

            # ----- crossover ---------------------------------------------------
            j_rand = rng.integers(n)
            trial = np.empty_like(mutant)
            for j in range(n):
                if rng.random() < CR or j == j_rand:
                    trial[j] = mutant[j]
                else:
                    trial[j] = pop[i][j]
            trial = _project_to_simplex(trial, target_sum)

            # ----- selection ---------------------------------------------------
            trial_score = _score(trial)
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_vec = trial.copy()
                    print(f"[{time.time():.2f}] DE gen={generation} new best={best_score:.6f}")

        generation += 1

        # occasional refresh of stagnating individuals
        if generation % 20 == 0:
            median = np.median(scores)
            for i in range(len(pop)):
                if scores[i] > median and rng.random() < 0.3:
                    a = _random_simplex(n, target_sum, rng)
                    pop[i] = a
                    scores[i] = _score(a)

    return best_vec, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step function with a small evaluation score.
    The routine runs a DE phase on a medium size, up‑scales, refines by
    annealed GD, does a short mass‑transfer pass and finally tries a few
    reorderings.  The best sequence found before the deadline is returned.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # 1️⃣  Differential Evolution on a modest problem size
    # ------------------------------------------------------------------
    n_de = 2000
    target_de = np.sqrt(2.0 * n_de)
    de_dead = min(deadline - 30.0, time.time() + 300.0)   # at most ~5 min for DE
    best_small, best_small_score = _de_optimize(
        n_de, target_de, de_dead, pop_size=80, F=0.6, CR=0.9
    )
    print(f"[{time.time():.2f}] DE completed, score={best_small_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Upscale to a larger size for fine polishing
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 120.0:
        n_final = 16000
    elif remaining > 60.0:
        n_final = 12000
    else:
        n_final = 8000
    target_final = np.sqrt(2.0 * n_final)

    seq = _resize_sequence(best_small, n_final)
    seq = _project_to_simplex(seq, target_final)

    # ------------------------------------------------------------------
    # 3️⃣  Annealed gradient descent (soft‑max) on the large problem
    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0:
        gd_dead = time.time() + min(30.0, deadline - time.time() - 12.0)
        seq, gd_score = _annealed_gradient_descent(
            seq, gd_dead, target_final,
            max_beta=1e6, step_factor=0.5, gamma=0.98
        )
        print(f"[{time.time():.2f}] GD post‑upscale score={gd_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Short deterministic mass‑transfer polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        mt_dead = time.time() + min(6.0, deadline - time.time() - 1.0)
        seq, mt_score = _deterministic_mass_transfer(
            seq, mt_dead, target_final,
            start_temp=0.3, end_temp=1e-5, low_frac=0.07
        )
        print(f"[{time.time():.2f}] MT polishing score={mt_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Reordering / simple linear combinations
    # ------------------------------------------------------------------
    candidates = [
        seq,
        _alternating_order(seq),
        _even_spread_order(seq),
    ]

    # a few random permutations – cheap but occasionally helpful
    rng = np.random.default_rng()
    for _ in range(4):
        perm = rng.permutation(len(seq))
        candidates.append(seq[perm])

    # try mixing with the reversed version (conv is symmetric, this
    # sometimes smooths the peak)
    rev = seq[::-1]
    for t in np.linspace(0.0, 0.5, 6):
        mixed = (1.0 - t) * seq + t * rev
        mixed = _project_to_simplex(mixed, target_final)
        candidates.append(mixed)

    seq = min(candidates, key=_max_convolution_fft)
    best_score = _score(seq)
    print(f"[{time.time():.2f}] After reordering score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Trim trailing near‑zero entries for compactness
    # ------------------------------------------------------------------
    arr = np.array(seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # 7️⃣  Final evaluation (purely for logging – not required for the harness)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose the result for possible warm‑start in a next run
    globals()["height_sequence_1"] = final_seq
    return final_seq
