"""
Hybrid optimizer – large‑scale simulated‑annealing refinement.

The sequence is first taken from the previous best (global
``height_sequence_1``) or generated afresh, projected onto the ℓ¹‑simplex
with   Σa = √(2 n).  The vector is up‑scaled to a larger length (≈30 k)
by linear interpolation, then a fast incremental simulated‑annealing
search (`_sa_mass_transfer`) moves tiny amounts of mass from the indices
that contribute most to the current convolution peak toward indices that
contribute little.  Each move updates the full self‑convolution with O(n)
slicing (no FFT per move), and the peak value is kept as the objective.
The temperature decays exponentially; uphill moves are accepted with a
Metropolis probability.  The best sequence found during the run is stored
and finally trimmed of trailing near‑zero entries.

The method works purely on the shape of the step function – scaling does
not affect the evaluation score, so the ℓ¹‑normalisation is only a
convenient way to keep the numbers bounded.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# FFT utilities – used only for the initial convolution and occasional
# full recomputations (e.g. after up‑scaling).
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (fast for the initial evaluation)."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Simplex projection (ℓ¹‑norm = s, non‑negative)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple sequence factories (used only for first‑time start)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _interpolate(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to size `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, … (helps to spread)."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


# ----------------------------------------------------------------------
# Incremental simulated‑annealing mass‑transfer
# ----------------------------------------------------------------------


def _sa_mass_transfer(a: np.ndarray,
                      deadline: float,
                      start_temp: float = 0.5,
                      end_temp: float = 1e-5,
                      low_frac: float = 0.06,
                      jitter_prob: float = 0.001) -> tuple[np.ndarray, float]:
    """
    Fast SA that keeps the full self‑convolution in a mutable array.
    Each move transfers a small δ from a high‑contributing index
    (donor) to a low‑contributing index (receiver) for the current peak.
    The convolution update is O(n) via slicing; the move can be accepted,
    rejected or accepted with Metropolis probability.
    Returns the best sequence seen and its max‑convolution value.
    """
    n = a.shape[0]
    conv = _conv_fft(a)                 # current full convolution
    cur_max = float(conv.max())
    best_a = a.copy()
    best_score = cur_max

    rng = np.random.default_rng()
    start = time.time()

    while time.time() < deadline:
        # ----- temperature schedule -----
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        # ----- locate the current peak -----
        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # product contribution a[i] * a[kmax-i] for all i in the interval
        prod = a[idx] * a[kmax - idx]

        # donor = index with maximal contribution
        donor = int(idx[int(np.argmax(prod))])

        # receiver = one of the low‑contributing indices (within the same
        # interval, low_frac fraction)
        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argpartition(prod, low_cnt - 1)[:low_cnt]]
        receiver = int(rng.choice(low_idxs))

        if donor == receiver:
            continue

        # ----- propose a small transfer -----
        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta

        # keep old state for possible rollback
        old_conv = conv.copy()
        old_donor_val = a[donor]
        old_receiver_val = a[receiver]

        # ----- incremental convolution update (using the *old* a values) -----
        conv[donor:donor + n] -= 2.0 * delta * a
        conv[receiver:receiver + n] += 2.0 * delta * a
        conv[2 * donor] += delta * delta
        conv[2 * receiver] += delta * delta
        conv[donor + receiver] -= 2.0 * delta * delta

        # ----- apply the mass move -----
        a[donor] = old_donor_val - delta
        a[receiver] = old_receiver_val + delta

        # Numerical safety
        a = np.maximum(a, 0.0)

        new_max = float(conv.max())

        # ----- acceptance test -----
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob = math.exp(-(new_max - cur_max) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_max = new_max
            if new_max < best_score:
                best_score = new_max
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA new best max={best_score:.6f}")
        else:
            # rollback
            conv = old_conv
            a[donor] = old_donor_val
            a[receiver] = old_receiver_val

        # occasional global jitter to break plateaus
        if rng.random() < jitter_prob:
            jitter = rng.normal(scale=cur_max * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            # recompute full convolution after a jitter (costly but rare)
            conv = _conv_fft(a)
            cur_max = float(conv.max())
            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA jitter improved max={best_score:.6f}")

    return best_a, best_score

# Main entry point for the competition harness

def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step function with a low benchmark score.
    The routine follows a simple pipeline:
      1. Load the previous best (global ``height_sequence_1``) or generate a
         power‑law seed.
      2. Upscale to a larger length (≈30 k) and project onto the ℓ¹‑simplex.
      3. Apply a fast simulated‑annealing mass‑transfer refinement that
         updates the self‑convolution incrementally.
      4. Trim trailing near‑zero entries and return the final list.
    The whole process respects the 1000 s time budget.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # 1. Initialise a seed sequence
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        seed = np.array(globals()["height_sequence_1"], dtype=np.float64)
        n0 = seed.shape[0]
        print(f"[{time.time():.2f}] Loaded previous sequence of length {n0}")
    else:
        n0 = 5000
        target0 = math.sqrt(2.0 * n0)
        seed = _power_law_sequence(n0, p=0.5, target_sum=target0)
        print(f"[{time.time():.2f}] Generated fresh power‑law seed (n={n0})")

    # ensure the seed satisfies the simplex constraint
    target0 = math.sqrt(2.0 * n0)
    seed = _project_to_simplex(seed, target0)

    # --------------------------------------------------------------
    # 2. Upscale to a larger problem size
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    # decide final size based on remaining time
    if remaining > 200:
        n_final = 30000
    elif remaining > 120:
        n_final = 20000
    elif remaining > 60:
        n_final = 15000
    else:
        n_final = n0

    target_final = math.sqrt(2.0 * n_final)

    # linear interpolation and simplex projection
    a = _interpolate(seed, n_final)
    a = _project_to_simplex(a, target_final)

    # a simple deterministic reordering that often lowers the max‑conv
    a = _alternating_order(a)

    print(f"[{time.time():.2f}] Upscaled to n={n_final}, initial max={_max_convolution_fft(a):.6f}")

    # --------------------------------------------------------------
    # 3. Simulated‑annealing mass‑transfer refinement (major work)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sa_deadline = deadline - 2.0
        best_a, best_score = _sa_mass_transfer(
            a,
            deadline=sa_deadline,
            start_temp=0.5,
            end_temp=1e-5,
            low_frac=0.06,
            jitter_prob=0.001,
        )
        a = best_a
        print(f"[{time.time():.2f}] SA finished, best max={best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Trim trailing near‑zero entries for compactness
    # --------------------------------------------------------------
    nz = np.nonzero(a > 1e-12)[0]
    if nz.size:
        a = a[: nz[-1] + 1]

    final_seq = a.tolist()

    # optional evaluation for logging (ignore exceptions)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for possible next runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
