def _random_local_search(a, deadline, target_sum):
    best = a.copy()
    best_score = _score(best)
    rng = np.random.default_rng()
    while time.time() < deadline:
        # choose random subset
        delta = rng.normal(scale=target_sum*0.01, size=best.shape)
        cand = best + delta
        cand = np.maximum(cand, 0)
        cand = _project_to_simplex(cand, target_sum)
        val = _score(cand)
        if val < best_score:
            best, best_score = cand, val
    return best, best_score

"""
Hybrid optimisation for the non‑negative step‑function problem.

Main ideas
----------
* Use a fairly small length (≈4‑5k) to explore many random seeds quickly with
  projected sub‑gradient descent on the exact max‑convolution.
* Keep the best seed, then upscale it (linear interpolation) to a larger
  length (≈12‑20k) where the additional degrees of freedom allow a flatter
  autocorrelation.
* On the large vector run a *long* projected sub‑gradient descent (hundreds
  of seconds) followed by a high‑β soft‑max gradient descent – both use
  momentum and occasional jitter to escape plateaus.
* Finish with cheap permutation‑search, deterministic mass‑transfer and a
  final trim of trailing zeros.
* All phases respect the global time budget (default 1000 s).  The best
  sequence found is stored in the global variable ``height_sequence_1`` for
  reuse by later runs.

The objective after normalisation (∑a = √(2n)) equals the peak of the
self‑convolution, so the optimisation directly minimises that peak.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – keeps ∑a = target and a_i ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple families of seed sequences
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_dirichlet(n: int, target_sum: float) -> np.ndarray:
    a = np.random.random(n)
    a *= target_sum / a.sum()
    return a


def _linear_decreasing(n: int, target_sum: float) -> np.ndarray:
    a = np.linspace(1.0, 0.0, n, dtype=np.float64)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation from old length to new_len."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Scoring utility (objective = max convolution entry)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Return the current objective value (peak of a*a)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))   # = max_b when s = √(2n)


# ----------------------------------------------------------------------
# Exact max‑convolution sub‑gradient
# ----------------------------------------------------------------------


def _max_conv_and_subgrad(a: np.ndarray):
    """
    Returns (conv, max_val, kmax, subgrad) where subgrad is the gradient of
    the peak entry of the convolution w.r.t. a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    kmax = int(np.argmax(conv))
    max_val = conv[kmax]

    grad = np.zeros_like(a)
    lo = max(0, kmax - (n - 1))
    hi = min(kmax, n - 1)
    idx = np.arange(lo, hi + 1)
    partner = a[kmax - idx]
    grad[lo:hi + 1] = 2.0 * partner
    return conv, max_val, kmax, grad


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.6,
                         gamma: float = 0.98) -> tuple[np.ndarray, float]:
    """
    Momentum projected sub‑gradient descent on max(conv(a,a)).
    Returns the best vector seen and its score.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0
    momentum = np.zeros_like(a)

    while time.time() < deadline:
        _, _, _, grad = _max_conv_and_subgrad(a)

        # momentum update
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_factor * target_sum / norm
            a -= lr * momentum
        else:
            a -= step_factor * target_sum * grad / (np.linalg.norm(grad) + 1e-12)

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] SubGD it={it:5d} score={best_val:.6f}")

        it += 1
    return best, best_val


# ----------------------------------------------------------------------
# Soft‑max (smooth max) gradient descent – annealed β
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(conv)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    grad = 2.0 * np.fft.irfft(np.fft.rfft(a_rev, n=size) *
                               np.fft.rfft(p, n=size), n=size)[: 2 * n - 1]
    grad = grad[n - 1: n - 1 + n]          # central n entries
    return grad, b


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e6,
                               step_factor: float = 0.5,
                               gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max approximation.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # jitter on the schedule
        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SoftGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        # increase β geometrically every few iterations
        if it % 300 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Targeted permutation search – only reorders the multiset of heights
# ----------------------------------------------------------------------


def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.12,
                                 end_temp: float = 1e-5,
                                 top_frac: float = 0.04,
                                 low_frac: float = 0.04) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    n = a.shape[0]
    start_time = time.time()

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # contribution of each position to the peak
        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)

        if donor == receiver:
            continue

        a[donor], a[receiver] = a[receiver], a[donor]

        new_score = _score(a)

        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] PermSearch new best={best_score:.6f}")

        # Metropolis acceptance
        if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / max(temp, 1e-12)):
            pass
        else:
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (simple hill‑climb)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.4,
                                end_temp: float = 1e-5,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution weight of each index to the current peak
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]
        # pick a low‑contributing receiver among the same neighbourhood
        low_cnt = max(1, int(len(idx) * low_frac))
        receiver = rng.choice(idx[np.argsort(w)[:low_cnt]])

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.4            # move at most 40 % of donor
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        new_score = _score(a)

        # accept if improves else accept with Boltzmann probability
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with a low peak of its
    self‑convolution.  The routine respects the time budget, stores the best
    vector in the global ``height_sequence_1`` and returns it as a plain
    Python list.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # 1. Choose an exploration length (fast to evaluate)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n_expl = 5000
    elif budget_s >= 500:
        n_expl = 4000
    else:
        n_expl = 3000
    target_expl = np.sqrt(2.0 * n_expl)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # 2. Build a pool of diverse seed vectors
    # ------------------------------------------------------------------
    pool = []

    for p in np.linspace(0.35, 0.75, 7):
        pool.append(_power_law_sequence(n_expl, float(p), target_expl))
    for lam in np.linspace(0.0015, 0.012, 5):
        pool.append(_exponential_sequence(n_expl, float(lam), target_expl))
    for _ in range(5):
        pool.append(_random_dirichlet(n_expl, target_expl))
    pool.append(_linear_decreasing(n_expl, target_expl))

    # reuse previous best if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        pool.append(_resize_sequence(prev, n_expl))

    # jittered copies for extra diversity
    for seq in list(pool):
        jitter = rng.normal(scale=target_expl * 0.008, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        pool.append(seq_j)

    # ------------------------------------------------------------------
    # 3. Quick sub‑gradient runs on the moderate size (exploration)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # allocate roughly half of the total budget to this phase
    expl_deadline = deadline - 250.0  # keep at least ~250 s for later stages
    while time.time() < expl_deadline and len(pool) > 0:
        seed = rng.choice(pool)
        run_dead = min(time.time() + 15.0, deadline - 120.0)   # ~15 s per run
        if run_dead <= time.time():
            break
        a_opt, val_opt = _subgradient_descent(seed, run_dead, target_expl,
                                              step_factor=0.6, gamma=0.98)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best score={best_score:.6f}")

    # fallback if nothing survived
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.55, target_expl)

    # ------------------------------------------------------------------
    # 4. Upscale to a larger length for extra freedom
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 200 and budget_s >= 700:
        n_big = 18000
    elif remaining > 120:
        n_big = 14000
    else:
        n_big = n_expl

    target_big = np.sqrt(2.0 * n_big)
    seq_big = _resize_sequence(best_seq, n_big)
    seq_big = _project_to_simplex(seq_big, target_big)

    # initial reordering heuristics
    candidates = [seq_big,
                  # alternating order often reduces the central peak
                  np.concatenate([seq_big[np.argsort(-seq_big)][:n_big // 2],
                                 seq_big[np.argsort(-seq_big)][n_big // 2:][::-1]])]
    best_seq = seq_big
    best_score = _score(seq_big)
    for cand in candidates[1:]:
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand
    print(f"[{time.time():.2f}] Upscaled to n={n_big}, start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Long projected sub‑gradient descent on the big vector
    # ------------------------------------------------------------------
    if time.time() < deadline - 120.0:
        sub_dead = time.time() + min(110.0, deadline - time.time() - 30.0)
        a_opt, val_opt = _subgradient_descent(best_seq, sub_dead,
                                              target_big,
                                              step_factor=0.6,
                                              gamma=0.985)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] SubGD (big) improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. High‑β soft‑max gradient descent (annealed)
    # ------------------------------------------------------------------
    if time.time() < deadline - 80.0:
        soft_dead = deadline - 30.0
        a_opt, val_opt = _annealed_gradient_descent(best_seq, soft_dead,
                                                    target_big,
                                                    max_beta=5e6,
                                                    step_factor=0.5,
                                                    gamma=0.995)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Softmax GD best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Permutation‑only simulated annealing
    # ------------------------------------------------------------------
    if time.time() < deadline - 40.0:
        perm_dead = time.time() + min(35.0, deadline - time.time() - 5.0)
        a_opt, val_opt = _permutation_targeted_search(best_seq, perm_dead,
                                                      start_temp=0.15,
                                                      end_temp=1e-5,
                                                      top_frac=0.04,
                                                      low_frac=0.04)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Permutation search best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Deterministic mass‑transfer fine polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        mt_dead = time.time() + min(8.0, deadline - time.time() - 2.0)
        a_opt, val_opt = _deterministic_mass_transfer(best_seq, mt_dead,
                                                      target_big,
                                                      start_temp=0.3,
                                                      end_temp=1e-5,
                                                      low_frac=0.05)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Mass‑transfer best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 9. Trim trailing zeros and final logging
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # optional final evaluation for our own information
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    globals()["height_sequence_1"] = final_seq
    return final_seq
