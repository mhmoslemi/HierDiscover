def permutation_anneal(arr, deadline, start_temp, end_temp, top_frac, low_frac, max_try=5):
    a = arr.copy()
    best = a.copy()
    best_score = score(a)
    n = len(a)
    rng = np.random.default_rng()
    start = time.time()
    while time.time() < deadline:
        # compute convolution and peak
        conv = _conv_fft(a) # shape 2n-1
        max_val = conv.max()
        # indices for kmax (could be multiple)
        kmax_indices = np.where(conv == max_val)[0]
        kmax = rng.choice(kmax_indices)
        # compute contributions
        # For each i in 0..n-1:
        partner = kmax - np.arange(n)
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[mask] * a[partner[mask]]
        # donor candidates: top "top_frac" by contrib.
        top_k = max(2, int(top_frac*n))
        donor_indices = np.argpartition(-contrib, top_k-1)[:top_k]
        # low contributors:
        low_k = max(2, int(low_frac*n))
        receiver_candidates = np.argpartition(contrib, low_k-1)[:low_k]
        # random sample of donors & receivers:
        # iterate over limited combos:
        best_local_score = best_score
        best_swap = None
        for donor in donor_indices:
            for receiver in rng.choice(receiver_candidates, size=min(len(receiver_candidates), max_try), replace=False):
                if donor == receiver: continue
                # swap and compute new max efficiently
                a[donor], a[receiver] = a[receiver], a[donor]
                new_score = _score(a)
                if new_score < best_local_score:
                    best_local_score = new_score
                    best_swap = (donor, receiver)
                # revert
                a[donor], a[receiver] = a[receiver], a[donor]
        # temperature schedule
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau
        if best_swap is not None:
            # perform best swap
            d, r = best_swap
            a[d], a[r] = a[r], a[d]
            cur_score = best_local_score
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(...)
        else:
            # random swap
            i,j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            new_score = _score(a)
            if new_score < best_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
                # accept
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best = a.copy()
            else:
                # revert
                a[i], a[j] = a[j], a[i]
    return best, best_score

"""
Improved step‑function optimiser.

Main ideas
----------
* A richer pool of initial shapes (power‑law, exponential, log, linear,
  random Dirichlet).
* Longer, higher‑β gradient‑descent runs on the smooth soft‑max
  approximation of max‑convolution.
* Two deterministic “mass‑transfer” refinements (before and after
  permutation).
* A more aggressive permutation annealing: for each peak we examine a
  small set of promising donor/receiver swaps (instead of a single
  random swap) and keep the best improving move.
* After the permutation phase we run one extra high‑β gradient pass.
* Finally we trim trailing zeros and return the best sequence seen.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Objective: 2·n·max(a*a) / (Σa)².
    The function works for any non‑negative a; it returns np.inf for a
    sum that is too small.
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _log_sequence(n: int, target_sum: float) -> np.ndarray:
    # 1 / log(i+e) shape – heavy tail but not too steep
    a = 1.0 / np.log(np.arange(1, n + 1, dtype=np.float64) + np.e)
    a *= target_sum / a.sum()
    return a


def _linear_decreasing(n: int, target_sum: float) -> np.ndarray:
    a = np.linspace(1.0, 0.0, n, dtype=np.float64)
    a *= target_sum / a.sum()
    return a


def _random_dirichlet(n: int, target_sum: float, alpha: float = 0.8) -> np.ndarray:
    rng = np.random.default_rng()
    a = rng.gamma(alpha, 1.0, size=n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place the largest, smallest, second‑largest, second‑smallest, …"""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth surrogate for max‑convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns the gradient of the soft‑max approximation of
    max(convolution) together with the raw convolution values.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * (a reversed) * p   (convolution with reversed a)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Annealed gradient descent on the soft‑max objective
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.5,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Momentum‑based projected gradient descent on a smooth approximation
    of the max‑convolution.  Beta grows exponentially up to `max_beta`,
    gradually turning the smooth surrogate into the true max.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        # increase β exponentially, but cap at max_beta
        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (small moves)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-4,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Small‑scale mass redistribution guided by the current peak of the
    convolution.  The algorithm moves a fraction of mass from a “donor”
    (high contribution) to a “receiver” (low contribution) using a
    simulated‑annealing acceptance rule.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # find the position of the current peak
        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # total contribution of each index to the peak diagonal
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Enhanced permutation search (targeted simulated annealing)
# ----------------------------------------------------------------------


def _enhanced_permutation_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-5,
                                 top_frac: float = 0.06,
                                 low_frac: float = 0.06,
                                 trial_per_move: int = 7) -> tuple[np.ndarray, float]:
    """
    Simulated annealing on the index order.
    For every peak we examine a small set of promising swaps (donor from
    the top‑fraction of contributors, receiver from the low‑fraction)
    and keep the best improving one.  If none improve, we still accept
    a random swap with the usual Metropolis probability.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    n = a.shape[0]
    start_time = time.time()

    while time.time() < deadline:
        # recompute convolution and locate the current peak
        conv = _conv_fft(a)
        max_val = conv.max()
        kmax_candidates = np.flatnonzero(conv == max_val)
        kmax = rng.choice(kmax_candidates)

        # contribution of each index to that peak
        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        # donor / receiver candidate sets
        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))
        donor_cands = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_cands = np.argpartition(contrib, low_k - 1)[:low_k]

        # try a handful of swaps and keep the best
        best_local_score = cur_score
        best_swap = None

        for donor in donor_cands:
            # sample a few receivers (without replacement)
            recv_sample = rng.choice(receiver_cands,
                                     size=min(trial_per_move, len(receiver_cands)),
                                     replace=False)
            for receiver in recv_sample:
                if donor == receiver:
                    continue
                a[donor], a[receiver] = a[receiver], a[donor]
                new_score = _score(a)
                if new_score < best_local_score:
                    best_local_score = new_score
                    best_swap = (donor, receiver)
                a[donor], a[receiver] = a[receiver], a[donor]   # revert

        # temperature schedule (slow exponential decay)
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        if best_swap is not None:
            # perform the improving swap
            d, r = best_swap
            a[d], a[r] = a[r], a[d]
            cur_score = best_local_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] PermSearch improve to {best_score:.6f}")
        else:
            # fallback: random swap with Metropolis acceptance
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            new_score = _score(a)
            if new_score < cur_score:
                accept = True
            else:
                accept = rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if accept:
                cur_score = new_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] PermSearch random improve to {best_score:.6f}")
            else:
                # revert
                a[i], a[j] = a[j], a[i]

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine combines:
      * diverse initial families,
      * several long gradient‑descent runs,
      * a deterministic mass‑transfer refinement,
      * up‑scaling,
      * a final high‑β gradient polishing,
      * an enhanced permutation annealing,
      * a last polishing pass.
    The best sequence found before the deadline is returned.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.6   # safety buffer

    # ------------------------------------------------------------------
    # 1. Choose an exploration length
    # ------------------------------------------------------------------
    if budget_s >= 900:
        n_expl = 9000
    elif budget_s >= 700:
        n_expl = 8000
    elif budget_s >= 500:
        n_expl = 7000
    else:
        n_expl = 6000
    target_expl = np.sqrt(2.0 * n_expl)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # 2. Build a diverse pool of initial candidates
    # ------------------------------------------------------------------
    init_pool: list[np.ndarray] = []

    # power‑law family
    for p in np.linspace(0.30, 0.80, 9):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))

    # exponential family
    for lam in np.linspace(0.001, 0.020, 7):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))

    # log‑type tail
    init_pool.append(_log_sequence(n_expl, target_expl))

    # linear decreasing
    init_pool.append(_linear_decreasing(n_expl, target_expl))

    # random Dirichlet draws (different concentration)
    for a in [0.5, 0.8, 1.2]:
        for _ in range(2):
            init_pool.append(_random_dirichlet(n_expl, target_expl, alpha=a))

    # reuse previous best if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_expl))

    # jittered copies to increase diversity
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.009, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    # ------------------------------------------------------------------
    # 3. Exploration: many gradient‑descent runs (short but many)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    while time.time() < deadline - 12.0:          # reserve time for later phases
        # pick a seed – sometimes jitter around the current best
        if best_seq is not None and rng.random() < 0.6:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.004, size=n_expl)
            seed = seed + jitter
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)

        # run a bounded GD instance (≈18 s each)
        run_dead = min(time.time() + 18.0, deadline - 10.0)
        if run_dead <= time.time():
            break

        a_opt, val_opt = _annealed_gradient_descent(
            seed,
            run_dead,
            target_expl,
            max_beta=5e5,
            step_factor=0.6,
            gamma=0.99,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. First deterministic mass‑transfer refinement (if time left)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 10.0:
        mt_dead = time.time() + (deadline - time.time()) * 0.10
        a_ref, val_ref = _deterministic_mass_transfer(
            best_seq,
            mt_dead,
            target_expl,
            start_temp=0.5,
            end_temp=1e-4,
            low_frac=0.05,
        )
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] After MT refinement best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Upscaling to a larger problem size (more degrees of freedom)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.55, target_expl)

    remaining = deadline - time.time()
    # Choose a larger length if we have enough time left.
    if remaining > 70.0 and budget_s >= 800:
        n_final = 13000
    elif remaining > 45.0:
        n_final = 11000
    else:
        n_final = n_expl

    target_final = np.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # try a few deterministic reorderings
    candidates = [
        seq_big,
        _alternating_order(seq_big),
        _reversed_alternating(seq_big),
    ]
    best_seq = seq_big
    best_score = _score(seq_big)
    for cand in candidates[1:]:
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand
    print(f"[{time.time():.2f}] Upscaled to {n_final} elems, start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. High‑β gradient polishing (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 15.0:
        a_final, val_final = _annealed_gradient_descent(
            best_seq,
            deadline - 7.0,
            target_final,
            max_beta=2e6,
            step_factor=0.5,
            gamma=0.995,
        )
        if val_final < best_score:
            best_score = val_final
            best_seq = a_final.copy()
            print(f"[{time.time():.2f}] High‑β polish best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Enhanced targeted permutation search (uses most of the remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 4.0:
        perm_dead = deadline - 1.0
        a_perm, val_perm = _enhanced_permutation_search(
            best_seq,
            perm_dead,
            start_temp=0.6,
            end_temp=1e-5,
            top_frac=0.06,
            low_frac=0.06,
            trial_per_move=9,
        )
        if val_perm < best_score:
            best_score = val_perm
            best_seq = a_perm.copy()
            print(f"[{time.time():.2f}] After enhanced permutation best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Final polishing gradient (short run) and tiny mass‑transfer
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.5:
        # short gradient to clean up after the swaps
        final_dead = time.time() + min(4.5, deadline - time.time() - 0.3)
        a_last, val_last = _annealed_gradient_descent(
            best_seq,
            final_dead,
            target_final,
            max_beta=1e6,
            step_factor=0.45,
            gamma=0.99,
        )
        if val_last < best_score:
            best_score = val_last
            best_seq = a_last.copy()
            print(f"[{time.time():.2f}] Final gradient polish best={best_score:.6f}")

    if time.time() < deadline - 1.0:
        mt_dead = time.time() + min(4.0, deadline - time.time() - 0.2)
        a_mt2, val_mt2 = _deterministic_mass_transfer(
            best_seq,
            mt_dead,
            target_final,
            start_temp=0.3,
            end_temp=1e-5,
            low_frac=0.05,
        )
        if val_mt2 < best_score:
            best_score = val_mt2
            best_seq = a_mt2.copy()
            print(f"[{time.time():.2f}] Last MT pass best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 9. Trim trailing near‑zero entries (compactness)
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # 10. Final evaluation (optional, for logging)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # Store globally for possible reuse in later calls
    globals()["height_sequence_1"] = final_seq
    return final_seq
