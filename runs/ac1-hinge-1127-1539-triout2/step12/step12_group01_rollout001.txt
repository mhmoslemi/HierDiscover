"""
Propose a low‑max‑convolution step function.

The optimizer consists of three stages designed to work well within a
1000 s budget:

1. **Multiple warm‑starts** – generate a few plausible initial sequences
   (resized previous best, several power‑law decays and a random Dirichlet).
2. **Additive projected sub‑gradient descent** – minimise the true
   `max(a*a)` while keeping the ℓ₁‑norm fixed (∑a = √(2 n)).  The method
   uses the exact sub‑gradient of the active convolution peak, a
   diminishing step‑size `η_t = 0.5·S/(√t·‖g‖)` and a cheap projection
   (clip + renormalise) onto the simplex.  This converges reliably for
   the convex objective.
3. **Aggressive annealing refinement** – a Metropolis‑style mass‑transfer
   move that updates the full convolution in O(n).  Temperature starts
   at 0.9·current score and follows a √‑cooling schedule; transfer size
   scales with temperature for rapid early changes and fine adjustments
   later.

The best sequence found is returned (trailing near‑zero entries are cut)
and stored in the global `height_sequence_1` for warm‑starting future runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Objective  2·n·max(a*a) / (sum(a)²).
    For Σa = √(2n) this equals max(a*a).
    """
    n = a.shape[0]
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Stage 2 – additive projected sub‑gradient descent on the true max‑conv.
# ----------------------------------------------------------------------


def additive_subgradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Projected additive sub‑gradient descent on `max(a*a)`.
    Returns the best sequence and its score seen before `deadline`.
    """
    n = a0.shape[0]
    a = a0.copy()
    a = a * (target_sum / max(a.sum(), 1e-12))

    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        # exact convolution and location of the current peak
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        # gradient of the active peak (2*a_{k_max-i})
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        # step size: 0.5·S / (sqrt(t)·‖g‖)  (S = target_sum)
        g_norm = np.linalg.norm(g)
        if g_norm == 0.0:
            break
        step = 0.5 * target_sum / ((np.sqrt(it + 1)) * g_norm)

        # gradient step + simple projection onto the simplex
        a = a - step * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a * (target_sum / s)
        else:  # pathological – re‑initialise uniformly
            a = np.full_like(a, target_sum / n)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGradAdd it={it:6d} score={best_val:.6f}")

        # occasional jitter to shake out of flat plateaus
        if it % 1000 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 3 – improved simulated‑annealing mass‑transfer refinement
# ----------------------------------------------------------------------


def anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Simulated‑annealing that moves tiny amounts of mass from “bad’’ to “good’’
    indices.  Temperature starts at 0.9·current score and decays with a
    square‑root schedule.
    """
    a = a0.copy()
    n = a.shape[0]

    b = _conv_fft(a)                     # exact convolution a*a
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.90

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        # locate current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – pick an index with relatively small height
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # size of the transfer (scaled by temperature)
        max_delta = min(a[donor] * 0.10, target_sum * 0.010)
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # --- O(n) update of the convolution --------------------------------
        b_new = b.copy()
        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        # Metropolis acceptance based on *current* energy
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal new best={best_val:.6f}")

        # periodic full recomputation to avoid drift
        if it % 5000 == 0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute new best={best_val:.6f}")

        it += 1

    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n gives finer granularity but costs time)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 500:
        n = 15000
    else:
        n = 10000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Assemble a small pool of warm‑starts
    # --------------------------------------------------------------
    init_seqs = []

    # 1) Previous best (if any)
    if "height_sequence_1" in globals():
        seq = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
        init_seqs.append(seq)

    # 2) Power‑law decays with a few exponents
    for exp in (0.55, 0.5, 0.6):
        seq = (np.arange(1, n + 1, dtype=np.float64) ** (-exp))
        seq = seq * (target_sum / seq.sum())
        init_seqs.append(seq)

    # 3) Random Dirichlet‑like start
    rng = np.random.default_rng()
    rand_seq = rng.random(n)
    rand_seq = rand_seq * (target_sum / rand_seq.sum())
    init_seqs.append(rand_seq)

    # --------------------------------------------------------------
    # Stage 2 – additive projected sub‑gradient descent on each start
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # allocate roughly 60 % of the remaining budget for this phase
    remaining = deadline - time.time()
    sub_budget = 0.60 * remaining
    per_start_budget = sub_budget / max(1, len(init_seqs))

    for idx, start_seq in enumerate(init_seqs):
        if time.time() >= deadline:
            break
        sub_deadline = time.time() + per_start_budget
        a_opt, val_opt = additive_subgradient_descent(start_seq, target_sum, sub_deadline)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] New best after sub‑grad start {idx} score={best_score:.6f}")

        # leave a tiny head‑room for the final stage
        if time.time() + 5.0 >= deadline:
            break

    # --------------------------------------------------------------
    # Stage 3 – aggressive annealing refinement (remaining time)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a_ann, val_ann = anneal_mass_transfer(best_seq, target_sum, deadline)
        if val_ann < best_score:
            best_score = val_ann
            best_seq = a_ann.copy()
            print(f"[{time.time():.2f}] New best after annealing score={best_score:.6f}")

    # --------------------------------------------------------------
    # Clean‑up, final evaluation and warm‑start storage
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
