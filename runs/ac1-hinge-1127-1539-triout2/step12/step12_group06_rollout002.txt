def _adam_descent(a0, deadline, target_sum, alpha=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_score = _score(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        t += 1
        _, _, _, grad = _max_conv_and_subgrad(a)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = alpha * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        # projection
        a = _project_to_simplex(a, target_sum)
        # jitter occasionally
        if t % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)
        val = _score(a)
        if val < best_score:
            best_score = val
            best = a.copy()
            print(f"[{time.time():.2f}] Adam it={t} score={best_score:.6f}")
    return best, best_score
```

"""
Hybrid optimiser for the non‑negative step‑function problem (improved).

Main ideas
----------
* Use several fast optimisation primitives:
  - Projected Adam (adaptive moment) on the exact max‑convolution sub‑gradient.
  - Soft‑max (log‑sum‑exp) gradient descent with geometric β‑increase.
  - Targeted permutation annealing.
  - Deterministic mass‑transfer (greedy Monte‑Carlo) for final polishing.
* Start with a diverse pool of seeds at a modest size (≈4000).
* Run many short Adam runs on these seeds, keep the best.
* Upscale the best seed to a larger length (12‑20k) – more degrees of freedom.
* Perform long Adam optimisation, followed by the soft‑max stage, then
  permutation search and a final mass‑transfer pass.
* All iterates are projected onto the simplex {x ≥ 0, Σx = √(2n)} so the
  evaluation score equals the maximal self‑convolution entry.
* The routine respects the 1000 s budget (a small safety margin is kept).

The function `propose_candidate` returns the final list of heights.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic building blocks
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) via FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(_conv_fft(a).max())


def _max_conv_and_subgrad(a: np.ndarray):
    """
    Returns (conv, max_val, k_max, grad) where grad is a sub‑gradient of
    max(conv) w.r.t. a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    kmax = int(np.argmax(conv))
    max_val = conv[kmax]

    # The gradient of the peak entry is 2 * a reversed restricted to the supporting interval.
    grad = np.zeros_like(a)
    lo = max(0, kmax - (n - 1))
    hi = min(kmax, n - 1)
    idx = np.arange(lo, hi + 1)
    partner = a[kmax - idx]
    grad[lo:hi + 1] = 2.0 * partner
    return conv, max_val, kmax, grad


def _score(a: np.ndarray) -> float:
    """Objective = max self‑convolution (sum is √(2n) ⇒ score = max_b)."""
    return _max_convolution_fft(a)

# Sequence generators

def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation from old to new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → left, second largest → right, third → next left, …."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    rev = _alternating_order(seq[::-1])
    return rev[::-1]

def _adam_descent(a0: np.ndarray,
                  deadline: float,
                  target_sum: float,
                  alpha: float = 0.15,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the exact max‑convolution sub‑gradient.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_val = _score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        _, _, _, grad = _max_conv_and_subgrad(a)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = alpha * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to leave flat regions
        if t % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] Adam it={t:5d}  best={best_val:.6f}")

    return best, best_val


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(conv). Returns (grad, conv)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(conv)
    exps = np.exp(beta * (conv - max_b))
    p = exps / np.sum(exps)

    # grad = 2 * (a_rev) ⋆ p
    a_rev = a[::-1]
    grad = 2.0 * np.fft.irfft(np.fft.rfft(a_rev, n=size) *
                               np.fft.rfft(p, n=size), n=size)[: 2 * n - 1]
    grad = grad[n - 1: n - 1 + n]
    return grad, conv


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 2e6,
                               step_factor: float = 0.5,
                               gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max approximation.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SoftGD it={it:5d} β={beta:8.2f} best={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.3,
                                end_temp: float = 1e-5,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy donor→receiver moves with an exponential temperature schedule."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]
        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_score


def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.15,
                                 end_temp: float = 1e-5,
                                 top_frac: float = 0.04,
                                 low_frac: float = 0.04) -> tuple[np.ndarray, float]:
    """Metropolis swaps focused on indices that contribute most/least to the current peak."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)

        if donor == receiver:
            continue

        a[donor], a[receiver] = a[receiver], a[donor]

        new_score = _score(a)

        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] PermSearch new best={best_score:.6f}")

        # Metropolis acceptance
        if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / max(temp, 1e-12)):
            pass
        else:
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with a low evaluation score.
    The routine respects the 1000 s time budget and stores the best
    sequence in the global variable ``height_sequence_1``.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # 1. Initialise a moderate size for quick exploration
    # ------------------------------------------------------------------
    n_expl = 4000
    target_expl = math.sqrt(2.0 * n_expl)

    # ------------------------------------------------------------------
    # 2. Build a diverse pool of seed vectors
    # ------------------------------------------------------------------
    init_pool = []

    # Power‑law family
    for p in np.linspace(0.35, 0.75, 9):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))

    # Exponential family
    for lam in np.linspace(0.0015, 0.015, 7):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))

    # Random Dirichlet‑like points
    for _ in range(6):
        a = rng.random(n_expl)
        a *= target_expl / a.sum()
        init_pool.append(a)

    # Linear decreasing shape
    lin = np.linspace(1.0, 0.0, n_expl)
    lin *= target_expl / lin.sum()
    init_pool.append(lin)

    # Include previous best if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_expl))

    # Jittered copies for extra diversity
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.008, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    # ------------------------------------------------------------------
    # 3. Quick Adam runs on the moderate size (exploration)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # Reserve roughly 200 s for the later, more expensive phases
    exploration_deadline = deadline - 200.0
    while time.time() < exploration_deadline:

    remaining = deadline - time.time()

    elif remaining > 350:
        n_final = 15000
    elif remaining > 200:
        n_final = 12000
    else:
        n_final = n_expl

    target_final = math.sqrt(2.0 * n_final)
    a_big = _resize_sequence(best_seq, n_final)
    a_big = _project_to_simplex(a_big, target_final)

    # Keep a couple of simple reorderings and retain the best start point
    candidates = [a_big,
                  _alternating_order(a_big),
                  _reversed_alternating(a_big)]
    best_score = _score(a_big)
    best_seq = a_big
    for cand in candidates[1:]:
        sc = _score(cand)

    # ------------------------------------------------------------------

        best_seq, best_score = _annealed_gradient_descent(best_seq,
                                                          soft_deadline,

                                                          max_beta=5e6,
                                                          step_factor=0.6,
                                                          gamma=0.995)

    # ------------------------------------------------------------------

        perm_deadline = deadline - 30.0
        best_seq, best_score = _permutation_targeted_search(best_seq,

    except Exception:
        pass

    return final_seq
