"""
Propose a step‑function with a low max‑self‑convolution score.

The search consists of three stages:

1. **Fast mirror‑descent** on a soft‑max surrogate (≈5 % of the time).
2. **Linear‑programming refinement** – solve a LP that maximises the total
   mass of a companion vector `g` while keeping the mixed convolution
   `f*g` below the current peak `M`.  The resulting `g` is mixed with the
   current `f` (convex combination) and the peak usually drops.
3. **Short simulated‑annealing “mass‑transfer’’** on the final length
   to polish the result.

All phases respect the global time budget (default 1000 s).  The best
sequence is stored in the global variable `height_sequence_1` for warm‑starts.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Convolution utilities (FFT based – O(n log n))
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a power‑of‑2 FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Exact objective: 2·n·max(a*a) / (sum(a)²).
    After normalising sum(a)=sqrt(2n) this equals max(a*a).
    """
    n = a.shape[0]
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of the max‑convolution.
    Returns (grad, full_convolution) where grad has the same length as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    grad_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(grad_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – exponential (mirror) descent on the soft‑max surrogate
# ----------------------------------------------------------------------
def exponential_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Mirror descent on the soft‑max surrogate."""
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpDescent it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – multiplicative sub‑gradient descent on the true max‑conv.
# ----------------------------------------------------------------------
def subgradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Sub‑gradient descent on the true max‑convolution entry."""
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        elapsed = time.time() - start
        total = deadline - start
        eta_start = 2.0
        eta_end = 0.2
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase D – simulated‑annealing mass‑transfer (shortened version)
# ----------------------------------------------------------------------
def anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """Mass‑transfer annealing – moves tiny amounts of mass from “bad’’ to “good’’."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.80  # initial temperature

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        # locate current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # donor – proportional to its contribution to the peak
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – favour low heights
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.05, target_sum * 0.005)
        if max_delta <= 0:
            continue

        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # O(n) update of the convolution
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved best={best_val:.6f}")

        if it % 5000 == 0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – Linear‑programming based refinement
# ----------------------------------------------------------------------
def lp_refine(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Repeatedly solve a LP that maximises total mass of a vector g
    under the constraint ‖a * g‖∞ ≤ M (M = current peak).
    The resulting g is mixed into a and the peak usually drops.
    """
    a = a0.copy()
    best_a = a.copy()
    best_M = _max_convolution(a)
    it = 0

    while time.time() < deadline:
        M = best_M
        n = a.shape[0]

        # Build Toeplitz constraint matrix: (f * g)_k = Σ_j f_{k-j} g_j
        col = np.r_[a, np.zeros(n - 1)]
        row = np.r_[a[0], np.zeros(n - 1)]
        A = toeplitz(col, row)  # shape (2n-1, n)

        b_ub = np.full(2 * n - 1, M)
        c = -np.ones(n)  # maximise sum(g)  <=> minimise -sum(g)

        # solve LP
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True})
        if not res.success:
            print("[LP] Solver failed – terminating LP phase")
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 < 1e-12:
            print("[LP] LP returned near‑zero mass – terminating")
            break

        # scale g to the required total sum
        scale = target_sum / sum_g0
        g = g0 * scale

        improved = False
        # try a few mixing rates; we keep the first that improves the peak
        for alpha in (0.4, 0.2, 0.1):
            a_new = (1.0 - alpha) * a + alpha * g
            M_new = _max_convolution(a_new)
            if M_new < best_M - 1e-9:
                a = a_new
                best_M = M_new
                best_a = a.copy()
                improved = True
                print(f"[LP] it={it} α={alpha:.2f} new M={best_M:.6f}")
                break

        if not improved:
            # No improvement with the tried alphas – stop early
            print("[LP] No improving mix found – terminating LP phase")
            break

        it += 1
        # safeguard: do not spend too much time on this phase
        if it >= 8:
            break

    return best_a, best_M


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence that minimises the benchmark score.
    The routine respects the supplied time budget (seconds) and returns the
    best sequence it found.
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Choose a working length.
    # Larger n gives finer granularity but makes the LP phase heavy.
    # We run a quick descent on a relatively large n, then down‑sample
    # for the LP refinement, and finally up‑sample again.
    # --------------------------------------------------------------
    if budget_s >= 800:
        n_big = 18000
    elif budget_s >= 500:
        n_big = 12000
    else:
        n_big = 8000

    target_big = np.sqrt(2.0 * n_big)

    # --------------------------------------------------------------
    # Warm‑start (reuse previous construction if available)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init_big = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n_big)
    else:
        init_big = (np.arange(1, n_big + 1, dtype=np.float64) ** (-0.55))

    init_big = init_big * (target_big / init_big.sum())

    # --------------------------------------------------------------
    # Phase A – cheap exponential descent (≈5 % of remaining budget)
    # --------------------------------------------------------------
    phase_a_deadline = time.time() + 0.05 * (deadline - time.time())
    a_big, _ = exponential_descent(init_big, target_big, phase_a_deadline)
    print(f"[{time.time():.2f}] After Phase A (exp‑descent)")

    # --------------------------------------------------------------
    # Phase B – sub‑gradient descent (≈5 % of remaining budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase_b_deadline = time.time() + 0.05 * (deadline - time.time())
        a_big, _ = subgradient_descent(a_big, target_big, phase_b_deadline)
        print(f"[{time.time():.2f}] After Phase B (sub‑gradient)")

    # --------------------------------------------------------------
    # Down‑sample for the LP refinement (n ≈ 2000 is a good trade‑off)
    # --------------------------------------------------------------
    n_lp = 2000
    a_lp = _resize_sequence(a_big, n_lp)
    target_lp = np.sqrt(2.0 * n_lp)
    a_lp = a_lp * (target_lp / a_lp.sum())

    # --------------------------------------------------------------
    # Phase C – linear‑programming refinement (≈30 % of remaining budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        lp_deadline = time.time() + 0.30 * (deadline - time.time())
        a_lp, _ = lp_refine(a_lp, target_lp, lp_deadline)
        print(f"[{time.time():.2f}] After Phase C (LP refinement)")

    # --------------------------------------------------------------
    # Upsample back to the original (big) length
    # --------------------------------------------------------------
    a_up = _resize_sequence(a_lp, n_big)
    a_up = a_up * (target_big / a_up.sum())

    # --------------------------------------------------------------
    # Phase D – short annealing to polish the final sequence
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_up, _ = anneal_mass_transfer(a_up, target_big, deadline)
        print(f"[{time.time():.2f}] After Phase D (annealing)")

    # --------------------------------------------------------------
    # Final clean‑up, trimming trailing zeros
    # --------------------------------------------------------------
    arr = np.array(a_up, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    globals()["height_sequence_1"] = final_seq

    # --------------------------------------------------------------
    # Final evaluation (for logging only – the harness will also call it)
    # --------------------------------------------------------------
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    return final_seq
