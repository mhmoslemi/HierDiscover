def _simulated_annealing(a0, target_sum, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    n = a.size

    T0 = (best_val) * 0.01 # maybe 1% of value.
    T = T0
    start = time.time()
    it = 0
    while time.time() < deadline and it < 200000:
        # progress for temperature schedule
        progress = (time.time() - start) / (deadline - start)
        T = T0 * (1 - progress)  # linear decay to 0

        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i: j += 1
        # epsilon fraction
        eps = rng.random() * min(0.1 * target_sum, a[i])  # up to 10% of total sum or available mass
        if eps <= 0:

        cand = a.copy()
        cand[i] -= eps
        cand[j] += eps

        val = _score(cand)
        if val < best_val or rng.random() < np.exp(-(val - best_val) / max(T, 1e-12)):
            a = cand
            if val < best_val:
                best_val = val
                best_a = cand.copy()
                print(f"[{time.time():.2f}] SA it={it} val={best_val:.6f}")
        it += 1
    return best_a, best_val

"""
Propose a non‑negative step‑function (a list of heights) that minimizes
the maximal self‑convolution.  The algorithm is a multi‑phase
heuristic:

0. Choose a problem size n (larger n gives more degrees of freedom).
1. Fast parametric sweep over several analytic families
   (power laws, power‑law+exponential tails, two‑sided power laws,
   simple sine).  The best candidate is retained.
2. A long Adam‑style projected gradient descent on a soft‑max
   approximation of the maximum (the “soft‑max” temperature β is
   increased gradually up to 2e5).  Multiple restarts are performed.
3. Projected (exact) sub‑gradient descent on the true max‑convolution.
4. Deterministic “mass‑transfer” moves that analytically drive the
   current peak down to the second‑largest convolution value.
5. A simulated‑annealing style random walk that moves a small amount
   of mass between two indices with a slowly cooling temperature.
6. Final polishing with a very large soft‑max temperature (β≈1e6).

All steps respect the global time budget (default 1000 s) and
project onto the simplex {a_i≥0, Σa_i = √(2 n)} after each move.
The routine can start from a previously discovered sequence stored in
the global variable ``height_sequence_1``.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, works for any n)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean, Σx = s, x ≥ 0)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx=s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families for the cheap pre‑search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0))
    a *= target_sum / a.sum()
    return a


def _two_sided_power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Large values near the two ends, small in the middle."""
    dist = np.minimum(np.arange(1, n + 1), np.arange(n, 0, -1))
    a = dist.astype(np.float64) ** (-p)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(a*a)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # convolution a_rev * p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f}  score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _adam_restart_manager(start_vec: np.ndarray,
                          target_sum: float,
                          overall_deadline: float) -> tuple[np.ndarray, float]:
    """Run several short Adam restarts and keep the best result."""
    best_a = start_vec.copy()
    best_val = _max_convolution_fft(best_a)
    rng = np.random.default_rng()

    while time.time() < overall_deadline - 5.0:
        remaining = overall_deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        a0 = best_a + jitter
        a0 = np.maximum(a0, 0.0)
        a0 = _project_to_simplex(a0, target_sum)

        a, val = _adam_gradient_descent(
            a0, time.time() + run_budget, target_sum,
            max_beta=2e5, step_factor=0.8, gamma=0.995)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (true max)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,

                         max_iter: int = 5000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_fft(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.flatnonzero(conv >= max_val - 1e-12)

        # second‑largest value
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()
        delta_target = max_val - second_max
        if delta_target < 1e-14:
            break

        # average sub‑gradient over all active maxima
        g = np.zeros_like(a)
        n = a.shape[0]
        for k in idx_max:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            g[idx] += 2.0 * a[k - idx]

        g /= len(idx_max)

        norm_g2 = np.dot(g, g)
        if norm_g2 == 0.0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = _project_to_simplex(a + jitter, target_sum)
            it += 1
            continue

        step = 0.9 * delta_target / norm_g2
        step = min(step, target_sum * 0.4 / (np.linalg.norm(g) + 1e-12))

        for _ in range(6):
            a_candidate = _project_to_simplex(a - step * g, target_sum)
            cand_score = _max_convolution_fft(a_candidate)
            if cand_score < best_score - 1e-12:
                a = a_candidate
                best_a = a_candidate.copy()
                best_score = cand_score
                print(f"[{time.time():.2f}] Subgrad it={it:5d} step={step:.3e} score={best_score:.6f}")
                break
            step *= 0.5
        else:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = _project_to_simplex(a + jitter, target_sum)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic mass‑transfer (exact analytic step)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                tol: float = 1e-12) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _max_convolution_fft(a)

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()
        if max_val - second_max < tol:
            break

        k = idx_max[0]          # pick the first maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]    # values that pair with a[idx] for this k
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        # Choose a donor with large partner (large contribution to the peak)
        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # Choose a receiver with a small partner (minimal increase of the peak)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _max_convolution_fft(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] DMT it={len(best_a)} score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing – random mass moves with a cooling schedule
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    rng = np.random.default_rng()
    n = a.shape[0]
    start = time.time()
    # starting temperature is a few percent of the current value
    T0 = best_val * 0.02
    it = 0

    while time.time() < deadline and it < 300000:
        progress = (time.time() - start) / (deadline - start)
        T = max(T0 * (1.0 - progress), 1e-12)

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer a random fraction of the donor mass (up to 5 % of total sum)
        max_transfer = min(a[i], target_sum * 0.05)
        if max_transfer <= 0:
            it += 1
            continue
        eps = rng.random() * max_transfer

        cand = a.copy()
        cand[i] -= eps
        cand[j] += eps

        # tiny numerical drift may happen – re‑project just in case
        cand = _project_to_simplex(cand, target_sum)

        cand_val = _max_convolution_fft(cand)

        if cand_val < best_val:
            best_val = cand_val
            best_a = cand.copy()
            a = cand
            print(f"[{time.time():.2f}] SA it={it:6d} new best={best_val:.6f}")
        else:
            # Metropolis acceptance
            if rng.random() < np.exp(-(cand_val - best_val) / T):
                a = cand

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal maximal self‑convolution.
    Returns a Python list of floats (trailing near‑zero entries are removed).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a tiny safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n → more flexibility, still cheap FFT)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 400:
        n = 15000
    else:
        n = 12000
    target_sum = np.sqrt(2.0 * n)   # Σa must equal this value

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric families, random starts, previous best
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    # power‑law sweep
    for p in np.linspace(0.25, 0.90, 27):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _max_convolution_fft(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # power‑law + exponential tail sweep
    for p in np.linspace(0.25, 0.90, 21):
        for q in np.linspace(0.0, 6.0, 21):
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _max_convolution_fft(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # two‑sided power law (large at both ends)
    for p in np.linspace(0.2, 0.8, 19):
        seq = _two_sided_power_law_sequence(n, float(p), target_sum)
        val = _max_convolution_fft(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # simple sine shape
    seq = _sin_sequence(n, target_sum)
    val = _max_convolution_fft(seq)
    if val < best_score:
        best_score = val
        best_seq = seq.copy()

    # previous global optimum if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev
        resized = _project_to_simplex(resized, target_sum)
        val = _max_convolution_fft(resized)
        if val < best_score:
            best_score = val
            best_seq = resized.copy()

    # a few jittered perturbations around the current best
    for _ in range(6):
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _max_convolution_fft(cand)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()

    print(f"[{time.time():.2f}] After pre‑search best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam‑style GD with many restarts (≈30 % of remaining time)
    # ------------------------------------------------------------------
    phase_a_deadline = time.time() + (deadline - time.time()) * 0.30
    best_seq, best_score = _adam_restart_manager(best_seq, target_sum, phase_a_deadline)
    print(f"[{time.time():.2f}] After Adam phase best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Exact sub‑gradient descent (≈20 % of remaining time)
    # ------------------------------------------------------------------
    phase_b_deadline = time.time() + (deadline - time.time()) * 0.20
    seq_sg, val_sg = _subgradient_descent(best_seq, target_sum, phase_b_deadline)
    if val_sg < best_score:
        best_score = val_sg
        best_seq = seq_sg.copy()
        print(f"[{time.time():.2f}] Sub‑gradient phase improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase C – Deterministic mass‑transfer (≈15 % of remaining time)
    # ------------------------------------------------------------------
    phase_c_deadline = time.time() + (deadline - time.time()) * 0.15
    seq_dm, val_dm = _deterministic_mass_transfer(best_seq, phase_c_deadline, target_sum)
    if val_dm < best_score:
        best_score = val_dm
        best_seq = seq_dm.copy()
        print(f"[{time.time():.2f}] Mass‑transfer phase improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase D – Simulated annealing (use whatever time is left, keep a few sec)
    # ------------------------------------------------------------------
    sa_deadline = deadline - 4.0
    if time.time() < sa_deadline:
        seq_sa, val_sa = _simulated_annealing(best_seq, target_sum, sa_deadline)
        if val_sa < best_score:
            best_score = val_sa
            best_seq = seq_sa.copy()
            print(f"[{time.time():.2f}] SA phase improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase E – Final high‑β Adam polish (≈3 s)
    # ------------------------------------------------------------------
    if time.time() + 2.5 < deadline:
        seq_fin, val_fin = _adam_gradient_descent(
            best_seq,
            deadline,
            target_sum,
            max_beta=1_000_000,
            step_factor=0.6,
            gamma=0.995,
        )
        if val_fin < best_score:
            best_score = val_fin
            best_seq = seq_fin.copy()
            print(f"[{time.time():.2f}] Final polish improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional sanity check (ignored if evaluate_sequence is not present)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
