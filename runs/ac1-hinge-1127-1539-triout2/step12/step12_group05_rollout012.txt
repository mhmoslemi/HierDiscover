Improved step‑function optimizer.

Main ideas
----------

1. **Rich parametric families** – besides the original power‑law,

4. **Two Adam passes** – an early low‑β Adam for global exploration
   and a final high‑β Adam for polishing.

All phases respect the simplex constraint Σa = √(2 n) and the

# ----------------------------------------------------------------------
# FFT based convolution utilities (linear convolution)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) using a real FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft_pair(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution x*y (full length 2·n‑1) using a real FFT."""
    n = x.size
    size = 1 << ((2 * n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Score = 2·n·max(a*a) / (sum(a)^2).  Scale‑invariant."""
    n = a.shape[0]
    max_b = _max_conv(a)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex Σx = s, x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    # sort descending and compute the threshold theta
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (dense pre‑search)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a = np.clip(a, 0.0, 1000.0)
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a = np.clip(a, 0.0, 1000.0)
    a *= target_sum / a.sum()
    return a


def _broken_power_law_sequence(
    n: int, p1: float, p2: float, alpha: float, target_sum: float
) -> np.ndarray:
    K = max(1, int(alpha * n))
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.empty_like(idx)
    a[:K] = idx[:K] ** (-p1)
    a[K:] = idx[K:] ** (-p2)
    a = np.clip(a, 0.0, 1000.0)
    a *= target_sum / a.sum()
    return a


def _beta_sequence(
    n: int, alpha: float, beta: float, target_sum: float
) -> np.ndarray:
    """Beta‑distribution shape on a uniform grid."""
    # centre points to avoid exact 0/1
    x = (np.arange(n) + 0.5) / n
    # avoid overflow for very small exponents
    a = np.power(x, max(alpha - 1.0, -10.0))
    a *= np.power(1.0 - x, max(beta - 1.0, -10.0))
    a = np.clip(a, 0.0, 1000.0)
    # If the parameters are extreme the mass may concentrate on the
    # extremes and clip to 1000 – re‑project to keep the correct sum.
    a = _project_to_simplex(a, target_sum)
    return a


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on a soft‑max approximation
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(a*a)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = conv_full.max()
    exps = np.exp(beta * (conv_full - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, conv_full


def _adam_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 5e5,
    step_factor: float = 0.8,
    gamma: float = 0.999,
) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max approximation of the max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}"
            )

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient guided mass‑transfer (Phase B)
# ----------------------------------------------------------------------


def _compute_gradient_for_max(a: np.ndarray, max_idx: np.ndarray) -> np.ndarray:
    """Exact sub‑gradient of max(conv) w.r.t. a (averaged over max indices)."""
    n = a.size
    g = np.zeros_like(a)
    for k in max_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g[idx] += 2.0 * a[k - idx]
    if max_idx.size > 0:
        g /= max_idx.size
    return g


def _guided_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Greedy exact reduction of the maximal convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        # second largest distinct entry
        mask_max = conv >= max_val - tol
        second_max = conv[~mask_max].max() if np.any(~mask_max) else 0.0

        max_idx = np.where(mask_max)[0]

        g = _compute_gradient_for_max(a, max_idx)

        sum_g = g.sum()
        d = -g + sum_g / n * np.ones_like(a)  # Σd = 0

        # maximal step respecting non‑negativity
        neg = d < 0
        if np.any(neg):
            t_nonneg = np.min(a[neg] / -d[neg])
        else:
            t_nonneg = np.inf

        delta = 2.0 * _conv_fft_pair(a, d)
        delta_max = delta[max_idx]
        delta_k = delta_max.min()
        if delta_k >= 0:
            break

        t_to_second = (max_val - second_max) / (-delta_k)
        t = min(t_nonneg, t_to_second) * 0.99
        if t <= 0:
            break

        a += t * d
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] GuidedMT it={it:5d} score={best_score:.6f}"
            )

        # tiny jitter to escape plateaus
        if it % 500 == 0 and rng.random() < 0.05:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Light global random exploration (optional)
# ----------------------------------------------------------------------


def _global_random_move(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
) -> tuple[np.ndarray, float]:
    """Light simulated‑annealing style mass moves."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    while time.time() < deadline - 0.5:
        top_k = max(1, int(0.02 * n))
        low_k = max(1, int(0.02 * n))

        donor = np.argsort(-a)[rng.integers(top_k)]
        receiver = np.argsort(a)[rng.integers(low_k)]
        if donor == receiver:
            continue

        delta = rng.random() * a[donor] * 0.05
        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] RandomMove new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic fine‑grained search in the parameter space
# ----------------------------------------------------------------------


def _stochastic_param_search(
    n: int,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Random walks in the space of the parametric families.
    Small Gaussian changes are applied to the parameters; the best
    sequence discovered is kept.  The loop stops when the remaining
    time drops below a safety margin.
    """
    # store current best
    a_best = best_seq.copy()
    score_best = best_score

    # parameter ranges (log‑space where appropriate)
    while time.time() < deadline - 5.0:
        # choose a family uniformly at random
        fam = rng.integers(4)
        if fam == 0:  # power‑law
            p = rng.normal(0.7, 0.15)  # centre around previously good values
            p = np.clip(p, 0.05, 2.5)
            seq = _power_law_sequence(n, float(p), target_sum)

        elif fam == 1:  # exponential‑power‑law
            p = rng.normal(0.7, 0.15)
            q = rng.normal(2.0, 0.8)
            p = np.clip(p, 0.05, 2.5)
            q = np.clip(q, 0.0, 12.0)
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)

        elif fam == 2:  # broken power‑law
            p1 = rng.normal(0.5, 0.15)
            p2 = rng.normal(1.5, 0.3)
            alpha = rng.normal(0.4, 0.08)
            p1 = np.clip(p1, 0.05, 2.5)
            p2 = np.clip(p2, 0.5, 3.0)
            alpha = np.clip(alpha, 0.05, 0.95)
            seq = _broken_power_law_sequence(
                n, float(p1), float(p2), float(alpha), target_sum
            )

        else:  # beta family
            alpha = rng.normal(0.8, 0.25)
            beta = rng.normal(0.8, 0.25)
            alpha = np.clip(alpha, 0.05, 5.0)
            beta = np.clip(beta, 0.05, 5.0)
            seq = _beta_sequence(n, float(alpha), float(beta), target_sum)

        val = _score(seq)
        if val < score_best:
            score_best = val
            a_best = seq.copy()
            print(
                f"[{time.time():.2f}] ParamSearch new best={score_best:.6f}"
            )

        # Occasionally (10 % of iterations) run a tiny Adam polish on the
        # candidate to see whether a local optimum is nearby.
        if rng.random() < 0.1:
            sub_deadline = time.time() + 0.5
            opt_seq, opt_val = _adam_gradient_descent(
                seq,
                sub_deadline,
                target_sum,
                max_beta=1e5,
                step_factor=0.9,
                gamma=0.998,
            )
            if opt_val < score_best:
                score_best = opt_val
                a_best = opt_seq.copy()
                print(
                    f"[{time.time():.2f}] ParamSearch Adam polish improved best={score_best:.6f}"
                )

    return a_best, score_best


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal benchmark score.
    The routine respects the supplied time budget and returns the best
    sequence found (trailing near‑zero entries are trimmed).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # tiny safety margin

    # ---------------------------------------------------------------
    # Choose problem size.
    # ---------------------------------------------------------------
    if budget_s >= 800:
        n = 15000
    elif budget_s >= 400:
        n = 10000
    else:
        n = 5000

    target_sum = math.sqrt(2.0 * n)  # Σa is fixed to this value

    # ---------------------------------------------------------------
    # Phase 0 – dense deterministic pre‑search
    # ---------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    rng = np.random.default_rng()

    # (a) power‑law grid
    for p in np.arange(0.20, 1.21, 0.01):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # (b) exponential‑power‑law grid
    for p in np.arange(0.20, 1.21, 0.02):
        for q in np.arange(0.0, 8.001, 0.25):
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # (c) broken power‑law grid
    for p1 in np.arange(0.20, 0.81, 0.1):
        for p2 in np.arange(0.80, 2.01, 0.2):
            for alpha in np.arange(0.1, 0.91, 0.1):
                seq = _broken_power_law_sequence(
                    n, float(p1), float(p2), float(alpha), target_sum
                )
                val = _score(seq)
                if val < best_score:
                    best_score, best_seq = val, seq.copy()

    # (d) beta family coarse grid
    for a in np.arange(0.4, 3.01, 0.3):
        for b in np.arange(0.4, 3.01, 0.3):
            seq = _beta_sequence(n, float(a), float(b), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # (e) warm start from external best if available
    if "height_sequence_1" in globals():
        prev = np.array(
            globals()["height_sequence_1"], dtype=np.float64
        )
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = np.clip(resized, 0.0, 1000.0)
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score, best_seq = val, resized.copy()

    print(
        f"[{time.time():.2f}] After dense pre‑search: best score = {best_score:.6f}"
    )

    # ---------------------------------------------------------------
    # Phase 1 – stochastic parametric fine‑tuning
    # ---------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 100.0:
        a_refined, val_refined = _stochastic_param_search(
            n,
            target_sum,
            deadline,
            rng,
            best_seq,
            best_score,
        )
        if val_refined < best_score:
            best_score, best_seq = val_refined, a_refined.copy()
            print(
                f"[{time.time():.2f}] After stochastic param‑search: best score = {best_score:.6f}"
            )

    # ---------------------------------------------------------------
    # Phase 2 – short Adam runs (exploratory restarts)
    # ---------------------------------------------------------------
    remaining = deadline - time.time()
    ada_deadline = time.time() + min(0.15 * remaining, 30.0)  # bounded time
    while time.time() < ada_deadline - 2.0:
        jitter = rng.normal(
            scale=target_sum * 0.015, size=n
        )  # broad jitter before each restart
        start_vec = best_seq + jitter
        start_vec = np.clip(start_vec, 0.0, 1000.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _adam_gradient_descent(
            start_vec,
            ada_deadline,
            target_sum,
            max_beta=5e5,
            step_factor=0.8,
            gamma=0.995,
        )
        if val_opt < best_score:
            best_score, best_seq = val_opt, a_opt.copy()
            print(
                f"[{time.time():.2f}] Adam restart improved best = {best_score:.6f}"
            )

    # ---------------------------------------------------------------
    # Phase 3 – guided exact mass‑transfer (core refinement)
    # ---------------------------------------------------------------
    while time.time() < deadline - 10.0:
        a_ref, val_ref = _guided_mass_transfer(
            best_seq, deadline, target_sum
        )
        if val_ref < best_score:
            best_score, best_seq = val_ref, a_ref.copy()
            print(
                f"[{time.time():.2f}] Guided mass‑transfer improved best = {best_score:.6f}"
            )

        # occasional random move to escape tiny plateaus
        a_rand, val_rand = _global_random_move(best_seq, deadline, target_sum)
        if val_rand < best_score:
            best_score, best_seq = val_rand, a_rand.copy()
            print(
                f"[{time.time():.2f}] Random move improved best = {best_score:.6f}"
            )

        if time.time() > deadline - 4.0:
            break

    # ---------------------------------------------------------------
    # Phase 4 – final polishing with a high‑β Adam run
    # ---------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(
            best_seq,
            deadline,
            target_sum,
            max_beta=1e6,
            step_factor=0.6,
            gamma=0.995,
        )
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(
                f"[{time.time():.2f}] Final polish improved best = {best_score:.6f}"
            )

    # ---------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return list
    # ---------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final evaluation for reporting (harmless if evaluate_sequence missing)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(
            f"[{time.time():.2f}] Final evaluated benchmark score = {val:.6f}"
        )
    except Exception:
        pass

    return final_seq
