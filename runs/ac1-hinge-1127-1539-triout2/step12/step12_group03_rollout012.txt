3.  Exact projected sub‑gradient descent on the true max‑convolution.
4.  Deterministic greedy mass‑transfer (exact DMT) +
    a cheap random mass‑transfer phase.
5.  Simulated annealing (random donor → receiver moves with
    Metropolis acceptance) to escape remaining local minima.
6.  Final high‑β Adam polishing.

All stages respect the wall‑clock budget (default 1000 s) and the
simplex constraint Σa = √(2 n).  The returned list is trimmed of
trailing near‑zero entries and can be fed directly into
``evaluate_sequence``.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# 0.  Core convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# 1.  Projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 2.  Parametric families for the quick pre‑search
# ----------------------------------------------------------------------
def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    """Simple positive sine shape."""
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# 3.  Soft‑max gradient (used by Adam)
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(b)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    ar_fft = np.fft.rfft(a_rev, n=size)
    p_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(ar_fft * p_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# 4.  Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------
def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """
    Adam on the soft‑max surrogate.
    Returns (best_sequence, best_max_convolution).
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 250

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} max={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(start_vec: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several short Adam restarts from jittered starts."""
    best_a = start_vec.copy()
    best_val = _max_convolution_fft(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        a0 = best_a + jitter
        a0 = np.maximum(a0, 0.0)
        a0 = _project_to_simplex(a0, target_sum)

        a_opt, val_opt = _adam_gradient_descent(
            a0,
            time.time() + run_budget,
            target_sum,
            max_beta=2e5,
            step_factor=0.8,
            gamma=0.995,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_a = a_opt.copy()
            print(f"[{time.time():.2f}] Adam restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# 5.  Exact projected sub‑gradient descent (true max)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         max_iter: int = 8000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent directly on max‑convolution.
    Returns (best_sequence, best_max_convolution).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_fft(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        max_val = conv.max()
        # indices attaining the max (tolerance 1e-12)
        idx_max = np.flatnonzero(conv >= max_val - 1e-12)

        # second largest value (ignore current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()
        delta_target = max_val - second_max
        if delta_target < 1e-14:
            break

        k = idx_max[rng.integers(len(idx_max))]
        n = a.shape[0]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k - idx]          # exact sub‑gradient for this k
        norm2 = np.dot(g, g)
        if norm2 == 0.0:
            # tiny random jitter
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = _project_to_simplex(a + jitter, target_sum)
            it += 1
            continue

        step = 0.9 * delta_target / norm2
        step = min(step, target_sum * 0.4 / (np.linalg.norm(g) + 1e-12))

        # simple backtracking (max 6 reductions)
        for _ in range(6):
            cand = _project_to_simplex(a - step * g, target_sum)
            cand_score = _max_convolution_fft(cand)
            if cand_score < best_score - 1e-12:
                a = cand
                best_a = cand.copy()
                best_score = cand_score
                print(f"[{time.time():.2f}] Subgrad it={it:5d} step={step:.3e} max={best_score:.6f}")
                break
            step *= 0.5
        else:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = _project_to_simplex(a + jitter, target_sum)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# 6.  Deterministic mass‑transfer (core refinement)
# ----------------------------------------------------------------------
def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the current peak by moving mass from a donor
    (large derivative) to a receiver (small derivative) using the
    analytically required amount.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _max_convolution_fft(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest entry
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()
        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # pick the first maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k‑i}
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break
        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # receiver – smallest partner value (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom

        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _max_convolution_fft(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] DMT it={it:5d} max={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# 7.  Global random mass‑transfer (exploratory)
# ----------------------------------------------------------------------
def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """Very cheap random donor→receiver moves."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _max_convolution_fft(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 0.5:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur = _max_convolution_fft(a)
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] Random move new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# 8.  Simulated annealing (new component)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         init_temp: float = 0.05,
                         final_temp: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Simple annealing: move a random fraction of mass from a donor
    (large a) to a receiver (small a).  Metropolis acceptance is used.
    """
    a = a0.copy()
    cur_val = _max_convolution_fft(a)
    best_a = a.copy()
    best_val = cur_val

    n = a.size
    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        # linear temperature schedule based on elapsed fraction
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        T = init_temp * (1.0 - frac) + final_temp * frac
        step_frac = 0.20 * (1.0 - frac) + 0.02 * frac   # maximal fraction of donor mass to move

        # pick a donor (biased towards larger entries)
        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        # pick a receiver ≠ i
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # random amount to shift
        delta = a[i] * step_frac * rng.random()
        if delta < 1e-12:
            continue

        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        # safety – keep non‑negative (numerical)
        if a_candidate[i] < 0.0:
            continue

        cand_val = _max_convolution_fft(a_candidate)
        dE = cand_val - cur_val

        accept = False
        if dE <= 0.0:
            accept = True
        else:
            # Metropolis criterion
            prob = np.exp(-dE / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_candidate
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal it={it:6d} T={T:.3e} max={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# 9.  Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal max self‑convolution.
    Returns the best sequence (list of floats) found before the time budget elapses.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size selection (larger n gives more freedom but larger FFT)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 15000                     # fits into a 2^15 FFT (32768)
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000
    target_sum = np.sqrt(2.0 * n)    # Σa = √(2 n)

    # ------------------------------------------------------------------
    # Phase 0 – quick parametric sweep (pure power‑law, exp‑tail, sine)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    p_grid = np.linspace(0.25, 1.10, 21)   # exponents
    q_grid = np.linspace(0.0, 8.0, 21)     # exponential tail strength

    # (a) pure power‑law
    for p in p_grid:
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _max_convolution_fft(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # (b) power‑law + exponential tail
    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _max_convolution_fft(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # (c) sine shape
    seq = _sin_sequence(n, target_sum)
    val = _max_convolution_fft(seq)
    if val < best_score:
        best_score = val
        best_seq = seq.copy()

    # (d) previous global best if available (resize if size mismatch)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev
        resized = _project_to_simplex(resized, target_sum)
        val = _max_convolution_fft(resized)
        if val < best_score:
            best_score = val
            best_seq = resized.copy()

    # (e) small jittered population around the current best
    for _ in range(6):
        jitter = rng.normal(scale=target_sum * 0.005, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _max_convolution_fft(cand)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()

    print(f"[{time.time():.2f}] After parametric sweep best max={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam‑style gradient descent with restarts (soft‑max)
    # ------------------------------------------------------------------
    gd_deadline = time.time() + (deadline - time.time()) * 0.30
    while time.time() < gd_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _gradient_descent_with_restarts(
            start_vec, target_sum, gd_deadline
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Adam restart improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Projected sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    subgrad_deadline = time.time() + (deadline - time.time()) * 0.20
    a_sg, val_sg = _subgradient_descent(best_seq, target_sum, subgrad_deadline)
    if val_sg < best_score:
        best_score = val_sg
        best_seq = a_sg.copy()
        print(f"[{time.time():.2f}] Subgradient descent improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase C – Exact deterministic mass‑transfer + random moves
    # ------------------------------------------------------------------
    while time.time() < deadline - 12.0:
        # deterministic greedy reduction
        a_ref, val_ref = _deterministic_mass_transfer_exact2(
            best_seq, deadline, target_sum
        )
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] DMT improved best={best_score:.6f}")

        # cheap random exploration
        a_rand, val_rand = _global_random_mass_transfer(
            best_seq, deadline, target_sum
        )
        if val_rand < best_score:
            best_score = val_rand
            best_seq = a_rand.copy()
            print(f"[{time.time():.2f}] Random move improved best={best_score:.6f}")

        if time.time() > deadline - 12.0:
            break

    # ------------------------------------------------------------------
    # Phase D – Simulated annealing (escape remaining plateaus)
    # ------------------------------------------------------------------
    if time.time() < deadline - 6.0:
        anneal_deadline = deadline - 4.0
        a_ann, val_ann = _simulated_annealing(
            best_seq,
            target_sum,
            anneal_deadline,
            init_temp=0.04,
            final_temp=1e-7,
        )
        if val_ann < best_score:
            best_score = val_ann
            best_seq = a_ann.copy()
            print(f"[{time.time():.2f}] Annealing improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase E – Final high‑β Adam polishing
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(
            best_seq,
            deadline,
            target_sum,
            max_beta=1e6,
            step_factor=0.6,
            gamma=0.995,
        )
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final Adam polished best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing – trim negligible trailing entries
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final evaluation (harmless if evaluate_sequence not defined)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated (full) score = {val:.6f}")
    except Exception:
        pass

    return final_seq
