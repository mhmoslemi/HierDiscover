import time, numpy as np

# Convolution util (FFT)
def _max_convolution_fft(a):
    n = a.shape[0]
    size = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    return float(conv.max())

def _conv_fft(a):
    n = a.shape[0]
    size = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    return conv

def _score(a):
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s*s))

def _project_to_simplex(v, s):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _power_law_sequence(n, p, target):
    a = (np.arange(1, n+1, dtype=np.float64) ** (-p))
    a *= target / a.sum()
    return a

def _exponential_sequence(n, lam, target):
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target / a.sum()
    return a

def _resize_sequence(seq, new_len):
    old_len = seq.shape[0]
    if old_len == new_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)

def _alternating_order(seq):
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left = 0
    right = n-1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out

def _reversed_alternating(seq):
    rev = _alternating_order(seq[::-1])
    return rev[::-1]

def _deterministic_mass_transfer(a, deadline, target_sum, start_temp=0.5, end_temp=1e-4, low_frac=0.05):
    a = a.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start = time.time()
    conv = _conv_fft(a)
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau
        # find peak index
        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi+1, dtype=np.int64)
        w = a[idx] + a[kmax - idx]
        donor = idx[np.argmax(w)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)
        if donor == receiver:
            it += 1
            continue
        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)
        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue
        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)
        conv = _conv_fft(a)
        new_score = _score(a)
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)
        it += 1
    return best_a, best_score

def _permutation_targeted_search(a0, deadline, start_temp=0.5, end_temp=1e-5, top_frac=0.05, low_frac=0.05):
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    n = a.shape[0]
    start = time.time()
    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        kmax_candidates = np.flatnonzero(conv == max_val)
        kmax = rng.choice(kmax_candidates)
        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]
        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))
        donor_candidates = np.argpartition(-contrib, top_k-1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k-1)[:low_k]
        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)
        if donor == receiver:
            continue
        a[donor], a[receiver] = a[receiver], a[donor]
        new_score = _score(a)
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] PermSearch new best={best_score:.6f}")
        else:
            a[donor], a[receiver] = a[receiver], a[donor]
    return best_a, best_score

def _lp_refine(f, target_sum, deadline, t_mix=0.2, max_iter=10):
    """Iteratively apply LP step to improve f."""
    n = f.shape[0]
    best_f = f.copy()
    best_val = _score(f)
    # pre-allocate A matrix builder
    K = 2*n - 1
    # create A matrix once for each iteration because f changes
    # but building each iteration is okay.
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        # compute current M
        M = _max_convolution_fft(f)
        # build constraint matrix A (K x n)
        # We'll allocate dense matrix; memory okay for n<=2000.
        A = np.zeros((K, n), dtype=np.float64)
        # fill using broadcasting
        # loop over columns
        for i in range(n):
            A[i:i+n, i] = f
        b = np.full(K, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)  # maximize sum -> minimize -sum
        bounds = [(0, None)] * n
        # solve LP
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if not res.success:
            print(f"[{time.time():.2f}] LP failed, breaking")
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        scale = target_sum / sum_g
        g_scaled = g * scale
        # mix
        f_new = (1 - t_mix) * f + t_mix * g_scaled
        # ensure feasibility due to numeric issues
        f_new = np.maximum(f_new, 0.0)
        # optional projection (should already sum to target_sum)
        f_new = _project_to_simplex(f_new, target_sum)
        val = _score(f_new)
        if val < best_val:
            best_val = val
            best_f = f_new.copy()
            print(f"[{time.time():.2f}] LP refine iteration new best={best_val:.6f}")
        f = f_new
        # optionally decrease t_mix gradually
        t_mix *= 0.9
    return best_f, best_val

    then up‑scales and runs a long permutation annealing.
    """
    start = time.time()

    n0 = 1800

        seeds.append(_exponential_sequence(n0, float(lam), target0))

            seeds.append(seq)

    best_val = np.inf
    for s in seeds:
        val = _score(s)
        if val < best_val:

            best_seed = s.copy()
    # small jitter and projection for robustness

        a_mt, val_mt = _deterministic_mass_transfer(best_seed, mt_dead, target0, start_temp=0.4, end_temp=1e-4, low_frac=0.05)
        if val_mt < best_val:
            best_val = val_mt
            best_seed = a_mt.copy()
            print(f"[{time.time():.2f}] After coarse MT best={best_val:.6f}")

    # decide final length based on remaining budget
    remain = deadline - time.time()
    if remain > 60.0 and budget_s >= 800:
        n_final = 12000
    elif remain > 30.0:

"""
LP‑enhanced step‑function optimiser.

The optimiser works in three stages:

1.  Build a modest‑size (≈ 1800) seed sequence and improve it by a
    small linear‑programming routine that spreads the mass while keeping
    the cross‑convolution with the current sequence bounded.
2.  Upscale the refined seed to a larger length (≈ 10 000–12 000) and try a
    few deterministic reorderings.
3.  A long targeted permutation‑annealing (plus a final tiny
    mass‑transfer pass) searches the ordering space on the large scale.

All steps keep the coefficients non‑negative and on the ℓ¹‑simplex
(∑a = √(2 n)).  The final list of floats is returned.
"""

import time
import numpy as np
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (∑a)² (scale‑invariant)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple families of initial sequences
# ----------------------------------------------------------------------
def _power_law_sequence(n: int, p: float, target: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    old_len = seq.shape[0]
    if old_len == new_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (small moves)
# ----------------------------------------------------------------------
def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-4,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start = time.time()
    conv = _conv_fft(a)
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]
        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted permutation search (simulated annealing on index order)
# ----------------------------------------------------------------------
def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-5,
                                 top_frac: float = 0.05,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    n = a.shape[0]
    start = time.time()
    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        kmax_candidates = np.flatnonzero(conv == max_val)
        kmax = rng.choice(kmax_candidates)

        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)
        if donor == receiver:
            continue

        a[donor], a[receiver] = a[receiver], a[donor]
        new_score = _score(a)

        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < cur_score or rng.random() < np.exp(
                -(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] PermSearch new best={best_score:.6f}")
        else:
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming based refinement (small scale)
# ----------------------------------------------------------------------
def _lp_refine(f: np.ndarray,
               target_sum: float,
               deadline: float,
               t_mix: float = 0.25,
               max_iter: int = 12) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve
        maximize   Σ g_i
        subject to f * g ≤ M   (M = max(f*f))
                    g_i ≥ 0, g_i ≤ target_sum
    then mix the result with the current f.
    """
    n = f.shape[0]
    best_f = f.copy()
    best_val = _score(f)
    K = 2 * n - 1

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        M = _max_convolution_fft(f)

        # build dense constraint matrix A (K×n),   A[k,i] = f[k-i] if in range
        A = np.zeros((K, n), dtype=np.float64)
        for i in range(n):
            A[i:i + n, i] = f

        b = np.full(K, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)                 # maximise Σg → minimise -Σg
        bounds = [(0.0, target_sum) for _ in range(n)]

        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if not res.success:
            print(f"[{time.time():.2f}] LP failed – stopping refinement")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        scale = target_sum / sum_g
        g_scaled = g * scale

        f_new = (1.0 - t_mix) * f + t_mix * g_scaled
        f_new = np.maximum(f_new, 0.0)
        f_new = _project_to_simplex(f_new, target_sum)

        val = _score(f_new)
        if val < best_val:
            best_val = val
            best_f = f_new.copy()
            print(f"[{time.time():.2f}] LP iteration new best={best_val:.6f}")

        f = f_new
        t_mix *= 0.9                # slowly reduce the mixing fraction

    return best_f, best_val

            print(f"[{time.time():.2f}] After coarse MT best={best_val:.6f}")

    # ------------------------------------------------------------------
    remain = deadline - time.time()
    if remain > 60.0 and budget_s >= 800:
        n_final = 12000
    elif remain > 30.0:
        n_final = 10000
    else:
        n_final = max(n0, int(remain * 20))
    target_final = np.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seed, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # try a few deterministic orderings
    candidates = [seq_big,
                  _alternating_order(seq_big),
                  _reversed_alternating(seq_big)]
    best_seq = seq_big
    best_score = _score(seq_big)
    for cand in candidates[1:]:
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand
    print(f"[{time.time():.2f}] Upscaled to {n_final} start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – long permutation‑annealing on the large sequence
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        perm_dead = deadline - 0.5
        a_perm, val_perm = _permutation_targeted_search(
            best_seq,
            perm_dead,
            start_temp=0.6,
            end_temp=1e-6,
            top_frac=0.04,
            low_frac=0.04,
        )
        if val_perm < best_score:
            best_score = val_perm
            best_seq = a_perm.copy()
            print(f"[{time.time():.2f}] After permutation search best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – final tiny mass‑transfer pass
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        mt_dead = time.time() + min(3.0, deadline - time.time() - 0.3)
        a_mt2, val_mt2 = _deterministic_mass_transfer(
            best_seq,
            mt_dead,
            target_final,
            start_temp=0.3,
            end_temp=1e-5,
            low_frac=0.05,
        )
        if val_mt2 < best_score:
            best_score = val_mt2
            best_seq = a_mt2.copy()
            print(f"[{time.time():.2f}] Final MT best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.flatnonzero(arr > 1e-12)
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Optional final evaluation logging
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for possible later reuse
    globals()["height_sequence_1"] = final_seq
    return final_seq
