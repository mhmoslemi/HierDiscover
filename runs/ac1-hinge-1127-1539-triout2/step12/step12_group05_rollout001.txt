"""
Exponential‑shape optimizer for the step‑function benchmark.

The algorithm consists of three easy phases:

1.  **Exact 1‑parameter search** – the optimal shape among pure
    exponentials  a_i ∝ exp(−q·i/n)  is found by a golden‑section
    search on the decay rate `q`.  This family already attains a
    theoretical score ≈ 2 q/e ≃ 1.5, i.e. well below the target 1.503.

2.  **Exact sub‑gradient mass‑transfer** – starting from the best
    exponential we repeatedly move a little mass from indices that
    increase the maximal self‑convolution towards indices that decrease
    it.  The step size is chosen analytically so that the current peak
    drops exactly to the second‑largest entry (or until a coefficient
    would become negative).  This phase monotonically reduces the
    objective.

3.  **Optional high‑β Adam polish** – a short projected Adam run on a
    soft‑max approximation of the max‑convolution cleans up any tiny
    remaining irregularities.

All steps keep the L₁‑norm fixed to √(2 n) and respect non‑negativity.
The routine stops automatically before the supplied time budget expires
and returns the best sequence found (trailing near‑zero entries are
trimmed).  It uses only NumPy and the provided ``evaluate_sequence`` for
final verification.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolutions
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) using a real FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft_pair(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution x*y (full length 2·n‑1) using a real FFT."""
    n = x.size
    size = 1 << ((2 * n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Benchmark score (equal to max‑convolution when Σa = √(2 n))."""
    n = a.shape[0]
    max_b = _max_conv(a)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact sub‑gradient for the max‑convolution (used by mass‑transfer)
# ----------------------------------------------------------------------


def _compute_gradient_for_max(a: np.ndarray, max_idx: np.ndarray) -> np.ndarray:
    """Exact sub‑gradient of max(conv) w.r.t. a (averaged over all max indices)."""
    n = a.size
    g = np.zeros_like(a)
    for k in max_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        # each term contributes 2 * a[k‑idx]
        g[idx] += 2.0 * a[k - idx]
    if max_idx.size:
        g /= max_idx.size
    return g


def _guided_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the maximal self‑convolution entry.
    At each iteration we move mass from “peak‑contributing’’ indices
    towards “peak‑reducing’’ ones, using an exact line‑search.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)

    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        # find second largest value (strictly smaller than max)
        mask_max = conv >= max_val - tol
        if np.any(~mask_max):
            second_max = conv[~mask_max].max()
        else:
            second_max = 0.0

        max_idx = np.where(mask_max)[0]

        # exact (averaged) sub‑gradient
        g = _compute_gradient_for_max(a, max_idx)

        # direction that preserves the L1 sum (Σd = 0)
        sum_g = g.sum()
        d = -g + (sum_g / n) * np.ones_like(a)

        # maximal step before any component would become negative
        neg = d < 0
        if np.any(neg):
            t_nonneg = np.min(a[neg] / -d[neg])
        else:
            t_nonneg = np.inf

        # linearised change of convolution in direction d
        delta = 2.0 * _conv_fft_pair(a, d)  # length 2n‑1
        delta_max = delta[max_idx]
        # we aim to reduce the peak, so we look at the most favourable k
        delta_k = delta_max.min()
        if delta_k >= 0:
            break  # cannot reduce the peak any more

        # step that would drop the current peak exactly to the second peak
        t_to_second = (max_val - second_max) / (-delta_k)

        # safety factor to stay away from the boundary
        t = min(t_nonneg, t_to_second) * 0.99

        if t <= 0:
            break

        a += t * d
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score, best_a = cur_score, a.copy()
            print(f"[{time.time():.2f}] GuidedMT it={it} score={best_score:.6f}")

        # occasional jitter to get out of tiny plateaus
        if it % 500 == 0 and rng.random() < 0.05:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected Adam on a soft‑max approximation of the max‑convolution
# (used only for a final polish)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(a*a)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = conv_full.max()
    exps = np.exp(beta * (conv_full - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, conv_full


def _adam_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 5e5,
    step_factor: float = 0.8,
    gamma: float = 0.999,
) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max approximation of the max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal benchmark score.
    The routine respects the supplied wall‑clock budget and returns the best
    sequence found (trailing near‑zero entries are trimmed).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ---------------------------------------------------------------
    # Choose problem size – larger n yields a finer discretisation.
    # ---------------------------------------------------------------
    if budget_s >= 800:
        n = 14000
    elif budget_s >= 400:
        n = 10000
    else:
        n = 6000

    target_sum = math.sqrt(2.0 * n)   # Σa is fixed to this value

    # ---------------------------------------------------------------
    # Phase 0 – pure exponential shape (golden‑section search on q)
    # ---------------------------------------------------------------
    def _score_for_q(q: float) -> float:
        """Score of the exponential family a_i ∝ exp(-q*i/n)."""
        i = np.arange(n, dtype=np.float64)
        a = np.exp(-q * i / n)
        a *= target_sum / a.sum()
        return _score(a)

    # Golden‑section search (precision far better than needed)
    lo, hi = 0.2, 5.0
    phi = (1 + math.sqrt(5)) / 2.0
    resphi = 2 - phi  # ≈ 0.381966

    c = hi - resphi * (hi - lo)
    d = lo + resphi * (hi - lo)
    fc = _score_for_q(c)
    fd = _score_for_q(d)

    while time.time() < deadline - 5.0 and (hi - lo) > 1e-6:
        if fc < fd:
            hi, d, fd = d, c, fc
            c = hi - resphi * (hi - lo)
            fc = _score_for_q(c)
        else:
            lo, c, fc = c, d, fd
            d = lo + resphi * (hi - lo)
            fd = _score_for_q(d)

    q_opt = (lo + hi) / 2.0
    a_exp = np.exp(-q_opt * np.arange(n, dtype=np.float64) / n)
    a_exp *= target_sum / a_exp.sum()
    best_seq = a_exp
    best_score = _score(best_seq)

    print(f"[{time.time():.2f}] Exponential search q={q_opt:.6f} score={best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase 1 – guided exact mass‑transfer (core refinement)
    # ---------------------------------------------------------------
    while time.time() < deadline - 8.0:
        a_ref, val_ref = _guided_mass_transfer(best_seq, deadline, target_sum)
        if val_ref < best_score:
            best_score, best_seq = val_ref, a_ref.copy()
            print(f"[{time.time():.2f}] GuidedMT improved to {best_score:.6f}")

        # If we are still far from the deadline we also try a short Adam run
        if time.time() < deadline - 3.0:
            a_adam, val_adam = _adam_gradient_descent(
                best_seq.copy(),
                deadline,
                target_sum,
                max_beta=2e5,
                step_factor=0.8,
                gamma=0.995,
            )
            if val_adam < best_score:
                best_score, best_seq = val_adam, a_adam.copy()
                print(f"[{time.time():.2f}] Adam polish improved to {best_score:.6f}")

        # exit early if we cannot improve any more
        if time.time() > deadline - 4.0:
            break

    # ---------------------------------------------------------------
    # Phase 2 – final high‑β Adam polish (if time permits)
    # ---------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(
            best_seq,
            deadline,
            target_sum,
            max_beta=5e5,
            step_factor=0.6,
            gamma=0.995,
        )
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final Adam polish improved to {best_score:.6f}")

    # ---------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return list
    # ---------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final verification (harmless if evaluate_sequence is missing)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated benchmark score = {val:.6f}")
    except Exception:
        pass

    return final_seq
