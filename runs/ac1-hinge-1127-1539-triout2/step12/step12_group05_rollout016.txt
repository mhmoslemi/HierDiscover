"""
Improved step‑function optimizer (≈ 100 % longer runtime).

Key ideas
---------

* **Extended guided mass‑transfer** – instead of using the averaged
  sub‑gradient of all active (max) constraints we evaluate the exact
  descent direction for each active index and pick the one that gives
  the largest linear reduction of its entry.  This often yields a much
  larger step size than the simple average and pushes the peak further
  down.

* **Adaptive schedule** – the budget is split dynamically between a
  tiny parametric pre‑search, a short “shake‑up” Adam run with a
  soft‑max approximation (β grows to 5·10⁵), the extended guided
  mass‑transfer, occasional random‑move perturbations and a final high‑β
  Adam polishing stage.

* **Larger problem size** – for a 1000 s budget we use 18 000 coefficients,
  which gives the optimiser many more degrees of freedom while keeping
  each FFT cheap (next power‑of‑two ≈ 2¹⁶).

The routine respects the simplex constraint Σa = √(2 n) and non‑negativity.
It returns the best sequence found before the deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) using a real FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft_pair(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution x*y (full length 2·n‑1) using a real FFT."""
    n = x.size
    size = 1 << ((2 * n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Benchmark score: 2·n·max(conv) / (sum(a)²)."""
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (cheap pre‑search)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _broken_power_law_sequence(
    n: int, p1: float, p2: float, alpha: float, target_sum: float
) -> np.ndarray:
    """Two‑segment power law, break point at α·n."""
    K = max(1, int(alpha * n))
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.empty_like(idx)
    a[:K] = idx[:K] ** (-p1)
    a[K:] = idx[K:] ** (-p2)
    a *= target_sum / a.sum()
    return a


def _uniform_sequence(n: int, target_sum: float) -> np.ndarray:
    return np.full(n, target_sum / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max approximation)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(a*a)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = conv_full.max()
    exps = np.exp(beta * (conv_full - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, conv_full


def _adam_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 5e5,
    step_factor: float = 0.8,
    gamma: float = 0.999,
) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max approximation of the max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient for a single active convolution entry
# ----------------------------------------------------------------------


def _compute_gradient_for_index(a: np.ndarray, k: int) -> np.ndarray:
    """Gradient of conv_k = Σ a_i a_{k-i} w.r.t. a."""
    n = a.size
    g = np.zeros_like(a)
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    g[idx] = 2.0 * a[k - idx]          # 2·a_{k−i}
    return g


def _compute_gradient_for_max(a: np.ndarray, max_idx: np.ndarray) -> np.ndarray:
    """Averaged sub‑gradient over all maximal indices."""
    n = a.size
    g = np.zeros_like(a)
    for k in max_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g[idx] += 2.0 * a[k - idx]
    if max_idx.size > 0:
        g /= max_idx.size
    return g


# ----------------------------------------------------------------------
# Extended exact guided mass‑transfer (Phase B)
# ----------------------------------------------------------------------


def _guided_mass_transfer_improved(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the maximal convolution entry.
    For each active (maximal) index we compute its exact gradient,
    form a zero‑sum feasible direction and pick the one that yields the
    largest linear decrease of the corresponding entry.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        mask_max = conv >= max_val - tol
        max_idxs = np.where(mask_max)[0]

        # second largest distinct value
        other = conv[~mask_max]
        second_max = other.max() if other.size else 0.0

        # ---- pick the most promising direction --------------------------------
        best_delta = None
        best_d = None
        best_t_to_second = None
        for k in max_idxs:
            g = _compute_gradient_for_index(a, k)
            sum_g = g.sum()
            d = -g + sum_g / n * np.ones_like(a)   # Σd = 0

            # maximal feasible step before hitting a zero bound
            neg = d < 0
            if np.any(neg):
                t_nonneg = np.min(a[neg] / -d[neg])
            else:
                t_nonneg = np.inf

            # linearised change of the whole convolution
            delta = 2.0 * _conv_fft_pair(a, d)
            delta_k = delta[k]                      # change of the chosen entry

            if delta_k < -1e-12:
                # step needed to bring this entry down to the second max
                t_second = (max_val - second_max) / (-delta_k) if delta_k != 0 else 0.0
                # keep the direction that offers the strongest negative slope
                if best_delta is None or delta_k < best_delta:
                    best_delta = delta_k
                    best_d = d
                    best_t_to_second = t_second
                    best_t_nonneg = t_nonneg

        # if no direction decreases the current peak, fall back to the
        # averaged gradient (same as the original method)
        if best_d is None:
            g = _compute_gradient_for_max(a, max_idxs)
            sum_g = g.sum()
            d = -g + sum_g / n * np.ones_like(a)
            neg = d < 0
            best_t_nonneg = np.min(a[neg] / -d[neg]) if np.any(neg) else np.inf
            delta = 2.0 * _conv_fft_pair(a, d)
            delta_k = delta[max_idxs].min()
            if delta_k >= -1e-12:
                break
            best_t_to_second = (max_val - second_max) / (-delta_k) if delta_k != 0 else 0.0
            best_d = d
            best_delta = delta_k

        # --------------------------------------------------------------
        t = min(best_t_nonneg, best_t_to_second) * 0.99
        if t <= 0:
            break

        a += t * best_d
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] GuidedMT it={it} score={best_score:.6f}")

        # occasional jitter to escape flat plateaus
        if it % 400 == 0 and rng.random() < 0.04:
            jitter = rng.normal(scale=target_sum * 0.0015, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Light global random exploration (optional)
# ----------------------------------------------------------------------


def _global_random_move(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
) -> tuple[np.ndarray, float]:
    """Very lightweight simulated‑annealing‑style mass moves."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    while time.time() < deadline - 0.5:
        top_k = max(1, int(0.02 * n))
        low_k = max(1, int(0.02 * n))
        donor = np.argsort(-a)[rng.integers(top_k)]
        receiver = np.argsort(a)[rng.integers(low_k)]
        if donor == receiver:
            continue
        delta = rng.random() * a[donor] * 0.05
        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] RandomMove new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with minimal benchmark score.
    The function respects the supplied time budget and returns the best
    sequence found (trailing near‑zero entries are trimmed).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose problem size – larger n gives more freedom.
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 15000
    else:
        n = 10000

    target_sum = math.sqrt(2.0 * n)          # simplex radius

    # ------------------------------------------------------------------
    # Initialisation – use previous best if available, otherwise a cheap
    # parametric family.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            seq = np.interp(new_x, old_x, prev)
        else:
            seq = prev.copy()
        seq = _project_to_simplex(seq, target_sum)
        val = _score(seq)
        best_seq, best_score = seq, val
        print(f"[{time.time():.2f}] Imported previous best, score={best_score:.6f}")

    # cheap parametric pre‑search (run only if time permits)
    if time.time() < deadline - 10:
        for p1 in (0.3, 0.5, 0.7):
            for p2 in (1.0, 1.5, 2.0):
                for alpha in (0.2, 0.5, 0.8):
                    seq = _broken_power_law_sequence(n, p1, p2, alpha, target_sum)
                    val = _score(seq)
                    if val < best_score:
                        best_seq, best_score = seq.copy(), val
        print(f"[{time.time():.2f}] After cheap pre‑search, best score={best_score:.6f}")

    # if still empty (unlikely) fall back to uniform
    if best_seq is None:
        best_seq = _uniform_sequence(n, target_sum)
        best_score = _score(best_seq)

    # ------------------------------------------------------------------
    # Phase A – short Adam “shake‑up” (≈10 % of remaining budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    adam_deadline = time.time() + 0.10 * remaining
    if adam_deadline - time.time() > 5.0:
        a_tmp, val_tmp = _adam_gradient_descent(
            best_seq.copy(),
            adam_deadline,
            target_sum,
            max_beta=5e5,
            step_factor=0.8,
            gamma=0.995,
        )
        if val_tmp < best_score:
            best_seq, best_score = a_tmp.copy(), val_tmp
            print(f"[{time.time():.2f}] Adam shake‑up improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – extended guided mass‑transfer (main refinement)
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        a_ref, val_ref = _guided_mass_transfer_improved(
            best_seq.copy(),
            deadline,
            target_sum,
        )
        if val_ref < best_score:
            best_seq, best_score = a_ref.copy(), val_ref
            print(f"[{time.time():.2f}] GuidedMT improved best={best_score:.6f}")

        # occasional random moves to jump out of tiny plateaus
        a_rand, val_rand = _global_random_move(best_seq.copy(), deadline, target_sum)
        if val_rand < best_score:
            best_seq, best_score = a_rand.copy(), val_rand
            print(f"[{time.time():.2f}] Random move improved best={best_score:.6f}")

        # early exit if we already beat the target
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Phase C – final polishing with a high‑β Adam run
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(
            best_seq.copy(),
            deadline,
            target_sum,
            max_beta=1e6,
            step_factor=0.6,
            gamma=0.995,
        )
        if val_fin < best_score:
            best_seq, best_score = a_fin.copy(), val_fin
            print(f"[{time.time():.2f}] Final Adam polish improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries and return
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final evaluation (harmless if evaluate_sequence missing)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated benchmark score = {val:.6f}")
    except Exception:
        pass

    return final_seq
