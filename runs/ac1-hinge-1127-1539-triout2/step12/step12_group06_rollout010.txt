def _adam_softmax_opt(a0, target_sum, deadline, max_beta=1e6, step_lr=0.1):
    a = _project_to_simplex(a0.copy(), target_sum)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    iteration = 1
    beta = 0.5
    while time.time()<deadline:
        grad,_ = _gradient_softmax(a, beta)
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad*grad)
        m_hat = m/(1-beta1**iteration)
        v_hat = v/(1-beta2**iteration)
        lr = step_lr * target_sum / (np.sqrt(v_hat)+eps)
        a -= lr * m_hat

"""
Hybrid optimizer for the non‑negative step‑function problem.

Main ideas
----------
1. Build a large diverse pool of seed sequences (power‑law, exponential,
   sinusoidal, random Dirichlet, linear) and keep the few best.
2. Run a long projected sub‑gradient descent (exact gradient of the peak
   of the convolution) on the chosen seeds.  Momentum and occasional jitter
   help to escape plateaus.
3. Upscale the current best vector to a larger length where more degrees of
   freedom are available; continue sub‑gradient descent on the enlarged
   vector.
4. Fine‑tune with an Adam optimiser on a smooth soft‑max approximation of the
   max‑convolution (annealed temperature).
5. Perform cheap random “re‑ordering’’ moves (rotations, swaps, random
   permutations) that keep the multiset of heights unchanged – these moves
   can lower the peak without changing the distribution.
6. A final deterministic mass‑transfer pass (greedy hill‑climbing) for a
   few seconds.
7. Trim trailing zeros and return the sequence.

The whole pipeline respects the 1000 s budget, always stopping before the
deadline.  The global variable ``height_sequence_1`` stores the best
candidate found so far for reuse in the next call.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i = (i+1)^(-p), normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i = exp(-lam*i), normalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _sinusoidal_sequence(n: int, power: float, target_sum: float) -> np.ndarray:
    """a_i = sin(pi*i/(n+1))**power, normalised."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = np.sin(np.pi * x / (n + 1)) ** power
    a *= target_sum / a.sum()
    return a


def _linear_decreasing_sequence(n: int, target_sum: float) -> np.ndarray:
    a = np.linspace(1.0, 0.0, n, dtype=np.float64)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place largest → left, 2nd largest → right, 3rd → next left, …."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


# ----------------------------------------------------------------------
# Scoring (objective = max convolution entry, sum normalised to √(2n))
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Return the objective value for a (max entry of a*a)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    # when s = sqrt(2n) the objective equals max_b
    # we keep the generic formula to be safe
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Exact max‑convolution and sub‑gradient of the peak
# ----------------------------------------------------------------------


def _max_conv_and_subgrad(a: np.ndarray):
    """
    Returns (conv, max_val, k_max, subgrad) where subgrad is the gradient
    of the peak entry of the convolution w.r.t. a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    kmax = int(np.argmax(conv))
    max_val = conv[kmax]

    grad = np.zeros_like(a)
    lo = max(0, kmax - (n - 1))
    hi = min(kmax, n - 1)
    idx = np.arange(lo, hi + 1)
    partner = a[kmax - idx]
    grad[lo:hi + 1] = 2.0 * partner
    return conv, max_val, kmax, grad


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact peak)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Momentum projected sub‑gradient descent on max(conv(a,a))."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0
    momentum = np.zeros_like(a)

    while time.time() < deadline:
        _, _, _, grad = _max_conv_and_subgrad(a)

        # momentum update
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_factor * target_sum / norm
            a -= lr * momentum
        else:
            # fallback – should rarely happen
            a -= step_factor * target_sum * grad / (np.linalg.norm(grad) + 1e-12)

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to shake off plateaus
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] SubGD it={it:5d}  best={best_val:.6f}")

        it += 1

    return best, best_val


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max) with Adam optimizer
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    Returns (grad, conv) where conv is the full convolution.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient: 2 * (a reversed) ⋆ p
    a_rev = a[::-1]
    grad = 2.0 * np.fft.irfft(np.fft.rfft(a_rev, n=size) *
                               np.fft.rfft(p, n=size), n=size)[: 2 * n - 1]
    grad = grad[n - 1: n - 1 + n]  # keep central n entries
    return grad, b


def _adam_softmax_opt(a0: np.ndarray,
                      deadline: float,
                      target_sum: float,
                      max_beta: float = 1e6,
                      base_lr: float = 0.5,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max approximation.  The temperature ``beta``
    is increased geometrically (annealing).
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 1
    beta = 0.5

    while time.time() < deadline:
        grad, _ = _softmax_gradient(a, beta)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        lr = base_lr * target_sum / (np.sqrt(v_hat) + eps)
        a -= lr * m_hat

        a = _project_to_simplex(a, target_sum)

        # jitter every few iterations
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        # update best
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam β={beta:.1e} it={it:5d} best={best_val:.6f}")

        # anneal beta
        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Random permutation hill‑climbing (swap / rotation)
# ----------------------------------------------------------------------


def _permutation_hill_climb(a0: np.ndarray,
                            deadline: float,
                            start_temp: float = 0.2,
                            end_temp: float = 1e-5,
                            top_frac: float = 0.04,
                            low_frac: float = 0.04) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style search that only permutes the multiset of
    heights.  Swaps that involve a high‑contributing index and a low‑contributing
    index are proposed.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # contributions of each index to the current peak
        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)

        if donor == receiver:
            continue

        # perform swap
        a[donor], a[receiver] = a[receiver], a[donor]
        new_score = _score(a)

        # temperature schedule
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] PermSwap new best={best_score:.6f}")

        # Metropolis acceptance
        if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / max(temp, 1e-12)):
            # keep the swap
            pass
        else:
            # revert
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score


def _random_rotation_search(a0: np.ndarray,
                            deadline: float,
                            start_temp: float = 0.15,
                            end_temp: float = 1e-5) -> tuple[np.ndarray, float]:
    """Try random cyclic rotations; accept if improving (SA)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    start_time = time.time()

    while time.time() < deadline:
        shift = rng.integers(0, a.shape[0])
        cand = np.roll(a, shift)
        new_score = _score(cand)

        # simple greedy – keep it if better
        if new_score < best_score:
            best_score = new_score
            best_a = cand.copy()
            a = cand
            print(f"[{time.time():.2f}] Rotation new best={best_score:.6f}")
        else:
            # occasional SA acceptance to escape local minima
            elapsed = time.time() - start_time
            tau = elapsed / (deadline - start_time)
            temp = start_temp * (end_temp / start_temp) ** tau
            if rng.random() < np.exp(-(new_score - best_score) / max(temp, 1e-12)):
                a = cand

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (greedy hill‑climbing)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.4,
                                end_temp: float = 1e-5,
                                low_frac: float = 0.04) -> tuple[np.ndarray, float]:
    """Greedy donor→receiver moves with an exponential temperature schedule."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    n = a.shape[0]

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # weight contributed by each index to the peak
        w = a[idx] + a[kmax - idx]
        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        # move a modest amount of mass
        max_delta = a[donor] * 0.5
        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with a low benchmark score.
    The routine respects the time budget and stores the best vector in the
    global variable ``height_sequence_1`` for reuse.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # tiny safety margin

    # ------------------------------------------------------------------
    # 1. Initial exploration length (moderate, fast)
    # ------------------------------------------------------------------
    n_expl = 6000  # a size that is quick for many short runs
    target_expl = np.sqrt(2.0 * n_expl)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # 2. Build a rich seed pool
    # ------------------------------------------------------------------
    seed_pool = []

    # power‑law family
    for p in np.linspace(0.35, 0.85, 11):
        seed_pool.append(_power_law_sequence(n_expl, float(p), target_expl))

    # exponential family
    for lam in np.linspace(0.0008, 0.018, 9):
        seed_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))

    # sinusoidal family
    for pw in np.linspace(0.5, 3.0, 7):
        seed_pool.append(_sinusoidal_sequence(n_expl, float(pw), target_expl))

    # linear decreasing
    seed_pool.append(_linear_decreasing_sequence(n_expl, target_expl))

    # random Dirichlet‑like points
    for _ in range(10):
        a = rng.random(n_expl)
        a *= target_expl / a.sum()
        seed_pool.append(a)

    # previous best, if any
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        seed_pool.append(_resize_sequence(prev, n_expl))

    # jittered copies of each seed (adds diversity)
    jittered = []
    for seq in seed_pool:
        jitter = rng.normal(scale=target_expl * 0.007, size=n_expl)
        js = seq + jitter
        js = np.maximum(js, 0.0)
        js = _project_to_simplex(js, target_expl)
        jittered.append(js)
    seed_pool.extend(jittered)

    # ------------------------------------------------------------------
    # 3. Quick evaluation of the pool – keep the K best seeds
    # ------------------------------------------------------------------
    K = 5
    scored = [(seq, _score(seq)) for seq in seed_pool]
    scored.sort(key=lambda x: x[1])
    best_candidates = [seq for seq, _ in scored[:K]]
    best_score = scored[0][1]
    best_seq = best_candidates[0].copy()
    print(f"[{time.time():.2f}] Initial best pool score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Short sub‑gradient runs on the moderate size (exploration)
    # ------------------------------------------------------------------
    while time.time() < deadline - 30.0:   # reserve time for later phases
        seed = rng.choice(best_candidates)
        run_dead = min(time.time() + 12.0, deadline - 20.0)
        if run_dead <= time.time():
            break
        a_opt, val_opt = _subgradient_descent(seed, run_dead, target_expl,
                                              step_factor=0.6, gamma=0.98)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Upscale to a larger length for extra freedom
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    # decide final length based on remaining budget
    if remaining > 400:
        n_final = 20000
    elif remaining > 250:
        n_final = 15000
    else:
        n_final = n_expl

    target_final = np.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # try a few systematic orderings
    candidates = [seq_big,
                  _alternating_order(seq_big),
                  _reversed_alternating(seq_big)]
    best_seq = seq_big
    best_score = _score(seq_big)
    for cand in candidates[1:]:
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand
    print(f"[{time.time():.2f}] Upscaled length={n_final} start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Long sub‑gradient descent on the enlarged vector
    # ------------------------------------------------------------------
    sub_dead = min(deadline - 120.0, time.time() + 150.0)
    if sub_dead > time.time():
        a_opt, val_opt = _subgradient_descent(best_seq, sub_dead,
                                              target_final,
                                              step_factor=0.5, gamma=0.99)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Sub‑GD (big) best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Adam optimiser on the smooth soft‑max (high β) for fine polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 70.0:
        adam_dead = min(deadline - 70.0, time.time() + 80.0)
        a_opt, val_opt = _adam_softmax_opt(best_seq, adam_dead,
                                           target_final,
                                           max_beta=2e6,
                                           base_lr=0.6)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Adam‑softmax best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Random rotation search (very cheap)
    # ------------------------------------------------------------------
    if time.time() < deadline - 50.0:
        rot_dead = time.time() + min(30.0, deadline - time.time() - 30.0)
        a_opt, val_opt = _random_rotation_search(best_seq, rot_dead)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Rotation search best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 9. Targeted permutation swaps (simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        perm_dead = time.time() + min(20.0, deadline - time.time() - 20.0)
        a_opt, val_opt = _permutation_hill_climb(best_seq, perm_dead,
                                                 start_temp=0.12,
                                                 end_temp=1e-5,
                                                 top_frac=0.03,
                                                 low_frac=0.03)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Permutation search best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 10. Final deterministic mass‑transfer pass
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        mt_dead = time.time() + min(8.0, deadline - time.time() - 5.0)
        a_opt, val_opt = _deterministic_mass_transfer(best_seq, mt_dead,
                                                      target_final,
                                                      start_temp=0.35,
                                                      end_temp=1e-5,
                                                      low_frac=0.03)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Final MT best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 11. Trim trailing zeros for compactness
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # 12. Log final evaluation (using the provided evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    globals()["height_sequence_1"] = final_seq
    return final_seq
