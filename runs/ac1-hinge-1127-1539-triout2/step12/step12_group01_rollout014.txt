def greedy_smooth(a, target_sum, max_iters=5000):
    n = a.shape[0]
    b = _conv_fft(a)
    cur_val = _score(a)
    for it in range(max_iters):
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n-1))
        hi = min(k_max, n-1)
        idx = np.arange(lo, hi+1)
        contrib = a[idx] * a[k_max-idx]
        donor = idx[np.argmax(contrib)]
        # also track other index j = k_max - donor
        # compute both contributions; maybe move from donor only
        receiver = int(np.argmin(a))
        if donor == receiver:
            # choose next min
            receiver = int(np.argsort(a)[1])
        max_delta = min(a[donor]*0.05, target_sum*0.005)
        if max_delta <=0:
            break
        delta = max_delta
        # compute new b after moving delta from donor to receiver:
        b_new = b.copy()
        # linear terms:
        # For each index i from 0..n-1, b[donor + i] -= 2 delta a[i]
        # b[receiver + i] += 2 delta a[i]
        # Quadratic terms: as before
        # But we also need to update a[donor] and a[receiver] after move.
        # However note that a_i used in linear update includes original a values.
        # We'll compute exactly as in annealing code.
        b_new[donor: donor + n] -= 2.0 * delta * a
        b_new[receiver: receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta
        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))
        if new_val < cur_val:
            # accept
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
        else:
            # try smaller delta maybe
            # if not improved break
            break
    return a, cur_val

"""
Improved non‑negative step‑function optimizer.

The algorithm combines several ideas:

1. **Exponent search** – a quick scan of power‑law exponents on a moderate
   size (≈ 4000) to find a good analytical shape for the sequence.

2. **Multi‑scale optimisation** – a small‑scale (≈ 1/5 of the final length)
   optimisation (mirror descent → sub‑gradient → annealing) is performed on
   the best exponent shape, then the result is linearly interpolated to the
   final length.

3. **Large‑scale annealing** – a long simulated‑annealing mass‑transfer
   phase (the same routine used previously) is run on the full‑size
   sequence.

4. **Greedy smoothing** – after annealing we repeatedly move a fraction of
   the most contributing “donor” mass (with respect to the current maximal
   convolution entry) to the smallest entry, accepting only improving
   moves.  This deterministic refinement often lowers the max‑convolution
   a little further.

5. **Multiple restarts** – the whole procedure is repeated a few times
   (time‑budget‑aware) and the best sequence is retained.

All phases respect the global time budget (default 1000 s).  The final
sequence is trimmed of trailing zeros and stored in the global variable
`height_sequence_1` for warm‑starting future runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Objective  2·n·max(a*a) / (sum(a)²).
    For Σa = √(2n) this equals max(a*a).
    """
    n = a.shape[0]
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of the max‑convolution.
    Returns (grad, full_convolution) where grad has the same length as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    grad_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(grad_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – exponential (mirror) descent on the soft‑max surrogate
# ----------------------------------------------------------------------


def exponential_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Mirror descent on the soft‑max surrogate."""
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        # occasional jitter
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpDescent it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Sub‑gradient descent on the true max‑convolution entry."""
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        elapsed = time.time() - start
        total = deadline - start
        eta_start = 2.0
        eta_end = 0.2
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase D – improved simulated‑annealing mass‑transfer refinement
# ----------------------------------------------------------------------


def anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Long‑run annealing that moves tiny amounts of mass from “bad’’ to “good’’
    indices.  Acceptance is based on the *current* energy (Metropolis) and
    the temperature decays with a √‑schedule.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    # start temperature at a large fraction of the current score
    T0 = cur_val * 0.80

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        # non‑linear cooling: keep temperature high longer
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        # locate the current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – pick an index with relatively small height
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # size of the transfer (scaled by temperature)
        max_delta = min(a[donor] * 0.05, target_sum * 0.005)
        if max_delta <= 0:
            continue
        # scale factor grows with temperature
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # --- exact O(n) update of the convolution -------------------------
        b_new = b.copy()
        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        # Metropolis acceptance based on *current* value
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved best={best_val:.6f}")

        # periodic full recomputation to avoid drift
        if it % 5000 == 0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy deterministic smoothing (post‑anneal)
# ----------------------------------------------------------------------


def greedy_smoothing(a0: np.ndarray, target_sum: float, deadline: float):
    """Deterministic moves that always improve the max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_val = _score(a)

    start = time.time()
    it = 0
    while time.time() < deadline:
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contributions of each participating index to the current peak
        contrib = a[idx] * a[k_max - idx]
        donor = idx[np.argmax(contrib)]

        # pick a receiver with a very small height
        receiver = int(np.argmin(a))
        if donor == receiver:
            # second smallest if necessary
            sorted_idx = np.argsort(a)
            for cand in sorted_idx:
                if cand != donor:
                    receiver = cand
                    break

        max_delta = min(a[donor] * 0.05, target_sum * 0.005)
        if max_delta <= 0:
            break
        delta = max_delta

        # propose the move (exact O(n) update of convolution)
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        if new_val < cur_val:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
        else:
            # try smaller fractions of the same delta (up to 4 attempts)
            improved = False
            for factor in (0.5, 0.25, 0.125):
                delta2 = delta * factor
                if delta2 <= 1e-12:
                    break
                b_test = b.copy()
                b_test[donor:donor + n] -= 2.0 * delta2 * a
                b_test[receiver:receiver + n] += 2.0 * delta2 * a
                b_test[2 * donor] += delta2 * delta2
                b_test[2 * receiver] += delta2 * delta2
                if donor != receiver:
                    b_test[donor + receiver] -= 2.0 * delta2 * delta2
                val_test = float(2.0 * n * b_test.max() / (target_sum * target_sum))
                if val_test < cur_val:
                    a[donor] -= delta2
                    a[receiver] += delta2
                    b = b_test
                    cur_val = val_test
                    improved = True
                    break
            if not improved:
                # no improving move – stop early
                break

        # occasional renormalisation to avoid numerical drift
        if it % 500 == 0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = _score(a)

        it += 1

    return a, cur_val


# ----------------------------------------------------------------------
# Main entry point – propose_candidate
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # small safety margin

    # --------------------------------------------------------------
    # Choose final problem size
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # QUICK exponent scan on a moderate size (≈4000)
    # --------------------------------------------------------------
    n_scan = min(4000, n)
    exponents = np.linspace(-0.9, -0.05, 18)   # 18 candidate exponents
    best_exp = -0.55
    best_exp_score = np.inf
    for e in exponents:
        a_tmp = np.arange(1, n_scan + 1, dtype=np.float64) ** e
        a_tmp = a_tmp * (target_sum / a_tmp.sum())
        val = _score(a_tmp)
        if val < best_exp_score:
            best_exp_score = val
            best_exp = e
    print(f"[{time.time():.2f}] Best exponent from scan: {best_exp:.3f} score≈{best_exp_score:.6f}")

    # --------------------------------------------------------------
    # Small‑scale multi‑restart optimisation (≈25 % of total time)
    # --------------------------------------------------------------
    small_deadline = min(deadline, start + 0.25 * (deadline - start))
    n_small = max(2000, n // 5)
    best_small_seq = None
    best_small_val = np.inf
    restarts_small = 3
    for rs in range(restarts_small):
        if time.time() >= small_deadline:
            break
        # initialise with the best power‑law exponent
        a0 = np.arange(1, n_small + 1, dtype=np.float64) ** best_exp
        # add a tiny random perturbation for diversity
        rng = np.random.default_rng()
        a0 *= 1.0 + rng.normal(scale=0.02, size=a0.shape)
        a0 = np.maximum(a0, 0.0)
        a0 = a0 * (target_sum / a0.sum())

        # Phase A – exponential descent
        a1, v1 = exponential_descent(a0, target_sum, small_deadline)
        # Phase B – sub‑gradient descent
        a2, v2 = subgradient_descent(a1, target_sum, small_deadline)
        # Phase D – annealing
        a3, v3 = anneal_mass_transfer(a2, target_sum, small_deadline)

        if v3 < best_small_val:
            best_small_val = v3
            best_small_seq = a3.copy()
            print(f"[{time.time():.2f}] Small‑scale restart {rs} best score={best_small_val:.6f}")

    # --------------------------------------------------------------
    # Upscale to the final size
    # --------------------------------------------------------------
    if best_small_seq is not None:
        a_full = _resize_sequence(best_small_seq, n)
        a_full = a_full * (target_sum / a_full.sum())
    else:
        # fall back to pure power‑law if the small scale failed
        a_full = (np.arange(1, n + 1, dtype=np.float64) ** best_exp)
        a_full = a_full * (target_sum / a_full.sum())

    # --------------------------------------------------------------
    # Large‑scale annealing + greedy refinement (remaining time)
    # --------------------------------------------------------------
    cur_seq = a_full
    best_seq = cur_seq.copy()
    best_val = _score(cur_seq)
    print(f"[{time.time():.2f}] Starting large‑scale score={best_val:.6f}")

    remaining = deadline - time.time()
    if remaining <= 0:
        # time exhausted – return what we have
        arr = np.array(best_seq)
        nz = np.where(arr > 1e-12)[0]
        if nz.size:
            arr = arr[: nz[-1] + 1]
        final_seq = arr.tolist()
        globals()["height_sequence_1"] = final_seq
        return final_seq

    # allocate ~70 % of remaining time to annealing,
    # the rest to greedy smoothing
    anneal_deadline = time.time() + 0.7 * remaining
    cur_seq, cur_val = anneal_mass_transfer(cur_seq, target_sum, anneal_deadline)
    if cur_val < best_val:
        best_val = cur_val
        best_seq = cur_seq.copy()
        print(f"[{time.time():.2f}] After annealing best score={best_val:.6f}")

    # Greedy deterministic smoothing on whatever time is left
    greedy_deadline = deadline
    cur_seq, cur_val = greedy_smoothing(cur_seq, target_sum, greedy_deadline)
    if cur_val < best_val:
        best_val = cur_val
        best_seq = cur_seq.copy()
        print(f"[{time.time():.2f}] After greedy smoothing best score={best_val:.6f}")

    # --------------------------------------------------------------
    # Final trimming and evaluation
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for warm‑starts in future runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
