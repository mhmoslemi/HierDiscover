"""
Improved step‑function optimizer.

The algorithm follows a multi‑stage pipeline:
1. **Seed generation** – a diverse pool (power‑law, exponential,
   quadratic‑exponential, random simplex) of modest size (`n_expl`).
2. **Quick evaluation** – keep the few best seeds by the true objective.
3. **Projected annealed gradient** – short ADAM‑style runs on the best
   seeds (soft‑max approximation of the max‑convolution).
4. **Greedy mass‑transfer** – simulated‑annealing moves of mass from
   heavy to light contributors.
5. **Upscaling** – linear interpolation to a larger length (`n_final`),
   projection onto the ℓ¹‑simplex, and a deterministic reordering that
   spreads large entries.
6. **Parametric family search** – exhaustively scan a few one‑parameter
   families (exponential, power‑law, quadratic‑exponential) on the
   large scale, keeping the ordering with the smallest max‑convolution.
7. **Permutation refinement** – many random reorderings (FFT‑based
   evaluation) to further flatten the convolution.
8. **Projected sub‑gradient descent** – exact sub‑gradient steps with
   momentum on the large problem.
9. **Final mass‑transfer** – a short SA pass to escape any remaining
   plateaus.
All steps respect non‑negativity and the ℓ¹‑simplex constraint.
The routine stops automatically before the given time budget.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (linear convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a)/(∑a)²."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised to the required ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _quad_exp_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i²) (Gaussian‑like) then renormalised."""
    a = np.exp(-lam * (np.arange(n, dtype=np.float64) ** 2))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex (Dirichlet‑like)."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """
    Spread the large values uniformly by placing them on a cyclic sequence
    with a step coprime to the length (≈√n).
    """
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Annealed gradient descent (soft‑max approximation) – Adam‑style
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.9,
                               gamma: float = 0.98) -> tuple[np.ndarray, float]:
    """Momentum projected GD on the soft‑max approximation (Adam‑like)."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    epsilon = 1e-8
    beta1 = 0.9
    beta2 = 0.999
    it = 0
    while time.time() < deadline:
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_b = np.max(b)
        exps = np.exp(beta * (b - max_b))
        p = exps / np.sum(exps)               # soft‑max over convolution entries

        # gradient of the smoothed max w.r.t. a
        a_rev = a[::-1]
        Ar_fft = np.fft.rfft(a_rev, n=size)
        Pr_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        # Adam update (projected)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        step = step_base / (np.sqrt(v_hat) + epsilon)

        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:7.1f} score={best_val:.6f}")

        # beta schedule – exponential growth
        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that uses the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        # indices attaining (almost) the maximum
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]

        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy mass‑transfer with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.5,
                                      end_temp: float = 1e-4,
                                      low_frac: float = 0.06) -> tuple[np.ndarray, float]:
    """
    Greedy mass moves from a “donor’’ (large contribution to the peak)
    to a “receiver’’ (small contribution) with a Metropolis acceptance rule.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()

    conv = _conv_fft(a)          # current convolution
    n = a.shape[0]

    while time.time() < deadline:
        # exponential temperature decay
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # locate the current peak
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to the peak (product a_i * a_{kmax-i})
        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        # propose moving a small amount of mass
        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        # apply tentative move
        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        # full recompute (still cheap with FFT)
        conv = _conv_fft(a)
        new_score = _score(a)

        # Metropolis acceptance
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Random permutation search (fast heuristic)
# ----------------------------------------------------------------------


def _permutation_search(base_seq: np.ndarray,
                        trials: int,
                        deadline: float) -> tuple[np.ndarray, float]:
    """Try a number of random reorderings of `base_seq` and keep the best."""
    best_seq = base_seq.copy()
    best_val = _max_convolution_fft(best_seq)
    rng = np.random.default_rng()

    for _ in range(trials):
        if time.time() >= deadline:
            break
        perm = rng.permutation(base_seq)
        val = _max_convolution_fft(perm)
        if val < best_val:
            best_val = val
            best_seq = perm.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple parametric family search on a given length
# ----------------------------------------------------------------------


def _param_family_search(n: int,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Scan a few one‑parameter families (exp, power, quadratic‑exp) on length `n`.
    For each candidate the three deterministic orderings are tried.
    Returns the best sequence found and its objective.
    """
    rng = np.random.default_rng()
    best_seq = None
    best_obj = np.inf

    # families: (name, generator, low, high, logspace_flag)
    families = [
        ("exp", _exponential_sequence, 1e-6, 0.05, True),
        ("pow", _power_law_sequence, 0.1, 2.0, False),
        ("quad", _quad_exp_sequence, 1e-6, 0.02, True),
    ]

    for name, gen, low, high, logscale in families:
        # coarse grid
        for i in range(30):
            if time.time() >= deadline:
                break
            if logscale:
                param = 10 ** np.random.uniform(np.log10(low), np.log10(high))
            else:
                param = low + (high - low) * i / 29.0
            seq = gen(n, param, target_sum)

            # try three deterministic orderings
            candidates = [seq, _alternating_order(seq), _even_spread_order(seq)]
            for cand in candidates:
                obj = _score(cand)
                if obj < best_obj:
                    best_obj = obj
                    best_seq = cand.copy()
        # optional refinement around the best param of this family (if time permits)
        # we skip for simplicity – the coarse grid already gives a good candidate.
    return best_seq, best_obj


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine follows a multi‑stage schedule and never exceeds `budget_s`
    seconds.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin
    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Phase 1 – seed generation and quick evaluation
    # ------------------------------------------------------------------
    # Choose an exploratory size that fits comfortably in the time budget.
    if budget_s >= 800:
        n_expl = 8000
    elif budget_s >= 500:
        n_expl = 6000
    else:
        n_expl = 4000
    target_expl = math.sqrt(2.0 * n_expl)

    init_pool = []

    # power‑law family
    for p in np.linspace(0.2, 2.0, 12):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))

    # exponential family
    for lam in np.linspace(0.0005, 0.03, 12):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))

    # quadratic‑exponential family
    for lam in np.linspace(0.0005, 0.02, 8):
        init_pool.append(_quad_exp_sequence(n_expl, float(lam), target_expl))

    # random simplex points
    for _ in range(14):
        init_pool.append(_random_simplex(n_expl, target_expl, rng))

    # include previous global best if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_expl))

    # jitter copies to increase diversity
    expanded = []
    for seq in init_pool:
        jitter = rng.normal(scale=target_expl * 0.03, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        expanded.append(seq_j)
    init_pool.extend(expanded)

    # quick evaluation to keep a handful of promising seeds
    best_seq = None
    best_score = np.inf
    for seq in init_pool:
        sc = _score(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
    print(f"[{time.time():.2f}] Initial best score from pool = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – short annealed‑gradient runs on the best few seeds
    # ------------------------------------------------------------------
    # we run ADAM‑style gradient on up to three seeds
    seeds_to_opt = [best_seq]
    # add a couple of random seeds for diversity
    for _ in range(2):
        seeds_to_opt.append(rng.choice(init_pool))

    for seed in seeds_to_opt:
        if time.time() >= deadline - 20.0:
            break
        run_deadline = min(time.time() + 12.0, deadline - 10.0)
        a_opt, val_opt = _annealed_gradient_descent(seed, run_deadline, target_expl,
                                                    max_beta=8e5, step_factor=0.9, gamma=0.98)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Gradient improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – greedy mass‑transfer refinement (small scale)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 12.0:
        mt_deadline = time.time() + min(8.0, deadline - time.time() - 6.0)
        a_ref, ref_val = _mass_transfer_simulated_annealing(best_seq, mt_deadline,
                                                            target_expl,
                                                            start_temp=0.45,
                                                            end_temp=1e-4,
                                                            low_frac=0.07)
        if ref_val < best_score:
            best_score = ref_val
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] MT refined best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – upscale to a larger problem size
    # ------------------------------------------------------------------
    # decide final length based on remaining budget
    remaining = deadline - time.time()
    if remaining > 180.0:
        n_final = 25000
    elif remaining > 120.0:
        n_final = 20000
    elif remaining > 80.0:
        n_final = 15000
    elif remaining > 50.0:
        n_final = 12000
    else:
        n_final = n_expl

    target_final = math.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # try deterministic reorderings and keep the best
    candidates = [seq_big, _alternating_order(seq_big), _even_spread_order(seq_big)]
    seq_big = min(candidates, key=_max_convolution_fft)
    print(f"[{time.time():.2f}] Upscaled to {n_final} elements, start max={_max_convolution_fft(seq_big):.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – parametric family search on the large scale
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        fam_deadline = time.time() + 20.0
        seq_fam, fam_val = _param_family_search(n_final, target_final, fam_deadline)
        if seq_fam is not None:
            # pick the ordering that gives the lowest max‑convolution
            cand_seq = min([seq_fam, _alternating_order(seq_fam), _even_spread_order(seq_fam)],
                           key=_max_convolution_fft)
            cand_val = _max_convolution_fft(cand_seq)
            cur_val = _max_convolution_fft(seq_big)
            if cand_val < cur_val:
                seq_big = cand_seq.copy()
                print(f"[{time.time():.2f}] Parametric family improved max={cand_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – permutation refinement (random shuffles)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        perm_deadline = deadline - 20.0
        # allocate a modest number of trials per second left
        trials = int((perm_deadline - time.time()) // 0.02)  # ~20 ms per trial
        trials = max(30, min(trials, 800))
        seq_big, _ = _permutation_search(seq_big, trials, perm_deadline)
        print(f"[{time.time():.2f}] After permutation search max={_max_convolution_fft(seq_big):.6f}")

    # ------------------------------------------------------------------
    # Phase 7 – projected sub‑gradient descent on the large scale
    # ------------------------------------------------------------------
    if time.time() < deadline - 12.0:
        sub_deadline = deadline - 8.0
        a_sub, sub_val = _subgradient_descent(seq_big, sub_deadline, target_final,
                                              step_factor=0.45, gamma=0.995)
        if sub_val < best_score:
            best_score = sub_val
            seq_big = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑grad polished best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 8 – final greedy mass‑transfer (if any time left)
    # ------------------------------------------------------------------
    if time.time() < deadline - 4.0:
        mt_deadline = deadline - 2.0
        a_mt, mt_val = _mass_transfer_simulated_annealing(seq_big, mt_deadline,
                                                          target_final,
                                                          start_temp=0.25,
                                                          end_temp=1e-5,
                                                          low_frac=0.05)
        if mt_val < best_score:
            best_score = mt_val
            seq_big = a_mt.copy()
            print(f"[{time.time():.2f}] Final MT best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # ------------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Optional final evaluation (purely for logging)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose the sequence for possible reuse in a later run
    globals()["height_sequence_1"] = final_seq
    return final_seq
