"""
Propose a step‑function sequence that minimises the benchmark score.

The algorithm proceeds in four stages:

1. **Problem size** – choose n (a power of two) based on the allotted time,
   and fix the L1‑norm Σa = √(2 n).

2. **Extensive parametric pre‑search** – sample many classic families
   (power‑law, exponential‑power‑law, broken power‑law, sine‑power,
   arcsine‑type, semicircle‑type).  The best sequence from this cheap
   scan is kept as the starting point.

3. **High‑β Adam on the soft‑max approximation** – a projected Adam
   optimiser uses a temperature‑controlled soft‑max of the convolution
   entries as a smooth surrogate for the max‑operator.  The temperature
   β is doubled periodically up to a very large value, so the optimiser
   eventually tracks the true maximum.

4. **Exact guided mass‑transfer** – using the exact sub‑gradient of the
   max‑convolution we move infinitesimal mass from indices that enlarge
   the peak towards indices that do not, with a line‑search that drives
   the current maximal entry down to the second‑largest one.  This step
   monotonically reduces the score and is applied until the time budget
   expires.

Finally the trailing near‑zero entries are trimmed and the list is
returned.  All operations respect non‑negativity and the L1‑norm
constraint.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft_pair(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution x * y using a real FFT."""
    n = x.size
    size = 1 << ((2 * n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the auto‑convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Benchmark score (equal to the maximal convolution entry when Σa=√(2n))."""
    n = a.shape[0]
    max_b = _max_conv(a)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (dense pre‑search)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _broken_power_law_sequence(
    n: int, p1: float, p2: float, alpha: float, target_sum: float
) -> np.ndarray:
    K = max(1, int(alpha * n))
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.empty_like(idx)
    a[:K] = idx[:K] ** (-p1)
    a[K:] = idx[K:] ** (-p2)
    a *= target_sum / a.sum()
    return a


def _sin_power_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    # symmetric sine bell: sin(π·t) where t∈[0,1]
    t = (np.arange(1, n + 1, dtype=np.float64) / (n + 1)) * np.pi
    a = np.sin(t) ** p
    a *= target_sum / a.sum()
    return a


def _arcsine_power_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    # arcsine‑type density ∝ 1/√(x(1‑x))
    i = np.arange(n, dtype=np.float64)
    # map i → x ∈ (0,1)
    x = (i + 0.5) / n
    a = (x * (1.0 - x)) ** (-p / 2.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a *= target_sum / a.sum()
    return a


def _semicircle_power_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    # semicircle‑type density ∝ √(x(1‑x))
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = (x * (1.0 - x)) ** (p / 2.0)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Adam‑style optimiser on the soft‑max surrogate
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(a*a)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = conv_full.max()
    exps = np.exp(beta * (conv_full - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, conv_full


def _adam_opt(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 5e5,
    step_factor: float = 0.8,
    gamma: float = 0.999,
) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max surrogate of the max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact guided mass‑transfer (monotone reduction of the peak)
# ----------------------------------------------------------------------


def _compute_gradient_for_max(a: np.ndarray, max_idx: np.ndarray) -> np.ndarray:
    """Exact sub‑gradient of max(conv) w.r.t. a (averaged over max indices)."""
    n = a.size
    g = np.zeros_like(a)
    for k in max_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g[idx] += 2.0 * a[k - idx]
    if max_idx.size:
        g /= max_idx.size
    return g


def _guided_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Greedy reduction of the current maximal convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        mask_max = conv >= max_val - tol
        # second largest (excluding the current plateau)
        if np.any(~mask_max):
            second_max = conv[~mask_max].max()
        else:
            second_max = 0.0

        max_idx = np.where(mask_max)[0]
        g = _compute_gradient_for_max(a, max_idx)

        # direction that preserves the simplex (Σd = 0)
        sum_g = g.sum()
        d = -g + sum_g / n * np.ones_like(a)

        # largest step before any component would become negative
        neg = d < 0
        if np.any(neg):
            t_nonneg = np.min(a[neg] / -d[neg])
        else:
            t_nonneg = np.inf

        # linearised effect on the convolution (ignoring quadratic δ² terms)
        delta = 2.0 * _conv_fft_pair(a, d)  # length 2n‑1
        delta_max = delta[max_idx]
        delta_k = delta_max.min()  # most favourable index among the plateau

        if delta_k >= 0:
            # we cannot push the peak down using this direction
            break

        # step that would bring the current peak down to the second‑largest value
        t_to_second = (max_val - second_max) / (-delta_k)
        t = min(t_nonneg, t_to_second) * 0.99  # safety factor

        if t <= 0:
            break

        a += t * d
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] GuidedMT it={it} score={best_score:.6f}")

        # occasional random jitter to escape flat plateaux
        if it % 500 == 0 and rng.random() < 0.05:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal benchmark score.
    The routine respects the supplied wall‑clock budget and returns the best
    sequence found (trailing near‑zero entries are trimmed).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.3  # tiny safety margin

    # ------------------------------------------------------------------
    # Choose a power‑of‑two size – FFTs are fastest for such lengths.
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 16384
    elif budget_s >= 400:
        n = 8192
    else:
        n = 4096

    target_sum = math.sqrt(2.0 * n)  # Σa is fixed to this value

    # ------------------------------------------------------------------
    # Phase 0 – exhaustive parametric pre‑search (cheap, dense scan)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    rng = np.random.default_rng()

    # (a) power‑law (grid)
    for p in np.linspace(0.15, 2.5, 30):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # (b) exponential‑power‑law (grid)
    for p in np.linspace(0.15, 2.5, 20):
        for q in np.linspace(0.0, 10.0, 12):
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # (c) broken power‑law (grid)
    for p1 in np.linspace(0.15, 1.2, 12):
        for p2 in np.linspace(0.8, 3.0, 12):
            for alpha in np.linspace(0.1, 0.9, 9):
                seq = _broken_power_law_sequence(
                    n, float(p1), float(p2), float(alpha), target_sum
                )
                val = _score(seq)
                if val < best_score:
                    best_score, best_seq = val, seq.copy()

    # (d) sine‑power (grid)
    for p in np.linspace(0.2, 5.0, 20):
        seq = _sin_power_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # (e) arcsine‑type (random)
    for _ in range(300):
        p = rng.uniform(0.2, 3.0)
        seq = _arcsine_power_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # (f) semicircle‑type (random)
    for _ in range(300):
        p = rng.uniform(0.2, 3.0)
        seq = _semicircle_power_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # (g) warm‑start from a previously stored best, if any
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score, best_seq = val, resized.copy()

    print(f"[{time.time():.2f}] Pre‑search finished: best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase A – moderate‑β Adam runs (exploratory restarts)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    adam_phase_end = time.time() + 0.20 * remaining

    while time.time() < adam_phase_end - 1.0:
        jitter = rng.normal(scale=target_sum * 0.015, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _adam_opt(
            start_vec,
            adam_phase_end,
            target_sum,
            max_beta=5e5,
            step_factor=0.85,
            gamma=0.995,
        )
        if val_opt < best_score:
            best_score, best_seq = val_opt, a_opt.copy()
            print(f"[{time.time():.2f}] Adam restart improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – guided exact mass‑transfer (core refinement)
    # ------------------------------------------------------------------
    while time.time() < deadline - 2.5:
        a_ref, val_ref = _guided_mass_transfer(
            best_seq,
            deadline,
            target_sum,
        )
        if val_ref < best_score:
            best_score, best_seq = val_ref, a_ref.copy()
            print(f"[{time.time():.2f}] Guided MT improved best = {best_score:.6f}")

        # Tiny random shake to avoid getting stuck
        if rng.random() < 0.03:
            jitter = rng.normal(scale=target_sum * 0.0015, size=n)
            best_seq = np.maximum(best_seq + jitter, 0.0)
            best_seq = _project_to_simplex(best_seq, target_sum)

    # ------------------------------------------------------------------
    # Phase C – final high‑β Adam polishing (few seconds left)
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_opt(
            best_seq,
            deadline,
            target_sum,
            max_beta=2e6,
            step_factor=0.6,
            gamma=0.995,
        )
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final Adam polish improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return list
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # Optional sanity check (harmless if evaluate_sequence is not defined)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated benchmark score = {val:.6f}")
    except Exception:
        pass

    return final_seq
