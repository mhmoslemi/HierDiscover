"""
Propose a non‑negative step function whose self‑convolution has a minimal
maximum entry.  

The search consists of six stages:

1. Fast parametric sweep (pure power‑law, power‑law+exponential tail,
   simple sine).  The best candidate is kept.
2. Adam‑style projected gradient descent on a soft‑max approximation
   of the max‑convolution (several restarts).
3. Exact projected sub‑gradient descent (true max‑convolution).
4. Deterministic “mass‑transfer” step that reduces the current peak to
   the second‑largest value.
5. Random hill‑climbing based on incremental convolution updates.
   Each move transfers a small amount of mass from a “donor’’ index to a
   “receiver’’ index chosen from the active maximal convolution indices.
   The update uses an O(n) incremental formula, so thousands of moves can
   be tried quickly.
6. Final polishing with a high‑β Adam run.

All steps respect the global time budget (default 1000 s).  The returned
list is trimmed of trailing near‑zero entries; the harness will compute
the official score with ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT based) – used for full evaluations
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Scoring (the objective is exactly the maximal convolution entry)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Return the evaluation score (max‑convolution value)."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for the quick pre‑search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    """A smooth sine‑shaped candidate (always positive)."""
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max based gradient (used by Adam)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run a few short Adam restarts from jittered starts."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=2e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         max_iter: int = 5000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.flatnonzero(conv >= max_val - 1e-12)

        # second‑largest value
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()
        delta_target = max_val - second_max

        if delta_target < 1e-14:
            break

        k = idx_max[rng.integers(len(idx_max))]

        n = a.shape[0]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k - idx]

        norm_g2 = np.dot(g, g)
        if norm_g2 == 0.0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = _project_to_simplex(a + jitter, target_sum)
            it += 1
            continue

        step = 0.9 * delta_target / norm_g2
        step = min(step, target_sum * 0.4 / (np.linalg.norm(g) + 1e-12))

        for _ in range(6):
            a_candidate = _project_to_simplex(a - step * g, target_sum)
            cand_score = _score(a_candidate)
            if cand_score < best_score - 1e-12:
                a = a_candidate
                best_a = a_candidate.copy()
                best_score = cand_score
                print(f"[{time.time():.2f}] Subgrad it={it:5d} step={step:.3e} score={best_score:.6f}")
                break
            step *= 0.5
        else:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = _project_to_simplex(a + jitter, target_sum)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (as in the reference code)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]                 # a_{k‑i}
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # choose receiver with smallest partner (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom

        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Random hill‑climbing with incremental convolution updates
# ----------------------------------------------------------------------


def _apply_move_incremental(a: np.ndarray,
                            b: np.ndarray,
                            donor: int,
                            receiver: int,
                            delta: float) -> tuple[np.ndarray, np.ndarray]:
    """Return new (a, b) after moving `delta` mass from donor to receiver."""
    n = a.size
    a_new = a.copy()
    b_new = b.copy()

    a_new[donor] -= delta
    a_new[receiver] += delta

    # linear part: -2δ·a shifted by donor, +2δ·a shifted by receiver
    b_new[donor: donor + n] -= 2.0 * delta * a
    b_new[receiver: receiver + n] += 2.0 * delta * a

    # quadratic correction terms
    b_new[2 * donor] += delta * delta
    b_new[2 * receiver] += delta * delta
    b_new[donor + receiver] -= 2.0 * delta * delta

    return a_new, b_new


def _hill_climbing(a0: np.ndarray,
                   b0: np.ndarray,
                   target_sum: float,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy random hill‑climbing.  At each iteration a maximal convolution
    index `k` is chosen; a donor (large partner value) and a receiver
    (small partner value) are selected from the overlap region of `k`,
    and a small amount of mass is transferred.  Updates to `b` are done
    incrementally in O(n) time.
    """
    a = a0.copy()
    b = b0.copy()
    best_a = a.copy()
    best_val = b.max()
    n = a.size
    rng = np.random.default_rng()

    while time.time() < deadline:
        cur_max = b.max()
        idx_max = np.flatnonzero(b >= cur_max - 1e-12)
        if idx_max.size == 0:
            break
        k = int(rng.choice(idx_max))

        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = a[k - idx]                     # a_{k‑i}

        donor_local = int(np.argmax(partner))
        donor = int(idx[donor_local])

        receiver_local = int(np.argmin(partner))
        receiver = int(idx[receiver_local])

        if donor == receiver or a[donor] <= 0.0:
            # fall back to a random move
            donor_candidates = np.where(a > 1e-12)[0]
            if donor_candidates.size == 0:
                break
            donor = int(rng.choice(donor_candidates))
            receiver = int(rng.integers(n))
            if receiver == donor:
                continue

        # transfer a modest fraction of the donor mass
        max_delta = a[donor] * 0.08                      # ≤8 % per move
        delta = rng.random() * max_delta + 1e-12

        a_new, b_new = _apply_move_incremental(a, b, donor, receiver, delta)
        new_max = b_new.max()

        if new_max < cur_max - 1e-12:
            a, b = a_new, b_new
            if new_max < best_val:
                best_val = new_max
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] HillClimb improved best={best_val:.6f}")
        # else discard the move (simple greedy strategy)

    return best_a, best_val


# ----------------------------------------------------------------------
# Global random mass‑transfer (exploration, unchanged)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a minimal max‑convolution.
    Returns the best sequence found (as a plain Python list of floats).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5       # safety margin

    # Problem size – a trade‑off between expressiveness and speed
    if budget_s >= 800:
        n = 15000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000
    target_sum = np.sqrt(2.0 * n)                # Σa is fixed to this value

    # --------------------------------------------------------------
    # Phase 0 – dense parametric pre‑search
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    p_grid = np.linspace(0.25, 0.90, 31)
    q_grid = np.linspace(0.0, 6.0, 31)

    # (a) pure power‑law
    for p in p_grid:
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # (b) power‑law + exponential tail
    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # (c) simple sine shape
    seq = _sin_sequence(n, target_sum)
    val = _score(seq)
    if val < best_score:
        best_score = val
        best_seq = seq.copy()

    # (d) previous best if available (resized if needed)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score = val
            best_seq = resized.copy()

    # (e) jittered small population around the current best
    for _ in range(8):
        jitter = rng.normal(scale=target_sum * 0.005, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _score(cand)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()

    print(f"[{time.time():.2f}] After pre‑search best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase A – Adam‑style GD with restarts
    # --------------------------------------------------------------
    gd_deadline = time.time() + (deadline - time.time()) * 0.30
    while time.time() < gd_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _gradient_descent_with_restarts(start_vec,
                                                         target_sum,
                                                         gd_deadline)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – Projected sub‑gradient descent (exact max)
    # --------------------------------------------------------------
    subgrad_deadline = time.time() + (deadline - time.time()) * 0.20
    a_sg, val_sg = _subgradient_descent(best_seq,
                                         target_sum,
                                         subgrad_deadline)
    if val_sg < best_score:
        best_score = val_sg
        best_seq = a_sg.copy()
        print(f"[{time.time():.2f}] Subgradient descent improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase C – Exact deterministic mass‑transfer (core refinement)
    # --------------------------------------------------------------
    while time.time() < deadline - 8.0:
        a_ref, val_ref = _deterministic_mass_transfer_exact2(best_seq,
                                                             deadline,
                                                             target_sum)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] Exact DMT2 improved best = {best_score:.6f}")

        a_rand, val_rand = _global_random_mass_transfer(best_seq,
                                                        deadline,
                                                        target_sum)
        if val_rand < best_score:
            best_score = val_rand
            best_seq = a_rand.copy()
            print(f"[{time.time():.2f}] Global random improved best = {best_score:.6f}")

        if time.time() > deadline - 8.0:
            break

    # --------------------------------------------------------------
    # Phase D – Random hill‑climbing with incremental updates
    # --------------------------------------------------------------
    hill_deadline = time.time() + (deadline - time.time()) * 0.15
    if hill_deadline > time.time() + 5.0:
        # compute full convolution once for the current best
        b_current = _conv_fft(best_seq)
        a_hc, val_hc = _hill_climbing(best_seq, b_current,
                                      target_sum, hill_deadline)
        if val_hc < best_score:
            best_score = val_hc
            best_seq = a_hc.copy()
            print(f"[{time.time():.2f}] Hill‑climb final best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase E – Final polishing with a higher β for the soft‑max
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                deadline,
                                                target_sum,
                                                max_beta=1e6,
                                                step_factor=0.6,
                                                gamma=0.995)
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final evaluation (harmless if evaluate_sequence is not defined)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
