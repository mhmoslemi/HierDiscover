def _subgradient_descent(a0, deadline, target_sum, step_factor=0.5, gamma=0.99):
    a = _project_to_simplex(a0.astype(np.float64).copy(), target_sum)
    best = a.copy()
    best_val = _score(a)
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        # find indices achieving max (allow tolerance)
        idxs = np.where(conv >= maxb * (1 - 1e-12))[0]
        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i = np.arange(lo, hi + 1)
            grad[i] += 2.0 * a[k - i]   # derivative w.r.t a[i] from term a_i a_{k-i}
        # momentum
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)
        # occasional jitter
        if np.random.rand() < 0.01:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)
        cur = _score(a)
        if cur < best_val:
            best_val = cur
            best = a.copy()
    return best, best_val

def _beta_distribution_sequence(n, alpha, target_sum):
    # alpha in (0,1). For symmetrical Beta(alpha, alpha)
    x = (np.arange(1, n+1) - 0.5) / n   # shape at midpoints avoid 0,1
    p = (x * (1 - x)) ** (alpha - 1.0)
    # handle possible infinities
    p = np.where(np.isfinite(p), p, 0.0)
    total = p.sum()
    if total <= 0:
        return np.full(n, target_sum / n)
    a = p * target_sum / total
    return a

"""
Hybrid search for a low‑peak non‑negative step function.

* Initialise a small‑size problem (≈1500) with a family of
  symmetric Beta‑shaped distributions (α∈[0.5,0.9]), power‑law,
  exponential and random seeds.
* Perform several iterations of the linear‑programming alternating
  refinement (LP) introduced in the reference code – each iteration
  solves a LP that maximises a random linear functional under the
  current convolution bound and mixes the result back.
* A short sub‑gradient polishing on the small problem removes any
  remaining spikes.
* Upscale the best small solution to a large size (12000‑16000) by
  linear interpolation and apply a simple reordering (alternating /
  even‑spread) to reduce the peak further.
* Final polishing on the large problem using projected sub‑gradient
  descent and a deterministic mass‑transfer (simulated‑annealing‑like)
  refinement.

All updates respect non‑negativity, the ℓ¹‑simplex constraint and the
hard 0 ≤ a_i ≤ 1000 bound enforced by ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# ℓ¹‑simplex projection
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """
    Return the evaluation score 2·n·max(a*a)/(Σa)².
    The function mimics ``evaluate_sequence`` – values are clipped to
    the range [0,1000] before the convolution.
    """
    a = np.clip(a, 0.0, 1000.0)
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _beta_distribution_sequence(n: int, alpha: float, target_sum: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on (0,1) sampled at mid‑points.
    α∈(0,1]; α=0.5 gives the arcsine distribution.
    The resulting vector is scaled to have ℓ¹‑norm = target_sum
    and projected onto the simplex.
    """
    # mid‑point of each bin, avoiding the exact 0/1 boundaries
    x = (np.arange(1, n + 1) - 0.5) / n
    # p_i ∝ (x_i·(1−x_i))^{α−1}
    p = np.power(x * (1.0 - x), alpha - 1.0)
    p[~np.isfinite(p)] = 0.0
    total = p.sum()
    if total == 0.0:
        a = np.full(n, target_sum / n)
    else:
        a = p * target_sum / total
    a = np.clip(a, 0.0, 1000.0)
    a = _project_to_simplex(a, target_sum)
    return a


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^{−p} then renormalised."""
    a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
    a *= target_sum / a.sum()
    a = np.clip(a, 0.0, 1000.0)
    a = _project_to_simplex(a, target_sum)
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(−lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    a = np.clip(a, 0.0, 1000.0)
    a = _project_to_simplex(a, target_sum)
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex."""
    a = rng.random(n)
    a = a * target_sum / a.sum()
    a = np.clip(a, 0.0, 1000.0)
    a = _project_to_simplex(a, target_sum)
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to a new length ``new_len``."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Simple re‑ordering heuristics
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread the large values uniformly by a coprime step."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that works directly with the max‑convolution."""
    a = _project_to_simplex(a0.astype(np.float64).copy(), target_sum)
    best = a.copy()
    best_val = _score(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        # indices where the convolution attains (almost) the maximum
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]
        grad = np.zeros_like(a)

        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i = np.arange(lo, hi + 1, dtype=np.int64)
            grad[i] += 2.0 * a[k - i]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional tiny stochastic perturbation to escape plateaus
        if rng.random() < 0.01:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()

    return best, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (simulated annealing style)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy mass moves with a decaying Metropolis acceptance rule."""
    a = a0.copy()
    cur_score = _score(a)
    best = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    n = a.shape[0]

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # locate the index of the current convolution peak
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # product a_i * a_{kmax-i} – the contribution of each pair
        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
        else:
            # revert the move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best, best_score


# ----------------------------------------------------------------------
# Linear‑programming alternating refinement (small dimensionality)
# ----------------------------------------------------------------------


def _lp_optimize(f: np.ndarray,
                 M: float,
                 target_sum: float,
                 rng: np.random.Generator) -> np.ndarray | None:
    """
    Solve a LP that maximises a random linear functional of `b`
    subject to `f * b ≤ M` and `b ≥ 0`.  The result is scaled to have
    ℓ¹‑norm = target_sum.  Returns None on failure.
    """
    n = f.shape[0]
    rows = []
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=np.int64)
        row = np.zeros(n, dtype=np.float64)
        row[i] = f[k - i]
        rows.append(row)
    A = np.vstack(rows)               # (2n‑1) × n
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Random objective – maximise c·b  ⇔ minimise –c·b
    c = -rng.random(n)

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0.0, None)] * n,
                      method='highs')
    except Exception as e:
        print(f"[{time.time():.2f}] LP exception: {e}")
        return None

    if not res.success:
        return None

    b = res.x
    total = b.sum()
    if total <= 0:
        return None
    b *= target_sum / total
    return b

    # ------------------------------------------------------------------
