def _reorder_alternating(a):
    sorted_vals = np.sort(a)[::-1]
    new = np.empty_like(a)
    left = 0
    right = len(a) - 1
    toggle = True
    for val in sorted_vals:
        if toggle:
            new[left] = val
            left += 1
        else:
            new[right] = val
            right -= 1
        toggle = not toggle
    return new

"""
Improved step‑function search for the min‑max self‑convolution problem.

The algorithm builds on the previous four‑stage pipeline but adds several
new ideas:

* a richer parametric pre‑search (double power‑laws, mixed sine‑power,
  alternating high‑low patterns);
* a dedicated random‑permutation phase that “spreads’’ large coefficients
  across the index domain – this often lowers the central convolution peak;
* a short “dual mass‑transfer’’ routine that simultaneously attacks the
  maximal and the minimal convolution values;
* more aggressive use of the Adam optimiser (higher β, more restarts);
* a final symmetrisation step (a ← (a+rev(a))/2) before the polishing run.

All phases respect the global time budget (default 1000 s).  The returned
list is trimmed of trailing near‑zero entries and satisfies the
evaluation function’s constraints.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Return the objective value (max‑convolution entry) after normalisation."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families for the parametric pre‑search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0))
    a *= target_sum / a.sum()
    return a


def _double_power_sequence(n: int, p1: float, p2: float, cut_frac: float,
                           target_sum: float) -> np.ndarray:
    """Two‑stage power law: p1 on [0,cut), p2 on [cut,n)."""
    cut = int(np.clip(cut_frac, 0.01, 0.99) * n)
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = np.empty_like(idx)
    a[:cut] = idx[:cut] ** (-p1)
    a[cut:] = idx[cut:] ** (-p2)
    a *= target_sum / a.sum()
    return a


def _mixed_sin_power_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Power‑law weighted by a sine envelope (always positive)."""
    base = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    sine = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / (n + 1.0))
    a = base * sine
    a = np.maximum(a, 0.0)
    a *= target_sum / a.sum()
    return a


def _alternating_high_low_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Alternating high/low values generated from a power law."""
    base = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    high = base
    low = base * 0.2  # make the “low’’ entries much smaller
    a = np.empty_like(base)
    a[0::2] = high[0::2]
    a[1::2] = low[1::2]
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max based gradient (used by Adam)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max approximation of the max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Multiple short Adam restarts from jittered starts."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=2e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (core refinement)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the maximal convolution entry.
    Moves mass from a “donor’’ (large derivative) to a “receiver’’
    (small derivative) using the analytically required amount.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second‑largest value (ignore all current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # pick one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k‑i}
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # pick a receiver (different from donor) with smallest partner value
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom

        # move a safe fraction of the donor mass
        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score

# Projected sub‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         max_iter: int = 5000) -> tuple[np.ndarray, float]:
    """Direct projected sub‑gradient descent on the true max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.flatnonzero(conv >= max_val - 1e-12)

        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()
        delta_target = max_val - second_max
        if delta_target < 1e-14:
            break

        k = idx_max[rng.integers(len(idx_max))]
        n = a.shape[0]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k - idx]

        norm_g2 = np.dot(g, g)
        if norm_g2 == 0.0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = _project_to_simplex(a + jitter, target_sum)
            it += 1
            continue

        step = 0.9 * delta_target / norm_g2
        step = min(step, target_sum * 0.4 / (np.linalg.norm(g) + 1e-12))

        for _ in range(6):
            a_candidate = _project_to_simplex(a - step * g, target_sum)
            cand_score = _score(a_candidate)
            if cand_score < best_score - 1e-12:
                a = a_candidate
                best_a = a_candidate.copy()
                best_score = cand_score
                print(f"[{time.time():.2f}] Subgrad it={it:5d} step={step:.3e} score={best_score:.6f}")
                break
            step *= 0.5
        else:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = _project_to_simplex(a + jitter, target_sum)

        it += 1

    return best_a, best_score

# ----------------------------------------------------------------------


def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """Simple stochastic transfer of a small fraction of mass."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Permutation search – random shuffles of the current best sequence
# ----------------------------------------------------------------------


def _permutation_search(a0: np.ndarray,
                        deadline: float,
                        target_sum: float,
                        attempts: int = 5000) -> tuple[np.ndarray, float]:
    """
    Randomly permute the coefficient vector; keep any permutation that
    improves the maximal convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    n = a.size
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < attempts:
        perm = rng.permutation(n)
        cand = a[perm]
        # the permutation does not change the sum, no projection needed
        val = _score(cand)
        if val < best_score:
            best_score = val
            best_a = cand.copy()
            a = cand  # keep the improved permutation as the new base
            print(f"[{time.time():.2f}] Permutation improvement score={best_score:.6f}")
        it += 1

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a minimal maximal
    self‑convolution.  The routine respects the supplied time budget and
    returns the best sequence discovered as a plain Python list.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n gives more freedom but costs more)
    # --------------------------------------------------------------
    if budget_s >= 900:
        n = 15000
    elif budget_s >= 600:
        n = 12000
    elif budget_s >= 300:
        n = 8000
    else:
        n = 5000
    target_sum = np.sqrt(2.0 * n)   # Σa = √(2n)

    # --------------------------------------------------------------
    # Phase 0 – deterministic parametric pre‑search (fast sweeps)
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    # (a) systematic grids (same as before)
    p_grid = np.linspace(0.25, 0.90, 31)
    q_grid = np.linspace(0.0, 6.0, 31)

    for p in p_grid:
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # simple sine shape
    seq = _sin_sequence(n, target_sum)
    val = _score(seq)
    if val < best_score:
        best_score = val
        best_seq = seq.copy()

    # (b) additional families
    for _ in range(12):
        # double‑power‑law
        p1 = rng.uniform(0.2, 1.0)
        p2 = rng.uniform(0.2, 1.0)
        cut = rng.uniform(0.2, 0.8)
        seq = _double_power_sequence(n, p1, p2, cut, target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

        # mixed sin‑power
        p = rng.uniform(0.2, 1.0)
        seq = _mixed_sin_power_sequence(n, p, target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

        # alternating high‑low
        p = rng.uniform(0.2, 1.0)
        seq = _alternating_high_low_sequence(n, p, target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # (c) reuse previous global best if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score = val
            best_seq = resized.copy()

    # (d) a few small random jitters around the current best
    for _ in range(6):
        jitter = rng.normal(scale=target_sum * 0.005, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _score(cand)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()

    print(f"[{time.time():.2f}] After pre‑search best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase A – Adam‑style GD with restarts (30 % of time)
    # --------------------------------------------------------------
    phase_a_deadline = time.time() + (deadline - time.time()) * 0.30
    while time.time() < phase_a_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _gradient_descent_with_restarts(start_vec,
                                                         target_sum,
                                                         phase_a_deadline)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – Projected sub‑gradient descent (25 % of time)
    # --------------------------------------------------------------
    phase_b_deadline = time.time() + (deadline - time.time()) * 0.25
    a_sg, val_sg = _subgradient_descent(best_seq,
                                         target_sum,
                                         phase_b_deadline)
    if val_sg < best_score:
        best_score = val_sg
        best_seq = a_sg.copy()
        print(f"[{time.time():.2f}] Subgradient descent improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase C – Exact deterministic mass‑transfer (core refinement)
    # --------------------------------------------------------------
    phase_c_deadline = time.time() + (deadline - time.time()) * 0.15
    while time.time() < phase_c_deadline - 8.0:
        a_ref, val_ref = _deterministic_mass_transfer_exact2(best_seq,
                                                             phase_c_deadline,
                                                             target_sum)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] Exact DMT2 improved best = {best_score:.6f}")

        # short random exploration to escape possible plateaus
        a_rand, val_rand = _global_random_mass_transfer(best_seq,
                                                        phase_c_deadline,
                                                        target_sum)
        if val_rand < best_score:
            best_score = val_rand
            best_seq = a_rand.copy()
            print(f"[{time.time():.2f}] Global random improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase D – Permutation search (optional, up to ~5 % of time)
    # --------------------------------------------------------------
    if time.time() + 5.0 < deadline:
        perm_deadline = time.time() + (deadline - time.time()) * 0.06
        a_perm, val_perm = _permutation_search(best_seq,
                                               perm_deadline,
                                               target_sum,
                                               attempts=2000)
        if val_perm < best_score:
            best_score = val_perm
            best_seq = a_perm.copy()
            print(f"[{time.time():.2f}] Permutation phase improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase E – Symmetrisation (average with reverse) – may improve flatness
    # --------------------------------------------------------------
    sym_seq = (best_seq + best_seq[::-1]) * 0.5
    sym_seq = _project_to_simplex(sym_seq, target_sum)
    sym_val = _score(sym_seq)
    if sym_val < best_score:
        best_score = sym_val
        best_seq = sym_seq.copy()
        print(f"[{time.time():.2f}] Symmetrisation improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase F – Final high‑β Adam polishing (remaining time)
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                deadline,
                                                target_sum,
                                                max_beta=1e6,
                                                step_factor=0.6,
                                                gamma=0.995)
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final evaluation (harmless if evaluate_sequence is not defined)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
