def adam_softmax_descent(a0, target_sum, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_val = _score(a)

    # Adam hyperparams
    lr = 0.02
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    beta = 1.0
    max_beta = 1e6
    beta_schedule_iter = 0

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)
        t += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = np.maximum(a, 0.0)
        a = a * (target_sum / a.sum())

        # adjust beta gradually
        if beta_schedule_iter % 200 == 0:
            beta = min(beta * 1.5, max_beta)
        beta_schedule_iter += 1

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamSoftmax step={t} β={beta:.1f} score={best_val:.6f}")

    return best_a, best_val

def subgradient_descent_max(a0, target_sum, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_val = _score(a)
    n = a.shape[0]

    rng = np.random.default_rng()
    start = time.time()
    it = 0
    while time.time() < deadline:
        b = _conv_fft(a)
        max_b = b.max()
        # active set indices where b within 1% of max
        threshold = max_b * 0.995 # 0.5% maybe
        active = np.where(b >= threshold)[0]

        # compute gradient as sum over active peaks
        g = np.zeros_like(a)
        for k_max in active:
            lo = max(0, k_max - (n - 1))
            hi = min(k_max, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            # contributions: 2*a_{k_max - idx}
            g[idx] += 2.0 * a[k_max - idx]

        # step size schedule linearly decreasing
        elapsed = time.time() - start
        total = deadline - start
        eta_start = 2.0
        eta_end = 0.05
        eta = eta_start * (1 - elapsed / total) + eta_end * (elapsed / total)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 200 == 0:
            # jitter
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGradMax it={it} score={best_val:.6f}")
        it += 1

    return best_a, best_val

"""
Improved optimisation of the non‑negative step function for the benchmark.

The search proceeds in three stages:

1. **Fast Adam‑mirrored descent on a soft‑max surrogate** (small problem size).
   This quickly reduces the peak while keeping the ℓ₁‑norm fixed.

2. **Sub‑gradient descent on the true max‑convolution** (large problem size).
   The sub‑gradient is summed over all near‑maximal convolution entries
   (instead of a single arg‑max) to obtain a richer descent direction.

3. **Aggressive simulated‑annealing mass‑transfer**.
   *Temperature* is kept high for longer, the allowed transfer size is
   enlarged (up to 20 % of the donor mass) and a simple Metropolis rule
   with the *current* energy is used.  Periodic full recomputation of the
   convolution eliminates drift.

The routine respects the supplied time budget (default 1000 s) and returns
the best sequence found.  A warm‑start from the previous global
`height_sequence_1` is used when available; otherwise a power‑law start
is generated.  The final sequence is trimmed of trailing zeros and stored
globally for future runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Objective  2·n·max(a*a) / (sum(a)²).
    For Σa = √(2n) this equals max(a*a).
    """
    n = a.shape[0]
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of the max‑convolution.
    Returns (grad, full_convolution) where grad has the same length as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # gradient = 2 * (a_rev * p) conv
    a_rev = a[::-1]
    grad_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(grad_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase 1 – Adam (mirror) descent on the soft‑max surrogate (small n)
# ----------------------------------------------------------------------


def _adam_softmax_descent(a0: np.ndarray,
                          target_sum: float,
                          deadline: float):
    """Adam optimiser on the soft‑max surrogate."""
    a = a0.copy()
    best_a = a.copy()
    best_val = _score(a)

    # Adam hyper‑parameters
    lr = 0.02
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    beta = 1.0
    max_beta = 1e6
    beta_inc_iter = 0

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)

        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a -= lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        a = a * (target_sum / a.sum())

        # slowly increase the soft‑max sharpness
        if beta_inc_iter % 200 == 0:
            beta = min(beta * 1.5, max_beta)
        beta_inc_iter += 1

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamDescent it={t:5d} β={beta:6.2f} score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – Sub‑gradient descent on the true max‑convolution (large n)
# ----------------------------------------------------------------------


def _subgradient_descent_multi(a0: np.ndarray,
                               target_sum: float,
                               deadline: float):
    """
    Sub‑gradient descent using *all* convolution entries that lie
    within 0.5 % of the current maximum.  This yields a richer descent
    direction than a single arg‑max.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _score(a)

    n = a.shape[0]
    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        b = _conv_fft(a)
        max_b = b.max()
        # active set – indices whose value is ≥ (1‑ε)·max
        eps = 0.005
        active = np.where(b >= max_b * (1.0 - eps))[0]

        # aggregate sub‑gradient over the active set
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            g[idx] += 2.0 * a[k - idx]

        # linear schedule for the multiplicative step size
        elapsed = time.time() - start
        total = deadline - start
        eta_start = 2.0
        eta_end = 0.05
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        # occasional jitter to escape flat regions
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGradMulti it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 3 – Aggressive simulated‑annealing mass‑transfer (large n)
# ----------------------------------------------------------------------


def _anneal_mass_transfer(a0: np.ndarray,
                         target_sum: float,
                         deadline: float):
    """Improved annealing with larger move sizes and slower cooling."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start

    # initial temperature: a generous fraction (120 %) of the current score
    T0 = cur_val * 1.20

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        # slower, sub‑linear cooling (exponent 0.25 keeps T high longer)
        temperature = max(0.0, T0 * (1.0 - frac) ** 0.25)

        # locate the current peak(s)
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contributions to the current peak
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # pick a receiver that is comparatively low
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # allowed transfer size – much larger than in the original code
        max_delta = min(a[donor] * 0.20, target_sum * 0.02)  # up to 20 % donor or 2 % total
        if max_delta <= 0:
            continue

        # scale with temperature (still 0 at the very end)
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * (0.3 + 0.7 * scale)  # always a baseline move
        if delta <= 0.0:
            continue

        # ----- exact O(n) update of the convolution -----
        b_new = b.copy()
        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        # Metropolis acceptance based on the *current* energy
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved best={best_val:.6f}")

        # every few thousand steps recompute everything from scratch to avoid numerical drift
        if it % 4000 == 0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # Problem sizes – a cheaper stage on a smaller grid, followed by
    # refinement on the full grid.
    # --------------------------------------------------------------
    n_small = 8000
    n_full = 18000

    target_small = np.sqrt(2.0 * n_small)
    target_full = np.sqrt(2.0 * n_full)

    # --------------------------------------------------------------
    # Warm‑start: either the previous best or a simple decreasing power law.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init_small = _resize_sequence(np.array(globals()["height_sequence_1"],
                                               dtype=np.float64), n_small)
    else:
        init_small = (np.arange(1, n_small + 1, dtype=np.float64) ** (-0.55))

    init_small = init_small * (target_small / init_small.sum())

    # --------------------------------------------------------------
    # Phase 1 – Adam on the soft‑max surrogate (fast, small n)
    # --------------------------------------------------------------
    phase1_deadline = time.time() + 0.12 * (deadline - time.time())
    a_small, best_val_small = _adam_softmax_descent(init_small,
                                                    target_small,
                                                    phase1_deadline)
    best_seq = a_small
    best_score = best_val_small
    print(f"[{time.time():.2f}] After Phase 1 (Adam) best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – (optional) a quick random‑restart check
    # --------------------------------------------------------------
    # a few random initialisations with a very short Adam run to possibly
    # escape a shallow basin.
    rng = np.random.default_rng()
    extra_deadline = time.time() + 0.07 * (deadline - time.time())
    while time.time() < extra_deadline:
        rnd = rng.random(n_small)
        rnd = rnd * (target_small / rnd.sum())
        a_tmp, val_tmp = _adam_softmax_descent(rnd, target_small, time.time() + 2.0)
        if val_tmp < best_score:
            best_score = val_tmp
            best_seq = a_tmp.copy()
            print(f"[{time.time():.2f}] Random restart improved score={best_score:.6f}")

    # --------------------------------------------------------------
    # Upscale to the full problem size.
    # --------------------------------------------------------------
    a_full = _resize_sequence(best_seq, n_full)
    a_full = a_full * (target_full / a_full.sum())
    best_seq = a_full
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] After upscale best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Sub‑gradient descent on the true max‑convolution.
    # --------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + 0.12 * (deadline - time.time())
        a_full, val_sub = _subgradient_descent_multi(best_seq,
                                                     target_full,
                                                     sub_deadline)
        if val_sub < best_score:
            best_score = val_sub
            best_seq = a_full.copy()
        print(f"[{time.time():.2f}] After Sub‑gradient phase best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – Aggressive simulated‑annealing refinement.
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_full, val_ann = _anneal_mass_transfer(best_seq,
                                                target_full,
                                                deadline)
        if val_ann < best_score:
            best_score = val_ann
            best_seq = a_full.copy()
        print(f"[{time.time():.2f}] After annealing best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final clean‑up: strip trailing zeros, evaluate with the provided
    # benchmark function and store a warm‑start for the next run.
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    globals()["height_sequence_1"] = final_seq

    try:
        final_val = evaluate_sequence(final_seq)   # noqa: F821 – assumed to be provided
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:                     # noqa: BLE001
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    return final_seq
