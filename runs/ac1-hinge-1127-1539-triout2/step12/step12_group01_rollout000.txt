def mirror_descent_momentum(a0, target_sum, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    # Init momentum vector
    m = np.zeros_like(a)
    beta1 = 0.9  # momentum factor
    eta0 = 0.05
    beta = 1.0
    max_beta = 1e5
    iter = 0
    while time.time() < deadline:
        grad, _ = _softmax_gradient(a, beta)
        # update momentum
        m = beta1 * m + (1 - beta1) * grad
        # effective step
        eta = eta0 / np.sqrt(iter + 1)  # diminishing step
        a = a * np.exp(-eta * m)  # multiplicative update (mirror descent)
        # extra jitter occasionally?
        if iter % 500 == 0 and iter > 0:
            jitter = rng.normal(scale=target_sum*0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
        # project
        a = a * (target_sum / a.sum())
        # optionally increase beta gradually
        if iter % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)
        # evaluate
        val = _score(a)
        if val < best_val:
            best_val = val; best_a = a.copy()
            print(f"[{time.time():.2f}] Mirror mom iter={iter} beta={beta:.2f} score={val:.6f}")
        iter += 1
    return best_a, best_val

def subgradient_momentum(a0, target_sum, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    beta1 = 0.9
    iter = 0
    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))
        # compute subgradient: 2*a[k_max - i] based on contributions
        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k_max - idx]
        # momentum update
        m = beta1 * m + (1 - beta1) * grad
        # step schedule: decaying
        eta = 0.5 / np.sqrt(iter + 1)
        a = a * np.exp(-eta * m)
        a = a * (target_sum / a.sum())
        # jitter occasional
        if iter % 400 == 0:
            jitter = rng.normal(scale=target_sum*0.0005, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_val:
            best_val = val; best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad mom iter={iter} score={val:.6f}")
        iter += 1
    return best_a, best_val

propose_candidate – heuristic search for a low‑max‑autocorrelation step
---------------------------------------------------------------------

We minimise the benchmark function

        E(a) = 2·n·max(a*a) / (sum(a)**2)

under the constraints a_i ≥ 0 and sum(a)=√(2 n).  For a fixed n the
objective is exactly the maximal entry of the self‑convolution, so the

The search proceeds in four stages:

   previous run are normalised to the required sum and the best of them
   (according to a fast FFT‑based score) is kept.

2) **Momentum mirror‑descent on a soft‑max surrogate** – we minimise a
   smooth approximation of the max using the gradient of the
   log‑sum‑exp of the convolution.  A momentum term (β₁≈0.9) and an
   exponentially growing smoothness parameter β steer the iterates
   quickly into a low‑energy region while keeping the simplex
   constraint via multiplicative updates.

3) **Momentum sub‑gradient descent on the true max** – once β is large
   we switch to the exact sub‑gradient of the maximal convolution entry.
   The same mirror‑descent style update with momentum is used.

4) **Aggressive simulated‑annealing mass transfer** – the final stage
   moves small chunks of mass from indices that heavily contribute to
   the current peak towards low‑height indices.  The temperature schedule
   is √‑type (slow early cooling) and the transfer size scales with the
   temperature, allowing large exploratory moves at the beginning and
   fine adjustments later.

All stages respect the global time budget (default 1000 s).  The best
sequence found is stored in the global variable ``height_sequence_1``
so that subsequent runs start from a warm‑start.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Benchmark objective 2·n·max(a*a) / (sum(a)²).
    For Σa = √(2 n) this equals max(a*a).
    """
    n = a.shape[0]
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the log‑sum‑exp surrogate of max(a*a).

    Returns (grad, full_convolution) where `grad` has the same shape as
    `a`.  The gradient follows from the chain rule:
        d/da log Σ exp(β·b_k) = Σ p_k · ∂b_k/∂a,
    where p_k are the softmax weights.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # convolution with reversed a gives the required Jacobian
    a_rev = a[::-1]
    grad_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(grad_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Simple helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq)


# ----------------------------------------------------------------------
# Stage 0 – generate a few candidate initialisations
# ----------------------------------------------------------------------


def _initial_candidates(n: int, target_sum: float, deadline: float):
    """Return the best of several cheap initialisations."""
    rng = np.random.default_rng()
    best_a = None
    best_val = np.inf

    # 1) warm‑start if available
    if "height_sequence_1" in globals():
        w = np.array(globals()["height_sequence_1"], dtype=np.float64)
        a = _resize_sequence(w, n)
        a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_val:
            best_val, best_a = val, a

    # 2) power‑law families
    for alpha in np.linspace(0.40, 0.80, 9):
        a = (np.arange(1, n + 1, dtype=np.float64) ** (-alpha))
        a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_val:
            best_val, best_a = val, a

    # 3) exponential decay families
    for lam in np.linspace(5e-4, 5e-3, 6):
        a = np.exp(-lam * np.arange(n, dtype=np.float64))
        a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_val:
            best_val, best_a = val, a

    # 4) few random Dirichlet vectors (concentrated mass)
    for _ in range(5):
        a = rng.dirichlet(np.full(n, 0.1))
        a = a * target_sum
        val = _score(a)
        if val < best_val:
            best_val, best_a = val, a

    # 5) a tiny jittered version of the current best (if any)
    if best_a is not None:
        jitter = rng.normal(scale=target_sum * 0.002, size=n)
        a = np.maximum(best_a + jitter, 0.0)
        a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_val:
            best_val, best_a = val, a

    print(f"[{time.time():.2f}] Initialisation best score={best_val:.6f}")
    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 1 – momentum mirror descent on soft‑max surrogate
# ----------------------------------------------------------------------


def _mirror_descent_momentum(a0: np.ndarray, target_sum: float, deadline: float):
    """Mirror descent with exponential updates, momentum and growing β."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    m = np.zeros_like(a)          # momentum term
    beta1 = 0.9                   # momentum coefficient
    eta0 = 0.05                   # base step size
    beta = 1.0
    max_beta = 1e5
    it = 0

    while time.time() < deadline:
        grad, _ = _softmax_gradient(a, beta)

        # momentum update (exponential moving average)
        m = beta1 * m + (1.0 - beta1) * grad

        # diminishing step size
        eta = eta0 / np.sqrt(it + 1.0)

        # multiplicative (mirror) update
        a = a * np.exp(-eta * m)

        # occasional jitter to escape plateaus
        if it % 500 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)

        # re‑project to the simplex
        a = a * (target_sum / a.sum())

        # increase smoothness parameter β gradually
        if it % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            print(f"[{time.time():.2f}] MirrorMom it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 2 – momentum sub‑gradient descent on true max‑convolution
# ----------------------------------------------------------------------


def _subgradient_momentum(a0: np.ndarray, target_sum: float, deadline: float):
    """Sub‑gradient descent on the true max entry, with momentum."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    m = np.zeros_like(a)
    beta1 = 0.9
    it = 0

    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k_max - idx]          # exact sub‑gradient

        # momentum
        m = beta1 * m + (1.0 - beta1) * grad

        # step size (diminishing)
        eta = 0.5 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * m)
        a = a * (target_sum / a.sum())

        # jitter occasionally
        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        cur_val = _score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            print(f"[{time.time():.2f}] SubGradMom it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 3 – aggressive simulated‑annealing mass‑transfer
# ----------------------------------------------------------------------


def _anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """Annealing with temperature proportional to current score."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_val = _score(a)

    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.80  # initial temperature as a large fraction of score

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        # √‑schedule keeps temperature high for longer
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        # locate the current convolution peak
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contributions to the peak
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # choose a low‑height receiver (inverse proportional to height)
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            it += 1
            continue

        # transfer amount – up to 10 % of donor mass or 1 % of total sum
        max_delta = min(a[donor] * 0.10, target_sum * 0.01)
        if max_delta <= 0.0:
            it += 1
            continue
        # scale with temperature (linear scaling works well)
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            it += 1
            continue

        # --- O(n) exact update of the convolution -----------------------
        b_new = b.copy()
        # linear contribution 2·a·Δ (removal from donor, addition to receiver)
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic terms Δ²
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        # Metropolis acceptance (based on current energy)
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[{time.time():.2f}] Anneal improved best={best_val:.6f}")

        # periodic full recomputation to prevent drift
        if it % 5000 == 0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[{time.time():.2f}] Full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # leave a small safety margin

    # ------------------------------------------------------------------
    # Choose problem size
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 500:
        n = 15000
    else:
        n = 10000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Stage 0 – generate a good starting point
    # ------------------------------------------------------------------
    init_seq, init_val = _initial_candidates(n, target_sum, deadline)

    # ------------------------------------------------------------------
    # Stage 1 – mirror descent on soft‑max surrogate (≈15 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        md_deadline = time.time() + 0.15 * (deadline - time.time())
        seq_md, val_md = _mirror_descent_momentum(init_seq, target_sum, md_deadline)
        best_seq, best_val = (seq_md, val_md) if val_md < init_val else (init_seq, init_val)
        print(f"[{time.time():.2f}] After mirror descent best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – sub‑gradient descent on true max (≈15 % of remaining)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sg_deadline = time.time() + 0.15 * (deadline - time.time())
        seq_sg, val_sg = _subgradient_momentum(best_seq, target_sum, sg_deadline)
        if val_sg < best_val:
            best_seq, best_val = seq_sg, val_sg
        print(f"[{time.time():.2f}] After sub‑gradient descent best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – long annealing refinement (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_ann, val_ann = _anneal_mass_transfer(best_seq, target_sum, deadline)
        if val_ann < best_val:
            best_seq, best_val = seq_ann, val_ann
        print(f"[{time.time():.2f}] After annealing best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Final clean‑up: cut trailing zeros, convert to Python list
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Verify with the provided evaluation function (optional but useful)
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # Store for warm‑starts in subsequent runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
