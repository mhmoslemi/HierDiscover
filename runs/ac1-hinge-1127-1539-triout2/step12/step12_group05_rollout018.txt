"""
Optimisation of a non‑negative step function for the benchmark.
The core ideas
----------------
1.  A lightweight parametric family   a_i = (i+1+d)^{-p}
    (power‑law with a positive offset *d*).  The offset removes the huge
    first entry that normally dominates the convolution when p>0.5.
2.  Exhaustive grid search over (p,d) (few hundred candidates) using a
    fast FFT based evaluation of the score.
3.  Small random refinement around the best grid point.
4.  A short projected‑Adam run starting from the best candidate to
    polish the result.
All phases respect the simplex constraint Σ a_i = √(2 n) and the
non‑negativity constraint.  The routine stops before the supplied wall‑clock
budget and returns the best sequence found (trailing near‑zero entries are
removed).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities (full linear convolution)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft_pair(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution x*y using a real FFT (full length 2·n‑1)."""
    n = x.size
    size = 1 << ((2 * n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Benchmark score; equal to the maximal convolution when Σa = √(2n)."""
    n = a.shape[0]
    max_b = _max_conv(a)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project onto {x_i ≥ 0, Σx_i = s} (Euclidean projection)."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric family – shifted power law
# ----------------------------------------------------------------------


def _shifted_power_law_sequence(
    n: int, p: float, d: float, target_sum: float
) -> np.ndarray:
    """
    a_i = (i+1+d)^{-p}, then projected onto the simplex with Σa = target_sum.
    The offset d≥0 removes the singularity at i=0 for p>0.5.
    """
    idx = np.arange(1, n + 1, dtype=np.float64) + d  # shape (n,)
    a = idx ** (-p)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on a soft‑max approximation
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(a*a)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = conv_full.max()
    exps = np.exp(beta * (conv_full - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, conv_full


def _adam_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 5e5,
    step_factor: float = 0.8,
    gamma: float = 0.999,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the soft‑max approximation of the max‑convolution.
    Returns the best point seen before the deadline.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with minimal benchmark score.
    The routine respects the supplied time budget and returns the best
    sequence found (trailing near‑zero entries are trimmed).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    # ---------------------------------------------------------------
    # Problem size (larger n gives a finer discretisation)
    # ---------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 9000
    else:
        n = 6000

    target_sum = math.sqrt(2.0 * n)               # Σa is fixed to this value

    # ---------------------------------------------------------------
    # Initial baseline (uniform) – guarantees a valid return
    # ---------------------------------------------------------------
    best_seq = np.full(n, target_sum / n, dtype=np.float64)
    best_score = _score(best_seq)
    best_p, best_d = 0.5, 0.0

    # allow warm start from external best if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score, best_seq = val, resized.copy()
            print(f"[{time.time():.2f}] Warm start improved best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase 0 – dense grid search over shifted power‑law family
    # ---------------------------------------------------------------
    print(f"[{time.time():.2f}] Starting grid search (n={n})")
    p_vals = np.linspace(0.5, 0.94, 23)                # 0.5 … 0.94
    max_d = min(400.0, n * 0.08)                      # keep d reasonably small
    d_vals = np.linspace(0.0, max_d, 41)               # 0 … max_d

    for p in p_vals:
        for d in d_vals:
            if time.time() > deadline - 2.0:
                break
            seq = _shifted_power_law_sequence(n, float(p), float(d), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()
                best_p, best_d = p, d
                print(f"[{time.time():.2f}] Grid new best p={p:.3f} d={d:.1f} score={val:.6f}")

    # ---------------------------------------------------------------
    # Phase 1 – local random refinement around the current best
    # ---------------------------------------------------------------
    rng = np.random.default_rng()
    while time.time() < deadline - 6.0:
        p = best_p + rng.normal(scale=0.02)
        d = best_d + rng.normal(scale=5.0)
        p = float(np.clip(p, 0.45, 0.99))
        d = float(max(d, 0.0))
        seq = _shifted_power_law_sequence(n, p, d, target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()
            best_p, best_d = p, d
            print(f"[{time.time():.2f}] Refine new best p={p:.3f} d={d:.1f} score={val:.6f}")

    # ---------------------------------------------------------------
    # Phase 2 – short Adam polishing (soft‑max, high β)
    # ---------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_opt, val_opt = _adam_gradient_descent(
            best_seq,
            deadline,
            target_sum,
            max_beta=1e6,
            step_factor=0.6,
            gamma=0.995,
        )
        if val_opt < best_score:
            best_score, best_seq = val_opt, a_opt.copy()
            print(f"[{time.time():.2f}] Adam polishing improved best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return list
    # ---------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final evaluation (harmless if evaluate_sequence is missing)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated benchmark score = {val:.6f}")
    except Exception:
        pass

    return final_seq
