def propose_candidate(budget_s: int = 1000, **kwargs):
    start = time.time()
    deadline = start + budget_s - 0.5

    # set n based on budget
    if budget_s >= 900:
        n = 20000
    elif budget_s >= 600:
        n = 15000
    elif budget_s >= 300:
        n = 12000
    else:
        n = 8000

    target_sum = math.sqrt(2.0 * n)

    best_seq = None; best_score = float('inf'); rng = np.random.default_rng()

    # Phase 0 parametric families

    # (1) pure power law
    for p in np.arange(0.2, 1.01, 0.02):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_seq = seq.copy(); best_score = val

    # (2) exponential power law grid p, q
    for p in np.arange(0.2, 1.01, 0.02):
        for q in np.arange(0.0, 10.01, 0.5):
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_seq = seq.copy(); best_score = val

    # (3) broken power law
    for p1 in np.arange(0.2, 0.81, 0.1):
        for p2 in np.arange(0.8, 2.01, 0.2):
            for alpha in np.arange(0.2, 0.91, 0.2):
                seq = _broken_power_law_sequence(n, float(p1), float(p2), float(alpha), target_sum)
                val = _score(seq)
                if val < best_score:
                    best_seq = seq.copy(); best_score = val

    # (4) geometric
    for r in np.arange(0.3, 0.95, 0.05):
        seq = np.array([r**i for i in range(n)], dtype=float)
        seq *= target_sum / seq.sum()
        val = _score(seq)
        if val < best_score:
            best_seq = seq.copy(); best_score = val

    # (5) warm start from global variable if exists
    global_seq = globals().get("height_sequence_1")
    if isinstance(global_seq, (list, tuple, np.ndarray)):
        prev = np.asarray(global_seq, dtype=float)
        if prev.size != n:
            # interpolate
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)

"""
Propose a step‑function sequence that minimizes
    score = 2 * n * max(a * a) / (sum(a)^2)

The sum is fixed to √(2 n) which makes the score equal to the
maximum entry of the self‑convolution.  The algorithm consists of:

1.  Choose a problem size *n* based on the time budget.
2.  A dense parametric pre‑search over several classic families
    (power‑law, exponential‑power‑law, broken power‑law, geometric)
    to obtain a good initial shape.
3.  Several short Adam runs on a soft‑max approximation of the max‑convolution
    with an ever‑increasing temperature β; each run starts from the
    current best sequence plus a small random jitter.
4.  Repeated guided mass‑transfer steps (exact sub‑gradient descent) that
    move mass from indices that increase the current peak towards indices
    that decrease it, using a line‑search that drives the peak down to the
    second‑largest value.
5.  Light random “shake‑up” moves to escape plateaus.
6.  A final high‑β Adam polishing phase.
7.  Trim trailing near‑zero entries and return the list.

All updates respect the simplex constraint Σa = √(2 n) and non‑negativity.
The routine respects the supplied wall‑clock budget and returns the best
sequence found.
"""

import time

import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) using a real FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft_pair(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution x*y (full length 2·n‑1) using a real FFT."""
    n = x.size
    size = 1 << ((2 * n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Benchmark score (equal to max convolution when Σa = √(2n))."""
    n = a.shape[0]
    max_b = _max_conv(a)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (dense pre‑search)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _broken_power_law_sequence(
    n: int, p1: float, p2: float, alpha: float, target_sum: float
) -> np.ndarray:
    K = max(1, int(alpha * n))
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.empty_like(idx)
    a[:K] = idx[:K] ** (-p1)
    a[K:] = idx[K:] ** (-p2)
    a *= target_sum / a.sum()
    return a


def _geometric_sequence(n: int, r: float, target_sum: float) -> np.ndarray:
    a = np.array([r ** i for i in range(n)], dtype=np.float64)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max approximation)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(a*a).
    Returns (grad, full_convolution).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = conv_full.max()
    exps = np.exp(beta * (conv_full - max_b))
    p = exps / exps.sum()                     # soft‑max distribution over entries

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]   # gradient w.r.t. each a_i
    return grad, conv_full


def _adam_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 5e5,
    step_factor: float = 0.8,
    gamma: float = 0.999,
) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max approximation of the max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient guided mass‑transfer (Phase B)
# ----------------------------------------------------------------------


def _compute_gradient_for_max(a: np.ndarray, max_idx: np.ndarray) -> np.ndarray:
    """Exact sub‑gradient of max(conv) w.r.t. a (averaged over max indices)."""
    n = a.size
    g = np.zeros_like(a)
    for k in max_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g[idx] += 2.0 * a[k - idx]
    if max_idx.size > 0:
        g /= max_idx.size
    return g


def _guided_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Greedy exact reduction of the maximal convolution entry.
    Moves mass along the exact sub‑gradient direction while preserving Σa.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()

        mask_max = conv >= max_val - tol
        # second largest entry (strictly smaller than the current max)
        if np.any(~mask_max):
            second_max = conv[~mask_max].max()
        else:
            second_max = 0.0
        max_idx = np.where(mask_max)[0]

        # sub‑gradient for the active max entries
        g = _compute_gradient_for_max(a, max_idx)

        # direction that keeps Σa unchanged
        sum_g = g.sum()
        d = -g + sum_g / n * np.ones_like(a)   # Σd = 0

        # largest step before any component would become negative
        neg = d < 0
        if np.any(neg):
            t_nonneg = np.min(a[neg] / -d[neg])
        else:
            t_nonneg = np.inf

        # linearised change of the convolution in direction d
        delta = 2.0 * _conv_fft_pair(a, d)   # length 2n‑1
        delta_max = delta[max_idx]
        # most favourable active index (largest reduction)
        delta_k = delta_max.min()

        if delta_k >= 0:
            # no descent direction – we are at a local optimum
            break

        # step that would bring the current max down exactly to the second max
        t_to_second = (max_val - second_max) / (-delta_k)

        # keep a safety margin to stay away from the boundary
        t = min(t_nonneg, t_to_second) * 0.99

        if t <= 0:
            break

        a += t * d
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] GuidedMT it={it} score={best_score:.6f}")

        # occasional jitter to escape plateaus
        if it % 500 == 0 and rng.random() < 0.05:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_score

def _global_random_move(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
) -> tuple[np.ndarray, float]:
    """Very lightweight simulated‑annealing‑style mass moves."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    while time.time() < deadline:
        # pick a donor among the largest 2 % and a receiver among the smallest 2 %
        top_k = max(1, int(0.02 * n))
        low_k = max(1, int(0.02 * n))
        donor = np.argsort(-a)[rng.integers(top_k)]
        receiver = np.argsort(a)[rng.integers(low_k)]
        if donor == receiver:
            continue
        # move up to 5 % of donor's mass
        delta = rng.random() * a[donor] * 0.05
        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] RandomMove new best={best_score:.6f}")

    return best_a, best_score

    # ---------------------------------------------------------------
    remaining = deadline - time.time()
    adam_phase_end = time.time() + 0.30 * remaining   # ~30 % of remaining time

    while time.time() < adam_phase_end - 2.0:
        # allocate a modest sub‑deadline for each Adam run (≈12 s)
        sub_deadline = min(time.time() + 12.0, adam_phase_end)
        jitter_scale = target_sum * 0.02
        start_vec = best_seq + rng.normal(scale=jitter_scale, size=n)
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _adam_gradient_descent(
            start_vec,
            sub_deadline,
            target_sum,
            max_beta=2e6,
            step_factor=0.7,
            gamma=0.995,
        )
        if val_opt < best_score:
            best_score, best_seq = val_opt, a_opt.copy()
            print(f"[{time.time():.2f}] Adam restart improved best to {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase B – guided exact mass‑transfer + random shake‑ups
    # ---------------------------------------------------------------
    while time.time() < deadline - 10.0:
        # a) Guided mass‑transfer (≈8 s per call)
        sub_deadline_mt = min(time.time() + 8.0, deadline)
        a_ref, val_ref = _guided_mass_transfer(best_seq, sub_deadline_mt, target_sum)
        if val_ref < best_score:
            best_score, best_seq = val_ref, a_ref.copy()
            print(f"[{time.time():.2f}] Guided mass‑transfer improved best to {best_score:.6f}")

        # b) Light random moves (≈4 s per call)
        sub_deadline_rm = min(time.time() + 4.0, deadline)
        a_rand, val_rand = _global_random_move(best_seq, sub_deadline_rm, target_sum)
        if val_rand < best_score:
            best_score, best_seq = val_rand, a_rand.copy()
            print(f"[{time.time():.2f}] Random move improved best to {best_score:.6f}")

        # c) Occasional tiny jitter to keep the search fluid
        if rng.random() < 0.08:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            best_seq += jitter
            best_seq = np.maximum(best_seq, 0.0)
            best_seq = _project_to_simplex(best_seq, target_sum)

    # ---------------------------------------------------------------
    # Phase C – final polishing with a high‑β Adam run
    # ---------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(
            best_seq,
            deadline,
            target_sum,
            max_beta=5e6,
            step_factor=0.5,
            gamma=0.995,
        )
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final polish improved best to {best_score:.6f}")

    # ---------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return list
    # ---------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final evaluation for reporting (harmless if evaluate_sequence missing)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated benchmark score = {val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.2f}] Returning sequence of length {len(final_seq)} with score {best_score:.6f}")
    return final_seq
