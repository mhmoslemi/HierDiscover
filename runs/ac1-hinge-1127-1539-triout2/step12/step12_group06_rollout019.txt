"""
Propose a step‑function (non‑negative vector) minimizing
    2·n·max(conv(a,a)) / (sum(a)**2)

The optimisation consists of three stages:

1.  Adam optimisation on a smooth (soft‑max) approximation of the
    maximal convolution entry.  The vector is represented as
    a = S·softmax(y) so the L1‑norm constraint is satisfied exactly.
    The temperature β of the soft‑max is increased geometrically,
    turning the smooth loss into the true L∞‑loss.

2.  Projected sub‑gradient descent on the exact peak of the convolution.
    This fine‑tunes the solution obtained by Adam.

3.  A short permutation annealing and a deterministic mass‑transfer
    pass for the final polishing.

The whole routine respects the 1000 s budget, stores the best
vector in the global variable `height_sequence_1` and returns it as a
plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
#  Utility: next power of two
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
#  Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
#  Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex of non‑negative vectors with sum s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
#  Scoring utility (objective = 2·n·max(conv)/sum^2)
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
#  Exact max‑convolution sub‑gradient (for the peak entry)
# ----------------------------------------------------------------------
def _max_conv_and_subgrad(a: np.ndarray):
    """Return (conv, max_val, k_max, subgrad) for the current peak."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    kmax = int(np.argmax(conv))
    max_val = conv[kmax]

    grad = np.zeros_like(a)
    lo = max(0, kmax - (n - 1))
    hi = min(kmax, n - 1)
    idx = np.arange(lo, hi + 1)
    partner = a[kmax - idx]
    grad[lo:hi + 1] = 2.0 * partner
    return conv, max_val, kmax, grad


# ----------------------------------------------------------------------
#  Projected sub‑gradient descent (exact peak)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0
    momentum = np.zeros_like(a)

    while time.time() < deadline:
        _, _, _, grad = _max_conv_and_subgrad(a)

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_factor * target_sum / norm
            a -= lr * momentum
        else:
            a -= step_factor * target_sum * grad / (np.linalg.norm(grad) + 1e-12)

        a = _project_to_simplex(a, target_sum)

        # occasional tiny jitter
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] SubGD it={it:5d} score={best_val:.6f}")

        it += 1

    return best, best_val


# ----------------------------------------------------------------------
#  Adam optimisation on a smooth soft‑max approximation of the max
# ----------------------------------------------------------------------
def _adam_softmax_descent(n: int,
                          deadline: float,
                          target_sum: float,
                          init_beta: float = 0.5,
                          max_beta: float = 1e6,
                          base_lr: float = 0.1) -> tuple[np.ndarray, float]:
    """
    Optimise a = target_sum * softmax(y) using Adam on the
    soft‑max of the convolution, gradually increasing the temperature.
    """
    size = _next_pow2(2 * n - 1)          # FFT size (used repeatedly)
    conv_len = 2 * n - 1

    # initialise y (log‑weights) uniformly -> a uniform
    y = np.zeros(n, dtype=np.float64)
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    beta = init_beta
    it = 1

    best_a = None
    best_val = np.inf
    rng = np.random.default_rng()

    while time.time() < deadline:
        # ---- a = target * softmax(y) ---------------------------------
        y_max = np.max(y)
        exp_y = np.exp(y - y_max)
        p = exp_y / np.sum(exp_y)                # probability vector (sum 1)
        a = target_sum * p

        # ---- convolution a*a ----------------------------------------
        fa = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(fa * fa, n=size)[:conv_len]

        max_conv = np.max(conv)
        cur_score = float(2.0 * n * max_conv / (target_sum * target_sum))
        if cur_score < best_val:
            best_val = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:.2e} score={best_val:.6f}")

        # ---- soft‑max weights for the smooth max --------------------
        # (subtract max_conv for numerical stability)
        w = np.exp(beta * (conv - max_conv))
        w_sum = np.sum(w)
        if w_sum == 0:
            w = np.full_like(w, 1.0 / conv_len)
        else:
            w /= w_sum

        # ---- gradient wrt a -----------------------------------------
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, n=size)
        fw = np.fft.rfft(w, n=size)
        grad_full = np.fft.irfft(fa_rev * fw, n=size)[:conv_len]
        grad_a = 2.0 * grad_full[n - 1:n - 1 + n]   # central n entries

        # ---- gradient wrt y (chain rule) ----------------------------
        # avg_grad = (grad_a · a) / target_sum
        avg_grad = np.dot(grad_a, a) / target_sum
        grad_y = a * (grad_a - avg_grad)          # shape (n,)

        # ---- Adam update --------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        y -= base_lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- schedule & occasional jitter ----------------------------
        if it % 400 == 0:
            # increase temperature geometrically
            if beta < max_beta:
                beta = min(beta * 2.0, max_beta)

        if it % 1200 == 0:
            # tiny random walk in y to escape flat regions
            y += rng.normal(scale=0.01, size=n)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
#  Simulated‑annealing permutation search (re‑ordering only)
# ----------------------------------------------------------------------
def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.15,
                                 end_temp: float = 1e-5,
                                 top_frac: float = 0.04,
                                 low_frac: float = 0.04) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor = rng.choice(np.argpartition(-contrib, top_k - 1)[:top_k])
        receiver = rng.choice(np.argpartition(contrib, low_k - 1)[:low_k])

        if donor == receiver:
            continue

        a[donor], a[receiver] = a[receiver], a[donor]

        new_score = _score(a)

        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] PermSearch new best={best_score:.6f}")

        if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / max(temp, 1e-12)):
            pass
        else:
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score


# ----------------------------------------------------------------------
#  Deterministic mass‑transfer refinement (small stochastic hill‑climb)
# ----------------------------------------------------------------------
def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.3,
                                end_temp: float = 1e-5,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]          # contribution weight to the peak
        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                cur_score = new_score
            else:
                a[donor] += delta
                a[receiver] -= delta
                a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            # occasional jitter to keep the dynamics alive
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
#  Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with a low benchmark score.
    The routine runs several optimisation phases within the available
    time budget and returns the best vector found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a length (larger length gives more degrees of freedom)
    # ------------------------------------------------------------------
    # for the full budget we use up to 15000 points; this still fits easily
    # in memory and makes each FFT cheap (size 32768).
    n = 15000
    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # 2. Adam optimisation on the smooth max‑convolution loss
    # ------------------------------------------------------------------
    adam_dead = time.time() + min(350.0, deadline - time.time() - 250.0)
    a_best, best_val = _adam_softmax_descent(
        n, adam_dead, target_sum,
        init_beta=0.5, max_beta=5e5, base_lr=0.12,
    )

    # ------------------------------------------------------------------
    # 3. Refine with exact sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline - 200.0:
        sub_dead = time.time() + min(150.0, deadline - time.time() - 150.0)
        a_best, sub_val = _subgradient_descent(
            a_best, sub_dead, target_sum,
            step_factor=0.5, gamma=0.99,
        )
        if sub_val < best_val:
            best_val = sub_val
            a_best = a_best.copy()
            print(f"[{time.time():.2f}] Sub‑GD refinement best={best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Permutation annealing (re‑ordering only)
    # ------------------------------------------------------------------
    if time.time() < deadline - 100.0:
        perm_dead = time.time() + min(80.0, deadline - time.time() - 80.0)
        a_best, perm_val = _permutation_targeted_search(
            a_best, perm_dead,
            start_temp=0.12, end_temp=1e-5,
            top_frac=0.03, low_frac=0.03,
        )
        if perm_val < best_val:
            best_val = perm_val
            a_best = a_best.copy()
            print(f"[{time.time():.2f}] Permutation search best={best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final deterministic mass‑transfer polish
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        mt_dead = time.time() + min(30.0, deadline - time.time() - 30.0)
        a_best, mt_val = _deterministic_mass_transfer(
            a_best, mt_dead,
            target_sum,
            start_temp=0.2, end_temp=1e-5,
            low_frac=0.04,
        )
        if mt_val < best_val:
            best_val = mt_val
            a_best = a_best.copy()
            print(f"[{time.time():.2f}] Mass‑transfer final best={best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Trim trailing zeros for compactness
    # ------------------------------------------------------------------
    arr = np.array(a_best, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # 7. Log the final evaluated score (optional, may be suppressed)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # store for possible reuse in subsequent calls
    globals()["height_sequence_1"] = final_seq
    return final_seq
