"""
Improved search for a step‑function that minimises the maximal self‑convolution.
The routine follows the original four‑phase scheme (parameter sweep → Adam → sub‑gradient
→ deterministic mass‑transfer) and adds a **penalty‑descent** stage that tries to
lower the current maximum by directly minimising the sum of violations
(max(conv)−M)_+.  All operations are performed on the simplex
{a_i≥0, Σa_i = √(2·n)} so the evaluation score is exactly the maximal convolution.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Small helper returning the next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of arbitrary length vectors x and y using FFT.
    Returns the first len(x)+len(y)-1 entries.
    """
    size = _next_pow2(x.size + y.size - 1)
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: x.size + y.size - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a (linear convolution)."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    inde = np.arange(1, n + 1)
    rho = np.nonzero(u - cssv / inde > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (quick pre‑search)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0))
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    """Centered Gaussian, clipped at the ends, normalised to the target sum."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Scoring (max‑convolution value)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Return the objective: maximal self‑convolution entry."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=size)
    b = np.fft.irfft(af * af, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    ar_f = np.fft.rfft(a_rev, n=size)
    pr_f = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(ar_f * pr_f, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape tiny plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several Adam restarts from jittered starts."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=2e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (true max‑convolution)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         max_iter: int = 5000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.flatnonzero(conv >= max_val - 1e-12)

        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()
        delta_target = max_val - second_max
        if delta_target < 1e-14:
            break

        k = idx_max[rng.integers(len(idx_max))]
        n = a.shape[0]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k - idx]

        norm2 = np.dot(g, g)
        if norm2 == 0.0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = _project_to_simplex(a + jitter, target_sum)
            it += 1
            continue

        step = 0.9 * delta_target / norm2
        step = min(step, target_sum * 0.4 / (np.linalg.norm(g) + 1e-12))

        for _ in range(6):
            a_candidate = _project_to_simplex(a - step * g, target_sum)
            cand_score = _score(a_candidate)
            if cand_score < best_score - 1e-12:
                a = a_candidate
                best_a = a_candidate.copy()
                best_score = cand_score
                print(f"[{time.time():.2f}] Subgrad it={it:5d} step={step:.3e} score={best_score:.6f}")
                break
            step *= 0.5
        else:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = _project_to_simplex(a + jitter, target_sum)

        it += 1

    return best_a, best_score

# Deterministic mass‑transfer (exact)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()
        if max_val - second_max < tol:
            break

        k = idx_max[0]      # pick the first maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # choose a receiver with the smallest partner (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score

def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Penalty‑descent: lower a given maximal convolution bound
# ----------------------------------------------------------------------


def _penalty_lower_max(a0: np.ndarray,
                       target_sum: float,
                       deadline: float,
                       target_max: float) -> tuple[np.ndarray, float]:
    """
    Attempt to find a feasible vector with max(conv) ≤ target_max.
    Gradient is taken from the penalty (max(conv)-target_max)_+.
    """
    a = a0.copy()
    best_a = a.copy()
    best_max = _max_convolution_fft(a)
    rng = np.random.default_rng()
    it = 0
    n = a.shape[0]

    while time.time() < deadline and best_max > target_max + 1e-12 and it < 20000:
        conv = _conv_fft(a)
        cur_max = conv.max()
        if cur_max <= target_max:
            best_max = cur_max
            best_a = a.copy()
            break

        # penalty vector (only entries above target_max)
        p = conv - target_max
        p[p < 0] = 0.0

        # gradient = 2 * (p * a_rev)_{n-1 … n-1+n-1}
        a_rev = a[::-1]
        grad_full = _conv_full_fft(p, a_rev)
        grad = 2.0 * grad_full[n - 1: n - 1 + n]

        norm2 = np.dot(grad, grad)
        if norm2 == 0.0:
            # inject a tiny random move and continue
            a = _project_to_simplex(a + rng.normal(scale=target_sum * 0.001, size=n), target_sum)
            it += 1
            continue

        # step size inspired by sub‑gradient theory
        step = 0.9 * (cur_max - target_max) / (norm2 + 1e-12)
        step = min(step, target_sum * 0.5 / (np.linalg.norm(grad) + 1e-12))

        # backtracking line search (max 5 reductions)
        improved = False
        for _ in range(5):
            a_candidate = _project_to_simplex(a - step * grad, target_sum)
            cand_max = _max_convolution_fft(a_candidate)
            if cand_max < cur_max - 1e-12:
                a = a_candidate
                best_max = cand_max
                best_a = a_candidate.copy()
                improved = True
                break
            step *= 0.5

        if not improved:
            # small random jitter to escape possible stall
            a = _project_to_simplex(a + rng.normal(scale=target_sum * 0.001, size=n), target_sum)

        it += 1

    return best_a, best_max

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:

    Search for a non‑negative step‑function that minimises the maximal
    self‑convolution (the evaluation score).  The routine follows the
    original four‑stage scheme and then applies a penalty‑descent refinement.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # problem size (larger n gives more freedom, here we stay with 12000)
    n = 12000 if budget_s >= 800 else (8000 if budget_s >= 400 else 5000)
    target_sum = np.sqrt(2.0 * n)   # Σa = √(2·n) → score = max‑convolution

    # ------------------------------------------------------------------
    # Phase 0 – dense parametric pre‑search
    # ------------------------------------------------------------------
    best_seq = None
