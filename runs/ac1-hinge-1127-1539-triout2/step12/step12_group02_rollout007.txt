from scipy.optimize import linprog
import scipy.sparse as sp

def lp_step(a, M, target_sum):
    n = len(a)
    rows = []
    cols = []
    data = []
    # for each k (row):
    for k in range(2*n-1):
        i_min = max(0, k-(n-1))
        i_max = min(n-1, k)
        # i from i_min..i_max
        for i in range(i_min, i_max+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(a[i])
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # constraints: A @ b <= M
    bounds = [(0, None)] * n
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if not res.success:
        # maybe fallback to zeros?
        b = np.zeros_like(a)
    else:
        b = res.x
    # rescale
    if b.sum() > 0:
        b = b * (target_sum / b.sum())
    else:
        b = np.full_like(a, target_sum/n)
    return b
```

"""
Propose a step‑function sequence with low benchmark value.

The algorithm combines several complementary optimisation ideas:

1. **Rapid low‑dimensional exploration** – several short gradient
   descents (soft‑max approximation of the max‑convolution) on a
   modest size (≈2000) to obtain a good shape.

2. **Linear‑programming refinement** – for the current shape solve
   a linear programme that maximises the ℓ¹‑mass of a new sequence
   under the same peak‑convolution bound.  A convex combination of the
   old and the LP solution strictly reduces the objective.

3. **Upscaling** – the refined shape is interpolated to a larger length
   (≈10000–12000) while preserving the ℓ¹‑norm  Σa = √(2n).

4. **High‑β soft‑max gradient descent** – fine‑tunes the values on the
   large scale.

5. **Targeted simulated‑annealing on permutations** – repeatedly swaps
   a high‑contributing index with a low‑contributing one, using a
   slow temperature schedule.  This exploits the fact that the
   convolution depends heavily on the ordering of the coefficients.

6. **Tiny deterministic mass‑transfer polishing** – final local moves
   that shift a small amount of mass from the strongest donor to the
   weakest receiver.

All steps respect non‑negativity and the ℓ¹‑simplex constraint
`∑a = sqrt(2·n)`.  The routine stores the best sequence in the global
`height_sequence_1` variable so that subsequent runs can start from the
current optimum.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (FFT version)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (FFT version)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (∑a)² (∑a is assumed positive)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projects v onto the ℓ¹‑simplex of radius s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric sequence generators (used for diverse initial pool)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation resize preserving shape."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv_vector).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_conv = np.max(conv)
    exps = np.exp(beta * (conv - max_conv))
    p = exps / np.sum(exps)                # softmax weights over convolution indices

    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    p_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1: n - 1 + n]   # only the central n entries affect a
    return grad, conv


# ----------------------------------------------------------------------
# Projected (momentum) gradient descent on the soft‑max objective
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.5,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Momentum gradient descent with exponential β‑annealing."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid flat plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (small local moves)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-4,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # Locate current peak and the contributing pairs
        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]
        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted permutation simulated‑annealing
# ----------------------------------------------------------------------


def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-5,
                                 top_frac: float = 0.07,
                                 low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Simulated annealing that swaps a high‑contributing index with a low‑contributing one.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    n = a.shape[0]
    start_time = time.time()

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        kmax_candidates = np.flatnonzero(conv == max_val)
        kmax = rng.choice(kmax_candidates)

        # contribution of each index to the current peak
        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)

        if donor == receiver:
            continue

        # swap
        a[donor], a[receiver] = a[receiver], a[donor]

        new_score = _score(a)

        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] PermSearch new best={best_score:.6f}")
        else:
            # revert swap
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (maximise ℓ¹‑mass under peak bound)
# ----------------------------------------------------------------------


def _lp_max_sum_under_conv(a: np.ndarray, M: float, target_sum: float) -> np.ndarray:
    """
    Solve   max Σ b_j   s.t.   (a * b)_k ≤ M  for all k,   b ≥ 0.
    Returns the solution scaled to the required ℓ¹‑norm.
    """
    n = a.shape[0]
    # Build sparse matrix A where A[k, j] = a[i] with i+j = k
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(a[i])
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise Σb  ↔ minimise -Σb
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback: uniform distribution
        g = np.full(n, target_sum / n, dtype=np.float64)
    else:
        g = res.x

    # Scale to exact target ℓ¹‑norm (the LP does not enforce it)
    total = g.sum()
    if total > 0:
        g = g * (target_sum / total)
    else:
        g = np.full_like(g, target_sum / n)
    return g

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimizing the benchmark score.
    The routine proceeds through the stages described in the module docstring.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5       # tiny safety margin

    # ------------------------------------------------------------------
    # 1. Choose a modest dimension for the quick LP / GD phase
    # ------------------------------------------------------------------
    n_small = 2000 if budget_s >= 300 else max(800, int(budget_s * 2))
    target_small = np.sqrt(2.0 * n_small)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # 2. Build a diverse initial pool of candidates
    # ------------------------------------------------------------------
    init_pool = []

    # power‑law family
    for p in np.linspace(0.35, 0.75, 9):
        init_pool.append(_power_law_sequence(n_small, float(p), target_small))

    # exponential family
    for lam in np.linspace(0.0015, 0.015, 7):
        init_pool.append(_exponential_sequence(n_small, float(lam), target_small))

    # random Dirichlet‑like points
    for _ in range(6):
        a = rng.random(n_small)
        a *= target_small / a.sum()
        init_pool.append(a)

    # linear decreasing shape
    lin = np.linspace(1.0, 0.0, n_small)
    lin *= target_small / lin.sum()
    init_pool.append(lin)

    # reuse previous best if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_small))

    # jittered copies for extra diversity
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_small * 0.008, size=n_small)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_small)
        init_pool.append(seq_j)

    # ------------------------------------------------------------------
    # 3. Short gradient‑descent explorations to obtain a decent shape
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    while time.time() < deadline - 30.0:   # reserve time for later phases
        # pick a seed (prefer the current best with probability)
        if best_seq is not None and rng.random() < 0.55:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_small * 0.004, size=n_small)
            seed = seed + jitter
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_small)
        else:
            seed = rng.choice(init_pool)

        # Run a bounded GD instance (≈15–20 s each)
        run_dead = min(time.time() + 18.0, deadline - 12.0)
        if run_dead <= time.time():
            break
        a_opt, val_opt = _annealed_gradient_descent(
            seed,
            run_dead,
            target_small,
            max_beta=3e5,
            step_factor=0.7,
            gamma=0.99,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best score={best_score:.6f}")

    # fallback if nothing was generated
    if best_seq is None:
        best_seq = _power_law_sequence(n_small, 0.55, target_small)

    # ------------------------------------------------------------------
    # 4. Linear‑programming refinement (few iterations)
    # ------------------------------------------------------------------
    for _ in range(3):
        if time.time() > deadline - 20.0:
            break
        M = _max_convolution_fft(best_seq)
        g = _lp_max_sum_under_conv(best_seq, M, target_small)
        # convex combination
        t = 0.25
        best_seq = (1 - t) * best_seq + t * g
        best_seq = _project_to_simplex(best_seq, target_small)
        cur_val = _score(best_seq)
        if cur_val < best_score:
            best_score = cur_val
            print(f"[{time.time():.2f}] LP refinement improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Upscale to a larger length (if time permits)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 50.0:
        n_large = 12000 if budget_s >= 800 else 10000
    elif remaining > 20.0:
        n_large = 8000
    else:
        n_large = n_small
    target_large = np.sqrt(2.0 * n_large)

    seq_big = _resize_sequence(best_seq, n_large)
    seq_big = _project_to_simplex(seq_big, target_large)

    # try a couple of simple reorderings before polishing
    cand_alt = [_alternating_order(seq_big), _alternating_order(seq_big[::-1])[::-1]]
    for cand in cand_alt:
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            seq_big = cand
            print(f"[{time.time():.2f}] Reordering gave score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. High‑β gradient descent on the large problem (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 15.0:
        a_opt, val_opt = _annealed_gradient_descent(
            seq_big,
            deadline - 8.0,
            target_large,
            max_beta=2e6,
            step_factor=0.5,
            gamma=0.995,
        )
        if val_opt < best_score:
            best_score = val_opt
            seq_big = a_opt.copy()
            print(f"[{time.time():.2f}] High‑β GD improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Extended permutation‑only simulated annealing (uses most remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        perm_dead = deadline - 0.5
        a_opt, val_opt = _permutation_targeted_search(
            seq_big,
            perm_dead,
            start_temp=0.8,
            end_temp=1e-6,
            top_frac=0.07,
            low_frac=0.07,
        )
        if val_opt < best_score:
            best_score = val_opt
            seq_big = a_opt.copy()
            print(f"[{time.time():.2f}] Permutation search final best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Tiny deterministic mass‑transfer polishing (final few seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        mt_dead = time.time() + min(5.0, deadline - time.time() - 0.3)
        a_opt, val_opt = _deterministic_mass_transfer(
            seq_big,
            mt_dead,
            target_large,
            start_temp=0.2,
            end_temp=1e-5,
            low_frac=0.05,
        )
        if val_opt < best_score:
            best_score = val_opt
            seq_big = a_opt.copy()
            print(f"[{time.time():.2f}] Final MT polishing best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 9. Trim trailing zeros and return a plain Python list
    # ------------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # 10. Log final evaluation (optional)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for possible later reuse
    globals()["height_sequence_1"] = final_seq
    return final_seq
