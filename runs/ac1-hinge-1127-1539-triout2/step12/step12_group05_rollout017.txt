"""
Improved step‑function optimizer.

Main ideas
----------
* Use a larger problem size (n up to 20000 for the full budget) – more
  degrees of freedom usually allow a lower maximal convolution.
* Expand the family of parametric seeds: in addition to the classic
  decreasing power/law families we also try symmetric “U‑shaped”
  constructions where the mass is concentrated near both ends of the
  interval.
* Perform a fast random‑Dirichlet sampling stage to discover promising
  shapes that are not covered by the analytic families.
* Refine the best seed with many runs of the exact‑sub‑gradient guided
  mass‑transfer (the “MT” phase).  If a run stalls we inject a tiny
  jitter and restart – this repeatedly escapes plateaus.
* A cheap hill‑climbing neighbour search (mass swaps) is applied after
  each MT run to squeeze out a few extra improvements.
* Finally a short projected‑Adam run with a huge soft‑max temperature
  polishes the result.

All phases respect the simplex constraint Σa = √(2 n) and non‑negativity.
The routine respects the supplied wall‑clock budget and returns the
best sequence found (trailing near‑zero entries are trimmed).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) using a real FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft_pair(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution x*y (full length 2·n‑1) using a real FFT."""
    n = x.size
    size = 1 << ((2 * n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Benchmark score ( equals the maximal convolution entry when Σa = √(2n) )."""
    max_b = _max_conv(a)
    return max_b


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (dense pre‑search)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _broken_power_law_sequence(
    n: int, p1: float, p2: float, alpha: float, target_sum: float
) -> np.ndarray:
    K = max(1, int(alpha * n))
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.empty_like(idx)
    a[:K] = idx[:K] ** (-p1)
    a[K:] = idx[K:] ** (-p2)
    a *= target_sum / a.sum()
    return a


def _sym_power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 1.0
    rev = (n - np.arange(n, dtype=np.float64))  # also 1..n
    a = i ** (-p) + rev ** (-p)
    a *= target_sum / a.sum()
    return a


def _sym_exp_sequence(n: int, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    rev = n - 1 - i
    a = np.exp(-q * i) + np.exp(-q * rev)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Exact sub‑gradient guided mass‑transfer (Phase MT)
# ----------------------------------------------------------------------


def _compute_gradient_for_max(a: np.ndarray, max_idx: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a (averaged over the
    anti‑diagonal indices where the maximum is attained).
    """
    n = a.size
    g = np.zeros_like(a)
    for k in max_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g[idx] += 2.0 * a[k - idx]  # contribution 2 * a[k-i]
    if max_idx.size:
        g /= max_idx.size
    return g


def _guided_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Greedy exact reduction of the maximal convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        # second‑largest (excluding all that are within tol of the max)
        mask_max = conv >= max_val - tol
        if np.any(~mask_max):
            second_max = conv[~mask_max].max()
        else:
            second_max = 0.0
        max_idx = np.where(mask_max)[0]

        # exact sub‑gradient (averaged over all maximal indices)
        g = _compute_gradient_for_max(a, max_idx)

        # direction that preserves Σa = 0
        d = -g + g.sum() / n * np.ones_like(a)

        # maximum step before any component becomes negative
        neg = d < 0
        if np.any(neg):
            t_nonneg = np.min(a[neg] / -d[neg])
        else:
            t_nonneg = np.inf

        # linearised change of the convolution in direction d
        delta = 2.0 * _conv_fft_pair(a, d)
        delta_max = delta[max_idx]
        delta_k = delta_max.min()

        if delta_k >= 0:
            # direction does not reduce the peak → give up for this run
            break

        # step needed to drive the current max down to the second max
        t_to_second = (max_val - second_max) / (-delta_k)

        # safe step (a little smaller to stay away from the boundary)
        t = min(t_nonneg, t_to_second) * 0.99
        if t <= 0:
            break

        a += t * d
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best = a.copy()
            print(f"[{time.time():.2f}] Guided MT it={it} score={best_val:.6f}")

        # occasional jitter to escape tiny plateaus
        if it % 500 == 0 and rng.random() < 0.07:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best, best_val


# ----------------------------------------------------------------------
# Simple hill‑climbing neighbour search
# ----------------------------------------------------------------------


def _hill_climb(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    move_fraction: float = 0.2,
) -> tuple[np.ndarray, float]:
    """
    Very cheap local search: repeatedly move a random fraction of mass
    from a source index to a destination index, keep the move if it
    improves the maximal convolution.
    """
    a = a0.copy()
    best = a.copy()
    best_val = _score(a)
    n = a.shape[0]
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j or a[i] == 0.0:
            continue
        delta = a[i] * move_fraction * rng.random()
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)

        val = _score(cand)
        if val < best_val - 1e-12:
            best_val = val
            best = cand.copy()
            a = cand  # continue from the improved point
            # print(f"[{time.time():.2f}] Hill climb new best={best_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal benchmark score.
    The routine respects the supplied time budget and returns the best
    sequence found (trailing near‑zero entries are trimmed).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ---------------------------------------------------------------
    # Choose problem size – a larger n gives a finer discretisation.
    # ---------------------------------------------------------------
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 400:
        n = 15000
    else:
        n = 10000

    target_sum = math.sqrt(2.0 * n)  # Σa is fixed to this value

    # ---------------------------------------------------------------
    # Phase 0 – dense parametric pre‑search (very cheap)
    # ---------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    rng = np.random.default_rng()

    # (a) pure power‑law
    for p in np.arange(0.20, 2.01, 0.05):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # (b) exponential‑power‑law
    for p in np.arange(0.20, 2.01, 0.10):
        for q in np.arange(0.0, 8.001, 0.5):
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # (c) broken power‑law (two‑segment)
    for p1 in np.arange(0.20, 1.01, 0.2):
        for p2 in np.arange(0.80, 2.31, 0.3):
            for alpha in np.arange(0.1, 0.91, 0.2):
                seq = _broken_power_law_sequence(
                    n, float(p1), float(p2), float(alpha), target_sum
                )
                val = _score(seq)
                if val < best_score:
                    best_score, best_seq = val, seq.copy()

    # (d) symmetric power‑law (U‑shaped)
    for p in np.arange(0.20, 2.01, 0.10):
        seq = _sym_power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # (e) symmetric exponential
    for q in np.arange(0.0, 8.001, 0.5):
        seq = _sym_exp_sequence(n, float(q), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # (f) warm start from a previously found construction, if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score, best_seq = val, resized.copy()

    print(f"[{time.time():.2f}] After parametric pre‑search: best score = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase R – random Dirichlet sampling (cheap but diverse)
    # ---------------------------------------------------------------
    rand_samples = 3000 if budget_s >= 800 else 1500
    alpha_dirichlet = 0.4  # <1 favours sparsity
    while rand_samples > 0 and time.time() < deadline - 5.0:
        raw = rng.gamma(alpha_dirichlet, size=n)
        seq = _project_to_simplex(raw, target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()
            print(f"[{time.time():.2f}] Random Dirichlet improved best = {best_score:.6f}")
        rand_samples -= 1

    # ---------------------------------------------------------------
    # Phase MT – repeated guided mass‑transfer with occasional jitter
    # ---------------------------------------------------------------
    while time.time() < deadline - 10.0:
        # allocate a short time slice for a single MT run
        slice_deadline = time.time() + min(40.0, deadline - time.time() - 10.0)
        cand, cand_val = _guided_mass_transfer(best_seq, slice_deadline, target_sum)

        if cand_val < best_score - 1e-12:
            best_score, best_seq = cand_val, cand.copy()
            print(f"[{time.time():.2f}] MT improved best = {best_score:.6f}")

        # After MT, try a quick hill‑climb to polish any remaining slack
        hc_deadline = time.time() + min(15.0, deadline - time.time() - 5.0)
        cand2, cand2_val = _hill_climb(best_seq, hc_deadline, target_sum)
        if cand2_val < best_score - 1e-12:
            best_score, best_seq = cand2_val, cand2.copy()
            print(f"[{time.time():.2f}] Hill‑climb improved best = {best_score:.6f}")

        # If the MT run made no progress, inject a small jitter to escape
        if cand_val >= best_score - 1e-12:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            best_seq = best_seq + jitter
            best_seq = np.maximum(best_seq, 0.0)
            best_seq = _project_to_simplex(best_seq, target_sum)

    # ---------------------------------------------------------------
    # Phase P – final polishing with a high‑β projected Adam run
    # ---------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        # a tiny Adam run (soft‑max temperature grows to a huge value)
        # we reuse the Adam implementation from the previous state‑of‑the‑art
        # but with a very aggressive schedule.
        def _adam_grad_softmax(a: np.ndarray, beta: float):
            n = a.size
            size = 1 << ((2 * n - 1).bit_length())
            a_fft = np.fft.rfft(a, n=size)
            conv_full = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

            max_b = conv_full.max()
            exps = np.exp(beta * (conv_full - max_b))
            p = exps / exps.sum()

            a_rev = a[::-1]
            Ar_fft = np.fft.rfft(a_rev, n=size)
            Pr_fft = np.fft.rfft(p, n=size)
            conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

            grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
            return grad, conv_full

        def _adam_run(a0: np.ndarray, dl: float, s: float) -> np.ndarray:
            a = _project_to_simplex(a0.copy(), s)
            rng_loc = np.random.default_rng()
            m = np.zeros_like(a)
            v = np.zeros_like(a)
            beta = 0.5
            gamma = 0.999
            step_factor = 0.6
            eps = 1e-8
            it = 0
            beta_inc_iter = 200
            while time.time() < dl:
                g, _ = _adam_grad_softmax(a, beta)
                m = gamma * m + (1 - gamma) * g
                v = gamma * v + (1 - gamma) * (g * g)
                m_hat = m / (1 - gamma ** (it + 1))
                v_hat = v / (1 - gamma ** (it + 1))
                step = step_factor * s / (np.sqrt(v_hat) + eps)
                a -= step * m_hat
                a = _project_to_simplex(a, s)

                if it % 250 == 0:
                    jitter = rng_loc.normal(
                        scale=s * 0.001 / (beta + 1.0), size=a.shape
                    )
                    a += jitter
                    a = _project_to_simplex(a, s)

                if it % beta_inc_iter == 0 and beta < 1e6:
                    beta = min(beta * 2.0, 1e6)

                it += 1
            return a

        final_deadline = deadline
        best_seq = _adam_run(best_seq, final_deadline, target_sum)

    # ---------------------------------------------------------------
    # Post‑processing
    # ---------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final evaluation for reporting
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated benchmark score = {val:.6f}")
    except Exception:
        pass

    return final_seq
