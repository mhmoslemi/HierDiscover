def _adam_subgradient(a0, deadline, target_sum):
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_val = _score_seq(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow = beta1
    beta2_pow = beta2
    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        # compute convolution
        conv = _conv_fft(a)
        maxb = conv.max()
        # indices reaching near max within tolerance
        tol = 1e-12
        active = np.where(conv >= maxb * (1 - tol))[0]
        # accumulate gradient
        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1)
            # contribution: 2 * a[k - idx]
            grad[idx] += 2.0 * a[k - idx]
        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)
        step = lr / (np.sqrt(v_hat) + epsilon)
        a = a - step * m_hat
        # project
        a = _project_to_simplex(a, target_sum)
        # occasional jitter
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = a + jitter
            a = _project_to_simplex(a, target_sum)
        cur_val = _score_seq(a)
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            print(f"[{time.time():.2f}] Adam it={it} score={best_val:.6f}")
        beta1_pow *= beta1
        beta2_pow *= beta2
        it += 1
    return best, best_val

def _adam_subgradient(a0, deadline, target_sum, lr=0.7, beta1=0.9, beta2=0.999, epsilon=1e-8):
    a = _project_to_simplex(a0.copy(), target_sum)
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_val = _score_seq(a)
    it = 0
    beta1_pow = beta1
    beta2_pow = beta2
    rng = np.random.default_rng()
    size = 1 << ((2 * n - 1).bit_length())
    while time.time() < deadline:
        # Convolution
        a_fft = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(a_fft * a_fft, n=size)[:2 * n - 1]
        maxb = conv.max()
        # active indices (within tolerance)
        # Use relative tolerance, but to avoid floating errors
        eps = 1e-12
        active = np.where(conv >= maxb * (1 - eps))[0]

        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)
        step = lr / (np.sqrt(v_hat) + epsilon)
        a = a - step * m_hat

        # projection
        a = _project_to_simplex(a, target_sum)

        # jitter occasionally to escape stagnation
        if it % 800 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score_seq(a)
        if cur_val < best_val:
            best_val = cur_val

"""
Step‑function optimizer (≈3 × speed, deeper refinement).

The algorithm works in three stages:

1. **Seed generation** – a diverse pool of candidate shapes
   (power‑law, exponential, Gaussian, random simplex) is created for the
   target length `n`.  The best seed (smallest max‑self‑convolution) is
   kept.

2. **Continuous refinement** – a projected Adam sub‑gradient method
   (exact max‑convolution gradient) is run for the bulk of the time
   budget.  The simplex projection guarantees `a_i ≥ 0` and
   `∑a_i = √(2 n)`.  Occasional jitter avoids stagnation.

3. **Discrete fine‑tuning** – the multiset of values is shuffled,
   deterministic reorderings (alternating/even‑spread) are tested,
   and a cheap simulated‑annealing mass‑transfer pass is applied.

All phases respect the constraints required by `evaluate_sequence`.
The final sequence is trimmed of trailing zeros and stored in the global
variable `height_sequence_1` for warm starts in future runs.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (linear convolution)."""
    return float(_conv_fft(a).max())


def _score_seq(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a)/(∑a)² (for ∑a=√(2n) this equals max‑conv)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project onto {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma_frac: float, target_sum: float) -> np.ndarray:
    """Discrete Gaussian centred on the middle of the array."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread large values with a step coprime to the length (≈√n)."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Annealed gradient descent (soft‑max approximation)
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.9,
                               gamma: float = 0.98) -> tuple[np.ndarray, float]:
    """Projected momentum GD on the soft‑max approximation of max‑conv."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score_seq(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    it = 0
    while time.time() < deadline:
        # --- soft‑max of the convolution ---
        fa = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
        max_b = conv.max()
        exps = np.exp(beta * (conv - max_b))
        p = exps / np.sum(exps)

        # --- gradient w.r.t. a (via reversed convolution) ---
        a_rev = a[::-1]
        Ar_fft = np.fft.rfft(a_rev, n=size)
        Pr_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        # --- momentum & projection ---
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score_seq(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AGD it={it:5d} β={beta:7.1f} score={best_val:.6f}")

        # beta schedule
        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Adam‑style projected sub‑gradient descent (exact max‑conv)
# ----------------------------------------------------------------------


def _adam_subgradient(a0: np.ndarray,
                      deadline: float,
                      target_sum: float,
                      lr: float = 0.7,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      epsilon: float = 1e-8) -> tuple[np.ndarray, float]:
    """Adam optimizer on the exact max‑convolution sub‑gradient."""
    a = _project_to_simplex(a0.copy(), target_sum)
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_val = _score_seq(a)

    it = 0
    beta1_pow = beta1
    beta2_pow = beta2
    rng = np.random.default_rng()

    while time.time() < deadline:
        # convolution (FFT)
        fa = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
        maxb = conv.max()

        # active indices (within tiny relative tolerance)
        eps = 1e-12
        active = np.where(conv >= maxb * (1 - eps))[0]

        # exact sub‑gradient of the max‑convolution
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)
        step = lr / (np.sqrt(v_hat) + epsilon)
        a = a - step * m_hat

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 800 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score_seq(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} score={best_val:.6f}")

        it += 1
        beta1_pow *= beta1
        beta2_pow *= beta2

    return best_a, best_val


# ----------------------------------------------------------------------
# Permutation search (random shuffles)
# ----------------------------------------------------------------------


def _permutation_search(base_seq: np.ndarray,
                        max_trials: int,
                        deadline: float) -> tuple[np.ndarray, float]:
    """Try many random reorderings and keep the one with smallest max‑conv."""
    best_seq = base_seq.copy()
    best_val = _max_convolution_fft(best_seq)
    rng = np.random.default_rng()
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        perm = rng.permutation(base_seq)
        val = _max_convolution_fft(perm)
        if val < best_val:
            best_val = val
            best_seq = perm.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Greedy mass‑transfer with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.3,
                                      end_temp: float = 1e-5,
                                      low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    """Greedy mass moves from heavy contributors to light ones."""
    a = a0.copy()
    cur_score = _score_seq(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start = time.time()
    n = a.shape[0]

    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        # current peak location
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score_seq(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score_seq(a)

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine follows a three‑stage pipeline described in the module
    docstring and never exceeds `budget_s` seconds.
    """
    start = time.time()
    deadline = start + budget_s - 0.5  # tiny safety margin
    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Choose a final length based on the time budget
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 800:
        n_final = 40000
    elif remaining > 600:
        n_final = 35000
    elif remaining > 400:
        n_final = 30000
    elif remaining > 250:
        n_final = 25000
    elif remaining > 150:
        n_final = 20000
    else:
        n_final = 15000

    target_sum = math.sqrt(2.0 * n_final)

    # ------------------------------------------------------------------
    # Build a pool of seeds (diverse shapes)
    # ------------------------------------------------------------------
    seed_pool = []

    # start from previous best if it exists
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.shape[0] != n_final:
            prev = _resize_sequence(prev, n_final)
        prev = _project_to_simplex(prev, target_sum)
        seed_pool.append(prev)

    # power‑law family
    for p in (0.3, 0.5, 0.8, 1.0, 1.5):
        seed_pool.append(_power_law_sequence(n_final, p, target_sum))

    # exponential family
    for lam in (0.001, 0.005, 0.01):
        seed_pool.append(_exponential_sequence(n_final, lam, target_sum))

    # Gaussian‑like families
    for sigma_frac in (0.12, 0.18, 0.25):
        seed_pool.append(_gaussian_sequence(n_final, sigma_frac, target_sum))

    # pure random points on the simplex
    for _ in range(6):
        seed_pool.append(_random_simplex(n_final, target_sum, rng))

    # ------------------------------------------------------------------
    # Pick the best seed according to raw max‑convolution
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    for seq in seed_pool:
        val = _max_convolution_fft(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
    print(f"[{time.time():.2f}] Initial seed max‑conv = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – short annealed‑gradient refinement (exploration)
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    agd_time = min(30.0, remaining * 0.15)  # ~15 % of time
    if agd_time > 5.0:
        best_seq, agd_score = _annealed_gradient_descent(
            best_seq, now + agd_time, target_sum,
            max_beta=8e5, step_factor=0.9, gamma=0.98)
        print(f"[{time.time():.2f}] After AGD best score = {agd_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – deterministic reorderings on the large scale
    # ------------------------------------------------------------------
    candidates = [
        best_seq,
        _alternating_order(best_seq),
        _even_spread_order(best_seq)
    ]
    best_seq = min(candidates, key=_max_convolution_fft)
    print(f"[{time.time():.2f}] After deterministic reorder max‑conv = "
          f"{_max_convolution_fft(best_seq):.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam projected sub‑gradient descent (main optimisation)
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    # reserve a few seconds for the final polishing phases
    sub_time = max(5.0, remaining - 12.0)
    if sub_time > 5.0:
        best_seq, sub_score = _adam_subgradient(
            best_seq, now + sub_time, target_sum,
            lr=0.7, beta1=0.9, beta2=0.999, epsilon=1e-8)
        print(f"[{time.time():.2f}] After Adam sub‑grad best score = {sub_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – random permutation search (fast heuristic)
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    if remaining > 3.0:
        max_trials = int(min(2500, remaining * 150))  # rough budget
        best_seq, perm_max = _permutation_search(best_seq, max_trials, deadline)
        perm_score = _score_seq(best_seq)
        print(f"[{time.time():.2f}] After permutation search best score = {perm_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final greedy mass‑transfer simulated annealing
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    if remaining > 2.0:
        best_seq, mt_score = _mass_transfer_simulated_annealing(
            best_seq, deadline, target_sum,
            start_temp=0.25, end_temp=1e-5, low_frac=0.07)
        print(f"[{time.time():.2f}] Final mass‑transfer best = {mt_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Store the result for warm‑starts in future runs
    # ------------------------------------------------------------------
    globals()["height_sequence_1"] = final_seq

    # Optional final evaluation (purely for logging)
    try:
        ev = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {ev:.6f}")
    except Exception as exc:
        print("Evaluation error:", exc)

    return final_seq
