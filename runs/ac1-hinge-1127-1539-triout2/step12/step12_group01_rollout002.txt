    deadline = start + budget_s - 0.1  # small safety margin

    # Choose n based on budget
    if budget_s >= 800:
        n = 15000  # a bit smaller than earlier for speed
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)

    # Warm start from global variable if present, else power law
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=np.float64)
        # If init length differs, resize via interpolation
        if init_seq.shape[0] != n:
            init_seq = np.interp(np.linspace(0,1,n), np.linspace(0,1,init_seq.shape[0]), init_seq)
    else:
        init_seq = (np.arange(1, n+1, dtype=np.float64) ** -0.55)

    # scale to target_sum
    init_seq = init_seq * (target_sum / init_seq.sum())

    # initial best
    a = init_seq.copy()
    b = conv_self(a)
    cur_val = score_from_conv(b, n, target_sum) # compute objective = max_b

    best_a = a.copy()
    best_val = cur_val
    print(f"[{time.time():.2f}] Initial score {best_val:.6f}")

    rng = np.random.default_rng()

    # parameters for annealing
    T0 = cur_val * 1.5  # high temp
    total_time = deadline - start
    max_iter = 0
    # We'll loop until deadline but also track iteration.
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total_time
        # temperature schedule: keep high longer: e.g., T = T0 * (1 - sqrt(frac))
        temperature = T0 * max(0.0, 1.0 - np.sqrt(frac))

        # find peak k
        k_max = int(np.argmax(b))
        # compute donor candidates: indices that contribute to k_max
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi+1, dtype=np.int64)

        contributions = a[idx] * a[k_max - idx]
        sum_contrib = contributions.sum()
        if sum_contrib > 0:
            donor = int(rng.choice(idx, p=contributions/sum_contrib))
        else:
            donor = int(rng.integers(lo, hi+1))

        # pick receiver: low a
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = int(rng.choice(n, p=inv))

        if donor == receiver:
            continue

        # size of transfer: delta_max based on temperature
        # earlier we can set delta_max = min(a[donor], target_sum * 0.02) * (temperature / T0 + 0.1)
        # ensure delta > 0.

        temp_ratio = temperature / T0 if T0 > 0 else 0.0
        # make base delta factor bigger early
        base_factor = 0.02  # up to 2% of total sum
        # increase with temp ratio
        max_delta = min(a[donor], target_sum * base_factor) * (0.3 + 0.7*temp_ratio)  # at start near full base_factor
        # add small random fraction
        delta = rng.random() * max_delta

        if delta <= 0.0:
            continue

        # update convolution b_new efficiently
        # b_new = b copy first
        b_new = b.copy()
        # linear part contributions
        # For index shift: contribution of adding/subtracting delta to/from positions:

        # Actually change from donor removal: a[donor] reduces by delta => b'[k] = b[k] - 2*delta*a[k-donor] for each k where a index k-donor exists (i.e., k in donor..donor+n-1)

"""
Optimization of a non‑negative step function via aggressive simulated
annealing with large early temperature and mass‑transfer moves.
The algorithm works in three logical stages:

1. **Warm‑start** – use the best known construction (`height_sequence_1`)
   if available, otherwise a simple power‑law.  The sequence is resized
   to the chosen length and projected onto the simplex
   Σa = √(2 n).

2. **Aggressive annealing** – repeatedly move a small amount of mass
   from a “donor’’ index that contributes most to the current maximal
   convolution entry to a “receiver’’ index that is (probabilistically)
   low.  The move size is scaled with a high initial temperature
   (≈ 2 × the current peak) and decays with a √‑schedule, allowing
   large exploratory steps early on.  Exact O(n) updates of the full
   self‑convolution are performed, and a Metropolis acceptance rule
   based on the *current* energy is used.

3. **Periodic correction** – every few thousand iterations the full
   convolution is recomputed via FFT to eliminate drift and a tiny
   random jitter is added to the sequence to escape shallow minima.

The best sequence found before the deadline is returned (trailing
zeros are removed) and stored in the global variable
`height_sequence_1` for warm‑starting future runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score_from_conv(b: np.ndarray) -> float:
    """
    Objective for a sequence with Σa = √(2 n).  In that case the factor
    2·n/Σa² equals 1, so the score is simply max(b).
    """
    return float(b.max())


def _project_to_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale a non‑negative vector to have the prescribed ℓ₁‑sum."""
    s = v.sum()
    if s == 0:
        return np.full_like(v, target_sum / len(v))
    return v * (target_sum / s)


# ----------------------------------------------------------------------
# Aggressive annealing
# ----------------------------------------------------------------------


def _aggressive_anneal(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Simulated‑annealing mass‑transfer with a high initial temperature and
    large early moves.
    Returns the best sequence (as a NumPy array) and its score.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    cur_val = _score_from_conv(b)        # equals max(b) because Σa = sqrt(2n)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    # start temperature (≈ 2× current peak)
    T0 = max(1e-12, cur_val * 2.0)

    it = 0
    while time.time() < deadline:
        # --------------------------------------------------------------
        # Temperature schedule – keep it hot longer (√‑schedule)
        # --------------------------------------------------------------
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        # --------------------------------------------------------------
        # Identify the current peak of the convolution
        # --------------------------------------------------------------
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # --------------------------------------------------------------
        # Choose a donor (large contribution to the peak)
        # --------------------------------------------------------------
        contrib = a[idx] * a[k_max - idx]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            donor = int(rng.choice(idx, p=contrib / total_contrib))
        else:
            donor = int(rng.integers(lo, hi + 1))

        # --------------------------------------------------------------
        # Choose a receiver (prefer a small height)
        # --------------------------------------------------------------
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = int(rng.choice(n, p=inv))

        if donor == receiver:
            it += 1
            continue

        # --------------------------------------------------------------
        # Size of the transferred mass – scaled with temperature
        # --------------------------------------------------------------
        # Allow up to 2 % of the total mass in a single move (large early)
        max_delta = min(a[donor], target_sum * 0.02)
        # Scale down as the temperature cools
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * (0.1 + 0.9 * scale)

        if delta <= 0.0:
            it += 1
            continue

        # --------------------------------------------------------------
        # Efficient O(n) update of the full convolution
        # --------------------------------------------------------------
        b_new = b.copy()
        # linear contribution of the transfer
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic terms
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(b_new.max())

        # --------------------------------------------------------------
        # Metropolis acceptance based on *current* energy
        # --------------------------------------------------------------
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}]  it={it:6d}  new best {best_val:.6f}")

        # --------------------------------------------------------------
        # Periodic full recomputation + tiny jitter to avoid drift
        # --------------------------------------------------------------
        if it % 4000 == 0:
            # re‑normalise (tiny numerical drift) and recompute convolution
            a = _project_to_sum(a, target_sum)
            b = _conv_fft(a)
            cur_val = _score_from_conv(b)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}]  it={it:6d}  recompute improved best {best_val:.6f}")

            # random jitter – promotes exploration after a while
            jitter = rng.normal(scale=target_sum * 0.0004, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_sum(a, target_sum)
            b = _conv_fft(a)
            cur_val = _score_from_conv(b)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}]  it={it:6d}  jitter improved best {best_val:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.25   # tiny safety margin

    # --------------------------------------------------------------
    # Choose problem size – larger n gives finer granularity but costs time
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 15000                # a compromise between quality and speed
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Warm‑start (reuse previous construction if available)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.shape[0] != n:
            # linear interpolation to the new length
            old_x = np.linspace(0.0, 1.0, prev.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            init = np.interp(new_x, old_x, prev)
        else:
            init = prev.copy()
    else:
        # simple decreasing power‑law as a generic starter
        init = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))

    # enforce the sum constraint
    init = _project_to_sum(init, target_sum)

    # --------------------------------------------------------------
    # Initial score
    # --------------------------------------------------------------
    b0 = _conv_fft(init)
    cur_score = _score_from_conv(b0)
    print(f"[{time.time():.2f}] initial score = {cur_score:.6f}")

    # --------------------------------------------------------------
    # Aggressive annealing (the heavy lifting)
    # --------------------------------------------------------------
    best_seq, best_score = _aggressive_anneal(init, target_sum, deadline)

    # --------------------------------------------------------------
    # Final clean‑up: trim trailing zeros
    # --------------------------------------------------------------
    arr = best_seq
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # --------------------------------------------------------------
    # Final evaluation (for diagnostics)
    # --------------------------------------------------------------
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] evaluation error: {exc}")

    # store for warm‑starting next run
    globals()["height_sequence_1"] = final_seq
    return final_seq
