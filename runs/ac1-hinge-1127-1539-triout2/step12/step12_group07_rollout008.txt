"""
Hybrid optimizer with extended large‑scale phase.

Main ideas
----------
1. **Exploration** – many short annealed‑gradient runs on a modest
   sequence length (≈5–8 k).  The pool of seeds contains power‑law,
   exponential, log‑normal and random simplex points.
2. **Upscaling** – the best small solution is interpolated to a larger
   length (12–24 k) and projected onto the ℓ¹‑simplex with the required
   sum √(2 n).  A few deterministic reorderings are tested.
3. **Massive permutation search** – thousands of random shuffles are
   evaluated (FFT based) and the best ordering is kept.
4. **Projected sub‑gradient descent** – exact max‑convolution sub‑gradients
   polish the large‑scale candidate.
5. **Swap‑annealing** – a simulated‑annealing on the permutation space
   (swap a large and a small value) which frequently improves the peak.
   The temperature decays linearly.
6. **Final trimming** – trailing near‑zero entries are removed; the
   sequence is stored globally for warm‑starting later runs.

The routine respects the time budget (`budget_s`), never exceeds it,
and always returns a non‑negative ℓ¹‑normalized sequence.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a (linear convolution) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """Objective = max convolution (the evaluate_sequence value for
    ℓ¹‑normalised vectors)."""
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:
        return np.inf
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _log_normal_sequence(n: int, sigma: float, target_sum: float,
                         rng: np.random.Generator) -> np.ndarray:
    """Log‑normal (exp of normal) distribution."""
    a = np.exp(rng.normal(scale=sigma, size=n))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float,
                    rng: np.random.Generator) -> np.ndarray:
    """Uniform point on the ℓ¹‑simplex."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Deterministic reorderings
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, …."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread large values uniformly using a step coprime to n."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Annealed gradient descent (soft‑max approximation)
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 2e6,
                               step_factor: float = 0.9,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected gradient descent on a smooth max‑approximation."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _objective(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_b = np.max(b)
        exps = np.exp(beta * (b - max_b))
        p = exps / np.sum(exps)               # soft‑max over convolution entries

        a_rev = a[::-1]
        Ar_fft = np.fft.rfft(a_rev, n=size)
        Pr_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]          # gradient w.r.t a

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:7.1f} "
                  f"score={best_val:.6f}")

        # increase temperature (beta) periodically
        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.45,
                         gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent using the true max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _objective(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        # indices attaining (almost) the maximum
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]

        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # jitter every few hundred iterations
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1
    return best_a, best_val

def _permutation_search(base_seq: np.ndarray,
                        trials: int,
                        deadline: float) -> tuple[np.ndarray, float]:
    """Search many random reorderings; keep the best."""
    best_seq = base_seq.copy()
    best_val = _max_convolution_fft(best_seq)
    rng = np.random.default_rng()

    for _ in range(trials):
        if time.time() >= deadline:
            break
        perm = rng.permutation(base_seq)
        val = _max_convolution_fft(perm)
        if val < best_val:
            best_val = val
            best_seq = perm.copy()
    return best_seq, best_val

def _swap_annealing(seq: np.ndarray,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing on the permutation space: repeatedly swap a large
    and a small entry and accept according to a linearly decaying temperature.
    """
    cur = seq.copy()
    cur_max = _max_convolution_fft(cur)
    best_seq = cur.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    n = cur.shape[0]

    # maintain two index pools (top 20 % and bottom 20 %)
    def _update_pools():
        idx = np.argsort(cur)
        top = idx[-max(1, int(0.2 * n)):]
        low = idx[:max(1, int(0.2 * n))]
        return top, low

    top_idx, low_idx = _update_pools()
    start = time.time()

    while time.time() < deadline:
        # linear temperature decay
        elapsed = time.time() - start
        total = deadline - start
        temp = max(1e-6, 0.5 * (1.0 - elapsed / total))

        i = rng.choice(top_idx)
        j = rng.choice(low_idx)
        if i == j:
            continue

        # propose swap
        cur[i], cur[j] = cur[j], cur[i]
        new_max = _max_convolution_fft(cur)

        if new_max < cur_max:
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_seq = cur.copy()
                print(f"[{time.time():.2f}] SwapAnneal new best={best_max:.6f}")
        else:
            # Metropolis rule
            if rng.random() >= math.exp(-(new_max - cur_max) / (temp + 1e-12)):
                # reject → revert
                cur[i], cur[j] = cur[j], cur[i]
            else:
                cur_max = new_max

        # refresh pools every few seconds to adapt to the evolving distribution
        if (elapsed % 5.0) < 0.1:
            top_idx, low_idx = _update_pools()

    return best_seq, best_max

    seq_big = _resize_sequence(best_seq, n_big)
    seq_big = _project_to_simplex(seq_big, target_big)

    # try a few deterministic reorderings and keep the best
    candidates = [
        seq_big,
        _alternating_order(seq_big),
        _even_spread_order(seq_big)
    ]
    seq_big = min(candidates, key=_max_convolution_fft)
    print(f"[{time.time():.2f}] Upscaled to {n_big} elems, start max="
          f"{_max_convolution_fft(seq_big):.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – massive random permutation search
    # ------------------------------------------------------------------
    if time.time() < deadline - 15.0:
        perm_deadline = time.time() + min(12.0, deadline - time.time() - 8.0)
        # number of trials roughly proportional to remaining time
        trials = int(2000 * (perm_deadline - time.time()))
        seq_big, _ = _permutation_search(seq_big, trials, perm_deadline)
        print(f"[{time.time():.2f}] After random perms max="
              f"{_max_convolution_fft(seq_big):.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – sub‑gradient descent on the large scale
    # ------------------------------------------------------------------
    if time.time() < deadline - 12.0:
        sub_deadline = time.time() + min(10.0, deadline - time.time() - 8.0)
        a_sub, val_sub = _subgradient_descent(
            seq_big,
            sub_deadline,
            target_big,
            step_factor=0.45,
            gamma=0.996,
        )
        if val_sub < best_val:
            best_val = val_sub
            seq_big = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑grad polished best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – swap‑annealing (permutation simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sa_deadline = time.time() + min(8.0, deadline - time.time() - 3.0)
        seq_big, best_max = _swap_annealing(seq_big, sa_deadline)
        if best_max < best_val:
            best_val = best_max
            print(f"[{time.time():.2f}] Swap‑anneal best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Final trimming of trailing near‑zero entries
    # ------------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Logging of the final evaluation (optional)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_score:.6f}")
    except Exception:
        pass

    # expose for warm‑starting subsequent runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
