"""
Improved step‑function optimizer for the benchmark.

Main ideas
----------

1. **Parametric warm‑start** – scan a few classic families (power‑law,
   exponential‑power‑law, broken power‑law) to obtain a good initial shape.

2. **Projected Adam on a soft‑max approximation** – several short Adam runs
   with a slowly increasing temperature β to explore the landscape while
   respecting the simplex constraint Σa = √(2 n).

3. **Guided exact mass‑transfer** – the classic “move mass from high‑gradient
   to low‑gradient positions” step, but with more frequent jitter to avoid
   plateaus.

4. **Incremental greedy hill‑climb** – after the previous phases we keep the
   convolution array up‑to‑date with O(n) slice updates and repeatedly move
   mass from the index that contributes most to the current peak towards the
   index that contributes least.  The step size is computed analytically so
   that the current maximal convolution entry is driven down to the second
   largest one.  Small random jitter and periodic full FFT recomputation keep
   the state accurate.

5. **Final high‑β Adam polish** – a short high‑temperature Adam run for fine
   tuning.

All phases respect the simplex constraints, use only one CPU and stop
automatically when the supplied budget (≈1000 s) expires.  The routine
returns the best sequence found (trailing near‑zero entries are trimmed).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (linear convolution)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) using a real FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft_pair(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution x*y (full length 2·n‑1) using a real FFT."""
    n = x.size
    size = 1 << ((2 * n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Benchmark score.
    For Σa = √(2 n) this equals the maximal entry of the convolution.
    """
    n = a.shape[0]
    max_b = _max_conv(a)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for fast pre‑search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _broken_power_law_sequence(
    n: int, p1: float, p2: float, alpha: float, target_sum: float
) -> np.ndarray:
    K = max(1, int(alpha * n))
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.empty_like(idx)
    a[:K] = idx[:K] ** (-p1)
    a[K:] = idx[K:] ** (-p2)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimizer on a soft‑max approximation of the max convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(a*a)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = conv_full.max()
    exps = np.exp(beta * (conv_full - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, conv_full


def _adam_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 5e5,
    step_factor: float = 0.8,
    gamma: float = 0.999,
) -> tuple[np.ndarray, float]:
    """Projected Adam on the smooth soft‑max of the max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Guided exact mass‑transfer (original version, with more jitter)
# ----------------------------------------------------------------------


def _compute_gradient_for_max(a: np.ndarray, max_idx: np.ndarray) -> np.ndarray:
    """Exact sub‑gradient of max(conv) w.r.t. a (averaged over max indices)."""
    n = a.size
    g = np.zeros_like(a)
    for k in max_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g[idx] += 2.0 * a[k - idx]
    if max_idx.size > 0:
        g /= max_idx.size
    return g


def _guided_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Classic greedy exact reduction of the maximal convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        mask_max = conv >= max_val - tol
        second_max = conv[~mask_max].max() if np.any(~mask_max) else 0.0
        max_idx = np.where(mask_max)[0]

        g = _compute_gradient_for_max(a, max_idx)

        sum_g = g.sum()
        d = -g + sum_g / n * np.ones_like(a)  # Σd = 0

        # keep non‑negativity
        neg = d < 0
        if np.any(neg):
            t_nonneg = np.min(a[neg] / -d[neg])
        else:
            t_nonneg = np.inf

        delta = 2.0 * _conv_fft_pair(a, d)
        delta_max = delta[max_idx]
        delta_k = delta_max.min()
        if delta_k >= 0:
            break

        t_to_second = (max_val - second_max) / (-delta_k)
        t = min(t_nonneg, t_to_second) * 0.99
        if t <= 0:
            break

        a += t * d
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] GuidedMT it={it} score={best_score:.6f}")

        # more aggressive jitter to escape stagnation
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0015, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Incremental greedy hill‑climb (new phase)
# ----------------------------------------------------------------------


def _incremental_hill_climb(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Fast O(n) hill‑climb that repeatedly moves mass from the index that
    contributes most to the current peak towards the index that contributes
    least.  The maximal convolution entry is reduced analytically to the
    second‑largest value while respecting non‑negativity.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_fft(a)                     # full convolution, length 2n‑1
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    it = 0
    recompute_every = 3000      # full FFT to eliminate drift
    jitter_every = 2000

    while time.time() < deadline:
        # locate the (first) maximal index
        k_max = int(conv.argmax())
        max_val = conv[k_max]

        # compute second largest value (ignore all entries that equal max)
        mask = conv >= max_val - 1e-12
        if np.any(~mask):
            second_max = conv[~mask].max()
        else:
            second_max = 0.0

        # gradient of this convolution entry w.r.t. a
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx_range = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(a)
        g[idx_range] = 2.0 * a[k_max - idx_range]

        # donor = index with largest gradient among entries having some mass
        donor_candidates = np.where(a > 1e-14)[0]
        if donor_candidates.size == 0:
            break
        donor = donor_candidates[g[donor_candidates].argmax()]

        # receiver = index with smallest gradient (could be zero)
        receiver = int(g.argmin())

        if donor == receiver:
            # fall back to a random move
            donor = int(rng.integers(n))
            receiver = int(rng.integers(n))
            if donor == receiver:
                continue

        d_val = g[donor] - g[receiver]
        if d_val <= 1e-14:
            # no progress possible with this pair – add a tiny jitter and continue
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_max = conv.max()
            continue

        # step needed to bring the current peak down to second_max
        step = (max_val - second_max) / d_val
        step = min(step, a[donor] * 0.999)      # keep donor non‑negative

        if step <= 0:
            # cannot move any mass – break
            break

        # store a copy for the incremental update
        a_old = a.copy()

        # apply the mass transfer
        a[donor] -= step
        a[receiver] += step

        # update the convolution incrementally (exact linear part)
        conv[donor : donor + n] -= 2.0 * step * a_old
        conv[receiver : receiver + n] += 2.0 * step * a_old

        # add the quadratic δ² term (exact)
        idx_dd = 2 * donor
        idx_rr = 2 * receiver
        idx_dr = donor + receiver
        if idx_dd < conv.size:
            conv[idx_dd] += step * step
        if idx_rr < conv.size:
            conv[idx_rr] += step * step
        if idx_dr < conv.size:
            conv[idx_dr] -= 2.0 * step * step

        # clean tiny negative values that may appear from rounding
        conv = np.maximum(conv, 0.0)

        cur_max = conv.max()
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] HillClimb it={it} max={best_max:.6f}")

        it += 1

        # periodic full recomputation to avoid accumulated numerical error
        if it % recompute_every == 0:
            conv = _conv_fft(a)
            cur_max = conv.max()

        # occasional jitter to shake out of plateaus
        if it % jitter_every == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_max = conv.max()

    return best_a, best_max


# ----------------------------------------------------------------------
# Light global random exploration (kept from the original code)
# ----------------------------------------------------------------------


def _global_random_move(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
) -> tuple[np.ndarray, float]:
    """Very lightweight simulated‑annealing‑style mass moves."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    while time.time() < deadline - 0.5:
        top_k = max(1, int(0.02 * n))
        low_k = max(1, int(0.02 * n))
        donor = np.argsort(-a)[rng.integers(top_k)]
        receiver = np.argsort(a)[rng.integers(low_k)]
        if donor == receiver:
            continue
        delta = rng.random() * a[donor] * 0.05
        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] RandomMove new best={best_score:.6f}")

    return best_a, best_score

# Main entry point required by the competition harness

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal benchmark score.
    The routine respects the supplied time budget and returns the best
    sequence found (trailing near‑zero entries are trimmed).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ---------------------------------------------------------------
    # Problem size – larger n gives a finer discretisation.
    # ---------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = math.sqrt(2.0 * n)  # Σa is fixed to this value (scale‑invariant)

    # ---------------------------------------------------------------
    # Phase 0 – dense parametric pre‑search (coarse but fast)
    # ---------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    rng = np.random.default_rng()

    # (a) pure power‑law
    for p in np.arange(0.20, 1.21, 0.01):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # (b) exponential‑power‑law
    for p in np.arange(0.20, 1.21, 0.02):
        for q in np.arange(0.0, 8.001, 0.25):
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # (c) broken power‑law (two‑segment)
    for p1 in np.arange(0.20, 0.81, 0.1):
        for p2 in np.arange(0.80, 2.01, 0.2):
            for alpha in np.arange(0.1, 0.91, 0.1):
                seq = _broken_power_law_sequence(
                    n, float(p1), float(p2), float(alpha), target_sum
                )
                val = _score(seq)
                if val < best_score:
                    best_score, best_seq = val, seq.copy()

    # (d) warm start from an externally supplied best if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score, best_seq = val, resized.copy()

    print(f"[{time.time():.2f}] After dense pre‑search: best score = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase A – short Adam runs (exploratory restarts)
    # ---------------------------------------------------------------
    remaining = deadline - time.time()
    adam_deadline = time.time() + 0.20 * remaining
    while time.time() < adam_deadline - 2.0:
        jitter = rng.normal(scale=target_sum * 0.015, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _adam_gradient_descent(
            start_vec, adam_deadline, target_sum, max_beta=5e5, step_factor=0.8, gamma=0.995
        )
        if val_opt < best_score:
            best_score, best_seq = val_opt, a_opt.copy()
            print(f"[{time.time():.2f}] Adam restart improved best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase B – guided exact mass‑transfer (core refinement)
    # ---------------------------------------------------------------
    while time.time() < deadline - 12.0:
        a_ref, val_ref = _guided_mass_transfer(
            best_seq, deadline, target_sum
        )
        if val_ref < best_score:
            best_score, best_seq = val_ref, a_ref.copy()
            print(f"[{time.time():.2f}] Guided mass‑transfer improved best = {best_score:.6f}")

        # occasional random moves
        a_rand, val_rand = _global_random_move(best_seq, deadline, target_sum)
        if val_rand < best_score:
            best_score, best_seq = val_rand, a_rand.copy()
            print(f"[{time.time():.2f}] Random move improved best = {best_score:.6f}")

        if time.time() > deadline - 12.0:
            break

    # ---------------------------------------------------------------
    # Phase C – incremental greedy hill‑climb (new phase)
    # ---------------------------------------------------------------
    while time.time() < deadline - 6.0:
        a_hc, score_hc = _incremental_hill_climb(best_seq, target_sum, deadline)
        if score_hc < best_score:
            best_score, best_seq = score_hc, a_hc.copy()
            print(f"[{time.time():.2f}] Hill‑climb improved best = {best_score:.6f}")

        # small random perturbation after hill‑climb to search new basins
        jitter = rng.normal(scale=target_sum * 0.001, size=n)
        best_seq = np.maximum(best_seq + jitter, 0.0)
        best_seq = _project_to_simplex(best_seq, target_sum)
        best_score = _score(best_seq)

        if time.time() > deadline - 6.0:
            break

    # ---------------------------------------------------------------
    # Phase D – final polishing with a high‑β Adam run
    # ---------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(
            best_seq,
            deadline,
            target_sum,
            max_beta=1e6,
            step_factor=0.6,
            gamma=0.995,
        )
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final polish improved best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return list
    # ---------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final evaluation for reporting (harmless if evaluate_sequence missing)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated benchmark score = {val:.6f}")
    except Exception:
        pass

    return final_seq
