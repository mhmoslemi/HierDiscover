"""
Hybrid optimiser for the non‑negative step‑function problem.

Main ideas
----------
1. Initialise a fairly large sequence (≈12–18 k) from several
   parametric families (power‑law, exponential, random simplex) and from the
   best solution of a previous run if it exists.
2. Run a short but aggressive annealed‑gradient optimiser on each seed.
   The gradient is computed via a soft‑max approximation of the max‑convolution.
3. Take the best seed and improve the *ordering* of its values by a long
   simulated‑annealing over swaps (permutation annealing).  The objective is
   evaluated with an FFT‑based max‑convolution, which is cheap enough for
   thousands of swaps.
4. Refine the continuous values on the final ordering with a few seconds of
   exact sub‑gradient descent.
5. Finish with a greedy mass‑transfer simulated‑annealing stage.

All stages respect non‑negativity and the ℓ¹‑simplex constraint; scaling of
the whole vector does not affect the score, so we normalise to
∑a = √(2 n) which makes the objective equal the convolution peak.
The programme runs for at most `budget_s` seconds and returns the best
sequence found.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (FFT)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (FFT)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a)/(∑a)²."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised to required ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform random point on the ℓ¹‑simplex (Dirichlet‑like)."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Annealed gradient descent (soft‑max approximation)
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.9,
                               gamma: float = 0.98) -> tuple[np.ndarray, float]:
    """Momentum projected GD on the soft‑max approximation of the max‑conv."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_b = np.max(b)
        exps = np.exp(beta * (b - max_b))
        p = exps / np.sum(exps)

        a_rev = a[::-1]
        Ar_fft = np.fft.rfft(a_rev, n=size)
        Pr_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:7.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Exact projected sub‑gradient descent on the true max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]

        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy mass‑transfer with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.5,
                                      end_temp: float = 1e-4,
                                      low_frac: float = 0.06) -> tuple[np.ndarray, float]:
    """
    Move a little mass from a “donor’’ (large contribution to the current peak)
    to a “receiver’’ (small contribution) with Metropolis acceptance.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    conv = _conv_fft(a)          # current convolution

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if rng.random() < 0.001:      # occasional global jitter
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Permutation annealing (swap‑based simulated annealing)
# ----------------------------------------------------------------------


def _permutation_annealing(seq: np.ndarray,
                           deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated–annealing over swaps of the entries of ``seq``.
    The values themselves are not changed, only their order.
    """
    n = seq.shape[0]
    cur = seq.copy()
    total = cur.sum()
    factor = 2.0 * n / (total * total)      # constant factor for the score
    cur_max = _max_convolution_fft(cur)
    cur_score = factor * cur_max
    best = cur.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_T = max(1e-6, cur_score * 0.2)
    end_T = 1e-8
    start_time = time.time()
    it = 0

    while time.time() < deadline:
        # temperature schedule (exponential)
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        T = start_T * (end_T / start_T) ** tau if start_T > 0 else 0.0

        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # propose swap
        cur[i], cur[j] = cur[j], cur[i]
        new_max = _max_convolution_fft(cur)
        new_score = factor * new_max

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if T > 0:
                if rng.random() < math.exp(-(new_score - cur_score) / T):
                    accept = True

        if accept:
            cur_max = new_max
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            # revert swap
            cur[i], cur[j] = cur[j], cur[i]

        it += 1
        if it % 5000 == 0:
            print(f"[{time.time():.2f}] PermAnneal it={it:7d} best={best_score:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine follows 5 phases and respects the `budget_s` second limit.
    Returns a plain list of floats.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # small safety margin
    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Choose a problem size (larger size gives finer granularity but costs more)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n_final = 18000
    elif budget_s >= 500:
        n_final = 15000
    elif budget_s >= 300:
        n_final = 12000
    else:
        n_final = 8000
    target_sum = math.sqrt(2.0 * n_final)

    # ------------------------------------------------------------------
    # Build a diverse set of initial seeds
    # ------------------------------------------------------------------
    seeds: List[np.ndarray] = []

    # parametric families
    for p in np.linspace(0.2, 2.0, 8):
        seeds.append(_power_law_sequence(n_final, float(p), target_sum))
    for lam in np.linspace(0.001, 0.025, 8):
        seeds.append(_exponential_sequence(n_final, float(lam), target_sum))

    # random points on the simplex
    for _ in range(8):
        seeds.append(_random_simplex(n_final, target_sum, rng))

    # previous best (if any)
    if globals().get("height_sequence_1"):
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        seeds.append(_resize_sequence(prev, n_final))

    # a few jittered copies to increase diversity
    for seq in list(seeds):
        jitter = rng.normal(scale=target_sum * 0.02, size=n_final)
        seq_j = np.maximum(seq + jitter, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        seeds.append(seq_j)

    # ------------------------------------------------------------------
    # Phase 1 – short gradient optimisation on each seed
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.2 * budget_s
    best_seq = None
    best_val = np.inf

    for seed in seeds:
        if time.time() > phase1_deadline:
            break
        run_dead = min(phase1_deadline, time.time() + 7.0)   # ~7 s per seed
        a_opt, val_opt = _annealed_gradient_descent(
            seed, run_dead, target_sum,
            max_beta=8e5, step_factor=0.9, gamma=0.98)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Phase1 new best={best_val:.6f}")

    # fallback if nothing succeeded
    if best_seq is None:
        best_seq = _random_simplex(n_final, target_sum, rng)

    # ------------------------------------------------------------------
    # Phase 2 – permutation annealing (swap‑based SA)
    # ------------------------------------------------------------------
    perm_deadline = deadline - 30.0      # reserve ~30 s for later refinement
    best_seq, best_val = _permutation_annealing(best_seq, perm_deadline)
    print(f"[{time.time():.2f}] After permutation SA score={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exact sub‑gradient refinement (few seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        sub_deadline = deadline - 10.0
        a_sub, val_sub = _subgradient_descent(
            best_seq, sub_deadline, target_sum,
            step_factor=0.45, gamma=0.995)
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑grad refined best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final greedy mass‑transfer SA (if time left)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        mt_deadline = deadline - 2.0
        a_mt, val_mt = _mass_transfer_simulated_annealing(
            best_seq, mt_deadline, target_sum,
            start_temp=0.25, end_temp=1e-5, low_frac=0.05)
        if val_mt < best_val:
            best_val = val_mt
            best_seq = a_mt.copy()
            print(f"[{time.time():.2f}] Final MT best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries and return
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional logging of the official evaluator
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    globals()["height_sequence_1"] = final_seq
    return final_seq
