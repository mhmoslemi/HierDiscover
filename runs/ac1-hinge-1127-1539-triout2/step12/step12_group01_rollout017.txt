    if full.shape[0] > n:

    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()

    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))
        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi+1, dtype=np.int64)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]
        # step size schedule relative to remaining time
        elapsed = time.time() - start
        total = deadline - start
        # schedule from eta_start to eta_end
        eta_start, eta_end = 2.0, 0.2
        eta = eta_start * (1 - elapsed/total) + eta_end * (elapsed/total)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_val:

            best_a = a.copy()

"""
Hybrid optimizer for the non‑negative step‑function benchmark.

The routine works in three stages within the given time budget:

1️⃣ **Fast parametric sweep** – a cheap grid / random search over a few
   families of analytically‑defined shapes (power‑law, exponential,
   combined, symmetric and triangular).  The best shape from this sweep
   supplies a high‑quality warm start.

2️⃣ **Continuous refinement** – a short mirror‑descent on a soft‑max
   surrogate of the max‑convolution followed by a true sub‑gradient
   descent.  Both updates keep the L¹‑norm fixed (by simple rescaling).

3️⃣ **Long‑run annealing** – a Metropolis mass‑transfer annealer that
   moves small amounts of height from indices that largely contribute
   to the current peak towards low‑height indices.  The temperature
   decays with a square‑root schedule and the allowed transfer size is
   large at high temperature to enable rapid exploration.

All steps respect the global 1000 s budget; the best sequence found is
saved in the global variable `height_sequence_1` for warm‑starts in
future runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Objective  2·n·max(a*a) / (sum(a)²).
    The value is invariant under scaling of ``a`` (except for clipping).
    """
    n = a.shape[0]
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Soft‑max surrogate (used for mirror descent)
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max‑convolution.
    Returns (grad, full_convolution) where grad has the same length as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    # soft‑max over the 2n‑1 convolution entries
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()                     # probability vector

    # grad = 2 * (a reversed) * p   (convolution)
    a_rev = a[::-1]
    grad_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(grad_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – exponential (mirror) descent on the soft‑max surrogate
# ----------------------------------------------------------------------


def exponential_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Mirror descent on the soft‑max surrogate."""
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        # occasional jitter to avoid stagnation
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 8e-4, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpDescent it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Sub‑gradient descent on the true max‑convolution entry."""
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        # linearly decreasing step size
        elapsed = time.time() - start
        total = deadline - start
        eta_start, eta_end = 2.0, 0.2
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase D – improved simulated‑annealing mass‑transfer refinement
# ----------------------------------------------------------------------


def anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Annealing that transfers tiny mass from “bad’’ (large contributors to
    the current peak) to “good’’ (low heights).  The temperature decays
    with a √‑schedule and the allowed transfer size shrinks with it.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.85                     # start temperature ~85 % of current score

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        # current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – favour low heights
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # size of the transfer (larger when temperature is high)
        # early phase allows up to 20 % of donor mass, later shrinks
        donor_frac = 0.20 if temperature > T0 * 0.5 else 0.05
        max_delta = min(a[donor] * donor_frac, target_sum * 0.01)
        if max_delta <= 0.0:
            continue

        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # ---- O(n) update of the convolution (exact) ----
        b_new = b.copy()
        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic self‑terms
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        # Metropolis acceptance based on *current* energy
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved best={best_val:.6f}")

        # periodic full recomputation to avoid numerical drift
        if it % 5000 == 0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Shape generators for the fast parametric sweep
# ----------------------------------------------------------------------


def shape_power(n: int, p: float, offset: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    return (idx + offset) ** (-p)


def shape_exp(n: int, r: float) -> np.ndarray:
    return np.exp(-r * np.arange(n, dtype=np.float64))


def shape_comb(n: int, p: float, r: float, offset: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    return (idx + offset) ** (-p) * np.exp(-r * idx)


def shape_sym(n: int, p: float, r: float, offset: float) -> np.ndarray:
    """Symmetric (mirror) version of the combined shape."""
    if n % 2 == 0:
        half_len = n // 2
        idx = np.arange(half_len, dtype=np.float64)
        half = (idx + offset) ** (-p) * np.exp(-r * idx)
        return np.concatenate([half, half[::-1]])
    else:
        half_len = n // 2 + 1
        idx = np.arange(half_len, dtype=np.float64)
        half = (idx + offset) ** (-p) * np.exp(-r * idx)
        return np.concatenate([half, half[-2::-1]])


def shape_triangular(n: int, width: float) -> np.ndarray:
    """Symmetric triangular bump of half‑width ``width``."""
    center = (n - 1) / 2.0
    idx = np.arange(n, dtype=np.float64)
    a = np.maximum(0.0, (width - np.abs(idx - center)) / width)
    return a
