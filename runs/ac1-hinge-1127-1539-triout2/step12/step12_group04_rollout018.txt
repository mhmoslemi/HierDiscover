    a = a0.copy()
    n = a.shape[0]
    best_score = _score(a)
    best_a = a.copy()
    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k = int(np.argmax(conv))
        # compute second max
        conv2 = conv.copy()
        conv2[k] = -np.inf
        second_max = conv2.max()
        if max_val - second_max < 1e-12:
            break
        # donors
        K = max(1, int(K_frac * n))
        donor_idx = np.argsort(-a)[:K]
        # compute partner values
        # partner for each donor: if 0 <= k - i < n else 0
        partners = np.zeros(K, dtype=np.float64)
        valid = (k - donor_idx >= 0) & (k - donor_idx < n)
        partners[valid] = a[k - donor_idx[valid]]
        # consider only donors with positive partner
        mask = (partners > 0)
        if not np.any(mask):
            # cannot reduce peak via donors with partner value
            break
        donor_idx = donor_idx[mask]
        partners = partners[mask]
        # compute alpha_max
        alpha_max = np.min(a[donor_idx] / (2.0 * partners))
        # compute sum of squares of partners
        sum_c2 = np.sum(partners ** 2)
        if sum_c2 == 0:
            break
        alpha_target = (max_val - second_max) / (4.0 * sum_c2)
        alpha = min(alpha_max * 0.99, alpha_target * 0.99)
        if alpha <= 0:
            break
        delta_i = 2.0 * alpha * partners
        a[donor_idx] -= delta_i
        total_removed = delta_i.sum()
        # receiver: choose index with minimum partner value (i.e., minimal a_{k - i})
        # For all i, compute partner_all = a[k - i] if valid else 0
        # Actually we compute using padded array

        partner_all = np.zeros(n, dtype=np.float64)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx_range = np.arange(lo, hi + 1, dtype=np.int64)
        partner_all[idx_range] = a[k - idx_range]
        # choose receiver index with minimal partner among those not donors
        # Excluding donors:
        mask_recv = np.ones(n, dtype=bool)
        mask_recv[donor_idx] = False
        # also ignore cases where moving mass might increase max beyond new max? We'll just pick minimal.
        recv_idx = np.argmin(partner_all * mask_recv)  # but this multiplication convert bool to 0/1; better to set large value for excluded.
        # Simpler: set large for excluded
        partner_all_masked = partner_all.copy()

        recv_idx = np.argmin(partner_all_masked)

        a = _project_to_simplex(a, target_sum)

            best_score = cur_score
            best_a = a.copy()

"""
Propose a step‑function with a low maximal self‑convolution.

Algorithm
---------

* Choose a problem size (n) based on the time budget.
* Initialise the sequence with a dense scan of power‑law / exponentially‑tilted
  power‑law families and keep the best.
* Refine aggressively with several optimisation phases:
    1. Projected Adam‑style gradient descent on a soft‑max approximation of the
       max‑convolution.
    2. Sub‑gradient descent that directly attacks the current peak of the
       convolution.
    3. Exact deterministic mass‑transfer (DMT2) as in the previous state‑of‑the‑art.
    4. Pairwise greedy mass‑transfer (search over a few donors/receivers).
    5. Multi‑donor/receiver transfer that moves mass from many large coefficients
       to a single small one while guaranteeing that the peak strictly drops.
    6. Final polishing with a larger soft‑max temperature.
* Trim trailing near‑zero entries and return the list.

The whole routine respects the supplied time budget (default 1000 s) and
periodically prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) via FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Objective value required by evaluate_sequence.
    With Σa = √(2n) the denominator equals 2·n, therefore the score is exactly
    max(a*a).  The function is robust for any a (non‑negative, not too small).
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex Σx = s,  x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for the fast pre‑search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(
    n: int, p: float, q: float, target_sum: float
) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam‑GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(a*a)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 2e5,
    step_factor: float = 0.8,
    gamma: float = 0.999,
) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max approximation."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}"
            )

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(
    init: np.ndarray, target_sum: float, deadline: float
) -> tuple[np.ndarray, float]:
    """Restarted Adam‑GD with short jittered restarts."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(
            start,
            time.time() + run_budget,
            target_sum,
            max_beta=2e5,
            step_factor=0.8,
            gamma=0.995,
        )
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}"
            )

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (the previous DMT2)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Greedy local mass‑transfer that strictly lowers the maximal entry."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest value (ignore current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]  # pick one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]  # a_{k-i}

        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # pick a receiver with minimal partner value (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.7, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}"
            )

        it += 1

    return best_a, best_score

# Pairwise optimal mass‑transfer (new refinement)
# ----------------------------------------------------------------------


def _pairwise_optimal_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    donor_frac: float = 0.015,
    receiver_frac: float = 0.015,
    min_improve: float = 1e-7,
) -> tuple[np.ndarray, float]:
    """Greedy pairwise moves that strictly lower the peak."""
    a = a0.copy()
    n = a.shape[0]
    best_score = _score(a)

    pad_len = n - 1
    base = None

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k = int(np.argmax(conv))

        nd = max(1, int(donor_frac * n))
        nr = max(1, int(receiver_frac * n))
        donor_idx = np.argsort(-a)[: nd]
        receiver_idx = np.argsort(a)[: nr]

        # padded view used for shifted copies
        base = np.concatenate(
            [np.zeros(pad_len, dtype=np.float64), a, np.zeros(pad_len, dtype=np.float64)]
        )

        shifted = {}
        for i in np.unique(np.concatenate((donor_idx, receiver_idx))):
            shifted[i] = base[i : i + 2 * n - 1]  # view, no copy

        best_new_max = max_val
        best_pair = None
        best_delta = 0.0

        for i in donor_idx:
            if a[i] <= 0.0:
                continue
            shift_i = shifted[i]
            for j in receiver_idx:
                if i == j:
                    continue
                shift_j = shifted[j]

                d = 2.0 * (shift_j - shift_i)  # effect of moving mass i → j
                d_k = d[k]
                if d_k >= 0.0:
                    continue

                mask = d > d_k
                if np.any(mask):
                    denom = d[mask] - d_k
                    numer = max_val - conv[mask]
                    feasible = numer / denom
                    delta_max = min(feasible.min(), a[i])
                else:
                    delta_max = a[i]

                if delta_max <= 1e-12:
                    continue

                new_max = max_val + d_k * delta_max
                if new_max < best_new_max - min_improve:
                    best_new_max = new_max
                    best_pair = (i, j)
                    best_delta = delta_max

        if best_pair is None:
            break

        i, j = best_pair
        a[i] -= best_delta
        a[j] += best_delta

        # re‑project periodically to keep Σa exact
        if int(time.time() * 10) % 50 == 0:
            a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            print(
                f"[{time.time():.2f}] Pairwise transfer i={i} j={j} Δ={best_delta:.3e} new_score={best_score:.6f}"
            )

    return a, best_score

def _global_random_mass_transfer(

) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random move new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent that directly attacks the current peak
# ----------------------------------------------------------------------


def _subgradient_descent(
    a0: np.ndarray, deadline: float, target_sum: float
) -> tuple[np.ndarray, float]:
    """Simple sub‑gradient step on the current maximal convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)

    step = target_sum * 0.05  # initial crude step size
    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        k = int(np.argmax(conv))
        max_val = conv[k]

        # gradient of conv[k] w.r.t a_i is 2*a_{k-i} (zero outside range)
        grad = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad[idx] = 2.0 * a[k - idx]

        # move opposite to gradient (reduce the peak)
        a -= step * grad
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] Sub‑gradient it={it:5d} step={step:.3e} score={best_val:.6f}"
            )
            # gradually increase step when improving
            step *= 1.02
        else:
            # shrink step if no improvement
            step *= 0.95
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Multi‑donor / single‑receiver transfer (more aggressive)
# ----------------------------------------------------------------------


def _multi_donor_transfer(
    a0: np.ndarray, deadline: float, target_sum: float, K_frac: float = 0.01
) -> tuple[np.ndarray, float]:
    """Move mass from many donors to a single receiver while guaranteeing a strict peak drop."""
    a = a0.copy()
    n = a.shape[0]
    best_score = _score(a)
    best_a = a.copy()

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k = int(np.argmax(conv))

        # compute second largest value
        conv2 = conv.copy()
        conv2[k] = -np.inf
        second_max = conv2.max()
        if max_val - second_max < 1e-12:
            break

        K = max(1, int(K_frac * n))
        donor_idx = np.argsort(-a)[:K]

        # partner values a_{k-i} for donors (zero if out of range)
        partner = np.zeros_like(donor_idx, dtype=np.float64)
        valid = (k - donor_idx >= 0) & (k - donor_idx < n)
        partner[valid] = a[k - donor_idx[valid]]

        # keep only donors that actually contribute to the peak
        mask = partner > 0
        if not np.any(mask):
            break
        donor_idx = donor_idx[mask]
        partner = partner[mask]

        # maximum feasible scaling α so that donors stay non‑negative
        alpha_max = np.min(a[donor_idx] / (2.0 * partner))
        sum_c2 = np.sum(partner ** 2)
        if sum_c2 == 0:
            break
        alpha_target = (max_val - second_max) / (4.0 * sum_c2)
        alpha = min(alpha_max * 0.99, alpha_target * 0.99)
        if alpha <= 0:
            break

        delta = 2.0 * alpha * partner
        a[donor_idx] -= delta
        total_removed = delta.sum()

        # choose a receiver that has the smallest partner value (i.e. minimal a_{k-?})
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx_range = np.arange(lo, hi + 1, dtype=np.int64)
        partner_all = np.empty(n, dtype=np.float64)
        partner_all.fill(np.inf)
        partner_all[idx_range] = a[k - idx_range]
        # exclude donors from being chosen as receiver
        partner_all[donor_idx] = np.inf
        recv = int(np.argmin(partner_all))
        a[recv] += total_removed

        a = _project_to_simplex(a, target_sum)
        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] Multi‑donor step improved score={best_score:.6f}"
            )

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    max self‑convolution (equivalently the score returned by evaluate_sequence).
    The routine respects the given time budget and returns the best sequence found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ----- problem size -------------------------------------------------
    if budget_s >= 800:
        n = 30000
    elif budget_s >= 500:
        n = 20000
    else:
        n = 12000

    target_sum = np.sqrt(2.0 * n)  # Σa = √(2 n)  →  score = max convolution

    # ----- Phase 0 – dense parametric pre‑search -----------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    p_grid = np.linspace(0.30, 0.85, 31)          # power‑law exponent grid
    q_grid = np.linspace(0.0, 6.0, 31)           # exponential tilt grid

    for p in p_grid:
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(
                n, float(p), float(q), target_sum
            )
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # incorporate previous construction if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score = val
            best_seq = resized.copy()

    # add a few jittered copies around the current best
    for _ in range(8):
        jitter = rng.normal(scale=target_sum * 0.005, size=n)
        seq_j = best_seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        val = _score(seq_j)
        if val < best_score:
            best_score = val
            best_seq = seq_j.copy()

    print(
        f"[{time.time():.2f}] Initial best after parametric search = {best_score:.6f}"
    )

    # ----- Phase A – Adam‑style projected gradient descent -------------
    gd_deadline = time.time() + (deadline - time.time()) * 0.25
    while time.time() < gd_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _gradient_descent_with_restarts(
            start_vec, target_sum, gd_deadline
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] GD improved best = {best_score:.6f}")

    # ----- Phase B – Sub‑gradient descent on the peak ------------------
    sub_deadline = time.time() + (deadline - time.time()) * 0.20
    if time.time() < sub_deadline:
        a_sub, val_sub = _subgradient_descent(best_seq, sub_deadline, target_sum)
        if val_sub < best_score:
            best_score = val_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑gradient improved best = {best_score:.6f}")

    # ----- Phase C – Exact deterministic mass‑transfer (core) ----------
    while time.time() < deadline - 8.0:
        a_ref, val_ref = _deterministic_mass_transfer_exact2(
            best_seq, deadline, target_sum
        )
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] Exact DMT2 improved best = {best_score:.6f}")

        # quick random escape moves
        a_rand, val_rand = _global_random_mass_transfer(
            best_seq, deadline, target_sum
        )
        if val_rand < best_score:
            best_score = val_rand
            best_seq = a_rand.copy()
            print(f"[{time.time():.2f}] Global random improved best = {best_score:.6f}")

        if time.time() > deadline - 5.0:
            break

    # ----- Phase D – Pairwise optimal mass‑transfer --------------------
    if time.time() < deadline - 4.0:
        a_pair, val_pair = _pairwise_optimal_transfer(
            best_seq,
            deadline,
            target_sum,
            donor_frac=0.015,
            receiver_frac=0.015,
        )
        if val_pair < best_score:
            best_score = val_pair
            best_seq = a_pair.copy()
            print(f"[{time.time():.2f}] Pairwise transfer improved best = {best_score:.6f}")

    # ----- Phase E – Multi‑donor transfer -------------------------------
    if time.time() < deadline - 3.0:
        a_multi, val_multi = _multi_donor_transfer(
            best_seq, deadline, target_sum, K_frac=0.01
        )
        if val_multi < best_score:
            best_score = val_multi
            best_seq = a_multi.copy()
            print(f"[{time.time():.2f}] Multi‑donor improved best = {best_score:.6f}")

    # ----- Phase F – Final polishing (high β Adam) --------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(
            best_seq,
            deadline,
            target_sum,
            max_beta=1e6,
            step_factor=0.6,
            gamma=0.995,
        )
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best = {best_score:.6f}")

    # ----- Post‑processing: trim trailing near‑zero entries ------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final check (harmless if evaluate_sequence is not defined)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
