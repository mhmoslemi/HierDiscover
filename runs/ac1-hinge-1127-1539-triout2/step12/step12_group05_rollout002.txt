"""
propose_candidate
-----------------

A multi‑stage optimiser for the step‑function benchmark.

* Stage 0 – select a problem size `n` based on the time budget.
* Stage 1 – coarse parametric sweep over several analytic families
  (pure power‑law, exponential × power‑law, broken power‑law,
  pure geometric).  The best‑scoring sequence is kept.
* Stage 2 – local stochastic refinement of the exponential‑power‑law
  parameters (p,q) and of the geometric‑ratio `r`.
* Stage 3 – a long projected Adam run on the soft‑max approximation
  of the max‑convolution (β is annealed up to 1e6).  The Adam run
  respects the simplex constraint Σa = √(2 n).
* Stage 4 – exact sub‑gradient guided mass‑transfer: at each iteration
  mass is moved from indices that increase the current peak towards
  indices that increase it the least, using a line‑search that drives
  the peak down to the second‑largest convolution entry.
* Stage 5 – a final high‑β Adam polishing step.

All stages are time‑aware; the routine stops just before the deadline
and returns the best sequence discovered (trailing near‑zero entries are
removed).  The algorithm works entirely with NumPy and the supplied
`evaluate_sequence` routine.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) using a real FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft_pair(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution x*y (full length 2·n‑1) using a real FFT."""
    n = x.size
    size = 1 << ((2 * n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Benchmark score (equal to max convolution when Σa = √(2 n))."""
    n = a.shape[0]
    max_b = _max_conv(a)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (dense pre‑search)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _broken_power_law_sequence(
    n: int, p1: float, p2: float, alpha: float, target_sum: float
) -> np.ndarray:
    K = max(1, int(alpha * n))
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.empty_like(idx)
    a[:K] = idx[:K] ** (-p1)
    a[K:] = idx[K:] ** (-p2)
    a *= target_sum / a.sum()
    return a


def _geometric_sequence(n: int, r: float, target_sum: float) -> np.ndarray:
    """Pure exponential a_i = C * r**i , 0 < r < 1."""
    if r <= 0.0 or r >= 1.0:
        # fallback to uniform if the ratio is out of bounds
        return np.full(n, target_sum / n, dtype=np.float64)
    # closed form sum of geometric series
    C = target_sum * (1 - r) / (1 - r ** n)
    i = np.arange(n, dtype=np.float64)
    a = C * (r ** i)
    return a


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max approximation)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(a*a).
    Returns (grad, full_convolution).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = conv_full.max()
    # stabilised soft‑max
    exps = np.exp(beta * (conv_full - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, conv_full


def _adam_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 5e5,
    step_factor: float = 0.8,
    gamma: float = 0.999,
) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max approximation of the max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient guided mass‑transfer (core refinement)
# ----------------------------------------------------------------------


def _compute_gradient_for_max(a: np.ndarray, max_idx: np.ndarray) -> np.ndarray:
    """Exact sub‑gradient of max(conv) w.r.t. a (averaged over max indices)."""
    n = a.size
    g = np.zeros_like(a)
    for k in max_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        # contribution 2 * a[k - idx]
        g[idx] += 2.0 * a[k - idx]
    if max_idx.size > 0:
        g /= max_idx.size
    return g


def _guided_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Greedy exact reduction of the maximal convolution entry.
    Moves mass from indices that increase the peak to indices that increase it the least.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()

        # second largest (ignoring entries that are (near) maximal)
        mask_max = conv >= max_val - tol
        second_max = conv[~mask_max].max() if np.any(~mask_max) else 0.0
        max_idx = np.where(mask_max)[0]

        # exact sub‑gradient averaged over all maximal entries
        g = _compute_gradient_for_max(a, max_idx)

        # direction with zero sum
        sum_g = g.sum()
        d = -g + sum_g / n * np.ones_like(a)  # Σd = 0

        # largest admissible step before any component turns negative
        neg = d < 0
        if np.any(neg):
            t_nonneg = np.min(a[neg] / -d[neg])
        else:
            t_nonneg = np.inf

        # linearised change of the convolution in direction d
        delta = 2.0 * _conv_fft_pair(a, d)
        delta_max = delta[max_idx]
        delta_k = delta_max.min()  # most favourable maximal index

        if delta_k >= 0:
            # no decrease possible in this direction
            break

        # step needed to bring the current max down to the second max
        t_to_second = (max_val - second_max) / (-delta_k)

        # safety factor
        t = min(t_nonneg, t_to_second) * 0.99
        if t <= 0:
            break

        a += t * d
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] GuidedMT it={it} score={best_score:.6f}")

        # occasional jitter to hop out of small plateaus
        if it % 500 == 0 and rng.random() < 0.05:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Very cheap random “shake” move
# ----------------------------------------------------------------------


def _global_random_move(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
) -> tuple[np.ndarray, float]:
    """Light simulated‑annealing style mass redistribution."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    while time.time() < deadline - 0.5:
        top_k = max(1, int(0.02 * n))
        low_k = max(1, int(0.02 * n))
        donor = np.argsort(-a)[rng.integers(top_k)]
        receiver = np.argsort(a)[rng.integers(low_k)]
        if donor == receiver:
            continue
        delta = rng.random() * a[donor] * 0.05
        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] RandomMove new best={best_score:.6f}")

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising the benchmark score.
    The routine respects the supplied time budget and returns the best sequence
    found (trailing near‑zero entries are trimmed).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Problem size selection (larger n offers a finer discretisation)
    # ------------------------------------------------------------------
    if budget_s >= 900:
        n = 20000
    elif budget_s >= 700:
        n = 16000
    elif budget_s >= 500:
        n = 12000
    elif budget_s >= 300:
        n = 8000
    else:
        n = 5000

    target_sum = math.sqrt(2.0 * n)  # Σa is fixed to this value

    # ------------------------------------------------------------------
    # Warm start from a previously discovered solution (if any)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    rng = np.random.default_rng()

    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            # linear interpolation onto the new grid
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        best_seq, best_score = resized, val
        print(f"[{time.time():.2f}] Warm‑start from previous best, score={val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – dense parametric pre‑search (coarse but fast)
    # ------------------------------------------------------------------
    # (a) pure power‑law
    for p in np.linspace(0.20, 1.20, 41):  # step 0.025
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # (b) exponential‑power‑law
    for p in np.linspace(0.20, 1.20, 41):
        for q in np.linspace(0.0, 8.0, 33):  # step 0.25
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # (c) broken power‑law (coarse grid)
    for p1 in np.linspace(0.20, 0.80, 7):
        for p2 in np.linspace(0.80, 2.00, 7):
            for alpha in np.linspace(0.1, 0.9, 9):
                seq = _broken_power_law_sequence(
                    n, float(p1), float(p2), float(alpha), target_sum
                )
                val = _score(seq)
                if val < best_score:
                    best_score, best_seq = val, seq.copy()

    # (d) pure geometric family
    for r in np.linspace(0.90, 0.999, 56):  # step ≈0.002
        seq = _geometric_sequence(n, float(r), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    print(f"[{time.time():.2f}] After coarse families: best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – stochastic local refinement of parameters (p,q) & r
    # ------------------------------------------------------------------
    # Limit this stage to roughly 10 % of the remaining budget
    remaining = deadline - time.time()
    param_deadline = time.time() + 0.10 * remaining

    # keep track of the best parameters we have seen
    best_p, best_q = None, None
    best_r = None

    # initialise from the best sequence we already have
    # Try to infer which family it came from by fitting a simple model:
    # we just store its parameters if it matches one of our families.
    # (Not essential – the loops below will explore anyway.)
    while time.time() < param_deadline:
        # ---- refine (p,q) of exponential‑power‑law -------------------
        # start from current best values (if not set pick random)
        if best_p is None:
            best_p = rng.uniform(0.2, 1.2)
        if best_q is None:
            best_q = rng.uniform(0.0, 8.0)

        sigma_p = max(0.02, 0.2 / (1 + (time.time() - param_deadline) ** 0.5))
        sigma_q = max(0.05, 0.5 / (1 + (time.time() - param_deadline) ** 0.5))

        p_candidate = np.clip(best_p + rng.normal(scale=sigma_p), 0.05, 3.0)
        q_candidate = np.clip(best_q + rng.normal(scale=sigma_q), 0.0, 15.0)

        seq = _exp_power_law_sequence(n, float(p_candidate), float(q_candidate), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()
            best_p, best_q = p_candidate, q_candidate
            print(f"[{time.time():.2f}] Refined (p,q) -> ({best_p:.4f},{best_q:.4f}) score={best_score:.6f}")

        # ---- refine geometric ratio ---------------------------------
        if best_r is None:
            best_r = rng.uniform(0.90, 0.999)

        sigma_r = max(0.001, 0.01 / (1 + (time.time() - param_deadline) ** 0.5))
        r_candidate = np.clip(best_r + rng.normal(scale=sigma_r), 0.80, 0.9999)

        seq = _geometric_sequence(n, float(r_candidate), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()
            best_r = r_candidate
            print(f"[{time.time():.2f}] Refined r={best_r:.6f} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – projected Adam (soft‑max) runs (exploratory restarts)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    # allocate about 30 % of the remaining time to Adam
    adam_deadline = time.time() + 0.30 * remaining
    while time.time() < adam_deadline - 2.0:
        jitter = rng.normal(scale=target_sum * 0.02, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _adam_gradient_descent(
            start_vec,
            adam_deadline,
            target_sum,
            max_beta=1e6,
            step_factor=0.8,
            gamma=0.995,
        )
        if val_opt < best_score:
            best_score, best_seq = val_opt, a_opt.copy()
            print(f"[{time.time():.2f}] Adam restart improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – guided exact mass‑transfer (core refinement)
    # ------------------------------------------------------------------
    while time.time() < deadline - 8.0:
        a_ref, val_ref = _guided_mass_transfer(
            best_seq, deadline, target_sum
        )
        if val_ref < best_score:
            best_score, best_seq = val_ref, a_ref.copy()
            print(f"[{time.time():.2f}] Guided mass‑transfer improved best = {best_score:.6f}")

        # occasional random shaking to escape tiny plateaus
        a_rand, val_rand = _global_random_move(best_seq, deadline, target_sum)
        if val_rand < best_score:
            best_score, best_seq = val_rand, a_rand.copy()
            print(f"[{time.time():.2f}] Random move improved best = {best_score:.6f}")

        if time.time() > deadline - 2.0:
            break

    # ------------------------------------------------------------------
    # Stage 5 – final polishing with a high‑β Adam run
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(
            best_seq,
            deadline,
            target_sum,
            max_beta=2e6,
            step_factor=0.6,
            gamma=0.995,
        )
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final polish improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final evaluation (harmless if evaluate_sequence missing)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated benchmark score = {val:.6f}")
    except Exception:
        pass

    return final_seq
