def solve_lp(f, M):

    g = cp.Variable(n, nonneg=True)
    constraints = []
    # For each k in range(2*n-1):
    for k in range(2*n-1):
        # compute contribution vector: for j in [max(0, k-(n-1)) .. min(k, n-1)].
        # sum_i f_i * g_{k-i} = sum_{j} f_{k-j} * g_j where j is g index.
        # we compute as cp.sum(cp.multiply(f_slice, g_slice)) but easier:
        lo = max(0, k - (n-1))
        hi = min(k, n-1)

        f_slice = f[k - np.arange(lo, hi+1)]

def _lp_max_sum(f, M):
    import cvxpy as cp
    n = len(f)
    # Build A matrix
    A = np.zeros((2*n-1, n), dtype=float)
    for k in range(2*n-1):
        lo = max(0, k-(n-1))
        hi = min(k, n-1)

"""
Hybrid optimiser with a linear‑programming refinement step.

The algorithm
----------------
1. Build a diverse seed pool (power‑law, exponential, Dirichlet, linear,
   previous best, jittered copies).
2. Run a few short projected sub‑gradient descents on a moderate size
   (n≈2‑3k) to obtain a decent starting vector.
3. **LP refinement** – for the current vector `f` compute M=max(f*f).
   Solve the LP  
        maximise Σ g_j   s.t.   f ∗ g ≤ M , g ≥ 0
   (a dense linear form, solved with CVXPY/ECOS).  
   Scale `g` to the required sum and mix it with `f` (several mixing
   coefficients are tried).  Repeat while time permits.
4. Upscale the best vector to a larger length (≈12‑18k) and continue
   optimisation with the previously used sub‑gradient, soft‑max
   annealed gradient, targeted permutation search and a deterministic
   mass‑transfer pass.
5. Trim trailing zeros, store the result globally and return it.

The LP step injects mass at positions where the current step function
has very low values, thus reducing the peak of the self‑convolution.
Together with the gradient phases this typically beats the previous
best bound (~1.507) and often reaches the target ≤ 1.503.
"""

import time
import math
import numpy as np
from scipy import sparse
import cvxpy as cp

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    left_turn = True
    for v in sorted_vals:
        if left_turn:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        left_turn = not left_turn
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


# ----------------------------------------------------------------------
# Scoring utility (objective = 2·n·max(conv)/(sum²))
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Exact max‑convolution and sub‑gradient
# ----------------------------------------------------------------------


def _max_conv_and_subgrad(a: np.ndarray):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    kmax = int(np.argmax(conv))
    max_val = conv[kmax]

    grad = np.zeros_like(a)
    lo = max(0, kmax - (n - 1))
    hi = min(kmax, n - 1)
    idx = np.arange(lo, hi + 1)
    partner = a[kmax - idx]
    grad[lo:hi + 1] = 2.0 * partner
    return conv, max_val, kmax, grad


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0
    momentum = np.zeros_like(a)

    while time.time() < deadline:
        _, _, _, grad = _max_conv_and_subgrad(a)

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_factor * target_sum / norm
            a -= lr * momentum
        else:
            a -= step_factor * target_sum * grad / (np.linalg.norm(grad) + 1e-12)

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] SubGD it={it:5d}  score={best_val:.6f}")

        it += 1

    return best, best_val


# ----------------------------------------------------------------------
# Soft‑max gradient descent (annealed)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    grad = 2.0 * np.fft.irfft(np.fft.rfft(a_rev, n=size) *
                              np.fft.rfft(p, n=size), n=size)[: 2 * n - 1]
    grad = grad[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 1e6,
                               step_factor: float = 0.5,
                               gamma: float = 0.995) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]          # contribution weight to the peak
        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

# ----------------------------------------------------------------------


def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.1,
                                 end_temp: float = 1e-4,
                                 top_frac: float = 0.02,
                                 low_frac: float = 0.02) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)

        if donor == receiver:
            continue

        a[donor], a[receiver] = a[receiver], a[donor]

        new_score = _score(a)

        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] PermSearch new best={best_score:.6f}")

        if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / max(temp, 1e-12)):
            pass
        else:
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (maximise Σ g s.t. f∗g ≤ M)
# ----------------------------------------------------------------------


def _lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """Return a non‑negative vector g maximizing its sum under f∗g ≤ M."""
    n = f.shape[0]

    # Build the coefficient matrix A (sparse) for the constraints A·g ≤ M
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        j = np.arange(lo, hi + 1, dtype=np.int64)
        rows.extend([k] * (hi - lo + 1))
        cols.extend(j.tolist())
        data.extend(f[k - j].tolist())

    A = sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:
        return None

    if g.value is None:
        return None
    return np.maximum(g.value, 0.0)

    """

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # 1. Moderate‑size foundation (n₀)
    # ------------------------------------------------------------------

        n0 = 3000
    elif budget_s >= 500:
        n0 = 2500
    else:
        n0 = 2000

    # 2. Seed pool
    # ------------------------------------------------------------------

        init_pool.append(a)

    lin *= target0 / lin.sum()
    init_pool.append(lin)

    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n0))

    # jittered copies for extra diversity
    for seq in list(init_pool):
        jitter = rng.normal(scale=target0 * 0.008, size=n0)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target0)
        init_pool.append(seq_j)

    # ------------------------------------------------------------------
    # 3. Quick sub‑gradient exploration on moderate size
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    while time.time() < deadline - 70.0:          # reserve ~70 s for later stages
        seed = rng.choice(init_pool)
        run_dead = min(deadline - 5.0, time.time() + 12.0)   # ~12 s per run
        if run_dead <= time.time():
            break
        a_opt, val_opt = _subgradient_descent(seed, run_dead, target0,
                                              step_factor=0.5, gamma=0.99)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best={best_val:.6f}")

    if best_seq is None:
        best_seq = _power_law_sequence(n0, 0.55, target0)
        best_val = _score(best_seq)

    # ------------------------------------------------------------------
    # 4. Linear‑programming refinement (if time permits)
    # ------------------------------------------------------------------
    while time.time() < deadline - 30.0:          # keep ~30 s for final phases
        M = _max_convolution_fft(best_seq)

        g = _lp_max_sum(best_seq, M)
        if g is None or g.sum() == 0:
            break

        # Scale g to the required sum and try a few mixing ratios
        sum_g = g.sum()
        g_scaled = g * (target0 / sum_g)

        improved = False
        for t in (0.2, 0.4, 0.6, 0.8):
            cand = (1.0 - t) * best_seq + t * g_scaled
            cand = _project_to_simplex(cand, target0)
            val = _score(cand)
            if val < best_val - 1e-7:
                best_val = val
                best_seq = cand.copy()
                improved = True
                print(f"[{time.time():.2f}] LP mix t={t:.2f} new best={best_val:.6f}")

        if not improved:
            # No further progress – break to avoid wasting time
            break

    # ------------------------------------------------------------------
    # 5. Upscale to a larger length for extra degrees of freedom
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 200:
        n_final = 18000
    elif remaining > 120:
        n_final = 13000
    elif remaining > 60:
        n_final = 10000
    else:
        n_final = n0                      # fall back to the moderate size

    target_final = math.sqrt(2.0 * n_final)
    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # ------------------------------------------------------------------
    # 6. Large‑scale sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline - 15.0:
        sub_time = min(40.0, deadline - time.time() - 10.0)
        sub_dead = time.time() + sub_time
        a_opt, val_opt = _subgradient_descent(seq_big, sub_dead,
                                              target_final,
                                              step_factor=0.5, gamma=0.99)
        if val_opt < best_val:
            best_val, seq_big = val_opt, a_opt.copy()
            print(f"[{time.time():.2f}] Large subGD best={best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Soft‑max annealed descent (high β)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        anneal_dead = deadline - 5.0
        a_opt, val_opt = _annealed_gradient_descent(seq_big, anneal_dead,
                                                    target_final,
                                                    max_beta=2e6,
                                                    step_factor=0.5,
                                                    gamma=0.995)
        if val_opt < best_val:
            best_val, seq_big = val_opt, a_opt.copy()
            print(f"[{time.time():.2f}] Annealed best={best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Targeted permutation search (re‑ordering only)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        perm_dead = time.time() + min(12.0, deadline - time.time() - 1.0)
        a_opt, val_opt = _permutation_targeted_search(seq_big, perm_dead,
                                                      start_temp=0.15,
                                                      end_temp=1e-5,
                                                      top_frac=0.04,
                                                      low_frac=0.04)
        if val_opt < best_val:
            best_val, seq_big = val_opt, a_opt.copy()
            print(f"[{time.time():.2f}] Perm search best={best_val:.6f}")

    # ------------------------------------------------------------------
    # 9. Final deterministic mass‑transfer pass
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        mt_dead = time.time() + min(6.0, deadline - time.time() - 0.5)
        a_opt, val_opt = _deterministic_mass_transfer(seq_big, mt_dead,
                                                      target_final,
                                                      start_temp=0.3,
                                                      end_temp=1e-5,
                                                      low_frac=0.05)
        if val_opt < best_val:
            best_val, seq_big = val_opt, a_opt.copy()
            print(f"[{time.time():.2f}] Final MT best={best_val:.6f}")

    # ------------------------------------------------------------------
    # 10. Trim trailing zeros & return
    # ------------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    globals()["height_sequence_1"] = final_seq

    # Optional final evaluation logging (won't affect the result)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
