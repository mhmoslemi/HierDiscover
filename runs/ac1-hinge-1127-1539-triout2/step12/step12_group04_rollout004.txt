"""
Propose a step‑function that minimises the maximum self‑convolution.
The algorithm uses three complementary ideas:

1. **Coarse optimisation on a moderate size** (n≈3000).  
   • A dense scan over power‑law families gives a good start.  
   • Projected Adam optimisation (soft‑max of the convolution) refines it.  
   • Deterministic / pairwise mass‑transfer steps lower the peak further.

2. **Linear‑programming alternation** (inspired by the “f–g” method).  
   With the current sequence `a`, the constraints `a * g ≤ M` are linear in `g`.  
   We maximise `∑ g` under those constraints, then mix `g` back into `a`.  
   Repeating a few times strictly reduces the maximal convolution value.

3. **Upsampling and final polishing** on the full target size (≈12000).  
   After the LP phase we interpolate the sequence to the final length,
   re‑project onto the simplex ( Σa = √(2 n) ) and run the previous
   gradient‑descent / mass‑transfer pipeline again.

The whole routine respects a 1000 s wall‑clock budget and prints progress.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective: 2·n·max(b)/(sum(a)^2) .  With Σa=√(2n) this equals max(b)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto the simplex Σx = s, x ≥ 0 (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (power‑law, exponentially‑tilted)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam‑GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max surrogate of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=2e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val

def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest value (ignore current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # pick one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k-i}

        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # pick a receiver with minimal partner value (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.7, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score

def _pairwise_optimal_transfer(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               donor_frac: float = 0.015,
                               receiver_frac: float = 0.015,
                               min_improve: float = 1e-7) -> tuple[np.ndarray, float]:
    """Greedy pairwise mass transfer that strictly reduces the peak."""
    a = a0.copy()
    n = a.shape[0]
    best_score = _score(a)

    pad_len = n - 1
    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k = int(np.argmax(conv))

        nd = max(1, int(donor_frac * n))
        nr = max(1, int(receiver_frac * n))
        donor_idx = np.argsort(-a)[:nd]
        receiver_idx = np.argsort(a)[:nr]

        base = np.concatenate([np.zeros(pad_len, dtype=np.float64),
                               a,
                               np.zeros(pad_len, dtype=np.float64)])

        shifted = {}
        for i in np.unique(np.concatenate((donor_idx, receiver_idx))):
            shifted[i] = base[i:i + 2 * n - 1]

        best_new_max = max_val
        best_pair = None
        best_delta = 0.0

        for i in donor_idx:
            shift_i = shifted[i]
            a_i = a[i]
            if a_i <= 0.0:
                continue
            for j in receiver_idx:
                if i == j:
                    continue
                shift_j = shifted[j]

                d = 2.0 * (shift_j - shift_i)   # effect of moving mass from i to j
                d_k = d[k]
                if d_k >= 0.0:
                    continue

                mask = d > d_k
                if np.any(mask):
                    denom = d[mask] - d_k
                    numer = max_val - conv[mask]
                    feasible = numer / denom
                    delta_max = min(feasible.min(), a_i)
                else:
                    delta_max = a_i

                if delta_max <= 1e-12:
                    continue

                new_max = max_val + d_k * delta_max
                if new_max < best_new_max - min_improve:
                    best_new_max = new_max
                    best_pair = (i, j)
                    best_delta = delta_max

        if best_pair is None:
            break

        i, j = best_pair
        a[i] -= best_delta
        a[j] += best_delta

        # re‑project occasionally to correct numeric drift
        if int(time.time() * 10) % 50 == 0:
            a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            print(f"[{time.time():.2f}] Pairwise i={i} j={j} Δ={best_delta:.3e} new_score={best_score:.6f}")

    return a, best_score

# ----------------------------------------------------------------------
# Linear‑programming refinement (f–g alternation)
# ----------------------------------------------------------------------


def _lp_alternation(a0: np.ndarray,
                    target_sum: float,
                    deadline: float,
                    max_iter: int = 5,
                    mix: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Perform a few iterations of the classic f–g alternation:
    given a, solve maximise Σg  subject to  a * g ≤ M  and g ≥ 0,
    then mix g back into a.
    """
    a = a0.copy()
    n = a.shape[0]

    for it in range(max_iter):
        if time.time() >= deadline - 4.0:
            break

        # current peak
        conv = _conv_fft(a)
        M = conv.max()

        # build linear convolution matrix (A @ g = a * g)
        # A has shape (2n‑1, n), column j contains a shifted by j.
        mrows = 2 * n - 1
        A = np.empty((mrows, n), dtype=np.float64)
        # each column j contributes a at rows j … j+n‑1
        for j in range(n):
            A[j:j + n, j] = a

        # Linear program: maximise Σg  <=> minimise -Σg
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(mrows, M, dtype=np.float64)
        bounds = [(0.0, None)] * n

        res = linprog(c,
                      A_ub=A,
                      b_ub=b_ub,
                      bounds=bounds,
                      method='highs',
                      options={'presolve': True})

        if not res.success:
            print(f"[{time.time():.2f}] LP iteration {it} failed: {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break

        # renormalise to the required total mass
        g = g * (target_sum / sum_g)

        # convex combination
        a = (1.0 - mix) * a + mix * g
        a = _project_to_simplex(a, target_sum)

        cur = _score(a)
        print(f"[{time.time():.2f}] LP iter {it} score={cur:.6f}")

    return a, _score(a)

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low maximum self‑convolution.
    The routine respects the given time budget and returns the best sequence.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a tiny safety margin

    # --------------------------------------------------------------
    # 0 – choose problem sizes (coarse then full)
    # --------------------------------------------------------------
    if budget_s >= 800:
        final_n = 12000
        base_n = 3000
    elif budget_s >= 400:
        final_n = 8000
        base_n = 2500
    else:
        final_n = 5000
        base_n = 1800

    target_sum_base = math.sqrt(2.0 * base_n)
    target_sum_final = math.sqrt(2.0 * final_n)

    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # 1 – dense parametric pre‑search on the coarse grid
    # --------------------------------------------------------------
    best_seq = _power_law_sequence(base_n, 0.5, target_sum_base)
    best_val = _score(best_seq)

    p_grid = np.linspace(0.30, 0.85, 31)
    q_grid = np.linspace(0.0, 6.0, 31)

    for p in p_grid:
        seq = _power_law_sequence(base_n, float(p), target_sum_base)
        val = _score(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()

    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(base_n, float(p), float(q), target_sum_base)
            val = _score(seq)
            if val < best_val:
                best_val, best_seq = val, seq.copy()

    # incorporate any previously discovered construction if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != base_n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, base_n)
            prev = np.interp(new_x, old_x, prev)
        prev = _project_to_simplex(prev, target_sum_base)
        val = _score(prev)
        if val < best_val:
            best_val, best_seq = val, prev.copy()

    # small jittered population around the current best
    for _ in range(8):
        jitter = rng.normal(scale=target_sum_base * 0.005, size=base_n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum_base)
        val = _score(cand)
        if val < best_val:
            best_val, best_seq = val, cand.copy()

    print(f"[{time.time():.2f}] After parametric search  score={best_val:.6f}")

    # --------------------------------------------------------------
    # 2 – Adam‑style projected gradient descent with restarts (coarse)
    # --------------------------------------------------------------
    gd_deadline = time.time() + (deadline - time.time()) * 0.25
    while time.time() < gd_deadline - 5.0:
        jitter = rng.normal(scale=target_sum_base * 0.01, size=base_n)
        start = best_seq + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum_base)

        a_opt, val_opt = _gradient_descent_with_restarts(start,
                                                         target_sum_base,
                                                         gd_deadline)
        if val_opt < best_val:
            best_val, best_seq = val_opt, a_opt.copy()
            print(f"[{time.time():.2f}] GD improved best={best_val:.6f}")

    # --------------------------------------------------------------
    # 3 – Exact deterministic mass‑transfer (coarse)
    # --------------------------------------------------------------
    while time.time() < deadline - 8.0:
        a_ref, val_ref = _deterministic_mass_transfer_exact2(best_seq,
                                                             deadline,
                                                             target_sum_base)
        if val_ref < best_val:
            best_val, best_seq = val_ref, a_ref.copy()
            print(f"[{time.time():.2f}] DMT2 improved best={best_val:.6f}")

        # a short random move to escape possible plateaus
        a_rand, val_rand = _global_random_mass_transfer(best_seq,
                                                        deadline,
                                                        target_sum_base)
        if val_rand < best_val:
            best_val, best_seq = val_rand, a_rand.copy()
            print(f"[{time.time():.2f}] Global random improved best={best_val:.6f}")

        if time.time() > deadline - 6.0:
            break

    # --------------------------------------------------------------
    # 4 – Linear‑programming alternation (core refinement) on the coarse grid
    # --------------------------------------------------------------
    lp_deadline = time.time() + (deadline - time.time()) * 0.20
    best_seq, best_val = _lp_alternation(best_seq,
                                         target_sum_base,
                                         lp_deadline,
                                         max_iter=4,
                                         mix=0.3)

    print(f"[{time.time():.2f}] After LP alternation score={best_val:.6f}")

    # --------------------------------------------------------------
    # 5 – Upsample to the final size
    # --------------------------------------------------------------
    up_seq = _upsample(best_seq, final_n)
    up_seq = _project_to_simplex(up_seq, target_sum_final)

    # --------------------------------------------------------------
    # 6 – Final polishing on the full size (gradient + transfers)
    # --------------------------------------------------------------
    # a) Adam‑style GD (short, high‑β)
    fin_deadline = time.time() + (deadline - time.time()) * 0.30
    if time.time() < fin_deadline - 5.0:
        a_gd, val_gd = _gradient_descent_with_restarts(up_seq,
                                                       target_sum_final,
                                                       fin_deadline)
        if val_gd < best_val:
            best_val, up_seq = val_gd, a_gd.copy()
            print(f"[{time.time():.2f}] Final GD improved best={best_val:.6f}")

    # b) Deterministic mass‑transfer (short)
    if time.time() < deadline - 4.0:
        a_dt, val_dt = _deterministic_mass_transfer_exact2(up_seq,
                                                           deadline,
                                                           target_sum_final)
        if val_dt < best_val:
            best_val, up_seq = val_dt, a_dt.copy()
            print(f"[{time.time():.2f}] Final DMT2 improved best={best_val:.6f}")

    # c) Pairwise optimal transfer
    if time.time() < deadline - 2.0:
        a_pt, val_pt = _pairwise_optimal_transfer(up_seq,
                                                  deadline,
                                                  target_sum_final,
                                                  donor_frac=0.02,
                                                  receiver_frac=0.02)
        if val_pt < best_val:
            best_val, up_seq = val_pt, a_pt.copy()
            print(f"[{time.time():.2f}] Final pairwise improved best={best_val:.6f}")

    # d) Very high‑β Adam polish (optional, only if time permits)
    if time.time() + 2.0 < deadline:
        a_last, val_last = _adam_gradient_descent(up_seq,
                                                  deadline,
                                                  target_sum_final,
                                                  max_beta=1e6,
                                                  step_factor=0.6,
                                                  gamma=0.995)
        if val_last < best_val:
            best_val, up_seq = val_last, a_last.copy()
            print(f"[{time.time():.2f}] Final high‑β Adam improved best={best_val:.6f}")

    # --------------------------------------------------------------
    # 7 – Trim trailing near‑zero entries and return
    # --------------------------------------------------------------
    arr = np.array(up_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
