def additive_gradient_descent(a0, target_sum, deadline):
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)
    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.5  # step size scale
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        grad, _ = _softmax_gradient(a, beta)
        eta = base_eta / (beta + 1.0) / max(1.0, np.linalg.norm(grad))
        # scale step with target sum magnitude
        step = eta * target_sum
        a = a - step * grad
        a = _project_to_simplex(a, target_sum)
        # jitter occasionally:
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum*0.001, size=a.shape)
            a = a + jitter
            a = _project_to_simplex(a, target_sum)
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AddGrad it={it} β={beta:.1f} score={best_val:.6f}")
        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)
        it += 1
    return best_a, best_val

def subgradient_descent_additive(a0, target_sum, deadline):
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    start = time.time()
    it=0
    while time.time()<deadline:
        b = _conv_fft(a)
        max_b = b.max()
        # active indices maybe utilize tolerance
        eps = max_b * 1e-6
        active = np.where(b >= max_b - eps)[0]
        n = a.shape[0]
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k-(n-1))
            hi = min(k, n-1)
            idx = np.arange(lo, hi+1)
            g[idx] += 2.0*a[k-idx]
        # compute step decreasing
        elapsed = time.time() - start
        total = deadline - start
        eta_start = 1.5
        eta_end = 0.1
        eta = eta_start * (1 - elapsed/total) + eta_end * (elapsed/total)
        a = a - eta * g
        a = _project_to_simplex(a, target_sum)
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum*0.0005, size=a.shape)
            a = a + jitter
            a = _project_to_simplex(a, target_sum)
        val = _score(a)
        if val < best_val:
            best_val=val
            best_a=a.copy()
            print(f"[{time.time():.2f}] SubGradAdd it={it} score={best_val:.6f}")
        it+=1
    return best_a,best_val

"""
Propose a sequence of non‑negative step heights that minimizes
evaluate_sequence.  The algorithm consists of several phases:

1.  Quick power‑law sweep to obtain a decent initial shape.
2.  Additive projected gradient descent on a soft‑max surrogate of the
    max‑convolution (mirrors the earlier mirror‑descent but uses an
    explicit projection onto the simplex).
3.  Additive sub‑gradient descent on the true max‑convolution (all
    active peak indices are used, not just a single one).
4.  Aggressive simulated‑annealing mass‑transfer with a reheating
    schedule and larger transfer steps.

All phases respect the global time budget.  The best sequence found is
stored in the global variable ``height_sequence_1`` for warm‑starting
future runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution / score utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Exact benchmark score: 2·n·max(a*a) / (sum(a)²)."""
    n = a.shape[0]
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max‑convolution.
    Returns (grad, full_convolution) where grad has the same length as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    grad_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(grad_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase 1 – cheap power‑law sweep (produces a good warm‑start)
# ----------------------------------------------------------------------


def _init_powerlaw(n: int, target_sum: float):
    """
    Try several exponents p and keep the best (lowest exact score).
    """
    best_a = None
    best_val = np.inf
    for p in np.linspace(0.3, 0.8, 11):
        a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
        a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – additive projected gradient descent on soft‑max surrogate
# ----------------------------------------------------------------------


def additive_gradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Gradient descent with Euclidean projection onto the simplex.
    Uses the soft‑max surrogate of the max‑convolution.
    """
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.5
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        grad, _ = _softmax_gradient(a, beta)

        # Normalise step size – we want a step of order target_sum,
        # but scaled down when the gradient is large.
        norm = np.linalg.norm(grad) + 1e-12
        step = base_eta / (beta + 1.0) / norm
        a = a - step * grad * target_sum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to break symmetry
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = a + jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AddGrad it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 3 – additive sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgradient_descent_additive(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Sub‑gradient descent where the sub‑gradient is the sum of contributions
    from *all* indices attaining (or nearly attaining) the maximal convolution entry.
    """
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        b = _conv_fft(a)
        max_b = b.max()
        # active set: all entries within 1e‑9 relative tolerance
        eps = max_b * 1e-9
        active = np.where(b >= max_b - eps)[0]

        n = a.shape[0]
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            g[idx] += 2.0 * a[k - idx]

        # linear schedule for step size
        elapsed = time.time() - start
        total = deadline - start
        eta_start = 2.0
        eta_end = 0.2
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        a = a - eta * g
        a = _project_to_simplex(a, target_sum)

        # small jitter every few hundred iterations
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = a + jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGradAdd it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 4 – aggressive simulated‑annealing mass‑transfer
# ----------------------------------------------------------------------


def anneal_mass_transfer_adv(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Long‑run annealing that transfers mass between indices.
    A reheating schedule is added and the allowed transfer size is larger.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.80          # start temperature
    last_improve = 0

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total

        # non‑linear cooling (sqrt) + occasional reheating
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))
        if it - last_improve > 20000:
            temperature = max(temperature, T0 * 0.5)   # reheat a bit

        # --- pick donor (high contribution to the current peak) ----------
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # --- pick receiver (prefer low heights) ---------------------------
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            it += 1
            continue

        # --- size of the transfer (larger than before) -------------------
        max_delta = min(a[donor] * 0.20, target_sum * 0.02)   # up to 20 % of donor or 2 % of total mass
        if max_delta <= 0:
            it += 1
            continue

        # scale transfer with temperature (keeps moves sizable early)
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * (0.1 + 0.9 * scale)   # ensure a minimal step even when cold
        if delta <= 0.0:
            it += 1
            continue

        # ---- exact O(n) update of the convolution ------------------------
        b_new = b.copy()
        # linear part
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        # ---- Metropolis acceptance ---------------------------------------
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            last_improve = it
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] AnnealAdv improved best={best_val:.6f}")

        # occasional full recomputation to avoid drift
        if it % 3000 == 0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recomputeAdv improved best={best_val:.6f}")

        it += 1

    return best_a, best_val

    # --------------------------------------------------------------
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
