import scipy.sparse as sp

def lp_refine_once(a, M, t):
    n = len(a)
    # build matrix A_ub
    i_vals = np.repeat(np.arange(n), n)
    j_vals = np.tile(np.arange(n), n)
    rows = i_vals + j_vals
    cols = i_vals
    data = a[i_vals]
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b = np.full(2*n-1, M)
    # Solve max sum g -> min -sum x
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if not res.success:
        return None, None
    g = res.x  # feasible
    sum_g = g.sum()
    if sum_g == 0:
        return a, M
    s = a.sum() / sum_g
    g_scaled = g * s
    a_new = (1 - t) * a + t * g_scaled
    # compute new M
    new_M = max_convolution(a_new)
    return a_new, new_M

def max_convolution(a):
    # compute linear convolution
    n = a.shape[0]
    size = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    return conv.max()

    a = _project_to_simplex(a, target_sum)
    best_a = a.copy()
    best_M = max_convolution(a)
    it = 0
    while it < max_iter and time.time() < deadline:

        n = len(a)

"""
Hybrid linear-programming refinement and sub‑gradient descent for the
non‑negative step‑function problem.  First a modest‑size seed is improved
by a Konyagin‑type LP iteration that mixes the current vector with a
max‑mass feasible competitor while respecting the current peak of the
auto‑convolution.  The refined shape is then interpolated to a larger
length and polished with projected sub‑gradient descent and a short
permutation search.  The whole search respects a 1000 s budget.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _max_convolution(a: np.ndarray) -> float:
    """Return the maximum entry of the linear convolution a*a."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _max_conv_and_subgrad(a: np.ndarray):
    """
    Returns (conv, max_val, kmax, subgrad) where subgrad is the gradient
    of the peak entry of the convolution w.r.t. a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    kmax = int(np.argmax(conv))
    max_val = conv[kmax]

    grad = np.zeros_like(a)
    lo = max(0, kmax - (n - 1))
    hi = min(kmax, n - 1)
    idx = np.arange(lo, hi + 1)
    partner = a[kmax - idx]
    grad[lo:hi + 1] = 2.0 * partner
    return conv, max_val, kmax, grad


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


# ----------------------------------------------------------------------
# Score utility (max convolution = objective after normalisation)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    return _max_convolution(a)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0
    momentum = np.zeros_like(a)

    while time.time() < deadline:
        _, _, _, grad = _max_conv_and_subgrad(a)

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_factor * target_sum / norm
            a -= lr * momentum
        else:
            a -= step_factor * target_sum * grad / (np.linalg.norm(grad) + 1e-12)

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] SubGD it={it:5d}  score={best_val:.6f}")

        it += 1

    return best, best_val


# ----------------------------------------------------------------------
# Targeted permutation search (simulated annealing on index order)
# ----------------------------------------------------------------------


def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.1,
                                 end_temp: float = 1e-4,
                                 top_frac: float = 0.02,
                                 low_frac: float = 0.02) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    while time.time() < deadline:
        conv = np.fft.ifft(np.fft.fft(a) * np.fft.fft(a)).real
        kmax = int(np.argmax(conv))

        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)

        if donor == receiver:
            continue

        a[donor], a[receiver] = a[receiver], a[donor]

        new_score = _score(a)

        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] PermSearch new best={best_score:.6f}")

        if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / max(temp, 1e-12)):
            pass
        else:
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (Konyagin style)
# ----------------------------------------------------------------------


def _build_conv_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Returns the matrix A such that (A @ g)[k] = (a * g)[k] (linear convolution).
    Shape is (2*n-1, n).
    """
    n = a.shape[0]
    i = np.repeat(np.arange(n), n)          # column index (repeated for each partner)
    j = np.tile(np.arange(n), n)            # partner index (used to compute row)
    rows = i + j                            # k = i + j
    cols = i
    data = a[i]
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _lp_refine_sequence(a0: np.ndarray,
                        target_sum: float,
                        mixing: float = 0.5,
                        max_iter: int = 12,

        res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs', options={'presolve': True})

        if sum_g == 0:

            print(f"[{time.time():.2f}] LP iter {it} new best M={best_M:.6f}")

        a = a_new
        it += 1

"""
Hybrid linear‑programming refinement and sub‑gradient polishing for the
non‑negative step‑function problem.

1.  A modest‑size seed (≈2000 entries) is generated from several families
    (power‑law, exponential, random Dirichlet, linear).  
2.  Konyagin‑style LP iteration: given the current vector a and its peak
    M = max(a*a), solve a linear program that maximises the total mass of
    a feasible companion g under the linear constraints (a∗g)_k ≤ M.
    After solving, g is scaled to have the same ℓ₁‑norm as a and mixed
    with a (parameter t).  Repeating this reduces the peak of the
    auto‑convolution.
3.  The refined shape is interpolated to a larger length (≈8000) and
    projected onto the simplex {x ≥ 0, Σx = √(2n)}.
4.  A projected momentum sub‑gradient descent on the exact max‑convolution
    further polishes the vector.
5.  A short simulated‑annealing permutation search swaps indices that
    heavily contribute to the peak.
6.  The result is trimmed and returned.  All phases respect the 1000 s
    budget; intermediate progress is printed.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _max_conv_and_subgrad(a: np.ndarray):
    """
    Returns (conv, max_val, kmax, subgrad) where subgrad is the gradient
    of the peak entry of the convolution w.r.t. a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    kmax = int(np.argmax(conv))
    max_val = conv[kmax]

    grad = np.zeros_like(a)
    lo = max(0, kmax - (n - 1))
    hi = min(kmax, n - 1)
    idx = np.arange(lo, hi + 1)
    partner = a[kmax - idx]
    grad[lo:hi + 1] = 2.0 * partner
    return conv, max_val, kmax, grad


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


# ----------------------------------------------------------------------
# Score utility (max convolution = objective after normalisation)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    return _max_convolution(a)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0
    momentum = np.zeros_like(a)

    while time.time() < deadline:
        _, _, _, grad = _max_conv_and_subgrad(a)

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_factor * target_sum / norm
            a -= lr * momentum
        else:
            a -= step_factor * target_sum * grad / (np.linalg.norm(grad) + 1e-12)

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] SubGD it={it:5d}  score={best_val:.6f}")

        it += 1

    return best, best_val

def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.1,
                                 end_temp: float = 1e-4,
                                 top_frac: float = 0.02,
                                 low_frac: float = 0.02) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    while time.time() < deadline:
        # linear convolution via zero‑padded FFT
        size = 1 << ((2 * n - 1).bit_length())
        fa = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
        kmax = int(np.argmax(conv))

        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)

        if donor == receiver:
            continue

        a[donor], a[receiver] = a[receiver], a[donor]

        new_score = _score(a)

        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] PermSearch new best={best_score:.6f}")

        if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / max(temp, 1e-12)):
            pass
        else:
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (Konyagin style)

def _build_conv_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Build a sparse matrix A such that (A @ g)[k] = (a ∗ g)[k] (linear convolution).
    Shape: (2*n‑1, n).
    """
    n = a.shape[0]
    i = np.repeat(np.arange(n), n)          # column index i
    j = np.tile(np.arange(n), n)            # partner index j
    rows = i + j
    cols = i
    data = a[i]
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    if time.time() < deadline - 5:
        perm_dead = deadline - 2
        a_opt, val_opt = _permutation_targeted_search(best_seq, perm_dead,
                                                      start_temp=0.15,
                                                      end_temp=1e-5,
                                                      top_frac=0.04,
                                                      low_frac=0.04)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Permutation search best={best_score:.6f}")

    # --------------------------------------------------------------
    # 7. Trim trailing zeros and return
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final evaluation logging
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    globals()["height_sequence_1"] = final_seq
    return final_seq
