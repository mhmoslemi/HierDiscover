"""
Improved search for a low‑max‑convolution step‑function.

Key ideas
---------
* Use a larger problem size (up to 30 000 coefficients) – the asymptotic
  optimum ≈ 1.5 improves with n.
* Enlarge the parametric pre‑search with truncated power‑laws and a
  beta‑shaped family, plus a few Dirichlet random starts.
* Adam‑style projected GD (soft‑max surrogate) with many restarts and a
  higher final β.
* Exact projected sub‑gradient descent.
* Deterministic greedy mass‑transfer (as in the reference code) combined
  with a cheap global random‑mass phase.
* A short simulated‑annealing phase that moves mass between coordinates
  while accepting occasional uphill moves.
* All phases respect the global time budget (default 1000 s).

The routine returns the best sequence found (as a plain ``list[float]``).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based – fast for the large n we use)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Scoring (the objective for a fixed sum is simply the maximal convolution)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Return the maximal self‑convolution (the value we minimise)."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for the initial fast sweep
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0))
    a *= target_sum / a.sum()
    return a


def _trunc_power_law_sequence(n: int, p: float, K: int, target_sum: float) -> np.ndarray:
    """Power‑law on the first K entries, zero afterwards."""
    a = np.zeros(n, dtype=np.float64)
    if K > 0:
        idx = np.arange(1, K + 1, dtype=np.float64)
        a[:K] = idx ** (-p)
        a *= target_sum / a.sum()
    return a


def _beta_shape_sequence(n: int, a: float, b: float, target_sum: float) -> np.ndarray:
    """Shape a_i ∝ (i+1)^a·(n-i)^b, a,b≥0."""
    i = np.arange(1, n + 1, dtype=np.float64)
    shape = (i ** a) * ((n + 1 - i) ** b)
    a = shape * (target_sum / shape.sum())
    return a


def _random_dirichlet_sequence(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex using a Dirichlet distribution."""
    vec = rng.gamma(0.5, 1.0, size=n)   # concentration < 1 → sparse vectors
    a = vec * (target_sum / vec.sum())
    return a


# ----------------------------------------------------------------------
# Soft‑max based gradient (used by Adam)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max surrogate)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                    target_sum: float,
                                    deadline: float) -> tuple[np.ndarray, float]:
    """Run several short Adam restarts from jittered starts."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        start = best_a + jitter
        start = np.maximum(start, 0.0)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,

                                         max_beta=2e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         max_iter: int = 5000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on the true max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.flatnonzero(conv >= max_val - 1e-12)

        # second‑largest value (ignore current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()
        delta_target = max_val - second_max
        if delta_target < 1e-14:
            break

        k = idx_max[rng.integers(len(idx_max))]   # pick a random active index

        n = a.shape[0]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k - idx]

        norm_g2 = np.dot(g, g)
        if norm_g2 == 0.0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = _project_to_simplex(a + jitter, target_sum)
            it += 1
            continue

        step = 0.9 * delta_target / norm_g2
        step = min(step, target_sum * 0.4 / (np.linalg.norm(g) + 1e-12))

        for _ in range(6):
            cand = _project_to_simplex(a - step * g, target_sum)
            cand_score = _score(cand)
            if cand_score < best_score - 1e-12:
                a = cand
                best_a = cand.copy()
                best_score = cand_score
                print(f"[{time.time():.2f}] Subgrad it={it:5d} step={step:.3e} score={best_score:.6f}")
                break
            step *= 0.5
        else:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = _project_to_simplex(a + jitter, target_sum)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (greedy peak reduction)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the current peak by moving mass from a donor with a
    large partner value to a receiver with a small partner value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest value (ignore all current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()
        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # pick first maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k‑i}
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        # donor = index with largest partner among those that can give mass
        donor_local = np.argmax(partner * donor_mask)
        donor = idx[donor_local]

        # receiver = index with smallest partner (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom

        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] DMT it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Global random mass‑transfer (exploratory)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        donor = np.argsort(-a)[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        receiver = np.argsort(a)[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing – cheap mass moves with probabilistic uphill steps
# ----------------------------------------------------------------------


def _annealing_opt(a0: np.ndarray,
                   target_sum: float,
                   deadline: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)
    cur_score = best_score

    start = time.time()
    while time.time() < deadline:
        # linear temperature schedule (cooling)
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        temperature = max(1e-8, (1.0 - frac) * best_score * 0.1)

        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        if a[i] <= 0.0:
            continue

        delta = rng.random() * a[i] * 0.5   # move up to 50 % of donor mass
        a[i] -= delta
        a[j] += delta
        # sum is unchanged, non‑negativity already guaranteed

        new_score = _score(a)
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # uphill acceptance with Boltzmann probability
            if rng.random() < math.exp(-(new_score - cur_score) / temperature):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal new best={best_score:.6f}")
        else:
            # revert move
            a[i] += delta
            a[j] -= delta

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The method adapts the problem size to the time
    budget, performs an extensive parametric sweep, several gradient‑based
    refinements and a short annealing phase, finally returns the trimmed
    sequence.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin for the harness

    # --------------------------------------------------------------
    # Choose problem size – larger n gives more degrees of freedom.
    # Heuristic: n grows with the allowed budget, capped at 40 000.
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 30000
    elif budget_s >= 500:
        n = 20000
    else:
        n = 12000
    target_sum = np.sqrt(2.0 * n)   # Σa = √(2 n) → evaluation = max convolution

    # --------------------------------------------------------------
    # Phase 0 – dense parametric pre‑search
    # --------------------------------------------------------------
    best_seq = np.full(n, target_sum / n, dtype=np.float64)   # uniform fallback
    best_score = _score(best_seq)
    rng = np.random.default_rng()

    pre_end = min(deadline, time.time() + max(5.0, budget_s * 0.10))

    # (a) pure power‑law
    for p in np.linspace(0.25, 0.90, 21):
        if time.time() >= pre_end:
            break
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # (b) power‑law + exponential tail
    for p in np.linspace(0.25, 0.90, 15):
        for q in np.linspace(0.0, 6.0, 15):
            if time.time() >= pre_end:
                break
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # (c) truncated power‑law
    for p in np.linspace(0.3, 0.9, 7):
        for frac in (0.25, 0.5, 0.75):
            if time.time() >= pre_end:
                break
            K = int(frac * n)
            seq = _trunc_power_law_sequence(n, float(p), K, target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # (d) beta‑shape family (flexible polynomial‑like shape)
    for a_par in np.linspace(0.0, 2.0, 5):
        for b_par in np.linspace(0.0, 2.0, 5):
            if time.time() >= pre_end:
                break
            seq = _beta_shape_sequence(n, float(a_par), float(b_par), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # (e) simple sine shape
    seq = _sin_sequence(n, target_sum)
    val = _score(seq)
    if val < best_score:
        best_score = val
        best_seq = seq.copy()

    # (f) random Dirichlet draws – give us diverse starting points
    for _ in range(5):
        if time.time() >= pre_end:
            break
        seq = _random_dirichlet_sequence(n, target_sum, rng)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # (g) previous solution if available (resize if necessary)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score = val
            best_seq = resized.copy()

    print(f"[{time.time():.2f}] After pre‑search best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase A – Adam‑style GD with restarts (soft‑max surrogate)
    # --------------------------------------------------------------
    gd_deadline = time.time() + (deadline - time.time()) * 0.30
    while time.time() < gd_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _gradient_descent_with_restarts(start_vec,
                                                         target_sum,
                                                         gd_deadline)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – Exact projected sub‑gradient descent
    # --------------------------------------------------------------
    subgrad_deadline = time.time() + (deadline - time.time()) * 0.25
    a_sg, val_sg = _subgradient_descent(best_seq,
                                         target_sum,
                                         subgrad_deadline)
    if val_sg < best_score:
        best_score = val_sg
        best_seq = a_sg.copy()
        print(f"[{time.time():.2f}] Subgradient descent improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase C – Deterministic greedy mass‑transfer + short global random
    # --------------------------------------------------------------
    while time.time() < deadline - 8.0:
        # deterministic greedy reduction
        a_ref, val_ref = _deterministic_mass_transfer(best_seq,
                                                      deadline,
                                                      target_sum)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] DMT improved best = {best_score:.6f}")

        # brief global random exploration
        a_rand, val_rand = _global_random_mass_transfer(best_seq,
                                                        deadline,
                                                        target_sum)
        if val_rand < best_score:
            best_score = val_rand
            best_seq = a_rand.copy()
            print(f"[{time.time():.2f}] Global random improved best = {best_score:.6f}")

        if time.time() > deadline - 2.0:
            break

    # --------------------------------------------------------------
    # Phase D – Simulated annealing (few seconds)
    # --------------------------------------------------------------
    anneal_deadline = time.time() + min(12.0, max(4.0, (deadline - time.time()) * 0.15))
    if anneal_deadline > time.time():
        a_anneal, val_anneal = _annealing_opt(best_seq,
                                             target_sum,
                                             anneal_deadline)
        if val_anneal < best_score:
            best_score = val_anneal
            best_seq = a_anneal.copy()
            print(f"[{time.time():.2f}] Annealing improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase E – Final polishing with a very high β (soft‑max)
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                deadline,
                                                target_sum,
                                                max_beta=1e6,
                                                step_factor=0.6,
                                                gamma=0.995)
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Post‑processing: trim trailing zeros and return
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final wall‑clock check with the provided evaluator
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
