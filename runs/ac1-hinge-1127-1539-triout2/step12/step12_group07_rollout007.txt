def _perm_anneal(a, deadline):
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    n = len(a)
    start = time.time()
    while time.time() < deadline:
        # compute temperature as function of elapsed
        elapsed_ratio = (time.time() - start) / (deadline - start)
        temp = max(0.1, (1 - elapsed_ratio)) # linear cooling; start 0.5 e.g.
        i, j = rng.integers(0, n, size=2)
        if i == j: continue
        # swap two positions
        a[i], a[j] = a[j], a[i]
        new_score = _max_convolution_fft(a)
        if new_score < cur_score or rng.random() < math.exp(-(new_score-cur_score)/temp):
            cur_score = new_score

"""
Hybrid optimizer – large‑scale upscaling + aggressive simulated annealing.

The algorithm:
1.  Take the best sequence known so far (global ``height_sequence_1``) or a
    simple power‑law seed.
2.  Project it onto the ℓ¹‑simplex (∑a = √(2n)).
3.  Upscale to a much larger length (30k‑50k) by linear interpolation,
    re‑project onto the simplex and apply an “alternating” ordering that
    pairs large values with small ones.
4.  Run a massive mass‑transfer simulated‑annealing pass (donor → receiver)
    while the convolution peak is monitored via FFT.
5.  Optional short projected sub‑gradient descent (exact max‑convolution).
6.  A final permutation annealing that swaps two positions with a Metropolis
    rule, refining the ordering.
7.  Trim trailing zeros, store the result globally and return it.

All steps respect non‑negativity and the ℓ¹‑norm constraint, and the code
keeps careful track of the remaining time so that it always finishes before
the 1000 s budget.
"""

import time
import math
import random
from typing import List

import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a (linear convolution) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Scoring (objective = max self‑convolution when sum = √(2n))
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Return the evaluate_sequence objective (max convolution)."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x ≥ 0, Σx = s}.
    """
    n = v.shape[0]
    if s < 0:
        raise ValueError("target sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(‑p) then renormalised to have ℓ¹‑norm = target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to length ``new_len``."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """
    Place the largest value leftmost, the second largest rightmost,
    the third largest next to the leftmost, etc.
    """
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    l, r = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[l] = v
            l += 1
        else:
            out[r] = v
            r -= 1
    return out


# ----------------------------------------------------------------------
# Simulated‑annealing mass transfer (no projection after each move)
# ----------------------------------------------------------------------


def _mass_transfer_anneal(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    start_temp: float = 0.5,
    end_temp: float = 1e-4,
    low_frac: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Greedy mass transfer from a “donor’’ (large contribution to the current peak)
    to a “receiver’’ (small contribution).  Acceptance follows the Metropolis rule.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_conv = _conv_fft(a)
    cur_score = cur_conv.max()
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()

    while time.time() < deadline:
        # temperature (exponential decay)
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # current peak index and the pair indices that create it
        kmax = int(np.argmax(cur_conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to the peak (a_i * a_{kmax-i})
        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        # choose a low‑contributing partner among the smallest ``low_frac`` of the
        # contributing indices
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        # propose moving a modest amount of mass
        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        # apply move
        a[donor] -= delta
        a[receiver] += delta

        # recompute convolution (full FFT – cheap enough)
        new_conv = _conv_fft(a)
        new_score = new_conv.max()

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < math.exp(
            -(new_score - cur_score) / max(temp, 1e-12)
        ):
            cur_conv = new_conv
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta

        # occasional global jitter to shake out of plateaus
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            # keep non‑negative & preserve sum via projection
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            cur_conv = _conv_fft(a)
            cur_score = cur_conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    step_factor: float = 0.45,
    gamma: float = 0.99,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent using the true max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        # indices where the max is attained (within tiny tolerance)
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]

        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Permutation annealing (swap‑based simulated annealing)
# ----------------------------------------------------------------------


def _swap_anneal(
    a0: np.ndarray,
    deadline: float,
    start_temp: float = 0.2,
    end_temp: float = 1e-5,
    swaps_per_sec: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Randomly swaps two positions; moves are accepted with a Metropolis rule.
    The aim is to improve the ordering of the (already‑shaped) values.
    """
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start

    while time.time() < deadline:
        # temperature (linear decay in log‑space)
        elapsed = time.time() - start
        tau = elapsed / total
        temp = start_temp * (end_temp / start_temp) ** tau

        # propose a swap
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]

        new_score = _max_convolution_fft(a)

        if new_score < cur_score or rng.random() < math.exp(
            -(new_score - cur_score) / max(temp, 1e-12)
        ):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # revert the swap
            a[i], a[j] = a[j], a[i]

        # throttle the loop a bit to avoid excessive CPU consumption
        if swaps_per_sec > 0:
            # target a certain number of swaps per second
            target_time = (elapsed + 1.0 / swaps_per_sec)
            while time.time() < start + target_time:
                pass

    return best_a, best_score

        # a modest power‑law seed
        n0 = 5000
        target0 = math.sqrt(2.0 * n0)
        seed = _power_law_sequence(n0, p=1.0, target_sum=target0)

    # ensure feasibility
    target0 = math.sqrt(2.0 * seed.shape[0])
    seed = _project_to_simplex(seed, target0)

    # ------------------------------------------------------------------
    # 2. decide final length according to remaining time
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 700:
        n_final = 50000
    elif remaining > 500:
        n_final = 45000
    elif remaining > 300:
        n_final = 40000
    elif remaining > 150:
        n_final = 30000
    else:
        n_final = seed.shape[0]  # fallback to the original size

    target_final = math.sqrt(2.0 * n_final)

    # ------------------------------------------------------------------
    # 3. upscale, re‑project, order
    # ------------------------------------------------------------------
    a = _resize_sequence(seed, n_final)
    a = _project_to_simplex(a, target_final)
    a = _alternating_order(a)  # pair large with small

    print(
        f"[{time.time():.2f}] Upscaled to {n_final} entries, initial max={_max_convolution_fft(a):.6f}"
    )

    # ------------------------------------------------------------------
    # 4. massive mass‑transfer annealing
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline - 80.0:
        mt_deadline = now + (deadline - now) * 0.55
        a, best_mt = _mass_transfer_anneal(a, mt_deadline, target_final)
        print(f"[{time.time():.2f}] Mass‑transfer best max={best_mt:.6f}")

    # ------------------------------------------------------------------
    # 5. brief sub‑gradient refinement (exact max)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline - 50.0:
        sub_deadline = now + (deadline - now) * 0.25
        a, best_sub = _subgradient_descent(a, sub_deadline, target_final)
        print(f"[{time.time():.2f}] Sub‑grad best max={best_sub:.6f}")

    # ------------------------------------------------------------------
    # 6. permutation annealing (swap based)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline - 10.0:
        perm_deadline = deadline - 5.0
        a, best_perm = _swap_anneal(a, perm_deadline)
        print(f"[{time.time():.2f}] Permutation anneal best max={best_perm:.6f}")

    # ------------------------------------------------------------------
    # 7. final tidy‑up
    # ------------------------------------------------------------------
    arr = np.array(a, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional logging of the official evaluation
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for possible reuse in a subsequent run
    globals()["height_sequence_1"] = final_seq
    return final_seq
