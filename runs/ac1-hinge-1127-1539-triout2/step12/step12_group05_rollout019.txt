            a[K:] = idx[K:] ** (-p2)

            # beta
            a_param = rng.uniform(0.2,5.0)
            b_param = rng.uniform(0.2,5.0)

        if np.all(a == 0):

"""
Step‑function optimizer for the benchmark.

The objective is 2·n·max(conv(a,a))/sum(a)² where
    n = len(a) and conv is the linear convolution.
For a fixed sum this is simply the maximal convolution entry.
The algorithm works in three stages:

1. **Broad parametric search** – many classic families (power law,
   exponential‑power‑law, broken power‑law, beta‑shape, sin‑power and
   Gaussian) are sampled randomly.  Each candidate is normalised to the
   simplex Σa = √(2·n) and evaluated with a fast FFT based convolution.
   The best candidate so far is stored.

2. **Guided exact mass transfer** – an exact sub‑gradient of the
   currently maximal convolution entry is computed (averaged over all
   maximal indices).  A feasible descent direction that preserves the
   L¹‑norm is taken and a line‑search moves the mass as far as possible
   towards the second‑largest entry.  This loop runs until the time
   budget expires.

3. **Simple hill‑climbing fine‑tuning** – random donor/receiver moves
   (up to a few percent of the donor’s mass) are accepted if they
   improve the score.  The loop finishes the remaining time.

All steps respect non‑negativity and the simplex constraint.
The final sequence is truncated (trailing ≈0 values are removed) and
returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities (linear convolution, full length 2·n‑1)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) via real FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft_pair(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution x*y (full length 2·n‑1) via real FFT."""
    n = x.size
    size = 1 << ((2 * n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Benchmark score = maximal entry of a*a (after simplex normalisation)."""
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    # the factor 2·n / s² equals 1 when s = √(2·n)
    return float(2.0 * a.size * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (dense pre‑search)
# ----------------------------------------------------------------------


def _power_law(n: int, p: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    return a


def _exp_power_law(n: int, p: float, q: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    return a


def _broken_power_law(n: int, p1: float, p2: float, alpha: float) -> np.ndarray:
    K = max(1, int(alpha * n))
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = np.empty_like(idx)
    a[:K] = idx[:K] ** (-p1)
    a[K:] = idx[K:] ** (-p2)
    return a


def _beta_shape(n: int, a_par: float, b_par: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = (x ** (a_par - 1)) * ((1 - x) ** (b_par - 1))
    return a


def _sin_power(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.sin(np.pi * x) ** p
    return a


def _gaussian_shape(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - mid) / (sigma * n)) ** 2)
    return a


# ----------------------------------------------------------------------
# Guided exact mass‑transfer (core refinement)
# ----------------------------------------------------------------------


def _compute_subgradient(a: np.ndarray, max_idx: np.ndarray) -> np.ndarray:
    """Exact sub‑gradient of max(conv) with respect to a (averaged)."""
    n = a.size
    g = np.zeros_like(a)
    for k in max_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g[idx] += 2.0 * a[k - idx]          # contribution a_{k-i}
    if max_idx.size:
        g /= max_idx.size
    return g


def _guided_mass_transfer(a0: np.ndarray, deadline: float, target_sum: float) -> tuple[np.ndarray, float]:
    """Repeatedly reduce the current maximal convolution entry."""
    a = a0.copy()
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0
    tol = 1e-12

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        # indices attaining (approximately) the maximum
        max_idx = np.where(conv >= max_val - tol)[0]

        # exact sub‑gradient (averaged over all maximal indices)
        g = _compute_subgradient(a, max_idx)

        # feasible descent direction (preserves Σa)
        d = -g + g.mean()

        # largest feasible step before any component becomes negative
        neg = d < 0
        if np.any(neg):
            t_nonneg = np.min(a[neg] / -d[neg])
        else:
            t_nonneg = np.inf

        # linearised change of the convolution in direction d
        delta = 2.0 * _conv_fft_pair(a, d)          # length 2n‑1
        delta_max = delta[max_idx]
        delta_k = delta_max.min()                  # most favourable peak

        if delta_k >= 0:
            # direction does not help – stop
            break

        # second largest entry (to aim for)
        mask = conv < max_val - tol
        if np.any(mask):
            second_max = conv[mask].max()
        else:
            second_max = max_val

        # step needed to bring the current maximum down to the second one
        t_to_second = (max_val - second_max) / (-delta_k)

        # safe step (slightly back‑off)
        t = min(t_nonneg, t_to_second) * 0.99
        if t <= 0:
            break

        a += t * d
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Guided MT iter={it} score={best_val:.6f}")

        # occasional jitter to escape tiny plateaus
        if it % 500 == 0 and rng.random() < 0.03:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Random hill‑climbing moves (light‑weight local search)
# ----------------------------------------------------------------------


def _hill_climb(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Random donor/receiver moves; accept only if the score improves."""
    a = a0.copy()
    best_val = _score(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j or a[i] < 1e-12:
            continue
        # move up to 5 % of donor's mass
        delta = rng.random() * a[i] * 0.05
        a[i] -= delta
        a[j] += delta

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            print(f"[{time.time():.2f}] Hill‑climb step={it} new best={best_val:.6f}")
        else:
            # revert
            a[i] += delta
            a[j] -= delta

        it += 1

    return a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal benchmark score.
    The routine respects the supplied time budget and returns the best
    sequence found (trailing near‑zero entries are trimmed).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # a small safety margin

    # ---------------------------------------------------------------
    # Choose a problem size – larger n gives a finer discretisation.
    # ---------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = math.sqrt(2.0 * n)                 # Σa is fixed to this value

    # ---------------------------------------------------------------
    # Phase 0 – Randomised parametric pre‑search (fast but diverse)
    # ---------------------------------------------------------------
    best_seq = np.full(n, target_sum / n, dtype=np.float64)   # uniform start
    best_score = _score(best_seq)
    rng = np.random.default_rng()

    # fixed small grid for the classic families (quick sweep)
    for p in np.arange(0.2, 1.6, 0.2):
        seq = _power_law(n, float(p))
        seq = _project_to_simplex(seq, target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()
    for p in np.arange(0.2, 1.6, 0.2):
        for q in np.arange(0.0, 6.01, 1.0):
            seq = _exp_power_law(n, float(p), float(q))
            seq = _project_to_simplex(seq, target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()
    for p1 in np.arange(0.2, 1.0, 0.2):
        for p2 in np.arange(1.0, 2.2, 0.2):
            for alpha in np.arange(0.1, 0.9, 0.2):
                seq = _broken_power_law(n, float(p1), float(p2), float(alpha))
                seq = _project_to_simplex(seq, target_sum)
                val = _score(seq)
                if val < best_score:
                    best_score, best_seq = val, seq.copy()

    # Random sampling of richer families
    while time.time() < deadline - 30.0:
        family = rng.integers(5)
        if family == 0:                                   # beta‑shape
            a_par = rng.uniform(0.2, 5.0)
            b_par = rng.uniform(0.2, 5.0)
            seq = _beta_shape(n, a_par, b_par)
        elif family == 1:                                 # sin‑power
            p = rng.uniform(0.2, 8.0)
            seq = _sin_power(n, p)
        elif family == 2:                                 # gaussian
            sigma = rng.uniform(0.08, 0.30)
            seq = _gaussian_shape(n, sigma)
        elif family == 3:                                 # heavy‑tailed power
            p = rng.uniform(0.5, 3.0)
            seq = _power_law(n, p)
        else:                                             # exponential‑power
            p = rng.uniform(0.2, 3.0)
            q = rng.uniform(0.0, 12.0)
            seq = _exp_power_law(n, p, q)

        # normalise to the simplex and evaluate
        if seq.sum() == 0:
            continue
        seq = _project_to_simplex(seq, target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()
            print(f"[{time.time():.2f}] Random family improve score={best_score:.6f}")

    print(f"[{time.time():.2f}] After parametric search: best score = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase 1 – Guided exact mass‑transfer (core refinement)
    # ---------------------------------------------------------------
    while time.time() < deadline - 15.0:
        a_ref, val_ref = _guided_mass_transfer(best_seq, deadline, target_sum)
        if val_ref < best_score:
            best_score, best_seq = val_ref, a_ref.copy()
            print(f"[{time.time():.2f}] Guided MT improved best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase 2 – Light hill‑climbing to escape possible plateaus
    # ---------------------------------------------------------------
    if time.time() < deadline - 2.0:
        a_hc, val_hc = _hill_climb(best_seq, deadline)
        if val_hc < best_score:
            best_score, best_seq = val_hc, a_hc.copy()
            print(f"[{time.time():.2f}] Hill‑climb improved best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Final trimming and return
    # ---------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final benchmark printout
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated benchmark score = {val:.6f}")
    except Exception:
        pass

    return final_seq
