"""Improved search for a low‑peak step function.

The algorithm builds on the previous hybrid approach but adds:
* a fast golden‑section search for the optimal power‑law (and exponential)
  exponent on a medium‑size problem,
* more aggressive annealed soft‑max gradient descent from this tuned seed,
* a down‑scaled LP alternating refinement with several mixing steps,
* a final up‑scale together with random permutation trials and deterministic
  reorderings,
* sub‑gradient polishing and a final deterministic mass‑transfer pass.

All vectors are projected onto the ℓ¹‑simplex (sum = √(2 n)), because
the evaluation score then equals the maximal self‑convolution entry.
The routine respects the 1000 s time budget and stores the best sequence
found in the global name ``height_sequence_1`` for possible warm‑starts.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Projection onto ℓ¹‑simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised to ℓ¹‑norm = target_sum."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    left_turn = True
    for v in sorted_vals:
        if left_turn:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        left_turn = not left_turn
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread large values uniformly using a step coprime to the length."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Soft‑max gradient (annealed GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent_v2_ext(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 1e5,
    step_factor: float = 0.7,
    gamma: float = 0.99,
) -> tuple[np.ndarray, float]:
    """Projected GD on the soft‑max approximation of the max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 200 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur = _max_convolution_fft(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} max={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent (exact max)
# ----------------------------------------------------------------------


def _subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    step_factor: float = 0.5,
    gamma: float = 0.99,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent using the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        # indices attaining the max (up to numerical tolerance)
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]
        grad = np.zeros_like(a)
        for k in max_idxs:
            lo = max(0, k - (a.shape[0] - 1))
            hi = min(k, a.shape[0] - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur = _max_convolution_fft(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} max={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (simulated‑annealing style)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    start_temp: float = 0.5,
    end_temp: float = 1e-4,
    low_frac: float = 0.05,
) -> tuple[np.ndarray, float]:
    """Greedy mass moves with a decaying Metropolis acceptance rule."""
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            it += 1
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming alternating refinement (small dimensionality)
# ----------------------------------------------------------------------


def _lp_optimize(
    f: np.ndarray, M: float, target_sum: float, rng: np.random.Generator
) -> np.ndarray | None:
    """
    Solve a LP that maximises a random linear functional of `b`
    under constraints `f * b ≤ M` and `b ≥ 0`.  Result is scaled to ℓ¹‑norm = target_sum.
    Returns None on failure.
    """
    n = f.shape[0]
    rows = []
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=np.int64)
        row = np.zeros(n, dtype=np.float64)
        row[i] = f[k - i]
        rows.append(row)
    A = np.vstack(rows)  # (2n‑1) × n
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Random objective (maximise c·b)
    c = -rng.random(n)  # linprog minimises; we negate

    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    except Exception as e:
        print(f"[{time.time():.2f}] LP exception: {e}")
        return None

    if not res.success:
        print(f"[{time.time():.2f}] LP failed: {res.message}")
        return None

    b = res.x
    total = b.sum()
    if total <= 0:
        return None
    b *= target_sum / total
    return b


# ----------------------------------------------------------------------
# Exponent optimisation (golden‑section)
# ----------------------------------------------------------------------


def _optimize_power_law(
    n: int,
    target_sum: float,
    lo: float = 0.15,
    hi: float = 0.90,
    max_iter: int = 30,
    deadline: float | None = None,
) -> tuple[np.ndarray, float, float]:
    """Golden‑section search for the exponent p of a power‑law sequence."""
    phi = (np.sqrt(5) - 1) / 2  # 0.618...
    a, b = lo, hi
    c = b - phi * (b - a)
    d = a + phi * (b - a)

    seq_c = _power_law_sequence(n, c, target_sum)
    val_c = _max_convolution_fft(seq_c)
    seq_d = _power_law_sequence(n, d, target_sum)
    val_d = _max_convolution_fft(seq_d)

    for _ in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        if val_c < val_d:
            b, d, val_d = d, c, val_c
            c = b - phi * (b - a)
            seq_c = _power_law_sequence(n, c, target_sum)
            val_c = _max_convolution_fft(seq_c)
        else:
            a, c, val_c = c, d, val_d
            d = a + phi * (b - a)
            seq_d = _power_law_sequence(n, d, target_sum)
            val_d = _max_convolution_fft(seq_d)

    if val_c < val_d:
        return seq_c, val_c, c
    else:
        return seq_d, val_d, d


def _optimize_exponential(
    n: int,
    target_sum: float,
    lo: float = 0.001,
    hi: float = 0.04,
    max_iter: int = 30,
    deadline: float | None = None,
) -> tuple[np.ndarray, float, float]:
    """Golden‑section search for the exponential decay λ."""
    phi = (np.sqrt(5) - 1) / 2
    a, b = lo, hi
    c = b - phi * (b - a)
    d = a + phi * (b - a)

    seq_c = _exponential_sequence(n, c, target_sum)
    val_c = _max_convolution_fft(seq_c)
    seq_d = _exponential_sequence(n, d, target_sum)
    val_d = _max_convolution_fft(seq_d)

    for _ in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        if val_c < val_d:
            b, d, val_d = d, c, val_c
            c = b - phi * (b - a)
            seq_c = _exponential_sequence(n, c, target_sum)
            val_c = _max_convolution_fft(seq_c)
        else:
            a, c, val_c = c, d, val_d
            d = a + phi * (b - a)
            seq_d = _exponential_sequence(n, d, target_sum)
            val_d = _max_convolution_fft(seq_d)

    if val_c < val_d:
        return seq_c, val_c, c
    else:
        return seq_d, val_d, d


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid optimisation that combines exponent tuning, annealed GD,
    LP alternation, up‑scaling, permutation trials and final polishing.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Phase 0 – decide a moderate problem size for the bulk of the work
    # ------------------------------------------------------------------
    n_expl = 8000 if budget_s >= 800 else 6000
    target_expl = np.sqrt(2.0 * n_expl)

    # ------------------------------------------------------------------
    # Phase 1 – continuous‑parameter search (power‑law + exponential)
    # ------------------------------------------------------------------
    param_dead = min(deadline - 30.0, time.time() + 20.0)
    seq_pow, val_pow, p_opt = _optimize_power_law(
        n_expl, target_expl, lo=0.15, hi=0.90, max_iter=35, deadline=param_dead
    )
    seq_exp, val_exp, lam_opt = _optimize_exponential(
        n_expl, target_expl, lo=0.001, hi=0.04, max_iter=35, deadline=param_dead
    )
    print(f"[{time.time():.2f}] Power‑law p={p_opt:.4f} max={val_pow:.6f}")
    print(f"[{time.time():.2f}] Exponential λ={lam_opt:.5f} max={val_exp:.6f}")

    if val_pow < val_exp:
        best_seq = seq_pow
        best_val = val_pow
    else:
        best_seq = seq_exp
        best_val = val_exp

    # ------------------------------------------------------------------
    # Phase 2 – build a diverse pool of seeds
    # ------------------------------------------------------------------
    init_pool = [best_seq]
    # jittered variants of the best
    for _ in range(6):
        jitter = rng.normal(scale=target_expl * 0.02, size=n_expl)
        seq_j = best_seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    # previous global solution (if any)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.shape[0] != n_expl:
            prev = _resize_sequence(prev, n_expl)
        init_pool.append(_project_to_simplex(prev, target_expl))

    # a few random simplex points
    for _ in range(4):
        init_pool.append(_random_simplex(n_expl, target_expl, rng))

    # ------------------------------------------------------------------
    # Phase 3 – aggressive annealed gradient descent on medium size
    # ------------------------------------------------------------------
    gd_budget = max(0.0, (deadline - time.time()) * 0.45)
    gd_deadline = time.time() + gd_budget
    best_seq_gd = None
    best_val_gd = np.inf

    while time.time() < gd_deadline:
        seed = rng.choice(init_pool)

        run_dead = min(time.time() + 12.0, gd_deadline)
        if run_dead <= time.time():
            break

        a_opt, val_opt = _annealed_gradient_descent_v2_ext(
            seed, run_dead, target_expl, max_beta=5e5, step_factor=0.7, gamma=0.99
        )
        if val_opt < best_val_gd:
            best_val_gd = val_opt
            best_seq_gd = a_opt.copy()
            print(f"[{time.time():.2f}] GD new best max={best_val_gd:.6f}")

    if best_seq_gd is not None:
        best_seq = best_seq_gd
        best_val = best_val_gd

    # ------------------------------------------------------------------
    # Phase 4 – down‑scale & LP alternating refinement
    # ------------------------------------------------------------------
    n_lp = min(1500, max(800, n_expl // 5))
    target_lp = np.sqrt(2.0 * n_lp)
    seq_lp = _resize_sequence(best_seq, n_lp)
    seq_lp = _project_to_simplex(seq_lp, target_lp)

    lp_deadline = min(deadline - 20.0, time.time() + 30.0)
    lp_iter = 0
    while time.time() < lp_deadline:
        M = _max_convolution_fft(seq_lp)
        b_opt = _lp_optimize(seq_lp, M, target_lp, rng)
        if b_opt is None:
            break
        t = 0.25  # simple mixing weight
        seq_lp = (1 - t) * seq_lp + t * b_opt
        seq_lp = _project_to_simplex(seq_lp, target_lp)
        cur = _max_convolution_fft(seq_lp)
        print(f"[{time.time():.2f}] LP iter={lp_iter} max={cur:.6f}")
        lp_iter += 1
        if lp_iter >= 8:
            break

    # optional tiny GD on the LP problem
    if time.time() < deadline - 10.0:
        gd_dead = time.time() + min(8.0, deadline - time.time() - 6.0)
        a_opt, val_opt = _annealed_gradient_descent_v2_ext(
            seq_lp, gd_dead, target_lp, max_beta=1e5, step_factor=0.6, gamma=0.98
        )
        if val_opt < _max_convolution_fft(seq_lp):
            seq_lp = a_opt.copy()
            print(f"[{time.time():.2f}] Post‑LP GD improved max={val_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – up‑scale to a large final size
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 80.0:
        n_final = 16000
    elif remaining > 40.0:
        n_final = 12000
    else:
        n_final = max(8000, n_lp * 2)
    target_final = np.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(seq_lp, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # ------------------------------------------------------------------
    # Phase 6 – permutation trials to spread large values
    # ------------------------------------------------------------------
    best_seq_big = seq_big
    best_max = _max_convolution_fft(seq_big)

    # try a handful of random permutations
    for _ in range(7):
        perm = rng.permutation(n_final)
        cand = seq_big[perm]
        cur = _max_convolution_fft(cand)
        if cur < best_max:
            best_max = cur
            best_seq_big = cand.copy()
            print(f"[{time.time():.2f}] Random perm improved max={best_max:.6f}")

    # deterministic reorderings
    candidates = [
        best_seq_big,
        _alternating_order(best_seq_big),
        _even_spread_order(best_seq_big),
    ]
    seq_big = min(candidates, key=_max_convolution_fft)
    best_max = _max_convolution_fft(seq_big)
    print(f"[{time.time():.2f}] After ordering max={best_max:.6f}")

    # ------------------------------------------------------------------
    # Phase 7 – sub‑gradient polishing on the large problem
    # ------------------------------------------------------------------
    if time.time() < deadline - 6.0:
        sub_dead = deadline - 4.0
        a_sub, val_sub = _subgradient_descent(
            seq_big, sub_dead, target_final, step_factor=0.45, gamma=0.995
        )
        if val_sub < best_max:
            best_max = val_sub
            seq_big = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑grad polished max={best_max:.6f}")

    # ------------------------------------------------------------------
    # Phase 8 – final deterministic mass‑transfer pass
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        mt_dead = time.time() + min(5.0, deadline - time.time() - 1.0)
        a_mt, val_mt = _deterministic_mass_transfer(
            seq_big,
            mt_dead,
            target_final,
            start_temp=0.25,
            end_temp=1e-5,
            low_frac=0.05,
        )
        if val_mt < best_max:
            best_max = val_mt
            seq_big = a_mt.copy()
            print(f"[{time.time():.2f}] Final MT improved max={best_max:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # ------------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Optional final evaluation (purely for logging)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    globals()["height_sequence_1"] = final_seq
    return final_seq
