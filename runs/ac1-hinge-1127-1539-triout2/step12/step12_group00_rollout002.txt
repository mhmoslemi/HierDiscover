def _adam_optimize(a0, deadline, target_sum, max_beta=1e5, lr=0.05, beta1=0.9, beta2=0.999, epsilon=1e-8):
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_a = a.copy()
    best_val = _max_convolution_fft(a) # over target_sum normalization?
    # (target sum ensures max conv equals score)
    beta = 0.5
    while time.time() < deadline:
        t += 1
        # compute gradient of smoothed max
        grad, b = _gradient_softmax(a, beta)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        a -= step
        # clip negative before projection?
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)
        # optional jitter
        if t % 200 == 0:
            # small random perturbation to escape
            jitter = np.random.default_rng().normal(scale=target_sum*0.001 / (beta+1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(np.maximum(a,0.0), target_sum)
        # evaluate actual max conv
        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"Adam it={t} β={beta:.2f} val={best_val:.6f}")
        # increase beta for sharper approximation
        if t % 300 == 0 and beta < max_beta:
            beta = min(beta*2, max_beta)
    return best_a, best_val

def _gradient_softmax(a, beta):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2*n - 1]
    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)  # softmax over b
    # to compute gradient of sum_i p_i * b_i w.r.t a = sum_i p_i * d b_i/da
    # we can compute convolution of a_rev and p as earlier
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2*n - 1]
    # grad = 2 * conv_rev[n-1: n-1+n]
    grad = 2.0 * conv_rev[n-1: n-1+n]
    return grad, b

"""
Hybrid optimizer for the convolution‑max problem.

The algorithm works in three stages:

1. **Shape discovery** – a moderate‑size (≈5 000) sequence is
   created from several families (power‑law, exponential, random).
   A projected Adam optimizer on a soft‑max approximation of the
   max‑convolution refines the shape.  The best shape is kept.

2. **Upscaling & exact polishing** – the shape is interpolated to a
   much larger length (≈15‑20 000) and the true max‑convolution is
   minimised by a projected sub‑gradient method (exact gradient for
   the max entry).  The large‑scale run lasts for the bulk of the
   time budget.

3. **Final shaking** – a deterministic mass‑transfer (simulated‑annealing‑style)
   local search finishes the optimisation, after which a few simple
   reorderings are tried and the best one is returned.

All updates respect the non‑negativity and ℓ¹‑simplex constraint
(∑a = √(2 n)).  The routine prints progress and stores the
result in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a (exact max, not smoothed)."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} then normalized to ℓ¹‑norm = target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then normalized."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform point on the ℓ¹‑simplex (Dirichlet‑like)."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Reordering heuristics
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    l, r = 0, n - 1
    left = True
    for v in sorted_vals:
        if left:
            out[l] = v
            l += 1
        else:
            out[r] = v
            r -= 1
        left = not left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread large values uniformly (step ≈ √n, coprime to n)."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns the gradient of the soft‑max approximation of max(a*a)
    and the raw convolution vector.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)            # soft‑max probabilities over conv entries

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1: n - 1 + n]   # gradient w.r.t. a
    return grad, b


# ----------------------------------------------------------------------
# Projected Adam on the smoothed objective
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   target_sum: float,
                   max_beta: float = 1e5,
                   lr: float = 0.05,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser for the soft‑max surrogate of max‑convolution.
    Returns the best sequence seen and its exact max‑convolution value.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    beta = 0.5
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        grad, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if t % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(np.maximum(a, 0.0), target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={t:4d} β={beta:.1e} max={best_val:.6f}")

        # sharpen the soft‑max
        if t % 300 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (uses true max‑convolution)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        # indices where the maximum is attained (within tiny tolerance)
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]

        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to keep the walk alive
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(np.maximum(a, 0.0), target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} max={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer (simulated‑annealing style)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-4,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy mass moves with a temperature‑controlled Metropolis acceptance.
    Returns the best sequence found and its max‑convolution.
    """
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    # pre‑compute convolution once; will be refreshed after each accepted move
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # pick the (one of) the current maximum index
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # product contributions for each pair producing the max entry
        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]                     # biggest contributor
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        # amount to move – up to half of the smaller of the two masses
        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            # Metropolis test
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT new best={best_score:.6f}")
        else:
            # roll back
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional re‑jitter to avoid stagnation
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = _project_to_simplex(np.maximum(a, 0.0), target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

        it = it + 1

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """

    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    rng = np.random.default_rng()

        seeds.append(_exponential_sequence(n_mid, float(lam), target_mid))
    for _ in range(6):
        seeds.append(_random_simplex(n_mid, target_mid, rng))

    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        seeds.append(_resize_sequence(prev, n_mid))

    # pick the best raw seed (according to exact max‑convolution)
    best_seed = None
    best_score = np.inf
    for s in seeds:
        sc = _max_convolution_fft(s)
        if sc < best_score:
            best_score = sc
            best_seed = s.copy()
    print(f"[{time.time():.2f}] Initial seed max={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam optimisation on the soft‑max surrogate
    # ------------------------------------------------------------------
    adam_dead = min(deadline - 30.0, time.time() + 120.0)   # up to 2 min
    if adam_dead > time.time():

                                          max_beta=5e5, lr=0.05,
                                          beta1=0.9, beta2=0.999)
    else:
        a_mid, mid_score = best_seed, best_score
    print(f"[{time.time():.2f}] After Adam max={mid_score:.6f}")

    # ------------------------------------------------------------------

    remaining = deadline - time.time()

    return final_seq
