def _hill_climb(a, deadline, target_sum):
    best_a = a.copy()
    best_score = _score(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        # pick donors and recipients
        i = rng.integers(n)
        # ensure donor mass > epsilon
        if best_a[i] < 1e-12:

        j = rng.integers(n)

"""
Optimiser for the step‑function benchmark.

The algorithm combines several complementary heuristics:

1. Fast parametric pre‑search over a few classic families
   (power‑law, exponential‑power‑law, broken power‑law, logistic)
   plus a few random Dirichlet starts.  The best seed is kept.

2. Many short Adam‑GD runs on the soft‑max approximation of the
   max‑convolution (β is increased up to 1e7).  Each run starts from a
   jittered version of the current best sequence.

3. Exact sub‑gradient guided mass‑transfer (greedy flattening of the
   maximal convolution entry).  Several runs are performed until time
   runs out.

4. Local hill‑climbing: random pairwise mass moves that are accepted
   only when they improve the true objective.  This fine‑tunes the
   solution after the bulk optimisation phases.

5. A final polishing Adam run with a very high temperature.

All steps respect the simplex constraint Σa = √(2 n) and non‑negativity.
The routine uses only NumPy and runs within the supplied wall‑clock
budget (default 1000 s).  The best sequence found is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, linear full convolution)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) using a real FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft_pair(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution x*y (full length 2·n‑1) using a real FFT."""
    n = x.size
    size = 1 << ((2 * n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Benchmark score (equal to max convolution when Σa = √(2n))."""
    n = a.shape[0]
    max_b = _max_conv(a)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (dense pre‑search)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _broken_power_law_sequence(
    n: int, p1: float, p2: float, alpha: float, target_sum: float
) -> np.ndarray:
    K = max(1, int(alpha * n))
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.empty_like(idx)
    a[:K] = idx[:K] ** (-p1)
    a[K:] = idx[K:] ** (-p2)
    a *= target_sum / a.sum()
    return a


def _logistic_sequence(n: int, mu: float, sigma: float, target_sum: float) -> np.ndarray:
    """Sigmoid shape a_i = 1/(1+exp((i-mu)/sigma))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max approximation)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of soft‑max approximation of max(a*a)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # stability: all entries ≤ max; clip possible numerical overshoot
    max_b = conv_full.max()
    diff = conv_full - max_b
    diff = np.minimum(diff, 0.0)          # ≤0
    exps = np.exp(beta * diff)           # soft‑max weights
    p = exps / exps.sum()

    # convolution of reversed a with p (needed for gradient)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]   # size n
    return grad, conv_full


def _adam_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 5e5,
    step_factor: float = 0.8,
    gamma: float = 0.999,
) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max approximation of the max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to help escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient guided mass‑transfer (Phase B)
# ----------------------------------------------------------------------


def _compute_gradient_for_max(a: np.ndarray, max_idx: np.ndarray) -> np.ndarray:
    """Exact sub‑gradient of max(conv) w.r.t. a (averaged over max indices)."""
    n = a.size
    g = np.zeros_like(a)
    for k in max_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        # contribution 2 * a[k - idx]
        g[idx] += 2.0 * a[k - idx]
    if max_idx.size > 0:
        g /= max_idx.size
    return g


def _guided_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Greedy exact reduction of the maximal convolution entry.
    At each step we move mass from indices that most increase the peak
    towards indices that least increase it, using an exact line‑search.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()

        mask_max = conv >= max_val - tol
        second_max = conv[~mask_max].max() if np.any(~mask_max) else 0.0
        max_idx = np.where(mask_max)[0]

        g = _compute_gradient_for_max(a, max_idx)

        # direction preserving total mass
        sum_g = g.sum()
        d = -g + sum_g / a.size * np.ones_like(a)   # Σd = 0

        # largest step before any component becomes negative
        neg = d < 0
        if np.any(neg):
            t_nonneg = np.min(a[neg] / -d[neg])
        else:
            t_nonneg = np.inf

        # effect on convolution along direction d
        delta = 2.0 * _conv_fft_pair(a, d)  # length 2n‑1
        delta_max = delta[max_idx]
        delta_k = delta_max.min()
        if delta_k >= 0:
            break   # cannot reduce the peak with this direction

        # step that would bring current max down to the second max
        t_to_second = (max_val - second_max) / (-delta_k)

        # take a safe fraction
        t = min(t_nonneg, t_to_second) * 0.99
        if t <= 0:
            break

        a += t * d
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] GuidedMT it={it} score={best_score:.6f}")

        # occasional jitter to move out of tiny plateaus
        if it % 500 == 0 and rng.random() < 0.05:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.size)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Local hill‑climbing refinement (pairwise mass moves)
# ----------------------------------------------------------------------


def _hill_climb(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
) -> tuple[np.ndarray, float]:
    """Random pairwise mass moves accepted only when they improve the true score."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    n = a.size
    rng = np.random.default_rng()
    # we move at most 0.1 % of the total mass in a single step
    max_step = max(target_sum * 0.001, 1e-12)

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-15:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        delta = min(a[i], max_step * rng.random())
        if delta <= 0:
            continue

        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        # Projection only to clean rounding errors
        cand = _project_to_simplex(cand, target_sum)

        val = _score(cand)
        if val < best_score:
            best_score = val
            best_a = cand
            a = cand  # continue from the improved point
            print(f"[{time.time():.2f}] HillClimb improved score={best_score:.6f}")

    return best_a, best_score

def _global_random_move(

    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    while time.time() < deadline - 0.5:
        # pick a donor among largest 2 % and a receiver among smallest 2 %
        top_k = max(1, int(0.02 * n))
        low_k = max(1, int(0.02 * n))
        donor = np.argsort(-a)[rng.integers(top_k)]
        receiver = np.argsort(a)[rng.integers(low_k)]
        if donor == receiver:
            continue
        delta = rng.random() * a[donor] * 0.05
        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] RandomMove new best={best_score:.6f}")

    return best_a, best_score
