"""
Propose a step‑function with a very low maximal self‑convolution.

The algorithm works in several stages, staying inside the 1000 s budget.

1. **Problem size** – a larger number of steps gives more freedom.
   The length `n` is chosen from the budget (up to 25000).

2. **Heavy parametric exploration** – a dense grid of simple families
   (power law, power‑law + exponential tail, shifted power law,
   truncated power law, pure exponential and a sine shape) is scanned.
   Afterwards a fast random search over the same families (≈2000
   candidates) is performed.  The best sequence so far is kept.

3. **Adam on a soft‑max approximation of the max‑convolution**.
   The temperature β is increased gradually up to 1e6.
   Adam’s adaptive step‑size and Euclidean projection onto the simplex
   (Σa = √(2n), aᵢ ≥ 0) keep the iterates feasible.

4. **Exact deterministic mass‑transfer** – a greedy reduction of the
   current convolution peak using the analytically required mass move.
   This step is run until the time budget is exhausted.

5. **Final polishing** – a short high‑β Adam run to squeeze the last
   slack.

All stages respect the global deadline; intermediate best values are
printed so one can see the progress.  The returned list is trimmed of
trailing zeros and is ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Scoring – because Σa = √(2n) the evaluation equals the maximal
# entry of the convolution.
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Return the maximal self‑convolution (the true objective)."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Families of candidate sequences (all returned already normalised)
# ----------------------------------------------------------------------


def _norm_and_scale(seq: np.ndarray, target_sum: float) -> np.ndarray:
    seq = np.maximum(seq, 0.0)
    s = seq.sum()
    if s == 0:
        return np.full_like(seq, target_sum / seq.size)
    return seq * (target_sum / s)


def _power_law(n: int, p: float, target_sum: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    return _norm_and_scale(a, target_sum)


def _shifted_power_law(n: int, p: float, shift: float, target_sum: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64) + shift
    a = i ** (-p)
    return _norm_and_scale(a, target_sum)


def _exp_tail(n: int, lam: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i / n)
    return _norm_and_scale(a, target_sum)


def _exp_power_law(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    return _norm_and_scale(a, target_sum)


def _truncated_power_law(n: int, p: float, cut_frac: float, target_sum: float) -> np.ndarray:
    cut = int(cut_frac * n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.where(i <= cut, i ** (-p), 0.0)
    return _norm_and_scale(a, target_sum)


def _sine_shape(n: int, target_sum: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0))
    return _norm_and_scale(a, target_sum)


def _mixture_power_exp(n: int, p: float, lam: float, w: float, target_sum: float) -> np.ndarray:
    """w * power + (1-w) * exponential, both non‑negative."""
    pow_part = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    exp_part = np.exp(-lam * np.arange(n, dtype=np.float64) / n)
    a = w * pow_part + (1.0 - w) * exp_part
    return _norm_and_scale(a, target_sum)


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(b) w.r.t. a."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    # soft‑max over the convolution values
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # compute gradient = 2 * (a convolved with p), trimmed to length n
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]          # centre part
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """Adam on the soft‑max surrogate of the max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 250                     # increase β every few hundred steps

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # Adam moments
        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam β={beta:.1f} it={it:5d} score={best_val:.6f}")

        # β schedule – exponential up to max_beta
        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (aggressive version)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the maximal convolution entry.
    Each iteration moves as much mass as analytically needed (up to 90 % of
    the donor’s current value) from a donor with the largest partner
    value to a receiver with the smallest partner value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second‑largest entry (ignore all current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        # pick the first maximal index (any works)
        k = idx_max[0]

        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]                     # values that multiply the donor entry
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        # donor = index with largest partner among positions having positive mass
        donor_local = np.argmax(partner * donor_mask)
        donor = idx[donor_local]

        # receiver = index with smallest partner (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        # compute analytical delta to bring the peak down to the second‑largest value
        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        # move as much as safely possible
        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] DMT it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Small random mass‑transfer phase (helps escape plateaus)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        # pick a donor among the biggest values
        k_top = max(1, int(top_frac * n))
        donor = np.argsort(-a)[rng.integers(k_top)]

        # pick a receiver among the smallest values
        k_low = max(1, int(low_frac * n))
        receiver = np.argsort(a)[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.2          # a bit larger than earlier version
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur = _score(a)
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] Random new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a minimal maximal
    self‑convolution.  Returns the best sequence (plain Python list).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n → more freedom, still cheap FFT)
    # --------------------------------------------------------------
    if budget_s >= 900:
        n = 25000
    elif budget_s >= 600:
        n = 20000
    elif budget_s >= 400:
        n = 15000
    else:
        n = 12000
    target_sum = np.sqrt(2.0 * n)                # Σa = √(2n)

    print(f"[{time.time():.2f}] Using n = {n}, target sum = {target_sum:.6f}")

    # --------------------------------------------------------------
    # Phase 0 – dense deterministic families
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # (a) power law grid
    for p in np.linspace(0.25, 0.95, 25):
        seq = _power_law(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # (b) shifted power law grid
    for p in np.linspace(0.25, 0.95, 15):
        for shift in np.linspace(0.0, n * 0.4, 9):
            seq = _shifted_power_law(n, float(p), float(shift), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # (c) power‑law + exponential tail
    for p in np.linspace(0.2, 1.0, 11):
        for q in np.linspace(0.0, 8.0, 13):
            seq = _exp_power_law(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # (d) pure exponential
    for lam in np.linspace(0.0, 6.0, 13):
        seq = _exp_tail(n, float(lam), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # (e) truncated power law
    for p in np.linspace(0.3, 1.2, 11):
        for cf in np.linspace(0.2, 0.8, 7):
            seq = _truncated_power_law(n, float(p), float(cf), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # (f) sine shape (always positive)
    seq = _sine_shape(n, target_sum)
    val = _score(seq)
    if val < best_score:
        best_score, best_seq = val, seq.copy()

    # (g) previous solution if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score, best_seq = val, resized.copy()

    print(f"[{time.time():.2f}] After deterministic families best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 0b – fast random search over the same families
    # --------------------------------------------------------------
    rng = np.random.default_rng()
    param_deadline = time.time() + (deadline - time.time()) * 0.20
    while time.time() < param_deadline:
        family = rng.integers(6)   # 0..5 different families
        if family == 0:                     # plain power‑law
            p = rng.uniform(0.15, 1.5)
            seq = _power_law(n, p, target_sum)

        elif family == 1:                   # shifted power‑law
            p = rng.uniform(0.15, 1.5)
            shift = rng.uniform(0.0, n * 0.5)
            seq = _shifted_power_law(n, p, shift, target_sum)

        elif family == 2:                   # power‑law + exp tail
            p = rng.uniform(0.15, 1.5)
            q = rng.uniform(0.0, 8.0)
            seq = _exp_power_law(n, p, q, target_sum)

        elif family == 3:                   # pure exponential
            lam = rng.uniform(0.0, 8.0)
            seq = _exp_tail(n, lam, target_sum)

        elif family == 4:                   # truncated power‑law
            p = rng.uniform(0.2, 1.5)
            cf = rng.uniform(0.15, 0.85)
            seq = _truncated_power_law(n, p, cf, target_sum)

        else:                               # mixture power + exp
            p = rng.uniform(0.2, 1.5)
            lam = rng.uniform(0.0, 8.0)
            w = rng.uniform(0.0, 1.0)
            seq = _mixture_power_exp(n, p, lam, w, target_sum)

        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()
            print(f"[{time.time():.2f}] Random family best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase A – Adam on smooth approximation (soft‑max)
    # --------------------------------------------------------------
    gd_deadline = time.time() + (deadline - time.time()) * 0.35
    a_opt, val_opt = _adam_gradient_descent(best_seq,
                                            gd_deadline,
                                            target_sum,
                                            max_beta=5e5,
                                            step_factor=0.7,
                                            gamma=0.996)
    if val_opt < best_score:
        best_score, best_seq = val_opt, a_opt.copy()
        print(f"[{time.time():.2f}] Adam improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – Exact deterministic mass‑transfer (aggressive)
    # --------------------------------------------------------------
    mt_deadline = time.time() + (deadline - time.time()) * 0.30
    a_mt, val_mt = _deterministic_mass_transfer_exact2(best_seq,
                                                       mt_deadline,
                                                       target_sum,
                                                       tol=1e-13)
    if val_mt < best_score:
        best_score, best_seq = val_mt, a_mt.copy()
        print(f"[{time.time():.2f}] Mass‑transfer improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase C – Short random rescue (helps to escape local plateaus)
    # --------------------------------------------------------------
    rand_deadline = time.time() + (deadline - time.time()) * 0.10
    a_rand, val_rand = _global_random_mass_transfer(best_seq,
                                                    rand_deadline,
                                                    target_sum)
    if val_rand < best_score:
        best_score, best_seq = val_rand, a_rand.copy()
        print(f"[{time.time():.2f}] Random rescue improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase D – Final high‑β Adam polish (if any time left)
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                deadline,
                                                target_sum,
                                                max_beta=1e6,
                                                step_factor=0.6,
                                                gamma=0.995)
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final check (harmless if evaluate_sequence is absent)
    try:
        val = evaluate_sequence(final_seq)    # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
