    best_f = f.copy()
    best_score = _score(f)
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        M = _max_convolution_fft(f)
        # Solve LP
        n = len(f)
        g_var = cp.Variable(n)
        conv_fg = cp.conv(cp.Constant(f), g_var)
        constraints = [conv_fg <= M, g_var >= 0]
        objective = cp.Maximize(cp.sum(g_var))
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000)
        if g_var.value is None:
            break
        g0 = np.maximum(g_var.value, 0.0)
        sum_g0 = g0.sum()
        if sum_g0 == 0:

        # Rescale g to have sum = target_sum

"""
Hybrid optimiser for the non‑negative step‑function problem (enhanced).

Main ideas
==========

1. Build a diverse pool of seed vectors (power‑law, exponential, Dirichlet,
   linear, previous best).
2. Perform fast projected sub‑gradient descent on the exact max‑convolution
   (the peak is a piece‑wise linear convex function, its sub‑gradient can be
   computed analytically).  A small momentum term and occasional jitter help
   escape flat regions.
3. Upscale the best vector to a larger length – more degrees of freedom often
   lower the peak.
4. Refine the up‑scaled vector with another round of sub‑gradient descent.
5. **LP refinement** – exactly the iteration described in the paper:
   given the current vector `f` and its peak `M = max(f*f)`,
   solve the linear program
        max   sum(g)
        s.t.  f * g ≤ M      (convolution, all shifts)
              g ≥ 0
   Scale the solution to the required L‑1 norm and blend it with `f`.
   Repeating this 4–5 times typically drives the peak noticeably lower.
6. A tiny hill‑climbing (random local) search for the last fraction of the
   time budget.
7. Trim trailing zeros, evaluate once for logging and return the best
   sequence found.

All helper functions are defined at module level – there are no closures,
no filesystem access and the implementation respects the 1000 s time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators (used for the initial pool)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence and renormalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """Create an exp(-lam*i) sequence, renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → left, 2nd largest → right, 3rd → next left, …."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


# ----------------------------------------------------------------------
# Scoring utility (objective = max convolution entry after normalising sum)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Objective value (the max entry of the self‑convolution)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    # When the L1‑norm is √(2n) the returned value equals max_b.
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Exact max‑convolution together with a sub‑gradient of the peak
# ----------------------------------------------------------------------


def _max_conv_and_subgrad(a: np.ndarray):
    """
    Returns (conv, max_val, k_max, subgrad) where subgrad is the gradient
    of the peak entry of the convolution w.r.t. a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    kmax = int(np.argmax(conv))
    max_val = conv[kmax]

    grad = np.zeros_like(a)
    lo = max(0, kmax - (n - 1))
    hi = min(kmax, n - 1)
    idx = np.arange(lo, hi + 1)
    partner = a[kmax - idx]
    grad[lo:hi + 1] = 2.0 * partner
    return conv, max_val, kmax, grad


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Momentum projected sub‑gradient descent on max(conv(a,a)).
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0
    momentum = np.zeros_like(a)

    while time.time() < deadline:
        _, _, _, grad = _max_conv_and_subgrad(a)

        # momentum accumulation
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_factor * target_sum / norm
            a -= lr * momentum
        else:
            a -= step_factor * target_sum * grad / (np.linalg.norm(grad) + 1e-12)

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] Sub‑GD it={it:5d}  score={best_val:.6f}")

        it += 1

    return best, best_val


# ----------------------------------------------------------------------
# Linear‑programming based refinement (the “f‑g” iteration)
# ----------------------------------------------------------------------


def _lp_refinement(f0: np.ndarray,
                   target_sum: float,
                   deadline: float,
                   t: float = 0.2,
                   max_iters: int = 6) -> tuple[np.ndarray, float]:
    """
    Perform a few iterations of the LP based refinement described in the
    statement: maximise sum(g) under f * g ≤ M.  The algorithm is
    *exactly* the one from the paper (no linearisation), so it often beats
    pure sub‑gradient descent.
    """
    try:
        import cvxpy as cp
    except Exception as e:
        print("cvxpy not available, skipping LP refinement:", e)
        return f0, _score(f0)

    f = f0.copy()
    best_f = f.copy()
    best_val = _score(f)

    n = len(f)
    for it in range(max_iters):
        if time.time() >= deadline:
            break

        M = _max_convolution_fft(f)

        # ----- linear programme -----
        g = cp.Variable(n)
        conv_fg = cp.conv(cp.Constant(f), g)           # linear in g
        constraints = [conv_fg <= M, g >= 0]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        # ECOS works well for medium‑size LPs; we keep the maximum number of
        # interior‑point iterations modest to stay inside the time budget.
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, feastol=1e-9, reltol=1e-9)

        if g.value is None:
            print(f"[{time.time():.2f}] LP iteration {it} failed to converge.")
            break

        g0 = np.maximum(g.value, 0.0)
        sum_g0 = g0.sum()
        if sum_g0 == 0:
            break

        # Scale g to the required L1‑norm
        g_scaled = g0 * (target_sum / sum_g0)

        # Convex combination with the old vector
        f_new = (1 - t) * f + t * g_scaled
        f_new = _project_to_simplex(f_new, target_sum)

        new_val = _score(f_new)
        if new_val < best_val:
            best_val = new_val
            best_f = f_new.copy()
            print(f"[{time.time():.2f}] LP it={it}  new best={best_val:.6f}")

        f = f_new

    return best_f, best_val


# ----------------------------------------------------------------------
# Simple hill‑climbing (random local moves)
# ----------------------------------------------------------------------


def _random_local_search(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a tiny amount of mass from a random
    donor to a random receiver.  Accept only improvements.
    """
    a = a0.copy()
    best = a.copy()
    best_val = _score(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] < 1e-12:
            continue

        # move at most 20 % of the donor's mass
        delta = rng.random() * (0.2 * a[i] + 1e-12)
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_to_simplex(a_new, target_sum)

        val = _score(a_new)
        if val < best_val:
            best_val = val
            best = a_new.copy()
            a = a_new
            print(f"[{time.time():.2f}] Local search improvement: {best_val:.6f}")
        # else discard the move
    return best, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with a low benchmark score.
    The routine follows the pipeline described in the module docstring
    and respects the given time budget.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # 1. Choose an exploration length
    # --------------------------------------------------------------
    if budget_s >= 800:
        n_expl = 8000
    elif budget_s >= 500:
        n_expl = 6000
    else:
        n_expl = 4000
    target_expl = np.sqrt(2.0 * n_expl)

    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # 2. Build a diverse pool of seed sequences
    # --------------------------------------------------------------
    init_pool = []

    # power‑law family
    for p in np.linspace(0.35, 0.75, 9):

    for lam in np.linspace(0.0015, 0.015, 7):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))

    # random Dirichlet‑like points
    for _ in range(6):
        a = rng.random(n_expl)
        a *= target_expl / a.sum()
        init_pool.append(a)

    # linear decreasing shape
    lin = np.linspace(1.0, 0.0, n_expl)
    lin *= target_expl / lin.sum()
    init_pool.append(lin)

    # previous best, if any
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.shape[0] != n_expl:
            prev = _resize_sequence(prev, n_expl)
        init_pool.append(prev)

    # jittered copies for extra diversity
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.008, size=n_expl)
        seq_j = seq + jitter

    # --------------------------------------------------------------

    best_seq = None

            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best score={best_score:.6f}")

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 15.0:

                                              target_expl, step_factor=0.5, gamma=0.99)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Refine sub‑GD best={best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Upscale to a larger length (extra degrees of freedom)
    # --------------------------------------------------------------
    if best_seq is None:

    # Pick a larger final size if we still have enough time
