    a = initial_seq.copy()
    best_seq = a.copy()

        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        T = T0 * (T_end / T0) ** tau  # exponential decay
        if T < 1e-12:
            T = 1e-12

def _max_convolution_fft(a):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())

def annealing_optimize(a0, deadline, target_sum):
    a = a0.copy()
    cur_val = _max_convolution_fft(a)
    best = a.copy()
    best_val = cur_val
    start = time.time()
    T0 = target_sum
    T_end = target_sum * 1e-4
    n = a.shape[0]

    while time.time() < deadline:
        elapsed = time.time() - start
        total = deadline - start
        tau = elapsed / total if total>0 else 0.0
        T = T0 * (T_end / T0) ** tau
        # operation selection
        if rng.random() < 0.3:
            # swap
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            new_val = _max_convolution_fft(a)
            dE = new_val - cur_val
            if dE <= 0 or rng.random() < np.exp(-dE / (T + 1e-12)):
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = a.copy()
            else:
                # revert swap
                a[i], a[j] = a[j], a[i]
        else:
            # mass transfer
            i = rng.integers(0, n)
            if a[i] <= 0:
                continue
            j = rng.integers(0, n-1)
            if j >= i:
                j += 1
            # compute max delta fraction
            # fraction = min(1.0, max(0.01, T / T0))  # at start fraction 1, later small.
            # max generate delta as a[i] * fraction * rng.random()
            # but also should not exceed capacity of 1000
            max_delta = a[i] * max(0.01, T / T0)  # scaling factor
            if max_delta < 1e-12:
                continue
            delta = rng.random() * max_delta
            # cap new a[j] to 1000
            if a[j] + delta > 1000.0:
                delta = 1000.0 - a[j]
                if delta <= 0:
                    continue
            a[i] -= delta
            a[j] += delta
            new_val = _max_convolution_fft(a)
            dE = new_val - cur_val
            if dE <= 0 or rng.random() < np.exp(-dE / (T + 1e-12)):
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = a.copy()
            else:
                # revert
                a[i] += delta

"""
Hybrid simulated‑annealing + projected sub‑gradient optimizer.

1. Build a diverse pool of seed vectors for a chosen length n
   (power‑law, exponential, Dirichlet, uniform, previous best).
   All seeds are projected onto the simplex
   Σ a_i = √(2n) (the normalization makes the evaluation equal to
   the maximal convolution entry).

2. Pick the seed with the smallest max‑convolution value and run a
   temperature‑controlled annealing that moves mass between two
   positions or swaps two entries.  The Metropolis rule allows uphill
   moves early on and gradually cools down.

3. Try a few deterministic reorderings (alternating large‑small)
   which are known to reduce the autocorrelation peak.

4. Finish with a short projected sub‑gradient descent on the exact
   peak to polish the result.

The whole search respects the 1000 s wall‑time budget and stores the
best sequence in the global variable `height_sequence_1`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σ x = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.ifft(fa * fa).real[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a * a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Exact max‑convolution gradient (used by sub‑gradient descent)
# ----------------------------------------------------------------------


def _max_conv_and_subgrad(a: np.ndarray):
    """Return (conv, max_val, k_max, subgrad) where subgrad is the
    gradient of the peak entry of a*a with respect to a."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    kmax = int(np.argmax(conv))
    max_val = conv[kmax]

    grad = np.zeros_like(a)
    lo = max(0, kmax - (n - 1))
    hi = min(kmax, n - 1)
    idx = np.arange(lo, hi + 1)
    partner = a[kmax - idx]
    grad[lo:hi + 1] = 2.0 * partner
    return conv, max_val, kmax, grad


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution peak
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Momentum projected sub‑gradient descent on max(conv(a,a))."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_val = _max_convolution_fft(a)

    rng = np.random.default_rng()
    it = 0
    momentum = np.zeros_like(a)

    while time.time() < deadline:
        _, _, _, grad = _max_conv_and_subgrad(a)

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_factor * target_sum / norm
            a -= lr * momentum
        else:
            a -= step_factor * target_sum * grad / (np.linalg.norm(grad) + 1e-12)

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] Sub‑GD it={it:5d}  best={best_val:.6f}")

        it += 1

    return best, best_val


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) shape, renormalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """exp(-lam*i) shape, renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Deterministic alternating reorderings
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → left, 2nd largest → right, 3rd → next left, …."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


# ----------------------------------------------------------------------
# Simulated annealing on mass transfers and swaps
# ----------------------------------------------------------------------


def _annealing_optimize(a0: np.ndarray,
                        deadline: float,
                        target_sum: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simulated annealing that moves mass between two positions or swaps
    two entries.  The Metropolis rule uses a temperature that decays
    exponentially from target_sum to 1e‑4·target_sum.
    """
    a = a0.copy()
    cur_val = _max_convolution_fft(a)
    best = a.copy()
    best_val = cur_val

    start = time.time()
    T0 = target_sum               # initial temperature
    T_end = target_sum * 1e-4    # final temperature
    n = a.shape[0]

    while time.time() < deadline:
        elapsed = time.time() - start
        total = deadline - start
        tau = elapsed / total if total > 0 else 0.0
        T = T0 * (T_end / T0) ** tau
        # choose move type
        if rng.random() < 0.30:          # swap move
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            new_val = _max_convolution_fft(a)
            dE = new_val - cur_val
            if dE <= 0 or rng.random() < np.exp(-dE / (T + 1e-12)):
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = a.copy()
            else:
                a[i], a[j] = a[j], a[i]   # revert
        else:                               # mass‑transfer move
            i = rng.integers(0, n)
            if a[i] <= 0.0:
                continue
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            # maximal transferable fraction (shrinks with temperature)
            max_frac = max(0.01, T / T0)   # start ≈1, end ≈0.01
            max_delta = a[i] * max_frac
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta
            # keep under the hard 1000 bound
            if a[j] + delta > 1000.0:
                delta = 1000.0 - a[j]
                if delta <= 0.0:
                    continue
            a[i] -= delta
            a[j] += delta
            new_val = _max_convolution_fft(a)
            dE = new_val - cur_val
            if dE <= 0 or rng.random() < np.exp(-dE / (T + 1e-12)):
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = a.copy()
            else:
                a[i] += delta
                a[j] -= delta

    return best, best_val

        print(f"[{time.time():.2f}] Final evaluated score = {final_score:.6f}")
    except Exception:
        pass

    # make the sequence available for the next iteration
    globals()["height_sequence_1"] = final_seq
    return final_seq
