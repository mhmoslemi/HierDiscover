"""Hybrid optimizer for the max‑convolution benchmark.

The algorithm has three layers:

1. **LP based partner improvement** – a small problem (≈1500 points) is
   repeatedly solved as a linear program:
   minimise the worst‑case value of `f*g` while keeping `∑g` fixed.
   The new partner `g` is mixed into the current `f` (with back‑tracking)
   which monotonically reduces the self‑convolution peak.

2. **Upscaling & ordering** – the best low‑dimensional shape is interpolated
   to a larger size (≈12 k–18 k points) and rearranged by several
   deterministic “spread’’ permutations (alternating, coprime‑step).  The
   best ordering is kept.

3. **Fine polishing** – a projected sub‑gradient descent (exact max‑convolution)
   runs until the time budget expires.

All steps respect the non‑negativity and ℓ¹‑simplex constraints, and the
final sequence is trimmed of trailing zeroes before being returned.
"""

import time
import math
import numpy as np
from typing import List

# optional, but SciPy is available in the competition environment
import scipy.sparse
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Convolution helpers (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Simple factories
# ----------------------------------------------------------------------
def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _spread_order(seq: np.ndarray, step: int) -> np.ndarray:
    """Place values on a cyclic lattice with step `step` (must be coprime to n)."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Random permutation search (fast heuristic)
# ----------------------------------------------------------------------
def _permutation_search(base_seq: np.ndarray, trials: int, deadline: float) -> np.ndarray:
    """Try a number of random reorderings of `base_seq` and keep the best."""
    best_seq = base_seq.copy()
    best_val = _max_convolution_fft(best_seq)
    rng = np.random.default_rng()
    n = base_seq.shape[0]

    for _ in range(trials):
        if time.time() >= deadline:
            break
        perm = rng.permutation(base_seq)
        val = _max_convolution_fft(perm)
        if val < best_val:
            best_val = val
            best_seq = perm.copy()
    return best_seq


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution (fine polishing)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.4,
                         gamma: float = 0.99) -> np.ndarray:
    """Projected sub‑gradient descent with momentum."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_val = _max_convolution_fft(best)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]

        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 600 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur = _max_convolution_fft(a)
        if cur < best_val:
            best_val = cur
            best = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} best={best_val:.6f}")

        it += 1

    return best


# ----------------------------------------------------------------------
# Linear‑programming partner improvement (small scale)
# ----------------------------------------------------------------------
def _build_A_ub(f: np.ndarray) -> scipy.sparse.csr_matrix:
    """
    Build the matrix A such that (A @ g)[k] = (f * g)[k] for linear convolution.
    Returns A with an extra column (last) containing –1 for the variable M.
    Shape: (2n‑1, n+1)
    """
    n = f.shape[0]
    rows = 2 * n - 1

    # Toeplitz‑like lower‑triangular part: constant diagonals equal to f[d]
    offsets = np.arange(n)                     # 0 … n‑1
    data = np.vstack([np.full(n, v) for v in f])  # shape (n, n)
    A_f = scipy.sparse.diags(data, offsets, shape=(rows, n), format='csr')

    # column for M (coefficient –1 on every row)
    Mcol = scipy.sparse.csr_matrix((-np.ones(rows), (np.arange(rows), np.zeros(rows))),
                                   shape=(rows, 1))

    A = scipy.sparse.hstack([A_f, Mcol], format='csr')
    return A


def _lp_refine(f: np.ndarray, target_sum: float, deadline: float,
               max_iters: int = 6) -> np.ndarray:
    """
    Repeatedly solve a small LP:
        minimise   M
        subject to f * g <= M   (all shifts)
                  Σ g = target_sum
                  g ≥ 0 , M ≥ 0

    The resulting g is mixed into f with a back‑tracking line‑search.
    Returns the improved sequence (same length as f).
    """
    n = f.shape[0]
    best_f = f.copy()
    best_max = _max_convolution_fft(best_f)
    it = 0

    while time.time() < deadline - 2.0 and it < max_iters:
        A_ub = _build_A_ub(f)                     # (2n‑1) × (n+1)
        b_ub = np.zeros(2 * n - 1)

        c = np.concatenate([np.zeros(n), [1.0]])   # minimise M

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target_sum])

        bounds = [(0, None)] * n + [(0, None)]

        # give the LP a few seconds at most – it is very small
        linprog_time = max(0.0, deadline - time.time() - 0.5)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': linprog_time})

        if not res.success:
            print(f"[{time.time():.2f}] LP iteration {it} failed ({res.message})")
            break

        g = res.x[:n]
        # ensure exact simplex (numerical error)
        g = _project_to_simplex(g, target_sum)

        # try to move from f towards g – back‑track on the step size
        improved = False
        t = 0.5
        while t > 1e-3:
            cand = (1 - t) * f + t * g
            cand_max = _max_convolution_fft(cand)
            if cand_max < best_max - 1e-9:
                f = cand
                best_f = cand.copy()
                best_max = cand_max
                improved = True
                print(f"[{time.time():.2f}] LP it={it}   t={t:.3f}   max={best_max:.6f}")
                break
            t *= 0.5

        if not improved:
            print(f"[{time.time():.2f}] LP it={it} – no improvement")
            break

        it += 1
    return best_f

# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine follows the three‑phase schedule described in the module
    docstring and never exceeds `budget_s` seconds.
    """
    start = time.time()
    deadline = start + budget_s - 0.3               # small safety margin
    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Phase 0 – choose a modest size for the LP stage
    # ------------------------------------------------------------------
    n_small = 1500                                 # fits comfortably into an LP
    target_small = math.sqrt(2.0 * n_small)

    # initialise from previous run if available, otherwise from a power‑law
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        f0 = _resize_sequence(prev, n_small)
    else:
        # simple decreasing power law as a generic seed
        f0 = (np.arange(1, n_small + 1, dtype=np.float64) ** -0.8)
    f0 = _project_to_simplex(f0, target_small)

    # ------------------------------------------------------------------
    # Phase 1 – LP based refinement on the small scale
    # ------------------------------------------------------------------
    lp_deadline = min(deadline, start + 0.25 * budget_s)   # spend at most 25 % time here
    f_small = _lp_refine(f0, target_small, lp_deadline, max_iters=8)

    # ------------------------------------------------------------------
    # Phase 2 – upscale to a larger problem size
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 200.0:
        n_big = 18000
    elif time_left > 120.0:
        n_big = 15000
    elif time_left > 80.0:
        n_big = 12000
    else:
        n_big = n_small                     # fall back to the small size

    target_big = math.sqrt(2.0 * n_big)
    f_big = _resize_sequence(f_small, n_big)
    f_big = _project_to_simplex(f_big, target_big)

    # deterministic reorderings (alternating + a few coprime‑step spreads)
    candidates = [f_big,
                  _alternating_order(f_big)]

    # a few coprime steps around sqrt(n) and random ones
    steps = set()
    steps.add(max(1, int(round(math.sqrt(n_big)))))
    steps.add(max(1, int(round(n_big / 3))))
    steps.add(max(1, int(round(n_big / 2))))
    for _ in range(8):
        s = rng.integers(1, n_big)
        while math.gcd(s, n_big) != 1:
            s = (s + 1) % n_big or 1
        steps.add(s)

    for s in steps:
        cand = _spread_order(f_big, s)
        candidates.append(cand)

    # keep the ordering with the smallest max‑convolution
    f_big = min(candidates, key=_max_convolution_fft)

    # ------------------------------------------------------------------
    # Phase 3 – random permutation search (fast heuristic)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        perm_deadline = time.time() + min(8.0, deadline - time.time() - 5.0)
        trials = 120
        f_big = _permutation_search(f_big, trials, perm_deadline)

    # ------------------------------------------------------------------
    # Phase 4 – final sub‑gradient polishing on the large scale
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        sub_deadline = deadline - 0.2
        f_big = _subgradient_descent(f_big, sub_deadline, target_big,
                                     step_factor=0.45, gamma=0.995)

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries and return
    # ------------------------------------------------------------------
    arr = np.array(f_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # store globally for possible warm‑starts in a next run
    globals()["height_sequence_1"] = final_seq

    # optional logging of the final score (does not affect the result)
    try:
        score = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {score:.6f}")
    except Exception:
        pass

    return final_seq
