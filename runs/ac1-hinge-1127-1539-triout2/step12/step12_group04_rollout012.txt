def propose_candidate(budget_s=1000):

    deadline = start + budget_s - 0.5

    # small problem
    n_small = 2000

    # initial shape via p grid quick
    best_small = None
    best_score_small = np.inf
    for p in [0.5, 0.6, 0.7]:
        seq = _power_law_sequence(n_small, p, target_small)
        val = _score(seq)
        if val < best_score_small:
            best_score_small = val
            best_small = seq.copy()
    # LP refinement loop
    while time.time() < deadline - 10:   # leave time for later phases
        # Solve LP to get x
        try:
            x, M_val = _lp_refine(best_small, target_small, deadline)
        except Exception as e:
            print("LP failed:", e)
            break
        # Evaluate mixing
        best_mix = best_small
        best_mix_score = best_score_small
        for t in [0.25, 0.5, 0.75]:
            a_new = (1 - t) * best_small + t * x
            a_new = _project_to_simplex(a_new, target_small)
            val = _score(a_new)
            if val < best_mix_score:
                best_mix = a_new.copy()
                best_mix_score = val
        if best_mix_score < best_score_small - 1e-6:
            best_small = best_mix
            best_score_small = best_mix_score
            print(f"LP mixing improved small score = {best_mix_score:.6f}")
        else:
            break
    # upscale to large n
    n_large = 12000 if budget_s >= 800 else 8000 if budget_s >= 400 else 5000
    target_large = np.sqrt(2.0 * n_large)
    # interpolate
    old_x = np.linspace(0.0, 1.0, best_small.shape[0])

def _lp_refine(a_old, target_sum, deadline):
    n = a_old.shape[0]
    # build A
    rows = 2*n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = a_old
    x = cp.Variable(n)
    M = cp.Variable()
    constraints = [x >= 0,
                   cp.sum(x) == target_sum,
                   A @ x <= M]
    prob = cp.Problem(cp.Minimize(M), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, verbose=False)
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        raise ValueError("LP not solved")
    x_val = np.array(x.value).flatten()
    M_val = float(M.value)
    # clip tiny negatives
    x_val = np.maximum(x_val, 0.0)
    return x_val, M_val
```

"""
Propose a step‑function with a low self‑convolution maximum.

Algorithm
---------

1. **Small‑scale LP refinement** – work on a modest size (≈2000) where a
   linear program can be solved quickly:
   * variables `x_i ≥ 0`, `Σ x_i = √(2 n)`.
   * constraints ` (a_old * x)_k ≤ M ` for every shift `k`.
   * objective: minimise `M`.
   The solution concentrates mass where the current coefficients are
   small, which tends to reduce the convolution peak.

2. **Mix & project** – convexly mix the LP solution with the current
   sequence, re‑project onto the simplex and keep the best mixture.

3. **Upsample** – linearly interpolate the improved small sequence to the
   large problem size (≈12000) and re‑normalise.

4. **Gradient‑descent polishing** – Adam‑style projected GD on a soft‑max
   surrogate of the max‑convolution.

5. **Exact deterministic mass‑transfer** – the classic DMT2 routine.

6. **Global random moves** – occasional stochastic mass‑transfer.

7. **Pairwise optimal transfer** – greedy donor/receiver moves that
   analytically maximise the reduction of the current peak.

8. **Final Adam polishing** – high‑temperature soft‑max Adam run.

The whole pipeline respects the 1000 s time budget and returns the
best sequence found as a plain Python list.
"""

import time
import math
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective value 2·n·max(b)/(sum(a)^2).  For Σa=√(2n) this equals max(b)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex Σx = s,  x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (used for the initial guess)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam‑GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=2e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (the previous DMT2)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest value (ignore current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = int(idx_max[0])
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k-i}

        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # pick a receiver with minimal partner value (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.7, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Pairwise optimal mass‑transfer (new refinement)
# ----------------------------------------------------------------------


def _pairwise_optimal_transfer(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               donor_frac: float = 0.015,
                               receiver_frac: float = 0.015,
                               min_improve: float = 1e-7) -> tuple[np.ndarray, float]:
    """Greedy pairwise mass transfer."""
    a = a0.copy()
    n = a.shape[0]
    best_score = _score(a)

    pad_len = n - 1

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k = int(np.argmax(conv))

        nd = max(1, int(donor_frac * n))
        nr = max(1, int(receiver_frac * n))
        donor_idx = np.argsort(-a)[:nd]          # largest coefficients
        receiver_idx = np.argsort(a)[:nr]        # smallest coefficients

        base = np.concatenate([np.zeros(pad_len, dtype=np.float64),
                               a,
                               np.zeros(pad_len, dtype=np.float64)])

        shifted = {}
        for i in np.unique(np.concatenate((donor_idx, receiver_idx))):
            shifted[i] = base[i:i + 2 * n - 1]     # view of length 2n‑1

        best_new_max = max_val
        best_pair = None
        best_delta = 0.0

        for i in donor_idx:
            shift_i = shifted[i]
            a_i = a[i]
            if a_i <= 0.0:
                continue
            for j in receiver_idx:
                if i == j:
                    continue
                shift_j = shifted[j]

                d = 2.0 * (shift_j - shift_i)   # effect of moving mass from i to j
                d_k = d[k]
                if d_k >= 0.0:
                    continue

                mask = d > d_k
                if np.any(mask):
                    denom = d[mask] - d_k
                    numer = max_val - conv[mask]
                    feasible = numer / denom
                    delta_max = min(feasible.min(), a_i)
                else:
                    delta_max = a_i

                if delta_max <= 1e-12:
                    continue

                new_max = max_val + d_k * delta_max
                if new_max < best_new_max - min_improve:
                    best_new_max = new_max
                    best_pair = (i, j)
                    best_delta = delta_max

        if best_pair is None:
            break

        i, j = best_pair
        a[i] -= best_delta
        a[j] += best_delta

        if int(time.time() * 10) % 50 == 0:
            a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            print(f"[{time.time():.2f}] Pairwise i={i} j={j} Δ={best_delta:.3e} score={best_score:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Global random mass‑transfer (exploration phase)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear programme for a small‑scale refinement
# ----------------------------------------------------------------------


def _lp_refine(a_old: np.ndarray, target_sum: float, deadline: float) -> tuple[np.ndarray, float]:
    """
    Solve
        minimise   M
        subject to (a_old * x)_k ≤ M  for all k,
                   Σ x_i = target_sum,
                   x_i ≥ 0.
    Returns the optimal x and the minimised M.
    """
    n = a_old.shape[0]
    rows = 2 * n - 1

    # Build the full convolution matrix A where A[k,i] = a_old[k-i] (zero if out‑of‑range)
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = a_old

    x = cp.Variable(n)
    M = cp.Variable()

    constraints = [
        x >= 0,
        cp.sum(x) == target_sum,
        A @ x <= M
    ]

    prob = cp.Problem(cp.Minimize(M), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, verbose=False)
    except (cp.error.SolverError, cp.error.DCPError):
        prob.solve(solver=cp.SCS, max_iters=3000, verbose=False)

    if prob.status not in ("optimal", "optimal_inaccurate"):
        raise RuntimeError(f"LP not solved (status: {prob.status})")

    x_val = np.maximum(np.array(x.value).flatten(), 0.0)
    M_val = float(M.value)

    # tiny re‑projection to keep the sum exactly correct (numerical drift)
    x_val = _project_to_simplex(x_val, target_sum)
    return x_val, M_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """Search for a non‑negative step‑function that minimises the max self‑convolution."""

    # ------------------------------------------------------------------
    n_small = 2000

        seq = _power_law_sequence(n_small, float(p), target_small)
        val = _score(seq)
        if val < best_small_score:
            best_small_score = val
            best_small = seq.copy()
    print(f"[{time.time():.2f}] Small start score = {best_small_score:.6f}")

    # LP‑based refinement loop (short)
    while time.time() < deadline - 30.0:      # keep ~30 s for the large phases
        try:
            x_lp, M_lp = _lp_refine(best_small, target_small, deadline)
        except Exception as e:
            print("LP failed:", e)
            break

        # test a few convex mixtures of the old and new vectors
        improved = False
        for t in (0.25, 0.5, 0.75):
            cand = (1.0 - t) * best_small + t * x_lp
            cand = _project_to_simplex(cand, target_small)
            val = _score(cand)
            if val < best_small_score - 1e-8:
                best_small_score = val
                best_small = cand.copy()
                improved = True
                print(f"[{time.time():.2f}] LP mixture t={t:.2f} improved small score = {best_small_score:.6f}")
        if not improved:

    # ------------------------------------------------------------------
    # Phase 1 – Upsample to the target problem size
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n_large = 12000

        n_large = 8000
    else:

    while time.time() < deadline - 12.0:

            best_large_score = val_pair

    # Phase 5 – Final high‑temperature Adam polishing

        a_fin, val_fin = _adam_gradient_descent(best_large,
                                                deadline,
                                                target_large,
                                                max_beta=5e5,
                                                step_factor=0.6,
                                                gamma=0.995)

            best_large_score = val_fin
            best_large = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish improved score = {best_large_score:.6f}")

    # ------------------------------------------------------------------
    arr = np.array(best_large, dtype=float)

    except Exception:
        pass

    return final_seq
