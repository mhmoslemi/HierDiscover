"""
Propose a low‑max‑convolution step sequence.

The algorithm works in three stages inside the given time budget:
1. Warm‑start – optionally resize the best sequence found so far.
2. Fast local optimisation – a short mirror‑descent (soft‑max) pass.
3. Intensive simulated‑annealing mass‑transfer refinement.
4. Random‑restart loop – generate many simple power‑law / exponential
   initial shapes, run a short anneal on each and keep the best.
All steps respect the global deadline (default 1000 s).  The final
sequence is returned as a plain list of floats; trailing near‑zeros are
removed before returning.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution a*a using an FFT of power‑of‑2 size.
    Returns an array of length 2*len(a)-1.
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Objective  2·n·max(a*a) / (sum(a)²).
    For Σa = √(2 n) this equals max(a*a); the function is scale‑invariant.
    """
    n = a.shape[0]
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of the max‑convolution.
    Returns (grad, full_convolution) where grad has the same length as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    grad_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(grad_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – exponential (mirror) descent on soft‑max surrogate
# ----------------------------------------------------------------------


def exponential_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Mirror descent on the soft‑max surrogate."""
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        # occasional jitter
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpDescent it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – sub‑gradient descent on true max‑convolution
# ----------------------------------------------------------------------


def subgradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Sub‑gradient descent on the true max‑convolution entry."""
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0
    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        elapsed = time.time() - start
        total = deadline - start
        eta_start = 2.0
        eta_end = 0.2
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase D – improved simulated‑annealing mass‑transfer refinement
# ----------------------------------------------------------------------


def anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float,
                         max_delta_factor: float = 0.20,
                         total_delta_factor: float = 0.01):
    """
    Simulated‑annealing that moves a small amount of mass from a “donor’’
    (contributing most to the current peak) to a “receiver’’ (low height).
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.80  # start temperature as a fraction of current score

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))  # √‑schedule

        # locate current peak
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contributions of each index to the peak
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – pick an index with relatively low height
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # size of the transfer (scaled by temperature)
        max_delta = min(a[donor] * max_delta_factor,
                        target_sum * total_delta_factor)
        if max_delta <= 0:
            continue

        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # --- O(n) update of the convolution -----------------------------
        b_new = b.copy()
        # linear part: 2·a·Δ (removing from donor, adding to receiver)
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        # Metropolis acceptance based on *current* energy
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved best={best_val:.6f}")

        # periodic renormalisation to combat drift
        if it % 5000 == 0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val

    deadline = start + budget_s - 0.5   # safety margin

    # choose a problem size (larger sizes give finer granularity)
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Warm‑start (if any)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init_warm = _resize_sequence(np.array(globals()["height_sequence_1"],
                                             dtype=np.float64), n)
        init_warm = init_warm * (target_sum / init_warm.sum())
        warm_starts = [init_warm]
    else:
        warm_starts = []

    # ------------------------------------------------------------------
    # Prepare a handful of deterministic initial shapes
    # ------------------------------------------------------------------
    init_shapes = []

    # power‑law families
    for p in [0.45, 0.55, 0.65, 0.75]:
        seq = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        init_shapes.append(seq)

    # exponential decay families
    for rate in [0.001, 0.002, 0.004, 0.008]:
        seq = np.exp(-rate * np.arange(n, dtype=np.float64))
        init_shapes.append(seq)

    # linear drop to zero
    seq = np.maximum(0.0, 1.0 - np.arange(n, dtype=np.float64) / (n * 1.2))
    init_shapes.append(seq)

    # random quadratic‑biased vectors (more mass at low indices)
    rng = np.random.default_rng()
    for _ in range(2):
        seq = rng.random(n) ** 2
        init_shapes.append(seq)

    # combine warm starts and deterministic ones
    candidates = warm_starts + init_shapes

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # First pass: try each deterministic candidate with a short
    # optimisation (mirror descent → sub‑gradient → anneal).
    # ------------------------------------------------------------------
    for a0 in candidates:
        if time.time() >= deadline:
            break

        a0 = a0.astype(np.float64)
        if a0.sum() == 0:
            continue
        a0 = a0 * (target_sum / a0.sum())

        # Phase A – mirror descent (≈5 % of remaining time)
        remaining = deadline - time.time()
        if remaining <= 2.0:
            break
        t_exp = min(remaining * 0.05, 30.0)
        a_mid, _ = exponential_descent(a0, target_sum, time.time() + t_exp)

        # Phase B – sub‑gradient descent (≈5 % of remaining time)
        remaining = deadline - time.time()
        if remaining <= 2.0:
            break
        t_sub = min(remaining * 0.05, 30.0)
        a_mid, _ = subgradient_descent(a_mid, target_sum,
                                       time.time() + t_sub)

        # Phase D – annealing (use whatever time remains for this candidate)
        if time.time() < deadline:
            a_opt, val_opt = anneal_mass_transfer(a_mid, target_sum, deadline)

            if val_opt < best_score:
                best_score = val_opt
                best_seq = a_opt.copy()
                print(f"[{time.time():.2f}] New global best (deterministic) "
                      f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Random‑restart loop: generate many simple shapes on the fly,
    # run a relatively cheap anneal on each, keep the best.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        time_left = deadline - time.time()
        if time_left < 10.0:
            break  # not enough time for another full attempt

        # Randomly pick a family and a parameter
        if rng.random() < 0.6:                     # power‑law
            p = rng.uniform(0.35, 0.85)
            a0 = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        else:                                      # exponential
            rate = rng.uniform(0.0005, 0.0100)
            a0 = np.exp(-rate * np.arange(n, dtype=np.float64))

        a0 = a0 * (target_sum / a0.sum())

        # Very short mirror‑descent to give a quick push
        t_exp = min(time_left * 0.03, 15.0)
        a_mid, _ = exponential_descent(a0, target_sum,
                                       time.time() + t_exp)

        # Anneal for the rest of the allocated slice (≈60 % of remaining)
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        t_ann = min(remaining * 0.60, remaining - 2.0)
        a_opt, val_opt = anneal_mass_transfer(a_mid, target_sum,
                                              time.time() + t_ann)

        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] New global best (random) "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – in the unlikely event we never produced a sequence
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, target_sum / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Final clean‑up: cut trailing zeros, store for warm‑starts,
    # evaluate with the provided security‑checked function.
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    globals()["height_sequence_1"] = final_seq

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:  # safety – should never happen
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    return final_seq
