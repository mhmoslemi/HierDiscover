def _solve_lp_for_g(f, M, target_sum):
    n = f.shape[0]
    # Build A_ub: shape (2*n-1, n)
    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n-1)
        for j in range(j_start, j_end+1):
            i = k - j
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    A_ub = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n - 1, M)
    # objective to maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    # bounds (0, None)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if not res.success:
        # If fails, maybe fallback to random.
        g0 = np.zeros(n)
    else:
        g0 = res.x
    return g0
```

"""
Hybrid optimizer with early LP‐based shape refinement.

The search runs in several stages:

1. **Small‑scale LP refinement** – work on a modest length (≈1200).
   For the current sequence f we compute M = max(f∗f) and solve the
   linear program:
       maximise   Σ g_i
       subject to f∗g ≤ M   (elementwise) ,   g_i ≥ 0 .
   The solution g₀ is scaled to the required ℓ¹‑norm and mixed with f
   (convex combination).  Repeating a few times flattens the
   self‑convolution peak.

2. **Upscaling** – the improved small sequence is linearly interpolated
   to a larger size (≈12–18 k) and projected back onto the ℓ¹‑simplex.
   A few deterministic reorderings (alternating, even‑spread) are tried
   and the best ordering is kept.

3. **Permutation search** – a short random shuffle search further
   reduces the max‑convolution.

4. **Projected sub‑gradient descent** – exact sub‑gradients of
   max convolution are applied with momentum to polish the solution.

5. **Greedy mass‑transfer with simulated annealing** – a final
   simulated‑annealing pass moves mass from entries that contribute most
   to the current peak toward entries that contribute little.

All steps respect non‑negativity and the ℓ¹‑simplex constraint.
The routine stops before the time budget expires and returns the best
sequence found.
"""

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (linear convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray, target_sum: float) -> float:
    """Objective 2·n·max(a*a)/(∑a)² (for ∑a=target_sum this equals max conv)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised to the required ℓ¹‑norm."""
    a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """
    Spread the large values uniformly by placing them on a cyclic sequence
    with a step coprime to the length (≈√n).
    """
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# LP sub‑routine (max‑sum g s.t. f*g ≤ M)
# ----------------------------------------------------------------------


def _solve_lp_for_g(f: np.ndarray, M: float) -> np.ndarray:
    """Solve maximise Σ g_i  subject to f∗g ≤ M  and g≥0."""
    n = f.shape[0]
    rows = []
    cols = []
    data = []

    # Build sparse convolution matrix A where (A g)[k] = (f∗g)[k]
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        # indices j in [j_start, j_end]
        for j in range(j_start, j_end + 1):
            i = k - j
            rows.append(k)
            cols.append(j)
            data.append(f[i])

    A_ub = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise Σ g_i  → minimise -Σ g_i
    c = -np.ones(n, dtype=np.float64)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if not res.success:
        # fallback: return zero vector
        return np.zeros(n, dtype=np.float64)
    return res.x

# Random permutation search (fast heuristic)
# ----------------------------------------------------------------------


def _permutation_search(base_seq: np.ndarray,
                        trials: int,
                        deadline: float) -> tuple[np.ndarray, float]:
    """
    Try a number of random reorderings of `base_seq` (keeping the multiset
    of values fixed) and return the ordering with smallest max‑convolution.
    """
    best_seq = base_seq.copy()
    best_val = _max_convolution_fft(best_seq)
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        perm = rng.permutation(base_seq)
        val = _max_convolution_fft(perm)
        if val < best_val:
            best_val = val
            best_seq = perm.copy()
    return best_seq, best_val

# Sub‑gradient descent on the exact max‑convolution

def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.45,
                         gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that uses the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a, target_sum)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        # indices attaining the maximum (within tolerance)
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]

        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a, target_sum)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val

# Greedy mass‑transfer with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.2,
                                      end_temp: float = 1e-5,
                                      low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy mass moves from a “donor’’ (large contribution to the peak)
    to a “receiver’’ (small contribution) with a Metropolis acceptance rule.
    """
    a = a0.copy()
    cur_score = _score(a, target_sum)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]
    conv = _conv_fft(a)

    while time.time() < deadline:
        # temperature schedule (exponential decay)
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # locate current peak
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a, target_sum)

        # Metropolis acceptance
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a, target_sum)

    return best_a, best_score

    The routine follows a 5‑stage schedule:
      1. LP‑based shape refinement on a modest size.
      2. Upscaling + deterministic reordering.
      3. Random permutation search.
      4. Projected sub‑gradient descent.
      5. Final greedy mass‑transfer (simulated annealing).
    The total wall‑clock time stays within `budget_s` seconds.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin
    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # Stage 1 – LP refinement on a modest length
    # --------------------------------------------------------------
    n_small = 1200
    target_small = math.sqrt(2.0 * n_small)

    # Initialise from previous best if available, otherwise from a power law
    if "height_sequence_1" in globals():
        seq0 = np.array(globals()["height_sequence_1"], dtype=float)
        seq_small = _resize_sequence(seq0, n_small)
    else:
        seq_small = _power_law_sequence(n_small, p=0.8, target_sum=target_small)

    seq_small = _project_to_simplex(seq_small, target_small)

    max_lp_iters = 8
    t_mix = 0.4
    for it in range(max_lp_iters):
        if time.time() >= deadline - 30.0:
            break
        M = _max_convolution_fft(seq_small)
        g0 = _solve_lp_for_g(seq_small, M)
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            break
        g = g0 * (target_small / sum_g0)      # rescale to required ℓ¹‑norm
        seq_small = (1.0 - t_mix) * seq_small + t_mix * g
        seq_small = _project_to_simplex(seq_small, target_small)
        t_mix *= 0.8
        new_M = _max_convolution_fft(seq_small)
        print(f"[LP it {it}] max={new_M:.6f}, mix={t_mix:.3f}")
        if new_M >= M - 1e-6:
            # stagnation – continue a couple more iterations anyway
            pass

    best_small = seq_small
    best_M_small = _max_convolution_fft(best_small)
    print(f"[LP finished] best small max={best_M_small:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – Upscaling to a larger problem size
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 200.0:
        n_final = 18000
    elif remaining > 80.0:
        n_final = 15000
    else:
        n_final = 12000

    target_final = math.sqrt(2.0 * n_final)
    seq_big = _resize_sequence(best_small, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # Try deterministic reorderings and keep the best
    candidates = [
        seq_big,
        _alternating_order(seq_big),
        _even_spread_order(seq_big)
    ]
    seq_big = min(candidates, key=_max_convolution_fft)
    print(f"[Upscale] n={n_final}, initial max={_max_convolution_fft(seq_big):.6f}")

    # --------------------------------------------------------------
    # Stage 3 – Random permutation search
    # --------------------------------------------------------------
    if time.time() < deadline - 10.0:
        perm_deadline = time.time() + min(8.0, deadline - time.time() - 5.0)
        seq_big, _ = _permutation_search(seq_big, trials=40, deadline=perm_deadline)
        print(f"[Permutation] post‑perm max={_max_convolution_fft(seq_big):.6f}")

    # --------------------------------------------------------------
    # Stage 4 – Sub‑gradient refinement on the large scale
    # --------------------------------------------------------------
    if time.time() < deadline - 6.0:
        sub_deadline = deadline - 4.0
        seq_big, sub_val = _subgradient_descent(seq_big, sub_deadline,
                                                target_final,
                                                step_factor=0.4,
                                                gamma=0.995)
        print(f"[SubGrad] best max={_max_convolution_fft(seq_big):.6f}")

    # --------------------------------------------------------------
    # Stage 5 – Final greedy mass‑transfer (simulated annealing)
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        mt_deadline = deadline - 1.0
        seq_big, mt_val = _mass_transfer_simulated_annealing(seq_big,
                                                             mt_deadline,
                                                             target_final,
                                                             start_temp=0.2,
                                                             end_temp=1e-5,
                                                             low_frac=0.05)
        print(f"[Final MT] best max={_max_convolution_fft(seq_big):.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # ------------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Store for possible future runs and report final evaluation
    # ------------------------------------------------------------------
    globals()["height_sequence_1"] = final_seq
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[final] evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
