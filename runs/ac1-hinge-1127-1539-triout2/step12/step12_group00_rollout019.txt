"""
Hybrid annealed‑gradient / LP / mass‑transfer optimizer.

* Phase 1 – long annealed soft‑max GD on a fairly large problem (≈20 k
  variables).  The loss is a smooth approximation of the max‑convolution.

* Phase 2 – a short deterministic mass‑transfer (greedy “move mass from the
  heaviest contributor to the weakest”) to clean up the current peak.

* Phase 3 – shrink to a moderate size (≈1500) where a linear programme can be
  solved quickly.  Solve the LP   max c·b   s.t.   f∗b ≤ M, b≥0,
  then mix the LP solution with the current vector (damped Frank‑Wolfe step).
  Repeat a few times.

* Phase 4 – upscale back to a large size (≈25 k) and polish with a second
  GD run, a final sub‑gradient descent and one more mass‑transfer pass.

* Phase 5 – reorder the final vector with two simple heuristics (alternating
  and “even‑spread”) and keep the best ordering.

All vectors are projected onto the ℓ¹‑simplex with sum √(2 n) after each
update, guaranteeing that the evaluation function reduces to the peak
convolution value.
"""

import time
import numpy as np

# -------------------------------------------------------------
#  FFT based convolution utilities
# -------------------------------------------------------------

def _max_convolution_fft(a: np.ndarray) -> float:
    """Return max entry of a*a (linear convolution) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) via FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)² .  With Σa=√(2n) this equals max(conv)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# -------------------------------------------------------------
#  Simplex projection
# -------------------------------------------------------------

def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx=s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
#  Sequence factories
# -------------------------------------------------------------

def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma_factor: float, target_sum: float) -> np.ndarray:
    """Symmetric discrete Gaussian centred in the middle of the vector."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# -------------------------------------------------------------
#  Reordering helpers
# -------------------------------------------------------------

def _alternating_order(seq: np.ndarray) -> np.ndarray:
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# -------------------------------------------------------------
#  Soft‑max gradient (annealed GD)
# -------------------------------------------------------------

def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the smooth max over a*a."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.7,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected momentum GD on the smooth‑max approximation."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape tiny plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} best={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best, best_val

def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.995) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_val = _score(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        # indices attaining the maximum (within a small tolerance)
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]
        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.0004, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur = _score(a)
        if cur < best_val:
            best_val = cur
            best = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} best={best_val:.6f}")

        it += 1

    return best, best_val

def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-5,
                                low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _score(a)
    best = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        # most offending convolution index
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            it += 1
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            # rollback
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2500 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best, best_score

def _lp_optimize(f: np.ndarray, M: float, target_sum: float,
                 rng: np.random.Generator) -> np.ndarray | None:
    """Solve   max c·b   s.t.   f∗b ≤ M,   b ≥0   and rescale to ℓ¹‑norm."""
    n = f.shape[0]
    rows = []
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=np.int64)
        row = np.zeros(n, dtype=np.float64)
        row[i] = f[k - i]
        rows.append(row)
    A = np.vstack(rows)                     # (2n‑1) × n
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # random objective – bias exploration
    c = -rng.random(n)
    bounds = [(0.0, None)] * n

    try:
        from scipy.optimize import linprog
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception as e:                   # pragma: no cover
        print(f"[{time.time():.2f}] LP failure: {e}")
        return None

    if not res.success:
        return None

    b = res.x
    total = b.sum()
    if total <= 0:
        return None
    b *= target_sum / total
    return b

#  Main entry point

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """Return a non‑negative step‑function (list of heights) with a low score.

    The routine follows the multi‑phase scheme described in the module
    docstring.  It constantly respects the ℓ¹‑simplex constraint
    Σa = √(2 n) and uses only NumPy / SciPy.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # tiny safety margin

    rng = np.random.default_rng()

    # -----------------------------------------------------------------
    # Phase 1 – large‑scale annealed GD
    # -----------------------------------------------------------------
    if budget_s >= 800:
        n_expl = 20000
    elif budget_s >= 400:
        n_expl = 12000
    else:
        n_expl = 8000
    target_expl = np.sqrt(2.0 * n_expl)

    # build a small pool of seeds
    seeds = []

    # power‑law family
    for p in np.linspace(0.2, 1.2, 8):
        seeds.append(_power_law_sequence(n_expl, float(p), target_expl))

    # exponential family (very flat)
    for lam in np.linspace(0.0005, 0.01, 7):
        seeds.append(_exponential_sequence(n_expl, float(lam), target_expl))

    # random Dirichlet‑like points
    for _ in range(8):
        seeds.append(_random_simplex(n_expl, target_expl, rng))

    # a few Gaussian‑shaped seeds (often already very good)
    for sigma_factor in (0.2, 0.35, 0.5):
        seeds.append(_gaussian_sequence(n_expl, sigma_factor, target_expl))

    # incorporate the previous best if it exists
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        seeds.append(_resize_sequence(prev, n_expl))

    # jitter all seeds a little to diversify
    enriched = []
    for seq in seeds:
        jitter = rng.normal(scale=target_expl * 0.02, size=n_expl)
        seq2 = seq + jitter
        seq2 = np.maximum(seq2, 0.0)
        seq2 = _project_to_simplex(seq2, target_expl)
        enriched.append(seq2)
    seeds.extend(enriched)

    best_seq = None
    best_val = np.inf

    # run many short GD runs until we run out of budget
    while time.time() < deadline - 30.0:
        # pick a seed: 60 % from current best (jittered), else random
        if best_seq is not None and rng.random() < 0.6:
            seed = best_seq.copy()
            seed += rng.normal(scale=target_expl * 0.003, size=n_expl)
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(seeds)

        # give each GD run about 15 s (or whatever fits)
        gd_dead = min(time.time() + 15.0, deadline - 20.0)
        if gd_dead <= time.time():
            break

        a_opt, val_opt = _annealed_gradient_descent(
            seed, gd_dead, target_expl,
            max_beta=8e5, step_factor=0.8, gamma=0.985)

        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] New global best after GD: {best_val:.6f}")

    # -----------------------------------------------------------------
    # Phase 2 – deterministic mass‑transfer on the large problem
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 20.0:
        mt_dead = time.time() + 0.15 * (deadline - time.time())
        a_mt, v_mt = _deterministic_mass_transfer(best_seq, mt_dead,
                                                  target_expl,
                                                  start_temp=0.4,
                                                  end_temp=1e-4,
                                                  low_frac=0.06)
        if v_mt < best_val:
            best_val = v_mt
            best_seq = a_mt.copy()
            print(f"[{time.time():.2f}] After mass‑transfer best={best_val:.6f}")

    # -----------------------------------------------------------------
    # Phase 3 – LP alternating refinement on a down‑scaled instance
    # -----------------------------------------------------------------
    if time.time() < deadline - 35.0:
        n_lp = min(1800, max(900, n_expl // 8))
        target_lp = np.sqrt(2.0 * n_lp)
        seq_lp = _resize_sequence(best_seq, n_lp)
        seq_lp = _project_to_simplex(seq_lp, target_lp)

        max_lp_iters = 12
        for it in range(max_lp_iters):
            if time.time() > deadline - 30.0:
                break
            M = _max_convolution_fft(seq_lp)
            b_opt = _lp_optimize(seq_lp, M, target_lp, rng)
            if b_opt is None:
                break
            # damped Frank‑Wolfe step; step size slowly shrinks
            t = 0.25 * (0.9 ** it)
            seq_lp = (1 - t) * seq_lp + t * b_opt
            seq_lp = _project_to_simplex(seq_lp, target_lp)

            cur = _score(seq_lp)
            print(f"[{time.time():.2f}] LP iter {it} score={cur:.6f}")
            if cur < best_val:
                best_val = cur
                best_seq = _resize_sequence(seq_lp, n_expl)  # keep a copy
                print(f"[{time.time():.2f}] LP improved global best={best_val:.6f}")

        # brief GD polishing on the reduced problem
        if time.time() < deadline - 20.0:
            gd_dead = time.time() + min(8.0, deadline - time.time() - 12.0)
            a_opt, val_opt = _annealed_gradient_descent(seq_lp, gd_dead,
                                                        target_lp,
                                                        max_beta=2e5,
                                                        step_factor=0.6,
                                                        gamma=0.99)
            if val_opt < best_val:
                best_val = val_opt
                best_seq = _resize_sequence(a_opt, n_expl)
                print(f"[{time.time():.2f}] GD after LP best={best_val:.6f}")

    # -----------------------------------------------------------------
    # Phase 4 – upscale, sub‑gradient and final mass‑transfer polishing
    # -----------------------------------------------------------------
    # Determine final size based on remaining time
    remaining = deadline - time.time()
    if remaining > 200.0:
        n_final = 26000
    elif remaining > 120.0:
        n_final = 21000
    else:
        n_final = max(8000, n_expl)

    target_final = np.sqrt(2.0 * n_final)
    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # a few reorderings before polishing
    candidates = [seq_big,
                  _alternating_order(seq_big),
                  _even_spread_order(seq_big)]
    seq_big = min(candidates, key=_max_convolution_fft)
    best_score = _score(seq_big)
    print(f"[{time.time():.2f}] Upscaled to {n_final} entries, start score={best_score:.6f}")

    # sub‑gradient descent on the big instance
    if time.time() < deadline - 12.0:
        sub_dead = deadline - 8.0
        a_sub, v_sub = _subgradient_descent(seq_big, sub_dead,
                                            target_final,
                                            step_factor=0.45,
                                            gamma=0.996)
        if v_sub < best_score:
            best_score = v_sub
            seq_big = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑grad polished best={best_score:.6f}")

    # final deterministic mass‑transfer
    if time.time() < deadline - 4.0:
        mt_dead = time.time() + min(6.0, deadline - time.time() - 2.0)
        a_mt, v_mt = _deterministic_mass_transfer(seq_big, mt_dead,
                                                  target_final,
                                                  start_temp=0.2,
                                                  end_temp=1e-6,
                                                  low_frac=0.05)
        if v_mt < best_score:
            best_score = v_mt
            seq_big = a_mt.copy()
            print(f"[{time.time():.2f}] Final mass‑transfer best={best_score:.6f}")

    # -------------------------------------------------------------
    # Trim negligible tail and return the result
    # -------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # store for possible reuse by subsequent runs
    globals()["height_sequence_1"] = final_seq

    # optional logging of the true evaluation (for debugging)
    try:
        val = evaluate_sequence(final_seq)       # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:  # pragma: no cover
        pass

    return final_seq
