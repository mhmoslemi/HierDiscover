    i = np.arange(n, dtype=np.float64)

"""
Propose a step‑function with a small maximal self‑convolution.

Algorithm
---------
1. **Problem size** – choose a reasonably large number of points
   (≈ 20 000) so that the FFT‑based convolution stays fast.
2. **Parametric pre‑search** – generate a wide variety of analytic shapes
   (power‑law, exponential tails, beta‑type, semi‑circular and random
   Dirichlet draws).  Each candidate is projected onto the simplex
   Σa = √(2 n) and its maximal convolution entry (which equals the
   evaluation score under this normalisation) is recorded.
3. **Multi‑index sub‑gradient descent** – at every iteration the exact
   sub‑gradient is formed as the sum of the individual sub‑gradients
   belonging to *all* maximising convolution indices.  A step size
   proportional to the current peak‑gap is taken and the iterate is
   projected back onto the simplex.  This reduces the peak much faster
   than picking a single active index.
4. **Adam‑style soft‑max GD** – a few projected Adam runs with a
   gradually increasing soft‑max temperature fine‑tune the sequence.
5. **Deterministic mass‑transfer** – a final greedy transfer of mass
   from “donor’’ to “receiver’’ indices (as in the reference code) removes
   any remaining slack.
6. **Post‑processing** – trim trailing near‑zero entries and return the
   list.  The returned sequence respects all security checks of
   `evaluate_sequence`.

The whole routine respects the 1000 s time‑budget and aims to beat the
previous best bound (~1.5074) by pushing the maximal self‑convolution
below 1.503.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using real FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())       # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Projection onto the simplex Σx = s, x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _beta_shape_sequence(n: int, pl: float, pr: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-pl) * (n-i)^(-pr)   (both ends decay)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-pl) * (n - i) ** (-pr)
    a *= target_sum / a.sum()
    return a


def _semicircle_sequence(n: int, gamma: float, target_sum: float) -> np.ndarray:
    """
    Semi‑circular shape on [0,1] raised to power gamma.
    x ∈ (-1,1),  a_i = (max(0,1−x²))^{γ/2}.
    """
    x = (np.arange(n, dtype=np.float64) + 0.5) / n * 2.0 - 1.0        # (-1,1)
    base = np.sqrt(np.maximum(0.0, 1.0 - x * x))
    a = base ** gamma
    a *= target_sum / a.sum()
    return a


def _sine_pow_sequence(n: int, alpha: float, target_sum: float) -> np.ndarray:
    """
    a_i = sin(π·(i+0.5)/(n+1))^{α}   (α≥1 sharpens the ends).
    """
    x = np.arange(n, dtype=np.float64) + 0.5
    base = np.sin(np.pi * x / (n + 1.0))
    a = base ** alpha
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 250

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a = _project_to_simplex(a + jitter, target_sum)

        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Multi‑index sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _subgradient_multi(a0: np.ndarray,
                       target_sum: float,
                       deadline: float,
                       eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent where the sub‑gradient is the sum of the
    gradients of *all* active maximal entries.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_fft(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.flatnonzero(conv >= max_val - eps)

        # second highest value
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()
        gap = max_val - second_max

        # build summed sub‑gradient
        n = a.shape[0]
        g = np.zeros_like(a)
        for k in idx_max:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            g[idx] += 2.0 * a[k - idx]

        norm_sq = float(np.dot(g, g))
        if norm_sq == 0.0:
            break

        # step chosen to eliminate the current gap (quadratic bound)
        step = 0.9 * gap / norm_sq
        # safeguard against too aggressive moves
        step = min(step, target_sum * 0.5 / (np.linalg.norm(g) + 1e-12))

        # simple back‑tracking line search
        for _ in range(6):
            cand = _project_to_simplex(a - step * g, target_sum)
            cand_score = _max_convolution_fft(cand)
            if cand_score < max_val - 1e-14:
                a = cand
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = cand.copy()
                    print(f"[{time.time():.2f}] Multi‑subgrad it={it:5d} step={step:.3e} score={best_score:.6f}")
                break
            step *= 0.5
        else:
            # no progress – tiny random jitter
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a = _project_to_simplex(a + jitter, target_sum)

        it += 1

        # early stop if gap already tiny
        if gap < 1e-12:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (as in the reference code)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the maximal convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _max_convolution_fft(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - tol)[0]

        # second largest value
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # pick one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k‑i}
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # receiver – smallest partner value (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _max_convolution_fft(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score

        lambda: _sine_pow_sequence(
            n, rng.uniform(1.0, 6.0), target_sum),
        # pure Dirichlet draw (Γ(α,1) with α∈[0.3,5])
        lambda: _project_to_simplex(
            rng.gamma(rng.uniform(0.3, 5.0), 1.0, size=n), target_sum)
    ]

    # systematic scan (≈300 candidates, negligible time)
    for _ in range(350):
        seq = families[rng.integers(len(families))]()
        seq = _project_to_simplex(seq, target_sum)
        val = _max_convolution_fft(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Pre‑search improved best = {best_score:.6f}")

    # include the previously discovered sequence if available
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev
        resized = _project_to_simplex(resized, target_sum)
        val = _max_convolution_fft(resized)
        if val < best_score:
            best_score = val
            best_seq = resized.copy()
            print(f"[{time.time():.2f}] Imported previous improved best = {best_score:.6f}")

    # ----- Phase A: multi‑index sub‑gradient descent --------------------
    subgrad_deadline = time.time() + (deadline - time.time()) * 0.44
    seq_sg, score_sg = _subgradient_multi(best_seq, target_sum, subgrad_deadline)
    if score_sg < best_score:
        best_score = score_sg
        best_seq = seq_sg.copy()
        print(f"[{time.time():.2f}] After multi‑subgrad best = {best_score:.6f}")

    # ----- Phase B: Adam‑style soft‑max GD (restarts) ------------------
    while time.time() < deadline - 6.0:
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        start_vec = best_seq + jitter
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_adam, val_adam = _adam_gradient_descent(
            start_vec,
            deadline,
            target_sum,
            max_beta=6e5,
            step_factor=0.7,
            gamma=0.995)

        if val_adam < best_score:
            best_score = val_adam
            best_seq = a_adam.copy()
            print(f"[{time.time():.2f}] Adam GD improved best = {best_score:.6f}")

    # ----- Phase C: deterministic mass‑transfer ------------------------
    if time.time() < deadline - 3.0:
        a_dt, val_dt = _deterministic_mass_transfer_exact2(
            best_seq, deadline, target_sum)
        if val_dt < best_score:
            best_score = val_dt
            best_seq = a_dt.copy()
            print(f"[{time.time():.2f}] Deterministic DMT improved best = {best_score:.6f}")

    # ----- Final polishing (high β) ------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(
            best_seq,
            deadline,
            target_sum,
            max_beta=1e6,
            step_factor=0.6,
            gamma=0.99)
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best = {best_score:.6f}")

    # ----- Post‑processing ---------------------------------------------
    arr = np.asarray(best_seq, dtype=float)
    nz = np.flatnonzero(arr > 1e-12)
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final evaluation (harmless if evaluate_sequence missing)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
