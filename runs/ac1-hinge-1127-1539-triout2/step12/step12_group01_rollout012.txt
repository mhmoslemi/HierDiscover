"""
Search for a low‑score non‑negative step function.

Algorithm
---------
1. Warm‑start from the previously best known sequence
   (global variable ``height_sequence_1``) and resize it
   to the current problem size.
2. Short “mirror‑descent’’ optimisation on a soft‑max
   surrogate of the max‑convolution (Phase A).
3. Optional sub‑gradient descent on the true max‑convolution
   (Phase B).
4. Long refined simulated‑annealing mass‑transfer (Phase C):
   * donor index is sampled proportionally to its contribution
     to the *current* peak of the convolution;
   * receiver is sampled **outside** the overlap interval of the
     current peak (this cannot increase the current peak);
   * step size is proportional to the temperature, which follows a
     square‑root cooling schedule;
   * after each accepted move the convolution is updated in O(1) time;
   * a full recomputation of the convolution and a renormalisation
     of the mass are performed periodically to avoid drift.

The routine respects the supplied time budget (default 1000 s) and
returns the best sequence found as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Objective  2·n·max(a*a) / (sum(a)²).
    For Σa = √(2n) this equals max(a*a).
    """
    n = a.shape[0]
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of the max‑convolution.
    Returns (grad, full_convolution) where grad has the same length as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    grad_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(grad_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b

def exponential_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Mirror descent on the soft‑max surrogate."""
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        # occasional jitter to avoid stagnation
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 8e-4, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpDescent it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

def subgradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Sub‑gradient descent on the true max‑convolution entry."""
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        # linear schedule for η
        elapsed = time.time() - start
        total = deadline - start
        eta_start = 2.0
        eta_end = 0.2
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        # jitter to escape flat regions
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val

        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))   # √‑schedule

        # location of the current maximal convolution entry
        kmax = int(np.argmax(b))

        # overlap interval for this shift
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # ---- donor selection (proportional to contribution) ----
        contrib = a[idx] * a[kmax - idx]
        if contrib.sum() > 0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.choice(idx)

        # ---- receiver selection (outside the overlap interval) ----
        if lo == 0 and hi == n - 1:
            # whole range overlaps – fall back to low‑value bias
            inv = 1.0 / (a + 1e-12)
            inv = inv / inv.sum()
            receiver = rng.choice(n, p=inv)
        else:
            cand = np.concatenate((np.arange(0, lo, dtype=np.int64),
                                   np.arange(hi + 1, n, dtype=np.int64)))
            # bias towards small heights
            inv = 1.0 / (a[cand] + 1e-12)
            inv = inv / inv.sum()
            receiver = rng.choice(cand, p=inv)

        if donor == receiver:
            continue

        # ---- step‑size (δ) ----
        max_delta = min(a[donor] * 0.12, target_sum * 0.012)   # a bit larger than before
        if max_delta <= 0.0:
            continue
        # scale with temperature, keep a tiny baseline to allow exploration at low T
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * (0.08 + 0.92 * scale)
        if delta <= 0.0:
            continue

        # ---- O(1) update of the convolution ----
        b_new = b.copy()
        # linear terms
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic terms
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        # ---- Metropolis acceptance (based on *current* value) ----
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            # numerical clean‑up
            a[donor] = max(a[donor], 0.0)
            a[receiver] = max(a[receiver], 0.0)

            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA improved best={best_val:.6f}")

        # ---- periodic full recomputation (drift control) ----
        if it % 3000 == 0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5      # safety margin

    # --------------------------------------------------------------
    # Choose sequence length (larger n → finer granularity but slower)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Warm‑start from the best known construction if available
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
    else:
        # simple decreasing power‑law fallback
        init = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))

    init = init * (target_sum / init.sum())

    # --------------------------------------------------------------
    # Phase A – exponential (mirror) descent (≈15 % of budget)
    # --------------------------------------------------------------
    exp_deadline = time.time() + 0.15 * (deadline - time.time())
    a_exp, val_exp = exponential_descent(init, target_sum, exp_deadline)
    best_seq, best_score = a_exp, val_exp
    print(f"[{time.time():.2f}] After exponential descent best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – subgradient descent (≈15 % of remaining budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + 0.15 * (deadline - time.time())
        a_sub, val_sub = subgradient_descent(best_seq, target_sum, sub_deadline)
        if val_sub < best_score:
            best_seq, best_score = a_sub, val_sub
        print(f"[{time.time():.2f}] After subgradient descent best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase C – refined simulated‑annealing (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_ann, val_ann = refined_anneal(best_seq, target_sum, deadline)
        if val_ann < best_score:
            best_seq, best_score = a_ann, val_ann
        print(f"[{time.time():.2f}] After annealing best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Clean‑up, final evaluation and warm‑start storage
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
