"""
Hybrid optimiser with extra post‑processing.

The algorithm follows roughly the same multi‑scale scheme as the
reference implementation but adds:

* a longer final length (up to 20 000) when time permits;
* a cheap random‑permutation search after up‑scaling to discover
  better placements of the same multiset of heights;
* a simulated‑annealing “mass‑move” phase that directly trades mass
  between indices that contribute heavily to the current convolution
  peak;
* slightly more aggressive hyper‑parameters in the annealed
  gradient‑descent phase and in the deterministic mass‑transfer
  refinement.

All steps respect non‑negativity and the ℓ¹‑simplex constraint
(∑a = √(2 n)).  The best sequence found is stored in the global
``height_sequence_1`` and returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    left_turn = True
    for v in sorted_vals:
        if left_turn:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        left_turn = not left_turn
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread large values uniformly using a coprime step."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


def _random_permutation(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Shuffle the entries uniformly."""
    perm = rng.permutation(seq.shape[0])
    return seq[perm]


# ----------------------------------------------------------------------
# Scoring utilities
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (∑a)² .  Used only for logging."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Soft‑max gradient (annealed GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(conv)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.7,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the soft‑max approximation.
    Returns the best sequence seen and its score.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur = _score(a)
        if cur < best_val:
            best_val = cur
            best = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent using the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_val = _score(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        # indices achieving the max (allow some tolerance)
        idxs = np.where(conv >= maxb * (1 - 1e-12))[0]
        grad = np.zeros_like(a)
        for k in idxs:
            lo = max(0, k - (a.shape[0] - 1))
            hi = min(k, a.shape[0] - 1)
            ind = np.arange(lo, hi + 1, dtype=np.int64)
            grad[ind] += 2.0 * a[k - ind]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur = _score(a)
        if cur < best_val:
            best_val = cur
            best = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (greedy)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-4,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy mass moves with a Metropolis acceptance schedule."""
    a = a0.copy()
    cur_score = _score(a)
    best = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start = time.time()
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time():.2f}] DMT new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Linear‑programming alternating refinement (small dimensionality)
# ----------------------------------------------------------------------


def _lp_optimize(f: np.ndarray,
                 M: float,
                 target_sum: float,
                 rng: np.random.Generator) -> np.ndarray | None:
    """Solve max c·b s.t. f*b ≤ M, b ≥ 0; then renormalise to ℓ¹‑norm = target_sum."""
    n = f.shape[0]
    rows = []
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=np.int64)
        row = np.zeros(n, dtype=np.float64)
        row[i] = f[k - i]
        rows.append(row)
    A = np.vstack(rows)               # (2n‑1) × n
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Random linear objective
    c = -rng.random(n)

    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception as e:
        print(f"[{time.time():.2f}] LP exception: {e}")
        return None

    if not res.success:
        # print(f"[{time.time():.2f}] LP failed: {res.message}")
        return None

    b = res.x
    total = b.sum()
    if total <= 0:
        return None
    b *= target_sum / total
    return b


# ----------------------------------------------------------------------
# Simulated‑annealing mass‑move phase (targets the current peak)
# ----------------------------------------------------------------------


def _annealed_mass_move(a0: np.ndarray,
                       deadline: float,
                       target_sum: float,
                       start_temp: float = 0.3,
                       end_temp: float = 1e-5,
                       max_move_frac: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a small amount of mass from a high‑contributing
    index (relative to the current max‑convolution location) to a
    low‑contributing index.  Acceptance follows a Metropolis rule.
    """
    a = a0.copy()
    cur_score = _score(a)
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start = time.time()

    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        # locate current peak
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to the peak (a_i * a_{kmax-i})
        contrib = a[idx] * a[kmax - idx]

        # donor = index with large contribution, receiver = small contribution
        donor = idx[np.argmax(contrib)]
        # pick a receiver from the low‑contributing tail (including possibly outside the peak)
        low_candidates = np.setdiff1d(np.arange(a.shape[0]), [donor])
        receiver = rng.choice(low_candidates)

        # amount to move – a fraction of the smaller of the two entries
        max_delta = min(a[donor] * max_move_frac, a[receiver] * max_move_frac)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time():.2f}] AnnealMass new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best, best_score


# ----------------------------------------------------------------------
# Random‑permutation post‑processing
# ----------------------------------------------------------------------


def _best_of_random_permutations(seq: np.ndarray,
                                 rng: np.random.Generator,
                                 max_trials: int,
                                 deadline: float) -> np.ndarray:
    """Try a number of random shuffles and keep the one with the smallest max‑conv."""
    best = seq.copy()
    best_val = _max_convolution_fft(best)
    it = 0
    while it < max_trials and time.time() < deadline:
        cand = _random_permutation(seq, rng)
        val = _max_convolution_fft(cand)
        if val < best_val:
            best_val = val
            best = cand.copy()
            print(f"[{time.time():.2f}] Permutation trial {it} improved max={best_val:.6f}")
        it += 1
    return best


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid search for a non‑negative step function with a low evaluation score.
    The routine adapts its work‑load to the time budget.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Choose problem size
    # --------------------------------------------------------------
    if budget_s >= 900:
        n_main = 20000
    elif budget_s >= 600:
        n_main = 15000
    elif budget_s >= 400:
        n_main = 12000
    else:
        n_main = 8000
    target_main = np.sqrt(2.0 * n_main)

    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # Build a diverse pool of seeds
    # --------------------------------------------------------------
    init_pool = []

    # power laws
    for p in np.linspace(0.15, 1.0, 9):
        init_pool.append(_power_law_sequence(n_main, float(p), target_main))
    # exponentials
    for lam in np.linspace(0.0005, 0.02, 6):
        init_pool.append(_exponential_sequence(n_main, float(lam), target_main))
    # random points on simplex
    for _ in range(8):
        init_pool.append(_random_simplex(n_main, target_main, rng))

    # incorporate previous run if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_main))

    # jittered variants
    augmented = []
    for seq in init_pool:
        jitter = rng.normal(scale=target_main * 0.02, size=n_main)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_main)
        augmented.append(seq_j)
    init_pool.extend(augmented)

    # --------------------------------------------------------------
    # Phase 1 – many short annealed‑gradient runs
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    while time.time() < deadline - 30.0:
        if best_seq is not None and rng.random() < 0.6:
            seed = best_seq.copy()
            seed += rng.normal(scale=target_main * 0.004, size=n_main)
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_main)
        else:
            seed = rng.choice(init_pool)

        sub_dead = min(time.time() + 22.0, deadline - 8.0)
        if sub_dead <= time.time():
            break

        a_opt, val_opt = _annealed_gradient_descent(
            seed, sub_dead, target_main,
            max_beta=8e5, step_factor=0.6, gamma=0.985
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – deterministic mass‑transfer refinement (medium run)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 15.0:
        mt_dead = time.time() + (deadline - time.time()) * 0.10
        a_ref, val_ref = _deterministic_mass_transfer(
            best_seq, mt_dead, target_main,
            start_temp=0.4, end_temp=1e-4, low_frac=0.06
        )
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] After MT refinement best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – LP alternating refinement on a down‑scaled problem
    # --------------------------------------------------------------
    if time.time() < deadline - 30.0:
        n_lp = min(2500, max(1000, n_main // 6))
        target_lp = np.sqrt(2.0 * n_lp)
        seq_lp = _resize_sequence(best_seq, n_lp)
        seq_lp = _project_to_simplex(seq_lp, target_lp)

        lp_iters = 5
        for it in range(lp_iters):
            if time.time() > deadline - 12.0:
                break
            M = _max_convolution_fft(seq_lp)
            b_opt = _lp_optimize(seq_lp, M, target_lp, rng)
            if b_opt is None:
                break
            # mixing step
            seq_lp = (0.75) * seq_lp + (0.25) * b_opt
            seq_lp = _project_to_simplex(seq_lp, target_lp)
            cur = _score(seq_lp)
            print(f"[{time.time():.2f}] LP iter={it} score={cur:.6f}")

        # optional short GD on the small problem
        if time.time() < deadline - 8.0:
            gd_dead = time.time() + min(9.0, deadline - time.time() - 6.0)
            a_opt, val_opt = _annealed_gradient_descent(
                seq_lp, gd_dead, target_lp,
                max_beta=2e5, step_factor=0.55, gamma=0.98
            )
            if val_opt < _score(seq_lp):
                seq_lp = a_opt.copy()
                print(f"[{time.time():.2f}] Post‑LP GD score={val_opt:.6f}")

        # --------------------------------------------------------------
        # Phase 4 – upscale to a large size for final polishing
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 90.0:
            n_final = 20000
        elif remaining > 45.0:
            n_final = 15000
        else:
            n_final = n_main
        target_final = np.sqrt(2.0 * n_final)
        seq_big = _resize_sequence(seq_lp, n_final)
        seq_big = _project_to_simplex(seq_big, target_final)

    else:
        # Not enough time – just upscale the best we have
        remaining = deadline - time.time()
        if remaining > 90.0:
            n_final = 20000
        elif remaining > 45.0:
            n_final = 15000
        else:
            n_final = n_main
        target_final = np.sqrt(2.0 * n_final)
        seq_big = _resize_sequence(best_seq, n_final)
        seq_big = _project_to_simplex(seq_big, target_final)

    # --------------------------------------------------------------
    # Phase 5 – cheap re‑ordering heuristics
    # --------------------------------------------------------------
    candidates = [
        seq_big,
        _alternating_order(seq_big),
        _even_spread_order(seq_big),
    ]
    seq_big = min(candidates, key=_max_convolution_fft)
    best_score = _score(seq_big)
    print(f"[{time.time():.2f}] After reordering length={len(seq_big)} score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 6 – sub‑gradient polishing on the large problem
    # --------------------------------------------------------------
    if time.time() < deadline - 6.0:
        sub_dead = deadline - 4.0
        a_sub, val_sub = _subgradient_descent(
            seq_big, sub_dead, target_final,
            step_factor=0.45, gamma=0.995
        )
        if val_sub < best_score:
            best_score = val_sub
            seq_big = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑grad polished best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 7 – random permutation search (cheap but effective)
    # --------------------------------------------------------------
    if time.time() < deadline - 4.0:
        perm_dead = time.time() + min(12.0, deadline - time.time() - 2.0)
        seq_big = _best_of_random_permutations(seq_big, rng, max_trials=120, deadline=perm_dead)
        best_score = _score(seq_big)
        print(f"[{time.time():.2f}] After permutation search score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 8 – final annealed mass‑move refinement
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        anneal_dead = time.time() + min(8.0, deadline - time.time() - 1.0)
        a_ann, val_ann = _annealed_mass_move(
            seq_big, anneal_dead, target_final,
            start_temp=0.25, end_temp=1e-5, max_move_frac=0.25
        )
        if val_ann < best_score:
            best_score = val_ann
            seq_big = a_ann.copy()
            print(f"[{time.time():.2f}] Final annealed mass move best={best_score:.6f}")

    # --------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # --------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Log final score (pure evaluation)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    globals()["height_sequence_1"] = final_seq
    return final_seq
