def _optimal_step(conv, delta, t_max, max_iters=30):
    lo = 0.0
    hi = t_max
    # Initially compute derivative at lo: ds = delta[conv.argmax()] may be negative?
    # We'll just binary search for zero crossing.
    for _ in range(max_iters):
        mid = (lo + hi) / 2.0
        vals = conv + mid * delta
        max_val = vals.max()
        max_idx = np.where(np.abs(vals - max_val) <= 1e-12)[0]  # tolerance
        # derivative is max(delta[max_idx])
        deriv = delta[max_idx].max()
        if deriv > 0:
            hi = mid
        elif deriv < 0:
            lo = mid
        else:
            lo = hi = mid
            break
    return (lo + hi) / 2.0
```

"""
Step‑function optimizer (≈100 s pre‑search + fast mass‑transfer).

Main ideas
----------

* A cheap parametric pre‑search on a small problem (n≈2000) scans three
  classic families (power‑law, exponential‑power‑law, broken power‑law)
  and keeps the best shape.

* The selected shape is interpolated to the final problem size
  (n≈5000–18000 depending on the time budget) and projected onto the
  simplex Σa = √(2 n).

* A refined guided‑mass‑transfer loop repeatedly:
      – computes the set of maximal convolution entries,
      – evaluates the exact (averaged) sub‑gradient,
      – builds a zero‑sum direction d,
      – obtains the maximal feasible step keeping a ≥ 0,
      – performs a binary line‑search on the piece‑wise linear function
        f(t)=max(conv(a+t d)) to drive the maximum as low as possible,
      – updates a and projects onto the simplex.
  This exact line‑search is more aggressive than the simple “to‑second‑max”
  step used previously and usually yields a larger reduction per iteration.

* Occasionally a tiny random jitter is added to escape plateaus.

* After the time budget is exhausted the sequence is trimmed (trailing
  near‑zero entries removed) and returned.

The code respects the competition interface: it defines `propose_candidate`
and may use the provided `evaluate_sequence` for final reporting only.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolutions (real input)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Small helper to get the next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft_pair(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution x * y (full length 2·n‑1) via real FFT."""
    n = x.shape[0]
    size = _next_pow2(2 * n - 1)
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of a * a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Parametric families (used only in the tiny pre‑search)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ i^{‑p}."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    """a_i ∝ i^{‑p}·exp(-q·i/n)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _broken_power_law_sequence(
    n: int, p1: float, p2: float, alpha: float, target_sum: float
) -> np.ndarray:
    """Two‑segment power law split at α·n."""
    K = max(1, int(alpha * n))
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = np.empty_like(idx)
    a[:K] = idx[:K] ** (-p1)
    a[K:] = idx[K:] ** (-p2)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D sequence to a new length."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, seq)


# ----------------------------------------------------------------------
# Exact sub‑gradient for the max‑convolution entry
# ----------------------------------------------------------------------


def _compute_gradient_for_max(a: np.ndarray, max_idx: np.ndarray) -> np.ndarray:
    """Average sub‑gradient of max(conv) w.r.t. a (over all maximal indices)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in max_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g[idx] += 2.0 * a[k - idx]
    if max_idx.size > 0:
        g /= max_idx.size
    return g


# ----------------------------------------------------------------------
# Optimal step for a given direction (binary search on a piece‑wise linear max)
# ----------------------------------------------------------------------


def _optimal_step(conv: np.ndarray, delta: np.ndarray, t_max: float, max_iters: int = 30) -> float:
    """Find t∈[0,t_max] minimising max(conv + t·delta) by binary search."""
    lo, hi = 0.0, t_max
    # Early exit checks
    # slope at lo
    max_val = conv.max()
    idx = np.where(conv >= max_val - 1e-12)[0]
    deriv = delta[idx].max()
    if deriv >= 0:          # cannot move right, optimum at 0
        return 0.0
    # slope at hi
    max_val_hi = (conv + hi * delta).max()
    idx_hi = np.where(conv + hi * delta >= max_val_hi - 1e-12)[0]
    deriv_hi = delta[idx_hi].max()
    if deriv_hi <= 0:       # still decreasing at hi, optimum at hi
        return hi

    for _ in range(max_iters):
        mid = (lo + hi) * 0.5
        vals = conv + mid * delta
        m = vals.max()
        idx_mid = np.where(np.abs(vals - m) <= 1e-12)[0]
        dmid = delta[idx_mid].max()
        if dmid > 0:
            hi = mid
        elif dmid < 0:
            lo = mid
        else:
            lo = hi = mid
            break
    return (lo + hi) * 0.5


# ----------------------------------------------------------------------
# Guided exact mass‑transfer with refined line search
# ----------------------------------------------------------------------


def _guided_mass_transfer(a0: np.ndarray, deadline: float, target_sum: float) -> tuple[np.ndarray, float]:
    """
    Reduce the maximal convolution entry by moving mass from contributors
    to non‑contributors. Uses the exact binary line search described above.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_conv(a)
    rng = np.random.default_rng()
    iteration = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        max_idx = np.where(conv >= max_val - 1e-12)[0]

        # exact (averaged) sub‑gradient for the current peak(s)
        g = _compute_gradient_for_max(a, max_idx)

        # direction that preserves Σa = 0
        sum_g = g.sum()
        d = -g + sum_g / n * np.ones_like(a)

        # maximal feasible step before any component becomes negative
        neg = d < 0
        if np.any(neg):
            t_nonneg = np.min(a[neg] / -d[neg])
        else:
            t_nonneg = np.inf
        if t_nonneg <= 0:
            break

        # linear change of the convolution in direction d
        delta = 2.0 * _conv_fft_pair(a, d)

        # optimal step for this direction
        t_opt = _optimal_step(conv, delta, t_nonneg, max_iters=30)
        if t_opt <= 0:
            break

        a += t_opt * d
        # tiny numerical corrections
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        cur_score = _max_conv(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time():.2f}] GuidedMT it={iteration:5d} score={best_score:.6f}")

        # occasional jitter to escape tiny plateaus
        if iteration % 500 == 0 and rng.random() < 0.05:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        iteration += 1

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal benchmark score.
    The routine respects the supplied time budget and returns the best
    sequence found (trailing near‑zero entries are trimmed).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Choose problem dimension – larger n gives a finer discretisation.
    # ------------------------------------------------------------------
    if budget_s >= 900:
        n = 18000
    elif budget_s >= 800:
        n = 15000
    elif budget_s >= 600:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Initialise best candidate (global warm‑start, if any)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    if "height_sequence_1" in globals():
        seq0 = np.array(globals()["height_sequence_1"], dtype=np.float64)
        seq0 = _resize_sequence(seq0, n)
        seq0 = _project_to_simplex(seq0, target_sum)
        best_seq = seq0.copy()
        best_score = _max_conv(seq0)
        print(f"[{time.time():.2f}] Warm‑start from global best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – very cheap parametric pre‑search on a small instance
    # ------------------------------------------------------------------
    # Work on a reduced size to explore many shapes quickly.
    n_small = max(2000, n // 6)
    target_sum_small = math.sqrt(2.0 * n_small)

    best_small_seq = None
    best_small_score = float("inf")

    # (a) pure power‑law
    for p in np.arange(0.20, 1.21, 0.02):
        seq = _power_law_sequence(n_small, float(p), target_sum_small)
        val = _max_conv(seq)
        if val < best_small_score:
            best_small_score, best_small_seq = val, seq.copy()

    # (b) exponential‑power‑law (moderately dense grid)
    for p in np.arange(0.20, 1.21, 0.04):
        for q in np.arange(0.0, 8.001, 0.5):
            seq = _exp_power_law_sequence(n_small, float(p), float(q), target_sum_small)
            val = _max_conv(seq)
            if val < best_small_score:
                best_small_score, best_small_seq = val, seq.copy()

    # (c) broken power‑law (two‑segment)
    for p1 in np.arange(0.20, 0.81, 0.1):
        for p2 in np.arange(0.80, 2.01, 0.2):
            for alpha in np.arange(0.1, 0.91, 0.1):
                seq = _broken_power_law_sequence(
                    n_small, float(p1), float(p2), float(alpha), target_sum_small
                )
                val = _max_conv(seq)
                if val < best_small_score:
                    best_small_score, best_small_seq = val, seq.copy()

    # Upsample the best small shape to the full size
    if best_small_seq is not None:
        seq_big = _resize_sequence(best_small_seq, n)
        seq_big = _project_to_simplex(seq_big, target_sum)
        val_big = _max_conv(seq_big)
        if val_big < best_score:
            best_score, best_seq = val_big, seq_big.copy()
            print(f"[{time.time():.2f}] Upsampled parametric shape score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – refined guided mass‑transfer (core optimisation)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0 and best_seq is not None:
        mt_deadline = time.time() + 0.85 * remaining  # reserve a little slack
        a_opt, val_opt = _guided_mass_transfer(best_seq, mt_deadline, target_sum)
        if val_opt < best_score:
            best_score, best_seq = val_opt, a_opt.copy()
            print(f"[{time.time():.2f}] Post‑MT improved score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – tiny random‑jitter polishing (optional, cheap)
    # ------------------------------------------------------------------
    # One last sweep of small random mass transfers to squeeze out any left‑over improvement.
    if best_seq is not None and time.time() < deadline - 0.5:
        rng = np.random.default_rng()
        a = best_seq.copy()
        n = a.shape[0]
        for _ in range(200):
            if time.time() > deadline - 0.5:
                break
            # pick a donor among the top 2 % and a receiver among the bottom 2 %
            k_top = max(1, int(0.02 * n))
            k_low = max(1, int(0.02 * n))
            donor = np.argsort(-a)[rng.integers(k_top)]
            receiver = np.argsort(a)[rng.integers(k_low)]
            if donor == receiver:
                continue
            delta = rng.random() * a[donor] * 0.03  # move up to 3 % of donor mass
            a[donor] -= delta
            a[receiver] += delta
            a = _project_to_simplex(a, target_sum)

            cur = _max_conv(a)
            if cur < best_score - 1e-12:
                best_score, best_seq = cur, a.copy()
                print(f"[{time.time():.2f}] Random jitter improvement = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return list
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final evaluation (harmless if evaluate_sequence is not present)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated benchmark score = {val:.6f}")
    except Exception:
        pass

    return final_seq
