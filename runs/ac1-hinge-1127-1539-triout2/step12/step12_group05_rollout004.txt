"""
Step‑function optimizer.

The algorithm proceeds in three stages:

1. **Randomised parametric pre‑search** – sample a few classic families
   (pure power law, exponential‑power‑law and a broken power‑law) to
   obtain a good initial shape.  The number of samples is kept modest
   so that this phase is fast even for large problem sizes.

2. **Projected Adam on a smooth soft‑max approximation** – minimise a
   smooth approximation of the maximal convolution while respecting the
   simplex constraint Σa = √(2 n) and non‑negativity.  The smoothing
   parameter β is increased exponentially, the step‑size is adapted
   with Adam‑type moments and occasional jitter is added to escape
   plateaus.  A long Adam run consumes the bulk of the remaining budget.

3. **Exact sub‑gradient guided mass‑transfer** – starting from the best
   Adam solution we iteratively move mass from indices that increase the
   current peak most (computed via the exact sub‑gradient) to indices
   that increase it least, using an exact line‑search that drives the
   current maximal convolution entry down to the second‑largest entry.
   This phase performs fine‑grained reductions of the peak.

A final short high‑β Adam run is performed for polishing.  The routine
adapts the sequence length `n` to the time budget (larger `n` for longer
budgets) and returns the best sequence found before the deadline.
"""

import time

import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (linear convolution a*a and a*b)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) using a real FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft_pair(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution x*y (full length 2·n‑1) using a real FFT."""
    n = x.size
    size = 1 << ((2 * n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Benchmark score (equal to 2·n·max(conv) / (Σa)²).
    Assumes a is already non‑negative; Σa must be > 0.
    """
    n = a.shape[0]
    max_b = _max_conv(a)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for the cheap pre‑search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _broken_power_law_sequence(
    n: int, p1: float, p2: float, alpha: float, target_sum: float
) -> np.ndarray:
    K = min(n, max(1, int(alpha * n)))
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.empty_like(idx)
    a[:K] = idx[:K] ** (-p1)
    a[K:] = idx[K:] ** (-p2)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max approximation of the max‑convolution and Adam optimizer
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(a*a).
    Returns (grad, full_convolution) where grad has shape (n,).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = conv_full.max()
    # soft‑max over the convolution entries
    exps = np.exp(beta * (conv_full - max_b))
    p = exps / exps.sum()

    # gradient w.r.t. a: 2 * (a_rev * p) restricted to the original length
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, conv_full


def _adam_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 5e5,
    step_factor: float = 0.8,
    gamma: float = 0.999,
) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max approximation of the max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 200  # increase β less often for stability

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1.0 - gamma) * g
        v = gamma * v + (1.0 - gamma) * (g * g)
        m_hat = m / (1.0 - gamma ** (it + 1))
        v_hat = v / (1.0 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid being stuck
        if it % 400 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient guided mass‑transfer (fine‑grained reduction)
# ----------------------------------------------------------------------


def _compute_gradient_for_max(a: np.ndarray, max_idx: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a,
    averaged over all maximal indices.
    """
    n = a.size
    g = np.zeros_like(a)
    for k in max_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        # contribution 2 * a[k - idx] (due to symmetry of convolution)
        g[idx] += 2.0 * a[k - idx]
    if max_idx.size > 0:
        g /= max_idx.size
    return g


def _guided_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the maximal convolution entry.
    At each iteration mass is moved from indices that increase the peak
    most towards indices that increase it least.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        # mask for entries that are (practically) maximal
        mask_max = conv >= max_val - tol
        # second‑largest value (ignoring ties)
        if np.any(~mask_max):
            second_max = conv[~mask_max].max()
        else:
            second_max = 0.0
        max_idx = np.where(mask_max)[0]

        # exact sub‑gradient (averaged)
        g = _compute_gradient_for_max(a, max_idx)

        # direction preserving Σa = 0
        sum_g = g.sum()
        d = -g + (sum_g / n) * np.ones_like(a)   # Σd = 0

        # maximum admissible step before any component would become negative
        neg = d < 0
        if np.any(neg):
            t_nonneg = np.min(a[neg] / -d[neg])
        else:
            t_nonneg = np.inf

        # linearised change of the convolution along d
        delta = 2.0 * _conv_fft_pair(a, d)          # length 2n‑1
        delta_max = delta[max_idx]
        delta_k = delta_max.min()                   # most favourable reduction

        if delta_k >= 0:
            # no descent direction – stop
            break

        # step that would bring the current maximal entry down to the second max
        t_to_second = (max_val - second_max) / (-delta_k)

        # safe step (a little margin)
        t = min(t_nonneg, t_to_second) * 0.99
        if t <= 0:
            break

        a += t * d
        a = np.maximum(a, 0.0)                     # enforce non‑negativity
        a = _project_to_simplex(a, target_sum)      # fix tiny sum drift

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] GuidedMT it={it} score={best_score:.6f}")

        # occasional random jitter to escape tiny plateaus
        if it % 400 == 0 and rng.random() < 0.05:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_score

def _global_random_move(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
) -> tuple[np.ndarray, float]:
    """Very cheap simulated‑annealing style mass moves."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    while time.time() < deadline - 0.5:
        # pick a donor among the top 2 % and a receiver among the bottom 2 %
        top_k = max(1, int(0.02 * n))
        low_k = max(1, int(0.02 * n))
        donor = np.argsort(-a)[rng.integers(top_k)]
        receiver = np.argsort(a)[rng.integers(low_k)]
        if donor == receiver:
            continue
        delta = rng.random() * a[donor] * 0.05
        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] RandomMove new best={best_score:.6f}")

    return best_a, best_score

    # Post‑processing: trim trailing near‑zero entries and return list
    # ---------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final evaluation (harmless if evaluate_sequence is absent)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated benchmark score = {val:.6f}")
    except Exception:
        pass

    return final_seq
