"""
Search for a non‑negative step function minimizing the benchmark.
Algorithm (all phases respect the given time budget):

1. **Parametric warm‑start** – sweep simple families (power‑law, exponential)
   and optionally reuse the previous best (`height_sequence_1`).  The best
   candidate is kept as the initial point.

2. **Mirror‑descent on a soft‑max surrogate** – exponential descent that
   quickly lowers the surrogate of the maximal convolution while keeping
   the ℓ₁‑norm fixed.

3. **Sub‑gradient descent on the true max‑convolution** – multiplicative
   updates using the exact sub‑gradient of the active peak.

4. **Aggressive simulated‑annealing mass‑transfer** – Metropolis moves that
   shift tiny amounts of mass from “heavy” donors to “light” receivers.
   Temperature starts at the current score (≈ 100 % of it) and cools with a
   slower α‑schedule (power ¼) to stay hot longer; transfer sizes are larger
   (up to 10 % of the donor or 2 % of the total sum).

5. **Projected gradient refinement** – a few Euclidean‑projected gradient
   steps with diminishing stepsizes to polish the final solution.

The routine returns the best sequence found as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Objective  2·n·max(a*a) / (sum(a)²).
    For Σa = √(2n) this equals max(a*a).
    """
    n = a.shape[0]
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Phase A – exponential (mirror) descent on the soft‑max surrogate
# ----------------------------------------------------------------------


def exponential_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Mirror descent on the soft‑max surrogate of the max‑convolution."""
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        # --- soft‑max gradient -------------------------------------------------
        n = a.size
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_b = b.max()
        exps = np.exp(beta * (b - max_b))
        p = exps / exps.sum()

        a_rev = a[::-1]
        grad_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(grad_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # --- multiplicative update -------------------------------------------
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * grad)
        a = a * (target_sum / a.sum())

        # occasional jitter to avoid stagnation
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpDescent it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – multiplicative sub‑gradient descent on the true max‑conv.
# ----------------------------------------------------------------------


def subgradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Sub‑gradient descent on the true max‑convolution entry."""
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        elapsed = time.time() - start
        total = deadline - start
        eta_start = 2.0
        eta_end = 0.2
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – improved simulated‑annealing mass‑transfer refinement
# ----------------------------------------------------------------------


def anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Simulated annealing that moves tiny amounts of mass from “heavy’’ to
    “light’’ indices.  Temperature starts at the current score (≈ 100 %)
    and follows a power‑¼ cooling schedule.  Transfers are up to 10 % of
    the donor (or 2 % of the total sum) and are scaled with temperature.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 1.0                 # start at 100 % of current score

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        # power‑¼ cooling (keeps temperature high for longer)
        temperature = max(0.0, T0 * (1.0 - frac ** 0.25))

        # locate current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – pick an index with relatively small height
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # size of the transfer (scaled by temperature, larger moves early)
        max_delta = min(a[donor] * 0.10, target_sum * 0.02)   # up to 10 % donor or 2 % total
        if max_delta <= 0:
            continue

        scale = temperature / T0 if T0 > 0 else 0.0
        # ensure a baseline of 20 % of max_delta even when temperature is tiny,
        # but shrink quickly as we cool.
        delta = rng.random() * max_delta * (0.2 + 0.8 * scale)
        if delta <= 0.0:
            continue

        # --- O(n) update of the convolution ----------------------------------
        b_new = b.copy()
        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        # Metropolis acceptance based on the *current* energy
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved best={best_val:.6f}")

        # periodic full recomputation to avoid drift
        if it % 5000 == 0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase D – projected gradient descent (fine‑tuning)
# ----------------------------------------------------------------------


def projected_gradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Euclidean projected gradient descent using the exact sub‑gradient of the
    active convolution peak.  Step size decays linearly from 0.5 to 0.005.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _score(a)
    cur_val = best_val

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    it = 0

    while time.time() < deadline:
        # compute current peak and its sub‑gradient
        n = a.shape[0]
        b = _conv_fft(a)
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        # diminishing stepsize
        elapsed = time.time() - start
        frac = elapsed / total
        eta = 0.5 * (1.0 - frac) + 0.005 * frac

        a_new = a - eta * g
        a_new = _project_to_simplex(a_new, target_sum)
        a_new = np.maximum(a_new, 0.0)

        val = _score(a_new)

        if val < best_val:
            best_val = val
            best_a = a_new.copy()
            print(f"[{time.time():.2f}] PGD improved best={best_val:.6f}")

        # accept only improvements (occasional jitter to escape flat spots)
        if val < cur_val:
            a = a_new
            cur_val = val
        else:
            if rng.random() < 0.001:
                a = a_new
                cur_val = val

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5      # safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n → finer granularity, but more work)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 24000
    elif budget_s >= 500:
        n = 18000
    else:
        n = 12000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Parametric warm‑start (simple families + previous best)
    # --------------------------------------------------------------
    best_a = None
    best_val = np.inf

    # power‑law sweep
    for p in np.linspace(0.45, 0.65, 21):
        a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
        a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()

    # exponential sweep
    for c in np.linspace(0.0002, 0.0012, 11):
        a = np.exp(-c * np.arange(n, dtype=np.float64))
        a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()

    # warm‑start from previous global if available
    if "height_sequence_1" in globals():
        prev = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
        val = _score(prev)
        if val < best_val:
            best_val = val
            best_a = prev.copy()

    # --------------------------------------------------------------
    # Phase A – exponential (mirror) descent
    # --------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    if remaining > 0:
        exp_deadline = now + 0.12 * remaining
        a_exp, val_exp = exponential_descent(best_a, target_sum, exp_deadline)
        if val_exp < best_val:
            best_val = val_exp
            best_a = a_exp.copy()
        print(f"[{time.time():.2f}] After exponential descent best score={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase B – subgradient descent
    # --------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    if remaining > 0:
        sub_deadline = now + 0.12 * remaining
        a_sub, val_sub = subgradient_descent(best_a, target_sum, sub_deadline)
        if val_sub < best_val:
            best_val = val_sub
            best_a = a_sub.copy()
        print(f"[{time.time():.2f}] After subgradient descent best score={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase C – simulated annealing mass transfer
    # --------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    if remaining > 0:
        a_ann, val_ann = anneal_mass_transfer(best_a, target_sum, deadline)
        if val_ann < best_val:
            best_val = val_ann
            best_a = a_ann.copy()
        print(f"[{time.time():.2f}] After annealing best score={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase D – projected gradient refinement (optional)
    # --------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    if remaining > 10.0:      # need a few seconds to be worthwhile
        a_pg, val_pg = projected_gradient_descent(best_a, target_sum, deadline)
        if val_pg < best_val:
            best_val = val_pg
            best_a = a_pg.copy()
        print(f"[{time.time():.2f}] After projected gradient best score={best_val:.6f}")

    # --------------------------------------------------------------
    # Clean‑up, final evaluation and warm‑start storage
    # --------------------------------------------------------------
    arr = best_a
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
