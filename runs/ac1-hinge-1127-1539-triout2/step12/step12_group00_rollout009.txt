def propose_candidate(budget_s=1000, **kwargs):
    start = time.time()
    deadline = start + budget_s - 0.5

    base_n_expl = 8000 if budget_s >= 800 else 6000 if budget_s >= 400 else 4000
    target_expl = np.sqrt(2*base_n_expl)

    rng = np.random.default_rng()

    init_pool = []

    # generate seeds
    for p in np.linspace(0.2, 1.0, 9):
        init_pool.append(_power_law_sequence(base_n_expl, p, target_expl))
    for lam in np.linspace(0.001,0.02,6):
        init_pool.append(_exponential_sequence(base_n_expl, lam, target_expl))
    for _ in range(5):
        init_pool.append(_random_simplex(base_n_expl, target_expl, rng))
    init_pool.append(_triangular_sequence(base_n_expl, target_expl))
    init_pool.append(_sine_sequence(base_n_expl, target_expl))
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        init_pool.append(_resize_sequence(prev, base_n_expl))

    # Jitter each maybe
    jittered = []
    for seq in init_pool:
        jitter = rng.normal(scale=target_expl*0.01,size=base_n_expl)
        seqj = np.maximum(seq + jitter, 0.0)
        seqj = _project_to_simplex(seqj, target_expl)
        jittered.append(seqj)
    init_pool.extend(jittered)

    # pick best initial via score
    best_seq = None
    best_score = np.inf
    for seq in init_pool:
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
    print(f"[{time.time():.2f}] Initial best score={best_score}")

    # Phase 1: Adam softmax descent using multiple runs until time left (maybe allocate 0.5 of time)
    phase1_deadline = min(deadline - 200, start + (deadline - start) * 0.5)   # allocate 50% time
    while time.time() < phase1_deadline:
        # decide seed: best with jitter or random from pool
        if rng.random() < 0.7:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl*0.005,size=base_n_expl)
            seed = np.maximum(seed + jitter, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)
        run_dead = min(time.time() + 12.0, phase1_deadline)  # 12 secs each run
        if run_dead <= time.time():
            break
        seq_opt, val_opt = _adam_softmax_descent(seed, run_dead, target_expl,
                                                max_beta=1e6, step_factor=0.6,
                                                beta_inc=2.0)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = seq_opt.copy()
            print(f"[{time.time():.2f}] Adam step new best={best_score:.6f}")

    # Phase 2: Optional subgradient polish to escape local minima
    if time.time() < deadline - 150:
        sub_dead = time.time() + (deadline - time.time()) * 0.15  # 15% time
        seq_sub, val_sub = _subgradient_descent(best_seq, sub_dead, target_expl,
                                                step_factor=0.5, gamma=0.995)
        if val_sub < best_score:
            best_score = val_sub
            best_seq = seq_sub.copy()
            print(f"[{time.time():.2f}] SubGrad phase new best={best_score:.6f}")

    # Phase 3: LP refinement on downscaled version
    if time.time() < deadline - 120:
        n_lp = 1500
        seq_lp = _resize_sequence(best_seq, n_lp)
        target_lp = np.sqrt(2.0 * n_lp)
        seq_lp = _project_to_simplex(seq_lp, target_lp)
        max_lp_iters = 20
        lp_iter = 0
        while lp_iter < max_lp_iters and time.time() < deadline - 100:
            M = _max_convolution_fft(seq_lp)
            b_opt = _lp_optimize(seq_lp, M, target_lp, rng)
            if b_opt is None:
                break
            # determine mixing factor adaptively:
            # try t=0.2, if not improved, reduce.
            t = 0.2
            new_seq = (1 - t) * seq_lp + t * b_opt
            new_seq = _project_to_simplex(new_seq, target_lp)
            new_score = _score(new_seq)
            # Accept if improvement else reduce t
            if new_score < _score(seq_lp):
                seq_lp = new_seq
                print(f"[{time.time():.2f}] LP iter {lp_iter} improved score {new_score:.6f}")
            else:
                # try smaller t
                t_small = 0.05
                new_seq2 = (1 - t_small) * seq_lp + t_small * b_opt
                new_seq2 = _project_to_simplex(new_seq2, target_lp)
                new_score2 = _score(new_seq2)
                if new_score2 < _score(seq_lp):
                    seq_lp = new_seq2
                    print(f"[{time.time():.2f}] LP iter {lp_iter} improved with small t score {new_score2:.6f}")
                else:
                    # stop early if not improving for few iterations
                    pass
            lp_iter += 1
        # after LP, maybe a brief GD on small problem
        if time.time() < deadline - 70:
            gd_dead = time.time() + min(15.0, deadline - time.time() - 60)
            seq_gd, val_gd = _adam_softmax_descent(seq_lp, gd_dead, target_lp,
                                                   max_beta=2e5, step_factor=0.5,
                                                   beta_inc=2.0)
            if val_gd < _score(seq_lp):
                seq_lp = seq_gd.copy()
                print(f"[{time.time():.2f}] Post-LP GD new best {val_gd:.6f}")

        # Upscale to final length
        remaining = deadline - time.time()
        if remaining > 90:
            n_final = 16000
        elif remaining > 45:
            n_final = 12000
        else:
            n_final = max(base_n_expl, int(remaining*200))  # roughly adjust
        target_final = np.sqrt(2.0 * n_final)
        seq_big = _resize_sequence(seq_lp, n_final)
        seq_big = _project_to_simplex(seq_big, target_final)
    else:
        # Not enough time; upscale best_seq directly
        remaining = deadline - time.time()
        if remaining > 90:
            n_final = 16000
        elif remaining > 45:
            n_final = 12000
        else:
            n_final = base_n_expl
        target_final = np.sqrt(2.0 * n_final)
        seq_big = _resize_sequence(best_seq, n_final)
        seq_big = _project_to_simplex(seq_big, target_final)

    # Phase 4: reorderings
    candidates = [seq_big,
                  _alternating_order(seq_big),
                  _even_spread_order(seq_big),
                  _random_permutation(seq_big, rng)]
    seq_big = min(candidates, key=_max_convolution_fft)
    best_score = _score(seq_big)
    print(f"[{time.time():.2f}] After reorder score {best_score:.6f}")

    # Phase 5: final subgradient polishing
    if time.time() < deadline - 6:
        sub_dead = deadline - 4
        seq_sub2, val_sub2 = _subgradient_descent(seq_big, sub_dead, target_final,
                                                  step_factor=0.45, gamma=0.995)
        if val_sub2 < best_score:
            best_score = val_sub2
            seq_big = seq_sub2.copy()
            print(f"[{time.time():.2f}] Final subgrad improved to {best_score:.6f}")

    # Phase 6: final deterministic mass transfer
    if time.time() < deadline - 2:
        mt_dead = time.time() + min(6.0, deadline - time.time() - 1)
        seq_mt, val_mt = _deterministic_mass_transfer(seq_big, mt_dead, target_final,
                                                      start_temp=0.25, end_temp=1e-5,
                                                      low_frac=0.05)
        if val_mt < best_score:
            best_score = val_mt
            seq_big = seq_mt.copy()
            print(f"[{time.time():.2f}] Final mass transfer best {best_score:.6f}")

    # Trim zeros
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    try:
        val = evaluate_sequence(final_seq)
        print(f"[{time.time():.2f}] Final evaluated {val:.6f}")
    except Exception:
        pass

    globals()["height_sequence_1"] = final_seq
    return final_seq

"""
Hybrid search for a non‑negative step function that minimises
    2·n·max(a*a) / (∑a)²   (the evaluate_sequence objective).

The pipeline:
1. Build a diverse pool of seed vectors (power‑law, exponential, random,
   triangular, sinusoidal, previous best).
2. Run several short Adam‑based soft‑max gradient descents on the full size
   problem, keeping the best vector seen.
3. Optional sub‑gradient polishing on the best vector.
4. Down‑scale to ~1500 entries, perform many LP refinements (random linear
   objectives, mixing the LP solution with the current one), and a short
   Adam polishing on the small problem.
5. Upscale back to a large size (≈16000), try a few deterministic reorderings
   that spread large values, and keep the ordering with the smallest
   convolution peak.
6. Final sub‑gradient and deterministic mass‑transfer polish.
7. Trim trailing near‑zero entries and return the list.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (∑a)² (for ∑a=√(2n) this equals the max‑conv)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised to the required ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform random point on the ℓ¹‑simplex (Dirichlet‑like)."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _triangular_sequence(n: int, target_sum: float) -> np.ndarray:
    """Tent shape rising then falling linearly."""
    half = n // 2
    inc = np.arange(1, half + 1, dtype=np.float64)
    dec = inc[::-1] if n % 2 == 0 else np.arange(half, 0, -1, dtype=np.float64)
    a = np.concatenate([inc, dec])
    a *= target_sum / a.sum()
    return a


def _sine_sequence(n: int, target_sum: float) -> np.ndarray:
    """Positive half‑wave of a sine (scaled to the simplex)."""
    a = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Deterministic reorderings
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """
    Spread the large values uniformly by placing them on a cyclic sequence
    with a step coprime to the length.
    """
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]          # descending
    step = max(1, int(round(np.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


def _random_permutation(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """A random shuffle that preserves values."""
    perm = rng.permutation(seq.shape[0])
    return seq[perm]


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (gradient, convolution) where the gradient corresponds to the
    smooth approximation of the max via a softmax with inverse temperature `beta`.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)                     # softmax distribution

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    # gradient w.r.t a (only the central n entries are needed)
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b

# Adam‑based soft‑max descent
# ----------------------------------------------------------------------


def _adam_softmax_descent(a0: np.ndarray,
                          deadline: float,
                          target_sum: float,
                          max_beta: float = 1e6,
                          step_factor: float = 0.6,
                          beta_inc: float = 2.0) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the soft‑max approximation of the max‑convolution.
    Returns the best vector seen and its objective value.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    step_base = target_sum * step_factor
    rng = np.random.default_rng()

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # Learning rate adapts to the magnitude of the second moment
        lr = step_base / (np.sqrt(np.mean(v_hat)) + 1e-12) / (beta + 1.0)

        a -= lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if t % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={t:6d} β={beta:6.1f} score={best_val:.6f}")

        # increase the soft‑max temperature
        if t % 300 == 0 and beta < max_beta:
            beta = min(beta * beta_inc, max_beta)

    return best_a, best_val

def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that uses the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]
        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val

def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy mass moves with a decaying Metropolis acceptance rule."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)
    n = a.shape[0]

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            it += 1
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming alternating refinement (small dimensionality)
# ----------------------------------------------------------------------


def _lp_optimize(f: np.ndarray, M: float, target_sum: float,
                 rng: np.random.Generator) -> np.ndarray | None:
    """
    Solve a LP that maximises a random linear functional of `b`
    subject to `f * b <= M` and `b >= 0`.  The result is scaled to
    have ℓ¹‑norm = target_sum.  Returns None on failure.
    """
    n = f.shape[0]
    rows = []
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=np.int64)
        row = np.zeros(n, dtype=np.float64)
        row[i] = f[k - i]
        rows.append(row)
    A = np.vstack(rows)               # (2n‑1) × n
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Random objective – maximise c·b  ⇔ minimise –c·b
    c = -rng.random(n)

    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception as e:
        print(f"[{time.time():.2f}] LP exception: {e}")
        return None

    if not res.success:
        print(f"[{time.time():.2f}] LP failed: {res.message}")
        return None

    b = res.x
    total = b.sum()
    if total <= 0:
        return None
    b *= target_sum / total
    return b

        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
    print(f"[{time.time():.2f}] Initial best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam soft‑max descents (≈50 % of time)
    # ------------------------------------------------------------------
    phase1_deadline = min(deadline - 200,
                          start_time + (deadline - start_time) * 0.5)

    while time.time() < phase1_deadline:
        if rng.random() < 0.7:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.005, size=n_expl)
            seed = np.maximum(seed + jitter, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)
        run_dead = min(time.time() + 12.0, phase1_deadline)
        if run_dead <= time.time():
            break
        seq_opt, val_opt = _adam_softmax_descent(
            seed, run_dead, target_expl,
            max_beta=1e6, step_factor=0.6, beta_inc=2.0,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = seq_opt.copy()
            print(f"[{time.time():.2f}] Adam new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – optional sub‑gradient polishing (≈15 % of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 150:
        sub_dead = time.time() + (deadline - time.time()) * 0.15
        seq_sub, val_sub = _subgradient_descent(
            best_seq, sub_dead, target_expl,
            step_factor=0.5, gamma=0.995,
        )
        if val_sub < best_score:
            best_score = val_sub
            best_seq = seq_sub.copy()
            print(f"[{time.time():.2f}] Sub‑grad improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – LP refinement on a down‑scaled problem
    # ------------------------------------------------------------------
    if time.time() < deadline - 120:
        n_lp = 1500
        seq_lp = _resize_sequence(best_seq, n_lp)
        target_lp = np.sqrt(2.0 * n_lp)
        seq_lp = _project_to_simplex(seq_lp, target_lp)

        max_lp_iters = 20
        lp_iter = 0
        while lp_iter < max_lp_iters and time.time() < deadline - 100:
            M = _max_convolution_fft(seq_lp)
            b_opt = _lp_optimize(seq_lp, M, target_lp, rng)
            if b_opt is None:
                break

            # try a few mixing factors
            improved = False
            for t_mix in (0.2, 0.1, 0.05):
                mixed = (1 - t_mix) * seq_lp + t_mix * b_opt
                mixed = _project_to_simplex(mixed, target_lp)
                new_score = _score(mixed)
                if new_score < _score(seq_lp):
                    seq_lp = mixed
                    improved = True
                    print(f"[{time.time():.2f}] LP iter {lp_iter} t={t_mix:.2f} score={new_score:.6f}")
                    break
            if not improved:
                # no improvement – stop early
                break
            lp_iter += 1

        # Small Adam polishing on the low‑dimensional problem
        if time.time() < deadline - 70:
            gd_dead = time.time() + min(15.0, deadline - time.time() - 60)
            seq_gd, val_gd = _adam_softmax_descent(
                seq_lp, gd_dead, target_lp,
                max_beta=2e5, step_factor=0.5, beta_inc=2.0,
            )
            if val_gd < _score(seq_lp):
                seq_lp = seq_gd.copy()
                print(f"[{time.time():.2f}] Post‑LP Adam new best={val_gd:.6f}")

        # ------------------------------------------------------------------
        # Upscale to final size
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 90:
            n_final = 20000
        elif remaining > 45:
            n_final = 16000
        else:
            n_final = max(n_expl, int(remaining * 200))
        target_final = np.sqrt(2.0 * n_final)
        seq_big = _resize_sequence(seq_lp, n_final)
        seq_big = _project_to_simplex(seq_big, target_final)
    else:
        # Not enough time – upscale the current best directly
        remaining = deadline - time.time()
        if remaining > 90:
            n_final = 20000
        elif remaining > 45:
            n_final = 16000
        else:
            n_final = n_expl
        target_final = np.sqrt(2.0 * n_final)
        seq_big = _resize_sequence(best_seq, n_final)
        seq_big = _project_to_simplex(seq_big, target_final)

    # ------------------------------------------------------------------
    # Phase 4 – deterministic reorderings
    # ------------------------------------------------------------------
    candidates = [
        seq_big,
        _alternating_order(seq_big),
        _even_spread_order(seq_big),
        _random_permutation(seq_big, rng),
    ]
    seq_big = min(candidates, key=_max_convolution_fft)
    best_score = _score(seq_big)
    print(f"[{time.time():.2f}] After reordering score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final sub‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 6:
        sub_dead = deadline - 4
        seq_sub2, val_sub2 = _subgradient_descent(
            seq_big, sub_dead, target_final,
            step_factor=0.45, gamma=0.995,
        )
        if val_sub2 < best_score:
            best_score = val_sub2
            seq_big = seq_sub2.copy()
            print(f"[{time.time():.2f}] Final sub‑grad improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – final deterministic mass‑transfer refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 2:
        mt_dead = time.time() + min(6.0, deadline - time.time() - 1)
        seq_mt, val_mt = _deterministic_mass_transfer(
            seq_big, mt_dead, target_final,
            start_temp=0.25, end_temp=1e-5, low_frac=0.05,
        )
        if val_mt < best_score:
            best_score = val_mt
            seq_big = seq_mt.copy()
            print(f"[{time.time():.2f}] Final mass‑transfer best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries
    # ------------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Optional final evaluation (purely for logging)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    globals()["height_sequence_1"] = final_seq
    return final_seq
