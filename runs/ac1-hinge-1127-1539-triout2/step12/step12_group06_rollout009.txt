def _spread_order(seq, stride):
    sorted_vals = np.sort(seq)[::-1]
    n = len(seq)
    out = np.empty_like(seq)
    pos = 0
    for v in sorted_vals:
        out[pos] = v
        pos = (pos + stride) % n
    return out

"""
Hybrid multi‑start optimizer for the non‑negative step‑function problem.

The algorithm proceeds in several phases:

1.  Choose a moderate length (≈8000) and generate a diverse pool of
    candidate shapes (power‑law, exponential, linear, quadratic,
    sinusoidal, Gaussian, random Dirichlet).  Each candidate is scaled
    to the required ℓ₁‑norm √(2n) by a Euclidean simplex projection.

2.  For every shape evaluate a handful of orderings:
    – identity,
    – alternating (large ↔ left / small ↔ right),
    – reversed alternating,
    – random permutations,
    – a “spread” ordering that places the sorted values on a
      stride coprime to the length (uniformly distributes large values).
    The best sequence among all shape + ordering pairs is kept.

3.  Upscale the current best vector to a larger length (≈15000) by linear
    interpolation and re‑project onto the simplex.

4.  Refine the large vector with three complementary continuous optimizers,
    all projected onto the simplex after each step:
    – a momentum sub‑gradient descent on the exact peak of the
      autocorrelation,
    – an annealed soft‑max gradient descent (high‑β limit approximates the
      true max),
    – a targeted permutation‑search that only swaps indices contributing
      heavily to the current peak.

5.  Perform a short deterministic mass‑transfer pass that moves mass from
    “donor” indices (large contribution to the peak) to “receiver”
    indices (small contribution), accepting moves with a simulated‑annealing
    schedule.

6.  Trim trailing zeros, store the result for possible reuse and return it.

The whole pipeline respects the 1000 s time budget by checking the wall‑clock
time before every expensive phase.
"""

import time
import math
import numpy as np

# Global RNG – used by all helper functions (no closures required)
_GLOBAL_RNG = np.random.default_rng()


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Scoring utility (objective = max convolution entry, because sum = √(2n))
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Return the benchmark score for a (maximum convolution entry)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    # With the simplex constraint s == sqrt(2n) the expression equals max_b
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Shape factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _linear_decrease_sequence(n: int, target_sum: float) -> np.ndarray:
    a = np.linspace(1.0, 0.0, n, dtype=np.float64)
    a *= target_sum / a.sum()
    return a


def _quadratic_decrease_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) / (n - 1)
    a = 1.0 - x ** 2
    a = np.maximum(a, 0.0)
    a *= target_sum / a.sum()
    return a


def _sinusoidal_sequence(n: int, target_sum: float) -> np.ndarray:
    a = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
    a = np.maximum(a, 0.0)
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, target_sum: float) -> np.ndarray:
    mu = (n - 1) / 2.0
    sigma = n / 6.0
    a = np.exp(-0.5 * ((np.arange(n, dtype=np.float64) - mu) / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _triangular_sequence(n: int, target_sum: float) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else np.arange(half, 0, -1, dtype=np.float64)
    a = np.concatenate([left, right])
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Ordering utilities
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → left, 2nd largest → right, 3rd → next left, …."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


def _random_permutation(seq: np.ndarray) -> np.ndarray:
    perm = _GLOBAL_RNG.permutation(seq.shape[0])
    return seq[perm]


def _find_coprime_stride(n: int) -> int:
    """Return a small integer ≥2 that is coprime with n."""
    for s in range(2, n):
        if math.gcd(s, n) == 1:
            return s
    return 1


def _spread_order(seq: np.ndarray, stride: int) -> np.ndarray:
    """Place sorted values on a cyclic stride (uniform spread)."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    pos = 0
    for v in sorted_vals:
        out[pos] = v
        pos = (pos + stride) % n
    return out


# ----------------------------------------------------------------------
# Exact max‑convolution sub‑gradient
# ----------------------------------------------------------------------


def _max_conv_and_subgrad(a: np.ndarray):
    """Return (conv, max_val, kmax, subgrad) for a*a."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    kmax = int(np.argmax(conv))
    max_val = conv[kmax]

    grad = np.zeros_like(a)
    lo = max(0, kmax - (n - 1))
    hi = min(kmax, n - 1)
    idx = np.arange(lo, hi + 1)
    partner = a[kmax - idx]
    grad[lo:hi + 1] = 2.0 * partner
    return conv, max_val, kmax, grad


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Momentum projected sub‑gradient descent on max(conv(a,a))."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_val = _score(a)

    momentum = np.zeros_like(a)
    it = 0
    while time.time() < deadline:
        _, _, _, grad = _max_conv_and_subgrad(a)

        # momentum
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_factor * target_sum / norm
            a -= lr * momentum
        else:
            a -= step_factor * target_sum * grad / (np.linalg.norm(grad) + 1e-12)

        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 250 == 0:
            jitter = _GLOBAL_RNG.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] SubGD it={it:5d} score={best_val:.6f}")

        it += 1

    return best, best_val


# ----------------------------------------------------------------------
# Soft‑max gradient descent (annealed)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    grad = 2.0 * np.fft.irfft(np.fft.rfft(a_rev, n=size) *
                               np.fft.rfft(p, n=size), n=size)[: 2 * n - 1]
    grad = grad[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 1e6,
                               step_factor: float = 0.5,
                               gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """Momentum projected gradient descent on the soft‑max of the convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # jitter
        if it % 250 == 0:
            jitter = _GLOBAL_RNG.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AnnealedGD it={it:5d} β={beta:.1f} score={best_val:.6f}")

        # increase β geometrically
        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy donor→receiver moves with exponential temperature schedule."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]          # contribution weight to the peak
        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = _GLOBAL_RNG.choice(low_idxs)

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = _GLOBAL_RNG.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = _GLOBAL_RNG.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score

def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.1,
                                 end_temp: float = 1e-4,
                                 top_frac: float = 0.02,
                                 low_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """Swap indices that heavily contribute to the peak; Metropolis acceptance."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)

    start_time = time.time()
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = _GLOBAL_RNG.choice(donor_candidates)
        receiver = _GLOBAL_RNG.choice(receiver_candidates)

        if donor == receiver:
            continue

        a[donor], a[receiver] = a[receiver], a[donor]

        new_score = _score(a)

        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] PermSearch new best={best_score:.6f}")

        if new_score < best_score or _GLOBAL_RNG.random() < np.exp(-(new_score - best_score) / max(temp, 1e-12)):
            # keep swap
            pass
        else:
            # revert
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimizes the benchmark score.
    The function respects the 1000 s budget and stores the best vector in the
    global variable `height_sequence_1` for possible reuse.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1: choose a modest length for rapid exploration
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n_expl = 8000
    elif budget_s >= 500:
        n_expl = 6000
    else:
        n_expl = 4000
    target_expl = np.sqrt(2.0 * n_expl)

    # ------------------------------------------------------------------
    # Phase 2: build a diverse pool of seed sequences
    # ------------------------------------------------------------------
    seed_pool = []

    # shapes
    for p in np.linspace(0.35, 0.75, 9):
        seed_pool.append(_power_law_sequence(n_expl, float(p), target_expl))
    for lam in np.linspace(0.0015, 0.015, 7):
        seed_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))
    seed_pool.append(_linear_decrease_sequence(n_expl, target_expl))
    seed_pool.append(_quadratic_decrease_sequence(n_expl, target_expl))
    seed_pool.append(_sinusoidal_sequence(n_expl, target_expl))
    seed_pool.append(_gaussian_sequence(n_expl, target_expl))
    seed_pool.append(_triangular_sequence(n_expl, target_expl))

    # random Dirichlet‑like points
    for _ in range(6):
        a = _GLOBAL_RNG.random(n_expl)
        a *= target_expl / a.sum()
        seed_pool.append(a)

    # previous best (if any)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        seed_pool.append(_resize_sequence(prev, n_expl))

    # jittered copies
    jittered = []
    for seq in list(seed_pool):
        jitter = _GLOBAL_RNG.normal(scale=target_expl * 0.008, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        jittered.append(seq_j)
    seed_pool.extend(jittered)

    # ------------------------------------------------------------------
    # Phase 3: evaluate each seed with several orderings
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    stride = _find_coprime_stride(n_expl)

    for seq in seed_pool:
        if time.time() > deadline - 5.0:
            break

        # identity
        cand = seq.copy()
        sc = _score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
            print(f"[{time.time():.2f}] New best (id) score={best_score:.6f}")

        # alternating orders
        for order_func in (_alternating_order,
                           _reversed_alternating):
            cand = order_func(seq)
            sc = _score(cand)
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time():.2f}] New best (alt) score={best_score:.6f}")

        # random permutations (few tries)
        for _ in range(3):
            cand = _random_permutation(seq)
            sc = _score(cand)
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time():.2f}] New best (rand) score={best_score:.6f}")

        # spread order
        cand = _spread_order(seq, stride)
        sc = _score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
            print(f"[{time.time():.2f}] New best (spread) score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4: upscale to a larger length (more degrees of freedom)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.55, target_expl)

    # Decide final length based on remaining budget
    remaining = deadline - time.time()
    if remaining > 200 and budget_s >= 800:
        n_final = 15000
    elif remaining > 120:
        n_final = 13000
    elif remaining > 60:
        n_final = 11000
    else:
        n_final = n_expl

    target_final = np.sqrt(2.0 * n_final)
    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # Evaluate a few systematic reorderings at the larger size
    candidates = [seq_big,
                  _alternating_order(seq_big),
                  _reversed_alternating(seq_big),
                  _spread_order(seq_big, _find_coprime_stride(n_final)),
                  _random_permutation(seq_big)]
    best_seq = seq_big
    best_score = _score(seq_big)
    for cand in candidates[1:]:
        sc = _score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
            print(f"[{time.time():.2f}] Upscale ordering improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5: continuous refinement on the enlarged vector
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        sub_dead = time.time() + min(12.0, deadline - time.time() - 5.0)
        a_opt, val_opt = _subgradient_descent(best_seq, sub_dead,
                                              target_final,
                                              step_factor=0.6, gamma=0.95)
        if val_opt < best_score:
            best_score, best_seq = val_opt, a_opt.copy()
            print(f"[{time.time():.2f}] SubGD (big) improved to {best_score:.6f}")

    if time.time() < deadline - 25.0:
        anneal_dead = time.time() + min(10.0, deadline - time.time() - 5.0)
        a_opt, val_opt = _annealed_gradient_descent(best_seq, anneal_dead,
                                                    target_final,
                                                    max_beta=2e6,
                                                    step_factor=0.5,
                                                    gamma=0.995)
        if val_opt < best_score:
            best_score, best_seq = val_opt, a_opt.copy()
            print(f"[{time.time():.2f}] Annealed GD improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 6: permutation‑only local search
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        perm_dead = time.time() + min(8.0, deadline - time.time() - 1.0)
        a_opt, val_opt = _permutation_targeted_search(best_seq, perm_dead,
                                                      start_temp=0.15,
                                                      end_temp=1e-5,
                                                      top_frac=0.04,
                                                      low_frac=0.04)
        if val_opt < best_score:
            best_score, best_seq = val_opt, a_opt.copy()
            print(f"[{time.time():.2f}] Permutation search improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 7: final deterministic mass‑transfer pass
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        mt_dead = time.time() + min(5.0, deadline - time.time() - 0.5)
        a_opt, val_opt = _deterministic_mass_transfer(best_seq, mt_dead,
                                                      target_final,
                                                      start_temp=0.3,
                                                      end_temp=1e-5,
                                                      low_frac=0.05)
        if val_opt < best_score:
            best_score, best_seq = val_opt, a_opt.copy()
            print(f"[{time.time():.2f}] Final MT improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 8: trim trailing zeros and finalize
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Optional final evaluation logging (uses the provided evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # pylint: disable=undefined-variable
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as exc:   # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation failed: {exc}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
