"""
Multi‑stage optimiser for the auto‑convolution L∞ problem.

The routine works on a single length (typically 12‑20k) and performs:

1.  Generation of a diverse pool of seed sequences (power‑law,
    exponential, semicircle, Gaussian, random Dirichlet).  Each seed is
    projected onto the ℓ¹‑simplex ∑a = √(2 n).

2.  A short scan of the pool + an alternating‑order re‑arrangement to
    pick the best initial candidate (lowest max‑convolution).

3.  Long soft‑max gradient descent (annealed β) on the smooth
    approximation of the max‑convolution, with projection onto the
    simplex after every step.

4.  Deterministic “mass‑transfer” local search that moves a small
    amount of mass from a high‑contributing index (to the current peak)
    to a low‑contributing one, occasionally accepting uphill moves.

5.  Extensive simulated‑annealing permutation search that swaps a
    high‑contributing index with a low‑contributing one, driven by the
    true contribution to the current peak.  This stage consumes the
    remaining time budget.

6.  A final tiny mass‑transfer polishing and removal of trailing zeros.

All steps respect the non‑negative and ℓ¹‑simplex constraints, and the
function stops automatically before the given time limit (default 1000 s).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)² .  For Σa = √(2n) this equals max(a*a)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection of v onto the simplex {x≥0, Σx = s}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.nonzero(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _semicircle_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.linspace(-1.0, 1.0, n, dtype=np.float64)
    a = np.sqrt(np.clip(1.0 - x * x, 0.0, None))
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, target_sum: float, sigma_factor: float = 0.2) -> np.ndarray:
    sigma = sigma_factor * n
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((idx - n / 2) / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place the values alternating from the left‑most and right‑most slots."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Projected gradient descent on the soft‑max objective
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.6,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Momentum‑based gradient descent with exponentially increasing β."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (small moves)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-4,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy + simulated‑annealing mass transfer between donor/receiver indices."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]          # contribution of each index to the peak

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted permutation search (simulated annealing on index order)
# ----------------------------------------------------------------------


def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-5,
                                 top_frac: float = 0.07,
                                 low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing that swaps a high‑contributing index with a low‑contributing one.
    Uses the true contribution to the current peak of the convolution.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    n = a.shape[0]
    start_time = time.time()

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        kmax_candidates = np.flatnonzero(conv == max_val)
        kmax = rng.choice(kmax_candidates)

        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)

        if donor == receiver:
            continue

        a[donor], a[receiver] = a[receiver], a[donor]

        new_score = _score(a)

        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] Perm it new best={best_score:.6f}")
        else:
            # revert the swap
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The algorithm automatically chooses a problem size (≈12‑20 k) based on
    the total time budget, then runs:
        * diverse initialisation + quick selection,
        * long soft‑max gradient descent,
        * deterministic mass‑transfer,
        * an extended permutation‑only annealing,
        * a final tiny mass‑transfer polishing and trimming.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # tiny safety margin

    # --------------------------------------------------------------
    # Choose target length based on total budget.
    # --------------------------------------------------------------
    if budget_s >= 850:
        n_target = 20000
    elif budget_s >= 650:
        n_target = 16000
    elif budget_s >= 450:
        n_target = 12000
    else:
        n_target = 8000

    target_sum = np.sqrt(2.0 * n_target)

    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # Build a diverse pool of seed sequences.
    # --------------------------------------------------------------
    seed_pool = []

    # power‑law family
    for p in [0.4, 0.5, 0.55, 0.6, 0.65]:
        seed_pool.append(_power_law_sequence(n_target, p, target_sum))

    # exponential family
    for lam in [0.0015, 0.003, 0.006, 0.009, 0.012]:
        seed_pool.append(_exponential_sequence(n_target, lam, target_sum))

    # semicircle shape
    seed_pool.append(_semicircle_sequence(n_target, target_sum))

    # Gaussian shape
    seed_pool.append(_gaussian_sequence(n_target, target_sum, sigma_factor=0.15))
    seed_pool.append(_gaussian_sequence(n_target, target_sum, sigma_factor=0.25))

    # random Dirichlet‑like points
    for _ in range(4):
        a = rng.random(n_target)
        a *= target_sum / a.sum()
        seed_pool.append(a)

    # linear decreasing (useful for alternating order)
    lin = np.linspace(1.0, 0.0, n_target)
    lin *= target_sum / lin.sum()
    seed_pool.append(lin)

    # If a previous best exists, upscale it
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        # simple linear interpolation to the new length
        old_len = prev.shape[0]
        old_x = np.linspace(0.0, 1.0, old_len)
        new_x = np.linspace(0.0, 1.0, n_target)
        up = np.interp(new_x, old_x, prev)
        up = _project_to_simplex(up, target_sum)
        seed_pool.append(up)

    # add alternating‑order versions of all seeds (helps spread large values)
    alt_pool = [_alternating_order(s) for s in list(seed_pool)]
    seed_pool.extend(alt_pool)

    # --------------------------------------------------------------
    # Quick evaluation of the pool, keep the best seed.
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    for seq in seed_pool:
        sc = _score(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
    print(f"[{time.time():.2f}] Initial best score (seed) = {best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – long annealed gradient descent.
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    gd_deadline = time.time() + min(remaining * 0.55, 500.0)   # never exceed half a minute if overall budget tiny
    if gd_deadline > deadline:
        gd_deadline = deadline - 0.5
    best_seq, best_score = _annealed_gradient_descent(
        best_seq, gd_deadline, target_sum,
        max_beta=2e6, step_factor=0.5, gamma=0.995,
    )
    print(f"[{time.time():.2f}] After gradient descent best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – deterministic mass‑transfer refinement.
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        mt_deadline = time.time() + min(remaining * 0.15, 120.0)
        best_seq, best_score = _deterministic_mass_transfer(
            best_seq, mt_deadline, target_sum,
            start_temp=0.5, end_temp=1e-4, low_frac=0.05,
        )
        print(f"[{time.time():.2f}] After mass‑transfer best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – extended permutation‑only annealing (uses all left time).
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        perm_deadline = deadline - 0.2
        best_seq, best_score = _permutation_targeted_search(
            best_seq,
            perm_deadline,
            start_temp=0.6,
            end_temp=1e-5,
            top_frac=0.07,
            low_frac=0.07,
        )
        print(f"[{time.time():.2f}] After permutation search best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – final tiny mass‑transfer polishing (if a few seconds remain).
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 1.0:
        mt2_dead = time.time() + min(5.0, remaining - 0.5)
        best_seq, best_score = _deterministic_mass_transfer(
            best_seq,
            mt2_dead,
            target_sum,
            start_temp=0.3,
            end_temp=1e-5,
            low_frac=0.05,
        )
        print(f"[{time.time():.2f}] Final MT polishing best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness.
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.flatnonzero(arr > 1e-12)
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # --------------------------------------------------------------
    # Optional logging of the actual evaluation (does not affect the harness).
    # --------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for possible reuse in a later run
    globals()["height_sequence_1"] = final_seq
    return final_seq
