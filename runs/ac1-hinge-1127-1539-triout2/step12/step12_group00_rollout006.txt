def _max_convolution_fft(a):
    n = a.shape[0]
    size = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa*fa, n=size)[:2*n-1]
    return float(conv.max())

def _project_to_simplex(v, s):
    # Euclidean projection onto simplex (nonnegative entries sum to s)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv/np.arange(1,n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a):
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))

def _mass_transfer(a, deadline, target_sum, start_temp=0.3, end_temp=1e-5, low_frac=0.05):
    # Implementation similar to earlier deterministic mass transfer but simplified.
    # uses stochastic moves.
    cur = a.copy()
    cur_score = _score(cur)
    best = cur.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start = time.time()
    conv = np.fft.irfft(np.fft.rfft(cur, n=1<<((2*len(cur)-1).bit_length())) * np.fft.rfft(cur, n=1<<((2*len(cur)-1).bit_length())), n=1<<((2*len(cur)-1).bit_length()))[:2*len(cur)-1]
    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp/start_temp) ** tau
        # choose index of max conv
        kmax = int(np.argmax(conv))
        # choose donor (index involved in max conv)
        lo = max(0, kmax - (len(cur)-1))
        hi = min(kmax, len(cur)-1)
        idx = np.arange(lo, hi+1)
        # compute products for each index
        prods = cur[idx] * cur[kmax-idx]
        donor = idx[np.argmax(prods)]
        # choose receiver among low product indices
        low_cnt = max(1, int(len(idx)*low_frac))
        low_idxs = idx[np.argsort(prods)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        delta = rng.random() * max_delta
        if delta <= 0:

        cur[donor] -= delta
        cur[receiver] += delta

def _de_optimize(population, target_sum, deadline):
    n = population[0].shape[0]
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()
    # compute initial scores
    scores = [ _score(ind) for ind in population ]
    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_score = scores[best_idx]
    while time.time() < deadline:
        for i in range(len(population)):
            # select three distinct other indices
            idxs = list(range(len(population)))
            idxs.remove(i)
            a,b,c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + 0.5 * (population[b] - population[c])
            # project to simplex
            mutant = _project_to_simplex(mutant, target_sum)
            # crossover
            cross_mask = rng.random(n) < 0.9
            trial = np.where(cross_mask, mutant, population[i])
            # ensure sum constraint
            trial = _project_to_simplex(trial, target_sum)
            trial_score = _score(trial)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()
        # optional early exit if time left less than small margin
    return best_seq, best_score

"""
Hybrid differential‑evolution optimizer for step‑function minimisation.

The algorithm:
1. Generate a diverse pool of seed sequences (power‑law, exponential, random,
   previous best). Project each seed to the ℓ¹‑simplex ‖a‖₁ = √(2n).
2. Initialise a population for a differential‑evolution (DE) search.
   The DE mutation, crossover and projection keep all points feasible.
3. Run DE until the time budget is almost exhausted (≈70 % of time).
4. Refine the best individual with a deterministic mass‑transfer
   (simulated‑annealing style) and a short sub‑gradient descent.
5. Apply a few simple reorderings (alternating / even‑spread) and keep the
   ordering with the smallest max‑convolution.
6. Trim trailing near‑zero entries and return the list of floats.
"""

import time
import numpy as np

def _max_convolution_fft(a: np.ndarray) -> float:
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())

def _conv_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a)/(∑a)²."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))

def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a

def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a

def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a

def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)

def _alternating_order(seq: np.ndarray) -> np.ndarray:
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    l, r = 0, n - 1
    left = True
    for v in sorted_vals:
        if left:
            out[l] = v
            l += 1
        else:
            out[r] = v
            r -= 1
        left = not left
    return out

def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out

def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy mass‑transfer with a Metropolis acceptance rule."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score

def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]
        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        it += 1
    return best_a, best_val

def _de_optimize(population: list[np.ndarray],
                 target_sum: float,
                 deadline: float,
                 F: float = 0.5,
                 CR: float = 0.9) -> tuple[np.ndarray, float]:
    """Differential‑evolution on the simplex (projected after each step)."""
    n = population[0].shape[0]
    rng = np.random.default_rng()
    scores = [_score(ind) for ind in population]
    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(len(population)):
            # pick three distinct individuals other than i
            idxs = list(range(len(population)))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = _project_to_simplex(mutant, target_sum)
            # binomial crossover
            cross = rng.random(n) < CR
            # ensure at least one component from mutant
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, population[i])
            trial = _project_to_simplex(trial, target_sum)

            trial_score = _score(trial)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()
        # optional early exit if only a few seconds left
        if time.time() > deadline - 2.0:
            break
    return best_seq, best_score

"""
Hybrid differential‑evolution optimiser for the step‑function problem.

The algorithm proceeds through the following stages:

1. **Seeding** – generate a diverse collection of feasible sequences
   (power‑law, exponential, uniform Dirichlet‑like, and the previous
   best if available) and project them onto the ℓ¹‑simplex
   ∑a = √(2 n).

2. **Differential Evolution (DE)** – maintain a modest population,
   apply the classic DE mutation `a + F·(b−c)`, binomial crossover and a
   projection onto the simplex.  All individuals stay feasible and the
   best individual is tracked.

3. **Local polishing** – a deterministic mass‑transfer routine
   (simulated‑annealing style) further reduces the max‑convolution,
   followed by a short sub‑gradient descent on the exact objective.

4. **Reordering** – three simple permutations (original, alternating,
   even‑spread) are tested; the ordering with the smallest
   max‑convolution is kept.

5. **Trimming & return** – trailing near‑zero entries are removed,
   the resulting list is stored globally (so that the next run may start
   from it) and returned.

The whole process respects the 1000‑second budget, automatically
adjusting the time allocated to each phase.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)² (for Σa=√(2n) this equals max‑conv)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised to the required ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex (Dirichlet‑like)."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)

def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """
    Spread the large values uniformly by placing them on a cyclic sequence
    with a step coprime to the length.
    """
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]          # descending
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:               # ensure a full cycle
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out

        for i in range(len(population)):
            # pick three distinct mates different from i
            idxs = list(range(len(population)))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = _project_to_simplex(mutant, target_sum)

            # binomial crossover (guarantee at least one component from mutant)
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, population[i])
            trial = _project_to_simplex(trial, target_sum)

            trial_score = _score(trial)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()
        # early exit when only a few seconds remain
        if time.time() > deadline - 2.0:
            break
    return best_seq, best_score

# Main entry point for the competition harness

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid search for a non‑negative step function with a low evaluation score.
    The routine builds a modest population, runs a projected differential‑evolution,
    then refines the best result with deterministic mass‑transfer and a short
    sub‑gradient descent.  Finally a few simple reorderings are tried and the
    best ordering is returned.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # keep a safety margin

    # ------------------------------------------------------------------
    # Problem size (larger n usually gives a richer search space)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 8000
    elif budget_s >= 400:
        n = 6000
    else:
        n = 4000
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Build a pool of diverse seeds
    # ------------------------------------------------------------------
    seeds = []

    for p in np.linspace(0.2, 1.0, 9):
        seeds.append(_power_law_sequence(n, float(p), target_sum))

    for lam in np.linspace(0.001, 0.02, 6):
        seeds.append(_exponential_sequence(n, float(lam), target_sum))

    for _ in range(8):
        seeds.append(_random_simplex(n, target_sum, rng))

    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        seeds.append(_resize_sequence(prev, n))

    # jitter every seed a little to increase diversity
    for seq in list(seeds):
        jitter = rng.normal(scale=target_sum * 0.02, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        seeds.append(seq_j)

    # ------------------------------------------------------------------
    # Initialise the DE population
    # ------------------------------------------------------------------
    pop_size = max(20, int(n / 400))   # about 20‑30 individuals
    population = []

    # start with the best seed we have
    best_seed = min(seeds, key=_score)
    population.append(best_seed.copy())

    while len(population) < pop_size:
        s = rng.choice(seeds)
        population.append(_project_to_simplex(s.copy(), target_sum))

    # ------------------------------------------------------------------
    # DE phase (≈68 % of the total budget)
    # ------------------------------------------------------------------
    de_deadline = start_time + 0.68 * budget_s
    if de_deadline > deadline - 5.0:
        de_deadline = deadline - 5.0
    best_seq, best_score = _de_optimize(population, target_sum, de_deadline)

    print(f"[{time.time():.2f}] DE finished, best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Deterministic mass‑transfer polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        mt_dead = time.time() + min(15.0, deadline - time.time() - 7.0)
        seq_mt, score_mt = _deterministic_mass_transfer(
            best_seq, mt_dead, target_sum,
            start_temp=0.4, end_temp=1e-5, low_frac=0.07)
        if score_mt < best_score:
            best_score, best_seq = score_mt, seq_mt.copy()
            print(f"[{time.time():.2f}] After MT best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Short sub‑gradient descent refinement (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sg_dead = time.time() + min(10.0, deadline - time.time() - 3.0)
        seq_sg, score_sg = _subgradient_descent(
            best_seq, sg_dead, target_sum,
            step_factor=0.45, gamma=0.995)
        if score_sg < best_score:
            best_score, best_seq = score_sg, seq_sg.copy()
            print(f"[{time.time():.2f}] After sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Try a few simple reorderings that often lower the max‑convolution
    # ------------------------------------------------------------------
    candidates = [best_seq,
                  _alternating_order(best_seq),
                  _even_spread_order(best_seq)]
    best_seq = min(candidates, key=_max_convolution_fft)
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] After reordering best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Record for the next run and return
    # ------------------------------------------------------------------
    globals()["height_sequence_1"] = final_seq
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass
    return final_seq
