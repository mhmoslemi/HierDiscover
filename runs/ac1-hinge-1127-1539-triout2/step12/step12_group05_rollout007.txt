"""
Propose a low‑score step‑function.

Algorithm (time‑budget aware)
------------------------------

1. **Problem size** – pick n≈12000 when the wall‑clock budget is large,
   otherwise a smaller n.  The simplex target sum is S=√(2 n).

2. **Parametric pre‑search** – generate many candidates from a few
   classic families (power‑law, broken power‑law with a scaling factor,
   exponential‑power‑law).  The best candidate after this cheap scan
   becomes the starting point.

3. **Optional L‑BFGS refinement** – if SciPy is available we re‑parameterise
   the sequence by a soft‑max (a_i=S·exp(y_i)/∑exp(y)) and run a few
   L‑BFGS iterations on the exact sub‑gradient of max(a∗a).  This
   quickly drives down the peak of the convolution.

4. **Guided exact mass‑transfer** – an aggressive line‑search that moves
   mass from indices that increase the current peak towards indices that
   decrease it, preserving the ℓ₁‑norm.  Repeated until no further
   reduction is possible.

5. **Light random moves** – occasional “swap‑mass” perturbations to jump
   out of plateaus.

6. **Final Adam polish** – a short projected Adam run on a soft‑max
   approximation (very high β) to fine‑tune the result.

All phases respect the simplex constraint (a_i≥0, Σa_i=S) and stop
before the supplied deadline.  The returned list is trimmed of
near‑zero tail entries.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (linear convolution)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft_pair(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution x*y using a real FFT."""
    n = x.size
    size = 1 << ((2 * n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Scoring (the benchmark objective)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Benchmark score (identical to evaluate_sequence up to the simplex scaling)."""
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x >= 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _broken_power_law_sequence(
    n: int, p1: float, p2: float, alpha: float, target_sum: float
) -> np.ndarray:
    K = max(1, int(alpha * n))
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.empty_like(idx)
    a[:K] = idx[:K] ** (-p1)
    a[K:] = idx[K:] ** (-p2)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (exact)
# ----------------------------------------------------------------------


def _compute_gradient_for_max(a: np.ndarray, max_idx: np.ndarray) -> np.ndarray:
    """Exact sub‑gradient of max(conv) w.r.t. a (averaged over all maximal indices)."""
    n = a.size
    g = np.zeros_like(a)
    for k in max_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        # each contribution is 2 * a[k - idx]
        g[idx] += 2.0 * a[k - idx]
    if max_idx.size > 0:
        g /= max_idx.size
    return g


# ----------------------------------------------------------------------
# Guided exact mass‑transfer (Phase B of the original pipeline)
# ----------------------------------------------------------------------


def _guided_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Greedy reduction of the maximal convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        # second largest distinct value
        mask_max = conv >= max_val - tol
        if np.any(~mask_max):
            second_max = conv[~mask_max].max()
        else:
            second_max = 0.0
        max_idx = np.where(mask_max)[0]

        # exact sub‑gradient (averaged)
        g = _compute_gradient_for_max(a, max_idx)

        # direction preserving the ℓ₁‑norm
        sum_g = g.sum()
        d = -g + sum_g / n * np.ones_like(a)  # Σd = 0

        # maximal admissible step before hitting a zero
        neg = d < 0
        if np.any(neg):
            t_nonneg = np.min(a[neg] / -d[neg])
        else:
            t_nonneg = np.inf

        # linearised change of the convolution in direction d
        delta = 2.0 * _conv_fft_pair(a, d)  # length 2n‑1
        delta_k = delta[max_idx].min()      # most favourable slope among maximal entries

        if delta_k >= 0:
            break   # cannot decrease the peak in this direction

        # step that would bring the current max down to the second max
        t_to_second = (max_val - second_max) / (-delta_k)
        t = min(t_nonneg, t_to_second) * 0.99   # safety factor

        if t <= 0:
            break

        a += t * d
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score, best_a = cur_score, a.copy()
            print(f"[{time.time():.2f}] GuidedMT it={it} score={best_score:.6f}")

        # occasional jitter to escape plateaus
        if it % 500 == 0 and rng.random() < 0.05:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_score

# Light random moves (simulated‑annealing style)
# ----------------------------------------------------------------------


def _global_random_move(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
) -> tuple[np.ndarray, float]:
    """Very lightweight random mass redistribution."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    while time.time() < deadline - 0.5:
        top_k = max(1, int(0.02 * n))
        low_k = max(1, int(0.02 * n))
        donor = np.argsort(-a)[rng.integers(top_k)]
        receiver = np.argsort(a)[rng.integers(low_k)]
        if donor == receiver:
            continue
        delta = rng.random() * a[donor] * 0.05
        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(f"[{time.time():.2f}] RandomMove new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected Adam with soft‑max smoothing (Phase A / C)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(a*a)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = conv_full.max()
    exps = np.exp(beta * (conv_full - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, conv_full


def _adam_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 5e5,
    step_factor: float = 0.8,
    gamma: float = 0.999,
) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max approximation."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        best_seq, best_score = resized, val
        print(f"Start from previous best: score={best_score:.6f}")

    # simple power‑law grid (coarse)
    for p in np.arange(0.20, 1.21, 0.10):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_seq, best_score = seq, val

    # exponential‑power‑law grid (coarse)
    for p in np.arange(0.20, 1.21, 0.20):
        for q in np.arange(0.0, 8.01, 0.5):
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_seq, best_score = seq, val

    # random broken power‑law sampling (more flexible)
    phase0_deadline = start_time + min(0.18 * (deadline - start_time), 150.0)
    while time.time() < phase0_deadline:
        p1 = rng.uniform(0.2, 1.0)
        p2 = rng.uniform(0.8, 2.5)
        alpha = rng.uniform(0.05, 0.9)
        scale2 = rng.uniform(0.5, 1.5)        # extra scaling for the tail
        seq = _broken_power_law_sequence(n, p1, p2, alpha, target_sum)
        K = max(1, int(alpha * n))
        seq[K:] *= scale2
        seq = _project_to_simplex(seq, target_sum)
        val = _score(seq)
        if val < best_score:
            best_seq, best_score = seq, val
            print(f"[{time.time():.2f}] New best (broken PL) score={best_score:.6f}")

    print(f"[{time.time():.2f}] After parametric search best score={best_score:.6f}")

    # ---- Phase A : optional L‑BFGS refinement (if SciPy available) ----
    try:
        from scipy.optimize import fmin_l_bfgs_b
        have_scipy = True
    except Exception as exc:
        print(f"SciPy not available: {exc}")
        have_scipy = False

    if have_scipy:
        # objective: max convolution (the denominator is constant)
        def obj_and_grad(y):
            # a = S * softmax(y)
            max_exp = np.max(y)
            exps = np.exp(y - max_exp)
            a = target_sum * exps / exps.sum()
            conv = _conv_fft(a)
            maxb = conv.max()
            max_idx = np.where(conv == maxb)[0]
            grad_a = _compute_gradient_for_max(a, max_idx)
            dot = np.dot(a, grad_a)
            grad_y = a * (grad_a - dot / target_sum)
            return maxb, grad_y.astype(np.float64)

        # initialise y from the current best sequence
        eps = 1e-30
        y0 = np.log(best_seq + eps)

        # decide how many function evaluations we can afford
        remaining = deadline - time.time()
        if remaining > 10.0:
            maxfun = max(20, int(remaining / 0.04))
            print(f"[{time.time():.2f}] Starting L‑BFGS (maxfun={maxfun})")
            try:
                y_opt, f_opt, info = fmin_l_bfgs_b(
                    obj_and_grad,
                    y0,
                    maxfun=maxfun,
                    maxiter=maxfun,
                    pgtol=1e-12,
                    factr=10.0,
                )
                # recover a from y_opt
                a_opt = target_sum * np.exp(y_opt - np.max(y_opt))
                a_opt = a_opt / a_opt.sum() * target_sum
                val_opt = _score(a_opt)
                if val_opt < best_score:
                    best_seq, best_score = a_opt, val_opt
                    print(f"[{time.time():.2f}] L‑BFGS improved best to {best_score:.6f}")
            except Exception as exc2:
                print(f"L‑BFGS failed: {exc2}")

    # ---- Phase B : guided exact mass‑transfer (core refinement) -----
    while time.time() < deadline - 20.0:
        a_ref, val_ref = _guided_mass_transfer(best_seq, deadline, target_sum)
        if val_ref < best_score - 1e-12:
            best_seq, best_score = a_ref, val_ref
            print(f"[{time.time():.2f}] GuidedMT improved best = {best_score:.6f}")

        a_rand, val_rand = _global_random_move(best_seq, deadline, target_sum)
        if val_rand < best_score:
            best_seq, best_score = a_rand, val_rand
            print(f"[{time.time():.2f}] RandomMove improved best = {best_score:.6f}")

        # safety break if we're about to exceed the budget
        if time.time() > deadline - 2.0:
            break

    # ---- Phase C : final Adam polish (high β) ------------------------
    if time.time() + 5.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(
            best_seq,
            deadline,
            target_sum,
            max_beta=1e6,
            step_factor=0.5,
            gamma=0.995,
        )
        if val_fin < best_score:
            best_seq, best_score = a_fin, val_fin
            print(f"[{time.time():.2f}] Final Adam polish improved best = {best_score:.6f}")

    # ---- Post‑processing ------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final evaluation for logging (harmless if not defined)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated benchmark score = {val:.6f}")
    except Exception:
        pass

    return final_seq
