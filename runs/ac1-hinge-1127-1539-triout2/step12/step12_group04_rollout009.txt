"""
Propose a step‑function with a low self‑convolution peak.

The search proceeds in several loosely coupled phases that respect the
given wall‑clock budget (default 1000 s).

* Phase 0 – cheap parametric sweeps (power‑law & exponentially‑tilted
  power‑law) to get a reasonable start.
* Phase 1 – Adam‑style projected gradient descent on a soft‑max
  approximation of the ‑max‑norm.
* Phase 2 – deterministic mass‑transfer (exact DMT) that moves mass
  from a “donor’’ coefficient to a “receiver’’ while keeping the current
  peak of the convolution, guaranteeing a strict improvement.
* Phase 3 – pairwise optimal transfer with a larger donor/receiver
  candidate pool (more aggressive than the original 0.015 × n).
* Phase 4 – a short simulated‑annealing stage that performs random
  donor‑receiver moves and occasionally accepts uphill steps; this helps
  to escape shallow local minima.
* Phase 5 – a final Adam run with an even larger soft‑max β to polish
  the result.

All operations respect the simplex constraint Σa_i = √(2 n) and the
non‑negativity constraint; after the search the trailing near‑zero
entries are trimmed and the plain Python list is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact objective returned by evaluate_sequence when Σa = √(2 n)."""
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto the simplex Σx = s,  x ≥ 0 (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s} (Euclidean projection)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for the fast pre‑search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(
    n: int, p: float, q: float, target_sum: float
) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam‑GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(conv(a,a))."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 2e5,
    step_factor: float = 0.8,
    gamma: float = 0.999,
) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-10
    it = 0
    beta_inc_iter = 250

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to help escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}"
            )

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(
    init: np.ndarray, target_sum: float, deadline: float
) -> tuple[np.ndarray, float]:
    """Short Adam runs with random jittered starts."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(
            start,
            time.time() + run_budget,
            target_sum,
            max_beta=2e5,
            step_factor=0.8,
            gamma=0.995,
        )
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}"
            )

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (core refinement)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Iteratively move mass from a donor (large partner) to a receiver (small
    partner) while preserving the current peak of the convolution."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest entry (ignore current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]  # pick an arbitrary maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]  # a_{k-i}
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        # pick donor with largest contribution partner*donor
        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # pick receiver with smallest partner value (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        # amount needed to equalise the two peak contributions
        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.7, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}"
            )
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Pairwise optimal mass‑transfer (more aggressive than the original)
# ----------------------------------------------------------------------


def _pairwise_optimal_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    donor_frac: float = 0.05,
    receiver_frac: float = 0.05,
    min_improve: float = 1e-7,
) -> tuple[np.ndarray, float]:
    """
    Greedy pairwise mass transfer with enlarged candidate pools.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = _score(a)

    pad_len = n - 1
    base = np.concatenate(
        [np.zeros(pad_len, dtype=np.float64), a, np.zeros(pad_len, dtype=np.float64)]
    )

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k = int(np.argmax(conv))

        nd = max(1, int(donor_frac * n))
        nr = max(1, int(receiver_frac * n))
        donor_idx = np.argsort(-a)[: nd]  # largest coefficients
        receiver_idx = np.argsort(a)[: nr]  # smallest coefficients

        # cache the shifted views for all candidates (views, no copy)
        shifted = {}
        cand = np.unique(np.concatenate((donor_idx, receiver_idx)))
        for i in cand:
            shifted[i] = base[i : i + 2 * n - 1]

        best_new_max = max_val
        best_pair = None
        best_delta = 0.0

        for i in donor_idx:
            shift_i = shifted[i]
            a_i = a[i]
            if a_i <= 0.0:
                continue
            for j in receiver_idx:
                if i == j:
                    continue
                shift_j = shifted[j]
                d = 2.0 * (shift_j - shift_i)  # linear effect of moving mass
                d_k = d[k]
                if d_k >= 0.0:
                    continue  # would not lower the peak

                mask = d > d_k
                if np.any(mask):
                    denom = d[mask] - d_k
                    numer = max_val - conv[mask]
                    feasible = numer / denom
                    delta_max = min(feasible.min(), a_i)
                else:
                    delta_max = a_i

                if delta_max <= 1e-12:
                    continue

                new_max = max_val + d_k * delta_max
                if new_max < best_new_max - min_improve:
                    best_new_max = new_max
                    best_pair = (i, j)
                    best_delta = delta_max

        if best_pair is None:
            break  # no improving move found

        i, j = best_pair
        a[i] -= best_delta
        a[j] += best_delta

        # occasional re‑projection to keep Σa exact
        if int(time.time() * 10) % 47 == 0:
            a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            print(
                f"[{time.time():.2f}] Pairwise transfer i={i} j={j} Δ={best_delta:.3e} new_score={best_score:.6f}"
            )

    return a, best_score


# ----------------------------------------------------------------------
# Simulated annealing exploration phase
# ----------------------------------------------------------------------


def _annealing_search(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    top_frac: float = 0.02,
    low_frac: float = 0.02,
    max_move_frac: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Random donor‑receiver moves with a simple temperature schedule.
    The current sequence is mutated; uphill moves are allowed with a
    probability decreasing linearly in time.
    """
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total = deadline - start_time
    while time.time() < deadline:
        # dynamic temperature – start about 0.01 * cur_score and go to 0
        elapsed = time.time() - start_time
        T = (1.0 - elapsed / total) * max(1e-4, 0.01 * cur_score)

        top_k = max(1, int(top_frac * n))
        low_k = max(1, int(low_frac * n))

        donors = np.argsort(-a)[:top_k]
        receivers = np.argsort(a)[:low_k]

        donor = donors[rng.integers(top_k)]
        receiver = receivers[rng.integers(low_k)]
        if donor == receiver:
            continue

        max_delta = a[donor] * max_move_frac
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        # maintain exact sum (the move itself preserves the sum)
        a_new = np.maximum(a_new, 0.0)  # just in case of numerical noise
        a_new = _project_to_simplex(a_new, target_sum)

        new_score = _score(a_new)

        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                print(
                    f"[{time.time():.2f}] Anneal improved best={best_score:.6f}"
                )
        else:
            # uphill move?
            accept_prob = np.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < accept_prob:
                a = a_new
                cur_score = new_score

    return best_a, best_score


# ----------------------------------------------------------------------
# Global random mass‑transfer (exploration – unchanged)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(
    a0: np.ndarray, deadline: float, target_sum: float
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random move new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the maximum
    self‑convolution (equivalently the score returned by evaluate_sequence).
    The routine respects the given time budget and returns the best sequence
    found as a plain Python list.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n gives more freedom but costs more time)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 24000
    elif budget_s >= 500:
        n = 16000
    else:
        n = 10000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Phase 0 – dense parametric pre‑search
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    p_grid = np.linspace(0.30, 0.85, 25)
    q_grid = np.linspace(0.0, 6.0, 25)

    for p in p_grid:
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # If a historical construction is present, use it as a seed
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score = val
            best_seq = resized.copy()

    # Small jittered population around the current best
    for _ in range(6):
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        seq_j = best_seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        val = _score(seq_j)
        if val < best_score:
            best_score = val
            best_seq = seq_j.copy()

    print(
        f"[{time.time():.2f}] Initial best after parametric search = {best_score:.6f}"
    )

    # --------------------------------------------------------------
    # Phase A – Adam‑style projected gradient descent with restarts
    # --------------------------------------------------------------
    gd_deadline = time.time() + (deadline - time.time()) * 0.30
    while time.time() < gd_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.008, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _gradient_descent_with_restarts(
            start_vec, target_sum, gd_deadline
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] GD improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – Exact deterministic mass‑transfer (core refinement)
    # --------------------------------------------------------------
    while time.time() < deadline - 8.0:
        a_ref, val_ref = _deterministic_mass_transfer_exact2(
            best_seq, deadline, target_sum
        )
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] Exact DMT2 improved best = {best_score:.6f}")

        # short random move to escape plateaus
        a_rand, val_rand = _global_random_mass_transfer(
            best_seq, deadline, target_sum
        )
        if val_rand < best_score:
            best_score = val_rand
            best_seq = a_rand.copy()
            print(f"[{time.time():.2f}] Global random improved best = {best_score:.6f}")

        if time.time() > deadline - 5.0:
            break

    # --------------------------------------------------------------
    # Phase C – New pairwise optimal mass transfer (more aggressive)
    # --------------------------------------------------------------
    if time.time() < deadline - 3.0:
        a_pair, val_pair = _pairwise_optimal_transfer(
            best_seq,
            deadline,
            target_sum,
            donor_frac=0.05,
            receiver_frac=0.05,
        )
        if val_pair < best_score:
            best_score = val_pair
            best_seq = a_pair.copy()
            print(f"[{time.time():.2f}] Pairwise transfer improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase D – Simulated annealing (global exploration)
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        a_anneal, val_anneal = _annealing_search(
            best_seq, deadline, target_sum, top_frac=0.02, low_frac=0.02, max_move_frac=0.07
        )
        if val_anneal < best_score:
            best_score = val_anneal
            best_seq = a_anneal.copy()
            print(f"[{time.time():.2f}] Annealing improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase E – Final polishing with a higher β for the soft‑max
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(
            best_seq,
            deadline,
            target_sum,
            max_beta=1e6,
            step_factor=0.6,
            gamma=0.995,
        )
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final evaluation (harmless if evaluate_sequence is not defined)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
