def propose_candidate(budget_s=1000):

    start = time.time()
    deadline = start + budget_s - 0.5

    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)

    # helper functions as in earlier code, but make them top-level outside propose_candidate

    # define initializers:
    init_seqs = []

    # Warm-start if available
    if 'height_sequence_1' in globals():

    # Uniform
    init_seqs.append(np.full(n, target_sum / n))

    # Power-law with random exponent
    rng = np.random.default_rng()
    for exp in rng.uniform(0.3, 1.0, size=3):
        seq = np.arange(1, n+1, dtype=np.float64) ** (-exp)
        seq = seq * (target_sum / seq.sum())
        init_seqs.append(seq)

    # Random Dirichlet
    for _ in range(2):
        seq = rng.random(n)
        seq = seq * (target_sum / seq.sum())
        init_seqs.append(seq)

    best_seq = None
    best_score = np.inf

    for i, init in enumerate(init_seqs):
        # Ensure time left
        now = time.time()
        if now >= deadline:
            break

        # Phase A: mirror descent for short time (maybe 5% of remaining)
        phase_deadline = now + min(0.05 * (deadline - now), 30.0)  # up to 30 sec
        a, scr = exponential_descent(init, target_sum, phase_deadline)
        if scr < best_score:
            best_score, best_seq = scr, a.copy()
            print(f"[{time.time():.2f}] Init {i} exp descent improved best: {best_score:.6f}")

        # Phase B: subgradient descent for short time (5% of remaining)
        now = time.time()
        if now >= deadline:

        phase_deadline = now + min(0.05 * (deadline - now), 30.0)
        a, scr = subgradient_descent(a, target_sum, phase_deadline)
        if scr < best_score:
            best_score, best_seq = scr, a.copy()
            print(f"[{time.time():.2f}] Init {i} subgrad improved best: {best_score:.6f}")

        # Phase C: annealing for remaining time per init
        now = time.time()
        if now >= deadline:
            break
        # allocate remaining budget proportionally among remaining inits
        remaining_budget = deadline - now

        # approximate: allocate half of remaining to this init
        phase_deadline = now + remaining_budget * 0.5

        if scr < best_score:
            best_score, best_seq = scr, a.copy()

"""
Non‑negative step‑function optimizer for the benchmark.

The algorithm combines several complementary strategies:

1. **Diverse restarts** – a handful of very different initial shapes
   (warm‑start, uniform, random Dirichlet, power‑law, “U‑shaped”)
   are generated.  This mitigates getting stuck in a local basin.

2. **Quick mirror‑descent** – a short multiplicative (mirror) descent
   on a soft‑max surrogate drives the peak down while keeping the
   ℓ₁‑norm fixed.

3. **Sub‑gradient descent** – an exact sub‑gradient step on the true
   max‑convolution further reduces the current worst entry.

4. **Aggressive annealing** – a mass‑transfer simulated‑annealing
   routine with a larger initial temperature, a slower √‑cooling
   schedule and a higher allowed transfer size (up to 10 % of the
   donor mass).  Moves are biased towards the current peak but also
   contain random components for exploration.

The whole search respects the supplied time budget (default 1000 s) and
keeps the best sequence found across all restarts in the global variable
`height_sequence_1`.  The final sequence is trimmed of trailing zeros and
returned as a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (sum(a)²).
    When Σa = √(2 n) this equals max(a*a).
    """
    n = a.shape[0]
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:          # safeguard – should never happen
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Soft‑max surrogate utilities (mirror descent)
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of the max‑convolution.
    Returns (grad, full_convolution) where grad has the same length as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    # soft‑max probabilities
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # gradient (see derivation in the reference implementation)
    a_rev = a[::-1]
    grad_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(grad_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Phase A – exponential (mirror) descent on the soft‑max surrogate
# ----------------------------------------------------------------------


def exponential_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Mirror descent on the soft‑max surrogate."""
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        # occasional jitter to avoid stagnation
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpDescent it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Sub‑gradient descent on the true max‑convolution entry."""
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # exact sub‑gradient for the max entry
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        # linear schedule for the learning rate
        elapsed = time.time() - start
        total = deadline - start
        eta_start = 2.0
        eta_end = 0.2
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – aggressive simulated‑annealing mass‑transfer refinement
# ----------------------------------------------------------------------


def anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Simulated‑annealing that moves small amounts of mass from “bad’’
    indices (large contribution to the current peak) to “good’’ indices
    (low height).  The temperature starts at 100 % of the current score
    and decays with a sqrt‑schedule; the maximal transfer per step is
    enlarged to 10 % of the donor mass.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 1.0           # start temperature = current score
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))   # √‑cooling

        # locate the current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – pick an index with relatively small height
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # size of the transfer (scaled by temperature)
        # allow up to 10 % of donor mass (or 1 % of the total sum)
        max_delta = min(a[donor] * 0.10, target_sum * 0.01)
        if max_delta <= 0:
            continue

        # scaling factor grows with temperature (linear in T/T0)
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # -----------------------------------------------------------------
        # O(n) update of the convolution (exact)
        # -----------------------------------------------------------------
        b_new = b.copy()

        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a

        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        # Metropolis acceptance based on *current* value
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved best={best_val:.6f}")

        # periodic full recomputation to avoid drift
        if it % 5000 == 0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val

# Main entry point – orchestrates multiple restarts and phases
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the supplied time budget (seconds) and returns
    the best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5       # small safety margin

    # Choose problem size (larger n gives finer granularity but costs time)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Build a diverse pool of initial sequences
    # --------------------------------------------------------------
    rng = np.random.default_rng()
    init_pool = []

    # 1) warm‑start if available
    if "height_sequence_1" in globals():
        init_pool.append(
            _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
        )

    init_pool.append(np.full(n, target_sum / n))

    # 3) random Dirichlet (simple random positive numbers)

        seq = rng.random(n)
        seq = seq * (target_sum / seq.sum())
        init_pool.append(seq)

        seq = np.arange(1, n + 1, dtype=np.float64) ** (-exp)

    for α in rng.uniform(0.4, 1.0, size=3):
        i = np.arange(n, dtype=np.float64)
        seq = (i + 1) ** (-α) + (n - i) ** (-α)
        seq = seq * (target_sum / seq.sum())
        init_pool.append(seq)

    # --------------------------------------------------------------
    # Main loop over the initial candidates
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    restarts_done = 0

    for init_idx, init_seq in enumerate(init_pool):
        now = time.time()
        if now >= deadline:
            break

        remaining = deadline - now
        # give each restart at most 30 % of the total budget (to keep
        # several restarts affordable)

        # -----------------------------------------------------------------

        if val_exp < best_score:
            best_score, best_seq = val_exp, a_exp.copy()
            print(f"[{time.time():.2f}] Restart {restart_idx} ExpDescent improved best={best_score:.6f}")

        # -----------------------------------------------------------------
        # Phase B – short sub‑gradient descent (≈5 % of allocated time)
        # -----------------------------------------------------------------
        if time.time() < restart_deadline:
            phase_deadline = time.time() + min(0.05 * (restart_deadline - time.time()), 12.0)
            a_sub, val_sub = subgradient_descent(a_exp, target_sum, phase_deadline)
            if val_sub < best_score:
                best_score, best_seq = val_sub, a_sub.copy()
                print(f"[{time.time():.2f}] Restart {restart_idx} SubGrad improved best={best_score:.6f}")

        # -----------------------------------------------------------------
        # Phase C – long annealing (consume the rest of the budget for this restart)
        # -----------------------------------------------------------------
        if time.time() < restart_deadline:
            a_ann, val_ann = anneal_mass_transfer(best_seq, target_sum, restart_deadline)
            if val_ann < best_score:
                best_score, best_seq = val_ann, a_ann.copy()
                print(f"[{time.time():.2f}] Restart {restart_idx} Anneal improved best={best_score:.6f}")

        globals()["height_sequence_1"] = best_seq.tolist()
        restarts_done += 1

    # --------------------------------------------------------------
    # Final clean‑up, trimming trailing zeros and final evaluation
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback: simple uniform distribution
        best_seq = np.full(n, target_sum / n)

    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    globals()["height_sequence_1"] = final_seq
    print(f"[{time.time():.2f}] Search finished after {restarts_done} restarts.")
    return final_seq
