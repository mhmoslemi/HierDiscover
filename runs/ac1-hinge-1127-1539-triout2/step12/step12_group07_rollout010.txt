"""
Hybrid optimizer with a long Adam‐smooth‑max phase.

The search proceeds in five stages:

1. **Quick cheap exploration** – on a modest size (≈4 k) a few
   annealed‑gradient runs from diverse seeds (power‑law, exponential,
   random simplex).  The best small sequence is kept.

2. **Upscaling** – the small solution is interpolated to a larger length
   (≈15 k–20 k) and re‑projected onto the ℓ¹‑simplex.  Several deterministic
   reorderings (alternating, even‑spread) are tried and the one with the
   smallest self‑convolution peak is chosen.

3. **Adam optimisation on the soft‑max approximation** – a large‑scale
   Adam‑style projected optimiser is run for as long as the remaining
   budget allows.  The temperature β of the soft‑max is slowly raised so
   that the optimisation finally targets the true maximum of the
   convolution.

4. **Exact sub‑gradient polishing** – a few thousand projected sub‑gradient
   steps (the same update used previously) further reduce the max‑conv.

5. **Final greedy mass‑transfer / permutation search** – a short simulated‑
   annealing mass‑transfer pass is performed, followed by a quick random
   permutation sweep to exploit any remaining ordering freedom.

All steps respect non‑negativity and the ℓ¹‑norm constraint
∑a = √(2 n).  The routine returns the best sequence found before the
global time limit (default 1000 s)."""

import time

import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (linear convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a)/(∑a)² (for ∑a=√(2n) this equals max conv)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised to the required ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex (Dirichlet‑like)."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)

# Reordering helpers

def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """
    Spread the large values uniformly by placing them on a cyclic sequence
    with a step coprime to the length (≈√n).
    """
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out

def _annealed_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 5e5,
    step_factor: float = 0.9,
    gamma: float = 0.98,
) -> tuple[np.ndarray, float]:
    """Momentum projected GD on the soft‑max approximation."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_b = np.max(b)
        exps = np.exp(beta * (b - max_b))
        p = exps / np.sum(exps)

        a_rev = a[::-1]
        Ar_fft = np.fft.rfft(a_rev, n=size)
        Pr_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AGD it={it:5d} β={beta:7.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

def _subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    step_factor: float = 0.45,
    gamma: float = 0.995,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that uses the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]

        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val

# Greedy mass‑transfer with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def _mass_transfer_simulated_annealing(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    start_temp: float = 0.5,
    end_temp: float = 1e-4,
    low_frac: float = 0.06,
) -> tuple[np.ndarray, float]:
    """Mass transfer between “donor’’ and “receiver’’ with Metropolis rule."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)
    n = a.shape[0]

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

    return best_a, best_score

# Random permutation search (fast heuristic)
# ----------------------------------------------------------------------


def _permutation_search(
    base_seq: np.ndarray,
    trials: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Try a number of random reorderings of `base_seq` (keeping the multiset
    of values fixed) and return the ordering with smallest max‑convolution.
    """
    best_seq = base_seq.copy()
    best_val = _max_convolution_fft(best_seq)
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        perm = rng.permutation(base_seq)
        val = _max_convolution_fft(perm)
        if val < best_val:
            best_val = val
            best_seq = perm.copy()
    return best_seq, best_val

    seeds = []

    # power‑law family
    for p in np.linspace(0.1, 2.0, 9):
        seeds.append(_power_law_sequence(n_small, float(p), target_small))

    # exponential family
    for lam in np.linspace(0.0005, 0.025, 7):
        seeds.append(_exponential_sequence(n_small, float(lam), target_small))

    # random simplex seeds
    for _ in range(8):
        seeds.append(_random_simplex(n_small, target_small, rng))

    # previous best, if any
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        seeds.append(_resize_sequence(prev, n_small))

    # jittered copies of all seeds
    jittered = []
    for seq in seeds:
        j = rng.normal(scale=target_small * 0.02, size=n_small)
        seq_j = np.maximum(seq + j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_small)
        jittered.append(seq_j)
    seeds.extend(jittered)

    best_small = None
    best_small_val = np.inf

    # Run a few short annealed‑gradient runs (≈10 s each) until ~30 s left
    while time.time() < deadline - 30.0:
        seed = rng.choice(seeds)
        run_deadline = min(time.time() + 10.0, deadline - 20.0)
        a_opt, val_opt = _annealed_gradient_descent(
            seed, run_deadline, target_small,
            max_beta=8e5, step_factor=0.9, gamma=0.98
        )
        if val_opt < best_small_val:
            best_small_val = val_opt
            best_small = a_opt.copy()
            print(f"[{time.time():.2f}] Small‑phase new best={best_small_val:.6f}")

    if best_small is None:
        best_small = _power_law_sequence(n_small, 0.8, target_small)

    # ------------------------------------------------------------------
    # Phase 1 – upscale to a larger size
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 600:
        n_big = 20000
    elif remaining > 400:
        n_big = 18000
    elif remaining > 250:
        n_big = 15000
    elif remaining > 150:
        n_big = 12000
    else:
        n_big = n_small

    target_big = math.sqrt(2.0 * n_big)

    seq_big = _resize_sequence(best_small, n_big)
    seq_big = _project_to_simplex(seq_big, target_big)

    # deterministic reorderings
    candidates = [seq_big,
                  _alternating_order(seq_big),
                  _even_spread_order(seq_big)]
    seq_big = min(candidates, key=_max_convolution_fft)

    # a few random permutations
    if time.time() < deadline - 5.0:
        perm_dead = time.time() + 5.0
        seq_big, _ = _permutation_search(seq_big, trials=15, deadline=perm_dead)

    print(f"[{time.time():.2f}] Upscaled to {n_big} items, start max={_max_convolution_fft(seq_big):.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation (soft‑max)
    # ------------------------------------------------------------------
    if time.time() < deadline - 200.0:
        adam_dead = deadline - 150.0
        seq_big, _ = _adam_optimize(
            seq_big, adam_dead, target_big,
            max_beta=1e6, init_beta=0.5, beta_factor=2.0,
            beta_step=300, lr_factor=0.5,
            beta1=0.9, beta2=0.999, eps=1e-8
        )
        print(f"[{time.time():.2f}] After Adam max={_max_convolution_fft(seq_big):.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exact sub‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 100.0:
        sub_dead = deadline - 70.0
        seq_big, _ = _subgradient_descent(
            seq_big, sub_dead, target_big,
            step_factor=0.45, gamma=0.995
        )
        print(f"[{time.time():.2f}] After sub‑grad max={_max_convolution_fft(seq_big):.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – permutation refinement (random shuffles)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        perm_dead = time.time() + min(20.0, deadline - time.time() - 20.0)
        trials = 40
        seq_big, _ = _permutation_search(seq_big, trials, perm_dead)
        print(f"[{time.time():.2f}] After permutation max={_max_convolution_fft(seq_big):.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final greedy mass‑transfer (simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        mt_dead = time.time() + min(10.0, deadline - time.time() - 1.0)
        seq_big, _ = _mass_transfer_simulated_annealing(
            seq_big, mt_dead, target_big,
            start_temp=0.3, end_temp=1e-5, low_frac=0.05
        )
        print(f"[{time.time():.2f}] Final MT max={_max_convolution_fft(seq_big):.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries
    # ------------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Optional final evaluation (only for logging)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for possible subsequent runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
