def _kaiser_sequence(n, beta, target_sum):
    i = np.arange(n)
    alpha = (2*i) / (n-1) - 1
    w = i0(beta * np.sqrt(1 - alpha**2)) / i0(beta)
    w /= w.sum()
    w *= target_sum
    return w

"""
Step‑function optimizer.

The objective is 2·n·max(conv(a,a)) / (sum(a)²).  It is scale‑invariant, so we
fix Σa = √(2n) and minimise the maximal self‑convolution entry.
The algorithm consists of:

1.  A fast parametric pre‑search over a variety of analytic families
    (power‑law, exponential, broken power‑law, classic window functions,
    exponential decay, triangular shape).  The best candidate is kept.

2.  Several short Adam‑style runs on a soft‑max approximation of the
    max‑convolution, with an exponentially increasing temperature β.
    These runs act as global “shakes”.

3.  An exact sub‑gradient guided mass‑transfer phase that moves mass from
    indices that increase the current peak towards those that decrease it.
    A conservative line‑search (towards the second‑largest convolution entry)
    guarantees non‑negativity and Σa‑preservation.

4.  A final high‑β Adam polishing phase.

All phases respect the simplex constraint Σa = √(2n) and a wall‑clock
budget (default 1000 s).  The best sequence found is trimmed of trailing
near‑zero entries and returned as a plain Python list.
"""

import time
import math
import numpy as np

# optional: use scipy for the Kaiser window if available
try:
    from scipy.special import i0  # type: ignore
except Exception:  # pragma: no cover
    i0 = None


# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) using a real FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft_pair(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution x*y (full length 2·n‑1) using a real FFT."""
    n = x.size
    size = 1 << ((2 * n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Benchmark score (equal to max(conv) when Σa = √(2n))."""
    n = a.shape[0]
    max_b = _max_conv(a)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (dense pre‑search)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _broken_power_law_sequence(
    n: int, p1: float, p2: float, alpha: float, target_sum: float
) -> np.ndarray:
    K = max(1, int(alpha * n))
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.empty_like(idx)
    a[:K] = idx[:K] ** (-p1)
    a[K:] = idx[K:] ** (-p2)
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _hann_sequence(n: int, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a *= target_sum / a.sum()
    return a


def _hamming_sequence(n: int, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    a *= target_sum / a.sum()
    return a


def _blackman_sequence(n: int, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    a *= target_sum / a.sum()
    return a


def _triangular_sequence(n: int, target_sum: float) -> np.ndarray:
    # symmetric triangle (peak in the middle)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.minimum(i, n - i + 1)
    a = a.astype(np.float64)
    a *= target_sum / a.sum()
    return a


def _kaiser_sequence(n: int, beta: float, target_sum: float) -> np.ndarray:
    if i0 is None:  # fallback to a simple exponential if scipy not available
        return _exponential_sequence(n, beta * 0.1, target_sum)
    i = np.arange(n, dtype=np.float64)
    alpha = (2.0 * i) / (n - 1) - 1.0
    w = i0(beta * np.sqrt(1.0 - alpha ** 2))
    w /= i0(beta)
    w *= target_sum / w.sum()
    return w


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max approximation)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(a*a)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = conv_full.max()
    exps = np.exp(beta * (conv_full - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, conv_full


def _adam_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 5e5,
    step_factor: float = 0.8,
    gamma: float = 0.999,
) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max approximation of the max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)

        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient guided mass‑transfer (Phase B)
# ----------------------------------------------------------------------


def _compute_gradient_for_max(a: np.ndarray, max_idx: np.ndarray) -> np.ndarray:
    """Exact sub‑gradient of max(conv) w.r.t. a (averaged over max indices)."""
    n = a.size
    g = np.zeros_like(a)
    for k in max_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        # contribution 2 * a[k - idx]
        g[idx] += 2.0 * a[k - idx]
    if max_idx.size > 0:
        g /= max_idx.size
    return g


def _guided_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    tol: float = 1e-12,
    alpha_step: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Greedy exact reduction of the maximal convolution entry.
    Moves mass from indices that increase the peak towards indices that
    decrease it, using a line‑search that drives the current peak partly
    towards the second‑largest entry.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        mask_max = conv >= max_val - tol
        # second biggest entry (excluding the current maximal plateau)
        if np.any(~mask_max):
            second_max = conv[~mask_max].max()
        else:
            second_max = 0.0
        max_idx = np.where(mask_max)[0]

        # exact sub‑gradient (averaged over the maximal plateau)
        g = _compute_gradient_for_max(a, max_idx)

        # direction that keeps Σa unchanged
        sum_g = g.sum()
        d = -g + sum_g / n * np.ones_like(a)  # Σ d = 0

        # maximal step before any component would become negative
        neg = d < 0
        if np.any(neg):
            t_nonneg = np.min(a[neg] / -d[neg])
        else:
            t_nonneg = np.inf

        # linearised change of the convolution in direction d
        delta = 2.0 * _conv_fft_pair(a, d)  # length 2n‑1
        delta_max = delta[max_idx]
        delta_k = delta_max.min()  # most favourable index

        if delta_k >= 0:
            # direction does not reduce the peak – abort
            break

        # step needed to bring the current peak down towards the second peak
        t_to_second = (max_val - second_max) / (-delta_k)

        # conservative step (avoid hitting the exact boundary)
        t = min(t_nonneg, t_to_second * alpha_step) * 0.99
        if t <= 0:
            break

        a += t * d
        a = np.maximum(a, 0.0)                     # numerical safety
        a = _project_to_simplex(a, target_sum)     # fix tiny sum drift

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] GuidedMT it={it:4d} score={best_score:.6f}")

        # occasional jitter to escape tiny plateaus
        if it % 500 == 0 and rng.random() < 0.05:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_score

def _global_random_move(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
) -> tuple[np.ndarray, float]:
    """Lightweight simulated‑annealing‑style mass moves."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    while time.time() < deadline - 0.5:
        top_k = max(1, int(0.02 * n))
        low_k = max(1, int(0.02 * n))
        donor = np.argsort(-a)[rng.integers(top_k)]

        if donor == receiver:

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] RandomMove new best={best_score:.6f}")

    return best_a, best_score

            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        _maybe_update(resized)

    print(f"[{time.time():.2f}] After dense pre‑search: best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase A – short Adam runs (exploratory restarts)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    ada_deadline = time.time() + max(5.0, 0.18 * remaining)

    while time.time() < ada_deadline - 1.0:
        jitter = rng.normal(scale=target_sum * 0.012, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _adam_gradient_descent(
            start_vec,
            ada_deadline,
            target_sum,
            max_beta=5e5,
            step_factor=0.75,
            gamma=0.997,
        )
        if val_opt < best_score:
            best_score, best_seq = val_opt, a_opt.copy()
            print(f"[{time.time():.2f}] Adam restart improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – guided exact mass‑transfer (core refinement)
    # ------------------------------------------------------------------
    mt_deadline = deadline - 4.0  # keep a few seconds for final polishing
    if time.time() < mt_deadline:
        a_ref, val_ref = _guided_mass_transfer(best_seq, mt_deadline, target_sum)
        if val_ref < best_score:
            best_score, best_seq = val_ref, a_ref.copy()
            print(f"[{time.time():.2f}] Guided mass‑transfer improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional lightweight random moves (helps escape tiny plateaus)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.5:
        a_rand, val_rand = _global_random_move(best_seq, deadline - 2.0, target_sum)
        if val_rand < best_score:
            best_score, best_seq = val_rand, a_rand.copy()
            print(f"[{time.time():.2f}] Random move improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase C – final polishing with a high‑β Adam run
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(
            best_seq,
            deadline,
            target_sum,
            max_beta=1e6,
            step_factor=0.6,
            gamma=0.995,
        )
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final polish improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return list
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final check (harmless if evaluate_sequence is missing)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated benchmark score = {val:.6f}")
    except Exception:
        pass

    return final_seq
