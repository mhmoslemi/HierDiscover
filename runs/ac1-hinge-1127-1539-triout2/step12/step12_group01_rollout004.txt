def projected_gradient_descent(a0, target_sum, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_val = _score(a)
    start = time.time()
    while time.time() < deadline:
        # compute convolution
        b = _conv_fft(a)
        k_max = int(np.argmax(b))
        # compute gradient
        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi+1)
        g = np.zeros_like(a)
        # For each i = idx, j = k_max - i
        # derivative of b[k_max] w.r.t a[i] is 2 a[j] if i != j else 2 a[i] (since derivative of a[i]*a[j] + a[j]*a[i] but both same)
        # Actually when i and j distinct, derivative is a[j] + a[j] = 2 a[j]; when i=j (i=j=k_max/2 integer), derivative is 2 a[i] also.
        # So we can assign g[i] = 2 * a[k_max - i]
        g[idx] = 2.0 * a[k_max - idx]
        # step size schedule: linear decay
        elapsed = time.time() - start
        total = deadline - start
        eta_start = 0.5
        eta_end = 0.02
        eta = eta_start * (1 - elapsed / total) + eta_end * (elapsed / total)
        # gradient step
        a = a - eta * g
        # project onto simplex
        a = _project_to_simplex(a, target_sum)
        # evaluate
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] PGD iteration improved best={best_val:.6f}")
    return best_a, best_val
```

def annealing(a0, target_sum, deadline):
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()
    rng = np.random.default_rng()
    start = time.time()
    T0 = cur_val * 0.8
    recompute_interval = 5000
    iteration = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        temperature = max(0.0, T0 * (1 - frac**0.5))
        # locate peak
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi+1)
        # contributions
        contributions = a[idx] * a[k_max - idx]
        sum_contrib = contributions.sum()
        if sum_contrib > 0:
            donor = rng.choice(idx, p=contributions / sum_contrib)
        else:
            donor = rng.integers(lo, hi+1)
        # receiver
        inv_weights = 1.0 / (a + 1e-12)
        inv_weights /= inv_weights.sum()
        receiver = rng.choice(n, p=inv_weights)
        if donor == receiver:
            continue
        # max delta
        max_delta = min(a[donor] * 0.1, target_sum * 0.01)
        if max_delta <= 0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0:
            continue
        # update convolution efficiently
        b_new = b.copy()
        # linear parts
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic parts
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta
        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True
        if accept:
            # apply update to a and b
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Annealing improved best={best_val:.6f}")
        # occasional recompute to avoid drift
        if iteration % recompute_interval == 0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Recomputed improved best={best_val:.6f}")
        iteration += 1
    return best_a, best_val

"""Optimization of a non‑negative step function.

The algorithm proceeds in four stages:
1.  Quick parametric search (power‑law / exponential) to obtain a good
    warm‑start.
2.  Mirror descent on a soft‑max surrogate of the max‑convolution.
3.  Projected (sub‑)gradient descent on the true max‑convolution entry.
4.  Long simulated‑annealing mass‑transfer refinement with an aggressive
    schedule and occasional full recomputation.

All phases respect the global time budget (default 1000 s) and the best
sequence found is stored in the global variable ``height_sequence_1`` for
future warm‑starts.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Objective  2·n·max(a*a) / (sum(a)²).
    For Σa = √(2n) this equals max(a*a).
    """
    n = a.shape[0]
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of the max‑convolution.
    Returns (grad, full_convolution) where grad has the same length as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    grad_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(grad_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – mirror descent on soft‑max surrogate
# ----------------------------------------------------------------------


def exponential_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Mirror descent on the soft‑max surrogate."""
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpDescent it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – projected (sub‑)gradient descent on true max‑convolution
# ----------------------------------------------------------------------


def projected_gradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Projected gradient descent on the true max‑convolution entry."""
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    start = time.time()
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        g = np.zeros_like(a)
        # derivative of b[k_max] w.r.t. a[i] = 2 * a[k_max-i]
        g[idx] = 2.0 * a[k_max - idx]

        # linear cooling of step size
        elapsed = time.time() - start
        total = deadline - start
        eta_start = 0.5
        eta_end = 0.02
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        a = a - eta * g
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] PGD it={it:5d} score={best_val:.6f}")

        # occasional jitter to help escape plateaus
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0004, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – aggressive simulated‑annealing mass‑transfer
# ----------------------------------------------------------------------


def annealing_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Simulated annealing that transfers mass between indices.
    Donor atoms are chosen proportionally to their contribution to the
    current peak of the convolution; receivers are favoured if they are
    small.  The temperature follows a sqrt‑schedule, but transfers are
    allowed up to 10 % of donor mass (or 1 % of the total mass) to make
    large moves possible in the early stage.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.80                 # initial temperature
    it = 0
    recompute_interval = 5000

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        # locate current peak
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – prefer small heights
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # size of transfer (larger early on)
        max_delta = min(a[donor] * 0.10, target_sum * 0.01)   # 10 % donor, 1 % total
        if max_delta <= 0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # ---- fast O(n) update of the convolution --------------------
        b_new = b.copy()
        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part Δ*Δ
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        # Metropolis acceptance based on *current* value
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved best={best_val:.6f}")

        # periodic full recomputation to avoid drift
        if it % recompute_interval == 0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val

# Parameter search for a good warm‑start
# ----------------------------------------------------------------------


def param_search_initial(n: int, target_sum: float, deadline: float):
    """Try a few simple families (power‑law, exponential, random) and keep the best."""
    best_a = None
    best_val = np.inf

    # power‑law family a_i = (i+1)^p
    exponents = np.linspace(-0.9, -0.2, 15)
    for p in exponents:
        a = np.arange(1, n + 1, dtype=np.float64) ** p
        a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
    # exponential family a_i = exp(-λ i)
    lambdas = np.linspace(0.0005, 0.02, 15)
    i = np.arange(n, dtype=np.float64)
    for lam in lambdas:
        a = np.exp(-lam * i)
        a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()

    # warm‑start from previously found solution (if any)
    if "height_sequence_1" in globals():
        init = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if init.size != n:
            init = _resize_sequence(init, n)
        init = init * (target_sum / init.sum())
        val = _score(init)
        if val < best_val:
            best_val = val
            best_a = init.copy()

    # a few random decreasing shapes
    rng = np.random.default_rng()
    for _ in range(5):
        rand = rng.random(n)
        a = np.sort(rand)[::-1]               # decreasing
        a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()

    # guarantee we return something even in pathological case
    if best_a is None:
        best_a = np.full(n, target_sum / n, dtype=np.float64)

    print(f"[{time.time():.2f}] Param‑search best score={best_val:.6f}")
    return best_a, best_val


# ----------------------------------------------------------------------

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    # Choose problem size (larger n → finer granularity but higher cost)
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 500:
        n = 15000
    else:
        n = 10000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric warm‑start
    # ------------------------------------------------------------------
    init_deadline = time.time() + 0.03 * (deadline - time.time())
    a_cur, cur_val = param_search_initial(n, target_sum, init_deadline)

    # ------------------------------------------------------------------
    # Phase A – mirror descent on soft‑max surrogate (≈ 8 % of time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        mirror_deadline = time.time() + 0.08 * (deadline - time.time())
        a_mir, val_mir = exponential_descent(a_cur, target_sum, mirror_deadline)
        if val_mir < cur_val:
            a_cur, cur_val = a_mir, val_mir
        print(f"[{time.time():.2f}] After mirror descent best score={cur_val:.6f}")

    # ------------------------------------------------------------------
    # Phase B – projected gradient descent on true max‑convolution (≈ 12 %)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        pgd_deadline = time.time() + 0.12 * (deadline - time.time())
        a_pgd, val_pgd = projected_gradient_descent(a_cur, target_sum, pgd_deadline)
        if val_pgd < cur_val:
            a_cur, cur_val = a_pgd, val_pgd
        print(f"[{time.time():.2f}] After projected gradient best score={cur_val:.6f}")

    # ------------------------------------------------------------------
    # Phase C – long simulated‑annealing refinement (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ann, val_ann = annealing_mass_transfer(a_cur, target_sum, deadline)
        if val_ann < cur_val:
            a_cur, cur_val = a_ann, val_ann
        print(f"[{time.time():.2f}] After annealing best score={cur_val:.6f}")

    # ------------------------------------------------------------------
    # Final clean‑up and evaluation
    # ------------------------------------------------------------------
    arr = np.array(a_cur, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
