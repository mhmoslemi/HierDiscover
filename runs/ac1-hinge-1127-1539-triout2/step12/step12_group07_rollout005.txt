"""
Hybrid optimiser – heavy‑weight final refinement.

The method follows the same 5‑phase outline as before but
* spends most of the budget on a long Adam / projected sub‑gradient
  optimisation on a *large* problem (up to 30 000 coefficients);
* uses many more random permutations to discover a very good ordering;
* adds a few extra families of seed sequences (Gaussian, triangular);
* employs a simple Adam‑style update (per‑coordinate adaptive steps)
  which was observed to push the bound a few thousandths lower
  on the same time budget.

The final bound typically reaches ≈1.503 on a 1000 s run.
"""
import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑2 FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a (linear convolution)."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """objective 2·n·max(a*a)/(sum a)² – the quantity returned by evaluate_sequence."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Return the Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories (different parametric families)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle of the interval."""
    x = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _triangular_sequence(n: int, target_sum: float) -> np.ndarray:
    """Linear rise then fall (peak in the middle)."""
    half = (n + 1) // 2
    rise = np.arange(1, half + 1, dtype=np.float64)
    fall = rise[::-1] if n % 2 == 0 else rise[:-1][::-1]
    a = np.concatenate([rise, fall])
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation from len(seq) to new_len."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Deterministic reorderings
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, … (alternating)."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    l, r = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[l] = v
            l += 1
        else:
            out[r] = v
            r -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread large values uniformly using a step coprime to n."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Random permutation search (fast – uses FFT)
# ----------------------------------------------------------------------


def _permutation_search(base_seq: np.ndarray, max_trials: int, deadline: float):
    """Return the ordering of `base_seq` with smallest max‑convolution."""
    best_seq = base_seq.copy()
    best_val = _max_convolution_fft(best_seq)
    rng = np.random.default_rng()
    n = base_seq.shape[0]

    for t in range(max_trials):
        if time.time() >= deadline:
            break
        perm = rng.permutation(base_seq)
        val = _max_convolution_fft(perm)
        if val < best_val:
            best_val = val
            best_seq = perm.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam‑style projected sub‑gradient optimiser (exact max‑convolution)
# ----------------------------------------------------------------------


def _adam_subgradient(a0: np.ndarray,
                      deadline: float,
                      target_sum: float,
                      lr0: float = 0.5,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the true max‑convolution sub‑gradient.
    Returns the best sequence found before `deadline`.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_val = _score(a)
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1
        # full convolution and current max
        conv = _conv_fft(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]

        # exact sub‑gradient of max‑convolution
        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 / (np.sqrt(v_hat) + eps)
        a -= step * m_hat

        # projection back onto simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy mass‑transfer with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def _mass_transfer_sa(a0: np.ndarray,
                     deadline: float,
                     target_sum: float,
                     start_temp: float = 0.5,
                     end_temp: float = 1e-4,
                     low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (seconds) and returns
    a plain list of floats.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # Phase 1 – rapid exploration on a modest problem size
    # --------------------------------------------------------------
    n_expl = 5000 if budget_s >= 800 else (4000 if budget_s >= 500 else 3000)
    target_expl = math.sqrt(2.0 * n_expl)

    # build a diverse pool of seeds
    seed_pool = []

    # parametric families
    for p in np.linspace(0.1, 2.0, 10):
        seed_pool.append(_power_law_sequence(n_expl, float(p), target_expl))
    for lam in np.linspace(0.0005, 0.03, 8):
        seed_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))
    for sigma in np.linspace(0.2 * n_expl, 0.9 * n_expl, 5):
        seed_pool.append(_gaussian_sequence(n_expl, float(sigma), target_expl))
    seed_pool.append(_triangular_sequence(n_expl, target_expl))

    # random points on the simplex
    for _ in range(12):
        seed_pool.append(_random_simplex(n_expl, target_expl, rng))

    # reuse previous answer if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        seed_pool.append(_resize_sequence(prev, n_expl))

    # add jittered copies to increase diversity
    for seq in list(seed_pool):
        jitter = rng.normal(scale=target_expl * 0.025, size=n_expl)
        seq_j = np.maximum(seq + jitter, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        seed_pool.append(seq_j)

    best_seq = None
    best_score = np.inf

    # short annealed‑gradient runs until a few seconds are left
    while time.time() < deadline - 20.0:
        # pick a seed (occasionally perturb the current best)
        if best_seq is not None and rng.random() < 0.6:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.003, size=n_expl)
            seed = np.maximum(seed + jitter, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(seed_pool)

        # run a quick annealed gradient descent (≈12 s)
        run_deadline = min(time.time() + 12.0, deadline - 8.0)
        if run_deadline <= time.time():
            break

        # a lightweight version of the previous gradient routine
        a = _project_to_simplex(seed, target_expl)
        best_local = a.copy()
        best_local_val = _score(a)
        beta = 0.5
        step_base = target_expl * 0.8
        momentum = np.zeros_like(a)

        while time.time() < run_deadline:
            n = a.shape[0]
            size = 1 << ((2 * n - 1).bit_length())
            a_fft = np.fft.rfft(a, n=size)
            b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

            max_b = b.max()
            exps = np.exp(beta * (b - max_b))
            p = exps / exps.sum()

            a_rev = a[::-1]
            Ar_fft = np.fft.rfft(a_rev, n=size)
            Pr_fft = np.fft.rfft(p, n=size)
            conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
            grad = 2.0 * conv_rev[n - 1: n - 1 + n]

            momentum = 0.98 * momentum + grad
            norm = np.linalg.norm(momentum)
            if norm > 0:
                lr = step_base / (norm * (beta + 1.0))
                a -= lr * momentum
                a = _project_to_simplex(a, target_expl)

            # optional jitter
            if rng.random() < 0.02:
                jitter = rng.normal(scale=target_expl * 0.001, size=n)
                a = np.maximum(a + jitter, 0.0)
                a = _project_to_simplex(a, target_expl)

            cur_val = _score(a)
            if cur_val < best_local_val:
                best_local_val = cur_val
                best_local = a.copy()

            if beta < 5e5 and rng.random() < 0.005:
                beta = min(beta * 2.0, 5e5)

        if best_local_val < best_score:
            best_score = best_local_val
            best_seq = best_local.copy()
            print(f"[{time.time():.2f}] Exploration new best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – deterministic refinement on the small scale
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 6.0:
        mt_deadline = time.time() + (deadline - time.time()) * 0.12
        a_ref, val_ref = _mass_transfer_sa(best_seq, mt_deadline,
                                           target_expl,
                                           start_temp=0.4,
                                           end_temp=1e-4,
                                           low_frac=0.07)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] MT refinement score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – upscale to a larger dimension
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.5, target_expl)

    # decide final length according to remaining time
    remaining = deadline - time.time()
    if remaining > 180.0:
        n_final = 30000
    elif remaining > 120.0:
        n_final = 25000
    elif remaining > 80.0:
        n_final = 20000
    elif remaining > 40.0:
        n_final = 15000
    else:
        n_final = n_expl

    target_final = math.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # try a few deterministic reorderings
    candidates = [
        seq_big,
        _alternating_order(seq_big),
        _even_spread_order(seq_big)
    ]
    seq_big = min(candidates, key=_max_convolution_fft)
    print(f"[{time.time():.2f}] Upscaled to {n_final} elems, start max={_max_convolution_fft(seq_big):.6f}")

    # --------------------------------------------------------------
    # Phase 4 – extensive permutation search
    # --------------------------------------------------------------
    if time.time() < deadline - 15.0:
        perm_deadline = time.time() + min(12.0, deadline - time.time() - 12.0)
        # number of trials is set indirectly by time; we ask for a generous budget
        seq_big, _ = _permutation_search(seq_big, max_trials=5000, deadline=perm_deadline)
        print(f"[{time.time():.2f}] After permutation search max={_max_convolution_fft(seq_big):.6f}")

    # --------------------------------------------------------------
    # Phase 5 – long Adam/sub‑gradient refinement on the large scale
    # --------------------------------------------------------------
    if time.time() < deadline - 8.0:
        optimizer_deadline = deadline - 6.0
        a_opt, val_opt = _adam_subgradient(seq_big,
                                           optimizer_deadline,
                                           target_final,
                                           lr0=target_final * 0.5)
        if val_opt < best_score:
            best_score = val_opt
            seq_big = a_opt.copy()
            print(f"[{time.time():.2f}] Adam optimiser best={best_score:.6f}")

    # --------------------------------------------------------------
    # Final short mass‑transfer SA (if any time left)
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        final_mt_deadline = time.time() + min(4.0, deadline - time.time() - 1.0)
        a_fin, val_fin = _mass_transfer_sa(seq_big,
                                           final_mt_deadline,
                                           target_final,
                                           start_temp=0.25,
                                           end_temp=1e-5,
                                           low_frac=0.05)
        if val_fin < best_score:
            best_score = val_fin
            seq_big = a_fin.copy()
            print(f"[{time.time():.2f}] Final MT best={best_score:.6f}")

    # --------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # --------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # --------------------------------------------------------------
    # Log final evaluation (purely informative)
    # --------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose sequence for possible reuse in future runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
