def _lp_refine(a, target_sum, deadline):
    n = len(a)
    M = _max_convolution_fft(a)  # compute max conv
    # build A matrix
    rows = np.empty(n*n, dtype=np.int32)
    cols = np.empty(n*n, dtype=np.int32)
    data = np.empty(n*n, dtype=np.float64)
    idx = 0
    for i in range(n):
        f_i = a[i]
        rows[idx:idx+n] = np.arange(i, i+n, dtype=np.int32)
        cols[idx:idx+n] = i
        data[idx:idx+n] = f_i
        idx += n
    # shape rows length 2n-1, columns n
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # b = M per row
    b_ub = np.full(2*n-1, M, dtype=np.float64)
    # objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n, dtype=np.float64)
    # bounds 0 <= g <= None
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
    if not res.success:
        return a, _score(a)
    g = res.x
    # project onto simplex of target_sum
    g = _project_to_simplex(g, target_sum)
    return g, _score(g)

"""
Improved optimiser for the step‑function benchmark.

The algorithm combines three ideas that were missing from the previous
pipeline:

1. **LP‑based shape refinement** – For a moderate problem size (≈ 1500–2000)
   we solve a linear programme that maximises the ℓ¹‑mass of a new
   sequence `g` under the linear constraints  (f * g)_k ≤ M,
   where `f` is the current candidate and `M` is its peak convolution.
   The solution often discovers a more favourable distribution of the
   coefficients.

2. **Sub‑gradient descent on the true max‑convolution** (via a smooth
   soft‑max approximation) to quickly reach a good local optimum.

3. **Large‑scale polishing** – after the LP step the best candidate is
   interpolated to a larger length, optionally reordered (alternating
   high/low), and refined with a short high‑β gradient run, a long
   permutation‑only simulated annealing, and a final tiny mass‑transfer
   pass.

All steps respect the ℓ¹‑simplex `∑a = √(2n)` and keep the coefficients
non‑negative.  The routine stops automatically before the time budget
expires and returns the best sequence found.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)² (with Σa=√(2n) this reduces to max)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project the vector v onto the ℓ¹‑simplex of radius s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place the largest, smallest, 2‑nd‑largest, 2‑nd‑smallest …."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Projected gradient descent on the soft‑max objective
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.5,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Momentum‑based gradient descent on the soft‑max surrogate."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (small moves)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-4,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted permutation search (simulated annealing on index order)
# ----------------------------------------------------------------------


def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-5,
                                 top_frac: float = 0.05,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Swap indices that contribute a lot to the current peak with indices that contribute little."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    n = a.shape[0]
    start_time = time.time()

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        # break ties randomly
        kmax_candidates = np.flatnonzero(conv == max_val)
        kmax = rng.choice(kmax_candidates)

        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)

        if donor == receiver:
            continue

        a[donor], a[receiver] = a[receiver], a[donor]

        new_score = _score(a)

        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] PermSearch new best={best_score:.6f}")
        else:
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (max‑convolution bound)
# ----------------------------------------------------------------------


def _lp_refine(a: np.ndarray, target_sum: float, deadline: float) -> tuple[np.ndarray, float]:
    """
    Solve  max Σ g_j   s.t.  g_j ≥ 0  and  (a * g)_k ≤ M  for all k,
    where M = max(a*a).  Return the projected solution and its score.
    """
    n = a.shape[0]
    M = _max_convolution_fft(a)

    # Build the (2n‑1) × n constraint matrix.
    # Row k contains a_i whenever 0 ≤ k‑i < n.
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)

    pos = 0
    for i in range(n):
        rows[pos:pos + n] = np.arange(i, i + n, dtype=np.int32)
        cols[pos:pos + n] = i
        data[pos:pos + n] = a[i]
        pos += n

    A = sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # minimise –Σ g  → maximise Σ g

    # Leave a tiny safety margin for the linear solver.
    time_now = time.time()
    if time_now > deadline:
        return a, _score(a)

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
    if not res.success:
        # solve failed – return the original sequence
        return a, _score(a)

    g = res.x
    g = _project_to_simplex(g, target_sum)
    return g, _score(g)

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:

      4. up‑scale to a larger length (≈ 10‑12 k)

    for _ in range(5):
        a = rng.random(n_base)
        a *= target_base / a.sum()
        init_pool.append(a)

    lin *= target_base / lin.sum()
    init_pool.append(lin)

    # reuse previous best if present

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    best_seq = None

    # ------------------------------------------------------------------

            target_base,
            start_temp=0.5,
            end_temp=1e-4,
            low_frac=0.05,
        )

            print(f"[{time.time():.2f}] Post‑GD MT → best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. LP‑based shape refinement (few iterations)
    # ------------------------------------------------------------------
    while time.time() < deadline - 10.0:
        g, val_g = _lp_refine(best_seq, target_base, deadline - time.time())
        if val_g < best_score - 1e-6:      # require a genuine improvement
            best_score = val_g
            best_seq = g.copy()
            print(f"[{time.time():.2f}] LP refinement → score={best_score:.6f}")
        else:
            break

    # 6. Upscaling to a larger problem size
    # ------------------------------------------------------------------

    seq_big = _project_to_simplex(seq_big, target_final)

        seq_big,
        _alternating_order(seq_big),
        _reversed_alternating(seq_big),
    ]
    best_seq = seq_big
    best_score = _score(seq_big)
    for cand in candidates[1:]:
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
    print(f"[{time.time():.2f}] Upscaled length={n_final} start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final high‑β gradient polishing (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 8.0:
        a_final, val_final = _annealed_gradient_descent(
            best_seq,
            deadline - 4.0,
            target_final,
            max_beta=5e5,
            step_factor=0.5,
            gamma=0.995,
        )
        if val_final < best_score:
            best_score = val_final
            best_seq = a_final.copy()
            print(f"[{time.time():.2f}] Final gradient polishing → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Extended permutation‑only simulated annealing
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        perm_dead = deadline - 0.5
        a_perm, val_perm = _permutation_targeted_search(
            best_seq,
            perm_dead,
            start_temp=0.5,
            end_temp=1e-5,
            top_frac=0.05,
            low_frac=0.05,
        )
        if val_perm < best_score:
            best_score = val_perm
            best_seq = a_perm.copy()
            print(f"[{time.time():.2f}] Permutation search final → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 9. Tiny deterministic mass‑transfer pass (final polishing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        mt_dead = time.time() + min(4.0, deadline - time.time() - 0.2)
        a_mt2, val_mt2 = _deterministic_mass_transfer(
            best_seq,
            mt_dead,
            target_final,
            start_temp=0.3,
            end_temp=1e-5,
            low_frac=0.05,
        )
        if val_mt2 < best_score:
            best_score = val_mt2
            best_seq = a_mt2.copy()
            print(f"[{time.time():.2f}] Final MT polishing → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 10. Trim trailing zeros and return
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional logging of the true benchmark value
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for possible reuse in later calls
    globals()["height_sequence_1"] = final_seq
    return final_seq
