"""
Hybrid search for a non‑negative step function with a low self‑convolution peak.

Strategy
--------
1. Generate a diverse pool of seed sequences (power‑law, exponential, random simplex).
2. Run many short projected sub‑gradient refinements on a moderate size problem
   (≈8k–10k entries) and keep the best candidate.
3. Down‑scale the best candidate to a tractable size (≈1500) and run a dedicated
   linear program that directly minimises the maximal cross‑convolution f∗g.
   The LP solution is mixed with the current sequence and the process is repeated.
4. Upscale the refined sequence to a large length (≈15k–20k), apply a few deterministic
   reorderings (alternating, even‑spread, cyclic shifts) and keep the ordering with
   the smallest convolution peak.
5. Polish the large‑scale candidate with a long sub‑gradient descent followed by a
   deterministic mass‑transfer simulated‑annealing pass.
All updates respect non‑negativity and the ℓ¹‑simplex constraint (the overall scaling
does not affect the objective).  The routine stops before the time budget expires and
returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex Σx = s, x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Scoring utilities (the evaluation function without the final division)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)² (for Σa=√(2n) this equals max conv)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised to the required ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex (Dirichlet‑like)."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread large values uniformly on a cyclic sequence."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Sub‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        # indices achieving (near) the max
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]
        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (simulated annealing)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy mass moves with a decaying Metropolis acceptance rule."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            it += 1
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement – minimise cross‑convolution bound
# ----------------------------------------------------------------------


def _lp_minimize_cross_max(f: np.ndarray,
                           target_sum: float) -> tuple[np.ndarray, float] | None:
    """
    Solve a LP that finds g (and scalar M) minimising   M
    subject to   (f * g)[k] ≤ M  for all k,   g ≥ 0,   Σg = target_sum.
    Returns (g, M) or None on failure.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    # Build dense A_ub matrix  (rows × (n+1))
    A_ub = np.zeros((rows, n + 1), dtype=np.float64)
    for k in range(rows):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        for i in range(lo, hi + 1):
            j = k - i
            A_ub[k, j] = f[i]                # coefficient of g_j
        A_ub[k, n] = -1.0                     # -M term

    b_ub = np.zeros(rows, dtype=np.float64)

    # Equality: Σ g_i = target_sum
    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum], dtype=np.float64)

    # Bounds: g_i ≥ 0, M ≥ 0
    bounds = [(0.0, None)] * n + [(0.0, None)]

    # Objective: minimise M
    c = np.zeros(n + 1, dtype=np.float64)
    c[-1] = 1.0

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception as e:
        print(f"[{time.time():.2f}] LP exception: {e}")
        return None

    if not res.success:
        # Occasionally the LP can be numerically problematic
        # – treat it as a failure and move on.
        return None

    g = res.x[:n]
    M = res.x[-1]
    # Normalise again to eliminate tiny round‑off drift
    total = g.sum()
    if total <= 0:
        return None
    g *= target_sum / total
    return g, float(M)


def _lp_alternating_refinement(f: np.ndarray,
                               target_sum: float,
                               max_iters: int,
                               deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve the LP that minimises the cross‑convolution bound.
    After each solve we mix the LP solution with the current sequence
    (mixing factor t = 0.2).  The best sequence encountered is returned.
    """
    seq = f.copy()
    best_M = _max_convolution_fft(seq)
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        result = _lp_minimize_cross_max(seq, target_sum)
        if result is None:
            break
        g, M_cross = result
        # Try the raw LP solution first
        M_g = _max_convolution_fft(g)
        if M_g < best_M:
            best_M = M_g
            seq = g.copy()
            print(f"[{time.time():.2f}] LP it={it} direct replace M={best_M:.6f}")
            continue

        # Mix with a modest step size and keep whichever improves the max.
        t = 0.2
        candidate = (1.0 - t) * seq + t * g
        candidate = _project_to_simplex(candidate, target_sum)
        M_cand = _max_convolution_fft(candidate)
        if M_cand < best_M:
            best_M = M_cand
            seq = candidate.copy()
            print(f"[{time.time():.2f}] LP it={it} mixed improve M={best_M:.6f}")

    return seq, best_M
