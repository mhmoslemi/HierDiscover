        # locate peak
        k_max = int(np.argmax(b))

        hi = min(k_max, n - 1)
        idx_range = np.arange(lo, hi + 1)
        # partner values:
        partner_vals = a[k_max - idx_range]
        # donor probabilities:
        contrib = a[idx_range] * partner_vals
        if contrib.sum() > 0:
            donor = rng.choice(idx_range, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)
        # receiver probabilities based on low partner:
        inv_partner = 1.0 / (partner_vals + 1e-12)

        # We'll compute global inv based on partner for each index (non-zero only if partner exists else treat as 1).
        # Build array for all n.

        partner_global[lo:hi+1] = partner_vals
        inv = 1.0 / (partner_global + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)
        if donor == receiver:
            continue
        max_delta = min(a[donor] * 0.1, target_sum * 0.02)
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0:
            continue
        # Update b efficiently:
        # b_new = b - 2*delta*a (donor) + 2*delta*a (receiver) + adjustments
        # Use a before update.
        b_new = b.copy()
        # linear part from donor removal:
        b_new[donor:donor + n] -= 2.0 * delta * a
        # linear part from receiver addition:
        b_new[receiver:receiver + n] += 2.0 * delta * a

        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta

            accept = True

            if rng.random() < prob:
                accept = True

            a[receiver] += delta
            b = b_new
            cur_val = new_val

"""
Improved step‑function optimizer.

The method consists of three stages:

1. **Shape exploration** – try a family of monotone power‑law
   sequences (single exponent and a two‑segment exponent) and keep the
   one with the smallest score (the objective is proportional to the
   peak of the self‑convolution).

2. **Mass‑transfer simulated annealing** – repeatedly move a small
   amount of mass from a “donor’’ (large contribution to the current
   peak) to a “receiver’’ (small partner value).  The move is accepted
   with the Metropolis rule; the temperature follows a √‑schedule.
   The convolution is updated in O(n) time per move.

3. **Final mirror‑descent refinement** – a short exponential‑descent
   (soft‑max surrogate of the max‑convolution) finishes the
   optimisation.

All steps respect the global time budget supplied to ``propose_candidate``.
The best sequence found is stored in ``height_sequence_1`` for warm‑starts.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution using an FFT of power‑of‑2 size.
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Objective  2·n·max(a*a) / (sum(a)²).
    If Σ a = √(2n) the value equals max(a*a).
    """
    n = a.shape[0]
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of the max‑convolution.
    Returns (grad, full_convolution) where grad has the same length as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    grad_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(grad_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – exponential (mirror) descent on soft‑max surrogate
# ----------------------------------------------------------------------


def exponential_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Mirror descent on the soft‑max surrogate."""
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        # occasional jitter
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 8e-4, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpDescent it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase D – improved simulated‑annealing mass‑transfer refinement
# ----------------------------------------------------------------------


def anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Simulated‑annealing mass transfer.
    Moves a small amount of mass from an index that contributes heavily
    to the current peak to an index whose partner value in that peak is
    tiny.  Updates the convolution in O(n) time.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.80                     # initial temperature

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        # locate current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx_range = np.arange(lo, hi + 1, dtype=np.int64)

        # partner values for the active peak
        partner_vals = a[k_max - idx_range]

        # donor: weighted by contribution a_i * a_{k-i}
        contrib = a[idx_range] * partner_vals
        if contrib.sum() > 0:
            donor = rng.choice(idx_range, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver: favour indices whose partner value is small
        partner_global = np.zeros(n)
        partner_global[lo:hi + 1] = partner_vals
        inv = 1.0 / (partner_global + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # size of transfer (scaled by temperature)
        max_delta = min(a[donor] * 0.10, target_sum * 0.02)
        if max_delta <= 0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # --- O(n) update of the convolution -------------------------
        b_new = b.copy()
        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic corrections
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        # Metropolis acceptance based on current value
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal iter={it} new best={best_val:.6f}")

        # periodic full recomputation to avoid drift
        if it % 4000 == 0:
            a = np.maximum(a, 0.0)
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5       # safety margin

    # --------------------------------------------------------------
    # Choose problem size
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = math.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Shape exploration (single exponent + two‑segment exponent)
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # 1) single power‑law candidates
    exponents = np.arange(0.45, 0.66, 0.01)
    for e in exponents:
        a = (np.arange(1, n + 1, dtype=np.float64)) ** (-e)
        a = a * (target_sum / a.sum())
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a
            print(f"[{time.time():.2f}] single‑exp e={e:.3f} score={sc:.6f}")

    # 2) two‑segment power‑law candidates
    L = n // 2
    for e1 in np.arange(0.45, 0.66, 0.03):
        for e2 in np.arange(0.55, 0.76, 0.03):
            left = (np.arange(1, L + 1, dtype=np.float64)) ** (-e1)
            right = (np.arange(L + 1, n + 1, dtype=np.float64)) ** (-e2)
            a = np.concatenate([left, right])
            a = a * (target_sum / a.sum())
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a
                print(f"[{time.time():.2f}] two‑exp e1={e1:.3f} e2={e2:.3f} score={sc:.6f}")

    # 3) warm‑start from previous run if available
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if old.shape[0] != n:
            old = _resize_sequence(old, n)
        old = old * (target_sum / old.sum())
        sc_old = _score(old)
        if sc_old < best_score:
            best_score = sc_old
            best_seq = old
            print(f"[{time.time():.2f}] warm‑start score={sc_old:.6f}")

    # --------------------------------------------------------------
    # Optional short exponential descent (quick refinement)
    # --------------------------------------------------------------
    if time.time() < deadline:
        exp_deadline = time.time() + min(0.07 * (deadline - time.time()), 30.0)
        a_exp, sc_exp = exponential_descent(best_seq.copy(), target_sum, exp_deadline)
        if sc_exp < best_score:
            best_seq = a_exp
            best_score = sc_exp
            print(f"[{time.time():.2f}] after short exp‑descent best={best_score:.6f}")

    # --------------------------------------------------------------
    # Main simulated‑annealing phase
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_ann, sc_ann = anneal_mass_transfer(best_seq.copy(), target_sum, deadline)
        if sc_ann < best_score:
            best_seq = a_ann
            best_score = sc_ann
            print(f"[{time.time():.2f}] after annealing best={best_score:.6f}")

    # --------------------------------------------------------------
    # Final mirror‑descent pass (few iterations, high β)
    # --------------------------------------------------------------
    if time.time() < deadline:
        final_seq, final_score = exponential_descent(best_seq.copy(),
                                                     target_sum, deadline)
        if final_score < best_score:
            best_seq = final_seq
            best_score = final_score
            print(f"[{time.time():.2f}] final mirror descent best={best_score:.6f}")

    # --------------------------------------------------------------
    # Clean‑up, final evaluation and warm‑start storage
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # store for the next run
    globals()["height_sequence_1"] = final_seq

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluate_sequence score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    return final_seq
