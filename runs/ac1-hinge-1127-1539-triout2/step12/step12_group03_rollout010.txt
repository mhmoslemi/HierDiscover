def _subgradient_descent_avg(a0, target_sum, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.flatnonzero(conv >= max_val - 1e-12)
        second_max = conv.copy()
        second_max[idx_max] = -np.inf
        second_max = second_max.max()
        # break if almost flat
        if max_val - second_max < 1e-14:
            break

        # compute gradient as average of active constraints
        g = np.zeros_like(a)
        n = a.shape[0]
        for k in idx_max:
            lo = max(0, k-(n-1))
            hi = min(k, n-1)
            idx = np.arange(lo, hi+1, dtype=np.int64)
            g[idx] += a[k - idx]   # partner contributions
        # Since derivative per constraint is 2*a_{k-i}, we multiply by 2 later
        g = 2.0 * g / len(idx_max)

        norm2 = np.dot(g, g)
        if norm2 == 0:
            # jitter
            a += rng.normal(scale=target_sum*0.001, size=n)
            a = _project_to_simplex(a, target_sum)
            continue

        # step size to bring max down to second max: delta_target/(||g||^2)
        step = (max_val - second_max) / norm2
        # optionally 0.9 factor
        step = 0.9 * step
        # ensure not too large
        max_step = target_sum * 0.4 / (np.linalg.norm(g) + 1e-12)
        if step > max_step:
            step = max_step

        a_candidate = _project_to_simplex(a - step * g, target_sum)
        cand_val = _score(a_candidate)
        if cand_val < best_val - 1e-12:
            a = a_candidate
            best_a = a_candidate.copy()
            best_val = cand_val
            # maybe enforce symmetry
            # a = (a + a[::-1])/2
            # a = _project_to_simplex(a, target_sum)
        else:
            # jitter small to escape
            a += rng.normal(scale=target_sum*0.0005, size=n)
            a = _project_to_simplex(a, target_sum)
    return best_a, best_val

def _deterministic_mass_transfer_improved(a0, deadline, target_sum):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.flatnonzero(conv >= max_val - 1e-12)
        # find second max:
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()
        if max_val - second_max < 1e-12:
            break
        # compute for first max index
        k = idx_max[0]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = a[k - idx]
        # donors: indices where a[idx] > 0
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break
        # compute capacity = a[idx] * partner
        capacity = a[idx] * partner
        donor_candidates = np.where(donor_mask)[0]  # local indices
        donor_local = donor_candidates[np.argmax(capacity[donor_candidates])]
        donor = idx[donor_local]

        # find receiver with smallest partner value (or smallest capacity?) 
        # want partner low
        # we avoid donor
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            if idx[r_loc] != donor:
                receiver = idx[r_loc]
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_recv = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_recv)
        if denom <= 0:
            break

        delta_needed = (max_val - second_max) / denom
        # max possible delta is donor's mass
        delta = min(a[donor], delta_needed)
        # also not exceed donor mass to maintain positivity
        if delta <= 0:

        a[receiver] += delta

        a = _project_to_simplex(a, target_sum)

"""
Improved search for a non‑negative step function that minimizes the
maximal self‑convolution (the evaluation function returns that maximum
once the L¹ norm is fixed to √(2 n)).  The algorithm follows several
stages:

1.  Very fast parametric sweep over several families (power‑law,
    power‑law + exponential tail, sine, triangular) and the symmetric
    version of each.  The best candidate is kept.

2.  Several restarts of Adam‑style projected gradient descent on a
    soft‑max smoothing of the max‑convolution.  After every projection
    the vector is symmetrised (a←(a+a[::-1])/2) which empirically
    yields a lower peak.

3.  Projected sub‑gradient descent using the *average* of the active
    gradients (all indices attaining the current maximum).  An exact
    line‑search step size is used.

4.  An exact deterministic mass‑transfer step that moves mass from the
    most “dangerous’’ donor (largest product a_i·partner_i) to the most
    “helpful’’ receiver (smallest partner value) and transfers the
    maximal amount that can eliminate the current peak.  Symmetry is
    enforced after each move.

5.  Final polishing with a high‑β Adam run.

All phases respect the global time budget (default 1000 s).  The final
sequence is trimmed of trailing near‑zero entries and returned as a
plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers (linear convolution a*a, full length 2·n‑1)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Scoring – because the sum is forced to √(2 n) elsewhere,
# the evaluation equals the maximal convolution entry.
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (all return a vector of length n whose sum is
# exactly target_sum)
# ----------------------------------------------------------------------
def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float,
                            target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0))
    a *= target_sum / a.sum()
    return a


def _triangular_sequence(n: int, target_sum: float) -> np.ndarray:
    """Linear decrease from the left end to 0 (not symmetric)."""
    a = 1.0 - np.arange(n, dtype=np.float64) / (n - 1)
    a[a < 0] = 0.0
    a *= target_sum / a.sum()
    return a


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force symmetry: v_i = v_{n-1-i}.  Keeps the L¹ norm unchanged."""
    return 0.5 * (v + v[::-1])


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam)
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max) with symmetry
# ----------------------------------------------------------------------
def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    a = _symmetrize(a)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)
        a = _symmetrize(a)

        # occasional jitter to escape flat regions
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                 size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)
            a = _symmetrize(a)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent – average active gradient, symmetric
# ----------------------------------------------------------------------
def _subgradient_descent_avg(a0: np.ndarray,
                             target_sum: float,
                             deadline: float,
                             max_iter: int = 10000) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    a = _symmetrize(a)
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    it = 0
    n = a.shape[0]

    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.flatnonzero(conv >= max_val - 1e-12)

        # second largest value (ignore all current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < 1e-14:
            break

        # ---- average gradient over the active set ----
        g = np.zeros_like(a)
        for k in idx_max:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            # 2·a_{k-i} is the sub‑gradient for this lag
            g[idx] += 2.0 * a[k - idx]

        # average (any convex combination works)
        g /= len(idx_max)

        norm2 = np.dot(g, g)
        if norm2 == 0.0:
            # jitter and continue
            a += rng.normal(scale=target_sum * 0.001, size=n)
            a = _project_to_simplex(a, target_sum)
            a = _symmetrize(a)
            it += 1
            continue

        # analytic step size that would bring the current max down to the second max
        step = (max_val - second_max) / norm2
        step = 0.9 * step  # a small safety factor

        # limit step size to keep the move moderate
        max_step = target_sum * 0.4 / (np.linalg.norm(g) + 1e-12)
        if step > max_step:
            step = max_step

        a_candidate = _project_to_simplex(a - step * g, target_sum)
        a_candidate = _symmetrize(a_candidate)

        cand_val = _score(a_candidate)
        if cand_val < best_val - 1e-12:
            a = a_candidate
            best_a = a_candidate.copy()
            best_val = cand_val
            print(f"[{time.time():.2f}] SubgradAvg it={it:5d} step={step:.3e} score={best_val:.6f}")
        else:
            # tiny jitter to potentially escape a plateau
            a += rng.normal(scale=target_sum * 0.0005, size=n)
            a = _project_to_simplex(a, target_sum)
            a = _symmetrize(a)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (improved version, symmetric)
# ----------------------------------------------------------------------
def _deterministic_mass_transfer_improved(a0: np.ndarray,
                                          deadline: float,
                                          target_sum: float,
                                          tol: float = 1e-12) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    a = _symmetrize(a)
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - tol)[0]

        # second largest value (ignore all maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        # work with the first maximal lag
        k = idx_max[0]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k‑i}
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        # “capacity’’ = a_i * partner_i, pick the most effective donor
        capacity = a[idx] * partner
        donor_local = np.argmax(capacity * donor_mask)
        donor = idx[donor_local]

        # pick a receiver with the smallest partner value (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_recv = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_recv)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        # move as much mass as possible (but not more than we own)
        delta = min(a[donor], delta_needed)
        if delta <= 0.0:
            break

        a[donor] -= delta
        a[receiver] += delta

        a = _project_to_simplex(a, target_sum)
        a = _symmetrize(a)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] DMTImproved it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Global random mass‑transfer (exploration)
# ----------------------------------------------------------------------
def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    a = _symmetrize(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        donor = np.argsort(-a)[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        receiver = np.argsort(a)[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)
        a = _symmetrize(a)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] GlobalRandom new best={best_score:.6f}")

    return best_a, best_score
