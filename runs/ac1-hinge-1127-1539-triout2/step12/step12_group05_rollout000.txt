"""
Propose a near‑optimal step‑function for the benchmark.

Algorithm outline
-----------------
1. **Problem size** – choose n based on the time budget (12000 for a
   1000 s run).  The L1‐norm is fixed to √(2 n).

2. **Fast parametric pre‑search** – evaluate a rich family of analytic
   shapes (shifted power‑law, exponential‑power‑law, broken power‑law
   with a shift, pure exponential).  First a coarse grid is scanned,
   then a fine local grid around the current best parameters.

3. **Short Adam warm‑starts** – a few projected‑Adam runs on the
   smooth soft‑max approximation of the max‑convolution, with a
   geometrically increasing temperature β.  These runs help escape
   poor local minima.

4. **Guided exact mass‑transfer (core refinement)** – repeatedly
   identify the single convolution entry attaining the global maximum
   and move a small amount of mass from indices that increase this
   entry the most toward indices that increase it the least.  The
   step size is chosen by an exact line‑search that drives the peak
   down to the second‑largest value without violating non‑negativity.
   This sub‑gradient step is cheap (two FFTs) and is performed for
   the majority of the time budget.

5. **Light random‑move perturbations** – occasional simulated‑annealing
   style swaps of mass between a large and a small coefficient help
   to break plateaus.

6. **Final high‑β Adam polishing** – one last Adam run with a very
   large β (≈10⁶) to fine‑tune the remaining tiny peaks.

All iterates are projected onto the simplex {x ≥ 0, Σx = √(2 n)}.
The routine prints intermediate scores and returns the best sequence
found (trailing near‑zero entries are trimmed).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) via real FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft_pair(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution x*y (full length 2·n‑1) via real FFT."""
    n = x.size
    size = 1 << ((2 * n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Benchmark score (identical to the evaluator when Σa = √(2n))."""
    n = a.shape[0]
    max_b = _max_conv(a)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (dense pre‑search)
# ----------------------------------------------------------------------


def _power_law_shifted(n: int, p: float, delta: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1+δ)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64) + delta
    a = idx ** (-p)
    a *= target_sum / a.sum()
    return a


def _exp_power_law_shifted(
    n: int, p: float, q: float, delta: float, target_sum: float
) -> np.ndarray:
    """a_i ∝ (i+1+δ)^{-p} · exp(-q·i/n)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0 + delta) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _broken_power_law_shifted(
    n: int, p1: float, p2: float, alpha: float, delta: float, target_sum: float
) -> np.ndarray:
    """Two‑segment power‑law with a shift."""
    K = max(1, int(alpha * n))
    idx = np.arange(1, n + 1, dtype=np.float64) + delta
    a = np.empty_like(idx)
    a[:K] = idx[:K] ** (-p1)
    a[K:] = idx[K:] ** (-p2)
    a *= target_sum / a.sum()
    return a


def _pure_exponential(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-λ·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max approximation)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of soft‑max approximation of max(a*a)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = conv_full.max()
    exps = np.exp(beta * (conv_full - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, conv_full


def _adam_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 5e5,
    step_factor: float = 0.8,
    gamma: float = 0.999,
) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max approximation of the max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to increase exploration
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient guided mass‑transfer (core refinement)
# ----------------------------------------------------------------------


def _single_peak_subgradient(a: np.ndarray, peak_idx: int) -> np.ndarray:
    """
    Exact sub‑gradient for the *single* convolution entry at `peak_idx`.
    Returns g_i = 2·a[peak_idx‑i] wherever the index is valid, else 0.
    """
    n = a.size
    g = np.zeros_like(a)
    lo = max(0, peak_idx - (n - 1))
    hi = min(peak_idx, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    # contribution 2 * a[peak_idx - idx]
    g[idx] = 2.0 * a[peak_idx - idx]
    return g


def _guided_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the maximal convolution entry.
    At each iteration we move mass from indices that increase the current
    peak most towards indices that increase it least, using an exact line‑search.
    This variant uses the sub‑gradient of a *single* maximal index (chosen at random)
    for a stronger descent direction.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()

        # mask of entries that are (almost) maximal
        mask_max = conv >= max_val - tol
        # second largest value (strictly smaller than max)
        if np.any(~mask_max):
            second_max = conv[~mask_max].max()
        else:
            second_max = max_val

        # pick one of the maximal indices uniformly at random
        max_idxs = np.where(mask_max)[0]
        peak_idx = rng.choice(max_idxs)

        # exact sub‑gradient for the chosen peak
        g = _single_peak_subgradient(a, peak_idx)

        # direction preserving Σa = 0
        sum_g = g.sum()
        d = -g + sum_g / n * np.ones_like(a)   # Σd = 0

        # maximum allowable step before any component becomes negative
        neg = d < 0
        if np.any(neg):
            t_nonneg = np.min(a[neg] / -d[neg])
        else:
            t_nonneg = np.inf

        # linearised change of the convolution in direction d
        delta = 2.0 * _conv_fft_pair(a, d)          # length 2n‑1
        delta_k = delta[peak_idx]                  # effect on the chosen peak

        if delta_k >= 0:
            # direction cannot lower the peak – abort this phase
            break

        # step needed to drive the current peak down to the second largest value
        t_to_second = (max_val - second_max) / (-delta_k)

        # safe step (leave a tiny margin to avoid numerical ties)
        t = min(t_nonneg, t_to_second) * 0.99
        if t <= 0:
            break

        # update
        a += t * d
        # enforce feasibility (tiny rounding errors)
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] GuidedMT it={it:5d} score={best_score:.6f}")

        # occasional random jitter to hop out of tiny plateaus
        if it % 500 == 0 and rng.random() < 0.07:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Light global random exploration (optional)
# ----------------------------------------------------------------------


def _global_random_move(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
) -> tuple[np.ndarray, float]:
    """Very cheap simulated‑annealing style mass swaps."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    while time.time() < deadline - 0.5:
        # pick donor from the top 2 % and receiver from the bottom 2 %
        top_k = max(1, int(0.02 * n))
        low_k = max(1, int(0.02 * n))
        donor = np.argsort(-a)[rng.integers(top_k)]
        receiver = np.argsort(a)[rng.integers(low_k)]
        if donor == receiver:
            continue
        # move up to 5 % of donor's mass
        delta = rng.random() * a[donor] * 0.05
        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] RandomMove new best={best_score:.6f}")

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal benchmark score.
    The routine respects the supplied time budget and returns the best
    sequence found (trailing near‑zero entries are trimmed).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    # ---------------------------------------------------------------
    # Choose a problem size – larger n yields finer resolution.
    # ---------------------------------------------------------------
    if budget_s >= 900:
        n = 12000
    elif budget_s >= 600:
        n = 10000
    else:
        n = 8000

    target_sum = math.sqrt(2.0 * n)   # Σa is fixed

    # ---------------------------------------------------------------
    # Phase 0 – dense parametric pre‑search (coarse + local refinement)
    # ---------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    rng = np.random.default_rng()

    # ---- coarse grids ------------------------------------------------
    # (a) shifted power‑law
    for p in np.arange(0.20, 1.21, 0.04):
        for delta in np.arange(0.0, 5.01, 0.5):
            seq = _power_law_shifted(n, float(p), float(delta), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # (b) exponential‑power‑law
    for p in np.arange(0.30, 1.21, 0.05):
        for q in np.arange(0.0, 7.01, 0.5):
            for delta in np.arange(0.0, 5.01, 0.5):
                seq = _exp_power_law_shifted(
                    n, float(p), float(q), float(delta), target_sum
                )
                val = _score(seq)
                if val < best_score:
                    best_score, best_seq = val, seq.copy()

    # (c) broken power‑law with shift
    for p1 in np.arange(0.20, 0.81, 0.1):
        for p2 in np.arange(0.80, 2.01, 0.2):
            for alpha in np.arange(0.15, 0.86, 0.15):
                for delta in np.arange(0.0, 3.01, 1.0):
                    seq = _broken_power_law_shifted(
                        n,
                        float(p1),
                        float(p2),
                        float(alpha),
                        float(delta),
                        target_sum,
                    )
                    val = _score(seq)
                    if val < best_score:
                        best_score, best_seq = val, seq.copy()

    # (d) pure exponential tail
    for lam in np.arange(0.0005, 0.0105, 0.001):
        seq = _pure_exponential(n, float(lam), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # ---- local refinement near the best parameters ------------------
    # Store the best parameters of the previous families (very cheap
    # random sampling around them) to squeeze a little more.
    refin_time = deadline - time.time()
    if refin_time > 20.0:  # allocate up to ~20 s for extra sampling
        for _ in range(2000):
            # Randomly pick a family
            family = rng.choice(["pl", "epl", "bp", "exp"])
            if family == "pl":
                p = rng.uniform(0.2, 1.2)
                delta = rng.uniform(0.0, 5.0)
                seq = _power_law_shifted(n, p, delta, target_sum)
            elif family == "epl":
                p = rng.uniform(0.3, 1.2)
                q = rng.uniform(0.0, 7.0)
                delta = rng.uniform(0.0, 5.0)
                seq = _exp_power_law_shifted(n, p, q, delta, target_sum)
            elif family == "bp":
                p1 = rng.uniform(0.2, 0.8)
                p2 = rng.uniform(0.8, 2.0)
                alpha = rng.uniform(0.1, 0.9)
                delta = rng.uniform(0.0, 3.0)
                seq = _broken_power_law_shifted(n, p1, p2, alpha, delta, target_sum)
            else:  # pure exponential
                lam = rng.uniform(0.0005, 0.0105)
                seq = _pure_exponential(n, lam, target_sum)

            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # also consider the external best if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            # linear interpolation to the new size
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score, best_seq = val, resized.copy()

    print(f"[{time.time():.2f}] After dense pre‑search: best score = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase A – short Adam restarts (exploratory)
    # ---------------------------------------------------------------
    remaining = deadline - time.time()
    ada_deadline = time.time() + 0.07 * remaining  # ~7 % of remaining time
    while time.time() < ada_deadline - 1.5:
        jitter = rng.normal(scale=target_sum * 0.02, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _adam_gradient_descent(
            start_vec,
            ada_deadline,
            target_sum,
            max_beta=4e5,
            step_factor=0.85,
            gamma=0.996,
        )
        if val_opt < best_score:
            best_score, best_seq = val_opt, a_opt.copy()
            print(f"[{time.time():.2f}] Adam restart improved best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase B – intensive guided exact mass‑transfer
    # ---------------------------------------------------------------
    # Allocate the bulk of the remaining budget to this phase.
    while time.time() < deadline - 8.0:
        a_ref, val_ref = _guided_mass_transfer(best_seq, deadline, target_sum)
        if val_ref < best_score:
            best_score, best_seq = val_ref, a_ref.copy()
            print(f"[{time.time():.2f}] GuidedMT improved best = {best_score:.6f}")

        # occasional random move to escape tiny plateaus
        a_rand, val_rand = _global_random_move(best_seq, deadline, target_sum)
        if val_rand < best_score:
            best_score, best_seq = val_rand, a_rand.copy()
            print(f"[{time.time():.2f}] Random move improved best = {best_score:.6f}")

        # break if only a few seconds left
        if time.time() > deadline - 6.0:
            break

    # ---------------------------------------------------------------
    # Phase C – final high‑β Adam polishing (if time permits)
    # ---------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(
            best_seq,
            deadline,
            target_sum,
            max_beta=1e6,
            step_factor=0.6,
            gamma=0.995,
        )
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final Adam polish improved best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return list
    # ---------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final evaluation (harmless if `evaluate_sequence` is missing)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated benchmark score = {val:.6f}")
    except Exception:
        pass

    return final_seq
