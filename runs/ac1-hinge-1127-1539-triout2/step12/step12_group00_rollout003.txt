def _adam_softmax_optimize(a0, target_sum, deadline, betas=(0.5, 1e5), beta_mult=2., lr=0.2, eps=1e-8):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta = betas[0]
    best = a.copy()
    best_score = _score(a)
    t_iter = 0
    beta_target = betas[1]
    while time.time() < deadline:
        grad, conv = _gradient_softmax(a, beta)
        # Adam update
        t_iter += 1
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad**2)
        m_hat = m / (1 - 0.9**t_iter)
        v_hat = v / (1 - 0.999**t_iter)
        step = lr / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        # project onto simplex
        a = _project_to_simplex(a, target_sum)
        # logging maybe every 200 iterations
        if t_iter % 100 == 0:
            cur = _score(a)
            if cur < best_score:
                best_score = cur
                best = a.copy()
                print(f"[{time.time():.2f}] Adam it={t_iter} β={beta:.1f} score={best_score:.6f}")
        # increase beta
        if beta < beta_target:
            beta = min(beta * beta_mult, beta_target)
    return best, best_score

"""
Hybrid optimizer for the step–function minimisation problem.

The algorithm proceeds in four stages:

1. **Exploratory Adam‑softmax optimisation** on a moderate size (≈8k).  
   Uses an Adam variant on a smooth log‑sum‑exp approximation of the
   max‑convolution and projects onto the ℓ¹‑simplex after each step.

2. **LP alternating refinement** on a down‑scaled problem (≈1 k).  
   Solves a linear program that maximises a random linear functional
   under the current convolution bound, then mixes the result with a
   line‑search on the mixing weight.

3. **Upscaling & fine‑tuning** – the refined sequence is interpolated to a
   larger length (≈12‑20k) and polished again with Adam and a short
   sub‑gradient descent.

4. **Local stochastic mass transfer** – a simulated‑annealing style
   neighbourhood search that moves small amounts of mass between
   coordinates to escape possible local minima.

All updates respect non‑negativity and the ℓ¹‑simplex constraint
(∑a = √(2 n)).  The routine returns the best sequence found before the
time budget expires.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Scoring
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)² (for Σa=√(2n) this equals maxconv)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers (used after up‑scaling)
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of the max)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (gradient, convolution) of the soft‑max approximation
    of max(a*a).  Gradient = 2 * (a_rev * p) where p is the soft‑max
    weights on the convolution vector.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max surrogate
# ----------------------------------------------------------------------


def _adam_softmax_optimize(a0: np.ndarray,
                           target_sum: float,
                           deadline: float,
                           beta_start: float = 0.5,
                           beta_target: float = 5e5,
                           beta_mult: float = 2.0,
                           lr: float = 0.3,
                           eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp approximation of max(a*a).
    Keeps the vector on the ℓ¹‑simplex with sum = target_sum.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta = beta_start
    best = a.copy()
    best_val = _score(a)
    t_iter = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        # Adam moment updates
        t_iter += 1
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad ** 2)

        m_hat = m / (1 - 0.9 ** t_iter)
        v_hat = v / (1 - 0.999 ** t_iter)

        step = lr / (np.sqrt(v_hat) + eps)
        a -= step * m_hat

        # projection onto the simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if t_iter % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # record best
        if t_iter % 100 == 0:
            cur = _score(a)
            if cur < best_val:
                best_val = cur
                best = a.copy()
                print(f"[{time.time():.2f}] Adam it={t_iter:5d} β={beta:.1f} best={best_val:.6f}")

        # increase beta (sharpen soft‑max)
        if beta < beta_target:
            beta = min(beta * beta_mult, beta_target)

    return best, best_val


# ----------------------------------------------------------------------
# Linear‑programming alternating refinement (small dimensionality)
# ----------------------------------------------------------------------


def _lp_optimize(f: np.ndarray, M: float, target_sum: float,
                 rng: np.random.Generator) -> np.ndarray | None:
    """
    Solve a LP that maximises a random linear functional of `b`
    subject to `f * b <= M` and `b >= 0`.  The result is scaled to
    have ℓ¹‑norm = target_sum.
    """
    n = f.shape[0]
    rows = []
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=np.int64)
        row = np.zeros(n, dtype=np.float64)
        row[i] = f[k - i]
        rows.append(row)
    A = np.vstack(rows)               # (2n‑1) × n
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    c = -rng.random(n)                # maximise c·b  ⇔ minimise –c·b
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    b = res.x
    total = b.sum()
    if total <= 0:
        return None
    b *= target_sum / total
    return b


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent using the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_val = _score(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]
        grad = np.zeros_like(a)
        for k in max_idxs:
            lo = max(0, k - (a.shape[0] - 1))
            hi = min(k, a.shape[0] - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur = _score(a)
        if cur < best_val:
            best_val = cur
            best = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} best={best_val:.6f}")

        it += 1

    return best, best_val


# ----------------------------------------------------------------------
# Local stochastic mass‑transfer (simulated annealing)
# ----------------------------------------------------------------------


def _local_search(a0: np.ndarray,
                  target_sum: float,
                  deadline: float,
                  rng: np.random.Generator,
                  max_move_frac: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing neighbourhood search.
    Moves a random amount of mass from one coordinate to another,
    accepting improvements or probabilistic uphill moves.
    """
    a = a0.copy()
    cur_score = _score(a)
    best = a.copy()
    best_score = cur_score

    n = a.shape[0]
    start_time = time.time()
    start_temp = 0.02
    end_temp = 1e-6
    it = 0

    while time.time() < deadline:
        # linear temperature schedule
        tau = (time.time() - start_time) / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # pick donor and receiver
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # amount to transfer (no more than a[i])
        max_delta = a[i] * max_move_frac
        if max_delta <= 0:
            it += 1
            continue
        delta = rng.random() * max_delta

        a[i] -= delta
        a[j] += delta

        new_score = _score(a)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time():.2f}] Local search it={it:5d} new best={best_score:.6f}")
        else:
            # revert move
            a[i] += delta
            a[j] -= delta

        it += 1

    return best, best_score

    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 200:
        n_final = 20000
    elif remaining > 120:
        n_final = 15000
    elif remaining > 60:
        n_final = 12000
    else:
        n_final = n_expl

    target_final = np.sqrt(2.0 * n_final)
    seq_big = _resize_sequence(seq_lp, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # simple reorderings to lower the peak
    candidates = [seq_big,
                  _alternating_order(seq_big),
                  _even_spread_order(seq_big)]
    seq_big = min(candidates, key=_max_convolution_fft)
    best_score = _score(seq_big)
    print(f"[{time.time():.2f}] Upscaled length={len(seq_big)} start score={best_score:.6f}")

    # Adam polishing on the large problem
    fine_deadline = deadline - 30.0
    if time.time() < fine_deadline:
        seq_big, fine_val = _adam_softmax_optimize(
            seq_big, target_final, fine_deadline,
            beta_start=1.0, beta_target=1e6,
            beta_mult=1.5, lr=0.15)
        if fine_val < best_score:
            best_score = fine_val
            print(f"[{time.time():.2f}] Fine Adam best={best_score:.6f}")

    # Sub‑gradient polishing (optional but cheap)
    sub_deadline = deadline - 15.0
    if time.time() < sub_deadline:
        seq_big, sub_val = _subgradient_descent(
            seq_big, sub_deadline, target_final,
            step_factor=0.45, gamma=0.995)
        if sub_val < best_score:
            best_score = sub_val
            print(f"[{time.time():.2f}] Sub‑grad best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final local stochastic search
    # ------------------------------------------------------------------
    final_deadline = deadline - 2.0
    if time.time() < final_deadline:
        seq_big, final_val = _local_search(
            seq_big, target_final, final_deadline, rng)
        if final_val < best_score:
            best_score = final_val
            print(f"[{time.time():.2f}] Final local search best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # ------------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # store for possible reuse
    globals()["height_sequence_1"] = final_seq

    # optional final evaluation for logging (ignore failures)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
