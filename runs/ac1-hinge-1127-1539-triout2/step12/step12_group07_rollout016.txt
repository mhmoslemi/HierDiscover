def _permutation_annealing(seq, deadline):
    best_seq = seq.copy()
    best_val = _max_convolution_fft(seq)
    cur_seq = seq.copy()
    cur_val = best_val
    n = len(seq)
    rng = np.random.default_rng()
    # compute initial convolution? Not needed for evaluation; we have cur_val.
    # But for acceptance we need new conv each time.
    # We'll generate random swaps.
    # We'll decay temperature.
    start = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = 0.5 * (0.001 / 0.5) ** tau  # start 0.5 -> near 0.001 maybe

        # propose swap
        i, j = rng.integers(0, n, size=2)
        if i == j: continue
        # swap values
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        new_val = _max_convolution_fft(cur_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / max(temp, 1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cur_seq.copy()
        else:
            # revert swap
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

    return best_seq, best_val

"""
Hybrid optimizer for the step‑function benchmark.

The algorithm:
1. Generate a diverse pool of initial sequences of moderate size (≈4‑5 k)
   using power‑law, exponential and random simplex distributions.
2. Run a fast Adam‑based soft‑max gradient descent on each seed to
   minimise the smooth approximation of the max‑self‑convolution while
   keeping the ℓ¹–norm fixed (√(2 n)).
3. Keep the best candidate, upscale it to a larger length (≈15‑25 k),
   re‑project onto the simplex and try a few deterministic reorderings.
4. Refine the ordering with a simulated‑annealing permutation search that
   swaps positions to lower the true convolution peak.
5. Apply projected sub‑gradient descent on the exact max‑convolution.
6. Finish with a greedy mass‑transfer (simulated annealing) phase.
All phases respect the non‑negativity and ℓ¹‑simplex constraints and
are terminated early if the total 1000 s budget is exhausted.
"""

import time
import math
import random
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (linear convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a)/(∑a)² (for ∑a=√(2n) this equals max conv)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised to the required ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex_sequence(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex (Dirichlet‑like)."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _piecewise_linear_sequence(n: int, target_sum: float) -> np.ndarray:
    """
    Simple piecewise linear shape: large values at the ends,
    linearly decreasing towards the centre.
    """
    half = n // 2
    left = np.linspace(1.0, 0.0, half, endpoint=False)
    right = left[::-1]
    if n % 2 == 1:
        middle = np.array([0.0])
        a = np.concatenate([left, middle, right])
    else:
        a = np.concatenate([left, right])
    a = np.maximum(a, 0.0)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """
    Spread the large values uniformly by placing them on a cyclic sequence
    with a step coprime to the length (≈√n).
    """
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Adam‑based soft‑max gradient descent
# ----------------------------------------------------------------------


def _adam_softmax_gradient_descent(a0: np.ndarray,
                                   deadline: float,
                                   target_sum: float,
                                   max_beta: float = 8e5,
                                   step_factor: float = 0.5,
                                   beta1: float = 0.9,
                                   beta2: float = 0.999,
                                   gamma: float = 0.98) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation of the soft‑max approximation of the max‑convolution.
    """
    rng = np.random.default_rng()
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta = 0.5
    it = 0
    while time.time() < deadline:
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_b = np.max(conv)
        # soft‑max probabilities
        exps = np.exp(beta * (conv - max_b))
        p = exps / np.sum(exps)

        # gradient of soft‑max (see original algorithm)
        a_rev = a[::-1]
        a_rev_fft = np.fft.rfft(a_rev, n=size)
        p_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = step_factor * target_sum / (np.sqrt(v_hat) + 1e-8)
        a -= lr * m_hat

        a = _project_to_simplex(a, target_sum)

        # jitter occasionally to escape flat regions
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        # beta schedule (increase temperature of soft‑max)
        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam GD it={it:5d} β={beta:7.1f} score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.45,
                         gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that uses the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]

        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy mass‑transfer with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.5,
                                      end_temp: float = 1e-4,
                                      low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Greedy mass moves from a “donor’’ (large contribution to the peak)
    to a “receiver’’ (small contribution) with a Metropolis acceptance rule.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # locate current peak
        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contributions of each index to the peak
        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        # propose moving a small amount of mass
        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        # Metropolis acceptance
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing permutation search
# ----------------------------------------------------------------------


def _permutation_annealing(base_seq: np.ndarray,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Starting from `base_seq`, repeatedly swap two positions (or move an element)
    using a Metropolis rule to lower the true max‑convolution.
    """
    n = base_seq.shape[0]
    rng = np.random.default_rng()
    cur_seq = base_seq.copy()
    cur_val = _max_convolution_fft(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val
    start = time.time()

    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        # temperature decays from 0.5 to 1e-4
        temp = 0.5 * (1e-4 / 0.5) ** tau

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose swap
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        new_val = _max_convolution_fft(cur_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / max(temp, 1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cur_seq.copy()
                print(f"[{time.time():.2f}] PermAnneal new best={best_val:.6f}")
        else:
            # revert
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

    return best_seq, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:

    sequence found so far.
    """

    rng = np.random.default_rng()

    # Phase 1 – rapid exploration on a modest size
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n_small = 5000
    elif budget_s >= 600:
        n_small = 4500

        n_small = 4000
    else:
        n_small = 3000

    for p in np.linspace(0.1, 3.0, 15):

    # include previous best if present

        jitter = rng.normal(scale=target_small * 0.04, size=n_small)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_small)
        init_pool.append(seq_j)

    best_seq = None
    best_score = np.inf

    if best_seq is not None and time.time() < deadline - 20.0:

            best_seq, mt_deadline, target_small,

    ]

    print(f"[{time.time():.2f}] Upscaled to {n_final} elements, start max={_max_convolution_fft(seq_big):.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

        sub_deadline = deadline - 4.0
        a_sub, val_sub = _subgradient_descent(
            seq_big, sub_deadline, target_final,
            step_factor=0.4, gamma=0.995)
        if val_sub < best_score:
            best_score = val_sub
            seq_big = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑grad polished best={best_score:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if time.time() < deadline - 3.0:
        mt_deadline = time.time() + min(6.0, deadline - time.time() - 1.0)

"""
Hybrid optimizer for the step‑function benchmark.

The algorithm:
1. Generate a diverse pool of initial sequences of moderate size (≈4‑5 k)
   using power‑law, exponential and random‑simplex families.
2. Run a fast Adam‑based soft‑max gradient descent on each seed to
   minimise a smooth approximation of the max‑self‑convolution while
   keeping the ℓ¹–norm fixed (√(2 n)).
3. Keep the best candidate, upscale it to a larger length (≈15‑25 k),
   re‑project onto the simplex and try a few deterministic reorderings.
4. Refine the ordering with a simulated‑annealing permutation search that
   swaps positions to lower the true convolution peak.
5. Apply projected sub‑gradient descent on the exact max‑convolution.
6. Finish with a greedy mass‑transfer (simulated annealing) phase.
All phases respect the non‑negativity and ℓ¹‑simplex constraints and
are stopped early if the total 1000 s budget is exhausted.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (linear convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a)/(∑a)² (for ∑a=√(2n) this equals max conv)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised to the required ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex_sequence(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex (Dirichlet‑like)."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _piecewise_linear_sequence(n: int, target_sum: float) -> np.ndarray:
    """
    Simple piecewise‑linear shape: large values at the ends,
    linearly decreasing towards the centre.
    """
    half = n // 2
    left = np.linspace(1.0, 0.0, half, endpoint=False)
    right = left[::-1]
    if n % 2:
        middle = np.array([0.0])
        a = np.concatenate([left, middle, right])
    else:
        a = np.concatenate([left, right])
    a = np.maximum(a, 0.0)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """
    Spread large values uniformly by placing them on a cyclic sequence
    with a step coprime to the length (≈√n).
    """
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Adam‑based soft‑max gradient descent
# ----------------------------------------------------------------------


def _adam_softmax_gradient_descent(a0: np.ndarray,
                                   deadline: float,
                                   target_sum: float,
                                   max_beta: float = 8e5,
                                   step_factor: float = 0.5,
                                   beta1: float = 0.9,
                                   beta2: float = 0.999,
                                   gamma: float = 0.98) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation of a soft‑max approximation of the max‑convolution.
    """
    rng = np.random.default_rng()
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta = 0.5
    it = 0

    while time.time() < deadline:
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_b = np.max(conv)
        exps = np.exp(beta * (conv - max_b))
        p = exps / np.sum(exps)

        a_rev = a[::-1]
        a_rev_fft = np.fft.rfft(a_rev, n=size)
        p_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = step_factor * target_sum / (np.sqrt(v_hat) + 1e-8)
        a -= lr * m_hat

        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam GD it={it:5d} β={beta:7.1f} score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.45,
                         gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent using the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]

        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy mass‑transfer with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.5,
                                      end_temp: float = 1e-4,
                                      low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Greedy mass moves from a “donor’’ (large contribution to the peak)
    to a “receiver’’ (small contribution) with a Metropolis acceptance rule.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing permutation search
# ----------------------------------------------------------------------


def _permutation_annealing(base_seq: np.ndarray,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Starting from `base_seq`, repeatedly swap two positions using a
    Metropolis rule to lower the true max‑convolution.
    """
    n = base_seq.shape[0]
    rng = np.random.default_rng()
    cur_seq = base_seq.copy()
    cur_val = _max_convolution_fft(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val
    start = time.time()

    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = 0.5 * (1e-4 / 0.5) ** tau  # 0.5 → 1e‑4

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        new_val = _max_convolution_fft(cur_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / max(temp, 1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cur_seq.copy()
                print(f"[{time.time():.2f}] PermAnneal new best={best_val:.6f}")
        else:
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine respects the total time budget and returns the best
    sequence found so far.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin
    rng = np.random.default_rng()

    # Phase 1 – rapid exploration on a modest size
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n_small = 5000
    elif budget_s >= 600:
        n_small = 4500
    elif budget_s >= 400:
        n_small = 4000
    else:
        n_small = 3000

    target_small = math.sqrt(2.0 * n_small)

    init_pool = []

    # power‑law family
    for p in np.linspace(0.1, 3.0, 15):
        init_pool.append(_power_law_sequence(n_small, float(p), target_small))

    # exponential family
    for lam in np.linspace(0.0005, 0.06, 12):
        init_pool.append(_exponential_sequence(n_small, float(lam), target_small))

    init_pool.append(_piecewise_linear_sequence(n_small, target_small))

    # random simplex points
    for _ in range(12):
        init_pool.append(_random_simplex_sequence(n_small, target_small, rng))

    # reuse previous best if available

        prev = np.array(globals()["height_sequence_1"], dtype=float)

    # jittered copies of everything for extra diversity
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_small * 0.04, size=n_small)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)

    # Phase 2 – brief mass‑transfer refinement on the small scale

    if best_seq is not None and time.time() < deadline - 20.0:

            best_seq, mt_deadline, target_small,
            start_temp=0.45, end_temp=5e-5, low_frac=0.08)

            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] MT refinement score={best_score:.6f}")

        n_final = 20000

        n_final = 15000

        n_final = 12000
    else:
        n_final = n_small

    target_final = math.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # deterministic reorderings
    candidates = [
        seq_big,
        _alternating_order(seq_big),
        _even_spread_order(seq_big)
    ]
    seq_big = min(candidates, key=_max_convolution_fft)
    print(f"[{time.time():.2f}] Upscaled to {n_final} elements, start max={_max_convolution_fft(seq_big):.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – permutation annealing (random swaps)
    # ------------------------------------------------------------------

        perm_deadline = time.time() + min(10.0, deadline - time.time() - 6.0)
        seq_big, perm_val = _permutation_annealing(seq_big, perm_deadline)
        print(f"[{time.time():.2f}] After perm anneal max={perm_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – sub‑gradient descent on the large scale
    # ------------------------------------------------------------------
    if time.time() < deadline - 8.0:
        sub_deadline = deadline - 4.0
        a_sub, val_sub = _subgradient_descent(
            seq_big, sub_deadline, target_final,
            step_factor=0.4, gamma=0.995)
        if val_sub < best_score:
            best_score = val_sub
            seq_big = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑grad polished best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final mass‑transfer pass (if time left)
    # ------------------------------------------------------------------
    if time.time() < deadline - 3.0:
        mt_deadline = time.time() + min(6.0, deadline - time.time() - 1.0)
        a_mt, val_mt = _mass_transfer_simulated_annealing(
            seq_big, mt_deadline, target_final,
            start_temp=0.25, end_temp=2e-5, low_frac=0.06)
        if val_mt < best_score:
            best_score = val_mt
            seq_big = a_mt.copy()
            print(f"[{time.time():.2f}] Final MT best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # ------------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]

        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Optional final evaluation (purely for logging)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass
