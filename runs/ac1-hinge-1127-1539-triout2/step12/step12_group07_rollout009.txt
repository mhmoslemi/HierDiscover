"""
Hybrid optimizer for the non‑negative step‑function problem.

The routine consists of five stages:

1. **Seed generation** – create a diverse pool of short sequences (≈5 k)
   using power‑law, exponential and random simplex families; add jittered
   copies and the previous best if available.

2. **Soft‑max gradient descent** – run several short “annealed” gradient
   descents on the smoothed max‑convolution (soft‑max) with projection onto
   the ℓ¹‑simplex. The best low‑dimensional candidate is retained.

3. **Optional sub‑gradient polishing** – a few iterations of exact
   sub‑gradient descent on the true max‑convolution.

4. **Upscaling** – linearly interpolate the best short sequence to a larger
   length (≈15 k–30 k) and try a few deterministic reorderings
   (alternating / even‑spread). The ordering with the smallest convolution
   peak is kept.

5. **Permutation hill‑climbing** – a simulated‑annealing loop that swaps
   two entries or transfers a small amount of mass between a “donor” and a
   “receiver”.  Each move is evaluated via an FFT‑based convolution; moves
   are accepted with a Metropolis rule whose temperature decays
   exponentially.  The loop runs as long as time permits.

Finally the sequence is trimmed of trailing zeros, stored globally for warm‑starts,
and returned.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (linear convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a)/(∑a)² (for ∑a=√(2n) this equals max conv)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised to the required ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex (Dirichlet‑like)."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """
    Spread the large values uniformly by placing them on a cyclic sequence
    with a step coprime to the length (≈√n).
    """
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Annealed gradient descent (soft‑max approximation)
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.9,
                               gamma: float = 0.98) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on a smooth approximation of max‑convolution.
    The smoothness is controlled by a temperature‑like parameter beta.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_b = np.max(b)
        # soft‑max weights
        exps = np.exp(beta * (b - max_b))
        p = exps / np.sum(exps)

        # gradient of weighted convolution wrt a
        a_rev = a[::-1]
        Ar_fft = np.fft.rfft(a_rev, n=size)
        Pr_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:7.1f} score={best_val:.6f}")

        # increase smoothness
        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that uses the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        # indices attaining (almost) the maximum
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]

        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # jitter occasionally
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val

        while time.time() < hill_end:
            elapsed = time.time() - start_hill
            tau = elapsed / (hill_end - start_hill)
            temp = T_start * (T_end / T_start) ** tau

            if rng.random() < 0.65:
                # --- swap two entries -------------------------------------------------
                i, j = rng.integers(0, n_final, size=2)
                if i == j:
                    continue
                new_seq = cur_seq.copy()
                new_seq[i], new_seq[j] = new_seq[j], new_seq[i]
                new_max = _max_convolution_fft(new_seq)
                delta = new_max - cur_max
                accept = delta < 0 or rng.random() < math.exp(-delta / (temp + 1e-12))
                if accept:
                    cur_seq = new_seq
                    cur_max = new_max
                    if cur_max < best_local_max:
                        best_local_max = cur_max
                        best_local_seq = cur_seq.copy()
                        print(f"[{time.time():.2f}] Swap hill new best max={best_local_max:.6f}")
            else:
                # --- guided mass transfer ---------------------------------------------
                conv = _conv_fft(cur_seq)
                kmax = int(np.argmax(conv))
                n = cur_seq.shape[0]
                lo = max(0, kmax - (n - 1))
                hi = min(kmax, n - 1)
                idx = np.arange(lo, hi + 1, dtype=np.int64)

                # contributions to the peak
                prod = cur_seq[idx] * cur_seq[kmax - idx]
                donor = idx[int(np.argmax(prod))]

                low_cnt = max(1, int(len(idx) * 0.06))
                low_candidates = idx[np.argsort(prod)[:low_cnt]]
                receiver = rng.choice(low_candidates)

                if donor == receiver:
                    continue

                max_delta = min(cur_seq[donor] * 0.5, cur_seq[receiver] * 0.5)
                delta = rng.random() * max_delta
                new_seq = cur_seq.copy()
                new_seq[donor] -= delta
                new_seq[receiver] += delta
                new_seq = np.maximum(new_seq, 0.0)
                new_seq = _project_to_simplex(new_seq, target_final)

                new_max = _max_convolution_fft(new_seq)
                delta_score = new_max - cur_max
                accept = delta_score < 0 or rng.random() < math.exp(-delta_score / (temp + 1e-12))
                if accept:
                    cur_seq = new_seq
                    cur_max = new_max
                    if cur_max < best_local_max:
                        best_local_max = cur_max
                        best_local_seq = cur_seq.copy()
                        print(f"[{time.time():.2f}] MT hill new best max={best_local_max:.6f}")

        seq_big = best_local_seq.copy()
        best_val = best_local_max   # because sum == sqrt(2n) ⇒ score = max
        print(f"[{time.time():.2f}] Hill‑climb finished, best max={best_val:.6f}")

    else:
        best_val = _max_convolution_fft(seq_big)

    # --------------------------------------------------------------
    # Phase 5 – final short sub‑gradient polish (if time left)
    # --------------------------------------------------------------
    if time.time() < deadline - 4.0:
        final_deadline = deadline - 2.0
        a_final, val_final = _subgradient_descent(
            seq_big, final_deadline, target_final,
            step_factor=0.45, gamma=0.995)
        if val_final < best_val:
            best_val = val_final
            seq_big = a_final.copy()
            print(f"[{time.time():.2f}] Final sub‑grad improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # --------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # --------------------------------------------------------------
    # Logging of the actual evaluation function (optional)
    # --------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for warm‑starts in subsequent runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
