"""
Improved step‑function optimiser.

The optimiser works in three phases:
  1. Exploration – many short sub‑gradient descents on the exact
     max‑convolution (rather than a soft‑max approximation) from a
     diverse pool of initial shapes.
  2. Upscaling – the best shape is interpolated to a larger length
     (up to ~20000) and polished with a high‑β soft‑max gradient descent.
  3. Permutation annealing – a long simulated‑annealing phase swaps
     indices with high/low contribution to the current peak.
A final tiny mass‑transfer pass trims any remaining excess.

The code is written for the competition harness: only a
`propose_candidate(budget_s: int = 1000) -> list[float]` function is
required.  All heavy work respects the time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)² .  For Σa = √(2n) this equals max(a*a)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (ℓ¹‑simplex Σx = s, x ≥ 0)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation from old length to new_len."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest‑smallest‑largest‑smallest ordering."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


# ----------------------------------------------------------------------
# 1. Exact sub‑gradient descent on max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step0: float = 1.0,
                         gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """
    Simple sub‑gradient descent for   f(a) = max_k (a*a)[k] .
    Uses the exact sub‑gradient at the current peak and projects onto
    the ℓ¹‑simplex after each step.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        # full convolution, peak index
        conv = _conv_fft(a)
        kmax = int(conv.argmax())

        # sub‑gradient: g_i = a_{kmax-i} (if inside range)
        n = a.shape[0]
        i = np.arange(n, dtype=np.int64)
        partner = kmax - i
        mask = (partner >= 0) & (partner < n)
        g = np.zeros_like(a)
        g[mask] = a[partner[mask]]

        # momentum update
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            # step size decays with iteration count
            lr = step0 / (np.sqrt(it + 1) * (norm + 1e-12))
            a -= lr * momentum

        # projection back onto simplex
        a = _project_to_simplex(a, target_sum)

        # occasional small jitter to escape plateaus
        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (it + 1), size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SGD it={it:5d} val={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# 2. Soft‑max gradient descent (high‑β polishing)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.5,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Momentum gradient descent on the soft‑max approximation of
    max‑convolution.  The temperature (β) is increased exponentially.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        # increase β
        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

# 3. Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-4,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # peak index and its contributing partners
        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

# 4. Targeted permutation simulated annealing
# ----------------------------------------------------------------------


def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-5,
                                 top_frac: float = 0.07,
                                 low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing that swaps a high‑contributing index with a
    low‑contributing one.  The whole remaining budget is used here.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    n = a.shape[0]
    start_time = time.time()

    while time.time() < deadline:
        # recompute convolution and the index of the peak
        conv = _conv_fft(a)
        max_val = conv.max()
        kmax_candidates = np.flatnonzero(conv == max_val)
        kmax = rng.choice(kmax_candidates)

        # contribution of each position to the peak
        idx = np.arange(n, dtype=np.int64)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n, dtype=np.float64)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        # choose donor/receiver from tails of distribution
        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)

        if donor == receiver:
            continue

        # swap
        a[donor], a[receiver] = a[receiver], a[donor]

        new_score = _score(a)

        # temperature schedule (slow exponential decay)
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12))

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] PermSearch new best={best_score:.6f}")
        else:
            # revert swap
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.

    Strategy:
      * Build a diverse pool of initial shapes (power‑law, exponential, random).
      * Run many short exact sub‑gradient descents (≈10 s each) until the
        exploration budget is exhausted.
      * Upscale the best shape to a larger length (up to ~20000) and add a
        short high‑β soft‑max gradient polish.
      * Spend the remaining time in a targeted permutation annealing.
      * Finish with a tiny mass‑transfer refinement, trim zeros and return.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Determine exploration size (moderate to keep each gradient cheap)
    # ------------------------------------------------------------------
    if budget_s >= 900:
        n_expl = 12000
    elif budget_s >= 700:
        n_expl = 10000
    else:
        n_expl = 8000
    target_expl = np.sqrt(2.0 * n_expl)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Build initial pool of candidate shapes
    # ------------------------------------------------------------------
    init_pool = []

    # Power‑law family
    for p in np.linspace(0.35, 0.75, 9):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))

    # Exponential family
    for lam in np.linspace(0.0015, 0.015, 7):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))

    # Random Dirichlet‑like points
    for _ in range(8):
        a = rng.random(n_expl)
        a = a / a.sum() * target_expl
        init_pool.append(a)

    # Linear decreasing shape
    lin = np.linspace(1.0, 0.0, n_expl)
    lin = lin / lin.sum() * target_expl
    init_pool.append(lin)

    # If a previous best exists, reuse it (after resizing)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_expl))

    # Add jittered copies for diversity
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.008, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    # ------------------------------------------------------------------
    # 1. Exploration: many short sub‑gradient descents
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    while time.time() < deadline - 12.0:   # keep margin for later phases
        # choose a seed: either jittered best or random pool member
        if best_seq is not None and rng.random() < 0.6:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.004, size=n_expl)
            seed = seed + jitter
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)

        # allocate ~10 s per sub‑gradient run (adjust if close to deadline)
        run_dead = min(time.time() + 10.0, deadline - 6.0)
        if run_dead <= time.time():
            break

        a_opt, val_opt = _subgradient_descent(
            seed,
            run_dead,
            target_expl,
            step0=target_expl * 0.8,
            gamma=0.995,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Brief deterministic mass‑transfer refinement (if time)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 6.0:
        mt_dead = time.time() + (deadline - time.time()) * 0.12
        a_ref, val_ref = _deterministic_mass_transfer(
            best_seq,
            mt_dead,
            target_expl,
            start_temp=0.5,
            end_temp=1e-4,
            low_frac=0.05,
        )
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] After MT refinement best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Upscaling to a larger length (more degrees of freedom)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.55, target_expl)

    # decide final length based on remaining time
    time_left = deadline - time.time()
    if time_left > 120.0 and budget_s >= 900:
        n_final = 20000
    elif time_left > 80.0:
        n_final = 15000
    elif time_left > 40.0:
        n_final = 12000
    else:
        n_final = n_expl

    target_final = np.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # evaluate candidates with simple reorderings
    candidates = [seq_big,
                  _alternating_order(seq_big),
                  _alternating_order(seq_big[::-1])[::-1]]
    best_seq = seq_big
    best_score = _score(seq_big)
    for cand in candidates[1:]:
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand

    print(f"[{time.time():.2f}] Upscaled length={n_final} start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. High‑β polishing via soft‑max gradient descent (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0:
        polish_dead = deadline - 12.0
        a_pol, val_pol = _annealed_gradient_descent(
            best_seq,
            polish_dead,
            target_final,
            max_beta=2e6,
            step_factor=0.5,
            gamma=0.995,
        )
        if val_pol < best_score:
            best_score = val_pol
            best_seq = a_pol.copy()
            print(f"[{time.time():.2f}] Polishing best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Extended permutation simulated annealing (uses most remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        perm_dead = deadline - 0.5
        a_perm, val_perm = _permutation_targeted_search(
            best_seq,
            perm_dead,
            start_temp=0.7,
            end_temp=2e-6,
            top_frac=0.07,
            low_frac=0.07,
        )
        if val_perm < best_score:
            best_score = val_perm
            best_seq = a_perm.copy()
            print(f"[{time.time():.2f}] After permutation search best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final tiny mass‑transfer refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        mt_dead = time.time() + min(5.0, deadline - time.time() - 0.4)
        a_mt2, val_mt2 = _deterministic_mass_transfer(
            best_seq,
            mt_dead,
            target_final,
            start_temp=0.3,
            end_temp=1e-5,
            low_frac=0.05,
        )
        if val_mt2 < best_score:
            best_score = val_mt2
            best_seq = a_mt2.copy()
            print(f"[{time.time():.2f}] Post‑polish MT best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Trim trailing near‑zero entries for compactness
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.flatnonzero(arr > 1e-12)
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # 8. Optional final evaluation (for logging)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for possible later runs in the same session
    globals()["height_sequence_1"] = final_seq
    return final_seq
