def propose_candidate(budget_s=1000):
    start = time.time()
    deadline = start + budget_s - 0.2
    # Determine n_expl: set based on budget.
    # Use 8000 if budget>600 else 5000 else 3000.
    # Compute target_sum = sqrt(2*n_expl)

    # Generate base sequences:
    seq_pool = []
    # Power law
    for p in np.linspace(0.1, 3.0, 10):
        seq_pool.append(_power_law_sequence(n_expl, p, target_sum))
    # Exponential
    for lam in np.linspace(0.0001, 0.05, 10):
        seq_pool.append(_exponential_sequence(n_expl, lam, target_sum))
    # Triangular
    seq_pool.append(_triangular_sequence(n_expl, target_sum))
    # Gaussian with varied sigma
    for sigma in np.linspace(0.1, 0.5, 5):
        seq_pool.append(_gaussian_sequence(n_expl, sigma, target_sum))
    # Beta
    for alpha in [0.5, 1.0, 2.0]:
        for beta in [0.5, 1.0, 2.0]:
            seq_pool.append(_beta_sequence(n_expl, alpha, beta, target_sum))

    # include previous best if exists.
    if 'height_sequence_1' in globals():
        seq_pool.append(_resize_sequence(np.array(globals()['height_sequence_1']), n_expl))

    # Evaluate all pool sequences, keep top ones
    best_seq = None
    best_val = np.inf
    for seq in seq_pool:
        # Already normalized to target sum.
        val = _max_convolution_fft(seq)  # for normalized sum.
        if val < best_val:
            best_val = val
            best_seq = seq.copy()

    print("Initial best val", best_val)

    # Phase A: quick annealed gradient runs using random perturbations.
    # allocate time
    while time.time() < deadline - 60: # keep time for later phases
        # Pick seed: maybe jitter around best_seq or random from pool

"""
Hybrid optimizer with extended shape families and deeper refinement.

The algorithm proceeds in 5 adaptive phases:

1. **Shape exploration** – generate a diverse pool of candidate
   sequences (power‑law, exponential, triangular, Gaussian, Beta
   distributions) on a modest size (≈5‑8 k).  The best shape (lowest
   self‑convolution peak) is kept.

2. **Soft‑max gradient descent** – from the current best shape run many
   short annealed‑gradient passes (smooth approximation of the max‑norm)
   with a rapidly increasing temperature parameter β.  The best
   iterate is retained.

3. **Upscaling** – linearly interpolate the best small‑scale solution to
   a larger length (up to 40 k when time permits) and project onto the
   ℓ¹‑simplex (∑a = √(2n)).  Several deterministic reorderings
   (alternating, even‑spread) are tried and the ordering with the
   smallest convolution peak is kept.

4. **Projected sub‑gradient descent** – exact sub‑gradient steps on the
   true max‑convolution are performed with momentum and a decaying step
   size.  This refines the large‑scale solution.

5. **Greedy mass‑transfer (simulated‑annealing)** – small amounts of mass
   are moved from indices that heavily contribute to the current peak
   toward low‑contributing indices, accepted with a Metropolis rule.
   If time remains a final short annealed‑gradient pass is applied.

All phases respect non‑negativity and the ℓ¹‑simplex constraint.
The resulting sequence is trimmed of trailing near‑zero entries and
returned as a plain list of floats.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (linear convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a)/(∑a)² (for ∑a=√(2n) this equals max conv)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories (all return a vector already normalized to the given ℓ¹‑norm)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _triangular_sequence(n: int, target_sum: float) -> np.ndarray:
    """Symmetric triangle: a_i = min(i+1, n-i) (1‑indexed)."""
    a = np.minimum(np.arange(1, n + 1), np.arange(n, 0, -1)).astype(np.float64)
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma_frac: float, target_sum: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle, σ expressed as a fraction of n."""
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _beta_sequence(n: int, alpha: float, beta: float, target_sum: float) -> np.ndarray:
    """Beta‑distribution shaped sequence on [0,1]."""
    xs = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = (xs ** alpha) * ((1.0 - xs) ** beta)
    a = np.maximum(a, 0.0)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """
    Spread the large values uniformly by placing them on a cyclic sequence
    with a step coprime to the length (≈√n).
    """
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Annealed gradient descent (soft‑max approximation)
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.9,
                               gamma: float = 0.98) -> tuple[np.ndarray, float]:
    """Momentum‑projected GD on the soft‑max approximation of the max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_b = np.max(b)
        exps = np.exp(beta * (b - max_b))
        p = exps / np.sum(exps)

        a_rev = a[::-1]
        Ar_fft = np.fft.rfft(a_rev, n=size)
        Pr_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:7.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent with momentum."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        # indices that are (almost) maximal
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]

        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy mass‑transfer with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.5,
                                      end_temp: float = 1e-4,
                                      low_frac: float = 0.06) -> tuple[np.ndarray, float]:
    """Move a small amount of mass from heavy contributors to light ones."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)
    n = a.shape[0]

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if rng.random() < 0.001:  # occasional global jitter
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

    return best_a, best_score

# Random permutation search (fast heuristic)

def _permutation_search(base_seq: np.ndarray,
                        trials: int,
                        deadline: float) -> tuple[np.ndarray, float]:
    """Search random reorderings of `base_seq` and keep the best."""
    best_seq = base_seq.copy()
    best_val = _max_convolution_fft(best_seq)
    rng = np.random.default_rng()
    n = base_seq.shape[0]

    for _ in range(trials):
        if time.time() >= deadline:
            break
        perm = rng.permutation(base_seq)
        val = _max_convolution_fft(perm)
        if val < best_val:
            best_val = val
            best_seq = perm.copy()
    return best_seq, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine follows the five‑phase schedule described in the module
    docstring and never exceeds `budget_s` seconds.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5        # small safety margin
    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # Phase 0 – choose a modest base size
    # --------------------------------------------------------------
    if budget_s >= 800:
        n_expl = 8000
    elif budget_s >= 500:
        n_expl = 6000
    else:
        n_expl = 4000
    target_expl = math.sqrt(2.0 * n_expl)

    # --------------------------------------------------------------
    # Phase 1 – generate a rich pool of seed shapes
    # --------------------------------------------------------------
    pool: List[np.ndarray] = []

    # power‑law family
    for p in np.linspace(0.1, 3.0, 12):
        pool.append(_power_law_sequence(n_expl, float(p), target_expl))

    # exponential family
    for lam in np.linspace(0.0001, 0.05, 12):
        pool.append(_exponential_sequence(n_expl, float(lam), target_expl))

    # triangular
    pool.append(_triangular_sequence(n_expl, target_expl))

    # Gaussian family (σ as a fraction of n)
    for sigma_frac in np.linspace(0.08, 0.30, 8):
        pool.append(_gaussian_sequence(n_expl, float(sigma_frac), target_expl))

    # Beta family (α,β ∈ {0.5,1,2,3})
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        for beta in [0.5, 1.0, 2.0, 3.0]:
            pool.append(_beta_sequence(n_expl, alpha, beta, target_expl))

    # random points on the simplex
    for _ in range(10):
        pool.append(_random_simplex(n_expl, target_expl, rng))

    # include previous best, if any
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        pool.append(_resize_sequence(prev, n_expl))

    # --------------------------------------------------------------
    # Phase 1b – pick the best raw shape (lowest max‑convolution)
    # --------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    for seq in pool:
        val = _max_convolution_fft(seq)          # sum already = target_expl
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
    print(f"[{time.time():.2f}] Initial best (raw) max={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – annealed gradient descent (soft‑max) on the small scale
    # --------------------------------------------------------------
    while time.time() < deadline - 60:  # leave generous time for later phases
        # jitter around the current best with probability 0.7
        if best_seq is not None and rng.random() < 0.7:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.008, size=n_expl)
            seed += jitter
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = pool[rng.integers(len(pool))]

        run_deadline = min(time.time() + 12.0, deadline - 48.0)
        if run_deadline <= time.time():
            break

        a_opt, val_opt = _annealed_gradient_descent(
            seed,
            run_deadline,
            target_expl,
            max_beta=2e6,
            step_factor=0.8,
            gamma=0.99,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] GD improved best={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – upscale to a larger problem size
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.7, target_expl)

    remaining = deadline - time.time()
    # decide final length based on remaining time
    if remaining > 300:
        n_final = 40000
    elif remaining > 200:
        n_final = 30000
    elif remaining > 120:
        n_final = 25000
    elif remaining > 80:
        n_final = 18000
    else:
        n_final = n_expl

    target_final = math.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # try a few deterministic reorderings
    candidates = [
        seq_big,
        _alternating_order(seq_big),
        _even_spread_order(seq_big),
        _alternating_order(_even_spread_order(seq_big)),
    ]
    seq_big = min(candidates, key=_max_convolution_fft)
    print(f"[{time.time():.2f}] Upscaled to {n_final} elems, start max={_max_convolution_fft(seq_big):.6f}")

    # --------------------------------------------------------------
    # Phase 4 – optional random permutation refinement
    # --------------------------------------------------------------
    if time.time() < deadline - 30:
        perm_deadline = time.time() + min(8.0, deadline - time.time() - 25.0)
        seq_big, _ = _permutation_search(seq_big, trials=60, deadline=perm_deadline)
        print(f"[{time.time():.2f}] After permutation search max={_max_convolution_fft(seq_big):.6f}")

    # --------------------------------------------------------------
    # Phase 5 – sub‑gradient descent on the large scale
    # --------------------------------------------------------------
    if time.time() < deadline - 20:
        sub_deadline = deadline - 12.0
        a_sub, val_sub = _subgradient_descent(
            seq_big,
            sub_deadline,
            target_final,
            step_factor=0.35,
            gamma=0.995,
        )
        if val_sub < best_val:
            best_val = val_sub
            seq_big = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑grad best={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 6 – greedy mass‑transfer with simulated annealing
    # --------------------------------------------------------------
    if time.time() < deadline - 8:
        mt_deadline = deadline - 3.0
        a_mt, val_mt = _mass_transfer_simulated_annealing(
            seq_big,
            mt_deadline,
            target_final,
            start_temp=0.6,
            end_temp=1e-5,
            low_frac=0.04,
        )
        if val_mt < best_val:
            best_val = val_mt
            seq_big = a_mt.copy()
            print(f"[{time.time():.2f}] MT best={best_val:.6f}")

    # --------------------------------------------------------------
    # Final short annealed‑gradient polishing (if time left)
    # --------------------------------------------------------------
    if time.time() < deadline - 2:
        final_deadline = deadline
        a_final, val_final = _annealed_gradient_descent(
            seq_big,
            final_deadline,
            target_final,
            max_beta=3e6,
            step_factor=0.6,
            gamma=0.98,
        )
        if val_final < best_val:
            best_val = val_final
            seq_big = a_final.copy()
            print(f"[{time.time():.2f}] Final GD polish best={best_val:.6f}")

    # --------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # --------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # --------------------------------------------------------------
    # Store for possible future runs and final logging
    # --------------------------------------------------------------
    globals()["height_sequence_1"] = final_seq
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    return final_seq
