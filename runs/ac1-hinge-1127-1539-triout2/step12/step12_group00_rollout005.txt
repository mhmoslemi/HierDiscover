"""
Hybrid optimizer with stronger LP refinement and longer polishing.

The algorithm consists of the following phases (time‑controlled):

1. **Exploration (annealed soft‑max GD).**  Many short runs from diverse
   seeds (power‑law, exponential, random, previous best).  The β
   temperature is increased aggressively up to 1e6, and a momentum
   scheme is used.

2. **Deterministic mass‑transfer (DMT).**  A short Metropolis‑style
   redistribution of mass from “hot” convolution positions to “cold”
   ones.

3. **Linear‑programming minimisation of the current max‑convolution.**
   For a down‑scaled copy of the current sequence we solve a genuine
   LP that *minimises* the peak M subject to linear constraints
   `f * b ≤ M`.  The solution `b` is mixed back into the current
   sequence.

4. **Upscaling & reordering.**  The improved short sequence is
   interpolated to a larger length (≈16k–20k) and three simple
   reorderings are tested; the one with smallest convolution peak is
   kept.

5. **Sub‑gradient polishing (large scale).**  Projected sub‑gradient
   descent on the exact max‑convolution, with a linearly decaying
   step‑size and momentum.

6. **Final soft‑max GD + DMT.**  A final short annealed GD run,
   followed by a longer DMT pass.

All updates respect non‑negativity and the ℓ¹‑simplex
(`Σ a = √(2n)`).  The routine stops before the allotted budget
(`budget_s` seconds) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)²."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection of a vector onto {x≥0, Σx=s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread large values uniformly using a coprime step."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Soft‑max gradient (annealed GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent_v2_ext(a0: np.ndarray,
                                     deadline: float,
                                     target_sum: float,
                                     max_beta: float = 1e6,
                                     step_factor: float = 0.8,
                                     gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected momentum GD on a smooth soft‑max approximation."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent on exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         start_step_factor: float = 0.8,
                         end_step_factor: float = 0.2,
                         gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with a linearly decaying step size."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    start = time.time()

    it = 0
    while time.time() < deadline:
        # exact max‑conv and sub‑gradient
        conv = _conv_fft(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]
        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        # momentum + decaying step
        momentum = gamma * momentum + grad
        elapsed = time.time() - start
        total = deadline - start
        if total <= 0:
            break
        # linear decay of the factor controlling learning‑rate magnitude
        step_factor = end_step_factor + (start_step_factor - end_step_factor) * (1 - elapsed / total)
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0003, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (Metropolis style)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-5,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy mass moves with a decaying temperature."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={int(time.time()-start_time)} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if int(time.time() - start_time) % 3000 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# LP refinement that *minimises* the current max‑convolution
# ----------------------------------------------------------------------


def _lp_optimize_minM(f: np.ndarray,
                      target_sum: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float] | None:
    """
    Solve  min M   s.t.  f * b ≤ M ,  b ≥ 0 , Σb = target_sum.
    Returns (b, M) or None on failure.
    """
    n = f.shape[0]
    # Build A such that A @ b ≤ M  (i.e. A b - M ≤ 0)
    rows = []
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=np.int64)
        row = np.zeros(n, dtype=np.float64)
        row[i] = f[k - i]
        rows.append(row)
    A = np.vstack(rows)                     # (2n‑1) × n
    # Append column for M with coefficient -1
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1), dtype=np.float64)])
    b_ub = np.zeros(2 * n - 1, dtype=np.float64)

    # Equality: Σ b = target_sum
    A_eq = np.hstack([np.ones((1, n), dtype=np.float64), np.zeros((1, 1), dtype=np.float64)])
    b_eq = np.array([target_sum], dtype=np.float64)

    # Objective: minimise M
    c = np.concatenate([np.zeros(n, dtype=np.float64), np.array([1.0], dtype=np.float64)])

    bounds = [(0.0, None)] * n + [(0.0, None)]   # M ≥ 0

    try:
        from scipy.optimize import linprog
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception as e:
        print(f"[{time.time():.2f}] LP exception: {e}")
        return None

    if not res.success:
        print(f"[{time.time():.2f}] LP failed: {res.message}")
        return None

    sol = res.x
    b = sol[:n]
    M = sol[-1]
    if b.sum() <= 0:
        return None
    # b already satisfies the ℓ¹‑norm, but numerical drift may happen
    b = _project_to_simplex(b, target_sum)
    return b, float(M)

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid search for a non‑negative step function with a low evaluation score.
    The routine follows the phases described in the module docstring.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Choose an exploratory size (large enough for good quality)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n_expl = 9000
    elif budget_s >= 400:
        n_expl = 7000
    else:
        n_expl = 5000
    target_expl = np.sqrt(2.0 * n_expl)

    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # Build a diverse pool of initial seeds
    # --------------------------------------------------------------
    init_pool = []

    # power‑law family
    for p in np.linspace(0.2, 1.2, 9):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))

    # exponential family
    for lam in np.linspace(0.0005, 0.025, 7):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))

    # random simplex points
    for _ in range(8):
        init_pool.append(_random_simplex(n_expl, target_expl, rng))

    # previous best (if any)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_expl))

    # jittered copies of everything in the pool
    pool_copy = list(init_pool)
    for seq in pool_copy:
        jitter = rng.normal(scale=target_expl * 0.02, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    # --------------------------------------------------------------
    # Exploration: many short annealed‑gradient runs
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    while time.time() < deadline - 120.0:   # keep ~2 min for later phases
        # choose seed
        if best_seq is not None and rng.random() < 0.6:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.005, size=n_expl)
            seed += jitter
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)

        # allocate a short window (≈25 s) for each GD run
        run_dead = min(time.time() + 25.0, deadline - 115.0)
        if run_dead <= time.time():
            break

        a_opt, val_opt = _annealed_gradient_descent_v2_ext(
            seed, run_dead, target_expl,
            max_beta=1e6, step_factor=0.8, gamma=0.99,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Light deterministic mass‑transfer before LP stage
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 100.0:
        mt_dead = time.time() + (deadline - time.time()) * 0.07   # ~7 % of remaining time
        a_ref, val_ref = _deterministic_mass_transfer(
            best_seq, mt_dead, target_expl,
            start_temp=0.4, end_temp=1e-4, low_frac=0.06,
        )
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] After MT refinement best={best_score:.6f}")

    # --------------------------------------------------------------
    # LP minimisation on a down‑scaled problem
    # --------------------------------------------------------------
    if time.time() < deadline - 80.0:
        n_lp = min(1300, max(900, n_expl // 6))
        target_lp = np.sqrt(2.0 * n_lp)
        seq_lp = _resize_sequence(best_seq, n_lp)
        seq_lp = _project_to_simplex(seq_lp, target_lp)

        lp_iters = 6
        for lp_it in range(lp_iters):
            if time.time() > deadline - 70.0:
                break
            M = _max_convolution_fft(seq_lp)
            res = _lp_optimize_minM(seq_lp, target_lp, rng)
            if res is None:
                break
            b_opt, M_opt = res
            # mix a bit towards the LP solution
            t = 0.25
            seq_lp = (1 - t) * seq_lp + t * b_opt
            seq_lp = _project_to_simplex(seq_lp, target_lp)
            cur_score = _score(seq_lp)
            print(f"[{time.time():.2f}] LP it={lp_it} score={cur_score:.6f}")

        # optional brief GD polishing on the small problem
        if time.time() < deadline - 60.0:
            gd_dead = time.time() + min(12.0, deadline - time.time() - 55.0)
            a_opt, val_opt = _annealed_gradient_descent_v2_ext(
                seq_lp, gd_dead, target_lp,
                max_beta=5e5, step_factor=0.6, gamma=0.98,
            )
            if val_opt < _score(seq_lp):
                seq_lp = a_opt.copy()
                print(f"[{time.time():.2f}] Post‑LP GD score={val_opt:.6f}")

        # Upscale to a larger size for final polishing
        remaining = deadline - time.time()
        if remaining > 120.0:
            n_final = 20000
        elif remaining > 70.0:
            n_final = 16000
        else:
            n_final = max(8000, n_expl)
        target_final = np.sqrt(2.0 * n_final)
        seq_big = _resize_sequence(seq_lp, n_final)
        seq_big = _project_to_simplex(seq_big, target_final)
    else:
        # Not enough time for LP – just upscale the best we have
        remaining = deadline - time.time()
        if remaining > 120.0:
            n_final = 20000
        elif remaining > 70.0:
            n_final = 16000
        else:
            n_final = max(8000, n_expl)
        target_final = np.sqrt(2.0 * n_final)
        seq_big = _resize_sequence(best_seq, n_final)
        seq_big = _project_to_simplex(seq_big, target_final)

    # --------------------------------------------------------------
    # Simple reorderings to reduce the convolution peak
    # --------------------------------------------------------------
    candidates = [seq_big,
                  _alternating_order(seq_big),
                  _even_spread_order(seq_big)]
    seq_big = min(candidates, key=_max_convolution_fft)
    best_score = _score(seq_big)
    print(f"[{time.time():.2f}] Upscaled length={len(seq_big)} start score={best_score:.6f}")

    # --------------------------------------------------------------
    # Sub‑gradient polishing on the large problem (majority of time)
    # --------------------------------------------------------------
    if time.time() < deadline - 30.0:
        sub_dead = deadline - 28.0
        a_sub, val_sub = _subgradient_descent(
            seq_big, sub_dead, target_final,
            start_step_factor=0.8, end_step_factor=0.2, gamma=0.995,
        )
        if val_sub < best_score:
            best_score = val_sub
            seq_big = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑grad polished best={best_score:.6f}")

    # --------------------------------------------------------------
    # Final soft‑max GD polishing (short)
    # --------------------------------------------------------------
    if time.time() < deadline - 12.0:
        gd_dead = deadline - 10.0
        a_gd, val_gd = _annealed_gradient_descent_v2_ext(
            seq_big, gd_dead, target_final,
            max_beta=5e5, step_factor=0.6, gamma=0.98,
        )
        if val_gd < best_score:
            best_score = val_gd
            seq_big = a_gd.copy()
            print(f"[{time.time():.2f}] Final GD polished best={best_score:.6f}")

    # --------------------------------------------------------------
    # Final deterministic mass‑transfer pass (few minutes)
    # --------------------------------------------------------------
    if time.time() < deadline - 4.0:
        mt_dead = time.time() + min(20.0, deadline - time.time() - 2.0)
        a_mt, val_mt = _deterministic_mass_transfer(
            seq_big, mt_dead, target_final,
            start_temp=0.25, end_temp=1e-5, low_frac=0.05,
        )
        if val_mt < best_score:
            best_score = val_mt
            seq_big = a_mt.copy()
            print(f"[{time.time():.2f}] Final MT best={best_score:.6f}")

    # --------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # --------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Optional final evaluation (purely for logging)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    globals()["height_sequence_1"] = final_seq
    return final_seq
