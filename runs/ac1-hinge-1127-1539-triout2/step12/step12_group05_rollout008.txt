"""
Improved step‑function optimizer.

Main ideas
----------

1. **Adaptive problem size** – pick a larger discretisation (up to 20 k)
   when the time budget allows it, because a finer grid gives a
   strictly larger feasible set.

2. **Rich parametric warm‑start** – scan several classic families
   (pure power‑law, exponential, exponential‑power‑law, broken power‑law)
   with a finer grid than before, keeping the best shape as an initial
   point.

3. **Multiple Adam restarts** – run projected Adam on the soft‑max
   approximation of the max‑convolution from many diverse starts
   (the best parametric shape, a uniform vector and a few random
   Dirichlet vectors).  The beta schedule is accelerated so that the
   true sub‑gradient of the max appears early.

4. **Heavy guided mass‑transfer** – after the Adam phase we spend the
   bulk of the remaining budget in the exact sub‑gradient “move mass
   from bad to good” loop.  Each iteration reduces the current peak to
   the second‑largest value (or stops when no reduction is possible).

5. **Final polishing** – one last Adam run with a very high softmax
   temperature (β≈10⁶) to clean up any remaining tiny irregularities.

All steps respect the simplex constraint Σa = √(2 n) and non‑negativity.
The routine returns the best sequence found (trailing ~0 entries trimmed)
as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) using a real FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft_pair(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution x*y (full length 2·n‑1) using a real FFT."""
    n = x.size
    size = 1 << ((2 * n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Benchmark score (equal to max convolution when Σa = √(2n)).
    """
    n = a.shape[0]
    max_b = _max_conv(a)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (dense pre‑search)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_sequence(n: int, q: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-q * i / (n-1))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-q * i / (n - 1))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} * exp(-q * i / (n-1))."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / (n - 1))
    a *= target_sum / a.sum()
    return a


def _broken_power_law_sequence(
    n: int, p1: float, p2: float, alpha: float, target_sum: float
) -> np.ndarray:
    """Two‑segment power law: exponent p1 for the first α·n entries,
    exponent p2 thereafter."""
    K = max(1, int(alpha * n))
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.empty_like(idx)
    a[:K] = idx[:K] ** (-p1)
    a[K:] = idx[K:] ** (-p2)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max approximation)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(a*a).

    Returns (grad, full_convolution) where grad has the same size as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = conv_full.max()
    # soft‑max over the 2n‑1 entries
    exps = np.exp(beta * (conv_full - max_b))
    p = exps / exps.sum()

    # we need the gradient of Σ p_k * (a*a)_k  w.r.t. a
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    # the gradient (size n) lives in the central n entries of conv_rev
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, conv_full


def _adam_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 5e5,
    step_factor: float = 0.8,
    gamma: float = 0.995,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the soft‑max approximation of the max‑convolution.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 200  # accelerate beta growth

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 300 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} score={best_val:.6f}"
            )

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient guided mass‑transfer (Phase B)
# ----------------------------------------------------------------------


def _compute_gradient_for_max(a: np.ndarray, max_idx: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a (averaged over maximal indices).
    """
    n = a.size
    g = np.zeros_like(a)
    for k in max_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g[idx] += 2.0 * a[k - idx]
    if max_idx.size:
        g /= max_idx.size
    return g


def _guided_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Greedy exact reduction of the maximal convolution entry.
    Each iteration moves a small amount of mass from indices that
    increase the current peak towards indices that decrease it, using an
    exact line‑search based on the sub‑gradient.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()

        # mask for all entries that are (essentially) maximal
        mask_max = conv >= max_val - tol
        max_idx = np.where(mask_max)[0]

        # second largest entry (ignore all maximal positions)
        if np.any(~mask_max):
            second_max = conv[~mask_max].max()
        else:
            second_max = 0.0

        # exact averaged sub‑gradient of the peak(s)
        g = _compute_gradient_for_max(a, max_idx)

        # direction preserving the simplex sum
        sum_g = g.sum()
        d = -g + (sum_g / n) * np.ones_like(a)  # Σd = 0

        # largest admissible step before any component becomes negative
        neg = d < 0
        if np.any(neg):
            t_nonneg = np.min(a[neg] / -d[neg])
        else:
            t_nonneg = np.inf

        # linearised change of the convolution in direction d
        delta = 2.0 * _conv_fft_pair(a, d)  # length 2n‑1
        delta_max = delta[max_idx].min()   # most favourable peak to reduce

        if delta_max >= 0:
            # no descent direction – stop
            break

        # step that would bring the current peak down to the second peak
        t_to_second = (max_val - second_max) / (-delta_max)

        # safe step (slightly back off from the border)
        t = min(t_nonneg, t_to_second) * 0.99

        if t <= 0:
            break

        a += t * d
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] GuidedMT it={it:5d} score={best_score:.6f}")

        # occasional jitter to shake out tiny plateaus
        if it % 500 == 0 and rng.random() < 0.05:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_score

def _global_random_move(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
) -> tuple[np.ndarray, float]:
    """Very lightweight simulated‑annealing style mass moves."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    while time.time() < deadline - 0.5:
        # donor from top 2 % mass, receiver from bottom 2 %
        top_k = max(1, int(0.02 * n))
        low_k = max(1, int(0.02 * n))
        donor = np.argsort(-a)[rng.integers(top_k)]
        receiver = np.argsort(a)[rng.integers(low_k)]
        if donor == receiver:
            continue
        delta = rng.random() * a[donor] * 0.05
        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] RandomMove new best={best_score:.6f}")

    return best_a, best_score

                best_score, best_seq = val, seq.copy()

    # (d) broken power‑law (two‑segment)
    for p1 in np.arange(0.0, 1.01, 0.1):
        for p2 in np.arange(0.8, 3.01, 0.2):
            for alpha in np.arange(0.05, 0.96, 0.05):
                seq = _broken_power_law_sequence(
                    n, float(p1), float(p2), float(alpha), target_sum
                )
                val = _score(seq)
                if val < best_score:
                    best_score, best_seq = val, seq.copy()

    # (e) reuse external best if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score, best_seq = val, resized.copy()

    print(f"[{time.time():.2f}] After dense pre‑search: best score = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase A – Adam restarts from diverse starts
    # ---------------------------------------------------------------
    remaining = deadline - time.time()
    # allocate roughly 20 % of the remaining budget to Adam restarts
    adam_deadline = time.time() + 0.20 * remaining

    # 1) start from the best parametric sequence (already have)
    # 2) uniform (constant) sequence
    uniform = np.full(n, target_sum / n, dtype=np.float64)
    # 3) a few random Dirichlet points
    random_starts = [uniform, best_seq]
    for _ in range(3):
        rnd = rng.dirichlet(np.ones(n))
        rnd *= target_sum
        random_starts.append(rnd)

    # Run Adam on each start (they share the same global deadline)
    for start_vec in random_starts:
        if time.time() >= adam_deadline:
            break
        a_opt, val_opt = _adam_gradient_descent(
            start_vec,
            adam_deadline,
            target_sum,
            max_beta=8e5,
            step_factor=0.85,
            gamma=0.995,
        )
        if val_opt < best_score:
            best_score, best_seq = val_opt, a_opt.copy()
            print(f"[{time.time():.2f}] Adam restart improved best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase B – intensive guided mass‑transfer
    # ---------------------------------------------------------------
    # The bulk of the remaining time is spent here.
    while time.time() < deadline - 8.0:
        # a) guided exact mass‑transfer
        a_ref, val_ref = _guided_mass_transfer(best_seq, deadline, target_sum)
        if val_ref < best_score:
            best_score, best_seq = val_ref, a_ref.copy()
            print(f"[{time.time():.2f}] GuidedMT improved best = {best_score:.6f}")

        # b) occasional cheap random moves
        a_rand, val_rand = _global_random_move(best_seq, deadline, target_sum)
        if val_rand < best_score:
            best_score, best_seq = val_rand, a_rand.copy()
            print(f"[{time.time():.2f}] RandomMove improved best = {best_score:.6f}")

        # safety exit
        if time.time() > deadline - 2.0:
            break

    # ---------------------------------------------------------------
    # Phase C – final polishing with very high β Adam run
    # ---------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(
            best_seq,
            deadline,
            target_sum,
            max_beta=1e6,
            step_factor=0.6,
            gamma=0.995,
        )
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final polish improved best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return list
    # ---------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final evaluation (harmless if evaluate_sequence missing)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated benchmark score = {val:.6f}")
    except Exception:
        pass

    return final_seq
