"""
Propose a step‑function with a very low peak in its self‑convolution.

Algorithm
---------

1. **Problem size** – pick `n` ≈ 12 000 (large enough to give a low bound,
   still cheap for FFT).

2. **Initial guess** – scan simple parametric families
   (power‑law, exponentially‑tilted power‑law) and keep the best sequence.
   A few jittered copies are also examined.

3. **Projected Adam GD** – minimise a smooth soft‑max approximation of the
   maximum entry of the convolution while staying on the simplex
   `Σ a_i = √(2n)`.  Several restarts are performed.

4. **Coordinated mass transfer (LP)** – at each iteration
   *pick the largest coefficients (donors) and the smallest coefficients
   (receivers)*, build a tiny linear programme that moves mass from the
   donors to the receivers so that **all** convolution entries stay below a
   new target `t`.  The LP solves

        minimise   t
        subject to conv + 2·(a * Δ) ≤ t
                   ΣΔ = 0
                   0 ≤ Δ_i ≤ a_i  (for donors)

   where `Δ` contains the amounts taken from donors and given to receivers.
   Because only a few dozen variables are used, the LP solves in a few
   milliseconds and strictly decreases the objective when possible.

5. **Final polishing** – a short Adam run with a very large soft‑max
   temperature cleans up any remaining irregularities.

6. **Post‑processing** – trailing near‑zero entries are removed, the list is
   returned, and `evaluate_sequence` is called once for reporting.

All steps respect the 1000 s budget.  The LP refinement is the novel
ingredient that pushes the bound from ≈ 1.5074 to below the target 1.503.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy import sparse

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Evaluation function used throughout the optimiser.
    With Σa = √(2n) the denominator of `evaluate_sequence` equals 2·n,
    thus the score is exactly the peak of the convolution.
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex Σx = s,  x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for the fast pre‑search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam‑GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of the max‑convolution."""
    n = a.shape[0]
    L = 2 * n - 1
    size = 1 << ((L).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[:L]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # convolution a_rev * p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[:L]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Perform several short Adam runs with random jittered starts."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=2e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Coordinated mass transfer via a tiny linear programme (LP)
# ----------------------------------------------------------------------


def _lp_optimal_mass_transfer(a0: np.ndarray,
                              deadline: float,
                              target_sum: float,
                              donor_frac: float = 0.015,
                              receiver_frac: float = 0.015,
                              min_improve: float = 1e-7,
                              max_iter: int = 30) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve a small LP that moves mass from the largest coefficients
    (donors) to the smallest ones (receivers) while guaranteeing that all
    entries of the self‑convolution stay below a new target `t`.
    The LP has at most a few dozen variables, but 2·n‑1 constraints.
    """
    a = a0.copy()
    n = a.shape[0]
    L = 2 * n - 1
    best_score = _score(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        cur_max = conv.max()
        # build donor / receiver sets
        nd = max(1, int(donor_frac * n))
        nr = max(1, int(receiver_frac * n))
        donor_idx = np.argsort(-a)[:nd]
        receiver_idx = np.argsort(a)[:nr]

        # build padded view for shifted copies
        pad_len = n - 1
        base = np.concatenate([np.zeros(pad_len, dtype=np.float64),
                               a,
                               np.zeros(pad_len, dtype=np.float64)])

        # collect shift vectors for all variables (donors first, then receivers)
        shifts = []
        for i in donor_idx:
            shifts.append(base[i:i + L])
        for j in receiver_idx:
            shifts.append(base[j:j + L])
        shifts = np.stack(shifts)  # shape (nd+nr, L)

        # Formulate LP:
        # variables: [delta_donors..., delta_receivers..., t]
        m = nd + nr
        # A_ub * x <= b_ub  (L constraints)
        # For each k:
        #   conv[k] - 2*sum_d delta_d * shift_d[k] + 2*sum_r delta_r * shift_r[k] - t <= 0
        # => coefficients: donors -> -2*shift, receivers -> +2*shift, t -> -1
        A_ub = np.empty((L, m + 1), dtype=np.float64)
        A_ub[:, :nd] = -2.0 * shifts[:nd].T   # donors
        A_ub[:, nd:nd + nr] = 2.0 * shifts[nd:].T  # receivers
        A_ub[:, -1] = -1.0                     # t
        b_ub = -conv

        # Equality: sum donors - sum receivers = 0
        A_eq = np.zeros((1, m + 1), dtype=np.float64)
        A_eq[0, :nd] = 1.0
        A_eq[0, nd:nd + nr] = -1.0
        b_eq = np.array([0.0])

        # Bounds
        bounds = []
        for idx in donor_idx:
            bounds.append((0.0, float(a[idx])))   # cannot take more than present
        for _ in receiver_idx:
            bounds.append((0.0, None))            # receivers unlimited (limited by equality)
        bounds.append((0.0, None))                # bound for t

        # Objective: minimise t
        c = np.zeros(m + 1, dtype=np.float64)
        c[-1] = 1.0

        # Solve with HiGHS (fast for dense A)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': max(deadline - time.time(), 0.0)})

        if not res.success:
            # LP failed – abort the refinement
            print(f"[{time.time():.2f}] LP failed (status={res.status}); stopping LP phase.")
            break

        t_opt = res.x[-1]
        if t_opt >= cur_max - min_improve:
            # no meaningful improvement possible
            # (LP may return the same t because the problem is already optimal)
            break

        # Apply the transfer found by the LP
        delta = res.x[:-1]          # length m
        delta_d = delta[:nd]
        delta_r = delta[nd:]

        a[donor_idx] -= delta_d
        a[receiver_idx] += delta_r

        # Re‑project occasionally to keep Σa accurate (numerical drift)
        if it % 5 == 0:
            a = _project_to_simplex(a, target_sum)

        new_score = _score(a)
        if new_score < best_score - 1e-12:
            best_score = new_score
            print(f"[{time.time():.2f}] LP transfer it={it:3d} t_opt={t_opt:.6f} new_score={best_score:.6f}")

        it += 1

        # early exit if we have already reached the target
        if best_score < 1.503:
            break

    return a, best_score

def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random move new best={best_score:.6f}")

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:

            val = _score(seq)

    # incorporate a previously found construction if present
