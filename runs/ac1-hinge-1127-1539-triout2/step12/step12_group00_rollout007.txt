def lp_minimize_max_convolution(f, target_sum):
    n = len(f)
    # Build A with shape (2*n-1, n+1)
    rows = []
    for k in range(2*n-1):
        lo = max(0, k-(n-1))
        hi = min(k, n-1)
        i = np.arange(lo, hi+1, dtype=int)
        row = np.zeros(n+1)
        row[i] = f[k-i]  # fill coefficients
        row[-1] = -1.0  # coefficient for M (since we require sum_i f_{k-i} b_i - M <= 0)
        rows.append(row)
    A_ub = np.vstack(rows)
    b_ub = np.zeros(2*n-1)
    # Equality sum b_i = target_sum
    A_eq = np.zeros((1, n+1))
    A_eq[0,:n] = 1.0
    b_eq = np.array([target_sum])
    # Bounds for b_i: (0, None), for M: (0, None)
    bounds = [(0, None)]*n + [(0, None)]
    c = np.zeros(n+1)
    c[-1] = 1.0  # minimize M
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if not res.success: return None
    b = res.x[:n]
    M = res.x[-1]
    return b, M
```

"""
Hybrid search for the minimisation of

    2·n·max(a*a) / (sum(a)**2)

with the ℓ¹‑normalisation Σa = √(2 n).  The method consists of

1.  Sampling a diverse pool of seed sequences (power‑law, exponential,
    random Dirichlet) and keeping the best.
2.  A short sub‑gradient polishing on the original size.
3.  Down‑scaling to a moderate size (≈1000) and solving **exact** linear
    programmes that minimise the current convolution bound  M  in

        f * b ≤ M ,   Σb = √(2 n) ,   b ≥ 0 .

   The LP gives the best possible b for the current f.  We mix the new
   b with the current f (mixing factor 0.3) and repeat while time permits.
4.  Upscaling back to a large size (≈12000), a few heuristic reorderings,
    a final large‑scale sub‑gradient descent and a deterministic
    mass‑transfer (Metropolis‑style) annealing.
5.  Optional random permutations for a few seconds to eliminate any
    remaining peak.

All updates respect non‑negativity and the ℓ¹‑simplex constraint.
The final sequence is stored in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (Σa)².
    For Σa = √(2·n) this equals max(a*a).
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → left, second largest → right, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread large values uniformly using a coprime step."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Sub‑gradient descent for the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]
        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (Metropolis style)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy mass moves with a decaying temperature."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            it += 1
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact LP to minimise the current convolution bound  M  for a fixed f
# ----------------------------------------------------------------------


def _lp_minimize_max_convolution(f: np.ndarray,
                                 target_sum: float) -> tuple[np.ndarray, float] | tuple[None, None]:
    """
    Solve
        minimise   M
        subject to f * b ≤ M ,   Σb = target_sum ,   b ≥ 0 .
    Returns the optimal b and the minimal M.
    """
    n = f.shape[0]
    rows = np.zeros((2 * n - 1, n + 1), dtype=np.float64)   # last column is -M
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=np.int64)
        rows[k, i] = f[k - i]           # coefficient f_{k-i}
        rows[k, -1] = -1.0               # -M
    A_ub = rows
    b_ub = np.zeros(2 * n - 1, dtype=np.float64)

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum], dtype=np.float64)

    bounds = [(0.0, None)] * n + [(0.0, None)]   # b_i ≥0 , M ≥0
    c = np.zeros(n + 1, dtype=np.float64)
    c[-1] = 1.0                                  # minimise M

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception as e:
        print(f"[{time.time():.2f}] LP exception: {e}")
        return None, None

    if not res.success:
        print(f"[{time.time():.2f}] LP failed: {res.message}")
        return None, None

    b_opt = res.x[:n]
    M_opt = res.x[-1]
    return b_opt, M_opt

        sub_dead = min(deadline - 2.0, time.time() + 25.0)
        a_sub, val_sub = _subgradient_descent(best_seq, sub_dead,
                                               target_expl,
                                               step_factor=0.6,
                                               gamma=0.99)
        if val_sub < best_score:
            best_score = val_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑grad polished seed score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Down‑scale for exact LP refinement
    # ------------------------------------------------------------------
    n_lp = min(1200, max(800, n_expl // 7))
    target_lp = math.sqrt(2.0 * n_lp)
    seq_lp = _resize_sequence(best_seq, n_lp)
    seq_lp = _project_to_simplex(seq_lp, target_lp)

    # ------------------------------------------------------------------
    # Exact LP alternating refinement (mixing)
    # ------------------------------------------------------------------
    mixing_factor = 0.3
    max_lp_iters = 30
    lp_iter = 0
    while time.time() < deadline - 30.0 and lp_iter < max_lp_iters:
        b_opt, M_opt = _lp_minimize_max_convolution(seq_lp, target_lp)
        if b_opt is None:
            break
        # Mix the new solution with the current one
        seq_lp = (1.0 - mixing_factor) * seq_lp + mixing_factor * b_opt
        seq_lp = _project_to_simplex(seq_lp, target_lp)
        cur_val = _score(seq_lp)
        if cur_val < best_score:
            best_score = cur_val
            best_seq = seq_lp.copy()
            print(f"[{time.time():.2f}] LP iter {lp_iter:2d} score = {cur_val:.6f}")
        lp_iter += 1

    # ------------------------------------------------------------------
    # Upscale to a large size for final polishing
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 150:
        n_big = 16000
    elif remaining > 80:
        n_big = 12000
    elif remaining > 40:
        n_big = 8000
    else:
        n_big = n_lp
    target_big = math.sqrt(2.0 * n_big)
    seq_big = _resize_sequence(seq_lp, n_big)
    seq_big = _project_to_simplex(seq_big, target_big)

    # ------------------------------------------------------------------
    # Simple heuristic reorderings
    # ------------------------------------------------------------------
    candidates = [seq_big,
                  _alternating_order(seq_big),
                  _even_spread_order(seq_big)]
    seq_big = min(candidates, key=_max_convolution_fft)
    best_score = _score(seq_big)
    print(f"[{time.time():.2f}] After up‑scale reordering length={n_big} start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Large‑scale sub‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        sub_dead = deadline - 4.0
        a_sub, val_sub = _subgradient_descent(seq_big, sub_dead,
                                               target_big,
                                               step_factor=0.45,
                                               gamma=0.995)
        if val_sub < best_score:
            best_score = val_sub
            seq_big = a_sub.copy()
            print(f"[{time.time():.2f}] Large sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Deterministic mass‑transfer annealing
    # ------------------------------------------------------------------
    if time.time() < deadline - 4.0:
        mt_dead = time.time() + min(6.0, deadline - time.time() - 1.0)
        a_mt, val_mt = _deterministic_mass_transfer(seq_big, mt_dead,
                                                    target_big,
                                                    start_temp=0.25,
                                                    end_temp=1e-5,
                                                    low_frac=0.05)
        if val_mt < best_score:
            best_score = val_mt
            seq_big = a_mt.copy()
            print(f"[{time.time():.2f}] Final MT best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional random permutations for a few seconds
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        perm_dead = time.time() + min(5.0, deadline - time.time() - 0.5)
        best_seq_local = seq_big.copy()
        best_local_score = best_score
        while time.time() < perm_dead:
            perm = rng.permutation(len(seq_big))
            cand = seq_big[perm]
            cand_score = _score(cand)
            if cand_score < best_local_score:
                best_local_score = cand_score
                best_seq_local = cand.copy()
                print(f"[{time.time():.2f}] Random perm improved to {best_local_score:.6f}")
        seq_big = best_seq_local
        best_score = best_local_score

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # ------------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Store globally and final logging
    # ------------------------------------------------------------------
    globals()["height_sequence_1"] = final_seq
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
