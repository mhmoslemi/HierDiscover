"""
Improved step‑function optimiser.

The search proceeds in three stages:

1. **Broad exploration** – many short Adam‑optimisations on a modest
   problem size (≈ 4 000).  A rich pool of initial shapes is used:
   decreasing power‑laws, exponentials, edge‑heavy “U‑shapes”, symmetric
   versions and random Dirichlet points.  Jittered copies increase
   diversity.  The best candidate is kept.

2. **Upscaling & deep polishing** – the best candidate is interpolated
   to a larger length (≈ 10 000–12 000) and refined with a long Adam run
   on the soft‑max approximation of the peak convolution.  The learning
   rate follows an Adam schedule and the smoothness parameter β is
   increased exponentially until a very large value (≈ 2·10⁶).

3. **Final local search** – an extended targeted permutation annealing
   (swap high‑contributing with low‑contributing indices) consumes the
   remaining time, followed by a tiny mass‑transfer refinement.  The
   result is trimmed and returned.

All updates stay on the ℓ¹‑simplex (∑a = √(2 n)) and respect the
non‑negativity constraints required by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Scoring (objective = max convolution entry)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a)/(∑a)² – with ∑a = √(2n) this is simply max."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the probability simplex scaled by s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _edge_heavy_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """U‑shaped sequence: a_i ∝ ((i+1)*(n-i))^{-p}."""
    idx = np.arange(n, dtype=np.float64) + 1
    product = idx * (n - np.arange(n, dtype=np.float64))
    a = product ** (-p)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to change the length of a sequence."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place the sorted values alternately at the left and right ends."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


def _symmetrize(seq: np.ndarray) -> np.ndarray:
    """Force symmetry a_i = a_{n-1-i}."""
    return (seq + seq[::-1]) * 0.5


# ----------------------------------------------------------------------
# Soft‑max smoothing of the max‑convolution and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the smooth approximation
        g(a) = logsumexp(beta * (a*a)) / beta
    Returns the gradient (shape n) and the raw convolution vector.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)                     # soft‑max probabilities

    # convolution of reversed a with p gives the gradient of the smoothed max
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]     # length n
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on the soft‑max objective
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 5e5,
                           step_factor: float = 0.5,
                           beta_factor: float = 2.0,
                           beta1: float = 0.9,
                           beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth max‑convolution objective.
    The iterate is projected onto the ℓ¹‑simplex after each step.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5                       # start smoothing
    m = np.zeros_like(a)             # first moment
    v = np.zeros_like(a)             # second moment
    t = 0
    eps = 1e-8
    rng = np.random.default_rng()

    # step size normaliser (roughly target_sum * step_factor)
    step_base = target_sum * step_factor

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning rate adapts to the smoothing parameter
        lr = step_base / ((np.sqrt(v_hat) + eps) * (beta + 1.0))

        a -= lr * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if t % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={t:5d} β={beta:6.1f} score={best_val:.6f}")

        # increase smoothing parameter
        if t % 400 == 0 and beta < max_beta:
            beta = min(beta * beta_factor, max_beta)

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (small moves)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-4,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted permutation simulated‑annealing
# ----------------------------------------------------------------------


def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-5,
                                 top_frac: float = 0.05,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing that swaps an index with a high contribution to
    the current peak with an index with a low contribution.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    n = a.shape[0]
    start_time = time.time()

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        kmax_candidates = np.flatnonzero(conv == max_val)
        kmax = rng.choice(kmax_candidates)

        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)

        if donor == receiver:
            continue

        a[donor], a[receiver] = a[receiver], a[donor]

        new_score = _score(a)

        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12))

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] PermSearch new best={best_score:.6f}")
        else:
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point for the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # tiny safety margin

    # ------------------------------------------------------------------
    # 1. Choose an exploration size (fast to allow many seeds)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n_expl = 4000
    elif budget_s >= 500:
        n_expl = 3000
    else:
        n_expl = 2500
    target_expl = np.sqrt(2.0 * n_expl)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # 2. Build a diverse pool of initial candidates
    # ------------------------------------------------------------------
    init_pool: list[np.ndarray] = []

    # decreasing power‑law
    for p in np.linspace(0.35, 0.75, 7):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))

    # decreasing exponential
    for lam in np.linspace(0.0015, 0.015, 5):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))

    # edge‑heavy (U‑shape) families
    for p in np.linspace(0.1, 0.6, 5):
        init_pool.append(_edge_heavy_sequence(n_expl, float(p), target_expl))

    # symmetric power‑law (mirror the decreasing power law)
    for p in np.linspace(0.2, 0.6, 5):
        half = _power_law_sequence(n_expl // 2 + 1, float(p), target_expl / 2.0)
        sym = np.concatenate([half[:-1], half[::-1]])
        sym = _project_to_simplex(sym, target_expl)
        init_pool.append(sym)

    # random Dirichlet‑like points
    for _ in range(8):
        a = rng.random(n_expl)
        a *= target_expl / a.sum()
        init_pool.append(a)

    # reuse previous best if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_expl))

    # jittered copies for extra diversity
    extra = []
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.01, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        extra.append(seq_j)
    init_pool.extend(extra)

    # ------------------------------------------------------------------
    # 3. Exploration: many short Adam optimiser runs
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    exploration_deadline = start_time + budget_s * 0.55  # about half of the budget
    while time.time() < exploration_deadline:
        # pick a seed
        if best_seq is not None and rng.random() < 0.6:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.004, size=n_expl)
            seed = seed + jitter
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)

        # run a short optimisation (≈12 s)
        run_deadline = min(time.time() + 12.0, exploration_deadline)
        if run_deadline - time.time() < 1.0:
            break

        a_opt, val_opt = _adam_gradient_descent(
            seed,
            run_deadline,
            target_expl,
            max_beta=5e5,
            step_factor=0.6,
            beta_factor=2.0,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best={best_score:.6f}")

    # fallback
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.55, target_expl)
        best_score = _score(best_seq)

    # ------------------------------------------------------------------
    # 4. One short deterministic mass‑transfer refinement (if time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 12.0:
        mt_dead = time.time() + remaining * 0.10
        a_ref, val_ref = _deterministic_mass_transfer(
            best_seq,
            mt_dead,
            target_expl,
            start_temp=0.5,
            end_temp=1e-4,
            low_frac=0.05,
        )
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] After MT refinement best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Upscaling to a larger length
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and budget_s >= 800:
        n_final = 12000
    elif remaining > 20.0:
        n_final = 10000
    else:
        n_final = n_expl

    target_final = np.sqrt(2.0 * n_final)
    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # try a few systematic reorderings
    candidates = [
        seq_big,
        _alternating_order(seq_big),
        _reversed_alternating(seq_big),
        _symmetrize(seq_big),
    ]
    best_seq = seq_big
    best_score = _score(seq_big)
    for cand in candidates[1:]:
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand
    print(f"[{time.time():.2f}] Upscaled length={n_final} start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final polishing – high‑β Adam descent (if time permits)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 12.0:
        a_final, val_final = _adam_gradient_descent(
            best_seq,
            deadline - 2.0,
            target_final,
            max_beta=2_000_000,
            step_factor=0.4,
            beta_factor=2.0,
        )
        if val_final < best_score:
            best_score = val_final
            best_seq = a_final.copy()
            print(f"[{time.time():.2f}] Final polish best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Extended targeted permutation search (uses remaining time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 6.0:
        perm_dead = deadline - 0.5
        a_perm, val_perm = _permutation_targeted_search(
            best_seq,
            perm_dead,
            start_temp=0.5,
            end_temp=1e-5,
            top_frac=0.07,
            low_frac=0.07,
        )
        if val_perm < best_score:
            best_score = val_perm
            best_seq = a_perm.copy()
            print(f"[{time.time():.2f}] After permutation search best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Tiny deterministic mass‑transfer pass (final polishing)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        mt_dead = time.time() + min(5.0, remaining - 0.5)
        a_mt2, val_mt2 = _deterministic_mass_transfer(
            best_seq,
            mt_dead,
            target_final,
            start_temp=0.3,
            end_temp=1e-5,
            low_frac=0.05,
        )
        if val_mt2 < best_score:
            best_score = val_mt2
            best_seq = a_mt2.copy()
            print(f"[{time.time():.2f}] Post‑polish MT best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 9. Trim trailing zeros for compactness
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # 10. Log final score (optional)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for possible later reuse
    globals()["height_sequence_1"] = final_seq
    return final_seq
