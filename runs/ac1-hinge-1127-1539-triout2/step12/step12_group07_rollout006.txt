start_time = time.time()
while time.time() < deadline:
    # propose swap
    i,j = rng.integers(0, len(seq), size=2)
    if i==j: continue
    # create new sequence by swapping values
    new_seq = cur_seq.copy()
    new_seq[i], new_seq[j] = new_seq[j], new_seq[i]
    new_max = _max_convolution_fft(new_seq)
    delta = new_max - cur_max
    # temperature schedule
    elapsed = time.time() - start_time
    tau = elapsed / (deadline - start_time)
    temp = start_temp * (end_temp/start_temp)**tau
    if delta <= 0 or rng.random() < math.exp(-delta/temp):
        cur_seq = new_seq
        cur_max = new_max
        if new_max < best_max:
            best_max = new_max
            best_seq = new_seq.copy()

"""
Hybrid optimizer with LP‑based rebalancing and heavy permutation annealing.

The algorithm proceeds in the following stages (all respecting the
ℓ¹‑normalisation Σa = √(2·n) so that the score coincides with the
maximum self‑convolution value).

1. Small‑scale exploration – a short annealed‑gradient run on a modest
   length (≈2000).  The best seed is kept.
2. Linear‑programming boost – given the current f, solve
   maximise Σg  s.t. f∗g ≤ max(f∗f).  The optimiser g is rescaled to the
   ℓ¹‑norm and mixed with f.  A few iterations dramatically lower the
   peak.
3. Upscaling – the refined small sequence is interpolated to a large
   length (≈15000) and projected onto the simplex.  Deterministic
   spread orderings are tried and the best is kept.
4. Permutation annealing – a long simulated‑annealing loop that swaps
   values to flatten the convolution.  Each swap is evaluated by an
   FFT‑based convolution (O(N log N)).
5. Final sub‑gradient polishing – projected sub‑gradient descent on the
   exact max‑convolution.

The routine respects a time budget (default 1000 s) and logs intermediate
scores.  The final sequence is stored in the global variable
`height_sequence_1` for reuse in subsequent calls.
"""

import time
import math
import numpy as np
from typing import List

# Optional import of scipy for the LP step.
try:
    import scipy.optimize
except Exception:  # pragma: no cover
    scipy = None

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (full linear convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a)/(∑a)² (equals max‑conv when Σa=√(2n))."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = s} in Euclidean norm."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), then renormalised to Σa = target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex (Dirichlet‑like)."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread large values uniformly using a step coprime to the length."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out

# Annealed gradient descent (soft‑max approximation)

def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.9,
                               gamma: float = 0.98) -> tuple[np.ndarray, float]:
    """Momentum projected GD on the soft‑max approximation of max‑conv."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_b = np.max(b)
        exps = np.exp(beta * (b - max_b))
        p = exps / np.sum(exps)

        a_rev = a[::-1]
        Ar_fft = np.fft.rfft(a_rev, n=size)
        Pr_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:7.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent using the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]

        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑programming boost (maximise Σg s.t. f∗g ≤ max(f∗f))
# ----------------------------------------------------------------------


def _lp_step(f: np.ndarray,
             target_sum: float,
             mix_t: float,
             deadline: float) -> tuple[np.ndarray, float]:
    """
    One LP‑based improvement step.
    Solve   max Σg   s.t.    f∗g ≤ M,   g ≥ 0,
    where M = max(f∗f).  The optimiser g is rescaled to the ℓ¹‑norm
    and mixed with f using convex weight `mix_t`.
    """
    if scipy is None:                     # pragma: no cover
        return f, _max_convolution_fft(f)

    n = f.shape[0]
    M = _max_convolution_fft(f)

    # Build dense constraint matrix A where (A g)_k = Σ_i f_i g_{k-i}
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i, fi in enumerate(f):
        A[i:i + n, i] = fi

    c = -np.ones(n)                     # maximise Σg   ↔ minimise -Σg
    b = np.full(rows, M, dtype=np.float64)
    bounds = [(0.0, None) for _ in range(n)]

    # Solve LP
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                                 method='highs', options={'presolve': True})
    if not res.success:
        # If LP fails, fall back to the current f
        return f, M

    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 == 0:
        # degenerate LP – do nothing
        return f, M

    # Rescale to target ℓ¹‑norm
    g = g0 * (target_sum / sum_g0)

    # Convex combination and projection
    new_f = (1.0 - mix_t) * f + mix_t * g
    new_f = _project_to_simplex(new_f, target_sum)

    new_M = _max_convolution_fft(new_f)
    return new_f, new_M


# ----------------------------------------------------------------------
# Permutation annealing (swap‑based simulated annealing)
# ----------------------------------------------------------------------


def _anneal_permutation(seq: np.ndarray,
                        deadline: float,
                        start_temp: float = 1.0,
                        end_temp: float = 1e-4) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing over permutations: repeatedly swap two entries,
    accept improvements and occasionally uphill moves.
    """
    rng = np.random.default_rng()
    cur_seq = seq.copy()
    cur_max = _max_convolution_fft(cur_seq)
    best_seq = cur_seq.copy()
    best_max = cur_max

    start_time = time.time()
    n = cur_seq.shape[0]

    while time.time() < deadline:
        # temperature schedule (exponential decay)
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose swap
        new_seq = cur_seq.copy()
        new_seq[i], new_seq[j] = new_seq[j], new_seq[i]
        new_max = _max_convolution_fft(new_seq)
        delta = new_max - cur_max

        if delta <= 0 or rng.random() < math.exp(-delta / max(temp, 1e-12)):
            # accept
            cur_seq = new_seq
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_seq = new_seq.copy()
                print(f"[{time.time():.2f}] PermAnneal new best={best_max:.6f}")

    return best_seq, best_max

        print(f"[{time.time():.2f}] LP step {lp_iters+1} score={best_small_score:.6f}")
        # a short local refinement after LP
        sub_deadline = min(time.time() + 6.0, deadline - 10.0)
        best_small, sub_score = _subgradient_descent(best_small, sub_deadline,
                                                     target_small, step_factor=0.45,
                                                     gamma=0.995)
        if sub_score < best_small_score:
            best_small_score = sub_score
        lp_iters += 1

    # -------------------- Phase 3: upscale to large problem size --------------------
    remaining = deadline - time.time()
    if remaining > 120.0:
        n_large = 18000
    elif remaining > 70.0:
        n_large = 12000
    elif remaining > 40.0:
        n_large = 8000
    else:
        n_large = n_small

    target_large = math.sqrt(2.0 * n_large)
    seq_big = _resize_sequence(best_small, n_large)
    seq_big = _project_to_simplex(seq_big, target_large)

    # try a few deterministic reorderings and keep the best
    candidates = [
        seq_big,
        _alternating_order(seq_big),
        _even_spread_order(seq_big)
    ]
    seq_big = min(candidates, key=_max_convolution_fft)
    print(f"[{time.time():.2f}] Upscaled to {n_large} items, initial max={_max_convolution_fft(seq_big):.6f}")

    # -------------------- Phase 4: heavy permutation annealing --------------------
    if time.time() < deadline - 10.0:
        perm_deadline = deadline - 8.0
        seq_big, perm_max = _anneal_permutation(seq_big, perm_deadline,
                                                start_temp=1.0, end_temp=1e-5)
        print(f"[{time.time():.2f}] After permutation annealing max={perm_max:.6f}")

    # -------------------- Phase 5: final sub‑gradient polishing --------------------
    if time.time() < deadline - 6.0:
        sub_deadline = deadline - 4.0
        seq_big, final_score = _subgradient_descent(seq_big, sub_deadline,
                                                    target_large,
                                                    step_factor=0.45,
                                                    gamma=0.995)
        print(f"[{time.time():.2f}] Final sub‑grad score={final_score:.6f}")

    # -------------------- Trim trailing near‑zero entries --------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # -------------------- Logging of the final evaluation --------------------
    try:
        val = evaluate_sequence(final_seq)          # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose the sequence for reuse in later calls
    globals()["height_sequence_1"] = final_seq
    return final_seq
