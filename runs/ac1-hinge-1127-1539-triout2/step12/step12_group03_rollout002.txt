"""
Propose a step‑function with a low maximal self‑convolution.

Algorithm outline
-----------------
1. **Rich parametric sweep** – power‑laws with an optional offset,
   optionally multiplied by an exponential tail.  The grid is much
   finer than the previous implementation and includes a small set of
   offset values to break the singularity at the origin.  The best
   candidate of this fast phase is kept.

2. **Projected Adam on a soft‑max surrogate** – several short Adam runs
   (with restarts) on the smooth approximation of the max‑convolution.
   The schedule for the soft‑max temperature β is exponential and the
   step size is scaled by the simplex radius, which yields a very
   stable descent.

3. **Exact projected sub‑gradient descent** – uses the true sub‑gradient
   of the max‑convolution (the “2·a_{k‑i}” formula) together with a
   back‑tracking line search.

4. **Deterministic mass‑transfer (greedy flattening)** – the same idea
   as the reference code but executed repeatedly until the peak cannot
   be lowered any more.

5. **Random local mass‑transfer** – a cheap stochastic hill‑climbing
   phase that swaps small amounts of mass between two indices and
   keeps any improvement.  It helps to escape shallow local minima.

6. **Final high‑β Adam polish** – a last Adam run with a very large
   soft‑max temperature (β up to 1e6) to squeeze out the remaining
   slack.

All phases respect a global time budget (default 1000 s) and work on
the simplex {a_i ≥ 0, Σa_i = √(2 n)}.  The returned list is trimmed of
trailing near‑zero entries.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Simple scoring (max‑convolution value)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Return the evaluation score (max‑convolution entry)."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for the fast pre‑search
# ----------------------------------------------------------------------


def _power_law_offset_sequence(
    n: int, p: float, offset: float, target_sum: float
) -> np.ndarray:
    """a_i = (i + offset)^(-p)   (i starts at 0) and then normalized."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    a *= target_sum / a.sum()
    return a


def _exp_power_law_offset_sequence(
    n: int, p: float, q: float, offset: float, target_sum: float
) -> np.ndarray:
    """a_i = (i+offset)^(-p) * exp(-q*i/n)   and then normalized."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    """A smooth sine‑shaped candidate (always positive)."""
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max based gradient (used by Adam)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 5e5,
    step_factor: float = 0.6,
    gamma: float = 0.995,
) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max surrogate."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 250

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 400 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _adam_restarts(
    init: np.ndarray,
    target_sum: float,
    deadline: float,
    max_beta: float = 5e5,
) -> tuple[np.ndarray, float]:
    """Multiple short Adam runs with jittered starts."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 4.0:
        remaining = deadline - time.time()
        run_budget = min(25.0, remaining * 0.3)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.015, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(
            start,
            time.time() + run_budget,
            target_sum,
            max_beta=max_beta,
            step_factor=0.6,
            gamma=0.995,
        )
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------


def _subgradient_descent(
    a0: np.ndarray, target_sum: float, deadline: float, max_iter: int = 5000
) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on the true max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng()
    it = 0
    n = a.shape[0]

    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.flatnonzero(conv >= max_val - 1e-12)

        # second‑largest value (ignore all current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()
        delta_target = max_val - second_max
        if delta_target < 1e-14:
            break

        k = idx_max[rng.integers(len(idx_max))]

        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k - idx]

        norm_g2 = float(np.dot(g, g))
        if norm_g2 == 0.0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = _project_to_simplex(a + jitter, target_sum)
            it += 1
            continue

        step = 0.9 * delta_target / norm_g2
        step = min(step, target_sum * 0.4 / (np.linalg.norm(g) + 1e-12))

        for _ in range(6):
            a_candidate = _project_to_simplex(a - step * g, target_sum)
            cand_score = _score(a_candidate)
            if cand_score < best_score - 1e-12:
                a = a_candidate
                best_a = a_candidate.copy()
                best_score = cand_score
                print(
                    f"[{time.time():.2f}] Subgrad it={it:5d} step={step:.3e} score={best_score:.6f}"
                )
                break
            step *= 0.5
        else:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = _project_to_simplex(a + jitter, target_sum)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (greedy flattening)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(
    a0: np.ndarray, deadline: float, target_sum: float, tol: float = 1e-12
) -> tuple[np.ndarray, float]:
    """Greedy reduction of the maximal convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest (ignore all maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()
        if max_val - second_max < tol:
            break

        k = idx_max[0]  # pick the first maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]  # a_{k‑i}
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        # pick donor that contributes most to the peak
        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # pick receiver with the smallest partner value (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] DMT it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score

def _random_local_hill_climb(
    a0: np.ndarray, target_sum: float, deadline: float, max_trials: int = 3000
) -> tuple[np.ndarray, float]:
    """Swap small random amounts of mass between two indices."""
    a = a0.copy()
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    n = a.shape[0]
    trial = 0

    while time.time() < deadline and trial < max_trials:
        # pick donor proportional to current mass (more mass → more chance)
        donor = rng.choice(n, p=a / target_sum)
        if a[donor] <= 0.0:
            trial += 1
            continue
        receiver = rng.integers(n)
        if receiver == donor:
            trial += 1
            continue

        # move a random fraction (up to 20%) of donor's mass
        delta = rng.random() * a[donor] * 0.20
        a_candidate = a.copy()
        a_candidate[donor] -= delta
        a_candidate[receiver] += delta
        a_candidate = _project_to_simplex(a_candidate, target_sum)

        val = _score(a_candidate)
        if val < best_val:
            best_val = val
            best_a = a_candidate.copy()
            a = a_candidate  # continue from the improvement
            print(f"[{time.time():.2f}] RandomHill new best={best_val:.6f}")

        trial += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:

    """

    if budget_s >= 850:
        n = 16000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)  # Σa is fixed to this value

    # ------------------------------------------------------------------
    # Phase 0 – enriched parametric sweep
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    # grids
    p_grid = np.linspace(0.30, 0.90, 31)          # exponent
    q_grid = np.linspace(0.0, 6.0, 21)           # exponential tail strength
    offset_grid = np.array([0.0, 0.1, 0.5, 1.0, 2.0, 5.0, 10.0])

    # pure power‑law with offset
    for offset in offset_grid:
        for p in p_grid:
            seq = _power_law_offset_sequence(n, float(p), float(offset), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()
        # early‑exit if we run out of time
        if time.time() >= deadline - 5.0:
            break

    # power‑law + exponential tail with offset
    if time.time() < deadline - 5.0:
        for offset in offset_grid:
            for p in p_grid:
                for q in q_grid:
                    seq = _exp_power_law_offset_sequence(
                        n, float(p), float(q), float(offset), target_sum
                    )
                    val = _score(seq)
                    if val < best_score:
                        best_score = val
                        best_seq = seq.copy()
            if time.time() >= deadline - 5.0:
                break

    # sine shape
    if time.time() < deadline - 5.0:
        seq = _sin_sequence(n, target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # previous global best (if any)
    if time.time() < deadline - 5.0 and "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score = val
            best_seq = resized.copy()

    # small jittered population around current best
    if time.time() < deadline - 5.0:
        for _ in range(6):
            jitter = rng.normal(scale=target_sum * 0.004, size=n)
            cand = best_seq + jitter
            cand = np.maximum(cand, 0.0)
            cand = _project_to_simplex(cand, target_sum)
            val = _score(cand)
            if val < best_score:
                best_score = val
                best_seq = cand.copy()

    print(f"[{time.time():.2f}] After parametric sweep best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam with restarts (soft‑max surrogate)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        gd_deadline = time.time() + (deadline - time.time()) * 0.30
        a_adam, val_adam = _adam_restarts(best_seq, target_sum, gd_deadline, max_beta=5e5)
        if val_adam < best_score:
            best_score = val_adam
            best_seq = a_adam.copy()
            print(f"[{time.time():.2f}] Adam restarts improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Exact projected sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sg_deadline = time.time() + (deadline - time.time()) * 0.22
        a_sg, val_sg = _subgradient_descent(best_seq, target_sum, sg_deadline)
        if val_sg < best_score:
            best_score = val_sg
            best_seq = a_sg.copy()
            print(f"[{time.time():.2f}] Sub‑gradient descent improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase C – Deterministic greedy mass‑transfer
    # ------------------------------------------------------------------
    while time.time() < deadline - 8.0:
        dmt_deadline = time.time() + 3.0  # short bursts
        a_dmt, val_dmt = _deterministic_mass_transfer(best_seq, dmt_deadline, target_sum)
        if val_dmt < best_score:
            best_score = val_dmt
            best_seq = a_dmt.copy()
            print(f"[{time.time():.2f}] DMT improved best={best_score:.6f}")

        # ----------------------------------------------------------------
        # Short random global transfer to escape possible plateaus
        # ----------------------------------------------------------------
        rand_deadline = time.time() + 2.0
        a_rand, val_rand = _random_local_hill_climb(best_seq, target_sum, rand_deadline)
        if val_rand < best_score:
            best_score = val_rand
            best_seq = a_rand.copy()
            print(f"[{time.time():.2f}] Random hill climb improved best={best_score:.6f}")

        if time.time() > deadline - 2.0:
            break

    # ------------------------------------------------------------------
    # Phase D – Random local hill climbing (longer)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        a_rl, val_rl = _random_local_hill_climb(best_seq, target_sum, deadline)
        if val_rl < best_score:
            best_score = val_rl
            best_seq = a_rl.copy()
            print(f"[{time.time():.2f}] Final random hill climb best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase E – Final high‑β Adam polish
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(
            best_seq,
            deadline,
            target_sum,
            max_beta=1e6,
            step_factor=0.5,
            gamma=0.990,
        )
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final evaluation (harmless if evaluate_sequence not defined)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
