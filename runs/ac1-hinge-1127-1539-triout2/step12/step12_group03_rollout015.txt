"""
Light‑weight search for a step‑function with a low maximal self‑convolution.

The routine works in four stages:

1. **Parametric sweep** – generate sequences of the form
   a_i ∝ (i+1+δ)^{-p}·exp(-q·i/n) and keep the best one.
2. **Adam‑style projected gradient descent** – minimise a soft‑max
   approximation of the max‑convolution; several restarts are performed.
3. **Deterministic mass‑transfer** – greedy reduction of the current peak.
4. **Final polishing** – one more Adam run with a very large soft‑max β.

All steps respect the time budget (default 1000 s). The returned list
is already normalised (∑a_i = √(2 n)) so that the competition evaluation
coincides with the maximal convolution entry.
"""

from __future__ import annotations
import time
import numpy as np
import math

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) using a power‑of‑two FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum value of the full linear convolution a*a."""
    return float(_conv_full_fft(a).max())


# ----------------------------------------------------------------------
# Simple objective (max‑convolution) – the evaluation function equals this
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Score = maximal self‑convolution entry (the quantity to minimise)."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric family: shifted power‑law with an exponential tail
# ----------------------------------------------------------------------


def _shifted_power_exp_sequence(
    n: int, p: float, q: float, delta: float, target_sum: float
) -> np.ndarray:
    """
    a_i ∝ (i+1+δ)^{-p}·exp(-q·i/n)  (δ = delta).
    The vector is finally rescaled to have Σa_i = target_sum.
    """
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0 + delta) ** (-p)
    if q != 0.0:
        a *= np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(a*a)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_val = conv.max()
    exps = np.exp(beta * (conv - max_val))
    p = exps / exps.sum()

    a_rev = a[::-1]
    rev_fft = np.fft.rfft(a_rev, n=size)
    p_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(rev_fft * p_fft, n=size)[: 2 * n - 1]

    # gradient corresponds to the central n entries of the reversed convolution
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, conv


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 2e5,
    step_factor: float = 0.8,
    gamma: float = 0.999,
) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max surrogate of max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-12
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1.0 - gamma) * g
        v = gamma * v + (1.0 - gamma) * (g * g)

        m_hat = m / (1.0 - gamma ** (it + 1))
        v_hat = v / (1.0 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow plateaus
        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}"
            )

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------


def _subgradient_descent(
    a0: np.ndarray, target_sum: float, deadline: float, max_iter: int = 5000
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    it = 0
    n = a.size

    while time.time() < deadline and it < max_iter:
        conv = _conv_full_fft(a)
        max_val = conv.max()
        idx_max = np.flatnonzero(conv >= max_val - 1e-12)

        # second‑largest value (ignore all current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()
        delta_target = max_val - second_max

        if delta_target < 1e-14:
            break

        # random maximal index
        k = idx_max[rng.integers(len(idx_max))]

        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k - idx]

        norm2 = np.dot(g, g)
        if norm2 == 0.0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = _project_to_simplex(a + jitter, target_sum)
            it += 1
            continue

        step = 0.9 * delta_target / norm2
        step = min(step, target_sum * 0.4 / (np.linalg.norm(g) + 1e-12))

        # backtracking line search
        for _ in range(6):
            cand = _project_to_simplex(a - step * g, target_sum)
            cand_val = _score(cand)
            if cand_val < best_val - 1e-12:
                a = cand
                best_a = cand.copy()
                best_val = cand_val
                print(
                    f"[{time.time():.2f}] Subgrad it={it:5d} step={step:.3e} score={best_val:.6f}"
                )
                break
            step *= 0.5
        else:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = _project_to_simplex(a + jitter, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (greedy peak reduction)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(
    a0: np.ndarray, deadline: float, target_sum: float, tol: float = 1e-12
) -> tuple[np.ndarray, float]:
    """Greedy reduction of the maximal convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_full_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest value (ignore all maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # pick one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]

        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # receiver – smallest partner value (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur = _score(a)
        if cur < best_val - 1e-12:
            best_val = cur
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_val:.6f}"
            )
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Global random mass‑transfer (exploratory)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(
    a0: np.ndarray, deadline: float, target_sum: float
) -> tuple[np.ndarray, float]:
    """Very cheap stochastic mass‑transfer that occasionally finds improvements."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_val = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur = _score(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random new best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal maximal self‑convolution.
    Returns the best sequence found (as a plain Python list of floats).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    # ------------------------------------------------------------------
    # Problem size – larger n gives more flexibility but a little slower.
    # Empirically n≈20000 is a good compromise for a 1000 s budget.
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 400:
        n = 14000
    else:
        n = 10000
    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Phase 0 – coarse parametric sweep (shifted power‑law + exp tail)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    p_grid = np.linspace(0.40, 0.80, 11)          # 0.40 … 0.80
    q_grid = np.linspace(0.0, 8.0, 17)           # 0 … 8
    delta_grid = np.linspace(0.0, 4.0, 9)        # shift δ

    for p in p_grid:
        for q in q_grid:
            for delta in delta_grid:
                seq = _shifted_power_exp_sequence(
                    n, float(p), float(q), float(delta), target_sum
                )
                val = _score(seq)
                if val < best_score:
                    best_score = val
                    best_seq = seq.copy()
    print(f"[{time.time():.2f}] After param sweep best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam‑style SGD with multiple restarts
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    phase_a_deadline = start_time + 0.55 * budget_s  # ~55 % of total time
    while time.time() < phase_a_deadline - 5.0:
        # jittered start
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _adam_gradient_descent(
            start_vec, phase_a_deadline, target_sum, max_beta=3e5, step_factor=0.9
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time():.2f}] Adam restart improved best = {best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Phase B – exact deterministic mass‑transfer (aggressive peak flattening)
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.80 * budget_s
    while time.time() < phase_b_deadline - 5.0:
        a_ref, val_ref = _deterministic_mass_transfer_exact2(
            best_seq, phase_b_deadline, target_sum
        )
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(
                f"[{time.time():.2f}] Deterministic mass‑transfer improved best = {best_score:.6f}"
            )

        # occasional random moves to hop out of shallow plateaus
        a_rand, val_rand = _global_random_mass_transfer(
            best_seq, phase_b_deadline, target_sum
        )
        if val_rand < best_score:
            best_score = val_rand
            best_seq = a_rand.copy()
            print(
                f"[{time.time():.2f}] Global random step improved best = {best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Phase C – optional sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    phase_c_deadline = start_time + 0.90 * budget_s
    if time.time() < phase_c_deadline - 5.0:
        a_sg, val_sg = _subgradient_descent(
            best_seq, target_sum, phase_c_deadline
        )
        if val_sg < best_score:
            best_score = val_sg
            best_seq = a_sg.copy()
            print(
                f"[{time.time():.2f}] Sub‑gradient descent improved best = {best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Phase D – final high‑β polishing
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(
            best_seq,
            deadline,
            target_sum,
            max_beta=1e6,
            step_factor=0.6,
            gamma=0.995,
        )
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(
                f"[{time.time():.2f}] Final polish best = {best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries and return the final candidate
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final evaluation for logging (harmless if evaluate_sequence missing)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
