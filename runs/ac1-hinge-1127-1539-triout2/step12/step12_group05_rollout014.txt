"""
Improved step‑function optimizer for the benchmark.

Main ideas
----------

* **Problem size** – Use a larger n when the time budget permits (up to 20000
  entries).  The scoring formula is scale‑invariant, therefore a finer grid
  gives the optimizer more freedom without changing the underlying objective.

* **Diverse parametric pre‑search** – Randomly sample several classic families
  (pure power laws, exponential‑power‑law, broken power‑law and a shifted
  power‑law).  The best outcome of this cheap scan is used as a warm‑start.

* **Projected Adam on a soft‑max approximation** – A standard Adam optimiser
  is applied to the smooth surrogate
        softmaxβ(conv) = log‑sum‑exp(β⋅conv) / β.
  The gradient is obtained analytically via the FFT.  β is increased
  exponentially, the step‑size follows an Adam schedule and we inject a
  small Gaussian jitter every few hundred iterations to shake the iterate
  out of plateaus.

* **Exact sub‑gradient guided mass‑transfer (enhanced)** – Using the exact
  sub‑gradient of the current maximal convolution entry we move mass from
  “dangerous’’ positions (large gradient) toward “safe’’ ones (small
  gradient).  A line‑search drives the peak down towards the second‑largest
  entry, but when the direction ceases to reduce the peak we perform a tiny
  random perturbation and continue.  This loop runs until the time budget
  expires.

* **Final high‑β Adam polish** – After the mass‑transfer phase we run a
  short Adam with a very large β (≈10⁶) to fine‑tune the remaining
  irregularities.

All phases respect the simplex constraint Σa = √(2 n) by Euclidean projection.
The routine stops exactly at the supplied time limit and returns the best
sequence discovered (trailing near‑zero entries are removed).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft_pair(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution x*y (full length 2·n‑1) using a real FFT."""
    n = x.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Exact benchmark score (identical to evaluate_sequence for Σa = √2n)."""
    n = a.shape[0]
    max_b = _max_conv(a)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (dense pre‑search)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _broken_power_law_sequence(
    n: int, p1: float, p2: float, alpha: float, target_sum: float
) -> np.ndarray:
    K = max(1, int(alpha * n))
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.empty_like(idx)
    a[:K] = idx[:K] ** (-p1)
    a[K:] = idx[K:] ** (-p2)
    a *= target_sum / a.sum()
    return a


def _offset_power_law_sequence(
    n: int, p: float, offset: float, target_sum: float
) -> np.ndarray:
    """(i+offset)^(-p)   with i counting from 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (idx + offset) ** (-p)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max approximation)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(a*a)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = conv_full.max()
    # soft‑max weights (numerically stable)
    exps = np.exp(beta * (conv_full - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, conv_full


def _adam_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    *,
    max_beta: float = 5e5,
    step_factor: float = 0.8,
    gamma: float = 0.999,
    beta_start: float = 0.5,
    beta_inc_iter: int = 250,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the soft‑max approximation of the max‑convolution.
    Returns the best solution seen before ``deadline``.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = beta_start
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if it % 200 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} best={best_val:.6f}")

        # raise beta gradually (more focus on the true max)
        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient for the maximal convolution entry
# ----------------------------------------------------------------------


def _compute_gradient_for_max(a: np.ndarray, max_idx: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a (averaged over all maximal
    indices).  For a given index k the contribution to the gradient at i is
    2·a_{k‑i} whenever i participates in the convolution for k.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in max_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g[idx] += 2.0 * a[k - idx]
    if max_idx.size > 0:
        g /= max_idx.size
    return g


# ----------------------------------------------------------------------
# Guided exact mass‑transfer (enhanced)
# ----------------------------------------------------------------------


def _guided_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    *,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the maximal convolution entry.
    At each iteration we move mass from gradient‑large to gradient‑small
    positions using an exact line‑search.
    If the current direction does not decrease the peak we apply a tiny
    random perturbation and try again.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()

        # second largest distinct value (if any)
        mask_max = conv >= max_val - tol
        if np.any(~mask_max):
            second_max = conv[~mask_max].max()
        else:
            second_max = 0.0

        max_idx = np.where(mask_max)[0]

        # exact sub‑gradient (averaged)
        g = _compute_gradient_for_max(a, max_idx)

        # direction that preserves Σa = 0
        sum_g = g.sum()
        d = -g + sum_g / n * np.ones_like(a)  # Σd = 0

        # maximum feasible step before any entry becomes negative
        neg = d < 0
        if np.any(neg):
            t_nonneg = np.min(a[neg] / -d[neg])
        else:
            t_nonneg = np.inf

        # linearised change of the convolution in direction d
        delta = 2.0 * _conv_fft_pair(a, d)  # length 2n‑1
        delta_max = delta[max_idx]
        delta_k = delta_max.min()  # most favourable peak

        if delta_k >= -1e-14:  # direction does not lower the peak
            # tiny random jiggle and continue
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)
            continue

        # step required to bring the current peak down to the second‑largest
        if max_val - second_max < 1e-15:
            t_to_second = np.inf
        else:
            t_to_second = (max_val - second_max) / (-delta_k)

        # safety factor (stay a little away from the boundary)
        t = min(t_nonneg, t_to_second) * 0.995
        if t <= 0:
            break

        a += t * d
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] GuidedMT it={it:5d} score={best_score:.6f}")

        # occasional random jitter to hop out of tiny plateaus
        if it % 400 == 0 and rng.random() < 0.03:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_score

def _global_random_move(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
) -> tuple[np.ndarray, float]:
    """Very cheap simulated‑annealing style mass moves."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    while time.time() < deadline - 0.5:
        # pick a donor among the largest 2 % and a receiver among the smallest 2 %
        top_k = max(1, int(0.02 * n))
        low_k = max(1, int(0.02 * n))
        donor = np.argsort(-a)[rng.integers(top_k)]
        receiver = np.argsort(a)[rng.integers(low_k)]
        if donor == receiver:

        delta = rng.random() * a[donor] * 0.07  # a slightly larger move than before
        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] RandomMove new best={best_score:.6f}")

    return best_a, best_score

                adam_deadline,
                target_sum,
                max_beta=8e5,
                step_factor=0.7,
                gamma=0.995,
                beta_start=1.0,
                beta_inc_iter=180,
            )
            if val_opt < best_score:
                best_score, best_seq = val_opt, a_opt.copy()
                print(f"[{time.time():.2f}] Adam restart improved best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase B – enhanced guided mass‑transfer
    # ---------------------------------------------------------------
    while time.time() < deadline - 10.0:
        a_ref, val_ref = _guided_mass_transfer(best_seq, deadline, target_sum)
        if val_ref < best_score:
            best_score, best_seq = val_ref, a_ref.copy()
            print(f"[{time.time():.2f}] GuidedMT improved best = {best_score:.6f}")

        # occasional lightweight random moves
        a_rand, val_rand = _global_random_move(best_seq, deadline, target_sum)
        if val_rand < best_score:
            best_score, best_seq = val_rand, a_rand.copy()
            print(f"[{time.time():.2f}] Random move improved best = {best_score:.6f}")

        # break early if we are within the last couple of seconds
        if time.time() > deadline - 4.0:
            break

    # ---------------------------------------------------------------
    # Phase C – final high‑β Adam polish
    # ---------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(
            best_seq,
            deadline,
            target_sum,
            max_beta=1e6,
            step_factor=0.55,
            gamma=0.997,
            beta_start=2.0,
            beta_inc_iter=150,
        )
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final polish improved best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return list
    # ---------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final evaluation (harmless if evaluate_sequence missing)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated benchmark score = {val:.6f}")
    except Exception:
        pass

    return final_seq
