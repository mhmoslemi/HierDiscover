def _solve_lp_max_sum_under_convolution(f, M, target, rng):
    n = len(f)
    b = cp.Variable(n)
    # Build convolution matrix T
    T = np.zeros((2*n-1, n), dtype=np.float64)
    for i in range(n):
        T[i:i+n, i] = f
    constraints = [b >= 0, T @ b <= M]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    # Set solver parameters, e.g., warm start if needed
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    if b.value is None:
        # fallback: return f
        return f.copy()
    return np.array(b.value, dtype=np.float64)

def _swap_and_transfer_sa(a0, deadline, target_sum, start_temp=0.3, end_temp=1e-5, rng=None):

        rng = np.random.default_rng()
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    conv = _conv_fft(a)
    n = len(a)
    start_time = time.time()
    while time.time() < deadline:
        # Temperature schedule
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # decide move type: swap or mass transfer
        if rng.random() < 0.7:
            # swap two indices
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # perform swap
            a[i], a[j] = a[j], a[i]
            # recompute conv
            conv = _conv_fft(a)
            new_score = _score(a)
        else:
            # mass transfer similar to earlier but with constraint projection
            # pick donor and receiver based on contributions to current peak
            kmax = int(np.argmax(conv))
            lo = max(0, kmax - (n - 1))
            hi = min(kmax, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            prod = a[idx] * a[kmax - idx]
            donor = idx[np.argmax(prod)]
            low_cnt = max(1, int(len(idx) * 0.06))
            low_candidates = idx[np.argsort(prod)[:low_cnt]]
            receiver = rng.choice(low_candidates)
            if donor == receiver:
                continue
            # propose delta
            max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
            delta = rng.random() * max_delta
            if delta <= 0:
                continue
            a[donor] -= delta
            a[receiver] += delta
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            new_score = _score(a)

"""
Hybrid optimizer for the step‑function benchmark.

The pipeline consists of five stages:

1. **Exploratory LP refinement on a modest size (≈1500).**
   For a fixed ℓ¹‑norm we solve a linear program that maximises the total
   mass while keeping the convolution with the current iterate below the
   current peak value.  A convex combination of the old iterate and the
   LP solution (projected onto the simplex) reduces the maximal
   self‑convolution.

2. **Upscaling.**
   The best low‑dimensional shape is linearly interpolated to a much
   larger length (≈12‑18 k).  Several deterministic reorderings that spread
   large entries uniformly are tried and the one with the smallest peak
   is kept.

3. **Projected sub‑gradient descent** on the full‑size problem – a fast
   exact sub‑gradient step for the max‑convolution.

4. **Greedy mass‑transfer with simulated‑annealing acceptance** – the
   same idea as before but limited to the remaining time budget.

5. **Swap‑based simulated annealing.**
   Random swaps of two positions (plus occasional mass moves) are accepted
   with a Metropolis rule; the temperature decays exponentially.
   Swaps preserve the ℓ¹‑norm and are cheap to evaluate using FFT.

All stages respect non‑negativity and the ℓ¹‑simplex constraint.
The final sequence is trimmed of trailing zeros and stored in the
global variable ``height_sequence_1`` for warm‑starting later runs.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Optional cvxpy import (used for the small‑scale LP stage)
# ----------------------------------------------------------------------
try:
    import cvxpy as cp

    _CVX_AVAILABLE = True
except Exception:  # pragma: no cover
    _CVX_AVAILABLE = False

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (linear convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a)/(∑a)² (for ∑a=√(2n) this equals max conv)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    if new_len == seq.shape[0]:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, seq.shape[0])
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Place the sorted values on a cyclic step (≈√n)."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution (used later)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]

        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy mass‑transfer with simulated‑annealing acceptance (used later)
# ----------------------------------------------------------------------


def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.5,
                                      end_temp: float = 1e-4,
                                      low_frac: float = 0.06) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        # propose move
        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        a_new = _project_to_simplex(a_new, target_sum)
        conv_new = _conv_fft(a_new)
        new_score = _score(a_new)

        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            a = a_new
            conv = conv_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")

        # very occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# LP stage – maximise total mass under a convolution bound
# ----------------------------------------------------------------------


def _solve_lp_max_sum_under_convolution(f: np.ndarray,
                                        M: float,
                                        target_sum: float) -> np.ndarray:
    """
    Solve: maximise Σ b_j  subject to  b≥0,  (f * b) ≤ M  (elementwise).
    Returns the optimal b (may have sum > target_sum).
    """
    if not _CVX_AVAILABLE:  # pragma: no cover
        # Fallback – return the current vector unchanged
        return f.copy()

    n = f.shape[0]

    # Build the convolution matrix T where (T @ b)[k] = Σ_i f_i * b_{k-i}
    T = np.zeros((2 * n - 1, n), dtype=np.float64)
    for i in range(n):
        T[i : i + n, i] = f

    b = cp.Variable(n, nonneg=True)
    constraints = [T @ b <= M]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:  # pragma: no cover
        # In case the solver crashes, fall back to the input.
        return f.copy()

    if b.value is None:
        return f.copy()
    return np.array(b.value, dtype=np.float64)

def _swap_and_transfer_sa(a0: np.ndarray,
                          deadline: float,
                          target_sum: float,
                          start_temp: float = 0.3,
                          end_temp: float = 1e-5,
                          rng: np.random.Generator = None) -> tuple[np.ndarray, float]:
    """
    Random swaps of two positions (plus occasional mass transfer).
    Acceptance follows a temperature‑controlled Metropolis rule.
    """
    if rng is None:
        rng = np.random.default_rng()

    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    conv = _conv_fft(a)

    start_time = time.time()

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        a_old = a.copy()
        conv_old = conv.copy()
        move_type = rng.random()

        if move_type < 0.65:
            # ---- simple swap -------------------------------------------------
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            a[i], a[j] = a[j], a[i]
            conv = _conv_fft(a)
            new_score = _score(a)
        else:
            # ---- mass transfer (same code as in MT) -------------------------
            kmax = int(np.argmax(conv))
            lo = max(0, kmax - (n - 1))
            hi = min(kmax, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            prod = a[idx] * a[kmax - idx]
            donor = idx[np.argmax(prod)]
            low_cnt = max(1, int(len(idx) * 0.06))
            low_candidates = idx[np.argsort(prod)[:low_cnt]]
            receiver = rng.choice(low_candidates)

            if donor != receiver:
                max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
                delta = rng.random() * max_delta
                a[donor] -= delta
                a[receiver] += delta
                a = _project_to_simplex(a, target_sum)
                conv = _conv_fft(a)
                new_score = _score(a)
            else:
                continue

        # ---- Metropolis acceptance -----------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] Swap‑SA new best={best_score:.6f}")
        else:
            # revert to previous state
            a = a_old
            conv = conv_old
            # cur_score stays unchanged

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine follows a mixed LP / up‑scaling / local‑search schedule
    and respects the `budget_s` time limit.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin
    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # Phase 1 – small‑scale LP driven shape optimisation
    # --------------------------------------------------------------
    n_small = 1500 if budget_s > 400 else 1000
    target_small = math.sqrt(2.0 * n_small)

    # initialise a pool of diverse seeds
    seed_pool = []

    for p in np.linspace(0.2, 1.8, 9):
        seed_pool.append(_power_law_sequence(n_small, float(p), target_small))

    for lam in np.linspace(0.0005, 0.025, 8):
        seed_pool.append(_exponential_sequence(n_small, float(lam), target_small))

    for _ in range(8):
        seed_pool.append(_random_simplex(n_small, target_small, rng))

    # warm‑start from previously discovered sequence if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        seed_pool.append(_resize_sequence(prev, n_small))

    best_small = None
    best_small_score = np.inf

    # run a handful of independent LP‑based refinements while time allows
    while time.time() < deadline - 30.0:
        seed = rng.choice(seed_pool)
        a = _project_to_simplex(seed.copy(), target_small)

        # number of LP iterations per run – a trade‑off between quality and speed
        lp_iters = 5
        for it in range(lp_iters):
            M = _max_convolution_fft(a)
            g = _solve_lp_max_sum_under_convolution(a, M, target_small)
            # scale g back to the simplex size
            s = g.sum()
            if s > 0:
                g = g * (target_small / s)
            # convex combination (tuned mixing factor)
            t = 0.3
            a = (1.0 - t) * a + t * g
            a = _project_to_simplex(a, target_small)

        cur_score = _score(a)
        if cur_score < best_small_score:
            best_small_score = cur_score
            best_small = a.copy()

    if best_small is None:

    # --------------------------------------------------------------

    remaining = deadline - time.time()
    if remaining > 120:

    elif remaining > 80:
        n_final = 15000
    elif remaining > 40:
        n_final = 12000
    else:

    target_final = math.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_small, n_final)

    # try a few deterministic reorderings that tend to spread the mass

        _alternating_order(seq_big),
        _even_spread_order(seq_big),
    ]
    seq_big = min(candidates, key=_max_convolution_fft)
    print(f"[{time.time():.2f}] Upscaled to {n_final} entries, init max={_max_convolution_fft(seq_big):.6f}")

    # --------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent on the full size
    # --------------------------------------------------------------
    if time.time() < deadline - 6.0:
        sub_deadline = deadline - 4.0
        seq_big, sub_val = _subgradient_descent(
            seq_big, sub_deadline, target_final, step_factor=0.45, gamma=0.995
        )
        print(f"[{time.time():.2f}] After sub‑grad best score={sub_val:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – greedy mass‑transfer (simulated annealing)
    # --------------------------------------------------------------
    if time.time() < deadline - 4.0:
        mt_deadline = time.time() + min(5.0, deadline - time.time() - 1.0)
        seq_big, mt_val = _mass_transfer_simulated_annealing(
            seq_big, mt_deadline, target_final, start_temp=0.3, end_temp=1e-5, low_frac=0.06
        )
        print(f"[{time.time():.2f}] After mass‑transfer best={mt_val:.6f}")

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        sa_deadline = deadline - 0.5
        seq_big, sa_val = _swap_and_transfer_sa(
            seq_big, sa_deadline, target_final, start_temp=0.25, end_temp=1e-5, rng=rng
        )
        print(f"[{time.time():.2f}] Final swap‑SA best={sa_val:.6f}")

    except Exception:
        pass

    return final_seq
