assistantfinal**Solution Overview**

The goal is to minimise  

\[
\text{score}= \frac{2\,n\;\max\limits_k (f*f)_k}{(\sum_i f_i)^2}
\]

for a non‑negative step‑function `f`.  
If we enforce  

\[
\sum_i f_i = \sqrt{2n},
\]

the denominator equals `2 n` and the score is simply the **maximum entry of the auto‑convolution** of `f`.  
Hence the whole task reduces to:  

*choose a non‑negative vector on the simplex of fixed L¹‑norm whose auto‑convolution is as flat as possible.*

The optimiser below follows a **multi‑scale stochastic descent**:

1. **Diverse seeds** – power‑law, exponential, Dirichlet, linear, and the previous best.
2. **Fast sub‑gradient descent** on a moderate size (`n≈5000`).  
   The exact sub‑gradient of the peak entry of the convolution is known in closed form; we use a momentum update and a decreasing learning‑rate schedule.
3. **Upscaling** – the best seed is linearly interpolated to a larger dimension (`n≈20000`).  Larger vectors give more degrees of freedom and the theoretical limit of the score is `1.5`.
4. **Long sub‑gradient refinement** on the large vector (major part of the time budget).
5. **Randomised continuous annealing** – small zero‑mean Gaussian perturbations are added, the vector is projected back onto the simplex and accepted with a Metropolis rule.  This explores directions that the pure sub‑gradient cannot see.
6. **Targeted permutation search** – the ordering of the heights strongly influences the central peak.  Swaps that move mass away from the most contributing indices are performed using simulated annealing.
7. **Deterministic mass‑transfer polishing** – a greedy “donor → receiver’’ move with an exponential temperature schedule.
8. **Trimming** – trailing zeros are removed for compactness and the final score is printed.

All steps respect the 1000 s time limit; each phase stops early if the remaining budget is insufficient.

```python
"""
Hybrid optimiser for the non‑negative step‑function problem.

The optimiser works in several phases:
1. create a diverse pool of seeds (power‑law, exponential, Dirichlet …)
2. short sub‑gradient descent on a medium size (≈5k)
3. upscale the best seed to a large size (≈20k)
4. long sub‑gradient refinement (major part of the budget)
5. random continuous annealing (Gaussian jitter + simplex projection)
6. permutation search (swap indices that heavily contribute to the peak)
7. deterministic mass‑transfer polishing
8. trim trailing zeros and return the best sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, length 2·n‑1)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the auto‑convolution of a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators (used for the seed pool)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i = (i+1)^(-p) and renormalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i = exp(-lam*i) and renormalise to target_sum."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _linear_sequence(n: int, target_sum: float) -> np.ndarray:
    """Linearly decreasing from 1 to 0, then renormalised."""
    a = np.linspace(1.0, 0.0, n, dtype=np.float64)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → left, second‐largest → right, third → next left, …."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    l, r = 0, n - 1
    left = True
    for v in sorted_vals:
        if left:
            out[l] = v
            l += 1
        else:
            out[r] = v
            r -= 1
        left = not left
    return out


# ----------------------------------------------------------------------
# Scoring utilities
# ----------------------------------------------------------------------


def _score(a: np.ndarray, target_sum: float) -> float:
    """
    Returns the evaluation score (2·n·max(conv)/target_sum²).
    For the normalisation target_sum = sqrt(2·n) this equals max(conv).
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    if target_sum < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (target_sum * target_sum))


# ----------------------------------------------------------------------
# Exact max‑convolution and its sub‑gradient
# ----------------------------------------------------------------------


def _max_conv_and_subgrad(a: np.ndarray):
    """
    Returns (conv, max_val, kmax, grad) where grad is a sub‑gradient of the
    peak entry of the convolution w.r.t. a (i.e. ∂ max(conv)/∂ a).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    kmax = int(np.argmax(conv))
    max_val = conv[kmax]

    grad = np.zeros_like(a)
    lo = max(0, kmax - (n - 1))
    hi = min(kmax, n - 1)
    idx = np.arange(lo, hi + 1)
    partner = a[kmax - idx]
    grad[lo:hi + 1] = 2.0 * partner
    return conv, max_val, kmax, grad


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         base_step: float = 0.6,
                         gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """
    Momentum projected sub‑gradient descent on the exact peak of the convolution.
    Returns the best vector seen and its (lower) score.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    n = a.shape[0]
    # initialise best
    _, max_val, _, _ = _max_conv_and_subgrad(a)
    best_val = float(2.0 * n * max_val / (target_sum * target_sum))
    best = a.copy()

    rng = np.random.default_rng()
    it = 0
    momentum = np.zeros_like(a)

    while time.time() < deadline:
        conv, max_val, kmax, grad = _max_conv_and_subgrad(a)

        # momentum on the raw sub‑gradient
        momentum = gamma * momentum + grad

        norm = np.linalg.norm(momentum)
        if norm == 0:
            break

        # decreasing learning rate (≈ 1/√it)
        decay = 1.0 / (1.0 + it / 5000.0)
        lr = base_step * target_sum * decay / norm

        # move opposite to the gradient (to decrease the peak)
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # evaluate current score – we can reuse max_val
        cur_val = float(2.0 * n * max_val / (target_sum * target_sum))
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            print(f"[{time.time():.2f}] SubGD it={it:5d}  best={best_val:.6f}")

        # occasional tiny jitter to escape plateaus
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.0004, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best, best_val


# ----------------------------------------------------------------------
# Random continuous annealing (Gaussian jitter + simplex projection)
# ----------------------------------------------------------------------


def _continuous_annealing(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         start_temp: float = 0.3,
                         end_temp: float = 1e-5) -> tuple[np.ndarray, float]:
    """
    Simulated annealing where each move adds a small zero‑mean Gaussian
    vector, projects onto the simplex and accepts with Metropolis rule.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_val = _score(a, target_sum)

    rng = np.random.default_rng()
    start_time = time.time()
    while time.time() < deadline:
        # temperature schedule (linear in log‑time)
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # propose a perturbation whose amplitude follows the current temperature
        scale = target_sum * 0.001 * temp / start_temp
        delta = rng.normal(scale=scale, size=n)
        a_prop = a + delta
        a_prop = _project_to_simplex(a_prop, target_sum)

        val_prop = _score(a_prop, target_sum)

        # Metropolis acceptance
        if val_prop < best_val:
            best_val = val_prop
            best = a_prop.copy()
            a = a_prop
            print(f"[{time.time():.2f}] Anneal new best={best_val:.6f}")
        elif rng.random() < np.exp(-(val_prop - best_val) / max(temp, 1e-12)):
            a = a_prop
        # else reject – keep current a

    return best, best_val


# ----------------------------------------------------------------------
# Targeted permutation search (simulated annealing on swaps)
# ----------------------------------------------------------------------


def _permutation_search(a0: np.ndarray,
                        deadline: float,
                        start_temp: float = 0.2,
                        end_temp: float = 1e-5,
                        top_frac: float = 0.03,
                        low_frac: float = 0.03) -> tuple[np.ndarray, float]:
    """
    Swaps indices that heavily contribute to the current peak.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_val = _score(a, np.sqrt(2.0 * n))

    rng = np.random.default_rng()
    start_time = time.time()

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # contribution of each index to the peak (a[i] * a[kmax-i])
        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        # choose donor from top contributors, receiver from low contributors
        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))
        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)
        if donor == receiver:
            continue

        # perform swap
        a[donor], a[receiver] = a[receiver], a[donor]

        val = _score(a, np.sqrt(2.0 * n))

        # temperature for Metropolis
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] PermSwap new best={best_val:.6f}")

        if val < best_val or rng.random() < np.exp(-(val - best_val) / max(temp, 1e-12)):
            # keep the swap
            pass
        else:
            # revert
            a[donor], a[receiver] = a[receiver], a[donor]

    return best, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer polishing
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.4,
                                end_temp: float = 1e-5,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy donor→receiver moves with an exponential temperature schedule.
    """
    a = a0.copy()
    cur_score = _score(a, target_sum)
    best = a.copy()
    best_score = cur_score

    n = a.shape[0]
    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # current peak location
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # weight for each position in the peak (a[i] + a[kmax-i])
        w = a[idx] + a[kmax - idx]
        donor = idx[np.argmax(w)]

        # low‑weight candidates
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = 0.5 * a[donor]
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a, target_sum)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with a low evaluation score.
    The function respects the 1000 s budget and stores the best vector
    in the global variable `height_sequence_1` for possible reuse.
    """
    start_time = time.time()
    # a tiny safety margin – we never want to exceed the budget
    deadline = start_time + budget_s - 0.4

    # ------------------------------------------------------------------
    # 1. Choose an exploration length (moderate size)
    # ------------------------------------------------------------------
    # The medium size must be small enough for many fast sub‑gradient runs
    n_mid = 5000
    target_mid = np.sqrt(2.0 * n_mid)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # 2. Build a diverse pool of seed sequences
    # ------------------------------------------------------------------
    seed_pool = []

    # power‑law family
    for p in np.linspace(0.4, 0.8, 9):
        seed_pool.append(_power_law_sequence(n_mid, float(p), target_mid))

    # exponential family
    for lam in np.linspace(0.001, 0.02, 7):
        seed_pool.append(_exponential_sequence(n_mid, float(lam), target_mid))

    # random Dirichlet‑like seeds
    for _ in range(8):
        a = rng.random(n_mid)
        a *= target_mid / a.sum()
        seed_pool.append(a)

    # linear decreasing shape
    seed_pool.append(_linear_sequence(n_mid, target_mid))

    # reuse previous best if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        seed_pool.append(_resize_sequence(prev, n_mid))

    # jittered copies of every seed (adds extra diversity)
    for seq in list(seed_pool):
        jitter = rng.normal(scale=target_mid * 0.008, size=n_mid)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_mid)
        seed_pool.append(seq_j)

    # ------------------------------------------------------------------
    # 3. Quick sub‑gradient runs on the medium size
    # ------------------------------------------------------------------
    best_mid = None
    best_mid_score = np.inf

    # allocate ~30 s for this stage
    quick_deadline = min(deadline - 30.0, start_time + 30.0)
    while time.time() < quick_deadline:
        seed = rng.choice(seed_pool)
        run_dead = time.time() + 5.0          # very short runs (≈5 s each)
        if run_dead > deadline - 5.0:
            break
        a_opt, val_opt = _subgradient_descent(seed, run_dead, target_mid,
                                              base_step=0.7, gamma=0.99)
        if val_opt < best_mid_score:
            best_mid_score = val_opt
            best_mid = a_opt.copy()
            print(f"[{time.time():.2f}] Quick SGD new best={best_mid_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Upscale to a large size (the main optimisation arena)
    # ------------------------------------------------------------------
    if best_mid is None:
        best_mid = _power_law_sequence(n_mid, 0.6, target_mid)

    # final length – larger gives more degrees of freedom, but we must keep
    # the whole pipeline inside the time budget.
    # Empirically ~20000 works well within <1000 s.
    n_big = 20000
    target_big = np.sqrt(2.0 * n_big)

    big_seq = _resize_sequence(best_mid, n_big)
    big_seq = _project_to_simplex(big_seq, target_big)

    # try a couple of deterministic reorderings and keep the best
    candidates = [big_seq,
                  _alternating_order(big_seq),
                  _alternating_order(_alternating_order(big_seq))]  # double alternation ≈ reverse
    best_big = big_seq
    best_big_score = _score(big_seq, target_big)
    for cand in candidates[1:]:
        sc = _score(cand, target_big)
        if sc < best_big_score:
            best_big_score = sc
            best_big = cand.copy()
    print(f"[{time.time():.2f}] Upscaled n={n_big} start score={best_big_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Long sub‑gradient refinement on the large vector
    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0:
        sub_dead = deadline - 12.0
        a_opt, val_opt = _subgradient_descent(best_big, sub_dead,
                                              target_big,
                                              base_step=0.6,
                                              gamma=0.993)
        if val_opt < best_big_score:
            best_big_score = val_opt
            best_big = a_opt.copy()
            print(f"[{time.time():.2f}] Long SubGD best={best_big_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Continuous annealing (random Gaussian jitter)
    # ------------------------------------------------------------------
    if time.time() < deadline - 12.0:
        anneal_dead = deadline - 6.0
        a_opt, val_opt = _continuous_annealing(best_big, anneal_dead,
                                               target_big,
                                               start_temp=0.35,
                                               end_temp=5e-6)
        if val_opt < best_big_score:
            best_big_score = val_opt
            best_big = a_opt.copy()
            print(f"[{time.time():.2f}] Continuous anneal best={best_big_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Targeted permutation search (swap order only)
    # ------------------------------------------------------------------
    if time.time() < deadline - 8.0:
        perm_dead = time.time() + min(12.0, deadline - time.time() - 2.0)
        a_opt, val_opt = _permutation_search(best_big, perm_dead,
                                             start_temp=0.18,
                                             end_temp=2e-6,
                                             top_frac=0.04,
                                             low_frac=0.04)
        if val_opt < best_big_score:
            best_big_score = val_opt
            best_big = a_opt.copy()
            print(f"[{time.time():.2f}] Permutation search best={best_big_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Deterministic mass‑transfer polishing (final fine‑tuning)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        mt_dead = time.time() + min(5.0, deadline - time.time() - 1.0)
        a_opt, val_opt = _deterministic_mass_transfer(best_big, mt_dead,
                                                      target_big,
                                                      start_temp=0.3,
                                                      end_temp=1e-6,
                                                      low_frac=0.05)
        if val_opt < best_big_score:
            best_big_score = val_opt
            best_big = a_opt.copy()
            print(f"[{time.time():.2f}] Mass‑transfer final best={best_big_score:.6f}")

    # ------------------------------------------------------------------
    # 9. Trim trailing zeros, produce final list
    # ------------------------------------------------------------------
    arr = np.array(best_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # 10. Final evaluation (optional logging)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # store for the next round
    globals()["height_sequence_1"] = final_seq
    return final_seq
```
