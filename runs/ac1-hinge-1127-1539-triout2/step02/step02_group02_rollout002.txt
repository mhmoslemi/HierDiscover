"""
Simple tail‑augmentation of the best known step function.

The idea
---------
Take the best sequence that we already have (available as the global
``height_sequence_1``).  Let

    S = Σ a_i                (total L1‑mass)
    B = max (a * a)          (max self‑convolution)

Adding a block of *m* entries of the same height ``ε`` modifies the
objective

    score = 2·n·B / (Σ a)^2

in a predictable way.  If we choose

    ε = B / S                              (so that ε·S = B)
    m = S² / B                             (so that ε²·m = B)

the three possible contributions to the new maximum convolution
(original, cross, and tail‑self) all equal ``B``.  The L1‑mass roughly
doubles while the denominator in the score grows quadratically,
yielding a new score of about

    0.5 + ¼·old_score  ≈ 0.88   (for the previous 1.52)

which is well below the target 1.503.

The code builds a few candidates around this design (slightly rounded
length, 0.99·ε for safety, ±1 on *m*) and returns the one with the
lowest external evaluation.  The whole process finishes in a fraction
of a second.
"""

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) yields a
    score far below the current bound of 1.503.

    The construction is the classic “best‑known sequence + uniform tail”
    described in the module docstring.
    """
    # ------------------------------------------------------------------
    # 1.  Base sequence -------------------------------------------------
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=float)
        if base.ndim != 1 or base.size == 0:
            raise ValueError("height_sequence_1 is not a valid 1‑D sequence")
    else:
        # fallback: a modest uniform sequence (length 4000)
        n0 = 4000
        base = np.full(n0, np.sqrt(2.0 * n0) / n0, dtype=float)

    # ------------------------------------------------------------------
    # 2.  Compute its total mass and maximal self‑convolution -----------
    # ------------------------------------------------------------------
    S = float(base.sum())
    conv_full = np.convolve(base, base)
    B = float(conv_full.max())

    if S <= 0 or B <= 0:
        # Degenerate case – just return the base sequence.
        return base.tolist()

    # ------------------------------------------------------------------
    # 3.  Build candidate tails -----------------------------------------
    # ------------------------------------------------------------------
    # Ideal parameters (may be fractional)
    eps_ideal = B / S                       # height that makes cross term = B
    m_ideal   = S * S / B                   # length that makes tail self‑conv = B

    # Round length to an integer, keep it reasonable
    m_base = max(1, int(round(m_ideal)))

    # Try a few safe variations
    scales = [1.0, 0.99, 0.95]               # scaling on ε (to keep cross term < B)
    dms    = [0, -1, 1]                     # tiny adjustments of the length

    best_seq   = None
    best_score = float('inf')

    # Helper for evaluating a concrete candidate
    def _eval(candidate: np.ndarray):
        try:
            sc = evaluate_sequence(candidate.tolist())
        except Exception as e:
            print("[eval error]", e)
            sc = float('inf')
        return sc

    # ------------------------------------------------------------------
    # 4.  Enumerate candidates and keep the best -----------------------
    # ------------------------------------------------------------------
    for scale in scales:
        eps = eps_ideal * scale
        # enforce the global caps (should never be hit here)
        eps = min(max(eps, 0.0), 1000.0)

        for dm in dms:
            m = max(1, m_base + dm)

            tail = np.full(m, eps, dtype=float)
            cand = np.concatenate([base, tail])

            # safety clip (again, just in case)
            cand = np.clip(cand, 0.0, 1000.0)

            sc = _eval(cand)
            print(f"[candidate] len={len(cand):6d}  eps={eps:.6f}  m={m:6d}  "
                  f"score={sc:.6f}")

            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"  ==> new global best (score={best_score:.6f})")

    # ------------------------------------------------------------------
    # 5.  Return the best sequence found --------------------------------
    # ------------------------------------------------------------------
    if best_seq is None:
        # Should never happen, but keep a safe fallback.
        best_seq = base.copy()
        best_score = _eval(best_seq)

    print(f"[final] best score = {best_score:.6f}  length = {len(best_seq)}")
    return best_seq.tolist()
