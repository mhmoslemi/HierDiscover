"""
Optimiser for the max‑convolution objective.

The evaluation function
    evaluate_sequence(seq)  -> 2·n·max(a*a) / (sum(a)^2)

is equal to the maximal entry of the self‑convolution when the L1‑norm of
the sequence is fixed to √(2 n).  Hence we minimise the quantity
    max(a*a)
under the constraints a_i ≥ 0 and Σ a_i = √(2 n).

The algorithm is a projected (multiplicative) gradient descent on a smooth
approximation of the “max’’ function.  For a given vector a we compute the
full convolution c = a*a, form a soft‑max weighting
    w_k ∝ exp(α·(c_k – max(c)))                (α > 0)
and obtain an exact gradient of the smooth objective:
    ∂/∂a_i  (log‑sum‑exp_α(c)) = 2·Σ_j a_j·w_{i+j}.
The gradient is applied multiplicatively (a ← a·exp(‑η·v)) so that
non‑negativity is automatic; after each step a is projected back onto the
affine L1‑sphere Σa = √(2 n).  Momentum (exponential moving average) and a
small random jitter are used to accelerate convergence and escape plateaus.

Multiple independent restarts with different lengths and seeds are performed
until the time budget (≈ 1000 s) is exhausted.  The best sequence seen so
far is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n = x.shape[0]
    m = y.shape[0]
    conv_len = n + m - 1
    fft_len = _next_pow2(conv_len)
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[:conv_len]
    return conv


def self_convolution(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Core optimisation of one restart
# ----------------------------------------------------------------------


def _run_one_restart(
    a0: np.ndarray,
    n: int,
    deadline: float,
    *,
    alpha: float = 10.0,
    eta0: float = 0.25,
    beta: float = 0.9,
    decay: float = 0.99995,
    jitter_prob: float = 0.015,
    max_iters: int = 20000,
) -> tuple[np.ndarray, float]:
    """
    Projected exponentiated gradient descent on the soft‑max surrogate of
    max(a*a).  Returns the best sequence seen and its external score.
    """
    a = a0.copy()
    S = np.sqrt(2.0 * n)                # required L1‑norm
    # external evaluation of the start point
    try:
        best_score = evaluate_sequence(a.tolist())
    except Exception:                 # pragma: no cover
        best_score = float("inf")
    best_seq = a.copy()
    best_obj = float("inf")            # best observed max(a*a)
    v = np.zeros_like(a)               # momentum buffer
    eta = eta0
    it = 0
    # Pre‑allocate arrays to avoid reallocation inside the loop
    a_rev = np.empty_like(a)

    while time.time() < deadline and it < max_iters:
        it += 1

        # ----- self‑convolution and soft‑max weights -----
        conv = self_convolution(a)                     # length 2n‑1
        cur_max = float(conv.max())
        # soft‑max over the convolution entries (stable version)
        exp_scaled = np.exp(alpha * (conv - cur_max))
        w = exp_scaled / exp_scaled.sum()              # Σ w = 1

        # ----- gradient of the smooth max -----
        a_rev[:] = a[::-1]                              # reversed copy
        # correlation g_i = 2 * Σ_j a_j * w_{i+j}
        corr = convolve_fft(w, a_rev)                   # length 3n‑2
        g = 2.0 * corr[n - 1 : 2 * n - 1]               # slice of length n

        # ----- momentum and multiplicative update -----
        v = beta * v + (1.0 - beta) * g
        a = a * np.exp(-eta * v)                       # element‑wise

        # ----- projection onto the L1‑sphere -----
        total = a.sum()
        if total == 0.0:
            a.fill(S / n)
        else:
            a *= (S / total)

        # ----- occasional jitter to escape tiny plateaus -----
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * 0.01
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()

        # ----- step‑size and temperature decay -----
        eta *= decay
        alpha *= 0.9999   # slowly sharpen the soft‑max

        # ----- record improvement -----
        if cur_max < best_obj - 1e-12:
            best_obj = cur_max
            try:
                ext = evaluate_sequence(a.tolist())
                if ext < best_score:
                    best_score = ext
                    best_seq = a.copy()
                    print(
                        f"[local improve] t={time.time():.1f}s  score={best_score:.6f}"
                        f"  n={n}  iter={it}  max={cur_max:.6f}"
                    )
            except Exception:                         # pragma: no cover
                pass

    return best_seq, best_score


# ----------------------------------------------------------------------
# Public entry point – used by the judging harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **_):
    """
    Search for a non‑negative step function with a low self‑convolution
    maximum.  The routine runs many independent restarts of the optimiser
    described above until the time budget is exhausted and returns the best
    sequence found (as a plain Python list).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # keep a small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Initialise from any pre‑existing construction (if the environment
    # already defines a variable named height_sequence_1).
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            if cand.ndim == 1 and cand.size > 0:
                sc = evaluate_sequence(cand.tolist())
                best_seq = cand.copy()
                best_score = sc
                print(
                    f"[baseline] height_sequence_1 score={best_score:.6f}"
                    f"  len={cand.size}"
                )
        except Exception:                         # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # Lengths to explore.  Larger lengths tend to give better bounds but
    # are more expensive per iteration, so we mix a few options.
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000, 12000, 16000]

    run_id = 0
    while time.time() < deadline:
        n = int(rng.choice(lengths))

        # ----- initialise a0 -------------------------------------------------
        if best_seq is not None and best_seq.size >= n and rng.random() < 0.35:
            a0 = best_seq[:n].copy()
        else:
            # Dirichlet‑like random start (positive entries)
            a0 = rng.random(n) + 0.1 * rng.random()  # avoid exact zeros
        # normalise to the required L1‑norm √(2 n)
        S = np.sqrt(2.0 * n)
        a0 = a0 / a0.sum() * S

        # compute a (weak) estimate of how many iterations we can afford
        # for this length – we keep it roughly proportional to 1/n
        max_iters = max(2000, int(20000 * 2000 / n))

        # ----- run a single optimisation instance ---------------------------
        seq_opt, score_opt = _run_one_restart(
            a0,
            n,
            deadline,
            alpha=12.0,
            eta0=0.22,
            beta=0.9,
            decay=0.99993,
            jitter_prob=0.012,
            max_iters=max_iters,
        )

        # ----- keep the global best -----------------------------------------
        if score_opt < best_score:
            best_score = score_opt
            best_seq = seq_opt.copy()
            print(f"[global improve] run={run_id} n={n} score={best_score:.6f}")

        run_id += 1

    # ------------------------------------------------------------------
    # Fallback (should never be required, but kept for safety)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        a = np.full(n, np.sqrt(2.0 * n) / n)
        best_seq = a
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] score={best_score:.6f}")

    print(f"[final] best score = {best_score:.6f}   length = {len(best_seq)}")
    # Return a plain Python list as required by the harness
    return best_seq.tolist()
