inside the 1000 s budget:

def _generate_powerlaw(p: float, sigma: float,
                       max_len: int = 5000, eps: float = 1e-10) -> np.ndarray:
    """a_i = (i + sigma) ** (-p), truncated when a_i < eps."""
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p)
    return a[a > eps]


def _generate_exponential(lam: float,
                          max_len: int = 5000, eps: float = 1e-10) -> np.ndarray:
    """a_i = exp(-lam * i), truncated when a_i < eps."""
    i = np.arange(max_len, dtype=np.float64)
    a = np.exp(-lam * i)
    return a[a > eps]


def _generate_mixed(p: float, sigma: float, lam: float,
                    max_len: int = 5000, eps: float = 1e-10) -> np.ndarray:
    """a_i = (i+sigma)^(-p) * exp(-lam*i), truncated when a_i < eps."""
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p) * np.exp(-lam * i)
    return a[a > eps]


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = 1}."""
    if v.ndim != 1:
        raise ValueError("simplex_projection works only on 1‑D arrays")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(x: np.ndarray) -> float:
    """
    Direct implementation of evaluate_sequence *without* the security
    wrappers (exactly the same arithmetic).  Returns a float;  np.inf
    is returned for illegal (too small sum) inputs.
    """
    if x.ndim != 1 or x.size == 0:
        return float('inf')
    if np.any(x < 0):
        return float('inf')
    s = x.sum()
    if s < 0.01:
        return float('inf')
    n = x.shape[0]

    # fast convolution – use FFT for n>1024, else np.convolve
    if n > 1024:
        size = 1 << (2 * n - 1).bit_length()   # next power of two
        fx = np.fft.rfft(x, n=size)
        conv = np.fft.irfft(fx * fx, n=size)[: 2 * n - 1]
        max_b = conv.max()
    else:
        max_b = np.convolve(x, x).max()

    return float(2.0 * n * max_b / (s * s))


def _subgradient(x: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of   g(x) = max_k (x*x)_k   with respect to x,
    computed using all active (maximising) convolution indices.
    """
    n = x.shape[0]
    conv = np.convolve(x, x)
    m = conv.max()
    active = np.where(conv >= m - 1e-14)[0]
    g = np.zeros_like(x)
    for k in active:
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        i = np.arange(i_min, i_max + 1, dtype=int)
        j = k - i
        # derivative of conv[k] w.r.t. x_i is 2 * x_j (the factor 2 absorbed)
        g[i] += x[j]
    return g


def _assign_interleaved(weights: np.ndarray, length: int = None) -> np.ndarray:
    """
    Place the given (non‑negative) weights on a line of ``length`` points.
    The algorithm picks the middle of the largest free interval,
    recurses on the left/right parts and therefore spreads large
    weights as far apart as possible – an inexpensive heuristic for a
    flatter auto‑convolution.
    """
    m = weights.shape[0]
    if length is None:
        length = max(2 * m - 1, m)      # enough room for a near‑unique sum pattern
    positions = np.empty(m, dtype=int)

    # intervals = list of (left, right) inclusive
    intervals = [(0, length - 1)]
    # place weights from largest to smallest
    order = np.argsort(-weights)        # descending order
    for idx in order:
        # pick the current longest interval
        intervals.sort(key=lambda iv: iv[1] - iv[0], reverse=True)
        l, r = intervals.pop(0)
        pos = (l + r) // 2
        positions[idx] = pos
        # split remaining space
        if pos - 1 >= l:
            intervals.append((l, pos - 1))
        if pos + 1 <= r:
            intervals.append((pos + 1, r))

    seq = np.zeros(length, dtype=np.float64)
    seq[positions] = weights
    return seq

            seq = seq / seq.sum()
            score = _objective(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.tolist()
                print(f'new best PL   p={p:.3f} σ={sigma:.5f} '
                      f'len={len(seq)} score={best_score:.6f}')
        if time.time() >= deadline:
            break

    # exponential
    lam_vals = np.logspace(-4, 0, 30)
    for lam in lam_vals:
        if time.time() >= deadline:
            break
        seq = _generate_exponential(lam, max_len, eps)
        if seq.size < 2:
            continue
        seq = seq / seq.sum()
        score = _objective(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.tolist()
            print(f'new best EXP  λ={lam:.5e} len={len(seq)} '
                  f'score={best_score:.6f}')

    # mixed family
    for p in (0.45, 0.5, 0.55):
        for sigma in (1e-3, 1e-2, 1e-1, 1.0):
            for lam in (1e-3, 5e-3, 1e-2):
                if time.time() >= deadline:
                    break
                seq = _generate_mixed(p, sigma, lam, max_len, eps)
                if seq.size < 2:
                    continue
                seq = seq / seq.sum()
                score = _objective(seq)
                if score < best_score:
                    best_score = score
                    best_seq = seq.tolist()
                    print(f'new best MIX p={p:.2f} σ={sigma:.5g} λ={lam:.5g} '
                          f'len={len(seq)} score={best_score:.6f}')
    # ------------------------------------------------------------------
    # 2.  Sparse weighted placement heuristics.
    # ------------------------------------------------------------------
    print('=== Phase 2 – sparse weighted placement ===')
    # draw a few Dirichlet vectors (α < 1 gives sparse, high‑contrast weights)
    for m in (200, 400, 800):
        for alpha in (0.2, 0.4, 0.7):
            if time.time() >= deadline:
                break
            # generate a Dirichlet sample
            weights = np.random.gamma(alpha, 1.0, size=m)
            weights /= weights.sum()
            # place using interleaved layout
            seq = _assign_interleaved(weights)
            # normalise (just in case rounding moved sum)
            seq = seq / seq.sum()
            score = _objective(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.tolist()
                print(f'new best PLmt m={m} α={alpha:.2f} len={len(seq)} '
                      f'score={best_score:.6f}')

    # ------------------------------------------------------------------
    # 3.  Projected sub‑gradient descent (momentum + occasional growth).
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform vector of length 1000
        n0 = 1000
        best_seq = np.full(n0, 1.0 / n0).tolist()
        best_score = _objective(np.array(best_seq))
        print('fallback to uniform sequence, score =', best_score)

    x = np.array(best_seq, dtype=np.float64)
    x = x / x.sum()
    step0 = 0.7
    beta = 0.86
    momentum = np.zeros_like(x)
    iteration = 0
    grow_interval = 3000
    max_total_len = 8000

    print('=== Phase 3 – projected sub‑gradient descent ===')
    while time.time() < deadline:
        iteration += 1
        g = _subgradient(x)

        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / np.sqrt(iteration)

        y = x - step * momentum
        x = _simplex_projection(y)

        # tiny random jitter
        if np.random.rand() < 0.015:
            j = np.random.randint(x.shape[0])
            delta = (np.random.rand() - 0.5) * 0.08 * x[j]
            x[j] = max(0.0, x[j] + delta)
            x = x / x.sum()

        # occasional length growth
        if iteration % grow_interval == 0 and x.shape[0] < max_total_len:
            extra = np.full(200, 1e-12, dtype=np.float64)
            x = np.concatenate([x, extra])
            x = x / x.sum()
            momentum = np.concatenate([momentum, np.zeros_like(extra)])

        cur_score = _objective(x)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = x.tolist()
            print(f'iter {iteration:6d} → improved score = {best_score:.6f}')
            if best_score < 1.45:        # a more ambitious early stop
                print('Excellent bound reached – stop sub‑gradient phase.')
                break

        # simple restart if no improvement for a while
        if iteration % 5000 == 0:
            n = x.shape[0]
            rnd = np.random.rand(n)
            x = 0.5 * x + 0.5 * rnd
            x = x / x.sum()
            momentum = np.zeros_like(x)

    # ------------------------------------------------------------------
    # 4.  Simulated annealing refinement.
    # ------------------------------------------------------------------
    print('=== Phase 4 – simulated annealing refinement ===')
    x = np.array(best_seq, dtype=np.float64)
    x = x / x.sum()
    cur_score = _objective(x)
    best_local_score = cur_score
    best_local_seq = x.copy()

    # temperature schedule – linear cooling from T0 to Tmin
    T0 = 0.08
    Tmin = 1e-6
    total_iters = 0

    while time.time() < deadline:
        total_iters += 1
        # linear interpolation based on remaining time
        remaining = deadline - time.time()
        frac = remaining / (budget_s - 5.0)          # approx fraction of total budget left
        T = max(Tmin, T0 * frac)

        proposal = x.copy()
        if np.random.rand() < 0.25:                   # swap two positions
            i, j = np.random.randint(proposal.size, size=2)
            proposal[i], proposal[j] = proposal[j], proposal[i]
        else:                                          # multiplicative perturbation of a single entry
            i = np.random.randint(proposal.size)
            factor = np.exp(np.random.normal(scale=0.12))   # log‑normal factor ≈[0.7,1.4]
            proposal[i] *= factor
            if proposal[i] < 0:
                proposal[i] = 0.0
            proposal = proposal / proposal.sum()

        new_score = _objective(proposal)

        # Metropolis acceptance
        if new_score < cur_score or np.random.rand() < math.exp(-(new_score - cur_score) / T):
            x = proposal
            cur_score = new_score

            if new_score < best_local_score:
                best_local_score = new_score
                best_local_seq = x.copy()
                print(f'  SA iter {total_iters:7d} → new best {best_local_score:.6f}')
                if best_local_score < best_score:
                    best_score = best_local_score
                    best_seq = best_local_seq.tolist()
                    print('*** Global best improved by SA ***')

    print('final best score:', best_score, 'length:', len(best_seq))
    return best_seq
