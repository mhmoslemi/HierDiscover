"""
Non‑negative step‑function optimizer (improved).

The score to minimise after normalising sum(a)=1 is
        score = 2 * n * max(conv(a, a))

where conv is the full linear convolution of the vector with itself.
The algorithm explores several lengths and initialisations,
then performs a mirror‑descent–like optimisation that
  * always uses the *exact* active (maximal) convolution entries,
  * builds a true sub‑gradient for those entries,
  * applies an exponentiated‑gradient update with a back‑tracking line search,
  * enforces symmetry (p[i]=p[-1-i]) and projects onto the probability simplex.
Random jitter and occasional pruning of tiny components help escaping plateaus.
The routine respects a 1000 s budget and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _full_conv_fft(p: np.ndarray) -> np.ndarray:
    """Linear convolution of a real vector with itself via FFT.

    Returns the real part of length 2*len(p)-1.
    """
    n = p.shape[0]
    # Next power of two for efficient FFT (zero‑padding)
    L = 1 << ((2 * n - 2).bit_length())
    ft = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(ft * ft, n=L)[: 2 * n - 1]
    conv = np.real(conv)
    conv[conv < 0] = 0.0          # clip tiny negatives caused by round‑off
    return conv


def _subgradient_max(p: np.ndarray, conv: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """Exact sub‑gradient of  max_k conv_k  w.r.t. the probability vector p.

    All k for which conv_k is within `eps` of the maximum are treated as active.
    """
    n = p.shape[0]
    max_val = np.max(conv)
    ks = np.where(conv >= max_val - eps)[0]          # active positions
    g = np.zeros_like(p)

    for k in ks:
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        idx = np.arange(i_min, i_max + 1, dtype=int)
        # contribution for each i in the sum p_i * p_{k-i}
        g[idx] += p[k - idx]

    g *= 2.0                                          # ordered‑pair derivative
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_from_prob(p: np.ndarray) -> float:
    """Score = 2 * n * max(conv(p,p))  for a probability vector p."""
    n = p.shape[0]
    conv = _full_conv_fft(p)
    return 2.0 * n * np.max(conv)
