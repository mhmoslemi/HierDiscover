"""
Optimisation of a non‑negative step‑function for the metric

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm uses a single Adam‑style optimiser on the smooth
approximation of the max (log‑sum‑exp).  The mass is always re‑projected
onto the ℓ₁‑simplex (∑a = √(2 n), a_i ≥ 0) after each update, so the
objective evaluated by ``evaluate_sequence`` is exactly the current
maximum of the self‑convolution.  A simple parametric “U‑shape”
initialiser is used unless a previously discovered sequence
(`height_sequence_1`) is available.  Random jitter and occasional
pair‑mass transfers keep the search from stagnating.  The routine runs
until the supplied time budget (≈1000 s) expires and returns the best
sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_self_fft(a: np.ndarray, L_self: int) -> np.ndarray:
    """
    Full linear self‑convolution of a using an FFT of length L_self.
    Returns a vector of length 2*len(a)-1.
    """
    fa = np.fft.rfft(a, n=L_self)
    conv_full = np.fft.irfft(fa * fa, n=L_self)
    return conv_full[: 2 * a.shape[0] - 1]


def conv_cross_fft(w: np.ndarray, a_rev: np.ndarray, L_cross: int) -> np.ndarray:
    """
    Full linear convolution of w with a_rev using an FFT of length L_cross.
    Returns a vector of length len(w)+len(a_rev)-1.
    """
    W = np.fft.rfft(w, n=L_cross)
    A = np.fft.rfft(a_rev, n=L_cross)
    conv_full = np.fft.irfft(W * A, n=L_cross)
    return conv_full[: len(w) + len(a_rev) - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex
        {x | x_i ≥ 0, sum_i x_i = target}.
    Implementation of Duchi et al. (2008), O(n).
    """
    if target <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries would be zero – return uniform distribution
        return np.full_like(v, target / len(v))

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a small fraction (scale) of the mass from a random index
    to another random index, then re‑project onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def generate_U_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
) -> np.ndarray:
    """
    Produce a symmetric “U‑shaped’’ sequence of length n.
    kind ∈ {linear, quadratic, exponential}.
    param ≥ 0 controls the edge enhancement.
    The result is projected onto the ℓ₁‑simplex of total mass target_sum.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid               # distance ∈ [0, 1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)
