"""
Propose a step‑function with a very small ∞‑norm of its self‑convolution.

The search consists of many short independent runs.  Each run starts from
a (possibly interpolated) previous best sequence, then performs a mixture
of

* a **multi‑peak projected exponentiated‑gradient step** – the gradient is
  built from the top‑k convolution entries (rather than only the single
  maximum), which flattens the whole high‑plateau of the autocorrelation,

* **simulated‑annealing swaps** – moving a small fraction of mass from an
  index that participates in the current maximal entry to a random index,

* occasional **jitter** – Gaussian noise to escape flat regions.

A simple linear‑interpolation routine lets us reuse the best sequence at
different lengths, allowing the algorithm to explore larger `n` (up to
~15000) without losing the shape discovered so far.  The whole routine
stops after the allotted time (≈1000 s) and returns the best list of
heights found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_max(a: np.ndarray) -> tuple[float, int]:
    """
    Return (maximum convolution entry, index where it occurs).
    Faster than computing the whole convolution when only the max is needed.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def _gradient_step_multi(a: np.ndarray,
                         eta: float,
                         top_k: int = 5) -> np.ndarray:
    """
    One multiplicative gradient step built from the gradients of the
    `top_k` largest convolution entries (if fewer entries exist all are used).

    The step is projected back onto the simplex Σa_i = √(2·n) and keeps the
    vector non‑negative.
    """
    n = a.shape[0]
    conv = _conv_full(a)

    # select the indices of the k largest entries
    if top_k >= conv.size:
        ks = np.arange(conv.size)
    else:
        ks = np.argpartition(-conv, top_k - 1)[:top_k]

    g = np.zeros_like(a)

    for k in ks:
        # partner index for each i
        idx = np.arange(n)
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        # double contribution for i != partner, single for i == partner
        g[valid] += 2.0 * a[partner[valid]]
        if k % 2 == 0:               # diagonal term was counted twice → subtract once
            d = k // 2
            if 0 <= d < n:
                g[d] -= a[d]

    # average over the selected peaks (stabilises the step size)
    g = g / len(ks)

    # multiplicative update (mirror descent with KL‑divergence)
    a_new = a * np.exp(-eta * g)

    # renormalise to the required L1‑norm √(2·n)
    S = np.sqrt(2.0 * n)
    total = a_new.sum()
    if total == 0.0:
        a_new = np.full_like(a, S / n)
    else:
        a_new *= (S / total)
    return a_new


def _initial_sequence(base_seq: np.ndarray,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting sequence of length `n` with Σa_i = √(2·n).

    If `base_seq` is supplied, it is linearly interpolated to the desired
    length and a tiny jitter is added; otherwise a uniform Dirichlet draw
    is used.
    """
    S = np.sqrt(2.0 * n)

    if base_seq is not None and base_seq.size > 1:
        # linear interpolation
        old_x = np.linspace(0.0, 1.0, base_seq.size)
        new_x = np.linspace(0.0, 1.0, n)
        a = np.interp(new_x, old_x, base_seq)
        # tiny noise to avoid exact copies across restarts
        a += rng.normal(scale=a.mean() * 0.001, size=n)
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)

    # normalise to the required L1‑norm
    total = a.sum()
    if total == 0.0:
        a = np.full(n, S / n)
    else:
        a = a / total * S
    return a


def _optimize_local(a: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Local optimiser mixing multi‑peak gradient steps, simulated‑annealing
    swaps and occasional jitter.  Returns the best sequence encountered and
    its maximal convolution value.
    """
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # current best
    cur_max, cur_idx = _conv_max(a)
    best_a = a.copy()
    best_score = cur_max

    # hyper‑parameters (tuned empirically)
    eta = 0.8                 # initial step‑size for the multiplicative step
    eta_decay = 0.9995        # exponential decay per iteration
    move_prob = 0.12          # probability of attempting a random mass move
    epsilon_move = 0.06       # fraction of the source mass to move
    jitter_prob = 0.008       # occasional Gaussian jitter
    temp = cur_max * 0.15     # annealing temperature
    temp_decay = 0.9993

    while time.time() < deadline:
        if rng.random() < move_prob:
            # ---- simulated‑annealing random move ----
            # indices that actually contribute to the current maximal entry
            partner = cur_idx - np.arange(n)
            source_idx = np.where((partner >= 0) & (partner < n))[0]
            non_source_idx = np.where(~((partner >= 0) & (partner < n)))[0]

            if source_idx.size and non_source_idx.size:
                i = rng.choice(source_idx)
                j = rng.choice(non_source_idx)

                delta = epsilon_move * a[i]
                delta = min(delta, a[i])          # keep non‑negative

                a_candidate = a.copy()
                a_candidate[i] -= delta
                a_candidate[j] += delta

                cand_max, cand_idx = _conv_max(a_candidate)

                # Metropolis acceptance
                accept = False
                if cand_max <= cur_max:
                    accept = True
                else:
                    prob = np.exp(-(cand_max - cur_max) / max(temp, 1e-12))
                    if rng.random() < prob:
                        accept = True

                if accept:
                    a = a_candidate
                    cur_max = cand_max
                    cur_idx = cand_idx
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                temp *= temp_decay
        else:
            # ---- multi‑peak gradient descent step ----
            a = _gradient_step_multi(a, eta, top_k=5)
            cur_max, cur_idx = _conv_max(a)
            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()

        # occasional jitter to escape tiny plateaus
        if rng.random() < jitter_prob:
            noise = rng.normal(scale=a.mean() * 0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = a / a.sum() * S

        eta *= eta_decay

    return best_a, best_score
