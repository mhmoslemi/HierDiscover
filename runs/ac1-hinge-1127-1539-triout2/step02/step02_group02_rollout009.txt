def proj_simplex(v, z=1.0):
    # Ensure v >= 0? algorithm works for any v.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
fast_softmax_gradient.py

A gradient‑based optimizer for the step‑function problem.

The algorithm works on a fixed length `n` and keeps the L‑1 norm
(Σ a_i) equal to √(2 n).  For a given vector `a`

    c = a ★ a          # linear self‑convolution (FFT, O(n log n))
    M = max(c)        # the quantity we want to minimise

Instead of a sub‑gradient based on the top‑K entries, we form a smooth
approximation of the max using a soft‑max weight

    w_k = exp(β·(c_k−M)),  w ← w / Σ w ,

and compute the exact gradient of the weighted sum

    g_j = Σ_k w_k·2·a_{k−j}  = 2·(a ⊛ w)[j] ,

where ⊛ is the (linear) cross‑correlation.  The gradient is obtained
with one extra FFT.  A momentum term (Adam‑style) is applied and the
iterates are projected back onto the simplex Σa = √(2 n), a ≥ 0.
The surrogate `M` is monitored; whenever it improves we call the
expensive `evaluate_sequence` and keep the best overall vector.

Multiple independent restarts with different lengths are run until the
time budget (≈ 1000 s) is exhausted.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _project_onto_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = z}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if z < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(len(v)) + 1
    # Find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries will be zero
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using a FFT of length L (L ≥ 2·len(a)−1).
    Returns the full convolution array of length 2·n−1.
    """
    Fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Fa * Fa, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _gradient_softmax(a: np.ndarray,
                     conv: np.ndarray,
                     beta: float,
                     L: int) -> np.ndarray:
    """
    Compute the gradient of the soft‑max surrogate.
    Returns a vector g with the same length as a.
    """
    n = a.shape[0]
    # soft‑max weights (normalised)
    w = np.exp(beta * (conv - conv.max()))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # cross‑correlation a ⊛ w  (linear, not circular)
    Fa = np.fft.rfft(a, n=L)
    Fw = np.fft.rfft(w, n=L)
    corr = np.fft.irfft(Fa * np.conj(Fw), n=L)

    offset = conv.shape[0] - 1        # = 2·n‑2
    grad = 2.0 * corr[offset : offset + n]
    return grad


def _run_optimize_one_start(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            init_lr: float = 0.01,
                            lr_decay: float = 0.9999,
                            momentum: float = 0.9,
                            beta_start: float = 0.5,
                            beta_inc: float = 1.001,
                            jitter_prob: float = 0.01,
                            max_iters: int = 200000) -> tuple[np.ndarray, float]:
    """
    Gradient descent with a soft‑max surrogate, projected onto the simplex.
    Returns the best sequence encountered and its external score.
    """
    a = a0.copy()
    S = np.sqrt(2.0 * n)                     # required L1‑norm

    # initial external evaluation
    try:
        best_score = evaluate_sequence(a.tolist())
    except Exception:                       # safety – should not happen
        best_score = float("inf")
    best_seq = a.copy()
    best_surrogate = float("inf")

    # momentum buffer
    vel = np.zeros_like(a)

    # constant FFT length (covers both convolution and correlation)
    L_fft = _next_pow2(3 * n - 2)

    lr = init_lr
    beta = beta_start
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # ---- 1. convolution and surrogate ----
        conv = _conv_full_fft(a, L_fft)          # length 2n‑1
        cur_max = float(conv.max())

        # ---- 2. gradient of the soft‑max ----
        grad = _gradient_softmax(a, conv, beta, L_fft)

        # ---- 3. momentum (Adam‑style) ----
        vel = momentum * vel + (1.0 - momentum) * grad

        # ---- 4. gradient step (additive) ----
        a = a - lr * vel

        # ---- 5. occasional jitter to escape plateaus ----
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * (a.mean() if a.mean() > 0 else 1.0) * 0.01
            a += noise

        # ---- 6. enforce non‑negativity & L1‑norm ----
        a = np.maximum(a, 0.0)
        total = a.sum()
        if total == 0.0:
            a.fill(S / n)
        else:
            a *= (S / total)        # simple scaling = projection onto the simplex

        # ---- 7. adapt learning rate and temperature ----
        lr *= lr_decay
        beta *= beta_inc

        # ---- 8. keep the best external score seen ----
        if cur_max < best_surrogate - 1e-12:
            best_surrogate = cur_max
            try:
                ext = evaluate_sequence(a.tolist())
                if ext < best_score:
                    best_score = ext
                    best_seq = a.copy()
                    print(f"[local improve] time={time.time():.1f}s  "
                          f"score={best_score:.6f}  n={n}  iter={it}  max={cur_max:.6f}")
            except Exception:
                pass

    return best_seq, best_score
