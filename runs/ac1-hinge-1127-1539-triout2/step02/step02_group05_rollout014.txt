"""
Propose a non‑negative step‑function that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The implementation uses a projected Adam optimiser that works directly
with the true ℓ∞‑objective.  At every iteration the convolution
`a*a` is computed, the (possibly several) indices where the maximum is
attained are identified and a hard‑max weight vector is built.
The resulting gradient is the exact sub‑gradient of the max‑convolution
objective, i.e.  ∇ maxₖ (a∗a)[k] = 2·(w⊛a_rev).  After a gradient step the
vector is projected onto the simplex
    {x ≥ 0, ∑x = √(2 n)}.
A simple restart strategy injects small random jitter when the search
stalls for many iterations.  After the main phase a fine‑tuning stage
with a very small step size further polishes the best sequence found.
All heavy work (convolutions) is done with FFTs for speed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ *x*."""
    return 1 << (x - 1).bit_length()


def _compute_score(a: np.ndarray) -> float:
    """
    Fast version of the provided ``evaluate_sequence``.
    Assumes ``a`` is non‑negative.
    Returns the same quantity: 2*n*max(conv)/sum(a)^2 = max(conv) when
    sum(a)=√(2n).
    """
    n = a.shape[0]
    nfft = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
    return float(conv.max())


def project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of *v* onto
        { x : x_i ≥ 0 , ∑ x_i = target_sum }.
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence with a low value of the evaluation
    function.  The algorithm proceeds in three stages:

    1.  Initialise from the supplied ``height_sequence_1`` (if any) and a
        handful of random power‑law / exponential candidates.
    2.  Main optimisation loop – projected Adam updates where the
        gradient is the exact sub‑gradient of the ℓ∞‑objective
        (hard‑max weight vector).
        A restart with jitter is triggered after a long period without
        improvement.
    3.  Fine‑tuning – a tiny plain gradient step for many iterations.

    The best sequence discovered before the time budget expires is
    returned as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # safety margin

    # --------------------------------------------------------------
    # 1. Determine problem size and target ℓ1‑norm (√(2n))
    # --------------------------------------------------------------
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        n = init_seq.shape[0]
    except Exception:
        n = 6000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # 2. Build a modest set of initial candidates
    # --------------------------------------------------------------
    best_a = None
    best_score = np.inf

    candidates = []
