def project_to_simplex(v, target):
    # v is non-negative or any real numbers.
    # We'll follow Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - target) / (np.arange(1, len(v)+1)) > 0)[0]
    if rho.size == 0:
        # all zeros, return uniform
        return np.full_like(v, target/len(v))
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1)
    w = np.maximum(v - theta, 0)
    # Due to numerical, we may need to renorm
    w_sum = w.sum()
    if w_sum > 0:
        w = w * (target / w_sum)
    else:
        w = np.full_like(v, target/len(v))
    return w

"""
Propose a non‑negative height sequence that (approximately) minimises
evaluate_sequence().  The search consists of three stages:

1.  A broad sweep of analytically‑defined families (symmetric “U” shapes,
    power‑U, exponential, linear, quadratic) together with random starts.
    The best candidate is kept as a seed.

2.  Several restarts of a projected Adam optimiser that minimises a
    smooth approximation of the ℓ∞‑norm of the self‑convolution.
    The temperature of the soft‑max is gradually lowered, and after each
    step the vector is projected onto the simplex (non‑negative, fixed ℓ₁‑norm).

3.  A short polishing phase that performs cheap random pair‑wise mass
    transfers (hill‑climbing) and occasional jitter until the time budget
    is exhausted.

All helper routines are defined at top level; no closures or lambdas are
used.  The algorithm respects the 1000 s wall‑clock limit and always returns
the best sequence discovered.

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level linear‑algebra utilities
# ----------------------------------------------------------------------


def fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors using an FFT of power‑of‑2 length."""
    n, m = a.shape[0], b.shape[0]
    L = 1 << ((n + m - 1).bit_length())           # next power of two
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)
    return conv[: n + m - 1]


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (length 2·len(a)‑1)."""
    return fft_convolve(a, a)


def project_to_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x ≥ 0, Σ x = target}.
    """
    if target <= 0:
        return np.full_like(v, 0.0)

    # Duchi et al. projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - target) / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all components are zero – return uniform distribution
        return np.full_like(v, target / len(v))
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # re‑normalise to guard against rounding errors
    s = w.sum()
    if s > 0:
        w *= target / s
    else:
        w = np.full_like(v, target / len(v))
    return w

# Shape generators (simple analytic families)

def generate_U_shape(
    n: int,
    target_sum: float,
    exponent: float = 1.0,
) -> np.ndarray:
    """
    Symmetric “U’’ shape: a_i ∝ |i‑mid|^exponent,
    centred at the middle, normalised to have ℓ₁‑norm = target_sum.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                # ∈[0,1]
    raw = np.power(d, exponent)
    # avoid the pathological all‑zeros case
    raw = raw + 1e-12
    return project_to_simplex(raw, target_sum)


def generate_linear_shape(
    n: int,
    target_sum: float,
    param: float = 0.0,
    direction: str = "symmetric",
    kind: str = "linear",
) -> np.ndarray:
    """
    Original family from the reference implementation
    (linear / quadratic / exponential, symmetric/decreasing/increasing).
    """
    idx = np.arange(n, dtype=np.float64)
    if direction == "symmetric":
        mid = (n - 1) / 2.0
        d = np.abs(idx - mid) / mid
        base = d
    elif direction == "decreasing":
        base = 1.0 - idx / (n - 1)
    elif direction == "increasing":
        base = idx / (n - 1)
    else:
        raise ValueError("unknown direction")

    if kind == "linear":
        raw = 1.0 + param * base
    elif kind == "quadratic":
        raw = 1.0 + param * base ** 2
    elif kind == "exponential":
        raw = np.exp(param * base)
    else:
        raise ValueError("unknown kind")
    raw = np.clip(raw, 0.0, None)
    return project_to_simplex(raw, target_sum)

# Small random perturbations used in later phases
# ----------------------------------------------------------------------


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """Transfer a fraction of mass from a random source index to a random target."""
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0.0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_to_simplex(a, target_sum)


def jitter(a: np.ndarray, target_sum: float, sigma: float = 1e-3) -> np.ndarray:
    """Add Gaussian noise, clip, and renormalise."""
    a = a + np.random.randn(a.shape[0]) * sigma
    a = np.clip(a, 0.0, None)
    return project_to_simplex(a, target_sum)
