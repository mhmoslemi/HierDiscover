    n = x.shape[0]
    size = 1 << ((2*n-1) -1).bit_length()
    X = np.fft.rfft(x, size)
    conv = np.fft.irfft(X*X, size)[:2*n-1]  # real conv

    shifted = alpha * (conv - maxv)   # shift so max becomes zero

"""
Propose a non‑negative step sequence that minimises the evaluation function.

The algorithm works in three stages inside the 1000 s wall‑clock budget.

1️⃣  **Broad parametric sweep** – families of analytic shapes (exponential,
    power‑law, mixed power‑law/exponential and Beta‑like) are sampled for a
    wide range of parameters and lengths.  Every candidate is normalised to
    ℓ¹‑norm = 1 (the objective is scale‑free) and its exact score is obtained
    by an FFT based convolution (O(n log n)).  The best candidate so far is
    retained.

2️⃣  **Smooth max optimisation** – the non‑differentiable term
    ``max(conv)`` is replaced by a smooth log‑sum‑exp surrogate:
        fα(x) = (2·n/α)·log Σ_k exp(α·(x*x)_k) .
    Its gradient can be written analytically as a cross‑correlation between a
    soft‑max weight vector and the current distribution.  Using FFT this
    gradient is computed in O(n log n).  An Adam optimiser with a slowly
    increasing smoothing parameter α drives the distribution towards a flat
    auto‑convolution while we stay on the probability simplex via the Euclidean
    simplex projection.

3️⃣  **Length growth & restarts** – every few thousand iterations a block of
    tiny entries is appended (allowing the mass to spread further) and,
    if no improvement is observed for a long stretch, a fresh random Dirichlet
    start is tried.  The best sequence throughout the run is finally returned.

The code contains no closures, no lambda functions and all helpers are top‑level
as required by the challenge framework.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex {x≥0, Σx=1}."""
    if v.ndim != 1:
        raise ValueError("simplex projection only works on 1‑D arrays")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Convolution via FFT (real‑valued inputs).  Returns full convolution."""
    n = a.shape[0] + b.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    C = A * B
    conv = np.fft.irfft(C, size)[:n]
    return conv


def _autoconv_max(x: np.ndarray) -> float:
    """Maximum entry of the auto‑convolution x*x (using FFT)."""
    n = x.shape[0]
    conv = _fft_convolve(x, x)                 # length 2·n‑1
    return float(np.max(conv))


def _fast_score(x: np.ndarray) -> float:
    """Exact score 2·n·max(x*x) for a normalised (ℓ¹‑norm = 1) vector."""
    total = x.sum()
    if total < 0.01:
        return float('inf')
    x = x / total
    n = x.shape[0]
    maxb = _autoconv_max(x)
    return 2.0 * n * maxb


# ----------------------------------------------------------------------
# Families of analytic shapes
# ----------------------------------------------------------------------


def _generate_powerlaw(p: float, sigma: float,
                       max_len: int = 8000, eps: float = 1e-12) -> np.ndarray:
    """a_i = (i+σ)^{-p}, truncated when a_i < eps."""
    idx = np.arange(max_len, dtype=np.float64)
    a = (idx + sigma) ** (-p)
    return a[a > eps]


def _generate_exponential(lam: float,
                          max_len: int = 8000, eps: float = 1e-12) -> np.ndarray:
    """a_i = exp(-λ·i), truncated when a_i < eps."""
    idx = np.arange(max_len, dtype=np.float64)
    a = np.exp(-lam * idx)
    return a[a > eps]


def _generate_mixed(p: float, sigma: float, lam: float,
                    max_len: int = 8000, eps: float = 1e-12) -> np.ndarray:
    """a_i = (i+σ)^{-p}·exp(-λ·i), truncated when a_i < eps."""
    idx = np.arange(max_len, dtype=np.float64)
    a = (idx + sigma) ** (-p) * np.exp(-lam * idx)
    return a[a > eps]


def _generate_beta(a: float, b: float, n: int) -> np.ndarray:
    """Discretised Beta(a,b) on {0,…,n‑1} (mid‑point rule)."""
    t = (np.arange(n, dtype=np.float64) + 0.5) / n
    vals = (t ** (a - 1)) * ((1 - t) ** (b - 1))
    return vals


# ----------------------------------------------------------------------
# Smooth‑max gradient (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def _smooth_gradient(x: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of  (2·n/α)·log Σ_k exp(α·(x*x)_k)  w.r.t. x.
    Implements the formula   g = 2·(w ⋆ x)   where w is the soft‑max over the
    auto‑convolution entries.
    """
    n = x.shape[0]

    # auto‑convolution
    conv = _fft_convolve(x, x)                     # length 2·n‑1

    # soft‑max weights (stable)
    maxv = np.max(alpha * conv)
    w_unnorm = np.exp(alpha * conv - maxv)
    w = w_unnorm / np.sum(w_unnorm)                # Σ w = 1

    # cross‑correlation w ⋆ x  =  (w * reverse(x))
    x_rev = x[::-1]
    corr_full = _fft_convolve(w, x_rev)            # length (2·n‑1)+n‑1 = 3·n‑2
    grad = 2.0 * corr_full[(n - 1):(n - 1 + n)]    # extract n entries
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def _adam_optimize(start: np.ndarray,
                   deadline: float,
                   best_score_ref: list,
                   best_seq_ref: list,
                   target: float = 1.503) -> None:
    """
    Refines ``start`` using Adam + simplex projection.
    Updates ``best_score_ref[0]`` and ``best_seq_ref[0]`` in‑place.
    """
    x = start.copy()
    n = x.shape[0]

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # scheduler for the smoothing parameter α
    alpha_start, alpha_end = 5.0, 200.0
    max_iters = 20000   # theoretical cap; we stop earlier by deadline

    # length‑growth schedule
    grow_interval = 3000
    extra_block = 200               # number of tiny entries to append

    last_improve = 0

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # linearly increase α
        progress = min(iteration / max_iters, 1.0)
        alpha = alpha_start + (alpha_end - alpha_start) * progress

        # gradient of the smooth‑max surrogate
        grad = _smooth_gradient(x, alpha)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # a modest learning‑rate schedule
        step = 0.5 / (np.sqrt(iteration) + 1.0)

        x = x - step * m_hat / (np.sqrt(v_hat) + eps)

        # keep on the simplex
        x = _simplex_projection(x)

        # occasional growth of the vector (adds extra tiny mass)
        if iteration % grow_interval == 0 and x.shape[0] < 8000:
            extra = np.full(extra_block, 1e-12, dtype=float)
            x = np.concatenate([x, extra])
            x = x / x.sum()
            # extend Adam moments with zeros
            m = np.concatenate([m, np.zeros_like(extra)])
            v = np.concatenate([v, np.zeros_like(extra)])

        # compute fast score (exact objective)
        cur_score = _fast_score(x)

        if cur_score < best_score_ref[0] - 1e-12:
            best_score_ref[0] = cur_score
            best_seq_ref[0] = x.copy()
            last_improve = iteration
            print(f'iter {iteration:6d} → new best score = {cur_score:.6f}')
            if cur_score < target:
                print('Target bound reached – early exit.')
                return

        # restart if stagnating for a long time
        if iteration - last_improve > 6000:
            # fresh Dirichlet start with same length
            rnd = np.random.rand(x.shape[0])
            rnd = rnd / rnd.sum()
            x = 0.6 * x + 0.4 * rnd
            x = x / x.sum()
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iteration
            print(f'iter {iteration:6d} – restart due to stagnation.')

    # finished because of deadline
    return


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list:
    """
    Search for a non‑negative step sequence that minimises ``evaluate_sequence``.
    Returns the best sequence discovered before the 1000 s wall‑clock limit.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0        # safety margin

    best_score = float('inf')
    best_seq = None

    # ------------------------------------------------------------------
    # 0.  Global seed (if supplied by the outer harness)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():

            seq0 = np.asarray(height_sequence_1, dtype=float)
            cur = _fast_score(seq0)
            if np.isfinite(cur) and cur < best_score:
                best_score = cur
                best_seq = seq0.copy()
                print(f'Initial global sequence score = {best_score:.6f}')
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Broad parametric sweep (fast FFT‑based evaluation)
    # ------------------------------------------------------------------
    print('=== Phase 1 – parametric sweep ===')
    max_len = 8000
    eps = 1e-12

    # power‑law family
    p_vals = np.linspace(0.2, 0.9, 35)
    sigma_vals = np.logspace(-6, 2, 45)
    for p in p_vals:
        for sigma in sigma_vals:
            if time.time() >= deadline:
                break
            seq = _generate_powerlaw(p, sigma, max_len, eps)
            if seq.size < 2:
                continue
            score = _fast_score(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f'new PL   p={p:.3f} σ={sigma:.3e} len={len(seq)} → {best_score:.6f}')
        if time.time() >= deadline:
            break

    # exponential family
    lam_vals = np.logspace(-5, 0, 40)          # from 1e‑5 to 1
    for lam in lam_vals:
        if time.time() >= deadline:
            break
        seq = _generate_exponential(lam, max_len, eps)
        if seq.size < 2:
            continue
        score = _fast_score(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f'new EXP  λ={lam:.3e} len={len(seq)} → {best_score:.6f}')

    # mixed power‑law / exponential
    for p in (0.4, 0.5, 0.6):
        for sigma in (1e-4, 1e-2, 1e-1, 1.0):
            for lam in (1e-4, 5e-4, 1e-3, 5e-3):
                if time.time() >= deadline:
                    break
                seq = _generate_mixed(p, sigma, lam, max_len, eps)
                if seq.size < 2:
                    continue
                score = _fast_score(seq)
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    print(f'new MIX p={p:.2f} σ={sigma:.1e} λ={lam:.1e} len={len(seq)} → {best_score:.6f}')

    # Beta‑like family (discretised Beta distribution)
    beta_ns = [2000, 3000, 4000, 5000, 6000, 8000]
    for n in beta_ns:
        for _ in range(20):                      # 20 random (a,b) per length
            if time.time() >= deadline:
                break
            a = 10 ** np.random.uniform(-1, 1)   # between 0.1 and 10
            b = 10 ** np.random.uniform(-1, 1)
            seq = _generate_beta(a, b, n)
            if seq.sum() < 1e-12:
                continue
            score = _fast_score(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f'new BETA a={a:.3f} b={b:.3f} n={n} → {best_score:.6f}')

    # Random Dirichlet draws (for extra diversity)
    for n in (2000, 3000, 4000, 5000, 6000, 8000):
        for _ in range(15):
            if time.time() >= deadline:
                break
            alpha = np.full(n, 0.5)               # moderate concentration
            dirichlet = np.random.gamma(alpha, 1.0)
            seq = dirichlet / dirichlet.sum()
            score = _fast_score(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f'new DIR n={n} → {best_score:.6f}')

    # ------------------------------------------------------------------
    # 2.  Local optimisation (smooth‑max + Adam)
    # ------------------------------------------------------------------
    if best_seq is None:

        n0 = 4000
        best_seq = np.full(n0, 1.0 / n0, dtype=float)
        best_score = _fast_score(best_seq)
        print('fallback to uniform sequence, score =', best_score)

    # normalise to ℓ¹‑norm = 1 (required by the optimiser)
    best_seq = best_seq / best_seq.sum()

    # mutable containers to allow in‑place updates from the optimiser
    best_score_ref = [best_score]
    best_seq_ref = [best_seq.copy()]

    print('=== Phase 2 – smooth‑max Adam optimisation ===')
    _adam_optimize(best_seq, deadline, best_score_ref, best_seq_ref,
                   target=1.503)

    final_best_seq = best_seq_ref[0]
    final_best_score = best_score_ref[0]

    print('final best score:', final_best_score,
          'length:', final_best_seq.shape[0])

    # Convert to plain Python list as required by the external checker
    return final_best_seq.tolist()
