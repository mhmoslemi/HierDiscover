"""
Adam‑based smoothed max‑convolution minimisation.

The evaluation metric for a non‑negative vector `a` of length *n* is

    2·n·max(conv(a,a)) / (sum(a)²)

If we enforce the ℓ₁‑normalisation  ∑a = √(2n)  the denominator equals
2·n, so the metric becomes exactly the maximal entry of the self‑convolution.
The algorithm works in three phases:

1. **Coarse parametric sweep** (linear / quadratic / exponential “U‑shapes”)
   to obtain a decent start point.

2. **Adam optimiser on a smooth surrogate** of the max.  The smooth
   surrogate is `τ·log(∑exp(conv/τ))`.  Its gradient w.r.t. the vector can be
   written as a convolution, which we evaluate with FFTs.  Positivity and the
   ℓ₁‑constraint are enforced by representing the vector as a soft‑max of an
   unrestricted log‑weight vector; Adam updates the log‑weights.

3. **Final polishing** – tiny random pair‑wise mass transfers are tried;
   any improvement is kept.

The whole procedure respects a 1000 s time budget and always returns the
best sequence seen so far.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself using a power‑of‑two FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())        # next power of two ≥ 2n‑1
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum x = target}."""
    if target <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback to uniform distribution
        return np.full_like(v, target / len(v))

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def generate_U_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
) -> np.ndarray:
    """
    Produce a symmetric “U‑shaped’’ sequence of length n.
    kind ∈ {linear, quadratic, exponential}.
    param ≥ 0 controls edge enhancement.
    The result is scaled to have ℓ₁‑norm `target_sum`.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def generate_gaussian_shape(
    n: int,
    target_sum: float,
    sigma_factor: float = 0.2,
) -> np.ndarray:
    """Symmetric Gaussian bump; sigma = sigma_factor·(n‑1)/2."""
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    sigma = sigma_factor * mid + 1e-12
    raw = np.exp(-0.5 * ((idx - mid) / sigma) ** 2)
    return project_onto_simplex(raw, target_sum)


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a tiny amount of mass from a random index to another
    and project back onto the ℓ₁‑simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    shift = logits - np.max(logits)
    exp_shift = np.exp(shift)
    return exp_shift / exp_shift.sum()


def smoothed_gradient(
    a: np.ndarray,
    tau: float,
) -> tuple[float, np.ndarray]:
    """
    Return (max_conv, grad_a) where grad_a is the gradient of the
    log‑sum‑exp smoothed max w.r.t. a.
    """
    n = a.shape[0]
    conv = conv_self_fft(a)                 # length 2n‑1
    max_conv = conv.max()

    # log‑sum‑exp weights (stable)
    w = np.exp((conv - max_conv) / tau)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient: (2/τ) * (w * a_rev) restricted to the n entries that touch a
    a_rev = a[::-1]
    L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
    grad_a = (2.0 / tau) * conv_full[n - 1 : n - 1 + n]
    return max_conv, grad_a
