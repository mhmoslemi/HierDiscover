    n = a.shape[0]
    conv_len = 2*n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)

def _gradient_softmax(a, conv, alpha):
    # conv = a*a (full)
    maxc = conv.max()
    w = np.exp(alpha * (conv - maxc))
    # Weighted convolution w*a
    conv1 = _conv_weighted_fft(w, a)  # length 3n-2
    conv2 = _conv_weighted_fft(w[::-1], a)  # same length
    L = conv.shape[0] # =2n-1
    # pick first n entries of conv1
    term1 = conv1[:a.shape[0]]          # shape n
    # term2: index L-1-j -> we extract conv2[L-1 - np.arange(n)]
    term2 = conv2[L-1 - np.arange(a.shape[0])]
    g = 2.0 * (term1 + term2)
    return g
```

def _optimize(a0, n, deadline, max_iters=5000):
    a = a0.copy()
    S = np.sqrt(2.0 * n)
    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5
    alpha = 0.1
    iteration = 0
    best_local = np.inf
    best_seq_local = a.copy()
    while time.time() < deadline and iteration < max_iters:
        iteration += 1
        conv = _conv_fft(a)
        cur_max = conv.max()
        if cur_max < best_local - 1e-12:
            best_local = cur_max
            best_seq_local = a.copy()
            # maybe evaluate external to ensure
            # but we can decide later.
        # gradient
        g = _gradient_softmax(a, conv, alpha)
        # Adam update (additive)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g**2)
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step   # subtract because we want to minimize max (conv) which gradient points direction to increase conv, I think? Actually gradient of max surrogate is positive direction; we need to move a in opposite direction of gradient to reduce conv. Yes subtract.
        # ensure nonnegative
        a = np.clip(a, 0.0, None)
        # renormalize to sum S
        total = a.sum()
        if total == 0:
            a.fill(S / n)
        else:
            a *= (S / total)
        # schedule
        # increase alpha gradually
        alpha = min(10.0, alpha * 1.005)   # escalate from 0.1 to around 10
        # decay lr after some steps
        if iteration % 1000 == 0:
            lr *= 0.9
    return best_seq_local, best_local

"""
Improved search for a step‑function with minimal self‑convolution peak.

The algorithm is a multi‑restart, soft‑max‑guided exponentiated‑gradient
optimiser:

* Choose a length n and initialise a non‑negative vector a (random or a
  decreasing shape).  Normalise it so that Σ a = √(2 n); with this
  normalisation the external evaluation equals the maximal entry of the
  linear self‑convolution (the quantity we want to minimise).

* In each iteration compute the full self‑convolution c = a*a via FFT.
  From c build a soft‑max weight vector w_i = exp(α·(c_i – max(c))), where
  the temperature α starts small and is slowly increased.  The weighted
  gradient of the (smoothed) maximum is
        g = 2·(w ⋆ a)      (cross‑correlation)
  which can be evaluated with a single FFT convolution of w and the
  reversed a.

* Apply momentum (exponential moving average) to the gradient and update
  a multiplicatively (exponentiated gradient):
        a ← a·exp(−η·v) .
  Renormalise a to keep Σ a = √(2 n) and keep it non‑negative.

* Keep the best vector seen (according to max(c)).  The routine runs many
  independent restarts with different lengths until the time budget (≈
  1000 s) is exhausted, returning the best sequence found.

The code uses only NumPy and the supplied `evaluate_sequence` for the
optional baseline/fallback check.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolutions
# ----------------------------------------------------------------------


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two 1‑D real arrays using the FFT.
    Returns the full convolution of length len(x)+len(y)-1.
    """
    n = x.shape[0]
    m = y.shape[0]
    conv_len = n + m - 1
    fft_len = 1 << (conv_len - 1).bit_length()      # next power of two
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[:conv_len]
    return conv


def _self_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear self‑convolution a*a."""
    return _fft_convolve(a, a)


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the weighted max surrogate  Σ_i w_i·(a*a)[i]  with respect
    to a.  The formula reduces to  g = 2·(w ⋆ a)  (cross‑correlation).

    It is evaluated as a convolution of w with the reversed a:
        (w ⋆ a)[j] = (w * a_rev)[j + n - 1] .
    """
    n = a.shape[0]
    a_rev = a[::-1]
    conv = _fft_convolve(w, a_rev)                 # length = (2n‑1)+n‑1 = 3n‑2
    start = n - 1
    grad = 2.0 * conv[start:start + n]             # slice of length n
    return grad


# ----------------------------------------------------------------------
# Optimisation of a single restart
# ----------------------------------------------------------------------


def _optimize_one_restart(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 20000,
                          eta0: float = 0.5,
                          beta: float = 0.9,
                          alpha0: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent with momentum, guided by a soft‑max
    approximation of the maximal convolution entry.

    Returns the best vector found and its corresponding max‑convolution
    value (which equals the external score after normalisation).
    """
    a = a0.copy()
    S = np.sqrt(2.0 * n)               # target L1‑norm
    best_a = a.copy()
    best_val = float('inf')
    v = np.zeros_like(a)               # momentum buffer
    eta = eta0
    alpha = alpha0                     # temperature for the soft‑max
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # --- self‑convolution and current objective ---
        conv = _self_convolution(a)
        cur_max = float(conv.max())

        # record improvement
        if cur_max < best_val - 1e-12:
            best_val = cur_max
            best_a = a.copy()
            print(f"[local improve] t={time.time():.1f}s  n={n} iter={it} max={cur_max:.6f}")
            if best_val < 1.503:                 # target reached – stop early
                break

        # --- soft‑max weights (smooth max) ---
        w = np.exp(alpha * (conv - cur_max))      # w_i ∈ (0,1]

        # --- gradient of the smoothed objective ---
        g = _gradient_from_weights(a, w)

        # momentum update (exponential moving average)
        v = beta * v + (1.0 - beta) * g

        # exponentiated (multiplicative) update
        a = a * np.exp(-eta * v)

        # enforce non‑negativity and renormalise to Σa = √(2 n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a.fill(S / n)
        else:
            a *= (S / total)

        # occasional jitter to escape tiny plateaus
        if np.random.rand() < 0.01:
            noise = np.random.randn(n) * a.mean() * 0.02
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()

        # schedule: increase soft‑max temperature slowly, decay step size
        alpha = min(10.0, alpha * 1.005)
        if it % 2000 == 0:
            eta *= 0.9

    return best_a, best_val
