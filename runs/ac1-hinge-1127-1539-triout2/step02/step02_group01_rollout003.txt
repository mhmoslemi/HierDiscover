"""
Propose a non‑negative step‑function that (approximately) minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

where conv is the linear self‑convolution.  

The algorithm works in three stages while respecting the 1000 s budget:

1. **Parametric sweep** – a handful of symmetric “U‑shapes’’ (linear,
   quadratic, exponential) and a random Dirichlet sample are projected
   onto the ℓ₁‑simplex (sum = √(2 n)).  The best among them seeds the
   optimiser.

2. **Projected Adam on a top‑K sub‑gradient** – in each iteration we
   compute the full self‑convolution with an FFT.  The indices with the
   largest convolution values (a small top‑K% of all 2 n‑1 entries) are
   given equal weight, yielding a (sub‑)gradient of the objective
   max(conv).  An Adam‑style update is taken, after which the vector is
   projected back onto the simplex.  The top‑K percentage decays
   gradually, focusing the search on the true maximum.

3. **Polishing** – a short final pass of tiny random mass transfers
   (pairwise transfers) is performed; any improvement is kept.

All heavy work (convolutions, FFTs) is O(n log n) and fits comfortably
within the time limit for n≈8192 (a power of two, which makes FFTs fast).
The routine returns the best sequence seen so far.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex
        {x | x_i ≥ 0, sum_i x_i = target}
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback: uniform distribution
        return np.full_like(v, target / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def generate_U_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
) -> np.ndarray:
    """
    Produce a symmetric “U‑shaped’’ sequence of length n.
    kind ∈ {linear, quadratic, exponential}.
    param ≥ 0 controls the edge reinforcement.
    The result is projected onto the ℓ₁‑simplex with total sum = target_sum.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def random_dirichlet(n: int, target_sum: float) -> np.ndarray:
    """
    Sample a random point on the simplex (Gamma(1,1) → Dirichlet(1,…,1))
    and scale it to have ℓ₁‑norm = target_sum.
    """
    g = np.random.gamma(1.0, 1.0, size=n)
    return project_onto_simplex(g, target_sum)


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a tiny amount of mass from one random index to another,
    then project back onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution of a with itself using an FFT of length L.
    Returns the first (2*len(a)-1) entries.
    """
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)
    return conv[: 2 * a.shape[0] - 1]


def gradient_topk(a: np.ndarray, top_k: int, L: int) -> np.ndarray:
    """
    Compute the (sub‑)gradient of max(conv(a,a)) using a uniform weight
    on the top_k largest convolution entries.
    Returns a gradient vector of length len(a).
    """
    n = a.shape[0]
    # Full self‑convolution
    conv = conv_self_fft(a, L)
    # Build weight vector w (length 2n-1)
    w = np.zeros_like(conv)
    if top_k >= conv.size:
        w[:] = 1.0 / conv.size
    else:
        # indices of the top_k entries (unsorted)
        top_idx = np.argpartition(-conv, top_k)[:top_k]
        w[top_idx] = 1.0 / top_k
    # Convolution of w with reversed a
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
    # Slice to obtain gradient (see derivation in analysis)
    grad = 2.0 * full[n - 1 : n - 1 + n]
    return grad, conv.max()


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    `evaluate_sequence`.  The routine respects a time budget (≈1000 s)
    and returns the best sequence encountered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Determine problem size (use provided baseline if present)
    # --------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        best_seq = np.array(base, dtype=np.float64)
    else:
        # Choose a power‑of‑two length – fast FFT and enough degrees of freedom
        n = 8192
        best_seq = None

    target_sum = np.sqrt(2.0 * n)   # conventional normalisation

    # --------------------------------------------------------------
    # Phase 1 – cheap parametric sweep / random start
    # --------------------------------------------------------------
    best_score = float("inf")
    candidates = []

    # uniform start (fallback)
    candidates.append(np.full(n, target_sum / n))

    # U‑shapes
    kinds = ("linear", "quadratic", "exponential")
    param_grid = {
        "linear": np.linspace(0.0, 5.0, 26),        # step 0.2
        "quadratic": np.linspace(0.0, 3.0, 16),
        "exponential": np.linspace(0.0, 2.5, 13),
    }
    for kind in kinds:
        for p in param_grid[kind]:
            cand = generate_U_shape(n, target_sum, kind=kind, param=p)
            candidates.append(cand)

    # a random Dirichlet point
    candidates.append(random_dirichlet(n, target_sum))

    # evaluate sweep candidates
    print(f"[{time.time():.2f}] Phase 1 – parametric sweep (n={n})")
    for cand in candidates:
        score = evaluate_sequence(cand.tolist())
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"  new best (score={best_score:.6f})")
        if time.time() >= deadline:
            print("Time exhausted during Phase 1.")
            return best_seq.tolist()

    # --------------------------------------------------------------
    # Phase 2 – Projected Adam with top‑K sub‑gradient
    # --------------------------------------------------------------
    # FFT length – must be ≥ 2n‑1 + n‑1 = 3n‑2; pick next power of two.
    L = 1 << ((3 * n - 2).bit_length())
    a = best_seq.copy()
    m = np.zeros_like(a)  # first moment
    v = np.zeros_like(a)  # second moment

    # Adam hyper‑parameters
    alpha0 = 0.45                     # initial learning rate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # Top‑K schedule (percentage of 2n‑1 entries)
    top_percent_start = 0.025   # 2.5 % initially
    top_percent_end = 0.004     # 0.4 % at the end
    decay_iters = 250000        # over how many iterations we linearly decay

    iteration = 0
    no_improve = 0
    best_iter = 0

    print(f"[{time.time():.2f}] Phase 2 – projected Adam")
    while time.time() < deadline:
        iteration += 1
        # determine current top‑K size
        cur_percent = max(top_percent_end,
                          top_percent_start * (1.0 - iteration / decay_iters))
        top_k = max(1, int(cur_percent * (2 * n - 1)))

        # gradient and current max‑conv
        grad, cur_max = gradient_topk(a, top_k, L)

        # Adam statistics update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # learning‑rate schedule (slow decay)
        alpha = alpha0 * (0.995 ** (iteration // 1000))

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # projection onto the simplex (keeps non‑negativity & sum)
        a = project_onto_simplex(np.maximum(a, 0.0), target_sum)

        # keep the best observed solution
        if cur_max + 1e-12 < best_score:
            best_score = cur_max
            best_seq = a.copy()
            best_iter = iteration
            no_improve = 0
            print(f"Iter {iteration:06d} – improved max_conv={best_score:.6f}")
        else:
            no_improve += 1

        # occasional random pair transfer to escape plateaus
        if np.random.rand() < 0.012:
            a = random_pair_transfer(a, target_sum, scale=0.006)

        # heavy jitter after long stagnation
        if no_improve >= 5000:
            jitter = np.random.randn(n) * 2e-4
            a = project_onto_simplex(np.maximum(a + jitter, 0.0), target_sum)
            no_improve = 0
            print(f"Iter {iteration:06d} – jitter applied")

        # safety break (in case we overshoot time)
        if time.time() >= deadline - 5.0:   # keep a couple seconds for polishing
            break

    # --------------------------------------------------------------
    # Phase 3 – final polishing with tiny pairwise transfers
    # --------------------------------------------------------------
    print(f"[{time.time():.2f}] Phase 2 finished (best score {best_score:.6f})")
    for _ in range(45):
        if time.time() >= deadline:
            break
        cand = random_pair_transfer(best_seq.copy(), target_sum, scale=0.004)
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"Polish improvement → {best_score:.6f}")

    print(f"[{time.time():.2f}] Returning best sequence (score {best_score:.6f})")
    return best_seq.tolist()
