"""
Propose a non‑negative step‑function minimizing

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three phases:

1. **Coarse parametric sweep** over simple symmetric “U‑shapes’’
   (linear, quadratic, exponential).  The best shape is kept as a warm‑start.

2. **Smoothed Adam optimisation** – the nonsmooth max is replaced by a
   log‑sum‑exp (temperature τ).  The gradient of the smoothed objective is
   obtained analytically via a convolution.  At each iteration an Adam step
   is taken and the vector is projected back onto the ℓ₁‑simplex.
   The iterate is forced to stay symmetric (a[i] = a[n‑1‑i]) which reduces
   the search space and improves the bound.  The temperature τ and learning
   rate are gradually decreased; occasional random jitter helps escaping
   plateaus.

3. **Final polishing** – a short burst of tiny random pair‑wise mass
   transfers is performed; any improvement is kept.

The whole search respects a 1000 s budget and always returns the best
sequence seen so far.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of `a` with itself using a power‑of‑two FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of `x` and `y` via FFT.
    Returns the full convolution of length len(x)+len(y)-1.
    """
    L = 1 << ((len(x) + len(y) - 1).bit_length())
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)
    return conv[: len(x) + len(y) - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex
        {x | x_i ≥ 0, Σ_i x_i = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries would become zero – fallback to uniform distribution
        return np.full_like(v, target / len(v))

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def generate_U_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
) -> np.ndarray:
    """
    Produce a symmetric “U‑shaped’’ sequence of length n.
    kind ∈ {linear, quadratic, exponential}.
    param ≥ 0 controls the edge enhancement.
    The result is projected onto the simplex of ℓ₁‑norm `target_sum`.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def random_sym_init(n: int, target_sum: float) -> np.ndarray:
    """
    Random symmetric non‑negative vector of length n,
    projected onto the ℓ₁‑simplex.
    """
    half = (n + 1) // 2
    r = np.random.rand(half)
    if n % 2 == 0:
        full = np.concatenate([r, r[::-1]])
    else:
        # odd length – mirror everything except the middle element
        full = np.concatenate([r, r[-2::-1]])
    return project_onto_simplex(full, target_sum)


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a tiny amount of mass from a random index to another,
    then project back onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    a = np.maximum(a, 0.0)
    return project_onto_simplex(a, target_sum)


def smooth_gradient_step(
    a: np.ndarray,
    target_sum: float,
    eta: float,
    tau_factor: float,
) -> tuple[np.ndarray, float]:
    """
    One gradient step of the smoothed (log‑sum‑exp) objective.
    Returns the gradient vector and the current maximum convolution value.
    """
    # 1) self‑convolution and its maximum
    conv = conv_self_fft(a)                      # length 2n‑1
    max_conv = conv.max()

    # 2) smoothing weights (stable log‑sum‑exp)
    tau = max(tau_factor * max_conv, 1e-15)
    w_raw = np.exp((conv - max_conv) / tau)
    w_sum = w_raw.sum()
    if w_sum == 0:
        w = np.full_like(w_raw, 1.0 / w_raw.size)
    else:
        w = w_raw / w_sum

    # 3) gradient = (2/τ) * (w * a_rev) restricted to the n coordinates
    a_rev = a[::-1]
    conv_full = conv_fft(w, a_rev)               # length 3n‑2
    n = a.shape[0]
    grad = (2.0 / tau) * conv_full[n - 1 : n - 1 + n]
    return grad, max_conv


def adam_update(
    a: np.ndarray,
    grad: np.ndarray,
    m: np.ndarray,
    v: np.ndarray,
    t: int,
    lr: float,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """
    Perform an Adam update step on the (unconstrained) vector `a`.
    Returns the new vector and the updated moment estimates.
    """
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad ** 2)

    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = a - step
    return a_new, m, v

                best_score = score
                best_seq = cand.copy()
                print(f"  New best {kind} (p={p:.2f}): {best_score:.6f}")
            if time.time() >= deadline:
                print("Time exhausted during parametric sweep.")
                return best_seq.tolist()

    # ------------------------------------------------------------------
    # Phase 2 – smoothed Adam optimisation (multiple restarts)
    # ------------------------------------------------------------------
    max_attempts = 4
    attempt = 0
    while time.time() < deadline and attempt < max_attempts:
        attempt += 1
        # ------------------------------------------------------------------
        # Initialise – first attempt uses the best from sweep, others random
        # ------------------------------------------------------------------
        if attempt == 1:
            a = best_seq.copy()
        else:
            a = random_sym_init(n, target_sum)

        # Adam state
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        t_iter = 0

        # hyper‑parameters (will be annealed)
        eta = 0.5 * (target_sum / n)   # base learning rate
        tau_factor = 0.03
        no_improve = 0

        print(f"[{time.time():.2f}] Attempt {attempt}/{max_attempts} – start optimisation")

        while time.time() < deadline:
            t_iter += 1

            # gradient of the smoothed objective
            grad, max_conv = smooth_gradient_step(a, target_sum, eta, tau_factor)

            # Adam update (unconstrained)
            a, m, v = adam_update(a, grad, m, v, t_iter, lr=eta)

            # enforce non‑negativity, symmetry and ℓ₁‑norm
            a = np.maximum(a, 0.0)
            a = (a + a[::-1]) * 0.5
            a = project_onto_simplex(a, target_sum)

            # current objective value (sum = target_sum ⇒ value = max_conv)
            cur_score = max_conv
            if cur_score + 1e-12 < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"  Attempt {attempt} iter {t_iter:06d} – improved to {best_score:.6f}")
                no_improve = 0
            else:
                no_improve += 1

            # schedule adjustments
            if t_iter % 1000 == 0:
                eta *= 0.97
                tau_factor = max(1e-4, tau_factor * 0.98)

            # occasional random jitter to escape plateaus
            if np.random.rand() < 0.01:
                jitter = np.random.randn(n) * 0.001 * target_sum / n
                a = project_onto_simplex(np.maximum(a + jitter, 0.0), target_sum)
                a = (a + a[::-1]) * 0.5

            # stop this attempt if we have been stagnant for a long time
            if no_improve >= 50000:
                print(f"  Attempt {attempt} stagnated after {t_iter} iterations.")
                break

    print(f"[{time.time():.2f}] Optimisation finished – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final polishing with tiny random transfers
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Starting final polishing")
    for _ in range(200):
        if time.time() >= deadline:
            break
        cand = random_pair_transfer(best_seq.copy(), target_sum, scale=0.004)
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"  Polishing improved to {best_score:.6f}")

    print(f"[{time.time():.2f}] Search completed – final score {best_score:.6f}")
    return best_seq.tolist()
