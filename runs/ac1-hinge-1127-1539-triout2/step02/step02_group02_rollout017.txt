"""
Propose a step‑function (non‑negative height sequence) that minimizes

    2 * n * max(convolution(a, a)) / (sum(a) ** 2)

where the external evaluator `evaluate_sequence` returns the same value
(after it normalises the L1‑norm).  The algorithm is a multi‑restart
Adam optimiser on a smooth soft‑max surrogate of the max‑convolution.
For each restart we

    * pick a length `n` (random from a small list)
    * initialise a probability vector with a random Dirichlet,
      power‑law or exponential distribution,
    * normalise it to the required L1‑norm `√(2 n)`,
    * repeatedly
        – compute the full self‑convolution by FFT,
        – build a soft‑max weighting of the convolution entries
          (parameter ε controls the smoothness),
        – obtain the exact gradient of the smooth surrogate by a second
          FFT (cross‑correlation of `a` with the soft‑max weights),
        – perform an Adam step, clip to non‑negative, and renormalise.
    * Whenever the true score (the maximal convolution entry) improves,
      store the sequence.

The optimisation runs until the time budget (≈ 1000 s) is exhausted.
The best sequence found over all restarts is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of appropriate length."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()      # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _softmax_weights(conv: np.ndarray, eps: float) -> np.ndarray:
    """Stable soft‑max of `conv / eps`.  Returns a probability vector of the same length."""
    # shift by the maximum to avoid overflow
    max_c = conv.max()
    shifted = (conv - max_c) / eps
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total == 0.0:               # numerical safety
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / total


def _gradient_from_weights_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Exact gradient of the smooth surrogate with respect to `a`.
    `w` must be the soft‑max weights (length 2n‑1).  Returns a vector of size n.
    """
    n = a.shape[0]
    L = w.shape[0]                # L = 2n‑1
    # correlation a ⋆ w  (g_i = 2 Σ_j a_j w_{i+j})
    w_rev = w[::-1]
    conv_len = n + L - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = (L - 1) - np.arange(n)          # indices giving Σ_j a_j w_{i+j}
    g = 2.0 * corr[idx]
    return g


def _initial_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random probability vector of length n.
    Several families (Dirichlet, power‑law, exponential, uniform) are mixed.
    """
    choice = rng.integers(0, 4)
    if choice == 0:                     # Dirichlet (nearly uniform)
        v = rng.dirichlet(np.ones(n))
    elif choice == 1:                   # Power‑law (heavy tail)
        alpha = rng.uniform(0.5, 2.0)
        idx = np.arange(1, n + 1)
        v = idx ** (-alpha)
        v = v / v.sum()
    elif choice == 2:                   # Exponential decay
        beta = rng.uniform(0.005, 0.05)
        idx = np.arange(n, dtype=float)
        v = np.exp(-beta * idx)
        v = v / v.sum()
    else:                               # Simple uniform random (non‑Dirichlet)
        v = rng.random(n)
        v = v / v.sum()
    return v


def _run_optimize_one_start(
    a0: np.ndarray,
    n: int,
    deadline: float,
    eps0: float = 0.5,
    eps_decay: float = 0.998,
    min_eps: float = 0.005,
    lr0: float = 0.05,
    lr_decay: float = 0.999,
    max_iters: int = 50000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max surrogate of the max‑convolution.
    Returns the best sequence seen during the run and its true score.
    """
    a = a0.copy()
    S = np.sqrt(2.0 * n)                         # required L1‑norm

    # Initialise Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    epsadam = 1e-8

    # initial true score (max of convolution)
    conv = _conv_full_fft(a)
    best_score = conv.max()
    best_seq = a.copy()

    eps = eps0
    lr = lr0
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # full convolution
        conv = _conv_full_fft(a)
        cur_max = conv.max()

        # soft‑max weighting of the convolution entries
        w = _softmax_weights(conv, eps)

        # gradient of the smooth surrogate
        g = _gradient_from_weights_fft(a, w)

        # Adam update (descent)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + epsadam)
        a -= step

        # keep non‑negative and renormalise to the fixed L1‑norm
        np.clip(a, 0.0, None, out=a)
        total = a.sum()
        if total == 0.0:
            a.fill(S / n)
        else:
            a *= (S / total)

        # occasional tiny random jitter (helps escape plateaus)
        if np.random.rand() < 0.012:
            jitter = np.random.randn(n) * a.mean() * 0.008
            a += jitter
            np.clip(a, 0.0, None, out=a)
            a *= (S / a.sum())

        # schedule
        eps = max(eps * eps_decay, min_eps)
        lr *= lr_decay

        # record improvement w.r.t. the *exact* objective
        if cur_max < best_score - 1e-14:
            best_score = cur_max
            best_seq = a.copy()
            print(
                f"[improve] t={time.time():.1f}s n={n} iter={it} "
                f"score={best_score:.6f} eps={eps:.5f}"
            )

    return best_seq, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑score step‑function within the allotted wall‑clock time.
    Returns a list of non‑negative heights (the candidate sequence).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # a few seconds safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Record any pre‑provided construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            if cand.ndim == 1 and cand.size > 0:
                sc = evaluate_sequence(cand.tolist())
                best_seq = cand.copy()
                best_score = sc
                print(f"[baseline] supplied height_sequence_1 score={best_score:.6f} len={cand.size}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Lengths to explore (the larger the length, the lower the theoretical bound)
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000, 20000]
    run_id = 0

    while time.time() < deadline:
        n = int(rng.choice(lengths))

        # initialise a probability vector
        if best_seq is not None and best_seq.size >= n and rng.random() < 0.35:
            # reuse the best known vector (truncated) as a warm start
            p0 = best_seq[:n].copy()
            p0 = p0 / p0.sum()
        else:
            p0 = _initial_vector(n, rng)

        # convert to the required L1‑norm √(2 n)
        S = np.sqrt(2.0 * n)
        a0 = p0 * S

        # run a single optimisation instance
        seq_opt, score_opt = _run_optimize_one_start(a0, n, deadline)

        # keep the best overall sequence
        if score_opt < best_score:
            best_score = score_opt
            best_seq = seq_opt.copy()
            print(f"[global improve] run={run_id} n={n} score={best_score:.6f}")

        run_id += 1

    # ------------------------------------------------------------------
    # Fallback (should never be necessary)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        a = np.full(n, np.sqrt(2.0 * n) / n)
        best_seq = a
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] score={best_score:.6f}")

    print(f"[final] best score = {best_score:.6f}   length = {len(best_seq)}")
    return best_seq.tolist()
