def _lp_optimize(a, M, target_sum):
    n = a.shape[0]
    rows = 2*n - 1
    # Build A
    A = _build_lp_matrix(a)   # using sparse
    c = -np.ones(n)
    b = np.full(rows, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    g = res.x
    sum_g = g.sum()
    if sum_g <= 0:
        return None
    # Scale to target sum
    g_scaled = g * (target_sum / sum_g)
    return g_scaled

def _guided_hill_climb(a, deadline):
    n = len(a)
    best_a = a.copy()
    best_val = _score(a)
    # certain number of attempts within deadline
    while time.time() < deadline:
        conv = np.convolve(a,a)
        k_max = np.argmax(conv)
        # compute sensitivity
        s = np.zeros(n)
        for i in range(n):
            j = k_max - i
            if 0 <= j < n:
                s[i] = a[j]
        # choose donor and receiver
        # donor: index with highest s* a_i? Perhaps weighted by a_i as well.
        # We'll define weight = s[i] * a[i] maybe.
        weight = s * a
        # pick donor index where weight is high
        donors = np.where(weight > np.percentile(weight, 75))[0]
        receivers = np.where(weight < np.percentile(weight, 25))[0]
        if donors.size == 0 or receivers.size ==0:
            # fall back to random
            i = np.random.randint(n)
            j = np.random.randint(n)
        else:
            i = np.random.choice(donors)
            j = np.random.choice(receivers)
        if a[i] < 1e-12:
            continue
        # move some fraction
        frac = np.random.rand() * 0.2
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta
        # evaluate
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
        else:
            # revert if not improved maybe keep? we could accept some uphill moves occasionally.
            # Accept if some probability? but maybe not needed.
            a[j] -= delta
            a[i] += delta
    return best_a, best_val

"""
A fast multi‑stage optimizer for the “max‑convolution / ℓ¹‑norm’’ benchmark.

Stage 1 – coarse grid search.
    For several lengths `n` we enumerate exponents `p` of a pure power‑law
    a_i = (i+1)^(-p) (the classic shape that gave ~1.52).  Each candidate is
    normalised to Σa = √(2·n) so that the evaluation function reduces to the
    maximal convolution entry.  The best few candidates per length are kept.

Stage 2 – annealed projected gradient descent.
    Starting from the short list we run the soft‑max annealing scheme from the
    reference implementation (gradient of a smooth approximation of max(conv)).
    The algorithm respects the simplex constraints (non‑negativity + fixed sum).

Stage 3 – guided hill‑climbing.
    The dominant convolution entry is located (index `k*`).  The sensitivity of
    each coordinate is `s_i = a_{k*‑i}` (zero if out of range).  Mass is moved
    from high‑sensitivity positions to low‑sensitivity ones; only improving
    moves are kept.  This “directional’’ search quickly squeezes the peak.

All three stages are time‑boxed; the whole routine stops before the global
budget (`budget_s`) expires and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """Fast version of the competition objective (exact for the normalised sum)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:  # should never happen
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    { x | x_i ≥ 0, Σ x_i = s }.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of the soft‑max approximation of the maximum convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Returns (g, b) where b = a*a and g = ∇(log‑sum‑exp_{β}(b))."""
    n = a.shape[0]

    # b = a*a (full convolution)
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)  # shape (2·n‑1,)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1  # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return g, b


def _annealed_gradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected gradient descent on the soft‑max approximation of the maximum.
    The temperature β is increased (annealed) while the step size adapts.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 256.0
    step_base = target_sum * 0.20
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            lr = step_base / (norm_g * (beta + 1.0))
            a = a - lr * g
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] AD   beta={beta:.2f}  score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Guided hill‑climbing (focuses on the current maximal convolution entry)
# ----------------------------------------------------------------------


def _guided_hill_climb(a: np.ndarray, deadline: float):
    """Move mass from indices that contribute heavily to the peak to lighter ones."""
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    cur_val = best_val

    rng = np.random.default_rng()
    max_frac = 0.07          # maximum fraction of a donor’s mass to move

    while time.time() < deadline:
        conv = np.convolve(a, a)          # full convolution (small; O(n))
        k_max = int(np.argmax(conv))

        # sensitivity vector: for each i, contribution to conv[k_max] is a[k_max-i]
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        # indices i in [lo,hi] have a partner j = k_max - i
        sens[lo : hi + 1] = a[k_max - np.arange(lo, hi + 1)]

        # choose a donor (large sens·a) and a receiver (small sens·a)
        weight = sens * a
        if weight.max() == 0:
            break
        donor = int(np.argmax(weight))
        # pick a receiver among the 10% lowest weights (avoid the same index)
        low_thr = np.percentile(weight, 10)
        candidates = np.where(weight <= low_thr)[0]
        if candidates.size == 0:
            break
        receiver = int(rng.choice(candidates))
        if donor == receiver:
            continue
        if a[donor] <= 1e-12:
            continue

        frac = rng.random() * max_frac
        delta = a[donor] * frac
        a[donor] -= delta
        a[receiver] += delta

        new_val = _score(a)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}] GH   improved = {best_val:.6f}")
        else:
            # revert the move
            a[receiver] -= delta
            a[donor] += delta

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple power‑law generator (used in the coarse grid)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point required by the competition
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine performs a quick grid search over power‑law exponents,
    refines the most promising candidates with annealed gradient descent
    and finishes with a guided hill‑climbing phase.
    The best sequence encountered before the time budget expires is returned.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    deadline = time.time() + budget_s - 2.0  # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Stage 1 – coarse grid over lengths & exponents
    # ------------------------------------------------------------------
    lengths = [1500, 2500, 3500, 4500, 5000, 8000]
    # number of p‑values to try per length
    p_grid = np.linspace(0.40, 0.70, 61)  # step 0.005

    # keep the top‑k raw candidates per length for deeper optimisation
    topk = 3
    candidates = []  # list of (score, array, n)

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # evaluate the whole grid
        scores = []
        arrays = []
        for p in p_grid:
            a = _power_law_sequence(n, p, target_sum)
            # small jitter to break ties / avoid exact symmetry
            a += rng.normal(scale=target_sum * 1e-4, size=n)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            val = _score(a)
            scores.append(val)
            arrays.append(a)

        # pick the best `topk` candidates for this length
        sorted_idx = np.argsort(scores)
        for idx in sorted_idx[:topk]:
            candidates.append((scores[idx], arrays[idx].copy(), n))

    # ------------------------------------------------------------------
    # Stage 2 – annealed gradient descent on the shortlisted candidates
    # ------------------------------------------------------------------
    for base_score, a0, n in candidates:
        if time.time() >= deadline:
            break
        # allocate a slice of the remaining budget
        remaining = deadline - time.time()
        # we give each candidate at most 30 seconds, but not more than what is left
        sub_deadline = time.time() + min(30.0, remaining / 2.0)

        # gradient‑descent phase
        a_opt, val = _annealed_gradient_descent(a0, sub_deadline)

        # ------------------------------------------------------------------
        # Stage 3 – guided hill‑climbing refinement
        # ------------------------------------------------------------------
        if time.time() < deadline:
            a_opt, val = _guided_hill_climb(a_opt, deadline)

        if val < best_score:
            best_score = val
            best_seq = a_opt.copy()
            print(f"** GLOBAL BEST **  n={n}  base_score={base_score:.6f}  "
                  f"final_score={best_score:.6f}")

    # fallback – a simple normalized constant sequence (never used unless everything fails)
    if best_seq is None:
        n = 1000
        a = np.ones(n, dtype=np.float64)
        a *= np.sqrt(2.0 * n) / n
        best_seq = a

    # trim trailing zeros (just in case)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    return best_seq.tolist()
