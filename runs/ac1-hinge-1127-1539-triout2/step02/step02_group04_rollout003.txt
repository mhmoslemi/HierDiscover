"""
Propose a step‑function (non‑negative vector) that minimizes the
provided ``evaluate_sequence`` objective.

The search consists of three stages:

1. **Fast parametric sweep** – families of analytic shapes (power‑law,
   exponential, beta‑type, sinusoidal) are sampled with a variety of
   parameters.  Every candidate is normalised to sum = 1 (the objective is
   scale‑free) and evaluated; the best candidate is kept.

2. **Projected Adam optimisation on the simplex** – starting from the best
   candidate, we minimise the maximal auto‑convolution using a smooth
   approximation of the ``max`` (soft‑max / log‑sum‑exp).  The gradient of
   the smoothed objective is obtained analytically via FFT‑based
   convolutions, which makes each iteration O(n log n).  An Adam step is
   taken and the iterate is projected back onto the probability simplex.
   The soft‑max temperature α is increased slowly so that the optimisation
   first sees a smooth landscape and later focuses on the true maximum.

3. **Length growth** – after a fixed number of Adam steps the vector is
   interpolated to a larger size (up to a preset maximum) and the Adam
   optimiser is continued.  Increasing the length lets the mass spread
   further and can reduce the factor ``n·max(conv)`` in the objective.

Random jitter, occasional momentum resets and a final fallback to a
uniform vector are included to avoid stagnation.  The routine respects a
hard 1000 s wall‑clock budget and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("simplex projection works on 1‑D arrays")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Convolution via FFT (real‐valued, returns length len(a)+len(b)-1)."""
    n = len(a) + len(b) - 1
    m = 1 << (n - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, m)
    B = np.fft.rfft(b, m)
    C = A * B
    conv = np.fft.irfft(C, m)[:n]
    return conv


def _softmax_gradient(x: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smoothed max (log‑sum‑exp) of the auto‑convolution
    ``conv = x*x`` with respect to ``x``.
    """
    # full auto‑convolution
    conv = _conv_fft(x, x)                     # length 2n‑1
    # soft‑max weights (numerically stable)
    maxc = conv.max()
    w = np.exp(alpha * (conv - maxc))
    w /= w.sum()
    # gradient = 2 * (x  *  w_rev) evaluated at appropriate indices
    w_rev = w[::-1]
    rev_conv = _conv_fft(x, w_rev)            # length n + (2n‑1) - 1 = 3n‑2
    L = len(w)
    grad = 2.0 * rev_conv[(L - 1) - np.arange(len(x))]
    return grad


def _evaluate(seq: np.ndarray) -> float:
    """Thin wrapper around the (already imported) ``evaluate_sequence``."""
    try:
        return evaluate_sequence(seq.tolist())
    except Exception:
        return float("inf")


# ----------------------------------------------------------------------
# Families of analytic candidates (used in the coarse sweep)
# ----------------------------------------------------------------------


def _gen_powerlaw(p: float, sigma: float, max_len: int = 5000) -> np.ndarray:
    """a_i = (i + sigma)^{-p} (truncated when very small)."""
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p)
    # prune tiny tails (saves time later)
    mask = a > 1e-12
    return a[mask]


def _gen_exponential(lam: float, max_len: int = 5000) -> np.ndarray:
    """a_i = exp(-lam * i) (truncated when very small)."""
    i = np.arange(max_len, dtype=np.float64)
    a = np.exp(-lam * i)
    mask = a > 1e-12
    return a[mask]


def _gen_beta(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Beta‑type shape:
        a_i ∝ (i+1)^{alpha} * (n-i)^{beta},
    for i = 0,…,n‑1.  ``n`` is the desired length.
    """
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta
    return a


def _gen_sine(p: float, n: int) -> np.ndarray:
    """
    Sinusoidal shape:
        a_i ∝ sin(π·(i+0.5)/n)^{p}
    """
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n) ** p
    return a


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list:
    """
    Search for a non‑negative step sequence that minimises
    ``evaluate_sequence``.  The routine respects a 1000 s wall‑clock budget
    and returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Use a globally supplied construction if present
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.array(height_sequence_1, dtype=float)
            if cand.size > 0:
                cand = cand / cand.sum()
                sc = _evaluate(cand)
                if np.isfinite(sc):
                    best_seq = cand.tolist()
                    best_score = float(sc)
                    print(f"[init] global seed score = {best_score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Broad parametric sweep (≈ first 150‑200 s)
    # ------------------------------------------------------------------
    print("=== Phase 1 – parametric sweep ===")
    max_len = 5000
    # --- power‑law ----------------------------------------------------
    p_vals = np.linspace(0.35, 0.65, 25)
    sigma_vals = np.logspace(-5, 2, 30)
    for p in p_vals:
        for sigma in sigma_vals:
            if time.time() >= deadline:
                break
            seq = _gen_powerlaw(p, sigma, max_len)
            if seq.size < 2:
                continue
            seq = seq / seq.sum()
            sc = _evaluate(seq)
            if np.isfinite(sc) and sc < best_score:
                best_score = float(sc)
                best_seq = seq.tolist()
                print(f"[sweep] PL  p={p:.3f} σ={sigma:.5f} len={len(seq)} → {best_score:.6f}")
        if time.time() >= deadline:
            break

    # --- exponential ---------------------------------------------------
    lam_vals = np.logspace(-4, 0, 25)
    for lam in lam_vals:
        if time.time() >= deadline:
            break
        seq = _gen_exponential(lam, max_len)
        if seq.size < 2:
            continue
        seq = seq / seq.sum()
        sc = _evaluate(seq)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_seq = seq.tolist()
            print(f"[sweep] EXP λ={lam:.5e} len={len(seq)} → {best_score:.6f}")

    # --- beta‑type (roughly triangular) --------------------------------
    for alpha in (0.0, 0.5, 1.0, 2.0):
        for beta in (0.0, 0.5, 1.0, 2.0):
            for n in (1000, 2000, 4000):
                if time.time() >= deadline:
                    break
                seq = _gen_beta(alpha, beta, n)
                seq = seq / seq.sum()
                sc = _evaluate(seq)
                if np.isfinite(sc) and sc < best_score:
                    best_score = float(sc)
                    best_seq = seq.tolist()
                    print(f"[sweep] BETA a={alpha} b={beta} n={n} → {best_score:.6f}")

    # --- sinusoidal ----------------------------------------------------
    for p in (1.0, 2.0, 3.0, 4.0):
        for n in (1200, 2500, 4000):
            if time.time() >= deadline:
                break
            seq = _gen_sine(p, n)
            seq = seq / seq.sum()
            sc = _evaluate(seq)
            if np.isfinite(sc) and sc < best_score:
                best_score = float(sc)
                best_seq = seq.tolist()
                print(f"[sweep] SINE p={p} n={n} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Projected Adam optimisation on the simplex
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: a modest length uniform vector
        n0 = 1500
        best_seq = np.ones(n0, dtype=float) / n0
        best_score = _evaluate(best_seq)
        best_seq = best_seq.tolist()
        print(f"[fallback] uniform len={n0} score={best_score:.6f}")

    # working copy (always kept on the simplex – sum == 1)
    x = np.array(best_seq, dtype=float)
    x = x / x.sum()

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.5

    m = np.zeros_like(x)
    v = np.zeros_like(x)

    iteration = 0
    grow_every = 3000               # steps after which we enlarge the vector
    max_len_allowed = 12000
    alpha_start = 5.0               # initial soft‑max temperature
    alpha_max = 40.0                # final temperature
    alpha_inc = (alpha_max - alpha_start) / 50000.0   # slow increase

    print("=== Phase 2 – Adam on simplex ===")
    while time.time() < deadline:
        iteration += 1

        # ---- dynamic temperature for soft‑max -----------------------
        alpha = min(alpha_max, alpha_start + alpha_inc * iteration)

        # ---- gradient of smoothed max --------------------------------
        grad = _softmax_gradient(x, alpha)

        # ---- Adam update ---------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = lr0 / (np.sqrt(v_hat) + eps)

        y = x - step * m_hat
        # projection back onto the simplex (ensures non‑negativity & sum=1)
        x = _simplex_projection(y)

        # ---- occasional jitter to escape plateaus --------------------
        if np.random.rand() < 0.01:
            j = np.random.randint(len(x))
            delta = (np.random.rand() - 0.5) * 0.02 * x[j]
            x[j] = max(0.0, x[j] + delta)
            x = x / x.sum()

        # ---- length growth -------------------------------------------
        if iteration % grow_every == 0 and len(x) < max_len_allowed:
            new_len = min(max_len_allowed, len(x) * 2)
            old_len = len(x)
            # linear interpolation preserves the shape
            new_x = np.interp(
                np.linspace(0, old_len - 1, new_len),
                np.arange(old_len),
                x,
            )
            x = new_x / new_x.sum()
            m = np.zeros_like(x)
            v = np.zeros_like(x)
            print(f"[grow] length increased to {new_len}")

        # ---- evaluate ------------------------------------------------
        cur_score = _evaluate(x)
        if np.isfinite(cur_score) and cur_score < best_score:
            best_score = float(cur_score)
            best_seq = x.tolist()
            print(f"[iter {iteration:6d}] improved score = {best_score:.6f} (len={len(x)})")
            # early exit if target already reached
            if best_score < 1.503:
                print("[target] achieved required bound – terminating early.")
                break

        # ---- restart if stagnating (simple heuristic) ---------------
        if iteration % 12000 == 0:
            # blend current point with fresh Dirichlet noise
            n = len(x)
            noise = np.random.rand(n)
            x = 0.6 * x + 0.4 * noise
            x = x / x.sum()
            m = np.zeros_like(x)
            v = np.zeros_like(x)

        # safety check for time
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 3.  Final reporting
    # ------------------------------------------------------------------
    print(f"final best score = {best_score:.6f}  length = {len(best_seq)}")
    return best_seq
