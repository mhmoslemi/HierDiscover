def compute_soft_grad(a, gamma):
    conv = compute_conv(a)   # length 2n-1
    max_c = conv.max()
    w = np.exp(gamma * (conv - max_c))
    w_sum = w.sum()
    w = w/w_sum
    # compute gradient of soft max: dm = sum_k w_k * d conv_k/da = 2*(a * w_rev reversed)
    # using convolution with w_rev
                    
    # Build w_rev = w[::-1]
    w_rev = w[::-1]
    n = len(a)
    # FFT length = next power of 2 >= n + len(w_rev) - 1
    conv_len = n + len(w_rev) - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    conv2 = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    # get dm: gm = 2 * conv2[U - j] where U = 2n-2
    U = 2*n - 2
    dm = 2.0 * conv2[U - np.arange(n)]
    return conv.max(), dm

"""
Propose a step‑function (non‑negative sequence) with a low evaluate_sequence score.

Algorithm
---------
1.  Multi‑restart: pick a length n (2000 … 12000) and generate a random
    positive start vector a (Dirichlet / Gamma draws).  The L1‑norm is
    normalised to √(2·n) – the evaluation score then equals the maximal
    entry of the self‑convolution (scale invariance).

2.  Smooth‑max gradient descent.
    •  Let 𝑚(a)=maxₖ (a∗a)ₖ .  Replace the true max by a soft‑max
       Mγ(a)= (1/γ)·log Σₖ exp(γ·(a∗a)ₖ) .
    •  The gradient of Mγ is  dm = 2·conv(a , w_rev) where
       wₖ ∝ exp(γ·((a∗a)ₖ–max)) are normalised weights.
    •  The score S = 2·n·m / (∑a)²; with a fixed L1‑norm it reduces to m.
       Using the soft‑max we minimise the surrogate
           J = 2·n·Mγ / (∑a)² .
       Its gradient is
           ∇J = (2·n/∑a²)·dm  –  (4·n·Mγ/∑a³)·1 .
    •  Perform a few thousand gradient steps with a linearly decaying
       step size and a γ that grows from a small to a larger value,
       thus moving from a smooth landscape to the true max.

3.  Local greedy refinement.
    After the smooth phase, repeatedly pick the shift k that attains the
    current maximal convolution entry.  Among the overlapping indices
    choose the pair (i,j) with the largest product aᵢ·aⱼ, move a small
    fraction of the larger entry to a low‑valued position, re‑normalise
    and accept the move if the maximal convolution decreases.

4.  Keep the best sequence seen across all restarts.
    Whenever the surrogate (or the exact) score improves, call the
    provided evaluate_sequence to record the official score.

The whole search stops slightly before the given time budget (≈5 s
margin).  The returned list of heights is guaranteed to be non‑negative
and to respect the evaluator’s limits.

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the linear self‑convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _softmax_gradient(a: np.ndarray, gamma: float) -> tuple[float, np.ndarray]:
    """
    Compute a smooth approximation of the max‑convolution and its gradient.
    Returns (Mγ , dm) where
        Mγ = (1/γ)·log Σ exp(γ·convₖ)   (soft‑max of the convolution)
        dm = ∂Mγ/∂a   (vector of length len(a))
    """
    n = a.shape[0]
    conv = _full_convolution(a)                # length 2n‑1
    max_c = conv.max()

    # soft‑max weights (stable version)
    w = np.exp(gamma * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:                # fallback – should not happen
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient of the soft‑max: dm = 2 * (a ⊛ w_rev)
    w_rev = w[::-1]                                 # reverse weights
    # convolution a * w_rev  (full length = n + (2n‑1) – 1 = 3n‑2)
    conv_len2 = n + w_rev.size - 1
    fft_len = 1 << (conv_len2 - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    conv2 = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len2]

    # Extract the required slice: dm[j] = 2·conv2[U‑j] ,  U = 2n‑2
    U = 2 * n - 2
    dm = 2.0 * conv2[U - np.arange(n)]

    # The smooth max value (not needed for the gradient but useful for logging)
    M_gamma = max_c + np.log(w_sum) / gamma

    return M_gamma, dm

# Local optimiser for a single start

def _optimize_one_start(
    a0: np.ndarray,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_iter_grad: int = 2500,
    max_iter_greedy: int = 500,
) -> tuple[np.ndarray, float]:
    """
    Smooth‑max gradient descent followed by a greedy local search.
    Returns the best vector found and its exact maximal self‑convolution
    value (which equals the evaluate_sequence score after normalisation).
    """
    a = a0.copy()
    target_sum = np.sqrt(2.0 * n)          # we keep the L1‑norm fixed
    a *= target_sum / a.sum()              # normalise exactly

    # ----- phase 1 : smooth‑max gradient descent -----------------------
    best_a = a.copy()
    best_m = np.inf

    C = 2.0 * n                           # constant in the objective
    start_time = time.time()

    for it in range(1, max_iter_grad + 1):
        if time.time() >= deadline:
            break

        # gradually increase the soft‑max sharpness
        gamma = 5.0 + 20.0 * (it / max_iter_grad)   # 5 → 25

        M_gamma, dm = _softmax_gradient(a, gamma)

        s = a.sum()
        # Gradient of the true objective with the smooth max in place
        grad = (C / (s * s)) * dm - (2.0 * C * M_gamma / (s * s * s)) * np.ones_like(a)

        # simple learning‑rate schedule (decays to a small value)
        eta = 0.30 * (1.0 - it / max_iter_grad) + 0.02

        a -= eta * grad
        a = np.maximum(a, 0.0)                          # enforce non‑negativity

        # renormalise the L1‑norm (exact scale invariance)
        total = a.sum()
        if total == 0.0:
            a.fill(target_sum / n)
        else:
            a *= (target_sum / total)

        # occasional small random jitter to escape plateaus
        if rng.random() < 0.015:
            noise = rng.normal(scale=a.mean() * 0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a *= target_sum / a.sum()

        # monitor improvement of the *exact* max‑convolution
        conv = _full_convolution(a)
        cur_m = conv.max()
        if cur_m < best_m - 1e-12:
            best_m = cur_m
            best_a = a.copy()
            try:
                sc = evaluate_sequence(best_a.tolist())
                print(f"[grad improve] time={time.time():.1f}s  n={n} it={it} "
                      f"max={best_m:.6f}  score={sc:.6f}")
            except Exception:
                pass

    a = best_a.copy()                     # start greedy phase from current best

    # ----- phase 2 : greedy local refinement ---------------------------
    for it in range(1, max_iter_greedy + 1):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a)
        cur_m = conv.max()
        k = int(np.argmax(conv))           # shift where the maximum occurs

        # overlapping index range for this shift
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1)
        jdx = k - idx

        products = a[idx] * a[jdx]
        max_pair = int(np.argmax(products))
        i1, i2 = idx[max_pair], jdx[max_pair]

        # choose the larger entry as donor
        donor = i1 if a[i1] >= a[i2] else i2

        # pick a low‑valued recipient (among the 10 % smallest entries)
        low_mask = a <= np.percentile(a, 10)
        low_candidates = np.nonzero(low_mask)[0]
        if low_candidates.size == 0:
            low_candidates = np.arange(n)
        recipient = rng.choice(low_candidates)

        if donor == recipient:
            continue

        delta = a[donor] * 0.07          # move 7 % of the donor mass
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[recipient] += delta

        # re‑normalise (exact sum is unchanged, but rounding errors accumulate)
        total = a.sum()
        if total == 0.0:
            a.fill(target_sum / n)
        else:
            a *= (target_sum / total)

        # accept only if the maximum decreased
        new_conv = _full_convolution(a)
        new_m = new_conv.max()
        if new_m < cur_m - 1e-12:
            cur_m = new_m
            if cur_m < best_m - 1e-12:
                best_m = cur_m
                best_a = a.copy()
                try:
                    sc = evaluate_sequence(best_a.tolist())
                    print(f"[greedy improve] time={time.time():.1f}s  n={n} it={it} "
                          f"max={best_m:.6f}  score={sc:.6f}")
                except Exception:
                    pass
        else:
            # revert the move
            a[recipient] -= delta
            a[donor] += delta

    # final exact maximal convolution (the value that evaluate_sequence will see)
    final_conv = _full_convolution(best_a)
    final_max = final_conv.max()
    return best_a, final_max

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that yields a low
    evaluate_sequence score.  The routine runs many independent restarts,
    each consisting of a smooth‑max gradient phase followed by a greedy
    local refinement.  The best sequence found before the time budget is
    exhausted is returned.
    """
    rng = np.random.default_rng(seed)
    # a tiny safety margin so we can always return a value
    deadline = time.time() + budget_s - 5.0

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # If a previous construction is already present, record its score.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            if cand.ndim == 1 and cand.size > 0:
                sc = evaluate_sequence(cand.tolist())
                best_seq = cand.copy()
                best_score = sc
                print(f"[baseline] existing height_sequence_1 score = {sc:.6f} "
                      f"length = {cand.size}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Parameter space for lengths.
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000, 10000, 12000]

    run_id = 0
    while time.time() < deadline:
        n = int(rng.choice(lengths))

        # ---------- initialisation ----------
        if best_seq is not None and best_seq.size >= n and rng.random() < 0.35:
            # truncate or randomly sample from the current best
            if best_seq.size == n:
                a0 = best_seq.copy()
            else:
                # random subset without replacement
                idx = rng.choice(best_seq.size, n, replace=False)
                a0 = best_seq[idx].copy()
        else:
            # Gamma distribution (shape < 1 gives a heavy‑tailed start)
            shape = rng.uniform(0.1, 1.0)
            a0 = rng.gamma(shape=shape, scale=1.0, size=n)

        # normalise L1‑norm to √(2·n)
        target = np.sqrt(2.0 * n)
        a0 = a0 / a0.sum() * target

        # ---------- optimise this start ----------
        a_opt, max_conv = _optimize_one_start(a0, n, deadline, rng)

        # exact external score (identical to max_conv because of normalisation)
        try:
            sc = evaluate_sequence(a_opt.tolist())
        except Exception:
            sc = float('inf')

        if sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"[global improve] run={run_id} n={n} score={best_score:.6f}")

        run_id += 1

        # early exit if we already beat the target bound
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback in the unlikely case that nothing was found.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] score={best_score:.6f}")

    print(f"[final] best score = {best_score:.6f}  length = {len(best_seq)}")
    return best_seq.tolist()
