def lp_phase(f, best_score, deadline):
    n_target = min(len(f), 1200)
    if len(f) > n_target:
        # downsample by grouping into n_target groups
        group = int(np.ceil(len(f) / n_target))
        f_small = np.add.reduceat(f, np.arange(0, len(f), group))
        # trim or pad to n_target
        if len(f_small) > n_target:
            f_small = f_small[:n_target]
        elif len(f_small) < n_target:
            # pad with tiny
            extra = np.full(n_target - len(f_small), 1e-12)
            f_small = np.concatenate([f_small, extra])
        f_small = f_small / f_small.sum()
    else:
        f_small = f.copy()
    
    n = len(f_small)
    # iterative LP update
    iteration = 0
    while time.time() < deadline and iteration < 200:
        iteration += 1

        use_full = (n <= 800)

def annealing_optimize(f, deadline, max_iters=50000):
    if f is None:
        n0 = 800
        f = np.full(n0, 1.0/n0)
    f = f / f.sum()

    total_time = deadline - time.time()
    start = time.time()
    T0 = 0.05  # initial temperature
    it = 0
    while time.time() < deadline:
        it += 1
        # linearly decaying temperature
        elapsed = time.time() - start
        fraction = elapsed / total_time
        T = T0 * (1 - fraction)
        # propose move
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j: continue
        # move up to 10% of mass from i to j
        delta = (np.random.rand() - 0.5) * 0.1 * f[i]
        if delta > f[i]:
            delta = f[i]
        if delta < -f[i]:
            delta = -f[i]  # ensure nonnegative after subtraction
        new = f.copy()
        new[i] -= delta
        new[j] += delta
        # ensure nonnegative
        if new[i] < 0:

"""
Propose a non‑negative step sequence that (approximately) minimises
evaluate_sequence().

Algorithmic outline
-------------------
1. **Initialisation** – reuse the global ``height_sequence_1`` if available,
   otherwise start from a simple power‑law shape.  All candidates are
   normalised to ℓ¹‑norm = 1, because the objective equals 2·n·max(conv) under
   this normalisation.

2. **Fast parametric sweep** – generate families (power law, exponential,
   mixed) of analytic sequences, evaluate them quickly with an FFT based
   convolution, and keep the best one found before the time budget expires.

3. **Linear‑programming improvement** – for the current vector *f* we solve

       max  Σ b_j      subject to   f * b ≤ M  ,  b ≥ 0

   where `M = max(f*f)` and `*` denotes linear convolution.
   The LP (solved with SciPy ‘highs’) yields a feasible vector *b* that spreads
   the mass more evenly.  We mix the result into *f* (convex combination)
   and repeat.  For modest lengths (≤ 800) the full set of constraints is used;
   for larger vectors only the top K convolution constraints are kept.

4. **Projected sub‑gradient descent** – starting from the best LP result we
   minimise the ℓ∞‑norm of the autocorrelation using a Nesterov‑style momentum,
   projecting after each step onto the probability simplex.

5. **Simulated‑annealing fine‑tuning** – random mass transfers between two
   positions are accepted according to a temperature schedule; this helps
   escape tiny plateaus left by the gradient method.

6. **Exact evaluation** – the final best candidate is passed once to the
   provided ``evaluate_sequence`` (which applies the official clipping rules).

All helper routines are top‑level, no closures or lambdas are used, and the
implementation respects the 1000 s wall‑clock budget.
"""

# Fast convolution utilities (linear convolution via FFT)
# ----------------------------------------------------------------------


def _fft_max_convolution(x: np.ndarray) -> float:
    """Return max_i (x*x)_i using a power‑of‑two FFT."""
    n = x.shape[0]
    L = 1 << ((2 * n - 1).bit_length())       # next power of two
    fx = np.fft.rfft(x, L)
    conv = np.fft.irfft(fx * fx, L)[: 2 * n - 1]
    return float(conv.max())


def _fast_score(x: np.ndarray) -> float:
    """Objective 2·n·max(conv) for a vector normalised to sum = 1."""
    x = np.asarray(x, dtype=np.float64)
    if x.sum() == 0:
        return float("inf")
    x = x / x.sum()
    n = x.shape[0]
    return 2.0 * n * _fft_max_convolution(x)

# Simple analytic families used in the parametric sweep
# ----------------------------------------------------------------------


def _generate_powerlaw(p: float, sigma: float,
                       max_len: int = 5000, eps: float = 1e-10) -> np.ndarray:
    """a_i = (i+σ)^(-p)  truncated when a_i < eps."""
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p)
    return a[a > eps]


def _generate_exponential(lam: float,
                          max_len: int = 5000, eps: float = 1e-10) -> np.ndarray:
    """a_i = exp(-λ·i)  truncated when a_i < eps."""
    i = np.arange(max_len, dtype=np.float64)
    a = np.exp(-lam * i)
    return a[a > eps]


def _generate_mixed(p: float, sigma: float, lam: float,
                    max_len: int = 5000, eps: float = 1e-10) -> np.ndarray:
    """a_i = (i+σ)^(-p)·exp(-λ·i)."""
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p) * np.exp(-lam * i)
    return a[a > eps]


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used by gradient methods
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx=1}."""
    if v.ndim != 1:
        raise ValueError("simplex_projection expects 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient of the ℓ∞‑norm of the autocorrelation
# ----------------------------------------------------------------------


def _subgradient(x: np.ndarray) -> np.ndarray:
    """Sub‑gradient of g(x)=max_k (x*x)_k."""
    n = x.shape[0]
    conv = np.convolve(x, x)            # linear conv, length 2n‑1
    m = conv.max()
    active = np.where(conv >= m - 1e-14)[0]
    g = np.zeros_like(x)
    for k in active:
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        i = np.arange(i_min, i_max + 1, dtype=int)
        j = k - i
        g[i] += x[j]                     # derivative of conv_k w.r.t. x_i
    return g


# ----------------------------------------------------------------------
# Linear‑programming helper (maximise Σb_j under f*b ≤ M)
# ----------------------------------------------------------------------


def _lp_max_sum(f: np.ndarray, M: float, use_full: bool = True,
                active_k: int = 150) -> np.ndarray | None:
    """Solve  max Σb_j  s.t. f*b ≤ M, b ≥ 0.  Returns the optimal b or None."""
    n = f.shape[0]
    if use_full or n <= 800:
        # build full Toeplitz‑like constraint matrix
        offsets = np.arange(n)                     # 0 … n‑1
        data = np.vstack([np.full(n, f[d]) for d in range(n)])  # (n,n)
        A = diags(data, offsets, shape=(2 * n - 1, n), format='csc')
    else:
        # only the K largest convolution constraints
        conv_full = np.convolve(f, f)
        K = min(active_k, conv_full.size)
        idx = np.argpartition(conv_full, -K)[-K:]      # indices of top K values
        rows, cols, vals = [], [], []
        for k in idx:
            i_min = max(0, k - (n - 1))
            i_max = min(k, n - 1)
            for i in range(i_min, i_max + 1):
                j = k - i
                rows.append(k)
                cols.append(j)
                vals.append(f[i])
        A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)                             # maximise Σb_j  → minimise –Σb_j
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'presolve': True})
    except Exception:                           # pragma: no cover
        return None
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# Phase 1 – fast parametric sweep
# ----------------------------------------------------------------------


def _parametric_sweep(initial_seq: np.ndarray,
                      initial_score: float,
                      deadline: float) -> tuple[np.ndarray, float]:
    """Search simple analytic families; keep the best candidate."""
    best_seq = initial_seq
    best_score = initial_score
    max_len = 3000
    eps = 1e-10

    # power‑law family
    p_vals = np.linspace(0.38, 0.62, 15)
    sigma_vals = np.logspace(-4, 1, 12)
    for p in p_vals:
        for sigma in sigma_vals:
            if time.time() >= deadline:
                return best_seq, best_score
            seq = _generate_powerlaw(p, sigma, max_len, eps)
            if seq.size < 2:
                continue
            seq = seq / seq.sum()
            sc = _fast_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f'PL sweep p={p:.3f} σ={sigma:.5g} len={len(seq)} score={sc:.6f}')

    # exponential family
    lam_vals = np.logspace(-4, -0.2, 12)
    for lam in lam_vals:
        if time.time() >= deadline:
            break
        seq = _generate_exponential(lam, max_len, eps)
        if seq.size < 2:
            continue
        seq = seq / seq.sum()
        sc = _fast_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f'EXP sweep λ={lam:.5e} len={len(seq)} score={sc:.6f}')

    # mixed family
    for p in (0.45, 0.5, 0.55):
        for sigma in (1e-3, 1e-2, 1e-1, 1.0):
            for lam in (1e-3, 5e-3, 1e-2):
                if time.time() >= deadline:
                    break
                seq = _generate_mixed(p, sigma, lam, max_len, eps)
                if seq.size < 2:
                    continue
                seq = seq / seq.sum()
                sc = _fast_score(seq)
                if sc < best_score:
                    best_score, best_seq = sc, seq.copy()
                    print(f'MIX sweep p={p:.2f} σ={sigma:.5g} λ={lam:.5g} len={len(seq)} score={sc:.6f}')
    return best_seq, best_score


# ----------------------------------------------------------------------
# Phase 2 – LP based mass redistribution
# ----------------------------------------------------------------------


def _lp_improvement(f: np.ndarray, deadline: float,
                    max_iters: int = 150) -> tuple[np.ndarray, float]:
    """Iteratively improve f by solving the LP max Σb subject to f*b ≤ max(f*f)."""
    if f is None:
        f = np.full(800, 1.0 / 800, dtype=np.float64)
    else:
        f = f.copy()
    f = f / f.sum()
    best = f.copy()
    best_sc = _fast_score(best)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        M = _fft_max_convolution(f)
        n = f.shape[0]
        use_full = (n <= 800)
        b = _lp_max_sum(f, M, use_full=use_full, active_k=150)
        if b is None or b.sum() == 0:
            # LP failed – abort the LP phase
            break
        g = b / b.sum()
        alpha = 0.35                     # mixing weight
        f_new = (1 - alpha) * f + alpha * g
        f_new = f_new / f_new.sum()
        sc_new = _fast_score(f_new)
        if sc_new < best_sc - 1e-9:
            best_sc, best = sc_new, f_new.copy()
            print(f'LP iter {it:3d} improved fast score {sc_new:.6f}')
        f = f_new

        # occasional length growth (tiny entries) – helps the factor n·max(conv)
        if it % 30 == 0 and n < 2000:
            extra = np.full(100, 1e-12, dtype=np.float64)
            f = np.concatenate([f, extra])
            f = f / f.sum()
    return best, best_sc


# ----------------------------------------------------------------------
# Phase 3 – Projected sub‑gradient descent with momentum
# ----------------------------------------------------------------------


def _gradient_descent(f: np.ndarray, deadline: float,
                      max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """Nesterov‑style projected sub‑gradient method."""
    if f is None:
        f = np.full(800, 1.0 / 800, dtype=np.float64)
    f = f / f.sum()
    best = f.copy()
    best_sc = _fast_score(best)

    n = f.shape[0]
    step0 = 1.0
    beta = 0.85
    momentum = np.zeros_like(f)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        g = _subgradient(f)
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / np.sqrt(it)
        y = f - step * momentum
        f = _simplex_projection(y)

        # small random jitter – helps escaping plateaus
        if np.random.rand() < 0.01:
            j = np.random.randint(n)
            delta = (np.random.rand() - 0.5) * 0.05 * f[j]
            f[j] = max(0.0, f[j] + delta)
            f = f / f.sum()

        sc = _fast_score(f)
        if sc < best_sc - 1e-9:
            best_sc, best = sc, f.copy()
            print(f'Grad iter {it:5d} new best {sc:.6f}')

        # length growth every few thousand iterations
        if it % 2000 == 0 and n < 2500:
            extra = np.full(200, 1e-12, dtype=np.float64)
            f = np.concatenate([f, extra])
            f = f / f.sum()
            momentum = np.concatenate([momentum, np.zeros_like(extra)])
            n = f.shape[0]
    return best, best_sc


# ----------------------------------------------------------------------
# Phase 4 – Simulated annealing fine‑tuning
# ----------------------------------------------------------------------


def _annealing(f: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Very low‑temperature random walk that moves mass between positions."""
    if f is None:
        f = np.full(800, 1.0 / 800, dtype=np.float64)
    f = f / f.sum()
    best = f.copy()
    best_sc = _fast_score(best)
    cur_sc = best_sc
    n = f.shape[0]

    start = time.time()
    total = max(1.0, deadline - start)
    T0 = 0.05                       # initial temperature

    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / total
        T = T0 * (1.0 - fraction)   # linear cooling

        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        # move a random fraction of the mass from i to j
        delta = (np.random.rand() - 0.5) * 0.1 * f[i]
        if delta > f[i]:
            delta = f[i]
        if delta < -f[i]:
            delta = -f[i]
        new = f.copy()
        new[i] -= delta
        new[j] += delta
        if new[i] < 0:               # safeguard against negative due to rounding
            continue
        new_sc = _fast_score(new)
        accept = False
        if new_sc < cur_sc:
            accept = True
        else:
            # Metropolis acceptance
            prob = math.exp(-(new_sc - cur_sc) / (T + 1e-12))
            if np.random.rand() < prob:
                accept = True
        if accept:
            f, cur_sc = new, new_sc
            if new_sc < best_sc:
                best_sc, best = new_sc, new.copy()
                print(f'Anneal iter {it} new best {best_sc:.6f}')
    return best, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list:
    """
    Search for a non‑negative step function that minimises the supplied
    evaluate_sequence.  The routine respects a 1000 s wall‑clock budget and
    returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # a small safety margin

    # ------------------------------------------------------------------
    # 0.  Use a globally supplied construction if available
    # ------------------------------------------------------------------
    best_seq = None
    best_fast = float("inf")
    if 'height_sequence_1' in globals():
        try:
            sc = evaluate_sequence(height_sequence_1)
            if np.isfinite(sc):
                arr = np.array(height_sequence_1, dtype=float)
                arr = arr / arr.sum()
                best_seq = arr
                best_fast = _fast_score(arr)
                print(f'Initial global sequence fast score = {best_fast:.6f}')
        except Exception:                     # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # 1.  Broad parametric sweep (fast)
    # ------------------------------------------------------------------
    best_seq, best_fast = _parametric_sweep(best_seq, best_fast, deadline)

    # ------------------------------------------------------------------
    # 2.  Linear‑programming based improvement (if SciPy is present)
    # ------------------------------------------------------------------
    if SCIPY_AVAILABLE:
        best_seq, best_fast = _lp_improvement(best_seq, deadline, max_iters=150)

    # ------------------------------------------------------------------
    # 3.  Projected sub‑gradient descent
    # ------------------------------------------------------------------
    best_seq, best_fast = _gradient_descent(best_seq, deadline, max_iters=20000)

    # ------------------------------------------------------------------
    # 4.  Simulated annealing fine‑tuning
    # ------------------------------------------------------------------
    best_seq, best_fast = _annealing(best_seq, deadline)

    # ------------------------------------------------------------------
    # 5.  Exact evaluation of the final best candidate
    # ------------------------------------------------------------------
    try:
        exact = evaluate_sequence(best_seq.tolist())
        print('Exact final score =', exact)
    except Exception:                        # pragma: no cover
        exact = None

    # Return as plain Python list – the harness expects a list of floats.
    return best_seq.tolist()
