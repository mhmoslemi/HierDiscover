    n = len(a)
    best = a.copy()
    best_score = _score(a)
    cur = a.copy()
    cur_score = best_score
    T0 = 0.1 * best_score  # initial temperature scaling
    t = 0
    while time.time() < deadline:
        # compute temperature schedule
        temperature = T0 * (0.999**t)  # decays
        # propose move
        i = rng.integers(n)
        if cur[i] < 1e-12:
            continue
        j = rng.integers(n)
        if i == j: continue
        # choose delta fraction of cur[i], e.g., up to 0.3
        frac = rng.random() * 0.3
        delta = cur[i] * frac

        cur[i] -= delta
        cur[j] += delta
        # compute new max
        new_score = _score(cur)
        # acceptance
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score)/max(temperature, 1e-12)):
            cur_score = new_score
            if new_score < best_score:

        else:
            # revert
            cur[j] -= delta
            cur[i] += delta
        t+=1
    return best, best_score

    n = a.shape[0]

    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]

    # compute initial conv
    b = compute_conv(a) # full conv

    max_val = b[max_idx]

    best_b = b.copy()
    best_max = max_val

        j = rng.integers(n)

"""
Simulated‑annealing optimiser for the step‑function bound.

The objective after normalising the ℓ¹‑norm to √(2·n) is simply
    max_k (a*a)[k] .
Therefore we keep the sum exactly fixed and repeatedly move a tiny
fraction of mass from one index to another.  The linear convolution
changes at *only three* positions (2·i, 2·j and i+j), which allows a
O(1) update of the maximum value.  A classic Metropolis schedule with
exponential cooling accepts worsening moves with probability
exp(−Δ/T) and gradually reduces the temperature T.  The algorithm
restarts on several lengths (1500 … 8000) and uses a known construction
(`height_sequence_1`) when available as a warm start.  The best
sequence found before the time budget expires is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (full convolution + max)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using an FFT (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (FFT‑based)."""
    return float(_full_convolution(a).max())


# ----------------------------------------------------------------------
# Simple up‑sampling / interpolation
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a new length."""
    old_len = seq.size
    x_old = np.linspace(0.0, 1.0, old_len, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_len, endpoint=False)
    return np.interp(x_new, x_old, seq)


# ----------------------------------------------------------------------
# Incremental simulated‑annealing (mass transfer moves)
# ----------------------------------------------------------------------


def _simulated_annealing_incremental(a0: np.ndarray,
                                    target_sum: float,
                                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Fast SA that updates the convolution in O(1) per move.
    Returns the best vector found and its objective value (max conv entry).
    """
    n = a0.size
    a = a0.copy()
    b = _full_convolution(a)                     # full convolution, length 2·n‑1
    cur_max_idx = int(b.argmax())
    cur_max_val = float(b[cur_max_idx])

    best_a = a.copy()
    best_max = cur_max_val
    best_idx = cur_max_idx

    rng = np.random.default_rng()
    # initial temperature – a fraction of the current peak
    T = max(cur_max_val * 0.1, 1e-6)
    decay = 0.9995
    it = 0

    while time.time() < deadline:
        # ---------- propose a move ----------
        i = rng.integers(0, n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # move a fraction (up to 20 %) of mass from i to j
        frac = rng.random() * 0.20
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a_i_old = a[i]
        a_j_old = a[j]

        # remember the present max (value & index)
        pre_max_val = cur_max_val
        pre_max_idx = cur_max_idx

        # ---------- apply the O(1) update ----------
        idx1 = 2 * i
        idx2 = 2 * j
        idx3 = i + j

        # three affected convolution entries (see derivation in analysis)
        b[idx1] += -2.0 * delta * a_i_old + delta * delta
        b[idx2] += 2.0 * delta * a_j_old + delta * delta
        b[idx3] += 2.0 * delta * (a_i_old - a_j_old) - 2.0 * delta * delta

        a[i] = a_i_old - delta
        a[j] = a_j_old + delta

        # ---------- compute new peak ----------
        # if the old peak was one of the changed entries we must recompute
        if pre_max_idx in (idx1, idx2, idx3):
            cur_max_val = float(b.max())
            cur_max_idx = int(b.argmax())
        else:
            # otherwise the new peak is either the old one or one of the three
            cur_max_val = pre_max_val
            cur_max_idx = pre_max_idx
            v1 = b[idx1]
            if v1 > cur_max_val:
                cur_max_val = float(v1)
                cur_max_idx = idx1
            v2 = b[idx2]
            if v2 > cur_max_val:
                cur_max_val = float(v2)
                cur_max_idx = idx2
            v3 = b[idx3]
            if v3 > cur_max_val:
                cur_max_val = float(v3)
                cur_max_idx = idx3

        # ---------- Metropolis acceptance ----------
        accept = False
        if cur_max_val <= pre_max_val:
            accept = True
        else:
            prob = math.exp(-(cur_max_val - pre_max_val) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            # keep the move – possibly update the global best
            if cur_max_val < best_max:
                best_max = cur_max_val
                best_a = a.copy()
                best_idx = cur_max_idx
                print(f"[{time.time():.1f}] SA new global best max = {best_max:.6f} (n={n})")
        else:
            # reject – revert everything
            a[i] = a_i_old
            a[j] = a_j_old
            b[idx1] -= -2.0 * delta * a_i_old + delta * delta
            b[idx2] -= 2.0 * delta * a_j_old + delta * delta
            b[idx3] -= 2.0 * delta * (a_i_old - a_j_old) - 2.0 * delta * delta
            cur_max_val = pre_max_val
            cur_max_idx = pre_max_idx

        # ---------- occasional global jitter ----------
        if it % 2000 == 0 and it > 0:
            # small Gaussian perturbation followed by renormalisation
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = a * (target_sum / a.sum())
            b = _full_convolution(a)
            cur_max_val = float(b.max())
            cur_max_idx = int(b.argmax())

        # ---------- temperature schedule ----------
        T *= decay
        if T < 1e-12:
            T = 1e-12
        it += 1

    # The objective after normalisation is exactly the peak value.
    return best_a, best_max


# ----------------------------------------------------------------------
# Main entry point required by the challenge
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising the supplied
    `evaluate_sequence` objective.  The routine runs for at most
    `budget_s` seconds, trying several vector lengths and random
    restarts.  The best sequence found is returned.
    """
    rng = np.random.default_rng(seed)
    # safety margin – stay a couple of seconds inside the budget
    deadline = time.time() + budget_s - 3.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Load a known construction if available (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        base_seq = np.array(height_sequence_1, dtype=np.float64)
    except NameError:
        base_seq = None

    # ------------------------------------------------------------------
    # Lengths to explore – trade‑off between flexibility and speed
    # ------------------------------------------------------------------
    lengths = [1500, 2500, 3500, 5000, 7500]
    # Number of restarts per length (will be limited by remaining time)
    restarts_per_len = 3

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        for restart in range(restarts_per_len):
            if time.time() >= deadline:
                break

            # ------------------- initialise a candidate -------------------
            if base_seq is not None:
                # Upsample the known construction to the current length
                a0 = _upsample(base_seq, n)
            else:
                # Random decreasing power‑law
                p = rng.uniform(0.35, 0.75)
                a0 = (np.arange(1, n + 1, dtype=np.float64) ** (-p))

            # Normalise ℓ¹‑norm to √(2·n) and add a tiny jitter
            a0 = a0 * (target_sum / a0.sum())
            a0 += rng.normal(scale=target_sum * 1e-3, size=n)
            a0 = np.maximum(a0, 0.0)
            a0 = a0 * (target_sum / a0.sum())

            # ------------------- allocate a sub‑budget -------------------
            remaining = deadline - time.time()
            # give at most 45 s to a single restart (adjust if little time left)
            sub_deadline = time.time() + min(45.0, remaining * 0.5)

            # ------------------- run the fast SA optimiser ----------------
            a_opt, max_val = _simulated_annealing_incremental(a0, target_sum, sub_deadline)

            # final objective according to the original evaluation function
            try:
                # evaluate_sequence is provided by the platform
                score = evaluate_sequence(a_opt.tolist())
            except Exception:
                # fall back to the internal formula (identical when sum is correct)
                score = 2.0 * n * max_val / (target_sum * target_sum)

            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"** NEW GLOBAL BEST **  n={n}  restart={restart}  score={best_score:.6f}")

            # If we already beat the target, stop early
            if best_score <= 1.503:
                break

        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback – in the unlikely event nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = math.sqrt(2.0 * n)
        a = np.full(n, target_sum / n, dtype=np.float64)
        best_seq = a

    # Trim trailing zeros that might have appeared through numerical noise
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    return best_seq.tolist()
