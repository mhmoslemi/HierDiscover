def gradient_softmax(p, conv, beta):
    # conv = p * p
    max_val = conv.max()
    # compute softmax weights:
    # Actually to avoid overflow, we compute exp(beta*(conv - max_val)): this yields numbers <=1.
    w = np.exp(beta * (conv - max_val))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones_like(w) / w.size
    else:
        w /= w_sum
    # compute convolution of w with reversed p
    n = p.shape[0]
    p_rev = p[::-1]
    L = 1 << ((len(w) + n - 1 - 1).bit_length())
    ft_w = np.fft.rfft(w, n=L)
    ft_p = np.fft.rfft(p_rev, n=L)
    conv_w_p = np.fft.irfft(ft_w * ft_p, n=L)[: len(w) + n - 1]
    g = 2.0 * conv_w_p[n-1 - np.arange(n)]
    return g

"""
Non‑negative step‑function optimiser (soft‑max gradient).

The objective after normalising Σ a_i = 1 is
    score = 2·n·max(conv(a,a))

We minimise the ℓ∞‑norm of the self‑convolution by a
projected gradient descent on the probability simplex.
The gradient is obtained from a smooth approximation of the max:
    w_k ∝ exp(β·(conv_k – max(conv)))   (β increases during optimisation)

The gradient of a weighted sum of convolutions can be written as a
single convolution, which we evaluate with FFTs (O(n log n)).
A diverse set of seeds (power‑law, exponential, random Dirichlet,
previous best, etc.) is explored; each seed is refined for a
large number of iterations or until the time budget (≈1000 s) expires.
The best sequence found is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array")
    n = v.shape[0]

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_conv_fft(p: np.ndarray) -> np.ndarray:
    """Linear convolution p * p (full, length 2·len(p)‑1) using FFT."""
    n = p.shape[0]
    L = 1 << ((2 * n - 2).bit_length())          # next power of two
    ft = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(ft * ft, n=L)[: 2 * n - 1]
    # eliminate tiny negative artefacts caused by round‑off
    conv[conv < 0] = 0.0
    return conv


def _gradient_softmax(p: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv(p,p)) w.r.t. p.
    The weights are w_k ∝ exp(β·(conv_k – max(conv))).
    """
    max_val = conv.max()
    # weights (stable soft‑max)
    w = np.exp(beta * (conv - max_val))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    n = p.shape[0]
    # convolution w * reverse(p) gives the needed sums
    p_rev = p[::-1]
    L = 1 << ((len(w) + n - 2).bit_length())
    ft_w = np.fft.rfft(w, n=L)
    ft_pr = np.fft.rfft(p_rev, n=L)
    tmp = np.fft.irfft(ft_w * ft_pr, n=L)[: len(w) + n - 1]
    # first n values of tmp, taken in reverse order, give the gradient
    g = 2.0 * tmp[n - 1 :: -1]   # length n
    return g


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0):
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises `evaluate_sequence`.  The routine respects the 1000 s
    time limit and returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0      # safety margin

    best_score = float("inf")
    best_seq = None

    # --------------------------------------------------------------
    # 1) build a varied collection of initial vectors (seeds)
    # --------------------------------------------------------------
    seeds = []

    # previous best (if it exists in the global namespace)
    try:
        seeds.append(np.asarray(height_sequence_1, dtype=np.float64))
    except NameError:
        pass

    lengths = (2000, 4000, 8000)                 # we stay below 8k for speed
    for n in lengths:

        for p_exp in (0.45, 0.50, 0.55, 0.60):
            i = np.arange(n, dtype=np.float64)
            seq = (i + 1) ** (-p_exp)
            seeds.append(seq)

        for lam in (0.0005, 0.001, 0.002, 0.003):
            i = np.arange(n, dtype=np.float64)

            seeds.append(seq)

        # uniform

        # random Dirichlet‑like
        seeds.append(np.random.rand(n))

        base = np.random.rand(n)
        for _ in range(3):
            perm = np.random.permutation(n)
            seeds.append(base[perm])

    # --------------------------------------------------------------
    # 2) optimise each seed (projected gradient with soft‑max)
    # --------------------------------------------------------------
    def _optimise_one(p: np.ndarray):
        nonlocal best_score, best_seq, deadline

        # normalise to a probability vector
        if p.sum() == 0.0:

        p = p / p.sum()
        n = p.shape[0]

        # hyper‑parameters

            # ---- gradient step (soft‑max of the max) ----
            g = _gradient_softmax(p, conv, beta)

            # additive update then projection onto the simplex
            p = p - eta * g
            p = _project_simplex(p)

            if it % 1000 == 0:

            # ---- occasional random jitter to escape plateaus ----
            if np.random.rand() < 0.01:

                eps = 0.01 * min(p[i], p[j])
                p[i] = max(p[i] - eps, 0.0)
                p[j] = p[j] + eps
                p = _project_simplex(p)

        return False

    # run optimisation on each seed
    for idx, seed_vec in enumerate(seeds, 1):
        if time.time() >= deadline:
            break
        p0 = np.asarray(seed_vec, dtype=np.float64)
        # truncate overly long vectors (keeps per‑iteration cost reasonable)
        if p0.shape[0] > 8000:
            p0 = p0[:8000]
        # normalise once (the optimiser will re‑normalise each iteration)
        p0 = p0 / p0.sum() if p0.sum() > 0 else p0
        print(f"=== Seed {idx}/{len(seeds)} | length={p0.shape[0]} ===")
        if _optimise_one(p0):
            break

    # --------------------------------------------------------------
    # 3) return the best sequence found
    # --------------------------------------------------------------
    print("Final best score:", best_score,
          "length:", len(best_seq) if best_seq is not None else None)
    return best_seq if best_seq is not None else []   # empty list signals failure
