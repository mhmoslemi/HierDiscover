"""
Propose a step‑function (non‑negative coefficient vector) that minimises

    2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm mixes three ideas:

1. **Coarse shape sweep** – a few parametric symmetric shapes
   (linear, quadratic, exponential, sinusoidal) are generated,
   projected onto the ℓ₁‑simplex Σa = √(2 n) and the best is kept as
   a warm start.

2. **Projected Adam on a smoothed max** – the nondifferentiable
   max‑convolution is approximated by a log‑sum‑exp.  The gradient of
   the smoothed objective can be written as a convolution of the
   soft‑max weights with the reversed current sequence.  An Adam
   optimiser (first‑ and second‑moment estimates) updates the vector,
   after each step it is projected back onto the simplex.

3. **Simulated‑annealing style random transfers** – occasionally a
   tiny amount of mass is moved from one index to another; the move
   is accepted if it improves the objective, otherwise it may be
   accepted with a temperature‑dependent probability.  This helps to
   escape plateaus.

All operations are FFT‑based (O(n log n) per iteration) and the
search respects a 1000 s time budget.  The best sequence seen so far
is returned at the end.
"""

import time
import math
import numpy as np

# Helper functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a with itself using an FFT of power‑of‑two
    length.  Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())      # next power of two ≥ 2n‑1
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex
        {x | x_i ≥ 0, Σ_i x_i = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]                     # sorted descending
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):                     # all zero case
        return np.full_like(v, target / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def generate_symmetric_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
) -> np.ndarray:
    """
    Produce a symmetric shape of length n.
    kind ∈ {linear, quadratic, exponential, sinusoid}.
    ``param`` controls the edge‑enhancement (≥0).
    The result is projected to the ℓ₁‑simplex with sum = target_sum.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "sinusoid":
        raw = 1.0 + param * (1 - np.cos(np.pi * idx / (n - 1)))
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.02) -> np.ndarray:
    """
    Transfer a small amount of mass from a random index to another,
    then project onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]                     # proportional to source mass
    if delta <= 0.0:
        return a
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    return project_onto_simplex(np.maximum(a2, 0.0), target_sum)


def smoothed_gradient(a: np.ndarray, max_conv: float, tau_factor: float) -> np.ndarray:
    """
    Compute the gradient of the log‑sum‑exp smooth approximation of the
    max self‑convolution.  Returns a vector of length n.
    """
    n = a.shape[0]
    # 1) compute the full convolution (length 2n‑1)
    conv = conv_self_fft(a)
    # 2) softmax weights
    tau = max(tau_factor * max_conv, 1e-12)
    w = np.exp((conv - max_conv) / tau)      # stability: subtract max_conv
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # 3) gradient = (2/τ) * (w * a_rev) restricted to indices n‑1 … n‑1+n‑1
    a_rev = a[::-1]
    L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
    grad = (2.0 / tau) * conv_full[n - 1 : n - 1 + n]
    return grad

            a = project_onto_simplex(np.maximum(a + jitter, 0.0), target_sum)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"Iter {iteration:07d} – jitter applied (stagnation)")

        # periodic logging (every ~30 s)
        if time.time() - last_log > 30.0:
            last_log = time.time()
            elapsed = time.time() - start_time
            print(f"[{elapsed:.1f}s] iter {iteration:,}  best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final polishing with tiny random transfers
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Gradient descent finished; best max_conv = {best_score:.6f}")

    for _ in range(40):
        if time.time() >= deadline:
            break
        cand = random_pair_transfer(best_seq.copy(), target_sum, scale=0.015)
        cand_max = conv_self_fft(cand).max()
        if cand_max < best_score:
            best_score = cand_max
            best_seq = cand.copy()
            print(f"Polish improved to {best_score:.6f}")

    # Return as plain Python list (evaluate_sequence expects a list)
    return best_seq.tolist()
