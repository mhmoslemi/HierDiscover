propose_candidate – exponentiated‑gradient search with soft‑max smoothing.

The algorithm starts from the best known sequence (the global
``height_sequence_1`` if present, otherwise a simple exponential shape).
The sequence lives on the probability simplex (non‑negative entries that
sum to one).  At each iteration we

1. compute the auto‑convolution `c = x * x` via an FFT,
2. form a soft‑max distribution `p_k ∝ exp(α·(c_k−max(c)))` (the
   smoothing parameter α is slowly increased from 0.5 to ≈200),
3. obtain the gradient of the smoothed maximum:
        ∇f_i = 2·∑_j p_{i+j}·x_j   (a correlation, also computed by FFT),
4. perform an exponentiated‑gradient (mirror‑descent) update
        x_i ← x_i·exp(−η·∇f_i)   with a decreasing step‑size η,
5. renormalise to keep the ℓ¹‑norm equal to one.

Every few thousand iterations a block of tiny entries is appended,
allowing the mass to spread over a longer support (the objective
contains the factor `n·max(conv)`).  If no improvement is seen for a
while the current vector is mixed with a fresh random Dirichlet
distribution to escape possible stagnation.

The objective being minimised is
        obj(x) = 2·n·max(x∗x)               (sum(x)=1),
which is exactly the value returned by ``evaluate_sequence`` after the
internal normalisations.  The routine respects a 1000 s wall‑clock
budget and returns the best sequence found.
"""

# Helper utilities – all top‑level, no closures

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray) -> np.ndarray:
    """
    Linear convolution of x with itself using a real FFT.
    Returns an array of length 2·len(x)−1.
    """
    n = x.shape[0]
    m = 2 * n - 1
    size = _next_pow2(m)
    X = np.fft.rfft(x, n=size)
    conv = np.fft.irfft(X * X, n=size)[:m]
    return conv

    """
    Linear cross‑correlation `corr[i] = Σ_j p[i+j]·x[j]` for i=0,…,len(x)−1.
    Here `p` has length 2·len(x)−1 (the same length as the auto‑convolution).
    The operation is performed via FFT for O(n log n) speed.
    """
    n = x.shape[0]
    L = p.shape[0]                     # L = 2·n−1
    size = _next_pow2(L + n - 1)       # length needed for linear convolution
    P = np.fft.rfft(p, n=size)
    X = np.fft.rfft(x, n=size)
    # correlation = ifft( P * conj(X) )
    corr_full = np.fft.irfft(P * np.conj(X), n=size)
    # the valid part corresponds to indices n‑1 … L‑1 (inclusive)
    start = n - 1
    end = L
    return corr_full[start:end]        # length = n


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Used only as a fallback when the exponentiated
    update would produce NaNs.
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _evaluate(seq: np.ndarray) -> float:
    """Thin wrapper around the supplied ``evaluate_sequence``."""
    try:
        return evaluate_sequence(seq.tolist())
    except Exception:
        return float("inf")

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the 1000 s time budget.

    Returns the best sequence (as a plain Python list) discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # 0 – initialise from a known good construction if available
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    if "height_sequence_1" in globals():
        try:
            h = np.asarray(height_sequence_1, dtype=float)
            if h.size >= 2:
                h = h / h.sum()                     # lie on the simplex
                cur_score = 2.0 * h.size * np.max(_conv_fft(h))
                if np.isfinite(cur_score):
                    best_seq = h.tolist()
                    best_score = float(cur_score)
                    print(f"[init] global seed score = {best_score:.6f}, n={h.size}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1 – fallback simple analytic seed if no global present
    # ------------------------------------------------------------------
    if best_seq is None:
        # Exponential decay – a surprisingly robust starter
        n0 = 2000
        lam = 0.0008
        idx = np.arange(n0, dtype=np.float64)
        h = np.exp(-lam * idx)
        h = h / h.sum()
        best_seq = h.tolist()
        best_score = float(2.0 * h.size * np.max(_conv_fft(h)))
        print(f"[init] exponential seed score = {best_score:.6f}, n={h.size}")

    # ------------------------------------------------------------------
    # 2 – main optimisation loop (exponentiated gradient + soft‑max)
    # ------------------------------------------------------------------
    x = np.asarray(best_seq, dtype=np.float64)
    x = x / x.sum()                               # ensure exact sum‑1

    iteration = 0
    last_improve = 0
    grow_interval = 3000
    max_len = 20000                               # generous upper bound
    extra_block = 200                             # entries added at each growth step

    # smoothing schedule: α starts small (smooth) and grows to ≈200
    alpha_start = 0.5
    alpha_target = 200.0
    alpha_ramp_iters = 6000                       # number of iters over which α grows

    while time.time() < deadline:
        iteration += 1
        n = x.shape[0]

        # ----- convolution & objective -----
        conv = _conv_fft(x)
        max_c = conv.max()
        cur_obj = 2.0 * n * max_c

        # ----- stopping test -----
        if cur_obj < best_score - 1e-9:
            best_score = float(cur_obj)
            best_seq = x.tolist()
            last_improve = iteration
            print(f"[iter {iteration:6d}] new best = {best_score:.6f}, n={n}")
            if best_score < 1.503:
                print("[info] target reached – exiting early.")
                break

        # ----- adapt smoothing factor α -----
        if iteration <= alpha_ramp_iters:
            alpha = alpha_start + (alpha_target - alpha_start) * (iteration / alpha_ramp_iters)
        else:
            alpha = alpha_target

        # ----- soft‑max weights over the convolution -----
        # weights sum to 1; the maximal entry gets weight 1, others <1.
        w = np.exp(alpha * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:                     # numerically impossible but guard
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # ----- gradient of the smoothed max (correlation) -----
        # ∇_i f = 2 * Σ_j w_{i+j} * x_j   = 2·corr(w, x)
        grad = 2.0 * _corr_fft(w, x)

        # ----- exponentiated‑gradient (mirror descent) step -----
        # Learning rate η decreases slowly.
        eta = 0.6 / (1.0 + 0.0002 * iteration)
        # Multiplicative update; guarantees non‑negativity.
        x = x * np.exp(-eta * grad)

        # ----- renormalise onto the simplex -----
        s = x.sum()
        if not np.isfinite(s) or s <= 0.0:
            # fall back to a uniform distribution of the current length
            x = np.full_like(x, 1.0 / n)
        else:
            x = x / s

        # ------------------------------------------------------------------
        # occasional length growth – distributing a few extra tiny masses
        # ------------------------------------------------------------------
        if iteration % grow_interval == 0 and n < max_len:
            extra = np.full(extra_block, 1e-12, dtype=np.float64)
            x = np.concatenate([x, extra])
            x = x / x.sum()                     # re‑normalise
            print(f"[grow] iteration {iteration} – length extended to {x.shape[0]}")

        # ------------------------------------------------------------------
        # occasional random restart if stagnation lasts too long
        # ------------------------------------------------------------------
        if iteration - last_improve > 8000:
            # blend current vector with a fresh Dirichlet sample
            rnd = np.random.rand(x.shape[0])
            rnd /= rnd.sum()
            x = 0.5 * x + 0.5 * rnd
            x /= x.sum()
            last_improve = iteration
            print(f"[restart] iteration {iteration} – mixing in random Dirichlet")

    # ------------------------------------------------------------------
    # final reporting
    # ------------------------------------------------------------------
    final_obj = 2.0 * len(best_seq) * np.max(_conv_fft(np.asarray(best_seq, dtype=np.float64)))
    print(f"[final] best score = {best_score:.6f} (re‑computed {final_obj:.6f}), length = {len(best_seq)}")
    return best_seq
