def _project_simplex(v, s):
    # v is numpy array, s target sum > 0.
    # return projection onto {x >=0, sum x = s}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def _full_convolution(a):
    n = a.shape[0]
    conv_len = 2*n -1
    fft_len = 1 << (conv_len-1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa*Fa, n=fft_len)[:conv_len]
    return conv

Algorithm Overview
------------------
* **Projected Gradient Descent on a smooth max** –  The true objective
  `max_k (a*a)[k]` is non‑differentiable.  We replace it by a
  smooth approximation
      f_β(a) = (1/β)·log∑_k exp(β·(a*a)[k])

      {a ≥ 0 , Σ a_i = √(2·n)}.
* **Annealing of the smoothness parameter** –  β starts small (the
  objective is very smooth) and is increased gradually, allowing the
  optimiser to first move globally and then focus on the true max.
* **Random restarts & diverse initial shapes** –  For each restart a

def _full_convolution(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n = x.shape[0] + y.shape[0] - 1
    fft_len = 1 << (n - 1).bit_length()  # next power of two
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[:n]
    return conv


def _conv_max(a: np.ndarray):
    """Return (full convolution, maximal value, index of the maximum)."""
    conv = _full_convolution(a, a)
    max_val = float(conv.max())
    max_idx = int(np.argmax(conv))
    return conv, max_val, max_idx


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

    if family == 0:
        # Dirichlet / uniform random
        a = rng.random(n)
    elif family == 1:
        # Exponential decay
        tau = rng.uniform(0.1, 5.0) * n
        a = np.exp(-np.arange(n) / tau)
    else:
        # Symmetric triangular peak
        centre = n // 2
        a = np.maximum(0.0, 1.0 - np.abs(np.arange(n) - centre) / (centre + 1))

    # Small jitter for diversity
    a += rng.normal(scale=a.mean() * 0.01, size=n)
    a = np.clip(a, 0.0, None)

    # Normalise to the required L¹ norm
    a = a / a.sum() * S
    return a


def _gradient_step_smooth(a: np.ndarray, beta: float, eta: float, S: float,
                         rng: np.random.Generator) -> np.ndarray:
    """
    One projected gradient descent step on the smoothed max objective.
    - beta : smoothing parameter (higher = closer to true max)
    - eta  : step size
    - S    : target L¹ norm (√(2·n))
    """
    n = a.shape[0]

    # 1) full convolution a*a
    conv = _full_convolution(a, a)                     # length 2n‑1
    max_val = conv.max()
    # 2) soft‑max weights (log‑sum‑exp trick)
    scaled = conv - max_val                             # max becomes 0
    exp_scaled = np.exp(beta * scaled)                  # all ≤ 1
    Z = exp_scaled.sum()
    w = exp_scaled / Z                                  # length 2n‑1, non‑negative

    # 3) gradient = 2 * (w * a_rev)  , a_rev = a[::-1]
    a_rev = a[::-1]
    grad_full = _full_convolution(w, a_rev)             # length 3n‑2
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]          # slice of length n

    # 4) additive update + projection onto simplex
    a_new = a - eta * grad
    a_proj = _project_simplex(a_new, S)
    return a_proj


def _max_subgradient_step(a: np.ndarray, eta: float) -> np.ndarray:
    """
    Multiplicative sub‑gradient step that only looks at the current maximal
    convolution entry.  This is the same operation used in the reference
    implementation.
    """
    n = a.shape[0]
    _, _, k = _conv_max(a)

    idx = np.arange(n)
    partner = k - idx
    valid = (partner >= 0) & (partner < n)

    g = np.zeros_like(a)
    g[valid] = 2.0 * a[partner[valid]]
    if k % 2 == 0:
        d = k // 2
        if 0 <= d < n:
            g[d] = a[d]

    a_new = a * np.exp(-eta * g)

    S = np.sqrt(2.0 * n)
    total = a_new.sum()
    if total == 0.0:
        a_new = np.full(n, S / n)
    else:
        a_new *= (S / total)
    return a_new


def _optimize_gradient(initial_a: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smoothed max followed by a short
    multiplicative sub‑gradient polishing phase.
    Returns the best vector encountered and its (true) maximal convolution.
    """
    a = initial_a.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # initialise best
    _, cur_max, _ = _conv_max(a)
    best_a = a.copy()
    best_score = cur_max

    # hyper‑parameters
    beta = 0.5               # start smoothness
    beta_inc = 1.02          # geometric increase (≈ annealing)
    beta_max = 20.0
    eta = 0.25               # step size for gradient descent
    eta_decay = 0.9995
    jitter_prob = 0.01

    # ----- gradient descent (smooth max) -----
    while time.time() < deadline:
        a = _gradient_step_smooth(a, beta, eta, S, rng)

        # occasional jitter to avoid stagnation
        if rng.random() < jitter_prob:
            a += rng.normal(scale=a.mean() * 0.01, size=n)
            a = np.clip(a, 0.0, None)
            a = _project_simplex(a, S)

        # evaluate true max (expensive but cheap enough for occasional logging)
        _, cur_max, _ = _conv_max(a)
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            # optional log
            # print(f"   [grad] new best max={best_score:.6f}")

        # update schedules
        beta = min(beta * beta_inc, beta_max)
        eta *= eta_decay

        # early exit if eta becomes tiny
        if eta < 1e-8:
            break

    # ----- short polishing with actual max sub‑gradient -----
    # Use a fixed small number of multiplicative steps; they are cheap.
    polishing_steps = 200
    polishing_eta = 0.1
    for _ in range(polishing_steps):
        a = _max_subgradient_step(a, polishing_eta)
        _, cur_max, _ = _conv_max(a)
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
        polishing_eta *= 0.995  # slight decay

    return best_a, best_score


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Entry point for the evaluation harness.
    Performs many independent short optimisation runs within the time budget
    and returns the best non‑negative step‑function found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 1.0   # a tiny safety margin

    best_overall_seq = None
    best_overall_score = np.inf
    attempt = 0

    # ------------------------------------------------------------------
    # Optional: try to improve the previously discovered sequence, if any.
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    # ------------------------------------------------------------------
    # Main loop: many random restarts.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        attempt += 1

        # Randomly pick a length (larger n gives more search space but is slower)
        n = rng.integers(800, 5000 + 1)

        # Build an initial candidate
        if base_seq is not None and base_seq.size >= n:
            # Use a slice of the known good sequence with a tiny jitter
            a0 = base_seq[:n].astype(float).copy()
            a0 += rng.normal(scale=a0.mean() * 0.001, size=n)
            a0 = np.clip(a0, 0.0, None)
            S = np.sqrt(2.0 * n)
            a0 = a0 / a0.sum() * S
        else:
            a0 = _generate_initial_seq(n, rng)

        # Allocate a time slice for this restart (≈15 s or whatever remains)
        slice_deadline = min(time.time() + 15.0, deadline)

        # Optimise
        a_opt, score = _optimize_gradient(a0, slice_deadline, rng)

        # Keep the best overall result
        if score < best_overall_score:
            best_overall_score = score
            best_overall_seq = a_opt.copy()
            print(f"[global] attempt={attempt} n={n} new best score={best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check through the provided black‑box evaluator.
    # ------------------------------------------------------------------
    if best_overall_seq is not None:
        final_seq = best_overall_seq.tolist()
        external_score = evaluate_sequence(final_seq)
        print(f"[final] best external score = {external_score:.6f} "
              f"(internal max = {best_overall_score:.6f}) "
              f"length = {len(final_seq)}")
        return final_seq
    else:
        # This should never happen, but return a trivial feasible vector.
        return [np.sqrt(2.0)]
