"""
Propose a step‑function (non‑negative sequence) that minimizes

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works in three stages:

1. **Diverse initialisation** – a small pool of random candidates
   (uniform, power‑law, Dirichlet, Gaussian bump) is built and the best
   one (according to a fast internal evaluator) is kept.

2. **Annealed Adam optimisation** – we minimise a smooth approximation
   of the “max‑convolution’’ using Adam with Euclidean projection onto
   the simplex `{x ≥ 0, sum(x)=√(2 n)}`.  Early iterations use a soft‑max
   (temperature γ) that spreads the gradient over many shifts; γ is
   multiplied each step until it reaches a large value (≈2000).  When
   γ becomes large we switch to a *top‑K* weighting, where only the K
   largest convolution entries receive weight (K is reduced together
   with γ).  This targets the active constraints more directly.

3. **Hard‑max fine‑tuning** – with γ fixed at a very large value we
   perform a few thousand tiny plain‑gradient steps (no Adam) that
   correspond to the true sub‑gradient of the max‑operator.  Small
   random jitter is added periodically to escape plateaus.

The whole search respects the 1000 s time budget and always returns the
best sequence found so far.  All arithmetic is performed with FFTs for
speed, and the sequence is kept on the simplex, so the evaluation value
is exactly the maximal convolution (because `sum = √(2 n)`).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of *v* onto the simplex
        { x : x_i ≥ 0 , Σ_i x_i = target_sum }.
    Implements the algorithm of Duchi et al. (ICML 2008).
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_score(a: np.ndarray) -> float:
    """
    Fast version of the supplied `evaluate_sequence`.  The vector *a* is
    assumed to be non‑negative; the sum is never zero (otherwise the
    function returns ``inf``).
    """
    n = a.shape[0]
    nfft = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    # because we keep the sum = √(2 n) this value equals max_c,
    # but we keep the original formula for robustness.
    return float(2.0 * n * max_c / (s * s))


def _initial_candidates(n: int, target_sum: float) -> list[np.ndarray]:
    """Generate a handful of diverse starting vectors."""
    rng = np.random.default_rng()
    cand = []

    # 1) uniform
    cand.append(np.full(n, target_sum / n))

    # 2) power‑law shapes
    for _ in range(12):
        p = rng.uniform(0.2, 1.0)
        off = rng.uniform(0.5, 5.0)
        a = 1.0 / np.power(np.arange(n, dtype=np.float64) + off, p)
        a = a * target_sum / a.sum()
        cand.append(a)

    # 3) Dirichlet draws (controlled concentration)
    for _ in range(8):
        alpha = rng.uniform(0.3, 5.0)
        a = rng.dirichlet(np.full(n, alpha)) * target_sum
        cand.append(a)

    # 4) Gaussian bump
    x = np.arange(n, dtype=np.float64)
    centre = n // 2
    sigma = rng.uniform(n / 20.0, n / 5.0)
    a = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    a = a * target_sum / a.sum()
    cand.append(a)

    # 5) reuse global construction if present
    try:
        global height_sequence_1  # noqa: PLW0603
        init = np.asarray(height_sequence_1, dtype=np.float64)
        if init.size > 0:

            init = np.maximum(init, 0.0)
            init = init * target_sum / init.sum()
            cand.append(init)
    except Exception:

    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    m = np.zeros_like(best_a)
    v = np.zeros_like(best_a)
    t_iter = 0

    # learning‑rate schedule
    lr = 0.5
    lr_min = 1e-4
    lr_decay = 0.9996

    # temperature (γ) schedule – we start gentle and anneal up to a
    # very large value that approximates the true max.
    gamma = 1.0
    gamma_max = 2000.0
    gamma_inc = 1.004

    iteration = 0
    a = best_a.copy()               # current iterate

    # --------------------------------------------------------------
    # 5. Main optimisation loop (soft‑max → top‑K annealing)
    # --------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1
        t_iter += 1

        # ----- convolution a*a -------------------------------------------------
        af = np.fft.rfft(a, n=nfft_conv)
        conv = np.fft.irfft(af * af, n=nfft_conv)[: 2 * n - 1]
        max_conv = conv.max()

        # ----- build weights ----------------------------------------------------
        if gamma < 50.0:                     # soft‑max region
            w = np.exp(gamma * (conv - max_conv))
            w_sum = w.sum()
            if w_sum == 0.0:
                w[:] = 1.0 / w.size
            else:
                w /= w_sum
        else:                                # top‑K region
            # K shrinks as γ grows – roughly K ≈ max(1, 2000/γ)
            K = int(max(1.0, 2000.0 / gamma))
            if K > conv.size:
                K = conv.size
            top_idx = np.argpartition(-conv, K - 1)[:K]
            w = np.zeros_like(conv)
            w[top_idx] = 1.0 / K

        # ----- gradient of the surrogate ---------------------------------------
        wf = np.fft.rfft(w, n=nfft_corr)
        af_rev = np.fft.rfft(a[::-1], n=nfft_corr)
        corr_full = np.fft.irfft(wf * af_rev, n=nfft_corr)
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]   # length‑n slice

        # ----- Adam update -----------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_onto_simplex(a_new, target_sum)

        # occasional tiny jitter to help escape flat regions
        if iteration % 127 == 0:
            jitter = 1e-5 * np.random.randn(n)
            a_new = project_onto_simplex(a_new + jitter, target_sum)

        # ----- evaluate candidate ------------------------------------------------
        score_new = _compute_score(a_new)

        # keep the best ever found
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            elapsed = time.time() - start_time
            print(
                f"Iter {iteration:6d}: new best {best_score:.6f} (elapsed {elapsed:.1f}s)"
            )

        # accept the update (Adam already does a kind of exploration)
        a = a_new

        # ----- schedule updates -------------------------------------------------
        gamma = min(gamma * gamma_inc, gamma_max)
        lr = max(lr * lr_decay, lr_min)

        # periodic time check
        if iteration % 500 == 0 and time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # 6. Fine‑tuning phase – true max sub‑gradient
    # --------------------------------------------------------------
    fine_iters = 5000
    # keep the Adam moments around – they are not needed any more
    lr_fine = 5e-4
    gamma_fine = gamma_max

    for _ in range(fine_iters):
        if time.time() >= deadline:
            break

        af = np.fft.rfft(a, n=nfft_conv)
        conv = np.fft.irfft(af * af, n=nfft_conv)[: 2 * n - 1]
        max_conv = conv.max()

        # hard‑max weighting (K = 1)
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0

        wf = np.fft.rfft(w, n=nfft_corr)
        af_rev = np.fft.rfft(a[::-1], n=nfft_corr)
        corr_full = np.fft.irfft(wf * af_rev, n=nfft_corr)
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]

        a = a - lr_fine * grad
        a = project_onto_simplex(a, target_sum)

        score_new = _compute_score(a)
        if score_new < best_score:
            best_score = score_new
            best_a = a.copy()
            elapsed = time.time() - start_time
            print(
                f"Fine‑tune: new best {best_score:.6f} (elapsed {elapsed:.1f}s)"
            )

    # --------------------------------------------------------------
    # 7. Return the best sequence discovered
    # --------------------------------------------------------------
    # as a plain Python list (the external driver expects list[float])
    return best_a.tolist()
