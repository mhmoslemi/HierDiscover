def _simulated_annealing(a0, deadline):
    a = a0.copy()
    n = a.size
    cur_val = _score(a)
    best_a = a.copy()
    best_val = cur_val

    # temperature schedule
    T0 = (cur_val) * 0.1  # maybe 10% of current
    T = T0
    decay = 0.9995

    rng = np.random.default_rng()
    while time.time() < deadline:
        # random move
        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        frac = rng.random() * 0.2   # up to 20% of a[i]
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta

        new_val = _score(a)
        delta_score = new_val - cur_val

        if delta_score < 0 or rng.random() < np.exp(-delta_score / (T+1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}]  SA improved = {best_val:.6f}")
        else:
            # revert
            a[j] -= delta
            a[i] += delta

        T *= decay
        # maybe occasionally reheat if stuck?
    return best_a, best_val

"""
Propose a step‑function (non‑negative heights) that minimises
    2·n·max(conv(a,a)) / (sum(a)**2)

The search consists of three stages:

1. **Parametric warm‑start** – random power‑law shapes a_i ∝ (i+δ)^(-p)
   (p≈0.5, δ>0) are generated, normalised to the required ℓ¹‑norm and
   the best one (lowest fast score) is kept as start point.

2. **Projected gradient descent** on a smooth log‑sum‑exp approximation
   of the maximum convolution.  The temperature β is annealed from 1 up
   to a large value (4096) while a step size inversely proportional to
   β·‖∇‖ is used.  After each gradient step we project onto the simplex
   Σa = √(2·n) and enforce symmetry a_i = a_{n‑1‑i} (the optimum is
   symmetric).  This stage runs until the allocated sub‑budget expires.

3. **Simulated‑annealing refinement** – random symmetric mass transfers
   are performed, accepting improving moves and occasionally worsening
   moves with probability exp(−Δ/Temp).  Temperature is slowly cooled.
   The best sequence seen during this phase is returned.

The whole routine tries several lengths and random restarts within the
overall time budget (default 1000 s) and finally returns the best
sequence found (trimmed of trailing zeros).  All helper functions are
top‑level; the fast `_score` function is used during the search.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """Fast evaluation of the competition objective (scale‑invariant)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:                                     # safety – should never happen
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i≥0, Σx_i = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of the soft‑max approximation of max(conv(a,a))
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b):
        b = a*a (full convolution, length 2·n‑1)
        g = ∇ (log‑sum‑exp_{β}(b)) = 2·(a ⋆ p)  with p_k ∝ exp(β·b_k)
    Both are computed with FFTs.
    """
    n = a.shape[0]

    # ---- b = a*a -------------------------------------------------------
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ---- soft‑max probabilities p ---------------------------------------
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)                # shape (2·n‑1,)

    # ---- gradient g = 2·(a ⋆ p) ---------------------------------------
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1        # = 3·n‑2
    size2 = 1 << ((conv_len).bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Projected gradient descent with annealed temperature
# ----------------------------------------------------------------------


def _annealed_gradient_descent_v2(a0: np.ndarray, deadline: float,
                                 max_beta: float = 4096.0) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the log‑sum‑exp surrogate of max(conv(a,a)).
    The simplex constraint (ℓ¹‑norm) and symmetry a_i = a_{n‑1‑i}
    are enforced after every step.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    step_base = target_sum * 0.25          # a bit larger than the previous 0.20
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        # gradient of the smooth max
        g, _ = _gradient_softmax(a, beta)

        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            lr = step_base / (norm_g * (beta + 1.0))
            a -= lr * g

        # project onto simplex and enforce symmetry
        a = _project_to_simplex(a, target_sum)
        a = (a + a[::-1]) * 0.5

        # evaluate
        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}]  beta={beta:.2f}  score={best_val:.6f}")

        # annealing schedule
        if it % 250 == 0 and beta < max_beta:
            beta = min(beta * 1.2, max_beta)

        # occasional jitter to escape plateaus
        if it % 1000 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            a = (a + a[::-1]) * 0.5

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (symmetric moves)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple SA that moves a random fraction of mass from one symmetric pair
    to another.  Accepts worsening moves with Metropolis probability.
    """
    a = a0.copy()
    n = a.size
    cur_val = _score(a)

    best_a = a.copy()
    best_val = cur_val

    # initial temperature proportional to current objective
    T = max(cur_val * 0.1, 1e-3)
    decay = 0.9995
    rng = np.random.default_rng()

    half = n // 2

    while time.time() < deadline:
        i = rng.integers(0, half)
        j = rng.integers(0, half)
        if i == j:
            continue

        # determine admissible transfer amount (must keep both sides ≥0)
        max_delta = min(a[i], a[n - 1 - i])
        if max_delta <= 0.0:
            continue

        frac = rng.random() * 0.15          # up to 15 % of the available mass
        delta = max_delta * frac

        # perform symmetric transfer
        a[i] -= delta
        a[j] += delta
        a[n - 1 - i] -= delta
        a[n - 1 - j] += delta

        new_val = _score(a)
        dE = new_val - cur_val
        if dE < 0.0 or rng.random() < np.exp(-dE / (T + 1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}]   SA improved = {best_val:.6f}")
        else:
            # revert the move
            a[i] += delta
            a[j] -= delta
            a[n - 1 - i] += delta
            a[n - 1 - j] -= delta

        T *= decay

    return best_a, best_val


# ----------------------------------------------------------------------
# Random parametric warm‑start (power‑law with shift)
# ----------------------------------------------------------------------


def _random_powerlaw_start(n: int, target_sum: float,
                           rng: np.random.Generator,
                           trials: int = 200) -> np.ndarray:
    """
    Sample a few (p,δ) pairs and keep the one with lowest fast score.
    a_i ∝ (i+δ)^(-p) ,  p≈0.5 , δ>0 .  The result is projected onto the simplex.
    """
    best_a = None
    best_val = np.inf

    i = np.arange(n, dtype=np.float64)

    for _ in range(trials):
        p = rng.uniform(0.35, 0.75)          # exponent around 0.5
        delta = rng.uniform(0.1, 10.0)       # shift to avoid singularity at 0
        raw = (i + delta) ** (-p)
        if raw.sum() == 0.0:
            continue
        a = raw * (target_sum / raw.sum())
        # small jitter for diversity
        a += rng.normal(scale=target_sum * 0.0005, size=n)
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()

    # fallback – simple uniform if nothing succeeded
    if best_a is None:
        best_a = np.ones(n, dtype=np.float64) * (target_sum / n)
    return best_a

# Main entry point required by the challenge
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the supplied
    `evaluate_sequence` objective.  The routine runs for at most
    `budget_s` seconds, trying several vector lengths and random restarts.
    Each restart consists of a parametric warm‑start, a gradient‑descent
    phase (annealed soft‑max) followed by a short simulated‑annealing
    refinement.  The best sequence found is returned.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    best_seq = None
    best_score = np.inf

    # vector lengths – longer vectors give more freedom at higher cost
    lengths = [1800, 2600, 3400, 4200, 5000]

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # we will perform a few restarts for each length
        for restart in range(5):
            if time.time() >= deadline:
                break

            # -------------------------------------------------------------
            # 1️⃣ warm‑start via random power‑law (p,δ) search
            # -------------------------------------------------------------
            a0 = _random_powerlaw_start(n, target_sum, rng, trials=250)

            # -------------------------------------------------------------
            # 2️⃣ gradient‑descent (annealed soft‑max)
            # -------------------------------------------------------------
            sub_deadline = min(deadline, time.time() + 25.0)   # ≤25 s per restart
            a_opt, opt_score = _annealed_gradient_descent_v2(a0, sub_deadline,
                                                             max_beta=4096.0)

            # -------------------------------------------------------------
            # 3️⃣ simulated‑annealing refinement
            # -------------------------------------------------------------
            if time.time() < deadline:
                a_opt, opt_score = _simulated_annealing(a_opt, deadline)

            # -------------------------------------------------------------
            # Keep global best
            # -------------------------------------------------------------
            if opt_score < best_score:
                best_score = opt_score
                best_seq = a_opt.copy()
                print(f"** NEW GLOBAL BEST ** n={n} restart={restart} "
                      f"score={best_score:.6f}")

    # Fallback – should never be needed
    if best_seq is None:
        n = 1000
        best_seq = np.ones(n, dtype=np.float64) * (np.sqrt(2.0 * n) / n)

    # Trim trailing zeros (if any) for a tidy output
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    return best_seq.tolist()
