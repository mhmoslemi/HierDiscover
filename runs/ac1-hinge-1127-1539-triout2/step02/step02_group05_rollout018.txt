        offset = np.random.uniform(0.5, 3.0)
        a = 1.0 / np.power(np.arange(n, dtype=np.float64) + offset, p)
        a = a[::-1]
        a = a * target_sum / a.sum()
        candidates.append(a.copy())
    # random Dirichlet
    for _ in range(10):
        raw = np.random.gamma(shape=0.5, scale=1.0, size=n)  # shape < 1 leads spiky
        a = raw / raw.sum() * target_sum
        candidates.append(a.copy())

    # evaluate each candidate, keep best
    best_a = None; best_score = np.inf
    for a in candidates:
        score = _compute_score(a)
        if score < best_score:
            best_score = score; best_a = a.copy()
            print(f"Init candidate score {score:.6f}")

    a = best_a.copy()

    # Precompute FFT sizes
    nfft_conv = _next_pow2(2 * n - 1)
    nfft_corr = _next_pow2(3 * n - 2)

    # Adam parameter
    beta1,beta2 = 0.9,0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_adam = 0

    # schedule parameters
    lr = 0.3
    lr_min = 0.001
    lr_decay = 0.9997

    gamma = 1.0
    gamma_max = 2000.0
    gamma_inc = 1.005

    # main loop
    iter_count = 0
    while time.time() < deadline:
        iter_count += 1
        t_adam += 1

        # convolution a*a
        af = np.fft.rfft(a, n=nfft_conv)
        conv = np.fft.irfft(af * af, n=nfft_conv)[:2*n-1]
        max_conv = conv.max()

        # softmax weights
        w = np.exp(gamma * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # gradient via convolution with reversed a
        wf = np.fft.rfft(w, n=nfft_corr)
        af_rev = np.fft.rfft(a[::-1], n=nfft_corr)
        corr_full = np.fft.irfft(wf * af_rev, n=nfft_corr)
        grad = 2.0 * corr_full[n-1:n-1+n]  # length n

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t_adam)
        v_hat = v / (1.0 - beta2 ** t_adam)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex
        a_new = project_onto_simplex(a_new, target_sum)

        # evaluate new candidate
        score_new = _compute_score(a_new)

        if score_new < best_score:
            best_score = score_new; best_a = a_new.copy()
            elapsed = time.time() - start_time
            print(f"Iter {iter_count:6d}: new best {best_score:.6f} (elapsed {elapsed:.1f}s)")

        a = a_new

        # schedule updates
        gamma = min(gamma * gamma_inc, gamma_max)
        lr = max(lr * lr_decay, lr_min)

        # occasional jitter
        if iter_count % 73 == 0:
            jitter = 1e-4 * np.random.randn(n)
            a = project_onto_simplex(a + jitter, target_sum)

        # time check each 1000
        if iter_count % 1000 == 0 and time.time() >= deadline:
            break

    # Phase 2: exact max subgradient descent
    # allocate remaining time (maybe up to 30% of total)
    fine_deadline = deadline
    lr2 = 5e-3
    lr2_decay = 0.9999
    eps_fine = 1e-8
    sub_iter = 0
    while time.time() < fine_deadline:
        sub_iter += 1

        # compute convolution
        af = np.fft.rfft(a, n=nfft_conv)
        conv = np.fft.irfft(af * af, n=nfft_conv)[:2*n-1]
        max_conv = conv.max()
        # find indices where conv close to max (within 1e-12 or relative)
        # Use absolute tolerance
        tol = max_conv * 1e-12 + 1e-12
        idxs = np.where(conv >= max_conv - tol)[0]
        # compute gradient sum over these indices
        grad = np.zeros_like(a)
        for k in idxs:
            # compute gradient for this k
            if k < n:
                grad[:k+1] += 2.0 * a[k::-1]
            else:
                start = k - (n - 1)
                # slice a[n-1: start-1: -1]
                grad[start:] += 2.0 * a[(n-1): (start-1): -1]
        # update a (simple gradient descent)
        a = a - lr2 * grad
        a = project_onto_simplex(a, target_sum)

        # evaluate
        score_new = _compute_score(a)
        if score_new < best_score:
            best_score = score_new; best_a = a.copy()
            elapsed = time.time() - start_time
            print(f"Fine {sub_iter:6d}: new best {best_score:.6f} (elapsed {elapsed:.1f}s)")

        # schedule lr decay
        lr2 = max(lr2 * lr2_decay, 1e-6)

        # time check
        if sub_iter % 1000 == 0 and time.time() >= fine_deadline:
            break

    # final return
    return best_a.tolist()

"""
Multi‑start optimisation of a non‑negative step‑function.

The objective is
    2 * n * max(conv(a, a)) / (sum(a)**2)

which, after normalising sum(a) = sqrt(2·n), simply equals
    max(conv(a, a))

The algorithm proceeds in three phases:

1. **Initialisation** – a collection of diverse feasible points
   (uniform, power‑law decreasing/increasing, Dirichlet‑sampled,
   and the optional globally supplied `height_sequence_1`) is generated
   and the best one is kept.

2. **Adam on a soft‑max surrogate** – the non‑smooth max is replaced by a
   temperature‑controlled soft‑max.  An Adam optimiser updates the
   coefficients and after each step the vector is projected back onto the
   simplex `{x ≥ 0, sum(x)=√(2n)}`.  The temperature (γ) is slowly increased
   up to a large value, turning the surrogate into the true max.

3. **Exact‑max fine‑tuning** – once the temperature is large the algorithm
   switches to a plain projected (sub‑)gradient descent that uses the true
   gradient of the maximal convolution entry (or the sum of gradients of all
   entries that are numerically tied for the maximum).  A small constant
   step size with a slow decay finishes the optimisation.

All heavy work (convolutions and their gradients) is performed with FFTs.
The routine respects the 1000 s time‑budget and prints intermediate scores.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ *x*."""
    return 1 << (x - 1).bit_length()


def _compute_score(a: np.ndarray) -> float:
    """
    Fast version of the provided `evaluate_sequence`.
    Assumes `a` is non‑negative and finite.
    """
    n = a.shape[0]
    nfft = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    # the returned value equals 2*n*max_c / s**2
    return float(2.0 * n * max_c / (s * s))


def project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of `v` onto the simplex
        { x : x_i ≥ 0 , sum_i x_i = target_sum }.
    """
    # ---- Duchi et al. (ICML 2008) ----
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(len(v)) + 1  # 1‑based
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    the evaluation function.  The routine runs for at most ``budget_s``
    seconds and prints progress information.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Choose the length of the sequence
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1")
    if init_seq is not None:
        try:
            init_seq = np.asarray(init_seq, dtype=np.float64)
            n = init_seq.shape[0]
        except Exception:
            init_seq = None
            n = 8000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # 2. Build a pool of initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # (a) uniform
    candidates.append(np.full(n, target_sum / n, dtype=np.float64))

    # (b) reuse the globally supplied construction if present
    if init_seq is not None:
        a = np.maximum(init_seq, 0.0)
        s = a.sum()
        if s > 0:
            a = a * target_sum / s
            candidates.append(a.copy())

    # (c) decreasing power‑law families
    for p in np.linspace(0.30, 0.90, 5):
        offset = np.random.uniform(0.5, 3.5)
        a = 1.0 / np.power(np.arange(n, dtype=np.float64) + offset, p)
        a = a * target_sum / a.sum()
        candidates.append(a)

    # (d) increasing power‑law families (reverse of decreasing)
    for p in np.linspace(0.30, 0.90, 5):
        offset = np.random.uniform(0.5, 3.5)
        a = 1.0 / np.power(np.arange(n, dtype=np.float64) + offset, p)
        a = a[::-1] * target_sum / a.sum()
        candidates.append(a)

    # (e) random Dirichlet / Gamma samples – useful for sparse starts
    for _ in range(12):
        raw = np.random.gamma(shape=0.5, scale=1.0, size=n)
        a = raw / raw.sum() * target_sum
        candidates.append(a)

    # ------------------------------------------------------------------
    # 3. Pick the best of the initial pool
    # ------------------------------------------------------------------
    best_a = None
    best_score = np.inf
    for a in candidates:
        sc = _compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"Initial candidate score {sc:.6f}")

    a = best_a.copy()   # current iterate

    # ------------------------------------------------------------------
    # 4. Pre‑compute FFT sizes
    # ------------------------------------------------------------------
    nfft_conv = _next_pow2(2 * n - 1)        # for a * a
    nfft_corr = _next_pow2(3 * n - 2)        # for w ⊛ a_rev

    # ------------------------------------------------------------------
    # 5. Adam optimisation on a smooth‑max surrogate
    # ------------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_adam = 0

    # learning‑rate schedule
    lr = 0.30
    lr_min = 0.001
    lr_decay = 0.9997

    # soft‑max temperature schedule
    gamma = 1.0
    gamma_max = 2000.0
    gamma_inc = 1.005

    iter_cnt = 0

    while time.time() < deadline:
        iter_cnt += 1
        t_adam += 1

        # ---- convolution a * a ------------------------------------------------
        af = np.fft.rfft(a, n=nfft_conv)
        conv = np.fft.irfft(af * af, n=nfft_conv)[: 2 * n - 1]
        max_conv = conv.max()

        # ---- soft‑max weights --------------------------------------------------
        w = np.exp(gamma * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the smooth max (via correlation) --------------------
        wf = np.fft.rfft(w, n=nfft_corr)
        af_rev = np.fft.rfft(a[::-1], n=nfft_corr)
        corr_full = np.fft.irfft(wf * af_rev, n=nfft_corr)
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]  # length‑n slice

        # ---- Adam parameter update --------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t_adam)
        v_hat = v / (1.0 - beta2 ** t_adam)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- projection onto the simplex (non‑negative, fixed L1) -------------
        a_new = project_onto_simplex(a_new, target_sum)

        # ---- occasional jitter to escape flat regions -------------------------
        if iter_cnt % 73 == 0:
            jitter = 1e-4 * np.random.randn(n)
            a_new = project_onto_simplex(a_new + jitter, target_sum)

        # ---- evaluate -----------------------------------------------------------
        score_new = _compute_score(a_new)

        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            elapsed = time.time() - start_time
            print(
                f"Iter {iter_cnt:6d}: new best {best_score:.6f} (elapsed {elapsed:.1f}s)"
            )

        a = a_new

        # ---- schedule updates --------------------------------------------------
        gamma = min(gamma * gamma_inc, gamma_max)
        lr = max(lr * lr_decay, lr_min)

        # safety check every 1000 iterations
        if iter_cnt % 1000 == 0 and time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 6. Fine‑tuning with the exact max (sub‑gradient) --------------------
    # ------------------------------------------------------------------
    # Remaining time (if any) goes to this phase.
    fine_deadline = deadline
    lr_fine = 5e-3
    lr_fine_decay = 0.9999
    sub_iter = 0

    while time.time() < fine_deadline:
        sub_iter += 1

        # current convolution
        af = np.fft.rfft(a, n=nfft_conv)
        conv = np.fft.irfft(af * af, n=nfft_conv)[: 2 * n - 1]
        max_conv = conv.max()

        # indices that are (numerically) maximal
        tol = max_conv * 1e-12 + 1e-12
        max_idx = np.where(conv >= max_conv - tol)[0]

        # gradient of the true max (sum of sub‑gradients if several ties)
        grad = np.zeros_like(a)
        for k in max_idx:
            if k < n:
                # overlap = 0 .. k
                grad[: k + 1] += 2.0 * a[k::-1]
            else:
                start = k - (n - 1)
                # overlap = start .. n‑1
                grad[start:] += 2.0 * a[(n - 1) : (start - 1) : -1]

        # simple projected gradient step
        a = a - lr_fine * grad
        a = project_onto_simplex(a, target_sum)

        # evaluate
        score_new = _compute_score(a)
        if score_new < best_score:
            best_score = score_new
            best_a = a.copy()
            elapsed = time.time() - start_time
            print(
                f"Fine {sub_iter:6d}: new best {best_score:.6f} (elapsed {elapsed:.1f}s)"
            )

        # decay learning rate gently
        lr_fine = max(lr_fine * lr_fine_decay, 1e-6)

        if sub_iter % 1000 == 0 and time.time() >= fine_deadline:
            break

    # ------------------------------------------------------------------
    # 7. Return the best sequence discovered
    # ------------------------------------------------------------------
    return best_a.tolist()
