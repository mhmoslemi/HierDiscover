"""
Non‑negative step‑function optimizer using a smooth‑max objective and Adam.

The original problem is
    f(a) = 2 * n * max(conv(a, a)) / (sum(a)**2)

Because the score is scale‑invariant we restrict the search to
probability vectors p (p_i ≥ 0, Σ p_i = 1).  Then
    f(p) = 2 * n * max(conv(p, p))

The non‑smooth max is replaced by a differentiable soft‑max
(β‑scaled log‑sum‑exp).  The gradient of the smoothed objective is
computed analytically using FFT‑based convolutions, and the parameters
are updated with Adam on the log‑softmax (logits) of p.  β is increased
gradually (continuation) so the optimisation first sees a smooth landscape
and finally approximates the true ℓ∞‑norm.

A diverse set of seeds (previous best, power‑law, exponential,
uniform, random Dirichlet) is generated, each is refined for a limited
number of iterations, and the best sequence encountered is returned.
The routine respects a 1000 s wall‑time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _full_conv_fft(p: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a real vector with itself using FFT.
    Returns a real array of length 2*len(p)-1.
    """
    n = p.shape[0]
    L = 1 << ((2 * n - 2).bit_length())
    ft = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(ft * ft, n=L)[: 2 * n - 1]
    conv = np.real(conv)
    conv[conv < 0] = 0.0
    return conv


def _conv_fft_general(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors a (len A) and b (len B)
    using FFT.  Returns an array of length A+B‑1.
    """
    A = a.shape[0]
    B = b.shape[0]
    L = 1 << ((A + B - 2).bit_length())
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)[: A + B - 1]
    conv = np.real(conv)
    conv[conv < 0] = 0.0
    return conv


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def _generate_powerlaw(p_exp: float, sigma: float,
                       max_len: int = 4000, eps: float = 1e-9) -> np.ndarray:
    """Truncated power‑law a_i = (i+σ)^{-p_exp}."""
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p_exp)
    mask = a > eps
    return a[mask]


def _generate_exponential(lam: float, max_len: int = 4000,
                          eps: float = 1e-9) -> np.ndarray:
    """Truncated exponential a_i = exp(-lam * i)."""
    i = np.arange(max_len, dtype=np.float64)
    a = np.exp(-lam * i)
    mask = a > eps
    return a[mask]

# ----------------------------------------------------------------------

    # random Dirichlet‑like seeds
    for n in (2000, 3000, 4000):

    # ---------------------------------------------------------------
    # 2) Optimise each seed with Adam on a smooth approximation of max
    # ---------------------------------------------------------------
    max_len = 4000          # we keep all vectors at most this long
    beta_start = 5.0
    beta_end = 50.0
    beta_steps = 6000       # number of inner iterations over which β grows
    base_lr = 0.05
    max_iters_per_seed = 8000

    for base in seeds:
        if time.time() >= deadline:
            break

        p0 = np.asarray(base, dtype=np.float64).ravel()
        if p0.sum() == 0.0:
            continue

        # truncate / pad to max_len
        if p0.size > max_len:
            p0 = p0[:max_len]
        elif p0.size < max_len:
            # pad with small uniform tail to reach max_len
            pad = np.full(max_len - p0.size, 1e-12, dtype=np.float64)
            p0 = np.concatenate([p0, pad])

        # normalise to a probability vector
        p0 = p0 / p0.sum()
        n = p0.shape[0]

        # initialise logits (log‑softmax) – start close to the seed
        y = np.log(p0 + 1e-15)

        v = np.zeros_like(y)          # second moment for Adam
        t_iter = 0

        conv0 = _full_conv_fft(p0)
        cur_score = 2.0 * n * np.max(conv0)

            t_iter += 1

            y_max = np.max(y)

            beta = beta_start + (beta_end - beta_start) * min(1.0, it / (beta_steps - 1))

            z = beta * conv
            z_max = np.max(z)
            w = np.exp(z - z_max)
            w_sum = w.sum()
            w = w / w_sum

            # ----- gradient of smooth objective w.r.t. p -----
            # g_i = 4 * n * Σ_k w_k * p_{k-i}
            w_rev = w[::-1]                         # reverse for convolution trick
            conv_rev = _conv_fft_general(p, w_rev)  # length = n + (2n-1) - 1 = 3n-2
            start = n - 1
            end = start + n
            g_raw = 4.0 * n * np.flip(conv_rev[start:end])

            # ----- convert gradient to logits (softmax parameter) -----
            pg = np.dot(p, g_raw)
            grad_y = p * (g_raw - pg)   # ∂J/∂y = p ⊙ (∂J/∂p - ⟨p,∂J/∂p⟩)

            # ----- Adam update on logits -----
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y * grad_y)
            m_hat = m / (1.0 - beta1 ** t_iter)
            v_hat = v / (1.0 - beta2 ** t_iter)
            y -= base_lr * m_hat / (np.sqrt(v_hat) + eps)

            # ----- occasional jitter to escape plateaus -----
            if np.random.rand() < 0.01:
                idx = np.random.randint(n)
                delta = (np.random.rand() - 0.5) * 0.02 * p[idx]
                p[idx] = max(p[idx] + delta, 0.0)
                p = p / p.sum()
                y = np.log(p + 1e-15)   # re‑initialise logits from jittered p

    # ---------------------------------------------------------------
    # 3) Return best sequence discovered
    # ---------------------------------------------------------------
    print("Final best score:", best_score, "length:", len(best_seq) if best_seq else None)
    return best_seq if best_seq is not None else []  # empty list signals failure
