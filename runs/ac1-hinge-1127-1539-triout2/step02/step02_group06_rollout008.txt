def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # determine n
    base = globals().get('height_sequence_1')
    if isinstance(base, (list, tuple)):
        n = len(base)
    else:
        n = 12000  # default

    target_sum = np.sqrt(2.0 * n)

    k = 30  # param count

    # Obtain initial param vector from base if available else random decreasing
    if isinstance(base, (list, tuple)):
        base_arr = np.array(base, dtype=np.float64)
        # Resample to k points using interpolation
        xs = np.linspace(0.0, 1.0, len(base_arr))
        knots = np.linspace(0.0, 1.0, k)
        v0 = np.interp(knots, xs, base_arr)
        # ensure nonnegative and monotonic decreasing
        v0 = np.clip(v0, 0.0, None)
        v0 = np.sort(v0)[::-1]
    else:
        # random decreasing initial vector
        v0 = np.random.rand(k)
        v0 = np.sort(v0)[::-1]

    # define helper to produce full seq and evaluate
    def full_seq_from_params(v):
        # v should be sorted decreasing else sort
        v = np.sort(v)[::-1]
        # linear interpolation
        t_knots = np.linspace(0.0, 1.0, len(v))
        xs = np.linspace(0.0, 1.0, n)
        seq = np.interp(xs, t_knots, v)
        # ensure nonnegative
        seq = np.clip(seq, 0.0, None)
        seq = project_to_sum(seq, target_sum)
        return seq

    # initial sequence
    a = full_seq_from_params(v0)
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    cur_seq = a.copy()
    cur_score = best_score
    cur_v = v0.copy()

    # Simulated annealing parameters
    start_temp = 0.1 * best_score  # temperature size relative to score
    end_temp = 1e-4
    total_sae_time = min(deadline - time.time(), 600.0) # allocate up to 600 s for SA
    # we will adjust temperature linearly with time
    # sigma for parameter perturbation
    sigma = 0.3 * np.mean(cur_v) if np.mean(cur_v) > 0 else 0.1

    iter_count = 0
    accept_count = 0
    last_improve_iter = 0

    print(f"[{time.time():.2f}] Starting SA optimization (n={n}, k={k})")
    while time.time() < deadline - 200:  # reserve time for final gradient
        # compute remaining time for SA
        elapsed = time.time() - (deadline - total_sae_time)  # not needed
        # Compute temperature schedule based on fraction of remaining SA time
        frac = (time.time() - (deadline - total_sae_time)) / total_sae_time
        # but if exceed, break
        # We'll compute t = max(0,1-frac)
        frac = max(0.0, min(1.0, (time.time() - (deadline - total_sae_time)) / total_sae_time))
        temperature = start_temp * (1 - frac) + end_temp * frac

        # propose new param
        cand_v = cur_v + sigma * np.random.randn(k)
        cand_v = np.clip(cand_v, 0.0, None)
        cand_v = np.sort(cand_v)[::-1]

        cand_seq = full_seq_from_params(cand_v)
        cand_score = evaluate_sequence(cand_seq.tolist())
        iter_count += 1

        if cand_score + 1e-12 < cur_score:
            cur_v = cand_v
            cur_seq = cand_seq
            cur_score = cand_score
            accept_count += 1
            if cand_score < best_score:
                best_score = cand_score

"""
Optimization of a non‑negative step function for the objective

    2·n·max(a⋆a) / (∑a)²

This implementation consists of two phases:

1. **Low‑dimensional simulated‑annealing** – the sequence is represented by a
   small set of monotone decreasing knot values (k≈30).  Linear interpolation
   produces a full‑length vector which is then normalised to the required ℓ₁‑norm.
   A simple Metropolis schedule explores the parameter space, keeps the best
   candidate and occasionally restarts after stagnation.

2. **Fine‑grained smooth‑max gradient descent** – the best vector from phase 1
   is refined with the soft‑max gradient algorithm used previously (FFT‑based
   convolution, projection onto the ℓ₁‑hyperplane, momentum and temperature
   cooling).  A few random pair‑wise mass transfers / jitter steps help to
   escape residual plateaus.

The whole search respects the supplied wall‑clock budget and always returns
the best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures / lambdas)
# ----------------------------------------------------------------------


def fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors a and b using an FFT of power‑of‑2 length."""
    n, m = a.shape[0], b.shape[0]
    L = 1 << ((n + m - 1).bit_length())           # next power of two
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)
    return conv[: n + m - 1]


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (length 2·len(a)-1)."""
    return fft_convolve(a, a)


def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector so that its ℓ₁‑norm equals *target*."""
    s = v.sum()
    if s > 0.0:
        return v * (target / s)
    # Should never happen – fallback to uniform distribution
    return np.full_like(v, target / v.shape[0])


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """Transfer a tiny fraction of mass from a random source to a random target."""
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0.0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_to_sum(a, target_sum)


def jitter(a: np.ndarray, target_sum: float, sigma: float = 1e-3) -> np.ndarray:
    """Add Gaussian noise (clipped) and renormalise."""
    a = a + np.random.randn(a.shape[0]) * sigma
    a = np.clip(a, 0.0, None)
    return project_to_sum(a, target_sum)


def smooth_max_step(
    a: np.ndarray,
    target_sum: float,
    lr: float,
    tau: float,
    momentum: float,
    v_prev: np.ndarray,
) -> tuple[np.ndarray, np.ndarray]:
    """
    One iteration of gradient descent on the smoothed max of the convolution.

    Returns the updated vector and the new momentum term.
    """
    # 1) self‑convolution
    conv = conv_self_fft(a)                     # length 2·n‑1

    # 2) soft‑max weights (log‑sum‑exp for robustness)
    max_c = conv.max()
    exp_arg = (conv - max_c) / tau
    # avoid overflow / underflow
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:                           # degenerate – use hard max
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    # 3) gradient of the smoothed max:
    #    g_i = 2 * Σ_k w_k * a_{k-i}
    a_rev = a[::-1]
    grad_full = fft_convolve(w, a_rev)         # length 3·n‑2
    n = a.shape[0]
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]   # slice i = 0 … n‑1

    # 4) project onto the ℓ₁‑hyperplane (preserve the sum)
    direction = -(grad - np.mean(grad))

    # 5) momentum + step
    v = momentum * v_prev + direction
    a_new = a + lr * v

    # 6) enforce feasibility
    a_new = np.clip(a_new, 0.0, None)
    a_new = project_to_sum(a_new, target_sum)

    return a_new, v


# ----------------------------------------------------------------------
# Helpers for the parametrised (low‑dimensional) search
# ----------------------------------------------------------------------


def shape_from_params(v: np.ndarray, n: int, target_sum: float) -> np.ndarray:
    """
    Given a decreasing knot vector `v` (length k) construct a full length‑n
    non‑negative sequence by linear interpolation and renormalise it to
    `target_sum`.
    """
    # guarantee monotone decreasing
    v = np.sort(v)[::-1]
    t_knots = np.linspace(0.0, 1.0, len(v))
    xs = np.linspace(0.0, 1.0, n)
    seq = np.interp(xs, t_knots, v)
    seq = np.clip(seq, 0.0, None)
    return project_to_sum(seq, target_sum)


def downsample_seq(seq: np.ndarray, k: int) -> np.ndarray:
    """
    Produce a decreasing vector of length `k` that interpolates the given
    sequence `seq`.  Used to initialise the low‑dimensional parameter vector.
    """
    xs = np.linspace(0.0, 1.0, len(seq))
    knots = np.linspace(0.0, 1.0, k)
    vals = np.interp(knots, xs, seq)
    vals = np.clip(vals, 0.0, None)
    return np.sort(vals)[::-1]

# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises ``evaluate_sequence``.
    The algorithm first performs a simulated‑annealing search in a 30‑dimensional
    knot space, then refines the best solution with a smooth‑max gradient descent.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # leave a small safety margin

    # --------------------------------------------------------------
    # Choose a length – reuse a known construction if it exists
    # --------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        base_arr = np.array(base, dtype=np.float64)
    else:
        # a reasonable default size (large enough for a good bound, cheap enough)
        n = 8400
        base_arr = None

    target_sum = np.sqrt(2.0 * n)                    # ℓ₁‑norm used in the literature
    k = 30                                            # number of knots for the parametric model

    # --------------------------------------------------------------
    # Initialise the low‑dimensional parameter vector
    # --------------------------------------------------------------
    if base_arr is not None:
        cur_v = downsample_seq(base_arr, k)
    else:
        # random decreasing start
        cur_v = np.sort(np.random.rand(k))[::-1]

    # Evaluate the starting full sequence
    cur_seq = shape_from_params(cur_v, n, target_sum)
    best_seq = cur_seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    cur_score = best_score

    print(f"[{time.time():.2f}] SA start – n={n}, init score={best_score:.6f}")

    # --------------------------------------------------------------
    # Simulated annealing on the knot parameters
    # --------------------------------------------------------------
    # Allocate roughly two‑thirds of the remaining budget to SA

    # --------------------------------------------------------------
    a = best_seq.copy()
