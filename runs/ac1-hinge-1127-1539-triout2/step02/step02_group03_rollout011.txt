    n = a.size
    conv_len = 2*n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)

def _optimize_local(a, deadline, rng):
    n = a.shape[0]
    S = np.sqrt(2.0*n)

    # compute initial conv and max
    conv = _full_conv(a)
    cur_max = float(conv.max())
    cur_idx = int(np.argmax(conv))
    best_a = a.copy()
    best_score = cur_max

    # parameters
    K = 5
    eta = 0.6
    decay = 0.9995
    epsilon_move = 0.08
    jitter_prob = 0.02
    move_prob = 0.5  # decide each iteration whether to do gradient or move
    temp = cur_max * 0.1
    temp_decay = 0.9997

    idx_arr = np.arange(n)

    while time.time() < deadline:
        # compute top K indices
        if K >= conv.shape[0]:
            topK = np.arange(conv.shape[0])
        else:
            topK = np.argpartition(conv, -K)[-K:]

        # compute gradient g
        g = np.zeros_like(a)
        for k in topK:
            partner = k - idx_arr
            mask = (partner >= 0) & (partner < n)
            g[mask] += 2.0 * a[partner[mask]]
        # gradient step
        if rng.random() < move_prob:
            # gradient
            a_new = a * np.exp(-eta * g)
            a_new = np.clip(a_new, 0.0, None)
            total = a_new.sum()
            if total == 0:
                a_new = np.full(n, S / n)
            else:
                a_new *= (S / total)

            # evaluate new max
            conv_new = _full_conv(a_new)
            max_new = float(conv_new.max())
            # accept if better or probabilistic
            if max_new <= cur_max or rng.random() < np.exp(-(max_new - cur_max) / max(temp, 1e-12)):
                a = a_new
                conv = conv_new
                cur_max = max_new
                cur_idx = int(np.argmax(conv))
                if cur_max < best_score:
                    best_score = cur_max
                    best_a = a.copy()
        else:
            # deterministic move
            # compute contributions as a * (g/2)
            contributions = a * (g * 0.5)
            i = int(np.argmax(contributions))
            # choose j as index with smallest a (or low)
            low_idx = np.where(a <= np.percentile(a, 20))[0]
            if low_idx.size:
                j = int(rng.choice(low_idx))
            else:
                j = int(np.argmin(a))
            if i != j:
                delta = epsilon_move * a[i]
                a_candidate = a.copy()
                a_candidate[i] -= delta
                a_candidate[j] += delta
                # evaluate
                conv_candidate = _full_conv(a_candidate)
                max_candidate = float(conv_candidate.max())
                # accept based on annealing
                accept = False
                if max_candidate <= cur_max:
                    accept = True
                else:
                    prob = np.exp(-(max_candidate - cur_max) / max(temp, 1e-12))
                    if rng.random() < prob:
                        accept = True
                if accept:
                    a = a_candidate
                    conv = conv_candidate
                    cur_max = max_candidate
                    cur_idx = int(np.argmax(conv))
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                temp *= temp_decay

"""
Improved stochastic optimiser for the self‑convolution ∞‑norm.

Main ideas
----------

*   Use a *multi‑peak* sub‑gradient: instead of following only the single
    maximal convolution entry we take the K largest entries (K≈5) and
    aggregate their gradients.  This drives down several peaks
    simultaneously and avoids the “single‑peak” stagnation of the
    previous method.

*   Deterministic mass‑transfer moves: from the index that contributes the
    most to the current top‑K peaks we move a fraction of its weight to a
    low‑weight index.  The move is accepted either if it lowers the
    maximum or with a small annealing probability.

*   Adaptive step‑size (exponential decay) and occasional Gaussian jitter
    keep the search robust.

*   Many short restarts (different lengths, Dirichlet‑alpha≈0.3 sparse
    initialisation or the previous best if available) explore the
    high‑dimensional space within the 1000 s budget.  The best sequence
    is refined in a final longer run.

The function `propose_candidate` follows the required signature and
returns the best sequence (as a plain list of floats) found before the
deadline.  The internal value `cur_max` is exactly the quantity
`evaluate_sequence` would return (the maximal entry of the self‑convolution)
because the L¹‑norm is kept at √(2·n) throughout the optimisation.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Compute the linear convolution a*a using FFT.
    Returned array has length 2·len(a)‑1.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()   # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_stats(a: np.ndarray, top_k: int = 5):
    """
    Return the full convolution, its maximal value and index,
    plus the indices of the top_k largest entries (unsorted).
    """
    conv = _full_convolution(a)
    max_idx = int(np.argmax(conv))
    max_val = float(conv[max_idx])
    if top_k >= conv.size:
        top_idx = np.arange(conv.size, dtype=int)
    else:
        top_idx = np.argpartition(conv, -top_k)[-top_k:].astype(int)
    return conv, max_val, max_idx, top_idx


# ----------------------------------------------------------------------
# Gradient + deterministic move optimiser (local)
# ----------------------------------------------------------------------


def _local_optimize(a: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator,
                    top_k: int = 5,
                    init_eta: float = 0.6,
                    decay: float = 0.9995,
                    move_prob: float = 0.5,
                    epsilon_move: float = 0.08,
                    jitter_prob: float = 0.02,
                    temp_decay: float = 0.9997) -> tuple[np.ndarray, float]:
    """
    Perform a stochastic descent on the maximal self‑convolution entry.
    Returns the best sequence encountered and its maximal convolution value.
    """
    n = a.shape[0]
    S = np.sqrt(2.0 * n)               # required L¹‑norm
    idx_arr = np.arange(n)

    # initialise statistics
    conv, cur_max, cur_idx, top_idx = _conv_stats(a, top_k=top_k)
    best_a = a.copy()
    best_score = cur_max

    eta = init_eta
    temp = cur_max * 0.1               # annealing temperature for moves

    while time.time() < deadline:
        # recompute top‑k indices if we have changed the convolution
        if move_prob > 0.0:
            # top indices already available from the previous iteration;
            # they will be refreshed after any accepted modification.
            pass

        # build the (aggregated) sub‑gradient for the chosen top‑k peaks
        g = np.zeros_like(a)
        for k in top_idx:
            partner = k - idx_arr
            mask = (partner >= 0) & (partner < n)
            g[mask] += 2.0 * a[partner[mask]]

        if rng.random() < move_prob:
            # ---------- multiplicative gradient step ----------
            a_candidate = a * np.exp(-eta * g)
            a_candidate = np.clip(a_candidate, 0.0, None)
            total = a_candidate.sum()
            if total == 0.0:
                a_candidate = np.full(n, S / n)
            else:
                a_candidate *= (S / total)

            conv_new = _full_convolution(a_candidate)
            max_new = float(conv_new.max())

            # accept if it helps or with a small annealed probability
            accept = False
            if max_new <= cur_max:
                accept = True
            else:
                prob = np.exp(-(max_new - cur_max) / max(temp, 1e-12))
                if rng.random() < prob:
                    accept = True

            if accept:
                a = a_candidate
                conv = conv_new
                cur_max = max_new
                cur_idx = int(np.argmax(conv))
                top_idx = np.argpartition(conv, -top_k)[-top_k:].astype(int)
                if cur_max < best_score:
                    best_score = cur_max
                    best_a = a.copy()
        else:
            # ---------- deterministic mass transfer ----------
            # contribution of each index to the current top‑k peaks
            contributions = a * (g * 0.5)          # a_i * Σ_j a_j over top‑k
            i = int(np.argmax(contributions))

            # pick a low‑mass target (among the 20 % smallest entries)
            low_thr = np.percentile(a, 20)
            low_candidates = np.where(a <= low_thr)[0]
            if low_candidates.size:
                j = int(rng.choice(low_candidates))
            else:
                j = int(np.argmin(a))

            if i != j:
                delta = epsilon_move * a[i]
                a_candidate = a.copy()
                a_candidate[i] -= delta
                a_candidate[j] += delta

                conv_candidate = _full_convolution(a_candidate)
                max_candidate = float(conv_candidate.max())

                accept = False
                if max_candidate <= cur_max:
                    accept = True
                else:
                    prob = np.exp(-(max_candidate - cur_max) / max(temp, 1e-12))
                    if rng.random() < prob:
                        accept = True

                if accept:
                    a = a_candidate
                    conv = conv_candidate
                    cur_max = max_candidate
                    cur_idx = int(np.argmax(conv))
                    top_idx = np.argpartition(conv, -top_k)[-top_k:].astype(int)
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                temp *= temp_decay

        # occasional jitter to escape plateaus
        if rng.random() < jitter_prob:
            noise = rng.normal(scale=a.mean() * 0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()
            conv = _full_convolution(a)
            cur_max = float(conv.max())
            cur_idx = int(np.argmax(conv))
            top_idx = np.argpartition(conv, -top_k)[-top_k:].astype(int)
            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()

        # decay the step‑size (and, implicitly, the temperature)
        eta *= decay

    return best_a, best_score

def _initial_sequence(base_seq: np.ndarray,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative starting vector of length n with L¹‑norm √(2·n).
    If a suitable base sequence is supplied its leading part (plus a tiny
    jitter) is used; otherwise a sparse Dirichlet draw (α=0.3) is generated.
    """
    S = np.sqrt(2.0 * n)

    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        a += rng.normal(scale=a.mean() * 0.001, size=n)
        a = np.clip(a, 0.0, None)
    else:
        # sparse Dirichlet (Gamma with shape < 1)
        a = rng.gamma(0.3, 1.0, size=n)

    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0)
        total = a.sum()
    a = a / total * S
    return a

            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_overall_seq = None
    best_overall_score = np.inf
    attempt = 0

    # ------------------------------------------------------------------
    # Phase 1 – many short independent restarts
    # ------------------------------------------------------------------
    while time.time() < deadline:
        attempt += 1

        # choose a length (between ~800 and ~5000)
        if base_seq is not None:
            max_len = max(800, min(5000, base_seq.size))
            n = rng.integers(800, max_len + 1)
        else:
            n = rng.integers(800, 4000)

        a0 = _initial_sequence(base_seq, n, rng)

        # 20 s (or the remaining budget) for the local optimiser
        slice_deadline = min(time.time() + 20.0, deadline)

        a_opt, score = _local_optimize(a0, slice_deadline, rng)

        if score < best_overall_score:
            best_overall_score = score
            best_overall_seq = a_opt.copy()
            print(f"[global] attempt={attempt} n={n} new best score={best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – final refinement of the best candidate (if time permits)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if best_overall_seq is not None and time_left > 30.0:
        a_refined, refined_score = _local_optimize(
            best_overall_seq,
            deadline,                 # use the full remaining budget
            rng,
            top_k=8,                 # a slightly larger top‑k set for fine‑tuning
            init_eta=0.3,            # smaller steps for precision
        )
        if refined_score < best_overall_score:
            best_overall_score = refined_score
            best_overall_seq = a_refined.copy()
            print(f"[refine] improved to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence as a plain list of floats
    # ------------------------------------------------------------------
    if best_overall_seq is not None:
        final_seq = best_overall_seq.tolist()
        external_score = evaluate_sequence(final_seq)   # black‑box check
        print(f"[final] best external score = {external_score:.6f} "
              f"(internal max = {best_overall_score:.6f}) "
              f"length = {len(final_seq)}")
        return final_seq
    else:
        # Should never happen – fallback to a trivial feasible vector
        return [1.0]
