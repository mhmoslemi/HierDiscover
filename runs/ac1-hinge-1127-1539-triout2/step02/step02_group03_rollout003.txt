from scipy.optimize import linprog

def solve_lp(a, M):
    n = len(a)
    # Build A matrix:
    # Each offset d from 0..n-1 gives diagonal of length n entries = a[d]
    diags_data = [np.full(n, a[d]) for d in range(n)]
    offsets = np.arange(n)
    A = diags(diags_data, offsets, shape=(2*n-1, n), format='csc')
    b_ub = np.full(2*n-1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        return None
```

def conv_fft(v):
    n = len(v)
    conv_len = 2*n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    F = np.fft.rfft(v, n=fft_len)
    conv = np.fft.irfft(F*F, n=fft_len)[:conv_len]
    return conv

"""
LP‑guided mixing for step‑function self‑convolution minimisation.

The algorithm works in two phases:

1. **LP direction** – given a current non‑negative sequence `a` with
   Σa = √(2·n), solve a linear programme that finds a complementary
   sequence `b` of maximal ℓ₁‑mass while respecting the convolution bound
   (a * b)ₖ ≤ M, where M = maxₖ (a*a)ₖ.  After scaling `b` to the same
   ℓ₁‑norm we obtain a candidate `g`.

2. **Optimal mixing** – the new sequence is taken as
   (1‑t)·a + t·g.  For each `t ∈ [0,1]` the self‑convolution is a quadratic
   function of `t`.  We evaluate the maximum entry on a fine grid,
   refine around the best point and keep the optimal mixture.

The whole procedure is embedded in a multi‑restart loop that tries
different lengths `n` (≈800…2000) and random initialisations.  After the
global search a short multiplicative sub‑gradient refinement (the
method used in the previous baseline) is applied to the best sequence
found so far.

All sequences are kept non‑negative and normalised to Σa = √(2·n); hence
`evaluate_sequence` simply returns the maximal convolution entry.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# Core linear‑algebra utilities
# ----------------------------------------------------------------------


def _conv_fft(v: np.ndarray) -> np.ndarray:
    """Full linear convolution v*v using an FFT (return length 2·len(v)-1)."""
    n = v.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()          # next power of two
    V = np.fft.rfft(v, n=fft_len)
    conv = np.fft.irfft(V * V, n=fft_len)[:conv_len]
    return conv


def _conv_cross(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b (full length)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    A = np.fft.rfft(a, n=fft_len)
    B = np.fft.rfft(b, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Linear programme (LP) that supplies a direction g
# ----------------------------------------------------------------------


def _solve_lp(a: np.ndarray, M: float) -> np.ndarray | None:
    """Solve  max Σ bᵢ   s.t.  a * b ≤ M  (element‑wise) ,  b ≥ 0.
    Returns the optimal vector b (or None if the solver fails)."""
    n = a.shape[0]

    # Build the Toeplitz matrix that implements the linear convolution a*b.
    # Entry (k,i) = a[k‑i]  for 0 ≤ i < n , 0 ≤ k‑i < n.
    # This is a sparse matrix with constant diagonals.
    data = [np.full(n, a[d], dtype=float) for d in range(n)]
    offsets = np.arange(n, dtype=int)                     # 0 … n‑1
    A = diags(data, offsets, shape=(2 * n - 1, n), format="csc")

    # Upper‑bound vector (all entries ≤ M)
    b_ub = np.full(2 * n - 1, M, dtype=float)

    # Linear objective: maximise Σb → minimise −Σb
    c = -np.ones(n, dtype=float)

    # Bounds for each variable: bᵢ ≥ 0
    bounds = [(0, None)] * n

    # Use HiGHS (fast interior‑point) via scipy.optimize.linprog
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
# Mixing – find the best t ∈ [0,1] for (1‑t)·a + t·g
# ----------------------------------------------------------------------


def _best_mix(conv_a: np.ndarray,
              conv_g: np.ndarray,
              conv_ag: np.ndarray) -> tuple[float, float]:
    """
    Returns the mixing weight t that minimises max_k
        (1‑t)²·conv_a + 2·t·(1‑t)·conv_ag + t²·conv_g
    as well as the corresponding maximal value.
    """
    # coarse grid
    ts = np.linspace(0.0, 1.0, 21)
    best_t = 0.0
    best_val = np.inf
    for t in ts:
        mixed = (1 - t) ** 2 * conv_a + 2 * t * (1 - t) * conv_ag + t ** 2 * conv_g
        val = mixed.max()
        if val < best_val:
            best_val = val
            best_t = t

    # refined search around the best coarse point
    lo = max(0.0, best_t - 0.05)
    hi = min(1.0, best_t + 0.05)
    ts2 = np.linspace(lo, hi, 21)
    for t in ts2:
        mixed = (1 - t) ** 2 * conv_a + 2 * t * (1 - t) * conv_ag + t ** 2 * conv_g
        val = mixed.max()
        if val < best_val:
            best_val = val
            best_t = t
    return best_t, best_val


# ----------------------------------------------------------------------
# Multiplicative sub‑gradient step (optional final polishing)
# ----------------------------------------------------------------------


def _gradient_step(a: np.ndarray, eta: float) -> np.ndarray:
    """One multiplicative sub‑gradient update to reduce the maximal convolution entry."""
    n = a.shape[0]
    conv = _conv_fft(a)
    k = int(np.argmax(conv))

    idx = np.arange(n)
    partner = k - idx
    valid = (partner >= 0) & (partner < n)

    g = np.zeros_like(a)
    g[valid] = 2.0 * a[partner[valid]]

    if k % 2 == 0:
        d = k // 2
        if 0 <= d < n:
            g[d] = a[d]                 # diagonal contributes only once

    a_new = a * np.exp(-eta * g)

    S = np.sqrt(2.0 * n)
    total = a_new.sum()
    if total == 0.0:
        a_new = np.full(n, S / n)
    else:
        a_new *= (S / total)
    return a_new


# ----------------------------------------------------------------------
# Initial sequence generator
# ----------------------------------------------------------------------


def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """Create a non‑negative length‑n sequence with Σ = √(2·n)."""
    S = np.sqrt(2.0 * n)

    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        # small jitter keeps restarts diverse
        jitter = rng.normal(scale=max(a.mean(), 1e-8) * 0.001, size=n)
        a += jitter
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)               # Dirichlet(1,…,1) – uniform on simplex

    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0)
        total = a.sum()
    a = a / total * S
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------

    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Global multi‑restart loop

    while time.time() < deadline:
        attempt += 1

        # Choose a problem size.  We cap at 2000 to keep the LP cheap.
        if base_seq is not None:

        else:
            n = rng.integers(800, 2000)

        # Initialise a feasible sequence.
        a = _initial_sequence(base_seq, n, rng)

        # inner optimisation – a few LP‑mixing steps

        for it in range(inner_iters):

            # Find optimal mixing coefficient
            t_opt, mixed_max = _best_mix(conv_a, conv_g, conv_ag)
