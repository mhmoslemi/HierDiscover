def conv_self_fft(a):
    n = len(a)
    L = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa*fa, n=L)
    return conv[:2*n-1]

def generate_piecewise_U(n, target_sum, w_frac=0.4, ratio=2.0):
    # w_frac in [0,1], fraction of n that is low region in middle.
    low_len = int(round(w_frac * n))
    high_len = n - low_len
    # split high_len equally to both edges?
    # Use symmetric: left high_len/2 at left, right high_len/2 at right.
    # We'll assign values high_val and low_val s.t. sum = target.
    # Let low_val = x, high_val = r*x.
    # Sum = low_len*x + high_len * (r*x) = target.
    # => x = target / (low_len + high_len*r). Then high_val = r*x.
    # Build array: [high_val]*left_len + [low_val]*low_len + [high_val]*right_len.
    left_len = high_len // 2
    right_len = high_len - left_len
    low_val = target_sum / (low_len + high_len*ratio)
    high_val = low_val * ratio
    seq = np.concatenate([np.full(left_len, high_val), np.full(low_len, low_val), np.full(right_len, high_val)])
    # In case length mismatched due to rounding, adjust last element.
    if len(seq)!=n:
        # pad/truncate
        seq = np.resize(seq, n) # quick
    # Ensure sum = target, project onto simplex just in case rounding errors.
    seq = project_onto_simplex(seq, target_sum)
    return seq

    target_sum = np.sqrt(2.0 * n)

    t = 0
    no_improve = 0

    while time.time() < deadline:

        max_conv = conv.max()

        # smoothing temperature
        tau = max(tau_factor * max_conv, 1e-8)
        w = np.exp((conv - max_conv) / tau)
        w_sum = w.sum()
        w /= w_sum

        # gradient
        a_rev = a[::-1]

"""
Propose a non‑negative step‑function (list of heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The code follows three stages:

1. **Extensive parametric sweep** – a collection of simple symmetric
   shapes (linear, quadratic, exponential, power‑law, piecewise “U”)
   is generated, normalised to the ℓ₁‑simplex with sum = √(2 n) and
   the best shape is kept.

2. **Accelerated smoothed optimisation** – the non‑smooth
   `max(conv)` is approximated by a log‑sum‑exp (temperature τ).
   The gradient of the smoothed objective is obtained analytically
   (a convolution of the soft‑max weights with the reversed sequence).
   An Adam optimiser performs many updates; after each step the vector
   is projected back onto the ℓ₁‑simplex.  The learning‑rate and the
   temperature are slowly decayed, and occasional random jitter
   prevents stagnation.

3. **Polishing** – a short burst of tiny random pair‑transfers are
   applied to the best solution found; any improvement is kept.

All convolutions use a power‑of‑two FFT, making each iteration
~O(n log n).  The routine respects a 1000 s wall‑time budget and
always returns the best sequence discovered so far.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a with itself using a power‑of‑two FFT.
    Returns a vector of length 2*len(a)‑1.
    """
    n = a.shape[0]
    L = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return conv_self_fft(a).max()


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex
        {x | x_i ≥ 0, Σ x_i = target}.
    Implements the O(n log n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – return uniform distribution
        return np.full_like(v, target / len(v))

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    param ≥ 0 controls edge enhancement.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    else:
        raise ValueError("unknown shape kind")

    return project_onto_simplex(raw, target_sum)


def generate_power_shape(
    n: int,
    target_sum: float,
    param: float = 1.0,
    power: float = 2.0,
) -> np.ndarray:
    """
    Shape a_i ∝ (1 + param·d) ** power,
    where d = |i‑mid|/mid.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid
    raw = (1.0 + param * d) ** power
    return project_onto_simplex(raw, target_sum)


def generate_piecewise_U(
    n: int,
    target_sum: float,
    middle_frac: float = 0.4,
    edge_ratio: float = 3.0,
) -> np.ndarray:
    """
    Piecewise constant “U’’:
      * middle_frac – fraction of the sequence that forms the low‑plateau.
      * edge_ratio   – edge value = edge_ratio × middle value.
    """
    middle_len = max(1, int(round(middle_frac * n)))
    edge_len = n - middle_len
    left_len = edge_len // 2
    right_len = edge_len - left_len

    # let low = x, high = r·x, solve x·middle_len + r·x·edge_len = target_sum
    x = target_sum / (middle_len + edge_len * edge_ratio)
    low = x
    high = low * edge_ratio

    seq = np.concatenate(
        [np.full(left_len, high), np.full(middle_len, low), np.full(right_len, high)]
    )
    # may be off‑by‑one because of rounding; correct by projection
    return project_onto_simplex(seq, target_sum)


def sample_dirichlet_seq(
    n: int,
    target_sum: float,
    alpha: float = 1.0,
) -> np.ndarray:
    """
    Random vector from Dirichlet(alpha,…,alpha) scaled to the ℓ₁‑simplex.
    """
    x = np.random.gamma(alpha, 1.0, size=n)
    return (x / x.sum()) * target_sum


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a tiny amount of mass from one random index to another,
    then project back onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


# ----------------------------------------------------------------------
# Optimisation core (Adam with log‑sum‑exp smoothing)
# ----------------------------------------------------------------------


def adam_optimize(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    init_lr: float = 0.6,
    init_tau_factor: float = 0.12,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed objective.
    Returns the best vector and its objective value (max convolution).
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # schedules
    lr = init_lr
    tau_factor = init_tau_factor

    best_seq = a.copy()
    best_score = max_self_convolution(a)

    no_improve = 0

    while time.time() < deadline:
        # ---- smoothed max ----
        conv_full = conv_self_fft(a)                # length 2n‑1
        max_conv = conv_full.max()

        # temperature τ
        tau = max(tau_factor * max_conv, 1e-12)

        # soft‑max weights (log‑sum‑exp)
        w = np.exp((conv_full - max_conv) / tau)
        w_sum = w.sum()
        if w_sum == 0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # ---- gradient (∂/∂a of the smoothed max) ----
        a_rev = a[::-1]
        L = next_pow2(len(w) + len(a_rev) - 1)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        grad_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]

        grad = (2.0 / tau) * grad_full[n - 1 : n - 1 + n]   # size n

        # ---- Adam update ----
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # enforce feasibility
        a_new = project_onto_simplex(np.maximum(a_new, 0.0), target_sum)

        a = a_new
        cur_score = max_self_convolution(a)

        # ---- bookkeeping ----
        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
            print(f"[{time.time():.2f}] Adam iter {t:07d} – new best {best_score:.6f}")
        else:
            no_improve += 1

        # ---- schedule adaptation ----
        if t % 500 == 0:
            lr *= 0.93
            tau_factor = max(0.001, tau_factor * 0.94)

        # occasional jitter to escape flat spots
        if np.random.rand() < 0.008:
            jitter = np.random.randn(n) * 0.0015
            a = project_onto_simplex(np.maximum(a + jitter, 0.0), target_sum)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[{time.time():.2f}] Adam iter {t:07d} – jitter injected")

        # heavy random reset after long stagnation
        if no_improve >= 4000:
            a = random_pair_transfer(best_seq.copy(), target_sum, scale=0.008)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[{time.time():.2f}] Adam iter {t:07d} – reset from best")
    return best_seq, best_score

# Main entry point (required by the harness)

def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative coefficient vector that minimises
    `evaluate_sequence`.  The routine respects a ≈1000 s time budget
    and returns the best vector found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # safety margin

    # ------------------------------------------------------------------
    # Determine problem size (use provided baseline if it exists)
    # ------------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        init_seq = np.array(base, dtype=np.float64)
    else:
        # a fairly large size that is still cheap for FFTs
        n = np.random.randint(12000, 15000)
        init_seq = None

    target_sum = np.sqrt(2.0 * n)   # usual normalisation (makes objective = max conv)

    # ------------------------------------------------------------------
    # Phase 1 – parametric sweep (U‑shapes, power‑law, piecewise)
    # ------------------------------------------------------------------
    best_seq = np.full(n, target_sum / n)   # uniform start
    best_score = max_self_convolution(best_seq)

    print(f"[{time.time():.2f}] Starting sweep (n={n}) – initial score {best_score:.6f}")

    # 1) classic U‑shapes
    for kind in ("linear", "quadratic", "exponential"):
        max_param = {"linear": 10.0, "quadratic": 6.0, "exponential": 4.0}[kind]
        for p in np.linspace(0.0, max_param, num=31):
            cand = generate_U_shape(n, target_sum, kind=kind, param=p)
            score = max_self_convolution(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"  New best {kind} p={p:.2f} → {best_score:.6f}")
            if time.time() >= deadline:
                print("Time exhausted during sweep.")
                return best_seq.tolist()

    # 2) power‑law shapes
    for power in (1.5, 2.0, 3.0):
        for param in np.linspace(0.0, 6.0, 21):
            cand = generate_power_shape(n, target_sum, param=param, power=power)
            score = max_self_convolution(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"  New best power p={param:.2f}, pow={power} → {best_score:.6f}")
            if time.time() >= deadline:
                return best_seq.tolist()

    # 3) piecewise U‑shapes
    for middle_frac in np.linspace(0.1, 0.8, 15):
        for edge_ratio in np.linspace(1.5, 6.0, 15):
            cand = generate_piecewise_U(
                n, target_sum, middle_frac=middle_frac, edge_ratio=edge_ratio
            )
            score = max_self_convolution(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(
                    f"  New best piecewise mfrac={middle_frac:.2f} ratio={edge_ratio:.2f} → {best_score:.6f}"
                )
            if time.time() >= deadline:
                return best_seq.tolist()

    # 4) random Dirichlet draws (quick Monte‑Carlo)
    alphas = [0.2, 0.5, 0.8, 1.0, 2.0, 5.0]
    while time.time() < deadline - 5.0:   # leave time for later phases
        alpha = np.random.choice(alphas)
        cand = sample_dirichlet_seq(n, target_sum, alpha=alpha)
        score = max_self_convolution(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"  New best Dirichlet α={alpha:.2f} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – accelerated Adam optimisation
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Entering Adam optimisation – best so far {best_score:.6f}")

    # allocate a slice of the remaining budget for Adam
    adam_deadline = deadline - 3.0   # keep a few seconds for polishing
    best_seq, best_score = adam_optimize(
        best_seq,
        target_sum=target_sum,
        deadline=adam_deadline,
        init_lr=0.6,
        init_tau_factor=0.12,
    )

    # ------------------------------------------------------------------
    # Phase 3 – final polishing (tiny random pair transfers)
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Polishing phase – current best {best_score:.6f}")

    for _ in range(60):
        if time.time() >= deadline:
            break
        cand = random_pair_transfer(best_seq.copy(), target_sum, scale=0.004)
        cand_score = max_self_convolution(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"Polish improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence as a plain Python list
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Finished – final best {best_score:.6f}")
    return best_seq.tolist()
