"""
Non‑negative step‑function optimizer (LP + smooth exponential descent).

The score to minimise is
    f(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

Because the objective is scale‑invariant we work with a probability
vector `p` (sum(p)=1).  For a given `p` we denote
    M(p) = max(conv(p,p))

The algorithm has two complementary phases:

1. **Smooth exponential descent**  
   We replace the non‑smooth `max` by a soft‑max
   `L(p)=log Σ exp(α·conv(p,p))`.  Its gradient can be written as a
   correlation and is computed by FFTs.  Using an exponentiated‑gradient
   update (`p_i ← p_i·exp(-η·∇L_i)`) together with Euclidean projection
   onto the simplex yields rapid reduction of the maximal convolution.

2. **Linear‑programming refinement**  
   Given a current vector `p` we solve
        maximise Σ g_j
        s.t.    g_j ≥ 0
                conv(p,g) ≤ M(p)   (2·n–1 linear constraints)
   The solution `g` is scaled back to a probability vector and mixed with
   `p` (`p ← (1‑t)·p + t·g`).  This step provably never increases the
   maximal convolution and removes remaining “flat” plateaus.

A number of diverse seeds (uniform, random Dirichlet, power‑law,
exponential and the previous best `height_sequence_1`) are generated.
Each seed is first refined by the exponential‑descent phase (up to several
thousand iterations) and then polished by a few LP‑updates.  The best
candidate discovered before the time budget expires is returned.
"""

import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]

    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_conv(p: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a real vector with itself using FFT.
    Returns length 2·len(p)‑1 array.
    """
    n = p.shape[0]
    L = 1 << ((2 * n - 2).bit_length())  # next power of two
    ft = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(ft * ft, n=L)[: 2 * n - 1]
    conv[conv < 0] = 0.0            # clip tiny negative round‑off
    return conv


def _correlation(p: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Correlation of w (length 2·n‑1) with p (length n).
    Returns a vector of length n:
        (corr)_i = Σ_j w_{i+j}·p_j   (valid indices only)
    Implemented by FFT (conv of w with reversed p) and extracting the
    “valid” part.
    """
    n = p.shape[0]
    p_rev = p[::-1]
    L = 1 << ((len(w) + n - 2).bit_length())
    ft_w = np.fft.rfft(w, n=L)
    ft_pr = np.fft.rfft(p_rev, n=L)
    conv_full = np.fft.irfft(ft_w * ft_pr, n=L)
    start = n - 1
    grad = 2.0 * conv_full[start:start + n]   # factor 2 from derivative of p·p
    return grad.real


def _smooth_exponential_descent(p: np.ndarray,
                                deadline: float,
                                max_iters: int = 5000,
                                alpha_start: float = 20.0,
                                alpha_end: float = 500.0,
                                eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Runs the smooth exponential‑descent optimisation on probability vector `p`.
    Returns the best vector encountered and its score (2·n·max_conv).
    """
    n = p.shape[0]
    best_score = float("inf")
    best_vec = p.copy()

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        conv = _full_conv(p)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_vec = p.copy()
            print(f"[Descent] iter {it:5d} | n={n} | score={best_score:.6f}")

            # early exit if we already beat the target
            if best_score < 1.503:
                return best_vec, best_score

        # anneal the smoothing parameter
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)

        # soft‑max weights – shifted by max to avoid overflow
        w = np.exp(alpha * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0:
            break
        w /= w_sum

        # gradient of the smoothed objective
        grad = _correlation(p, w)

        # exponentiated‑gradient update
        eta = eta0 / np.sqrt(it)
        p = p * np.exp(-eta * grad)

        # project back onto the simplex
        p = _project_simplex(p)

        # occasional stochastic jitter to escape tiny plateaus
        if np.random.rand() < 0.01:
            idx = np.random.randint(n)
            delta = (np.random.rand() - 0.5) * 1e-3 * p[idx]
            p[idx] = max(p[idx] + delta, 0.0)
            p = _project_simplex(p)

    return best_vec, best_score


def _build_conv_matrix(p: np.ndarray) -> np.ndarray:
    """
    Constructs the (2·n‑1) × n matrix A such that (A·g)_k = (p∗g)[k].
    """
    n = p.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    idx = np.arange(n)
    for i in range(n):
        A[i + idx, idx] += p[i]
    return A


def _lp_refine(p: np.ndarray,
               max_conv: float,
               mixing: float = 0.2) -> np.ndarray:
    """
    One LP‑based refinement step.
    Maximises Σ g_j subject to g ≥ 0 and conv(p,g) ≤ max_conv.
    The resulting g is renormalised to sum(p) (i.e. 1) and mixed with p.
    """
    try:
        from scipy.optimize import linprog
    except Exception as e:         # scipy not available – skip refinement
        return p

    n = p.shape[0]
    A = _build_conv_matrix(p)               # shape (2n‑1, n)
    b = np.full(2 * n - 1, max_conv, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)       # minimise -Σg → maximise Σg
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs',
                  options={'presolve': True, 'time_limit': 30})

    if not res.success:
        return p

    g = res.x
    sum_g = g.sum()
    if sum_g <= 0.0:
        return p

    # renormalise to a probability vector (sum = 1)
    g = g / sum_g

    # convex combination – the sum stays 1
    new_p = (1.0 - mixing) * p + mixing * g
    return new_p

    # --------------------------------------------------------------
    seed_vectors = []

    # previous best curve, if any
    if "height_sequence_1" in globals():
        hs = np.asarray(height_sequence_1, dtype=np.float64)
        seed_vectors.append(hs)

    # power‑law seeds
    for p_exp in np.linspace(0.3, 0.7, 7):
        for sigma in np.logspace(-3.0, 0.2, 12):
            i = np.arange(4000, dtype=np.float64)
            a = (i + sigma) ** (-p_exp)
            a = a[a > 1e-9]
            seed_vectors.append(a)

    # exponential decay seeds
    for lam in np.linspace(0.0004, 0.006, 12):
        i = np.arange(4000, dtype=np.float64)
        a = np.exp(-lam * i)
        a = a[a > 1e-9]
        seed_vectors.append(a)

    # uniform seeds of several lengths
    for n in (500, 1000, 1500, 2000, 2500):
        seed_vectors.append(np.ones(n, dtype=np.float64))

    # random Dirichlet‑like seeds
    for n in (600, 1200, 1800, 2400, 3000):
        rnd = np.random.rand(n)
        seed_vectors.append(rnd)

    # --------------------------------------------------------------
    # 2) optimisation for each seed
    # --------------------------------------------------------------
    for raw in seed_vectors:
        if time.time() >= deadline:
            break

        # truncate overly long vectors for speed
        max_len = 2000
        a = np.asarray(raw, dtype=np.float64)
        if a.size > max_len:
            a = a[:max_len]

        # protect against all‑zero or negative entries
        a = np.maximum(a, 0.0)
        if a.sum() == 0.0:
            continue

        # normalise to a probability vector (sum = 1)
        p = a / a.sum()

        n = p.shape[0]
        print(f"\n[Seed] length={n}")

        # ---- Phase 1: smooth exponential descent ----
        p_opt, score_opt = _smooth_exponential_descent(
            p.copy(),
            deadline,
            max_iters=8000,
            alpha_start=20.0,
            alpha_end=500.0,
            eta0=0.6
        )
        if score_opt < best_score:
            best_score = score_opt
            best_seq = p_opt.copy()
            print(f"[Update] new global best after descent: {best_score:.6f}")

        # ---- Phase 2: a few LP polishing steps ----
        p_lp = p_opt.copy()
        for lp_step in range(5):
            if time.time() >= deadline:
                break
            conv = _full_conv(p_lp)
            cur_max = conv.max()
            p_lp = _lp_refine(p_lp, cur_max, mixing=0.2)
            new_score = 2.0 * n * _full_conv(p_lp).max()
            if new_score < best_score:
                best_score = new_score
                best_seq = p_lp.copy()
                print(f"[LP] step {lp_step+1} | new global best: {best_score:.6f}")

            # if LP step made no improvement, break early
            if abs(new_score - score_opt) < 1e-12:
                break
            score_opt = new_score

    # --------------------------------------------------------------
    # 3) final reporting
    # --------------------------------------------------------------
    print("\n=== Final Result ===")
    print(f"Best score found: {best_score:.6f}")
    print(f"Length of best sequence: {len(best_seq) if best_seq is not None else 'None'}")
    # return as a plain Python list (the evaluation function accepts list[float])
    return best_seq.tolist() if best_seq is not None else []  # empty list signals failure
