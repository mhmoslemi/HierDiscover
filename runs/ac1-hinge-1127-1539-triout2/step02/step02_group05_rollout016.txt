```
def optimize_sequence(a0, target_sum, deadline):
    n = a0.shape[0]
    a = a0.copy()
    best_a = a.copy()
    best_score = _compute_score(a)
    # FFT sizes
    nfft_conv = _next_pow2(2 * n - 1)
    nfft_corr = _next_pow2(3 * n - 2)

    # Adam stats
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    # Learning rate
    lr = 0.3
    lr_min = 0.001
    lr_decay = 0.9995

    # Gamma schedule
    gamma = 1.0
    gamma_max = 200.0
    gamma_inc = 1.002

    smooth_phase_iters = 3000  # after this switch to hard max
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        t += 1

        # compute conv and max
        af = np.fft.rfft(a, n=nfft_conv)
        conv = np.fft.irfft(af * af, n=nfft_conv)[: 2 * n - 1]
        max_conv = conv.max()

        # determine weights
        if iteration <= smooth_phase_iters:
            w = np.exp(gamma * (conv - max_conv))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.ones_like(conv) / conv.size
            else:
                w = w / w_sum
            gamma = min(gamma * gamma_inc, gamma_max)
        else:
            # hard max
            mask = conv == max_conv
            cnt = np.count_nonzero(mask)
            w = np.zeros_like(conv)
            w[mask] = 1.0 / cnt

        # gradient via convolution with reversed a
        wf = np.fft.rfft(w, n=nfft_corr)
        af_rev = np.fft.rfft(a[::-1], n=nfft_corr)
        corr_full = np.fft.irfft(wf * af_rev, n=nfft_corr)
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_onto_simplex(a_new, target_sum)

        # occasional jitter
        if iteration % 97 == 0:
            jitter = 1e-4 * np.random.randn(n)
            a_new = project_onto_simplex(a_new + jitter, target_sum)

        # evaluate
        score_new = _compute_score(a_new)

        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            elapsed = time.time() - start_time
            print(f"Iter {iteration:6d}: new best {best_score:.6f} (elapsed {elapsed:.1f}s)")

        a = a_new

        # decrease lr
        lr = max(lr * lr_decay, lr_min)

        # optional random pair transfer every 500 iterations
        if iteration % 500 == 0:
            # attempt deterministic transfer from high to low
            i = np.argmax(a)
            j = np.argmin(a)
            # delta limited to a[i] * 0.02
            delta = min(a[i] * 0.02, target_sum * 0.01)
            if delta > 0:
                a_tmp = a.copy()
                a_tmp[i] -= delta
                a_tmp[j] += delta
                a_tmp = project_onto_simplex(a_tmp, target_sum)
                score_tmp = _compute_score(a_tmp)
                if score_tmp < best_score:
                    best_score = score_tmp
                    best_a = a_tmp.copy()
                    a = a_tmp
                    print(f"Transfer step improvement: {best_score:.6f}")

        # stop condition if near deadline
        if time.time() >= deadline:
            break

    return best_a, best_score
```

```
def fine_tune_sa(a, target_sum, deadline, init_score=None):
    n = a.shape[0]
    best_a = a.copy()
    best_score = init_score if init_score is not None else _compute_score(a)
    cur_a = a.copy()
    cur_score = best_score

    T0 = 0.1 * best_score
    T = T0
    decay = 0.9999

    while time.time() < deadline:
        # random transfer
        i, j = np.random.choice(n, 2, replace=False)
        # size of transfer relative to cur_a[i]
        max_delta = min(cur_a[i], target_sum * 0.01)
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand()
        cur_a[i] -= delta
        cur_a[j] += delta
        # enforce non-neg & sum via projection (though sum should be same)
        cur_a = project_onto_simplex(cur_a, target_sum)
        new_score = _compute_score(cur_a)
        delta_score = new_score - cur_score
        accept = False
        if delta_score < 0:
            accept = True
        else:
            if np.random.rand() < np.exp(-delta_score / T):
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cur_a.copy()
                elapsed = time.time() - start_time
                print(f"SA improve: {best_score:.6f} (elapsed {elapsed:.1f}s)")
        else:
            # revert
            cur_a[i] += delta
            cur_a[j] -= delta
        T = max(T * decay, 1e-6)

    return best_a, best_score
```

assistantfinal**Solution Overview**

The program searches for a non‑negative sequence `a = (a₀,…,aₙ₋₁)` that minimises  

\[
\frac{2\,n\,\max (a*a)}{(\sum a)^2},
\]

where `a*a` is the linear convolution of the sequence with itself.
If the sequence is normalised to  

\[
\sum a = \sqrt{2n},
\]

the objective becomes simply `max(a*a)`.  

The optimiser consists of three parts

1. **Initialisation** – a few random power‑law / exponential shapes are built
   and the best of them (or the supplied `height_sequence_1`) is kept.
2. **Projected Adam optimisation** – we minimise the maximum convolution.
   * For the first few thousand iterations a *soft‑max* surrogate is used
     (`softmax` with a temperature γ).  
   * After the “smooth” phase we switch to the true *hard‑max* (sub‑gradient)
     which gives an exact gradient of `max(a*a)`.  
   * Every step is projected onto the simplex  
     `{x ≥ 0 , Σx = √(2n)}`.
3. **Fine‑tuning with simulated‑annealing style moves** – random transfers of a
   small amount of mass between two positions are accepted if they improve the
   score or with a temperature–controlled probability. This helps to escape the
   local basin found by Adam.

The routine repeatedly restarts the optimisation with different random
initialisations (subject to the remaining time budget) and keeps the best
sequence found over all runs.

All heavy calculations (convolution, gradient) are performed with FFTs, so a
run with a few‑thousand‑element vectors easily fits into the 1000 s budget.



```python
"""
Search for a non‑negative step‑function that minimises
    2 * n * max(conv(a, a)) / (sum(a) ** 2)

Algorithm
---------
1. Build several random initial shapes (power‑law, exponential etc.) and
   reuse the globally supplied height_sequence_1 when available.
2. Optimise with a projected Adam method.
   * First phase – smooth‑max surrogate (softmax over the convolution) to
     obtain a good descent direction.
   * Second phase – hard‑max (sub‑gradient of the true maximum) for precise
     convergence.
   * After each Adam update the vector is projected onto the simplex
     {x ≥ 0, Σx = sqrt(2 n)}.
3. Fine‑tune the best vector with a cheap simulated‑annealing style local
   search (random mass transfers).

The routine runs as many independent restarts as time permits and returns the
best sequence found.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _compute_score(a: np.ndarray) -> float:
    """
    Fast version of `evaluate_sequence` – assumes `a` is non‑negative.
    Returns 2 * n * max(conv(a, a)) / (sum(a) ** 2).
    """
    n = a.shape[0]
    nfft = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))


def project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of *v* onto the simplex
        { x : x_i ≥ 0 , Σ x_i = target_sum }.
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def generate_initial_seq(n: int, target_sum: float) -> np.ndarray:
    """
    Produce a random, well‑behaved initial candidate.
    Mixes power‑law and exponential shapes.
    """
    if np.random.rand() < 0.6:
        # power‑law a_i = 1/(i+offset)^p
        p = np.random.uniform(0.3, 1.0)
        offset = np.random.uniform(0.3, 5.0)
        a = 1.0 / np.power(np.arange(n, dtype=np.float64) + offset, p)
    else:
        # exponential a_i = exp(-c * i / n)
        c = np.random.uniform(0.0, 2.0)
        a = np.exp(-c * np.arange(n, dtype=np.float64) / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a = a * target_sum / s
    else:
        a = np.full(n, target_sum / n)
    return a


def optimize_sequence(a0: np.ndarray,
                      target_sum: float,
                      deadline: float,
                      base_time: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation.
    First phase – smooth max (softmax) → second phase – hard max (sub‑gradient).
    Returns the best vector found and its score.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _compute_score(a)

    # FFT sizes (convolution and gradient)
    nfft_conv = _next_pow2(2 * n - 1)
    nfft_corr = _next_pow2(3 * n - 2)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    # learning‑rate schedule
    lr = 0.30
    lr_min = 0.001
    lr_decay = 0.9995

    # soft‑max temperature schedule (used only in the first phase)
    gamma = 1.0
    gamma_max = 200.0
    gamma_inc = 1.002

    smooth_iters = 3000            # after this we use the hard‑max sub‑gradient
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        t += 1

        # ----- convolution a * a -------------------------------------------------
        af = np.fft.rfft(a, n=nfft_conv)
        conv = np.fft.irfft(af * af, n=nfft_conv)[: 2 * n - 1]
        max_conv = conv.max()

        # ----- weights -----------------------------------------------------------
        if iteration <= smooth_iters:
            w = np.exp(gamma * (conv - max_conv))
            w_sum = w.sum()
            if w_sum == 0.0:
                w[:] = 1.0 / w.size
            else:
                w /= w_sum
            gamma = min(gamma * gamma_inc, gamma_max)
        else:
            # hard‑max: equal weight on all active indices
            mask = conv == max_conv
            cnt = np.count_nonzero(mask)
            w = np.zeros_like(conv)
            w[mask] = 1.0 / max(cnt, 1)

        # ----- gradient of (soft or hard) max ------------------------------------
        wf = np.fft.rfft(w, n=nfft_corr)
        af_rev = np.fft.rfft(a[::-1], n=nfft_corr)
        corr_full = np.fft.irfft(wf * af_rev, n=nfft_corr)
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]      # length‑n slice

        # ----- Adam update -------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_onto_simplex(a_new, target_sum)

        # ----- occasional jitter to escape plateaus ------------------------------
        if iteration % 97 == 0:
            jitter = 1e-4 * np.random.randn(n)
            a_new = project_onto_simplex(a_new + jitter, target_sum)

        # ----- evaluate -----------------------------------------------------------
        score_new = _compute_score(a_new)

        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            elapsed = time.time() - base_time
            print(
                f"Iter {iteration:6d}: new best {best_score:.6f} "
                f"(elapsed {elapsed:.1f}s)"
            )

        a = a_new

        # ----- learning‑rate decay ------------------------------------------------
        lr = max(lr * lr_decay, lr_min)

        # ----- deterministic transfer of a little mass from large to small ----------
        if iteration % 500 == 0:
            i_big = np.argmax(a)
            i_small = np.argmin(a)
            delta = min(a[i_big] * 0.02, target_sum * 0.01)
            if delta > 0:
                a_tmp = a.copy()
                a_tmp[i_big] -= delta
                a_tmp[i_small] += delta
                a_tmp = project_onto_simplex(a_tmp, target_sum)
                s_tmp = _compute_score(a_tmp)
                if s_tmp < best_score:
                    best_score = s_tmp
                    best_a = a_tmp.copy()
                    a = a_tmp
                    print(f"Transfer step improved to {best_score:.6f}")

        if time.time() >= deadline:
            break

    return best_a, best_score


def fine_tune_sa(a: np.ndarray,
                 target_sum: float,
                 deadline: float,
                 base_time: float,
                 init_score: float | None = None) -> tuple[np.ndarray, float]:
    """
    Very cheap simulated‑annealing / random‑transfer fine‑tuning.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = _compute_score(cur_a) if init_score is None else init_score
    best_a = cur_a.copy()
    best_score = cur_score

    # temperature schedule
    T = 0.1 * best_score
    T_min = 1e-6
    decay = 0.9999

    while time.time() < deadline:
        i, j = np.random.choice(n, 2, replace=False)
        max_delta = min(cur_a[i], target_sum * 0.01)
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand()
        cur_a[i] -= delta
        cur_a[j] += delta
        # keep feasible
        cur_a = project_onto_simplex(cur_a, target_sum)
        new_score = _compute_score(cur_a)
        diff = new_score - cur_score

        accept = False
        if diff < 0:
            accept = True
        else:
            if np.random.rand() < np.exp(-diff / max(T, 1e-12)):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cur_a.copy()
                elapsed = time.time() - base_time
                print(f"SA improved to {best_score:.6f} (elapsed {elapsed:.1f}s)")
        else:
            # revert the move
            cur_a[i] += delta
            cur_a[j] -= delta

        T = max(T * decay, T_min)

    return best_a, best_score
```
