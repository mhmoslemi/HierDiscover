"""
Hybrid projected (smoothed) sub‑gradient descent + simulated‑annealing.

* Initialise with a diverse set of candidates:
  - uniform,
  - several parametric “U‑shapes’’ (linear, quadratic, exponential),
  - a few random Dirichlet draws,
  - the provided `height_sequence_1` if present.
* Keep the best candidate (according to the fast max‑convolution score).
* Main loop (until the time‑budget expires):
    – With probability `p_grad` perform a smoothed gradient step.
      The gradient is obtained by a soft‑max weighting of the self‑convolution
      entries:  w_k ∝ exp((conv_k‑max_conv)/τ).  The step size (η) and the
      smoothing parameter τ decay linearly with elapsed time.
    – Otherwise perform a random mass‑transfer move (Metropolis acceptance).
    – After every move project onto the ℓ₁‑simplex (sum = √(2 n)) to respect the
      normalisation.
    – Accept moves that improve the objective, otherwise accept with the
      standard SA probability exp(−Δ/T) where the temperature T also decays.
    – Track the best sequence seen.
* After the time runs out, polish the best sequence with a few hundred
  low‑temperature gradient steps.
The function `propose_candidate` returns the best list of heights found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers (no closures, no lambdas)
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself using an FFT of power‑of‑two length."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x_i ≥ 0, sum x_i = target}."""
    if target <= 0:
        return np.zeros_like(v)

    # Duchi et al. O(n) projected simplex algorithm (implemented via sorting)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.nonzero(rho_arr > 0)[0]
    if len(rho) == 0:
        theta = cssv[-1] / len(v)
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_max_conv(a: np.ndarray) -> float:
    """Return max entry of the self‑convolution of a (no normalisation)."""
    return float(conv_self_fft(a).max())


def evaluate_fast(a: np.ndarray, target_sum: float) -> float:
    """
    Fast version of evaluate_sequence for a fixed sum target.
    Since we keep sum(a) = target_sum, the objective simplifies to max(conv).
    """
    max_conv = compute_max_conv(a)
    # safeguard against division by zero (should never happen)
    if target_sum < 1e-8:
        return float("inf")
    # the original evaluate_sequence returns 2*n*max_conv/(sum^2)
    n = a.shape[0]
    return 2.0 * n * max_conv / (target_sum ** 2)


def generate_U_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
) -> np.ndarray:
    """Symmetric U‑shape (linear / quadratic / exponential) of length n."""
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def generate_gaussian(
    n: int,
    target_sum: float,
    sigma: float = 0.25,
) -> np.ndarray:
    """Symmetric Gaussian bump centred at the middle of the sequence."""
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    raw = np.exp(-0.5 * ((idx - mid) / (sigma * n)) ** 2)
    return project_onto_simplex(raw, target_sum)


def random_dirichlet(n: int, target_sum: float) -> np.ndarray:
    """Sample a random point on the simplex using Gamma(1) variables."""
    g = np.random.gamma(shape=1.0, scale=1.0, size=n)
    return project_onto_simplex(g, target_sum)


def random_mass_transfer(a: np.ndarray, scale: float = 0.1) -> np.ndarray:
    """
    Transfer a fraction `scale` of mass from a random index i to a different j.
    The operation preserves the ℓ₁‑norm.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    return a2


def smoothed_gradient(a: np.ndarray, tau: float) -> np.ndarray:
    """
    Compute the gradient of the smoothed max‑convolution objective.
    `tau` is the smoothing temperature (absolute, not a factor).
    Returns a vector of length n (the gradient w.r.t. each entry of a).
    """
    n = a.shape[0]
    conv = conv_self_fft(a)                 # length 2n‑1
    max_conv = conv.max()

    # Soft‑max weights (log‑sum‑exp)
    w = np.exp((conv - max_conv) / tau)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # Gradient = (2/τ) * convolution(w, a_rev) over the central n positions
    a_rev = a[::-1]
    L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    W = np.fft.rfft(w, n=L)
    AR = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(W * AR, n=L)[: len(w) + len(a_rev) - 1]
    grad = (2.0 / tau) * conv_full[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    `evaluate_sequence`.  The routine respects a time budget (≈1000 s)
    and returns the best sequence observed.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.25   # tiny safety margin

    # --------------------------------------------------------------
    # Determine problem size and normalisation
    # --------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        init_from_base = np.array(base, dtype=np.float64)
    else:
        n = np.random.randint(7500, 8700)   # a manageable size
        init_from_base = None

    target_sum = math.sqrt(2.0 * n)          # conventional normalisation

    # --------------------------------------------------------------
    # Phase 0 – generate a pool of diverse starting points
    # --------------------------------------------------------------
    candidates = []

    # uniform
    uniform = np.full(n, target_sum / n)
    candidates.append(uniform)

    # parametric U‑shapes (coarse grid)
    shape_kinds = ("linear", "quadratic", "exponential")
    param_grids = {
        "linear": np.linspace(0.0, 4.0, 21),          # 0.2 step
        "quadratic": np.linspace(0.0, 3.0, 16),      # 0.2 step
        "exponential": np.linspace(0.0, 2.0, 11),    # 0.2 step
    }
    for kind in shape_kinds:
        for p in param_grids[kind]:
            cand = generate_U_shape(n, target_sum, kind=kind, param=p)
            candidates.append(cand)

    # Gaussian bumps with a few sigmas
    for sigma in (0.15, 0.20, 0.25, 0.30):
        cand = generate_gaussian(n, target_sum, sigma=sigma)
        candidates.append(cand)

    # Random Dirichlet draws
    for _ in range(12):
        cand = random_dirichlet(n, target_sum)
        candidates.append(cand)

    # optional previous construction
    if init_from_base is not None:
        candidates.append(init_from_base)

    # --------------------------------------------------------------
    # Evaluate initial candidates, keep the best one
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    for idx, cand in enumerate(candidates):
        # use fast evaluation to keep the outset cheap
        score = evaluate_fast(cand, target_sum)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"[init] candidate {idx:02d} improves to {best_score:.6f}")

        if time.time() >= deadline:
            print("[init] time exhausted during initial sweep")
            return best_seq.tolist()

    # --------------------------------------------------------------
    # Phase 1 – hybrid annealing / projected smoothed gradient descent
    # --------------------------------------------------------------
    a = best_seq.copy()
    cur_score = best_score

    # Hyper‑parameters (all decay linearly with elapsed fraction)
    eta_initial = target_sum / n            # about 0.015 for n≈8600
    tau_initial = 0.3                      # multiplier of max_conv
    tau_min = 1e-6
    T_initial = 0.02                       # SA temperature (absolute)
    p_grad = 0.68                           # probability to take a gradient step
    step_scale_initial = 0.30              # initial random transfer size

    iteration = 0
    stagnation = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        fraction = elapsed / budget_s
        # Linear decays
        eta = eta_initial * max(0.01, 1.0 - fraction)
        tau_factor = max(tau_min, tau_initial * (1.0 - fraction))
        temperature = max(1e-12, T_initial * (1.0 - fraction))
        step_scale = max(0.005, step_scale_initial * (1.0 - fraction))

        # ----------------------------------------------------------
        # Propose a new state
        # ----------------------------------------------------------
        if np.random.rand() < p_grad:
            # Gradient step (smoothed)
            conv = conv_self_fft(a)
            max_conv = conv.max()
            tau = tau_factor * max_conv
            grad = smoothed_gradient(a, tau)
            a_new = a - eta * grad
            # keep non‑negative before projection
            a_new = np.maximum(a_new, 0.0)
            a_new = project_onto_simplex(a_new, target_sum)
        else:
            # Random mass transfer (Metropolis move)
            a_new = random_mass_transfer(a, scale=step_scale)
            a_new = project_onto_simplex(a_new, target_sum)

        # ----------------------------------------------------------
        # Evaluate proposal
        # ----------------------------------------------------------
        new_score = evaluate_fast(a_new, target_sum)

        # ----------------------------------------------------------
        # Acceptance rule (SA)
        # ----------------------------------------------------------
        accept = False
        if new_score <= cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / temperature)
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = a_new
            cur_score = new_score
            stagnation = 0
        else:
            stagnation += 1

        # ----------------------------------------------------------
        # Update global best
        # ----------------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{elapsed:6.1f}s] iter {iteration:06d} – new best {best_score:.6f}")

        # occasional jitter to escape deep plateaus
        if stagnation >= 4000:
            jitter = np.random.randn(n) * 0.0004
            a = project_onto_simplex(np.maximum(a + jitter, 0.0), target_sum)
            cur_score = evaluate_fast(a, target_sum)
            stagnation = 0
            print(f"[{elapsed:6.1f}s] iter {iteration:06d} – jitter applied")

        iteration += 1

    # --------------------------------------------------------------
    # Phase 2 – final fine‑tuning (low‑temperature gradient descent)
    # --------------------------------------------------------------
    a = best_seq.copy()
    eta_fine = eta_initial * 0.05
    tau_fine = tau_min
    print(f"[final] polishing start – best score {best_score:.6f}")

    polish_iters = 800
    for it in range(polish_iters):
        grad = smoothed_gradient(a, tau_fine)
        a_new = a - eta_fine * grad
        a_new = project_onto_simplex(np.maximum(a_new, 0.0), target_sum)
        new_score = evaluate_fast(a_new, target_sum)
        if new_score < best_score:
            best_score = new_score
            best_seq = a_new.copy()
            print(f"[polish] iter {it:04d} → {best_score:.6f}")
        a = a_new
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Return the best sequence observed (as a plain Python list)
    # ------------------------------------------------------------------
    print(f"[done] best final score {best_score:.6f}, length {len(best_seq)}")
    return best_seq.tolist()
