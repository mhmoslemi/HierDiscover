import time
import numpy as np

def _generate_powerlaw(p, sigma, max_len=8000, eps=1e-12):
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p)
    mask = a > eps
    return a[mask]

def _generate_exponential(lam, max_len=8000, eps=1e-12):
    i = np.arange(max_len, dtype=np.float64)
    a = np.exp(-lam * i)
    mask = a > eps
    return a[mask]

def _full_conv_fft(p):
    n = p.shape[0]
    L = 1 << ((2*n -2).bit_length())
    ft = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(ft * ft, n=L)[:2*n-1]
    conv = np.real(conv)
    conv[conv < 0] = 0.0
    return conv

def _score(p):
    # assumes p normalized to sum 1
    conv = _full_conv_fft(p)
    return 2.0 * p.shape[0] * np.max(conv)

def _subgradient(p, conv, top_k=10):
    n = p.shape[0]
    g = np.zeros_like(p)
    # find indices of top_k largest conv entries
    if top_k >= conv.size:
        ks = np.arange(conv.size)
    else:
        ks = np.argpartition(conv, -top_k)[-top_k:]
    for k in ks:
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        idx = np.arange(i_min, i_max + 1, dtype=int)
        # p_{k-i}
        g[idx] += p[k - idx]
    g *= 2.0
    return g

def _project_simplex(v):
    """ Euclidean projection onto simplex. """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n+1)
    rho = np.where(u - cssv/ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _refine(p, time_deadline, top_k=None):
    # p is a probability vector (sum=1)
    n = p.shape[0]
    if top_k is None:
        top_k = max(5, n // 200)   # about 0.5% of possible conv entries
    best_local_score = _score(p)
    best_local = p.copy()
    max_iters = 8000
    eta0 = 0.6
    it = 0
    while it < max_iters and time.time() < time_deadline:
        it += 1
        conv = _full_conv_fft(p)
        cur_score = 2.0 * n * np.max(conv)
        if cur_score < best_local_score - 1e-12:
            best_local_score = cur_score
            best_local = p.copy()

        # subgradient
        g = _subgradient(p, conv, top_k=top_k)
        eta = eta0 / np.sqrt(it)
        p = p * np.exp(-eta * g)
        # projection to simplex (normalize)
        # due to multiplicative, using simple division is enough, but we ensure sum=1
        sum_p = p.sum()
        if sum_p <= 0 or not np.isfinite(sum_p):
            # reinitialize
            p = np.abs(np.random.randn(n))
            sum_p = p.sum()
        p = p / sum_p
        # occasional jitter for exploration

            delta = (np.random.rand() - 0.5) * 0.01 * p[idx]
            p[idx] = max(p[idx] + delta, 0.0)
            p = p / p.sum()
    return best_local, best_local_score

"""
Optimiser for the step‑function evaluation problem.

The score after normalising a vector `p` to a probability distribution
(sum = 1) is  `score(p) = 2 * n * max(conv(p,p))`.  The algorithm works as
follows:

1.  Build a diverse pool of seed vectors:
    - uniform vectors,
    - Dirichlet draws with several concentration parameters,
    - truncated power‑law sequences,
    - truncated exponentials,
    - mixtures of two exponentials,
    - the previously best sequence (if available).

2.  For each seed, run a mirror‑descent / exponentiated‑gradient loop:
    * keep the vector on the probability simplex,
    * at each iteration compute the full self‑convolution by FFT,
    * take a sub‑gradient of the (approximate) ℓ∞‑norm of the convolution
      by summing the true gradients for the `top_k` largest convolution
      entries,
    * multiplicatively update the probabilities with a diminishing step
      size `η = η0 / sqrt(t)`,
    * occasionally jitter a random coordinate to escape plateaus.

   The loop stops when the global time budget is exhausted or after a
   fixed number of iterations.  The best vector seen during the whole
   run is returned.

The implementation respects the 1000 s time limit, prints intermediate
progress and finally yields a list of non‑negative heights that can be
passed directly to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def _generate_powerlaw(p: float, sigma: float,
                       max_len: int = 8000, eps: float = 1e-12) -> np.ndarray:
    """Truncated power‑law a_i = (i+σ)^{-p}."""
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p)
    return a[a > eps]


def _generate_exponential(lam: float,
                          max_len: int = 8000, eps: float = 1e-12) -> np.ndarray:
    """Truncated exponential a_i = exp(-λ·i)."""
    i = np.arange(max_len, dtype=np.float64)
    a = np.exp(-lam * i)
    return a[a > eps]


def _full_conv_fft(p: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution of a real vector with itself using FFT.
    Returns the real part of length 2·len(p)‑1 and clips tiny negatives.
    """
    n = p.shape[0]
    L = 1 << ((2 * n - 2).bit_length())          # next power of two
    ft = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(ft * ft, n=L)[: 2 * n - 1]
    conv = np.real(conv)
    conv[conv < 0] = 0.0
    return conv


def _score(p: np.ndarray) -> float:
    """Score = 2·n·max(conv(p,p)).  ``p`` must be normalised (sum=1)."""
    conv = _full_conv_fft(p)
    return 2.0 * p.shape[0] * np.max(conv)


def _subgradient(p: np.ndarray, conv: np.ndarray, top_k: int = 10) -> np.ndarray:
    """
    Sub‑gradient of the ℓ∞‑norm of the convolution.
    It sums the true gradients of the ``top_k`` largest convolution entries.
    """
    n = p.shape[0]
    g = np.zeros_like(p)
    if top_k >= conv.size:
        ks = np.arange(conv.size)
    else:
        ks = np.argpartition(conv, -top_k)[-top_k:]
    for k in ks:
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        idx = np.arange(i_min, i_max + 1, dtype=int)
        g[idx] += p[k - idx]
    g *= 2.0               # each pair appears twice
    return g


def _refine(p: np.ndarray, deadline: float,
            top_k: int | None = None) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent refinement of a probability vector ``p``.
    Returns the best vector seen and its score.
    """
    n = p.shape[0]
    if top_k is None:
        top_k = max(5, n // 200)          # about 0.5 % of all entries
    best_vec = p.copy()
    best_sc = _score(p)

    eta0 = 0.6
    it = 0
    max_iters = 8000

    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _full_conv_fft(p)
        cur_sc = 2.0 * n * np.max(conv)

        if cur_sc < best_sc - 1e-12:
            best_sc = cur_sc
            best_vec = p.copy()

        # sub‑gradient step
        g = _subgradient(p, conv, top_k=top_k)
        eta = eta0 / np.sqrt(it)
        p = p * np.exp(-eta * g)

        # re‑normalise onto the probability simplex (simple division suffices)
        s = p.sum()
        if s <= 0 or not np.isfinite(s):
            p = np.abs(np.random.randn(n))
            s = p.sum()
        p = p / s

        # tiny random jitter to encourage exploration
        if np.random.rand() < 0.02:
            idx = np.random.randint(n)
            delta = (np.random.rand() - 0.5) * 0.01 * p[idx]
            p[idx] = max(p[idx] + delta, 0.0)
            p = p / p.sum()

    return best_vec, best_sc

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The routine respects a ~1000 s time budget and
    returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    best_score = float("inf")
    best_seq = None

    # --------------------------------------------------------------
    # 1) Build a collection of diverse starting vectors
    # --------------------------------------------------------------
    seeds: list[np.ndarray] = []

    # Previous best (if provided by the environment)
    if "height_sequence_1" in globals():
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.size:
            seeds.append(prev)

    lengths = [2000, 4000, 6000, 8000]          # we keep all ≤ 8000
    for n in lengths:
        # -- Uniform ------------------------------------------------
        seeds.append(np.ones(n, dtype=np.float64))

        # -- Dirichlet (sparse ↔ dense) ---------------------------
        for alpha in (0.2, 1.0, 5.0):
            seeds.append(np.random.dirichlet(np.full(n, alpha)))

        # -- Power‑law families ------------------------------------
        for p_exp in (0.4, 0.5, 0.6):
            for sigma in (1e-3, 1e-2, 1e-1, 1.0):
                seq = _generate_powerlaw(p_exp, sigma, max_len=n, eps=1e-12)
                if seq.size >= max(200, n // 2):
                    seeds.append(seq)

        # -- Simple exponentials ------------------------------------
        for lam in (0.0005, 0.001, 0.002, 0.003):
            seq = _generate_exponential(lam, max_len=n, eps=1e-12)
            if seq.size >= max(200, n // 2):
                seeds.append(seq)

        # -- Mixture of two exponentials ---------------------------
        mixes = [
            (0.0005, 0.002, 0.3),
            (0.0010, 0.004, 0.5),
            (0.0015, 0.0035, 0.7),
        ]
        i = np.arange(n, dtype=np.float64)
        for lam1, lam2, w in mixes:
            a = w * np.exp(-lam1 * i) + (1.0 - w) * np.exp(-lam2 * i)
            a = a[a > 1e-12]
            if a.size >= max(200, n // 2):
                seeds.append(a)

    # --------------------------------------------------------------
    # 2) Refine each seed and keep the best vector found
    # --------------------------------------------------------------
    for base in seeds:
        if time.time() >= deadline:
            break

        p = np.asarray(base, dtype=np.float64)
        if p.sum() == 0.0:
            continue

        # Normalise to a probability vector
        p = p / p.sum()

        # Truncate length if somehow larger than 8000
        if p.shape[0] > 8000:
            p = p[:8000]
            p = p / p.sum()

        n = p.shape[0]

        # Initial score
        cur_sc = _score(p)
        if cur_sc < best_score:
            best_score = cur_sc
            best_seq = p.tolist()
            print(f"Seed init | n={n:5d} | score={best_score:.6f}")

        # Refine
        refined_vec, refined_sc = _refine(p, deadline)
        if refined_sc < best_score:
            best_score = refined_sc
            best_seq = refined_vec.tolist()
            print(f"Refined  | n={n:5d} | new best {best_score:.6f}")

            if best_score < 1.503:
                print("Target reached – terminating early.")
                return best_seq

    # --------------------------------------------------------------
    # 3) Return the best sequence discovered
    # --------------------------------------------------------------
    print("Final best score:", best_score,
          "length:", len(best_seq) if best_seq else None)
    return best_seq if best_seq is not None else []  # empty list signals failure
