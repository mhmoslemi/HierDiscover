    conv = conv_self_fft(a)  # length 2n-1
    max_val = conv.max()

"""
Multi‑phase optimizer for the non‑negative step‑function problem.

Phases
------
1.  Broad parametric sweep over several “U‑shaped” families (linear,
    quadratic, exponential) and a few random simplex vectors.
2.  L2‑flattening stage – minimise the squared deviation of the
    auto‑convolution from its mean, using an FFT‑based gradient.
3.  Smoothed max‑gradient descent (log‑sum‑exp), similar to the
    reference implementation but with slower decay of the temperature.
4.  Hard‑max sub‑gradient descent – directly follows the gradient of the
    currently maximal convolution entry.
5.  A final polishing stage consisting of tiny random pair‑mass transfers.

All updates keep the vector on the ℓ₁‑simplex (non‑negative, prescribed
sum) using the Euclidean projection of Duchi et al.  The routine respects
the 1000 s time budget and returns the best sequence seen so far.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a with itself using a power‑of‑two FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex
        {x | x_i ≥ 0, sum_i x_i = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, target / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def generate_U_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
) -> np.ndarray:
    """
    Produce a symmetric “U‑shaped’’ sequence of length n.
    kind ∈ {linear, quadratic, exponential}.
    param ≥ 0 controls the edge enhancement.
    The result is scaled to have ℓ₁‑norm `target_sum`.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a tiny amount of mass from one random index to another,
    then project back onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def smooth_gradient_step(
    a: np.ndarray,
    target_sum: float,
    eta: float,
    tau_factor: float,
) -> tuple[np.ndarray, float]:
    """
    One step of the log‑sum‑exp (soft‑max) smoothed gradient descent.
    Returns the new vector and the current max‑convolution value.
    """
    n = a.shape[0]

    conv = conv_self_fft(a)
    max_conv = conv.max()

    tau = max(tau_factor * max_conv, 1e-12)
    w = np.exp((conv - max_conv) / tau)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    a_rev = a[::-1]
    L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]

    grad = (2.0 / tau) * conv_full[n - 1 : n - 1 + n]

    a_new = a - eta * grad
    a_new = project_onto_simplex(np.maximum(a_new, 0.0), target_sum)
    return a_new, max_conv


def l2_flatten_step(
    a: np.ndarray,
    target_sum: float,
    eta: float,
) -> tuple[np.ndarray, float]:
    """
    Gradient step that reduces the squared deviation of the convolution
    from its mean value (i.e. flattens the auto‑correlation).
    """
    n = a.shape[0]
    conv = conv_self_fft(a)
    mean = target_sum ** 2 / (2 * n - 1)
    diff = conv - mean

    a_rev = a[::-1]
    L = 1 << ((len(diff) + len(a_rev) - 1).bit_length())
    D = np.fft.rfft(diff, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(D * A_rev, n=L)[: len(diff) + len(a_rev) - 1]

    grad = 4.0 * conv_full[n - 1 : n - 1 + n]

    a_new = a - eta * grad
    a_new = project_onto_simplex(np.maximum(a_new, 0.0), target_sum)
    new_max = conv_self_fft(a_new).max()
    return a_new, new_max


def subgradient_step_max(
    a: np.ndarray,
    target_sum: float,
    eta: float,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient step that directly follows the gradient of the
    currently maximal convolution entry.
    """
    n = a.shape[0]
    conv = conv_self_fft(a)
    max_val = conv.max()
    k_max = int(np.argmax(conv))

    low = max(0, k_max - (n - 1))
    high = min(k_max, n - 1)
    i_range = np.arange(low, high + 1)
    grad = np.zeros_like(a)
    grad[i_range] = 2.0 * a[k_max - i_range]

    a_new = a - eta * grad
    a_new = project_onto_simplex(np.maximum(a_new, 0.0), target_sum)
    new_max = conv_self_fft(a_new).max()
    return a_new, new_max
