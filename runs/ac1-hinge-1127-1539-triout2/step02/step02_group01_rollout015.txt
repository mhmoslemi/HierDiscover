"""
Propose a step‑function (non‑negative vector) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

where conv is the full linear self‑convolution.

The algorithm works in three stages.

1️⃣  A cheap parametric sweep over simple symmetric “U‑shapes’’
    (linear, quadratic, exponential) gives a decent starting point.

2️⃣  An *alternating LP / convex‑combination* loop (the idea from the
    statement).  For the current vector f we compute M = max(conv(f,f)).
    All indices where conv(f,f) is within a tiny tolerance of M are kept
    as an *active set*.  With those rows we solve a tiny linear programme

          maximise   Σ b_j
          subject to  (f * b)[k] ≤ M   for k in active set
                       b_j ≥ 0

    (the convolution (f*b) is linear in b).  The solution b is rescaled
    to have the required ℓ₁‑norm √(2 n) and mixed with the old vector:

          f ← (1‑t)·f + t·b_scaled

    The mixing factor t is slowly decreased; after each step we keep the
    best vector seen so far.

3️⃣  A short final polishing phase consisting of tiny random “pair
    transfers’’ (move a small fraction of mass from one index to another)
    and a few gradient‑descent steps with log‑sum‑exp smoothing.  The
    whole search respects a 1000 s budget and always returns the best
    sequence observed.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog  # linear programming solver (HiGHS)

# ----------------------------------------------------------------------
# Basic linear‑convolution helpers (FFT based, power‑of‑two size)
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution of a with itself (length 2 n‑1)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def conv_full_fft(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Full linear convolution of u with v."""
    L = 1 << ((len(u) + len(v) - 1).bit_length())
    fu = np.fft.rfft(u, n=L)
    fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fu * fv, n=L)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
# Geometry helpers (simplex projection, random perturbations)
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = target}."""
    if target <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, target / len(v))

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """Transfer a tiny amount of mass from one random index to another."""
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


# ----------------------------------------------------------------------
# Parametric “U‑shape’’ generator (used in the coarse sweep)
# ----------------------------------------------------------------------


def generate_U_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
) -> np.ndarray:
    """Symmetric U‑shaped sequence (linear / quadratic / exponential)."""
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance in [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)

def build_active_A_ub(f: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Build the matrix A_ub for constraints (f * b)[k] ≤ M,
    k ∈ active_idx.  Each row corresponds to a shifted copy of f.
    """
    n = f.shape[0]
    rows = []
    for k in active_idx:
        start = max(0, k - n + 1)
        end = min(k, n - 1)
        row = np.zeros_like(f)
        # coefficients are f[i] for i ∈ [start, end]
        row[start : end + 1] = f[start : end + 1]
        rows.append(row)
    if rows:
        return np.vstack(rows)
    else:
        # no active constraints – return a dummy row (won’t be used)
        return np.zeros((0, n), dtype=f.dtype)

    best_score = float("inf")

    shape_kinds = ("linear", "quadratic", "exponential")
    param_grids = {
        "linear": np.linspace(0.0, 4.0, 41),
        "quadratic": np.linspace(0.0, 3.0, 31),
        "exponential": np.linspace(0.0, 2.0, 21),
    }

    print(f"[{time.time():.2f}] Starting coarse sweep (n={n})")
    for kind in shape_kinds:
        for p in param_grids[kind]:
            cand = generate_U_shape(n, target_sum, kind=kind, param=p)
            score = evaluate_sequence(cand.tolist())
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"  New best {kind} (p={p:.2f}) → {best_score:.6f}")
            if time.time() >= deadline:
                print("Timeout during coarse sweep.")
                return best_seq.tolist()

    # baseline from previous run (if any)
    if base_arr is not None:
        score = evaluate_sequence(base_arr.tolist())
        if score < best_score:
            best_score = score
            best_seq = base_arr.copy()
            print(f"  Baseline improves score to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – alternating LP / mixing loop
    # ------------------------------------------------------------------
    f = best_seq.copy()
    t_mix = 0.25                     # initial mixing coefficient
    iteration = 0
    no_improve = 0

    print(f"[{time.time():.2f}] Entering LP‑mixing phase")
    while time.time() < deadline:
        iteration += 1

        conv_f = conv_self_fft(f)
        M = conv_f.max()
        cur_score = M                     # because sum = √(2 n) → objective = M

        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best_seq = f.copy()
            no_improve = 0
            print(f"Iter {iteration:06d} – new best score {best_score:.6f}")
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # Build a small active set (indices where the convolution is close
        # to its maximum) – we keep at most 150 constraints for speed.
        # --------------------------------------------------------------
        eps = max(1e-8, M * 1e-4)        # tolerance for “active’’ indices
        active_idx = np.where(conv_f >= M - eps)[0]
        if active_idx.size > 150:
            # keep the 150 largest entries only
            active_idx = np.argsort(conv_f)[-150:]

        # --------------------------------------------------------------
        # Linear programme: maximise Σ b_j  s.t. (f * b)[k] ≤ M for k∈active,
        #                    b_j ≥ 0.
        # --------------------------------------------------------------
        try:
            A_ub = build_active_A_ub(f, active_idx)
            b_ub = np.full(A_ub.shape[0], M)
            c = -np.ones(n)               # minimise –Σ b_j  ⇔ maximise Σ b_j
            bounds = [(0, None)] * n
            lp_res = linprog(
                c,
                A_ub=A_ub,
                b_ub=b_ub,
                bounds=bounds,
                method="highs",
                options={"presolve": True},
            )
            if lp_res.success and lp_res.x.sum() > 0:
                g = lp_res.x
                # rescale to the required ℓ₁‑norm
                g = g * (target_sum / g.sum())
                # convex combination
                f = (1.0 - t_mix) * f + t_mix * g
                f = project_onto_simplex(f, target_sum)   # numerical safety
            else:
                # LP failed – fall back to a tiny random move
                f = random_pair_transfer(f, target_sum, scale=0.005)
        except Exception as exc:
            print(f"LP exception: {exc}")
            f = random_pair_transfer(f, target_sum, scale=0.005)

        # --------------------------------------------------------------
        # Scheduling – slowly reduce the mixing weight, occasionally jitter
        # --------------------------------------------------------------
        if iteration % 50 == 0:
            t_mix = max(0.02, t_mix * 0.96)   # decay but keep a minimum
        if no_improve >= 3000:
            # heavy jitter to escape a plateau
            jitter = np.random.randn(n) * 0.0004
            f = project_onto_simplex(np.maximum(f + jitter, 0.0), target_sum)
            no_improve = 0
            print(f"Iter {iteration:06d} – jitter applied")
        if np.random.rand() < 0.012:
            f = random_pair_transfer(f, target_sum, scale=0.007)

        # safety break – avoid endless tight loops when time is almost over
        if time.time() + 1.0 >= deadline:
            break

    print(f"[{time.time():.2f}] LP‑mixing finished, best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – final polishing with tiny random pair transfers
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Starting final polishing")
    for _ in range(60):
        if time.time() >= deadline:
            break
        cand = random_pair_transfer(best_seq.copy(), target_sum, scale=0.004)
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"Polish improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Search finished – final best {best_score:.6f}")
    return best_seq.tolist()
