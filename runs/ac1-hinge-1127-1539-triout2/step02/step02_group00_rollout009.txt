import scipy.optimize as opt
import scipy.sparse as sp

def solve_lp(f, M):
    n = len(f)
    rows = []
    cols = []
    data = []
    # for each constraint k (0..2n-2)
    # for each j such that 0 <= k-j < n => j in [max(0, k-(n-1)), min(k, n-1)]
    # coefficient = f[k-j]
    for k in range(2*n-1):
        j_start = max(0, k-(n-1))
        j_end = min(k, n-1)
        # loop over j in this range
        # but we can vectorize: indices j = range(j_start, j_end+1)
        # corresponding i = k - j
        for j in range(j_start, j_end+1):
            i = k - j
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b = np.full(2*n-1, M)
    c = -np.ones(n)   # maximize sum
    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs', options={'presolve': True, 'time_limit': remaining_time})
    # check success
    if not res.success:
        return None, None
    b_opt = res.x
    opt_sum = b_opt.sum()
    return b_opt, opt_sum

"""
Improved step‑function optimizer.

The algorithm combines three ideas:
1.  A projected sub‑gradient descent based on the exact max‑convolution
    (instead of the log‑sum‑exp smoothing used previously).
2.  An LP “mass‑redistribution” step: for the current sequence `a`,
    solve the linear program
        max   Σ_j b_j
        s.t.  (a * b)_k ≤ M  ∀k ,   b ≥ 0
    where `M = max(a * a)`.  The optimum pushes mass into positions where
    the convolution with `a` is cheap, which typically lowers the max‑
    convolution after a convex combination.
3.  A short stochastic hill‑climber that swaps tiny amounts of mass
    between two positions while keeping the ℓ¹‑norm fixed.

The optimiser runs as many restarts as the time budget allows, trying a
few vector lengths (≈1 k‑2 k).  For each restart it performs a few LP
updates, a quick projected sub‑gradient descent, and finally the hill‑
climber.  The best sequence found (according to the fast‐evaluation
`_score`) is returned.

All helper functions are top‑level; no closures or lambdas are used.
The code works with a single CPU and respects the 1000 s budget.
"""

# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """Fast version of the competition objective (no safety checks)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x≥0, Σx = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_step(a: np.ndarray, step: float) -> np.ndarray:
    """
    Perform one projected sub‑gradient step to minimise max(a*a).

    The sub‑gradient at the maximal index `k` is
        g_i = 2 * a_{k-i}   (if 0 ≤ k-i < n, else 0).
    """
    n = a.size
    # compute full convolution and locate the maximal index
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    k = int(np.argmax(conv))

    # build the sub‑gradient vector
    g = np.zeros_like(a)
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    # indices i in [start, end] contribute to g_{k-i}
    i_vals = np.arange(start, end + 1, dtype=int)
    g[k - i_vals] = 2.0 * a[i_vals]

    # gradient step and projection
    a_next = a - step * g
    target = np.sqrt(2.0 * n)
    a_next = _project_to_simplex(a_next, target)
    return a_next


def _subgradient_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simple projected sub‑gradient descent with a decreasing step size.
    The routine runs until the deadline and returns the best sequence seen.
    """
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)
    best = a.copy()
    best_val = _score(a)

    it = 0
    while time.time() < deadline:
        # step size: 1 / sqrt(iteration) (standard scheme)
        step = 0.5 / np.sqrt(it + 1)
        a = _subgradient_step(a, step)
        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.1f}] subgrad improved = {best_val:.6f}")
        it += 1
    return best


# ----------------------------------------------------------------------
# Linear‑programming mass redistribution step
# ----------------------------------------------------------------------


def _build_convolution_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1)×n matrix A where A_{k,j} = f_{k‑j} (or 0 if out of range).
    For any non‑negative vector b, (A @ b)_k = (f * b)_k.
    """
    n = f.shape[0]
    rows = []
    cols = []
    data = []

    # iterate over all rows (convolution indices)
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        # i = k - j runs over a contiguous block of f
        i_vals = np.arange(k - j_end, k - j_start + 1)  # same length as j_range
        j_vals = np.arange(j_start, j_end + 1)
        rows.extend([k] * len(j_vals))
        cols.extend(j_vals.tolist())
        data.extend(f[i_vals].tolist())
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _solve_lp_max_sum(f: np.ndarray, M: float, time_limit: float):
    """
    Solve   max Σ b_j   s.t.  (f * b)_k ≤ M  ∀k,  b ≥ 0.
    Returns the optimal b and its ℓ¹‑norm.
    """
    n = f.shape[0]
    A = _build_convolution_matrix(f)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # maximise sum → minimise -sum

    # HiGHS handles sparse A efficiently
    res = opt.linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=[(0, None)] * n,
        method="highs",
        options={"presolve": True, "time_limit": time_limit},
    )
    if not res.success:
        return None, None
    b_opt = res.x
    return b_opt, b_opt.sum()


def _lp_update(a: np.ndarray, deadline: float, max_iters: int = 5, alpha: float = 0.3):
    """
    Perform a few LP‑based updates.
    Each iteration solves the LP for the current a, rescales the solution,
    and mixes it back with mixing coefficient `alpha` (decreasing over time).
    The best vector seen is returned.
    """
    n = a.size
    target = np.sqrt(2.0 * n)
    best = a.copy()
    best_val = _score(a)

    cur_alpha = alpha
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        M = _max_convolution_fft(a)

        # give the LP a small time budget (the remaining time divided by remaining iters)
        remaining = deadline - time.time()
        lp_time = max(0.2, remaining / (max_iters - it + 1))

        b_opt, sum_b = _solve_lp_max_sum(a, M, lp_time)
        if b_opt is None or sum_b == 0:
            break

        # scale to the fixed ℓ¹‑norm
        b_scaled = b_opt * (target / sum_b)

        # convex combination
        a_new = (1 - cur_alpha) * a + cur_alpha * b_scaled
        a_new = _project_to_simplex(a_new, target)

        val_new = _score(a_new)
        if val_new < best_val:
            best_val = val_new
            best = a_new.copy()
            print(f"[{time.time():.1f}] LP iter {it} improved = {best_val:.6f}")

        a = a_new
        cur_alpha = max(cur_alpha * 0.9, 0.05)   # slowly decay mixing weight
    return best


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing (already used before)
# ----------------------------------------------------------------------


def _random_hill_climb(a: np.ndarray, deadline: float):
    """
    Very cheap hill‑climber that preserves the ℓ¹‑norm.
    Randomly transfers a fraction of mass from i to j and keeps improving moves.
    """
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    cur_val = best_val

    max_frac = 0.12
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        frac = rng.random() * max_frac
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        new_val = _score(a)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill‑climb improved = {best_val:.6f}")
        else:
            a[j] -= delta
            a[i] += delta
    return best_a, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimizes the
    `evaluate_sequence` objective.  The routine respects the given time
    budget, tries several vector lengths and random restarts, and combines
    three optimisation phases: an LP‑based redistribution, a sub‑gradient
    descent, and a final hill‑climb.  The best sequence discovered is
    returned as a plain Python list.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # one‑second safety margin
    deadline = time.time() + budget_s - 1.0

    best_seq = None
    best_score = np.inf

    # if a previous construction is available, use it as a fallback
    try:
        height_sequence_1  # noqa: F821
        prev = np.array(height_sequence_1, dtype=float)
        prev_score = _score(prev)
        best_seq = prev.copy()
        best_score = prev_score
        print(f"[{time.time():.1f}] using provided baseline score = {prev_score:.6f}")
    except NameError:
        pass

    # lengths to explore – larger lengths give more freedom but cost more time
    lengths = [1000, 1500, 2000]

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # decide how many restarts we can afford for this length
        # each restart will consume roughly a few dozen seconds
        max_restarts = 4
        for restart in range(max_restarts):
            if time.time() >= deadline:
                break

            # ----------- initialise a random decreasing profile -------------
            # exponent p drawn from a modest interval around the known good value
            p = 0.5 + rng.uniform(-0.08, 0.08)
            a0 = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
            a0 *= target_sum / a0.sum()
            a0 += rng.normal(scale=target_sum * 0.001, size=n)
            a0 = np.maximum(a0, 0.0)
            a0 = _project_to_simplex(a0, target_sum)

            # ----------- Phase 1: LP redistribution -------------------------
            phase_deadline = min(deadline, time.time() + 20.0)
            a_lp = _lp_update(a0, phase_deadline, max_iters=5, alpha=0.35)

            # ----------- Phase 2: Sub‑gradient descent ----------------------
            if time.time() < deadline:
                phase_deadline = min(deadline, time.time() + 25.0)
                a_sg = _subgradient_descent(a_lp, phase_deadline)
            else:
                a_sg = a_lp

            # ----------- Phase 3: fine hill‑climbing -----------------------
            if time.time() < deadline:
                a_hc, hc_score = _random_hill_climb(a_sg, deadline)
            else:
                a_hc = a_sg
                hc_score = _score(a_hc)

            # Keep the best found so far
            if hc_score < best_score:
                best_score = hc_score
                best_seq = a_hc.copy()
                print(f"** NEW GLOBAL BEST ** n={n} p={p:.4f} restart={restart} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # final sanity checks and return
    # ------------------------------------------------------------------
    if best_seq is None:
        # very crude fallback – uniform distribution
        n = 1000
        a = np.ones(n, dtype=np.float64)
        a *= np.sqrt(2.0 * n) / n
        best_seq = a

    # trim trailing zeros (they have no effect on the score)
    nonzeros = np.where(best_seq > 1e-12)[0]
    if nonzeros.size:
        best_seq = best_seq[: nonzeros[-1] + 1]

    # Ensure the output is a plain list of Python floats
    return best_seq.tolist()
