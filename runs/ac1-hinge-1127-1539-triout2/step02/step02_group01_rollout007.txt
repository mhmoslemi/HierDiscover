"""
Propose a non‑negative step‑function (vector of heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

under the usual normalisation sum(a)=sqrt(2 n).
The algorithm consists of three phases:

1. **Diverse initialisation** – uniform, a few parametric “U‑shapes’’,
   the supplied baseline (if any) and several random Dirichlet draws.
   All candidates are projected onto the ℓ₁‑simplex with the required sum.

2. **Symmetric Adam optimisation** – each candidate is optimised with an
   Adam‑style update on a smoothed version of the max‑convolution.
   * Early iterations use a log‑sum‑exp smoothing (temperature τ
     proportional to the current max‑convolution).
   * After a warm‑up the smoothing switches to a power‑norm (conv**(p‑1))
     with the exponent p slowly growing (≈2 → 64).  
   After every gradient step the vector is (a) forced to be non‑negative,
   (b) forced to be symmetric (a_i = a_{n‑1‑i}) and (c) projected back
   onto the ℓ₁‑simplex.

   Adaptive schedules decay the learning rate, the temperature factor
   and increase the exponent p.  If progress stalls a small random jitter
   is injected.

3. **Final polishing** – tiny random pair‑mass transfers (simulated‑annealing‑style)
   are applied to the best sequence until the time budget expires.

The best sequence seen across all starts is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a with itself using FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex
        {x | x_i ≥ 0, sum_i x_i = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, target / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def symmetrize(v: np.ndarray) -> np.ndarray:
    """Force symmetry a[i] = a[n-1-i] by averaging with its mirror."""
    return (v + v[::-1]) * 0.5


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a tiny amount of mass from one random index to another,
    then project back onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def generate_U_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
) -> np.ndarray:
    """
    Produce a symmetric “U‑shaped’’ sequence of length n.
    kind ∈ {linear, quadratic, exponential}.
    param ≥ 0 controls the edge enhancement.
    The result is projected onto the simplex with sum `target_sum`.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def random_dirichlet(n: int, target_sum: float, alpha: float) -> np.ndarray:
    """Draw a random point on the simplex from a Dirichlet(alpha) distribution."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return project_onto_simplex(x, target_sum)

# Optimisation primitives
# ----------------------------------------------------------------------


def grad_logexp(a: np.ndarray, max_conv: float, tau_factor: float) -> tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp smoothed max‑convolution.
    Returns (gradient, tau) where tau = tau_factor * max_conv.
    """
    tau = max(tau_factor * max_conv, 1e-12)
    # weight vector w_i ∝ exp((conv_i - max_conv) / tau)
    conv = conv_self_fft(a)
    w = np.exp((conv - max_conv) / tau)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # convolution w * a_rev, slice to obtain gradient
    a_rev = a[::-1]
    L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
    grad = (2.0 / tau) * conv_full[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, tau


def grad_power(a: np.ndarray, max_conv: float, p: float) -> np.ndarray:
    """
    Gradient of the p‑norm smoothed max‑convolution (p >= 2).
    Returns the gradient vector.
    """
    eps = 1e-12
    conv = conv_self_fft(a)
    # w_i ∝ conv_i^{p-1}
    w = np.power(conv + eps, p - 1.0)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    a_rev = a[::-1]
    L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
    grad = 2.0 * conv_full[len(a) - 1 : len(a) - 1 + len(a)]
    return grad


def adam_update(
    a: np.ndarray,
    grad: np.ndarray,
    m: np.ndarray,
    v: np.ndarray,
    t: int,
    lr: float,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """
    Perform one Adam update step (with projection handled outside).
    Returns (new_a, new_m, new_v).
    """
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad ** 2)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    return a, m, v

        # occasional jitter to escape plateaus
        if no_improve >= 2500:
            jitter = np.random.randn(n) * (target_sum / n) * 0.001
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = symmetrize(a)
            a = project_onto_simplex(a, target_sum)
            no_improve = 0
            lr *= 0.9
            print(f"Iter {t_iter:06d} – jitter injected")

        # periodic learning‑rate decay
        if t_iter % 1000 == 0:
            lr *= 0.98

        if time.time() >= deadline:
            break

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    `evaluate_sequence`.  The routine respects the time budget (≈1000 s)
    and returns the best vector seen so far.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.4   # safety margin

    # ------------------------------------------------------------------
    # Determine problem size (use supplied baseline if possible)
    # ------------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        base_arr = np.array(base, dtype=np.float64)
        # rescale baseline to the canonical L1‑norm sqrt(2 n)
        target_sum = np.sqrt(2.0 * n)
        base_arr = base_arr * (target_sum / base_arr.sum())
        init_sequences = [base_arr]
    else:
        # Choose a power‑of‑two length for cheap FFTs (≈8 k is a good trade‑off)
        n = 8192
        target_sum = np.sqrt(2.0 * n)
        init_sequences = []

    # ------------------------------------------------------------------
    # Build a modest but diverse set of starting points
    # ------------------------------------------------------------------
    # 1. uniform
    init_sequences.append(np.full(n, target_sum / n, dtype=np.float64))

    # 2. three simple U‑shapes
    for kind, param in (("linear", 2.0), ("quadratic", 1.5), ("exponential", 1.0)):
        init_sequences.append(generate_U_shape(n, target_sum, kind=kind, param=param))

    # 3. random Dirichlet draws with various concentrations
    for alpha in (0.2, 0.5, 1.0, 3.0):
        init_sequences.append(random_dirichlet(n, target_sum, alpha))

    # ------------------------------------------------------------------
    # Evaluate each start and keep the best as we go
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    for idx, seq in enumerate(init_sequences):
        if time.time() >= deadline:
            break
        # project onto simplex (in case of tiny numerical drift)
        seq = project_onto_simplex(seq, target_sum)
        # quick evaluation
        cur_score = conv_self_fft(seq).max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"Init {idx:02d} – initial max conv = {best_score:.6f}")

        # optimise this start
        seq_opt, score_opt = optimise_one_start(seq, target_sum, deadline)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = seq_opt.copy()
            print(f"Init {idx:02d} – improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final short polishing phase (random pair transfers)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – should never happen
        best_seq = np.full(n, target_sum / n, dtype=np.float64)

    print(f"[{time.time():.2f}] Polishing phase – current best max conv = {best_score:.6f}")

    while time.time() < deadline:
        cand = random_pair_transfer(best_seq.copy(), target_sum, scale=0.005)
        cand = symmetrize(cand)
        cand = project_onto_simplex(cand, target_sum)
        cand_score = conv_self_fft(cand).max()
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"Polish – new best max conv = {best_score:.6f}")
        # keep polishing until time runs out

    # ------------------------------------------------------------------
    # Return result as a plain Python list
    # ------------------------------------------------------------------
    print(f"Finished – best max conv achieved: {best_score:.6f}")
    return best_seq.tolist()
