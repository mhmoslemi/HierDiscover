"""
propose_candidate
=================

A multi‑restart optimizer that minimizes the evaluation function by flattening the
self‑convolution of a non‑negative step function.

The core idea
-------------
The score equals the maximum entry of the (linear) self‑convolution when the
vector is normalised to L1 = √(2 n).  We therefore minimise a smooth approximation
of the max‑convolution:  

    F(a) = (1/β)·log Σ_k exp(β·c_k) ,   c = a * a .

For large β this approaches max(c).  The gradient of F is a soft‑weighted sum of
the gradients of the individual convolution entries and can be computed
efficiently with one FFT‑based convolution.

Algorithm outline
-----------------
* pick a length n and an initial non‑negative vector a (random Dirichlet or a
  truncated version of the current best);
* normalise a so that Σ a = √(2 n);
* repeatedly
      – compute the full convolution c = a*a (FFT);
      – form softmax weights w_k ∝ exp(β·c_k) (β is slowly increased);
      – compute the gradient g_i = 2·Σ_j a_j·w_{i+j}  (one extra FFT);
      – apply an exponential (multiplicative) update a ← a·exp(−η·g);
      – renormalise to keep Σ a fixed;
      – occasional jitter to escape plateaus;
      – monitor the true max(c); when it improves, call the expensive
        ``evaluate_sequence`` and keep the best vector.
* run many independent restarts until the 1000 s budget is exhausted and
  return the best sequence found.

The use of a softmax surrogate gives a dense (rather than a top‑K) gradient,
which typically yields faster convergence than the previous “top‑K” method.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_full(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a using a real FFT.
    Returns the full convolution array of length 2·n‑1.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _softmax_weights(c: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute numerically stable softmax weights proportional to
    exp(beta * c).  The weights sum to 1.
    """
    # shift by max to avoid overflow
    shift = np.max(beta * c)
    exp_scaled = np.exp(beta * c - shift)
    return exp_scaled / exp_scaled.sum()


def _gradient_softmax(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the softmax surrogate with respect to a.
    g_i = 2 * Σ_j a_j * w_{i + j},
    where w has length 2·n‑1 (the same length as the convolution).
    Computed via a single FFT convolution.
    """
    n = a.shape[0]

    # reverse a for correlation → convolution
    a_rev = a[::-1]

    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    fft_len = _next_pow2(conv_len)

    Fa = np.fft.rfft(a_rev, n=fft_len)
    Fw = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]

    # the required slice is conv[n-1 : n-1 + n]
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


def _run_optimize_one_start(
    a0: np.ndarray,
    n: int,
    deadline: float,
    *,
    beta_init: float = 0.5,
    beta_final: float = 8.0,
    beta_inc: float = 0.0008,
    eta0: float = 0.6,
    decay: float = 0.99994,
    beta_momentum: float = 0.85,
    jitter_prob: float = 0.012,
    max_iters: int = 30000,
) -> tuple[np.ndarray, float]:
    """
    Optimise a single start using the softmax‑surrogate gradient.
    Returns the best vector seen (as np.ndarray) and its external score.
    """
    a = a0.copy()
    S = np.sqrt(2.0 * n)                 # required L1‑norm

    # external evaluation of the start point
    try:
        best_score = evaluate_sequence(a.tolist())
    except Exception:   # pragma: no cover
        best_score = float("inf")
    best_seq = a.copy()
    best_surrogate = float("inf")

    v = np.zeros_like(a)                 # momentum buffer
    eta = eta0
    beta = beta_init
    it = 0

    # main optimisation loop
    while time.time() < deadline and it < max_iters:
        it += 1

        # full convolution and true max entry
        c = _conv_full(a)
        cur_max = float(c.max())

        # soft‑max surrogate to obtain a dense gradient
        w = _softmax_weights(c, beta)

        # gradient of the surrogate (dense)
        g = _gradient_softmax(a, w)

        # momentum
        v = beta_momentum * v + (1.0 - beta_momentum) * g

        # multiplicative (exponentiated) update
        a = a * np.exp(-eta * v)

        # renormalise to keep Σ a = S
        total = a.sum()
        if total == 0.0:                     # safety – should not happen
            a.fill(S / n)
        else:
            a *= (S / total)

        # occasional jitter to leave tiny plateaus
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * 0.009
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()

        # step‑size decay and beta schedule
        eta *= decay
        beta = min(beta_final, beta + beta_inc)

        # monitor improvement of the true max‑convolution entry
        if cur_max < best_surrogate - 1e-12:
            best_surrogate = cur_max
            # evaluate external score only when we see a genuine surrogate
            # improvement – it is relatively cheap compared to a full call.
            try:
                ext = evaluate_sequence(a.tolist())
                if ext < best_score:
                    best_score = ext
                    best_seq = a.copy()
                    print(
                        f"[local improve] t={time.time():.1f}s  score={best_score:.6f}"
                        f"  n={n}  iter={it}  cur_max={cur_max:.6f}"
                    )
            except Exception:   # pragma: no cover
                pass

    return best_seq, best_score

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) yields a score
    lower than the current best bound (≈ 1.503).

    The routine spends the allotted wall‑clock time on many independent
    restarts of the optimiser described above.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin for clean exit

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # If a construction has been provided globally, record its score.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            if cand.ndim == 1 and cand.size > 0:
                sc = evaluate_sequence(cand.tolist())
                best_seq = cand.copy()
                best_score = sc
                print(
                    f"[baseline] height_sequence_1 score = {best_score:.6f}"
                    f"  length = {cand.size}"
                )
        except Exception:  # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # Search parameters
    # ------------------------------------------------------------------
    # Choose a set of lengths.  Larger lengths give more flexibility but are
    # slower; we keep the range modest to stay within the time budget.
    lengths = [2500, 4000, 6000, 8000, 12000, 16000]
    run_id = 0

    # ------------------------------------------------------------------
    # Main restart loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        n = int(rng.choice(lengths))

        # Initialise a0:
        # * with probability 0.35 reuse a prefix of the current best;
        # * otherwise draw a fresh Dirichlet (Gamma) sample.
        if best_seq is not None and best_seq.size >= n and rng.random() < 0.35:
            a0 = best_seq[:n].copy()
        else:
            # Gamma(1) = exponential gives a Dirichlet after normalisation.
            a0 = rng.random(n) + 1e-8

        # Normalise to the required L1‑norm √(2 n)
        S = np.sqrt(2.0 * n)
        a0 = a0 / a0.sum() * S

        # Run the optimiser for this start
        a_opt, score_opt = _run_optimize_one_start(a0, n, deadline)

        # Keep the global best
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[global improve] run={run_id} n={n} score={best_score:.6f}")

        run_id += 1

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        a = np.full(n, np.sqrt(2.0 * n) / n)
        best_seq = a
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] score={best_score:.6f}")

    print(f"[final] best score = {best_score:.6f}  length = {len(best_seq)}")
    return best_seq.tolist()
