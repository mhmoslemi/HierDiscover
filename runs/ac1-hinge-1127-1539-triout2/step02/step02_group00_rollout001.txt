def _subgradient_descent(a0, deadline):
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2*n)
    best_val = _score(a)
    best_a = a.copy()
    lr = target_sum * 0.5  # step scaling
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        # compute convolution
        n = a.size
        size = 1 << ((2*n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
        max_b = b.max()
        # compute argmax set
        ks = np.where(b == max_b)[0]
        # build p indicator vector
        p = np.zeros(2*n-1)
        p[ks] = 1.0 / len(ks)
        # compute gradient via correlation a_rev * p
        a_rev = a[::-1]
        conv_len = n + (2*n - 1) - 1  # 3n-2
        size2 = 1 << ((conv_len).bit_length())
        Ar_fft = np.fft.rfft(a_rev, n=size2)
        Pr_fft = np.fft.rfft(p, n=size2)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
        g = 2.0 * conv_rev[n-1 : n-1 + n]
        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            # step size decays progressively
            step = lr / (norm_g * (1.0 + it*0.001))  # or just lr/(norm_g*(1+it*something))
            a = a - step * g
        # projection
        a = _project_to_simplex(a, target_sum)
        # evaluate
        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        # jitter occasionally
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0)
            a = _project_to_simplex(a, target_sum)
        it += 1
    return best_a, best_val
```

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0
    best_seq = None
    best_score = np.inf

    # Optionally include prior best from global
    try:
        if 'height_sequence_1' in globals():
            seq0 = np.array(height_sequence_1, dtype=np.float64)
            # ensure sum normalization
            n0 = seq0.size
            target0 = np.sqrt(2*n0)
            seq0 = seq0 * (target0/seq0.sum())
            # project just in case
            seq0 = _project_to_simplex(seq0, target0)
            # refine this candidate
            remaining = deadline - time.time()
            if remaining > 30:
                a_opt, val = _annealed_gradient_descent(seq0, deadline - 5)
                a_opt, val = _subgradient_descent(a_opt, deadline - 2)
                if time.time() < deadline:
                    a_opt, val = _random_hill_climb(a_opt, deadline)
                if val < best_score:
                    best_score = val
                    best_seq = a_opt.copy()
    except Exception as e:
        pass

    lengths = [2000, 3000, 4000, 6000, 8000]
    restarts_per_len = 3

    total_runs = len(lengths) * restarts_per_len
    # compute per run allocation (approx)
    # but we will compute dynamically each iteration using remaining time / remaining runs
    runs_done = 0

    for n in lengths:
        target_sum = np.sqrt(2.0 * n)
        for r in range(restarts_per_len):
            runs_left = total_runs - runs_done
            remaining_time = deadline - time.time()
            if remaining_time <= 0:
                break
            allocated = remaining_time / runs_left * 0.9  # allocate most but leave buffer
            # at least small...
            if allocated < 5:
                allocated = remaining_time
            # allocate stages
            soft_time = allocated * 0.45
            subgrad_time = allocated * 0.35
            # remainder for hill climb
            hill_time = allocated * 0.18
            # also small buffer for next runs
            # initialize a0
            p = 0.5 + rng.uniform(-0.08, 0.08)
            a0 = (np.arange(1, n+1, dtype=np.float64) ** (-p))
            a0 *= target_sum / a0.sum()
            # jitter
            a0 += rng.normal(scale=target_sum*0.001, size=n)
            a0 = np.maximum(a0, 0.0)
            a0 = _project_to_simplex(a0, target_sum)

            # sub-deadlines:
            soft_deadline = min(deadline, time.time() + soft_time)
            a_opt, opt_score = _annealed_gradient_descent(a0, soft_deadline)

            # subgradient stage
            subgrad_deadline = min(deadline, time.time() + subgrad_time)
            a_opt, opt_score = _subgradient_descent(a_opt, subgrad_deadline)

            # hill climbing final stage
            if time.time() < deadline:
                hill_deadline = min(deadline, time.time() + hill_time)
                a_opt, opt_score = _random_hill_climb(a_opt, hill_deadline)

            if opt_score < best_score:
                best_score = opt_score
                best_seq = a_opt.copy()
                print(f"** NEW GLOBAL BEST ** len={n} restart={r} score={best_score:.6f}")

            runs_done += 1

    # final refinement on best_seq if time remains
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        if remaining > 10:
            a_opt, opt_score = _subgradient_descent(best_seq, deadline)
            if opt_score < best_score:
                best_score = opt_score
                best_seq = a_opt.copy()

"""
Propose a step‑function (non‑negative sequence) that minimizes

    2·n·max(conv(a,a)) / (sum(a)^2)

subject to a_i ≥ 0.  Because the sum is fixed to √(2·n) the objective
is exactly the maximal entry of the self‑convolution, so we try to make
the convolution as flat as possible.

Algorithm
---------

* We explore several vector lengths (2000 … 8000).  Longer vectors give
  more freedom to flatten the convolution but are more expensive.

* For each length we perform a few random restarts.  Initial vectors are
  generated either as a power‑law, a Dirichlet‑sample (sparse) or by
  interpolating the current global best sequence.

* Each restart consists of three phases

  1. **Soft‑max gradient descent** – a smooth approximation of the
     max‑convolution is minimised using projected gradient descent with
     an annealing schedule for the temperature β (the code from the
     previous run, slightly retuned).

  2. **Sub‑gradient descent** – once β is large the soft‑max gradient
     becomes an accurate sub‑gradient of the true max.  We finish the
     optimisation with a pure sub‑gradient method that directly uses
     the set of indices attaining the maximal convolution value.
     This converges quickly to a very flat convolution.

  3. **Random hill‑climbing** – a cheap local search that moves a tiny
     fraction of mass between two positions and keeps improving moves
     (helps escape small plateaus).

* All updates are projected onto the simplex
  `{x | x_i ≥ 0, Σ x_i = √(2·n)}` using the Euclidean projection of
  Duchi et al. (2008).  The projection guarantees the objective value
  stays comparable to the evaluation function (which rescales the
  sequence).

* The whole routine respects the global time budget (≈1000 s).  Time
  for each restart is allocated dynamically from the remaining budget
  (≈45 s per restart).  The best sequence found is returned.

The implementation uses only NumPy, FFTs for fast convolutions and
standard Python libraries.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """Fast version of the competition objective (no safety checks)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    # For a correctly normalised vector (s = sqrt(2*n)) the score equals max_b.
    return 2.0 * n * max_b / (s * s)


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    { x | x_i ≥ 0, Σ x_i = s }.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of the soft‑max approximation of the max‑convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Return (g, b) where
        b = a*a (full convolution, length 2·n‑1)
        g = ∇ (log‑sum‑exp_{β}(b))  = 2·(a ⋆ p)    with p_k ∝ exp(β·b_k)
    Both are computed with FFTs for speed.
    """
    n = a.shape[0]

    # ---- b = a*a -------------------------------------------------------
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ---- soft‑max probabilities p ---------------------------------------
    max_b = np.max(b)
    # numerical stability: shift by max_b before exponentiation
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)                # shape (2·n‑1,)

    # ---- gradient g = 2·(a ⋆ p) ---------------------------------------
    # cross‑correlation a ⋆ p can be obtained as the central part of
    # conv(a_rev, p) where a_rev = a[::-1]
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1        # = 3·n‑2
    size2 = 1 << ((conv_len).bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    # slice the required part: indices n‑1 … n‑1+n‑1 (total n entries)
    g = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Sub‑gradient for the true max‑convolution
# ----------------------------------------------------------------------


def _subgradient_max_conv(a: np.ndarray):
    """
    Compute a sub‑gradient of max(conv(a,a)) at `a`.  Returns gradient `g`
    of shape (n,) and the current maximal convolution value `max_b`.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = float(b.max())
    # indices where the maximum is attained (could be several)
    max_idx = np.where(b == max_b)[0]

    # build probability vector that equally splits weight over the argmax set
    p = np.zeros(2 * n - 1, dtype=np.float64)
    if max_idx.size == 0:
        # should never happen, but guard against zero division
        return np.zeros_like(a), max_b
    p[max_idx] = 1.0 / max_idx.size

    # gradient = 2 * (a_rev ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1        # 3·n‑2
    size2 = 1 << ((conv_len).bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return g, max_b


# ----------------------------------------------------------------------
# Stochastic hill‑climbing (local refinement)
# ----------------------------------------------------------------------


def _random_hill_climb(a: np.ndarray, deadline: float):
    """
    Very cheap hill‑climber that preserves the ℓ¹‑norm.
    Randomly transfers a fraction of mass from i to j and keeps improving moves.
    """
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    cur_val = best_val

    max_frac = 0.12
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        frac = rng.random() * max_frac
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        new_val = _score(a)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill‑climb improved = {best_val:.6f}")
        else:
            # revert
            a[j] -= delta
            a[i] += delta
    return best_a, best_val


# ----------------------------------------------------------------------
# Gradient descent with annealing (soft‑max)
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected gradient descent on the soft‑max approximation of the
    maximum convolution.  The temperature β is increased (annealed)
    while the step size adapts to the magnitude of the gradient.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 256.0
    step_base = target_sum * 0.20          # base step magnitude
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        # gradient at current temperature
        g, _ = _gradient_softmax(a, beta)

        # step size proportional to 1/(β·‖g‖) (the factor β keeps later stages safe)
        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            lr = step_base / (norm_g * (beta + 1.0))
            a = a - lr * g

        # projection onto the simplex (keeps non‑negativity and L¹‑norm)
        a = _project_to_simplex(a, target_sum)

        # occasional tiny random jitter to help escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # evaluate
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] beta={beta:.2f}  score={best_val:.6f}")

        # annealing schedule
        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent for the true max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected sub‑gradient descent directly on max(conv(a,a)).
    The step size decays slowly with the iteration counter.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0
    # initial step magnitude (chosen empirically)
    base_step = target_sum * 0.5

    while time.time() < deadline:
        g, max_b = _subgradient_max_conv(a)

        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            # decay the step with iteration count
            lr = base_step / (norm_g * (1.0 + it * 0.001))
            a = a - lr * g

        # projection back onto the simplex
        a = _project_to_simplex(a, target_sum)

        # evaluate
        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] subgrad iter={it}  score={best_val:.6f}")

        # tiny jitter every few hundred iterations
        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the provided
    `evaluate_sequence` objective.  The routine respects the time budget,
    tries several vector lengths and random restarts, combines soft‑max
    gradient descent, sub‑gradient descent and a final hill‑climb, and
    returns the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_seq = None
    best_score = np.inf

    # If a previous construction is available, give it a quick refinement
    # ------------------------------------------------------------------
    try:

            n0 = seq0.size
            target0 = np.sqrt(2.0 * n0)

            seq0 = _project_to_simplex(seq0, target0)

            if time.time() < deadline:
                seq0, val0 = _random_hill_climb(seq0, deadline)

            if val0 < best_score:
                best_score = val0

    except Exception:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    lengths = [2000, 3000, 4000, 6000, 8000]
    restarts_per_len = 3
    total_runs = len(lengths) * restarts_per_len
    runs_done = 0

            elif init_mode < 0.5:

            a0 = np.maximum(a0, 0.0)
