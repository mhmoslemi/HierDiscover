"""
Propose a step‑function (non‑negative sequence) that minimizes the evaluation
score 2·n·max(conv) / (sum(a)^2).  The score is equal to the maximal entry of
the linear self‑convolution when the L1‑norm of the vector is fixed, so we
directly minimise the maximal convolution coefficient.

The algorithm consists of two phases.

* Phase 1 – many independent restarts of a smooth‑max exponential‑gradient
  descent.  At each iteration we compute the full self‑convolution by FFT,
  obtain a soft‑max weighting of all shifts (temperature annealed
  progressively) and use the resulting weighted gradient (computed by an FFT
  correlation) in a multiplicative update with momentum.  Whenever the true
  maximum of the convolution improves we call the provided ``evaluate_sequence``
  and keep the best vector found.

* Phase 2 – a cheap local refinement around the best vector seen so far.  Random
  transfers of a tiny amount of mass between two positions are accepted only
  if they strictly lower the maximal convolution coefficient (computed again by
  FFT).  This simple hill‑climbing often pushes the score a little further.

The whole search respects a 1000 s wall‑clock budget and prints occasional
progress messages.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _conv_self_fft(a: np.ndarray):
    """
    Compute the full linear self‑convolution a*a using a single FFT.
    Returns the convolution array (length 2·n‑1), the FFT of ``a``, and the
    FFT length used.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv, Fa, fft_len


def _gradient_from_softmax(Fa, w: np.ndarray, fft_len: int, n: int):
    """
    Compute the sub‑gradient of the soft‑max surrogate.
    The gradient is 2 * (w ⋆ a) where “⋆” denotes the linear correlation
    (correlation of w with a).  The correlation is obtained via FFT:
        w ⋆ a = ifft( Fw * conj(Fa) )
    The returned vector has length n.
    """
    # FFT of the weight vector w (same length as the convolution)
    Fw = np.fft.rfft(w, n=fft_len)
    # Correlation w ⋆ a  (real‑valued, length = n + len(w) - 1)
    corr_full = np.fft.irfft(Fw * np.conj(Fa), n=fft_len)
    # The valid part starts at offset = len(w) - 1
    offset = len(w) - 1
    # Extract exactly n entries
    grad = 2.0 * corr_full[offset:offset + n]
    return grad


def _max_conv_fft(a: np.ndarray) -> float:
    """Return max entry of the linear self‑convolution of a (FFT version)."""
    conv, _, _ = _conv_self_fft(a)
    return float(conv.max())


# ----------------------------------------------------------------------
# Core optimisation for a single restart
# ----------------------------------------------------------------------


def _optimize_one_restart(a0: np.ndarray,
                          deadline: float,
                          max_iters: int = 50000,
                          eta0: float = 0.8,
                          beta_mom: float = 0.9,
                          decay_eta: float = 0.99995,
                          jitter_prob: float = 0.015,
                          beta_start: float = 0.05,
                          beta_factor: float = 1.001,
                          beta_max: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Exponential‑gradient descent with a soft‑max surrogate.
    Returns the best vector found in this run and its external score.
    """
    n = a0.shape[0]
    a = a0.copy()
    S = math.sqrt(2.0 * n)               # target L1‑norm (scale‑invariant)
    a *= S / a.sum()                     # normalise

    best_seq = a.copy()
    best_score = float('inf')
    best_surrogate = float('inf')        # smallest observed max(conv)

    v = np.zeros_like(a)                 # momentum buffer
    eta = eta0
    beta_tmp = beta_start                # current temperature

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # ----- convolution & current max -----
        conv, Fa, fft_len = _conv_self_fft(a)
        cur_max = float(conv.max())

        # keep track of the best surrogate (exact max of convolution)
        if cur_max < best_surrogate - 1e-12:
            best_surrogate = cur_max
            try:
                ext = evaluate_sequence(a.tolist())
                if ext < best_score:
                    best_score = ext
                    best_seq = a.copy()
                    print(f"[improve] t={time.time():.1f}s  n={n} it={it} "
                          f"score={best_score:.6f} maxconv={cur_max:.6f}")
            except Exception:
                pass

        # ----- soft‑max weights -----
        # w_k ∝ exp(beta_tmp * (conv_k - cur_max))  (stable because max exponent = 0)
        w_exp = np.exp(beta_tmp * (conv - cur_max))
        w = w_exp / w_exp.sum()   # shape (2n‑1,)

        # ----- gradient via FFT correlation -----
        g = _gradient_from_softmax(Fa, w, fft_len, n)

        # ----- momentum and multiplicative update -----
        v = beta_mom * v + (1.0 - beta_mom) * g
        a = a * np.exp(-eta * v)

        # ----- re‑normalise (maintain L1‑norm) -----
        total = a.sum()
        if total <= 0.0:
            a.fill(S / n)
        else:
            a *= (S / total)

        # ----- occasional jitter to escape plateaus -----
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * 0.01
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()

        # ----- schedules -----
        eta *= decay_eta
        beta_tmp = min(beta_max, beta_tmp * beta_factor)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Local refinement (simple hill‑climbing)
# ----------------------------------------------------------------------


def _local_refine(seq: np.ndarray,
                  time_limit: float,
                  max_steps: int = 50000,
                  step_scale: float = 0.02) -> np.ndarray:
    """
    Small random mass‑transfer moves that strictly lower max‑convolution.
    The routine stops when the time limit is reached or when no improvement
    occurs for many attempts.
    """
    n = seq.shape[0]
    a = seq.copy()
    cur_max = _max_conv_fft(a)
    best_max = cur_max
    best_a = a.copy()
    attempts_without_improve = 0
    it = 0

    while time.time() < time_limit and it < max_steps:
        it += 1
        # pick two distinct indices
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # transfer a small amount from i to j
        delta = step_scale * a.mean() * (np.random.rand() * 2 - 1)  # signed
        if delta > a[i]:
            delta = a[i]               # cannot go negative
        new_a_i = a[i] - delta
        new_a_j = a[j] + delta
        if new_a_i < 0.0:
            continue
        a[i] = new_a_i
        a[j] = new_a_j

        new_max = _max_conv_fft(a)
        if new_max < best_max - 1e-12:
            best_max = new_max
            best_a = a.copy()
            attempts_without_improve = 0
            # keep the improvement
            # (no external evaluate here – we will call it later)
        else:
            # revert
            a[i] = a[i] + delta
            a[j] = a[j] - delta
            attempts_without_improve += 1
            if attempts_without_improve > 2000:
                # give up early if we are stuck
                break

    return best_a

# Main entry point required by the environment
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that yields a low evaluation score.
    The function respects the 1000 s wall‑clock budget and returns the best
    sequence (as a plain Python list) found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # small safety margin

    best_seq = None
    best_score = float('inf')

    # --------------------------------------------------------------
    # If a construction is already present, record its score.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            if cand.ndim == 1 and cand.size > 0:
                sc = evaluate_sequence(cand.tolist())
                best_seq = cand.copy()
                best_score = sc
                print(f"[baseline] existing height_sequence_1 score = {best_score:.6f} "
                      f"length = {cand.size}")
        except Exception:
            pass

    # --------------------------------------------------------------
    # Search over several lengths.
    # --------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000, 20000]
    run_id = 0

    while time.time() < deadline:
        n = int(rng.choice(lengths))

        # ----- initialise a0 -------------------------------------------------
        if best_seq is not None and best_seq.size >= n and rng.random() < 0.4:
            # take a random slice of the current best and renormalise
            start = rng.integers(0, best_seq.size - n + 1)
            a0 = best_seq[start:start + n].copy()
        else:
            a0 = rng.random(n) + 1e-6  # ensure strictly positive

        # normalise to the L1‑norm √(2·n) (scale‑invariant for the score)
        S = math.sqrt(2.0 * n)
        a0 = a0 / a0.sum() * S

        # ----- run a single optimisation instance ----------------------------
        a_opt, score_opt = _optimize_one_restart(a0, deadline)

        # --------------------------------------------------------------
        # Keep the best overall candidate
        # --------------------------------------------------------------
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[global improve] run={run_id} n={n} score={best_score:.6f}")

        run_id += 1

    # --------------------------------------------------------------
    # Phase 2 – local refinement around the best candidate obtained.
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("[refine] starting local refinement")
        refined = _local_refine(best_seq, deadline, max_steps=200000)
        # evaluate the refined candidate for the final score
        try:
            refined_score = evaluate_sequence(refined.tolist())
            if refined_score < best_score:
                best_score = refined_score
                best_seq = refined.copy()
                print(f"[refine improve] new best score = {best_score:.6f}")
        except Exception:
            pass

    # --------------------------------------------------------------
    # Fallback – uniform vector (should never be needed)
    # --------------------------------------------------------------
    if best_seq is None:
        n = 1000
        a = np.full(n, math.sqrt(2.0 * n) / n)
        best_seq = a
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] score={best_score:.6f}")

    print(f"[final] best score = {best_score:.6f}  length = {len(best_seq)}")
    return best_seq.tolist()
