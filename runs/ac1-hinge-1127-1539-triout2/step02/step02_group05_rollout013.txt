"""
Propose a non‑negative step‑function (a list of heights) that minimises

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works in three phases:

1. **Initialisation** – a few analytically‑motivated candidates
   (uniform, power‑law, exponential) are generated and the best one is kept.

2. **Adam optimisation with a smooth‑max surrogate** – we minimise a
   temperature‑controlled soft‑max of the convolution entries.
   The temperature (γ) is increased until the surrogate is almost the true max.
   After γ reaches a pre‑set cap we switch to a *hard* max (binary mask)
   which gives the exact sub‑gradient of the true objective.
   Adam’s adaptive learning‑rate together with a projection onto the simplex
   (non‑negative, fixed L1‑norm) keeps the iterates feasible.

3. **Fine‑tuning** – a few thousand plain sub‑gradient steps with a very
   small constant step‑size are performed to polish the solution.

The whole routine runs for the amount of time given by ``budget_s`` (default
1000 s) and returns the best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, top‑level)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fast_score(a: np.ndarray) -> float:
    """
    Fast version of the objective when ``sum(a) = sqrt(2*n)``.
    Returns max(conv(a, a)) which equals the original metric.
    """
    n = a.shape[0]
    nfft = _next_pow2(3 * n - 2)          # large enough for both conv & corr
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
    return float(conv.max())


def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Simple (non‑sorting) projection onto the simplex:
    clip negatives, then renormalise to the required L1‑sum.
    The result is feasible (non‑negative, sum = target_sum).
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # fallback: uniform vector
        return np.full_like(v, target_sum / v.shape[0])
    return v * (target_sum / s)


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises the evaluation
    function.  The routine follows the three‑phase strategy described in
    the module docstring.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # a small safety margin

    # --------------------------------------------------------------
    # 1. Choose the length of the sequence
    # --------------------------------------------------------------
    # If the runner supplies a previous construction we can reuse it,
    # otherwise we target a moderately large length (≈12 k) which gives
    # the optimiser more flexibility without a huge cost per iteration.
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
    except Exception:
        init_seq = None

    # target length
    MIN_LEN = 12000
    if init_seq is not None:
        n = max(init_seq.shape[0], MIN_LEN)
    else:
        n = MIN_LEN

    target_sum = np.sqrt(2.0 * n)   # the normalisation used in the metric

    # --------------------------------------------------------------
    # 2. Build a few deterministic/random seeds and keep the best
    # --------------------------------------------------------------
    best_a = None
    best_score = np.inf

    # (a) start from the supplied construction (padded/truncated if needed)
    if init_seq is not None:
        if init_seq.shape[0] < n:
            a = np.concatenate([init_seq, np.zeros(n - init_seq.shape[0])])
        else:
            a = init_seq[:n].copy()
        a = _project_onto_simplex(a, target_sum)
        sc = _fast_score(a)
        best_a, best_score = a.copy(), sc

    # (b) uniform vector – cheap baseline
    a = np.full(n, target_sum / n)
    sc = _fast_score(a)
    if sc < best_score:
        best_a, best_score = a.copy(), sc

    # (c) a few random power‑law / exponential candidates
    for _ in range(30):
        if time.time() >= deadline:
            break
        p = rng.uniform(0.2, 1.2)                          # exponent
        offset = rng.uniform(0.1, 5.0)                     # shift to avoid division by zero
        a = 1.0 / np.power(np.arange(n, dtype=np.float64) + offset, p)
        a = _project_onto_simplex(a, target_sum)
        sc = _fast_score(a)
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # (d) exponential candidates
    for _ in range(10):
        if time.time() >= deadline:
            break
        r = rng.uniform(0.001, 0.02)                      # decay rate
        a = np.exp(-r * np.arange(n, dtype=np.float64))
        a = _project_onto_simplex(a, target_sum)
        sc = _fast_score(a)
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # --------------------------------------------------------------
    # 3. Optimisation – Adam on soft‑max → hard‑max sub‑gradient
    # --------------------------------------------------------------
    a = best_a.copy()

    # FFT size (single size works for both convolutions and correlations)
    nfft = _next_pow2(3 * n - 2)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    lr = 0.25                # initial learning rate
    lr_min = 0.001
    lr_decay = 0.9996

    # soft‑max temperature schedule
    gamma = 1.0
    gamma_inc = 1.002
    gamma_max = 500.0        # after this we switch to a hard max

    iteration = 0
    last_improve_iter = 0

    while time.time() < deadline:
        iteration += 1
        t += 1

        # ---- convolution a * a -------------------------------------------------
        af = np.fft.rfft(a, n=nfft)
        conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
        max_conv = conv.max()

        # ---- decide between soft‑max and hard‑max -------------------------------
        if gamma < gamma_max:
            # soft‑max weights (numerically stable)
            w = np.exp(gamma * (conv - max_conv))
            w_sum = w.sum()
            if w_sum == 0.0:
                w[:] = 1.0 / w.size
            else:
                w /= w_sum
        else:
            # exact sub‑gradient: uniform weight on all maximisers
            tol = 1e-12
            mask = conv >= (max_conv - tol)
            cnt = np.count_nonzero(mask)
            w = np.zeros_like(conv)
            if cnt == 0:                # extremely unlikely, fallback
                w[conv.argmax()] = 1.0
            else:
                w[mask] = 1.0 / cnt

        # ---- gradient of the (soft‑)max ---------------------------------------
        wf = np.fft.rfft(w, n=nfft)
        af_rev = np.fft.rfft(a[::-1], n=nfft)
        corr_full = np.fft.irfft(wf * af_rev, n=nfft)
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]   # length‑n slice

        # ---- Adam update -------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- project back onto the simplex (positivity + fixed L1) ------------
        a_new = _project_onto_simplex(a_new, target_sum)

        # ---- occasional jitter to escape plateaus ------------------------------
        if iteration % 101 == 0:
            jitter = 1e-4 * rng.standard_normal(n)
            a_new = _project_onto_simplex(a_new + jitter, target_sum)

        # ---- evaluate -----------------------------------------------------------
        score_new = _fast_score(a_new)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            last_improve_iter = iteration
            elapsed = time.time() - start
            print(
                f"Iter {iteration:6d}: new best {best_score:.6f} (elapsed {elapsed:.1f}s)"
            )

        # ---- move to the new point ---------------------------------------------
        a = a_new

        # ---- schedule updates ---------------------------------------------------
        gamma = min(gamma * gamma_inc, gamma_max)
        lr = max(lr * lr_decay, lr_min)

        # ---- early‑restart if stuck for a long while ----------------------------
        if iteration - last_improve_iter > 20000 and gamma >= gamma_max:
            # re‑inject a tiny random perturbation around the current best
            a = best_a * (1.0 + 1e-3 * rng.standard_normal(n))
            a = _project_onto_simplex(a, target_sum)
            last_improve_iter = iteration   # reset counter

        # safety break
        if iteration % 5000 == 0 and time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 4. Fine‑tuning phase – plain sub‑gradient steps with a tiny step
    # ------------------------------------------------------------------
    fine_iters = 3000
    lr_fine = 5e-4
    for _ in range(fine_iters):
        if time.time() >= deadline:
            break
        # hard‑max weight (exact sub‑gradient)
        af = np.fft.rfft(a, n=nfft)
        conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
        max_conv = conv.max()
        mask = conv >= (max_conv - 1e-12)
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv)
        w[mask] = 1.0 / max(cnt, 1)

        wf = np.fft.rfft(w, n=nfft)
        af_rev = np.fft.rfft(a[::-1], n=nfft)
        corr_full = np.fft.irfft(wf * af_rev, n=nfft)
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]

        a = a - lr_fine * grad
        a = _project_onto_simplex(a, target_sum)

        score_new = _fast_score(a)
        if score_new < best_score:
            best_score = score_new
            best_a = a.copy()
            elapsed = time.time() - start
            print(
                f"Fine‑tune: new best {best_score:.6f} (elapsed {elapsed:.1f}s)"
            )

    # ------------------------------------------------------------------
    # 5. Return the best sequence discovered
    # ------------------------------------------------------------------
    return best_a.tolist()
