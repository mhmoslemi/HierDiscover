assistantfinal**Solution Overview**

The evaluation function  

\[
\text{score}(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2}
\]

is *scale free*: multiplying the whole vector by a constant does not change the
score.  Hence we may (and will) keep the vector on the probability simplex  
\(\sum a_i = 1,\;a_i\ge 0\) and minimise  

\[
\Phi(a)=2\,n\;\max (a*a) .
\]

The maximum of the auto‑convolution can be written as the
ℓ∞‑norm of a quadratic form.  It is non‑smooth, but can be smoothed with a
soft‑max:

\[
\Psi_\tau(a)=\frac{1}{\tau}\,
\log\!\Bigl(\sum_{k} e^{\tau (a*a)_k}\Bigr)
\qquad(\tau>0) .
\]

When \(\tau\to\infty\) we recover \(\max (a*a)\).  The gradient of
\(\Psi_\tau\) is easy to obtain:

\[
\frac{\partial\Psi_\tau}{\partial a_i}
  = 2\;\sum_{k}p_k\,a_{k-i},
\qquad 
p_k=\frac{e^{\tau (a*a)_k}}{\sum_j e^{\tau (a*a)_j}} .
\]

The inner sum is a **cross‑correlation** between the soft‑max weights `p`
and the current vector `a`.  Both the auto‑convolution and the correlation can
be computed in \(O(n\log n)\) time by FFT, which makes the method fast enough
to run thousands of iterations inside the 1000 s budget.

The optimisation proceeds in three stages

1. **Coarse parametric sweep** – scan a few analytic families
   (power‑law, exponential, mixed) and keep the best candidate.
2. **Soft‑max gradient descent** – start from that candidate,
   use Adam (β₁=0.9, β₂=0.999) on the smoothed objective,
   increase the temperature τ gradually (annealing) so the algorithm
   first smooths the landscape and later focuses on the true maximum.
   After every step the iterate is projected onto the simplex.
3. **Final polishing** – occasional random jitter, a restart
   using a Dirichlet mixture, and periodic evaluation of the exact score.
   The best sequence seen so far is returned.

All helper functions are top‑level, no closures or lambdas are used, and the
code respects the 1000 s wall‑clock limit.

```python
"""
Proposes a step‑function (non‑negative sequence) that minimises
evaluate_sequence.  The algorithm consists of:

1) a quick parametric sweep over several analytic families,
2) a temperature‑annealed Adam optimisation of a soft‑max
   approximation of max(a*a) using FFT‑based convolutions,
3) occasional random jitter / restarts and evaluation of the exact
   objective.  The best sequence found before the time budget expires
   is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simple families – used for the initial coarse sweep
# ----------------------------------------------------------------------


def _generate_powerlaw(p: float, sigma: float,
                       max_len: int = 8000, eps: float = 1e-12) -> np.ndarray:
    """a_i = (i+σ)^(-p)   truncated when a_i < eps."""
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p)
    return a[a > eps]


def _generate_exponential(lam: float,
                          max_len: int = 8000, eps: float = 1e-12) -> np.ndarray:
    """a_i = exp(-λ·i)   truncated when a_i < eps."""
    i = np.arange(max_len, dtype=np.float64)
    a = np.exp(-lam * i)
    return a[a > eps]


def _generate_mixed(p: float, sigma: float, lam: float,
                    max_len: int = 8000, eps: float = 1e-12) -> np.ndarray:
    """a_i = (i+σ)^(-p)·exp(-λ·i)   truncated when a_i < eps."""
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p) * np.exp(-lam * i)
    return a[a > eps]


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto {x≥0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works only on 1‑D arrays")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution / correlation helpers
# ----------------------------------------------------------------------


def _fft_convolve(x: np.ndarray, y: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution of x and y using FFT, result length = len(x)+len(y)-1."""
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    return np.fft.irfft(X * Y, size)[:len(x) + len(y) - 1]


def _softmax_gradient(a: np.ndarray, tau: float, fft_size: int):
    """
    Compute the gradient of  Ψ_τ(a)= (1/τ) log Σ exp(τ·(a*a))
    together with the (a*a) vector.
    Returns (grad, conv), where grad has the same shape as a.
    """
    n = a.shape[0]

    # auto‑convolution a*a
    conv = _fft_convolve(a, a, fft_size)          # length 2n‑1

    # soft‑max weights   p_k = exp(τ·conv_k) / Σ exp(τ·conv)
    scaled = tau * conv
    scaled -= np.max(scaled)                      # for numerical stability
    exp_scaled = np.exp(scaled)
    p = exp_scaled / np.sum(exp_scaled)           # length 2n‑1, sums to 1

    # correlation p * a (cross‑correlation = convolution with reversed a)
    a_rev = a[::-1]
    corr_full = _fft_convolve(p, a_rev, fft_size)   # length 3n‑2

    # gradient_i = 2 * corr[i + n – 1]
    grad = 2.0 * corr_full[n - 1: n - 1 + n]
    return grad, conv


# ----------------------------------------------------------------------
# Exact score (identical to evaluate_sequence but cheaper)
# ----------------------------------------------------------------------


def _exact_score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)²)   (sum(a) is 1 after the projection)."""
    n = a.shape[0]
    conv = np.convolve(a, a)          # plain numpy convolution, length 2n‑1
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:                 # safety – same threshold as evaluate_sequence
        return float("inf")
    return 2.0 * n * max_b / (sum_a ** 2)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list:
    """
    Search for a non‑negative step sequence that minimises the supplied
    evaluate_sequence.  A 1000 s wall‑clock budget is honoured; the best
    sequence found so far is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0      # 5 s safety margin

    best_score = float("inf")
    best_seq = None

    # --------------------------------------------------------------
    # 0) optional start from a previously discovered global sequence
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.array(height_sequence_1, dtype=float)
            cand = cand / cand.sum()
            sc = _exact_score(cand)
            if np.isfinite(sc):
                best_score, best_seq = sc, cand.tolist()
                print(f"Initial global sequence score = {best_score:.6f}")
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1) Broad parametric sweep (very cheap, gives a decent seed)
    # --------------------------------------------------------------
    max_len = 6000
    eps = 1e-12
    print("=== Phase 1 – parametric sweep ===")
    # power‑law family
    for p in np.linspace(0.4, 0.6, 9):
        for sigma in np.logspace(-4, 1, 12):
            if time.time() >= deadline:
                break
            seq = _generate_powerlaw(p, sigma, max_len, eps)
            if seq.size < 3:
                continue
            a = seq / seq.sum()
            sc = _exact_score(a)
            if sc < best_score:
                best_score, best_seq = sc, a.tolist()
                print(f"new best PL  p={p:.3f} σ={sigma:.5g} len={len(a)} score={sc:.6f}")
        if time.time() >= deadline:
            break

    # exponential family
    for lam in np.logspace(-4, -0.5, 10):
        if time.time() >= deadline:
            break
        seq = _generate_exponential(lam, max_len, eps)
        if seq.size < 3:
            continue
        a = seq / seq.sum()
        sc = _exact_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.tolist()
            print(f"new best EXP λ={lam:.5e} len={len(a)} score={sc:.6f}")

    # mixed family (few points – cheap)
    for p in (0.45, 0.5, 0.55):
        for sigma in (1e-3, 1e-2, 1e-1, 1.0):
            for lam in (1e-3, 5e-3, 1e-2):
                if time.time() >= deadline:
                    break
                seq = _generate_mixed(p, sigma, lam, max_len, eps)
                if seq.size < 3:
                    continue
                a = seq / seq.sum()
                sc = _exact_score(a)
                if sc < best_score:
                    best_score, best_seq = sc, a.tolist()
                    print(f"new best MIX p={p:.2f} σ={sigma:.4g} λ={lam:.4g} len={len(a)} score={sc:.6f}")

    # --------------------------------------------------------------
    # 2) Soft‑max Adam optimisation on the simplex
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform vector of moderate size
        n0 = 3000
        a = np.full(n0, 1.0 / n0, dtype=np.float64)
        best_score = _exact_score(a)
        best_seq = a.tolist()
        print("fallback to uniform sequence, score =", best_score)
    else:
        a = np.array(best_seq, dtype=np.float64)
        a = a / a.sum()                         # ensure exact simplex placement

    n = a.shape[0]
    # FFT size – must be >= 3·n to cover correlation length
    fft_sz = 1 << ((3 * n).bit_length())

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_lr = 0.05                              # base learning rate (before Adam scaling)
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule (annealing)
    tau_start, tau_end = 1.0, 30.0
    tau_ramp_iters = 3000     # linear increase over first ~3000 steps

    iteration = 0
    eval_every = 200          # evaluate exact score every this many iterations
    restart_every = 5000
    jitter_prob = 0.015

    print("=== Phase 2 – soft‑max Adam optimisation ===")
    while time.time() < deadline:
        iteration += 1

        # --- temperature (annealing) ---
        if iteration <= tau_ramp_iters:
            tau = tau_start + (tau_end - tau_start) * (iteration / tau_ramp_iters)
        else:
            tau = tau_end

        # --- gradient of the smoothed objective ---
        grad, conv = _softmax_gradient(a, tau, fft_sz)

        # --- Adam update ---
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = base_lr / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step * m_hat

        # --- projection onto simplex ---
        a = _simplex_projection(a_new)

        # --- occasional random jitter (helps escaping plateaus) ---
        if np.random.rand() < jitter_prob:
            idx = np.random.randint(n)
            delta = (np.random.rand() - 0.5) * 0.2 * a[idx]   # up to ±10 % of the entry
            a[idx] = max(0.0, a[idx] + delta)
            a = a / a.sum()

        # --- periodic exact evaluation ---
        if iteration % eval_every == 0 or iteration == 1:
            # we already have conv from the current iteration
            max_b = conv.max()
            score = 2.0 * n * max_b / (a.sum() ** 2)
            if np.isfinite(score) and score < best_score:
                best_score = float(score)
                best_seq = a.copy()
                print(f"iter {iteration:6d} → score = {best_score:.6f}")
                if best_score < 1.503:
                    print("Target bound reached – terminating early.")
                    break

        # --- occasional restart (mix with fresh Dirichlet) ---
        if iteration % restart_every == 0:
            rnd = np.random.rand(n)
            a = 0.6 * a + 0.4 * rnd
            a = a / a.sum()
            m.fill(0.0)
            v.fill(0.0)

    # --------------------------------------------------------------
    # Return the best sequence discovered
    # --------------------------------------------------------------
    print("final best score:", best_score, "length:", len(best_seq))
    return best_seq
```
