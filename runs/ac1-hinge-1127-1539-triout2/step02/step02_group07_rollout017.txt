import cvxpy as cp

def solve_lp_for_g(f, M):
    n = len(f)
    g = cp.Variable(n)
    constraints = [g >= 0]
    # for each shift k:
    for k in range(2*n - 1):
        # compute sum_i f_i * g_{k-i}
        # i ranges from max(0, k - (n - 1)) to min(k, n - 1)
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        # create list of indices pairwise
        # expression = sum f[i] * g[k-i]
        expr = 0
        for i in range(i_min, i_max+1):
            expr += f[i] * g[k-i]
        constraints.append(expr <= M)
    # objective maximize sum(g)
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    prob.solve(solver='ECOS', warm_start=True) # maybe other solvers
    return g.value, prob.value

"""
Step‑function optimizer based on the original alternating‑LP scheme
described in the problem statement.

For a non‑negative vector a we want to minimise

    score(a) = 2 * n * max(conv(a,a)) / (sum(a)**2)

which is scale‑invariant.  After normalising `sum(a)=1` the objective
becomes `2*n*max(conv(p,p))`.  The term `max(conv(p,p))` is the
largest entry of the autocorrelation of p.  The classic approach is to
alternately optimise the two factors of the convolution while keeping
the other fixed – this leads to a linear programme because
`(f*g)[k] = Σ_i f_i g_{k-i}` is linear in g when f is fixed.

The routine below

1. generates several seeds (uniform, random Dirichlet and the previous
   best sequence if available),
2. for each seed runs a few alternating‑LP iterations,
3. keeps the best sequence found (according to the provided
   `evaluate_sequence` function).

The linear programmes are solved with SciPy’s high‑performance ``linprog``
solver; the problem size is kept moderate (n≤800) so that many
restarts can be performed within the 1000 s budget.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _max_convolution(x: np.ndarray) -> float:
    """
    Returns max(conv(x, x)) for a real 1‑D array x.
    """
    # small vectors – direct convolution is fine and avoids FFT artefacts.
    c = np.convolve(x, x)
    return float(np.max(c))


def _build_toeplitz_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n Toeplitz matrix T_f such that T_f @ g = conv(f, g)
    (full linear convolution).  Used for the linear programme.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    T = np.zeros((rows, n), dtype=np.float64)

    # Each column i of T contains f shifted down by i positions.
    for i in range(n):
        T[i : i + n, i] = f
    return T


def _alternating_lp(f: np.ndarray,
                    max_iters: int = 12,
                    step: float = 0.35,
                    min_improve: float = 1e-8) -> np.ndarray:
    """
    Perform a few alternating LP updates starting from vector ``f``.
    ``f`` is assumed to be non‑negative with arbitrary sum.
    The algorithm repeatedly
        • computes M = max(conv(f,f)),
        • solves   max Σ g_j   s.t. (f * g) ≤ M , g ≥ 0,
        • rescales g to the proper normalisation,
        • updates f ← (1‑step)·f + step·g .
    The function returns the final vector f.
    """
    n = f.shape[0]
    # Normalise to a probability vector (sum = 1) – the scaling does
    # not affect the objective.
    f = f / np.sum(f)

    # Pre‑compute the Toeplitz matrix for the current f only once per
    # outer iteration (it changes after the update).
    for it in range(max_iters):
        M = _max_convolution(f)
        # Build linear constraints: T_f @ g ≤ M
        T = _build_toeplitz_matrix(f)

        # Linear programme: maximise Σ g_j   ⇔ minimise -Σ g_j
        c = -np.ones(n, dtype=np.float64)                # objective
        A_ub = T
        b_ub = np.full(T.shape[0], M, dtype=np.float64)  # upper bounds
        bounds = [(0.0, None)] * n

        res = linprog(c,
                      A_ub=A_ub,
                      b_ub=b_ub,
                      bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': 30})

        if not res.success or res.x is None:
            # LP failed – abort the inner loop.
            break

        g = res.x
        s = np.sum(g)

        if s == 0.0:
            break

        # Rescale g to the standard normalisation sqrt(2n)
        scale = np.sqrt(2.0 * n) / s
        g = g * scale

        # Convex combination with the previous f (small step)
        f = (1.0 - step) * f + step * g
        # Re‑normalise to sum = 1 (scale‑invariance)
        f = f / np.sum(f)

        # Optional early stop if the improvement stalls
        new_M = _max_convolution(f)
        if abs(new_M - M) < min_improve:
            break

    # Return a version that respects the clipping rules of
    # ``evaluate_sequence`` (0 ≤ x ≤ 1000).  The values are already tiny,
    # but we make the conversion explicit.
    f = np.clip(f, 0.0, 1000.0)
    return f.tolist()


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 2024, budget_s: float = 1000.0):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the 1000 s time limit
    and returns the best sequence discovered.

    The method uses several random seeds and the alternating‑LP scheme
    described above.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # tiny safety margin

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # 1) Different lengths to explore – keep them small enough for the LP.
    # ------------------------------------------------------------------
    lengths = (200, 400, 600, 800)   # 800 → ~1.3 M LP coefficients, fast enough

    for n in lengths:
        if time.time() >= deadline:
            break

        # ----------------------------------------------------------------
        # 2) Seed generation for this length
        # ----------------------------------------------------------------
        seeds = []

        # a) uniform vector (the theoretical optimum for the plain problem)
        seeds.append(np.full(n, 1.0 / n, dtype=np.float64))

        # b) random Dirichlet samples
        for _ in range(6):
            dir_sample = np.random.rand(n)
            dir_sample /= dir_sample.sum()
            seeds.append(dir_sample)

        # c) previous best if it matches the current length (truncate / pad)
        if "height_sequence_1" in globals():
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            if prev.size >= n:
                seeds.append(prev[:n] / prev[:n].sum())
            else:
                pad = np.zeros(n, dtype=np.float64)
                pad[:prev.size] = prev
                pad[:prev.size] /= prev.sum()
                seeds.append(pad)

        # ----------------------------------------------------------------
        # 3) Run the alternating‑LP optimisation from each seed
        # ----------------------------------------------------------------
        for base in seeds:
            if time.time() >= deadline:
                break

            seq = _alternating_lp(base,
                                  max_iters=15,
                                  step=0.35)

            # ----------------------------------------------------------------
            # 4) Evaluate the obtained candidate (using the provided routine)
            # ----------------------------------------------------------------
            try:
                score = evaluate_sequence(seq)
            except Exception:        # safety – any unexpected error is ignored
                continue

            if np.isfinite(score) and score < best_score:
                best_score = float(score)
                best_seq = seq
                print(f"New best | n={n} | score={best_score:.6f}")

                # If we reach the (optimistic) target we can stop early.
                if best_score < 1.503:
                    print("Target achieved – exiting.")
                    return best_seq

    # ------------------------------------------------------------------
    # 5) Final fallback: if nothing succeeded, return a uniform vector.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 400
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)
        print("Fallback uniform sequence used, score =", best_score)

    print("Final best score:", best_score, "length:", len(best_seq))
    return best_seq
