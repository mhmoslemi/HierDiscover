"""
Propose a step‑function (list of non‑negative heights) that minimises
evaluate_sequence(sequence).  

The algorithm is a large‑scale stochastic optimisation that repeatedly
starts from simple “shapes” (uniform, linear, exponential, Dirichlet)
and improves them by exponentiated‑gradient descent on a surrogate
objective – the maximum entry of the self‑convolution.  The surrogate
gradient is obtained from the K largest convolution entries (K is
increased during the run).  The best sequence found during the whole
budget is returned.

The optimisation is fully vectorised and uses FFT‑based convolution, so
even sequences of length 20 000 can be handled many times within the
~1000 s time limit.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_max(a: np.ndarray) -> tuple[np.ndarray, float, int]:
    """
    Linear self‑convolution a*a via FFT.
    Returns (full_conv, max_value, index_of_max).
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    # next power‑of‑two for speed
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    idx = int(np.argmax(conv))
    return conv, float(conv[idx]), idx


def _gradient_from_shifts(a: np.ndarray, shifts: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max‑convolution w.r.t. a, summed over the supplied shift
    indices.  For each shift k we add 2·a_{k‑j} to component j whenever the
    partner index lies inside the vector.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    idx = np.arange(n)
    for k in shifts:
        partner = k - idx
        ok = (partner >= 0) & (partner < n)
        g[ok] += 2.0 * a[partner[ok]]
    return g


# ----------------------------------------------------------------------
# One optimisation run (exponentiated gradient + momentum)
# ----------------------------------------------------------------------


def _run_one_start(
    a0: np.ndarray,
    n: int,
    deadline: float,
    top_k: int = 5,
    eta0: float = 0.7,
    beta: float = 0.9,
    decay: float = 0.99997,
    jitter_prob: float = 0.01,
    max_iters: int = 100_000,
) -> tuple[np.ndarray, float]:
    """
    Optimise a single start vector ``a0``.
    Returns (best_vector, its external score).  The external score is
    computed only when the surrogate (max convolution) improves.
    """
    a = a0.copy()
    # keep the L1‑norm fixed at sqrt(2·n); the external score is scale‑free,
    # but fixing the norm avoids overflow and improves numerical stability.
    target_l1 = np.sqrt(2.0 * n)
    a = a / a.sum() * target_l1

    # initial external evaluation
    try:
        best_score = evaluate_sequence(a.tolist())
    except Exception:                     # defensive – should not happen
        best_score = float("inf")
    best_seq = a.copy()
    best_surrogate = float("inf")

    momentum = np.zeros_like(a)
    eta = eta0
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv, cur_max, _ = _conv_max(a)

        # choose the K largest convolution entries (indices)
        if conv.size <= top_k:
            shifts = np.arange(conv.size)
        else:
            # argpartition is O(n) – cheap even for large n
            shifts = np.argpartition(conv, -top_k)[-top_k:]

        # sub‑gradient w.r.t. these K active shifts
        g = _gradient_from_shifts(a, shifts)

        # momentum (exponential moving average)
        momentum = beta * momentum + (1.0 - beta) * g

        # exponentiated (multiplicative) update – guarantees positivity
        a = a * np.exp(-eta * momentum)

        # renormalise to keep the L1 norm fixed
        total = a.sum()
        if total == 0.0:
            a.fill(target_l1 / n)
        else:
            a *= target_l1 / total

        # occasional tiny jitter to escape tiny plateaux
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * 0.008
            a = np.clip(a + noise, 0.0, None)
            a *= target_l1 / a.sum()

        # learning‑rate decay
        eta *= decay

        # monitor improvement of the surrogate objective
        if cur_max < best_surrogate - 1e-13:
            best_surrogate = cur_max
            try:
                ext = evaluate_sequence(a.tolist())
                if ext < best_score:
                    best_score = ext
                    best_seq = a.copy()
                    print(
                        f"[improve] t={time.time():.1f}s  score={best_score:.6f} "
                        f"n={n} iter={it} cur_max={cur_max:.6f}"
                    )
            except Exception:
                pass

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function that gives a low value of evaluate_sequence.
    The function runs for at most ``budget_s`` seconds and returns the best
    sequence found.
    """
    rng = np.random.default_rng(12345)
    deadline = time.time() + budget_s - 5.0   # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # If a previous construction is already present, record its score.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            if cand.ndim == 1 and cand.size > 0:
                sc = evaluate_sequence(cand.tolist())
                best_seq = cand.copy()
                best_score = sc
                print(f"[baseline] existing score={best_score:.6f} len={cand.size}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Lengths to explore – larger lengths give finer granularity but cost
    # more per iteration.  The set is a compromise between coverage and speed.
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000, 20000]

    run_id = 0
    while time.time() < deadline:
        n = int(rng.choice(lengths))

        # ------------------------------------------------------------------
        # Initialisation strategies.
        #   1) uniform (the trivial construction)
        #   2) linear decreasing   a_i ∝ (n-i)
        #   3) exponential decay   a_i ∝ exp(-λ i)   (λ drawn)
        #   4) random Dirichlet
        #   5) a slice of the current global best (if long enough)
        # ------------------------------------------------------------------
        strategy = rng.integers(5)
        if strategy == 0:
            a0 = np.full(n, 1.0, dtype=float)
        elif strategy == 1:
            a0 = np.arange(n, 0, -1, dtype=float)          # n, n‑1, …, 1
        elif strategy == 2:
            lam = rng.uniform(0.0005, 0.005)               # decay rate
            a0 = np.exp(-lam * np.arange(n, dtype=float))
        elif strategy == 3:
            a0 = rng.random(n) + 1e-6                      # Dirichlet‑like
        else:   # strategy == 4
            if best_seq is not None and best_seq.size >= n and rng.random() < 0.3:
                a0 = best_seq[:n].copy()
            else:
                a0 = np.full(n, 1.0, dtype=float)

        # Normalise L1‑norm to sqrt(2·n) – this keeps numbers in a numerically
        # convenient range and makes the external score equal to the maximal
        # convolution entry (though this is not required for correctness).
        target_l1 = np.sqrt(2.0 * n)
        a0 = a0 / a0.sum() * target_l1

        # ------------------------------------------------------------------
        # Run a single optimisation instance.
        # ------------------------------------------------------------------
        # The inner optimiser starts with a modest top_k, then slowly raises
        # it to capture more of the convolution landscape as the search
        # proceeds.
        a_opt, score_opt = _run_one_start(
            a0, n, deadline, top_k=5, eta0=0.7, beta=0.9,
            decay=0.99997, jitter_prob=0.012, max_iters=80_000
        )

        # Keep the best overall vector.
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(
                f"[global‑improve] run={run_id} n={n} score={best_score:.6f}"
            )
        run_id += 1

    # ------------------------------------------------------------------
    # Fallback – uniform sequence (guaranteed to be feasible).
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        a = np.full(n, np.sqrt(2.0 * n) / n)
        best_seq = a
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] score={best_score:.6f}")

    print(f"[final] best score = {best_score:.6f} length = {len(best_seq)}")
    # Convert to plain Python list as required by the evaluation harness.
    return best_seq.tolist()
