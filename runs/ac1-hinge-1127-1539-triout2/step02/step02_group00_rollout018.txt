"""
Simple power‑law grid search + subgradient refinement.

The algorithm explores a family of decreasing power‑law sequences
a_i ∝ (i+1)^{-p}, normalises them so that Σ a_i = √(2n) (this fixes the
objective 2·n·max(conv)/Σa² to the maximum convolution value), picks
the best exponent, then performs a few rounds of exact sub‑gradient
descent on the *max* of the linear convolution (the gradient is a
simple cross‑correlation with a one‑hot vector at the active index).  After
projecting each iterate onto the ℓ¹‑simplex the method finishes with a
short stochastic hill‑climb.  The whole routine returns the best
sequence found before the time budget expires.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    return float(_full_convolution(a).max())


def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)²) (same as evaluate_sequence)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Projection onto the simplex Σx_i = s, x_i ≥ 0   (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto {x | x_i ≥ 0, Σ x_i = s}.
    Uses the O(n log n) algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray):
    """
    Compute a sub‑gradient of f(a)=max(conv(a,a)).
    Returns (g, max_b, k) where
        g is the sub‑gradient,
        max_b = max(conv(a,a)),
        k   = index of a (first) maximum entry.
    """
    n = a.size
    conv = _full_convolution(a)
    max_b = conv.max()
    k = int(conv.argmax())
    # Build gradient: for each i where the product a_i a_{k-i} appears,
    # add contributions a_{k-i} (to ∂/∂a_i) and a_i (to ∂/∂a_{k-i}) .
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    for i in range(i_start, i_end + 1):
        j = k - i
        if i == j:
            g[i] += a[i]  # derivative of a_i * a_i = 2 a_i, but we count once; later we multiply by 2
        else:
            g[i] += a[j]
            g[j] += a[i]
    # The above counts each product once; the true gradient of the
    # *full* quadratic term is 2 * a_{paired}.  Multiplying by 2 gives the exact
    # sub‑gradient of the maximal entry.
    g *= 2.0
    return g, max_b, k

def _subgradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    The ℓ¹‑norm Σ a_i is kept fixed to √(2·n) by projection after each step.
    A diminishing step size  η_t = α / √t is used.
    """
    n = a0.size
    target_sum = np.sqrt(2.0 * n)
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    alpha = target_sum * 0.1  # base step size
    t = 1
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, max_b, k = _subgradient_max(a)

        # step size: α / sqrt(t) / ||g||
        norm_g = np.linalg.norm(g)
        if norm_g == 0:
            break
        eta = alpha / (np.sqrt(t) * norm_g)

        a = a - eta * g
        # occasional random jitter to avoid stagnation
        if t % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.002, size=n)
            a = a + jitter

        # enforce feasibility
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        val = _objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}]  sub‑grad step {t}  max_idx={k}  score={best_val:.6f}")

        # simple diminishing schedule
        if t % 2000 == 0:
            alpha *= 0.8  # slowly shrink the base step
        t += 1

    return best_a, best_val

def _random_hill_climb(a: np.ndarray, deadline: float):
    """
    Transfer a random fraction of mass between two positions,
    keeping the ℓ¹‑norm unchanged, and keep improving moves.
    """
    n = a.size
    best_a = a.copy()
    best_val = _objective(a)
    cur_val = best_val
    max_frac = 0.12
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        frac = rng.random() * max_frac
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta

        new_val = _objective(a)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}]  hill‑climb improved = {best_val:.6f}")
        else:
            a[j] -= delta
            a[i] += delta
    return best_a, best_val

"""
Simple power‑law grid search + exact sub‑gradient refinement.

The search proceeds in three stages:

1.  Grid‑search a one‑parameter family
        a_i ∝ (i+1)^{-p}
    for several lengths `n` and exponents `p`.  The vector is scaled so
    that Σ a_i = √(2·n) – this fixes the objective
        2·n·max(conv(a,a))/Σa² = max(conv(a,a)).
    The best (n,p) pair gives a strong warm‑start.

2.  Starting from the warm‑start (with a tiny random jitter) we run a
    projected sub‑gradient descent on the *exact* max‑convolution.
    The sub‑gradient of `max(conv(a,a))` at the active index `k`
    is simply `g_j = a_j + a_{k-j}` for all `j` where both indices are
    inside the vector.  After each step the iterate is projected back
    onto the ℓ¹‑simplex `{x ≥ 0, Σx = √(2·n)}`.

3.  Finally a short stochastic hill‑climb moves a random fraction of
    mass between two positions and keeps improving moves.

The whole routine respects the 1000 s budget and returns the best
sequence found.  The code uses only NumPy and runs on a single CPU.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) via FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_full_convolution(a).max())


def _objective(a: np.ndarray) -> float:
    """
    Fast evaluation of the competition objective:
        2·n·max(conv(a,a)) / (sum(a)²)
    (identical to `evaluate_sequence` for the ranges we use).
    """
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Project `v` onto {x | x_i ≥ 0, Σ x_i = s}.
    Implements the Duchi‑Shalev‑Shwartz algorithm (O(n log n)).
    """
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv(a,a))
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray):
    """
    Compute a sub‑gradient of
        f(a) = max_k (a * a)_k .

    Returns (g, max_b, k) where
        g   – sub‑gradient vector,
        max_b – the maximal convolution entry,
        k   – index at which the maximum is attained (first one).
    """
    n = a.size
    conv = _full_convolution(a)
    max_b = conv.max()
    k = int(conv.argmax())

    idx = np.arange(n)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    mask = (idx >= lo) & (idx <= hi)
    partner = k - idx[mask]

    g = np.zeros_like(a)
    # for all admissible indices: g_j = a_j + a_{k-j}
    g[mask] = a[mask] + a[partner]
    return g, max_b, k


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected sub‑gradient descent on `max(conv(a,a))`.
    The ℓ¹‑norm Σa_i is kept fixed to √(2·n) by projection.
    A diminishing step size η_t = α / (√t·‖g‖) is used.
    """
    n = a0.size
    target_sum = np.sqrt(2.0 * n)
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    alpha = target_sum * 0.20          # base magnitude of the step
    t = 1
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, max_b, k = _subgradient_max(a)
        norm_g = np.linalg.norm(g)
        if norm_g == 0:
            break

        eta = alpha / (np.sqrt(t) * norm_g)
        a_new = a - eta * g
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_to_simplex(a_new, target_sum)

        val = _objective(a_new)
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
            print(
                f"[{time.time():.1f}]  sub‑grad iter {t}  max_idx={k}  score={best_val:.6f}"
            )

        # accept the new point (even if not better – the projection keeps feasibility)
        a = a_new

        # occasional tiny random jitter to kick the method out of plateaus
        if t % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        t += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing (keeps ℓ¹‑norm fixed)
# ----------------------------------------------------------------------


def _random_hill_climb(a: np.ndarray, deadline: float):
    """
    Randomly move a fraction of mass from index i to j.
    Accept only improving moves (strict descent).
    """
    n = a.size
    best_a = a.copy()
    best_val = _objective(a)
    cur_val = best_val
    max_frac = 0.12
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        frac = rng.random() * max_frac
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta

        new_val = _objective(a)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}]  hill‑climb improved = {best_val:.6f}")
        else:
            # revert
            a[j] -= delta
            a[i] += delta
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a non‑negative step‑function (list of heights) that minimises
    the supplied `evaluate_sequence` objective.

    Strategy
    --------
    1.  Grid‑search over vector lengths and power‑law exponents.
    2.  Refine the best candidate with exact sub‑gradient descent.
    3.  Finish with a short stochastic hill‑climb.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    deadline = time.time() + budget_s - 1.0   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 1. Grid search over lengths and exponents
    # ------------------------------------------------------------------
    lengths = [800, 1200, 1500, 2000, 2500, 3000, 3500, 4000]
    p_grid = np.linspace(0.45, 0.68, 48)      # step ≈0.005

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        for p in p_grid:
            # power‑law shape a_i ∝ (i+1)^{-p}
            a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
            # scale to the ℓ¹‑simplex
            a *= target_sum / a.sum()
            score = _objective(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"** GRID BEST **  n={n}  p={p:.5f}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Sub‑gradient descent refinement (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None:
        n_best = best_seq.size
        # give the refinement up to half of the remaining budget
        refine_deadline = min(deadline, time.time() + (deadline - time.time()) * 0.5)

        for restart in range(3):
            if time.time() >= refine_deadline:
                break
            a0 = best_seq.copy()
            # small random jitter before starting descent
            jitter = rng.normal(scale=0.002 * np.sqrt(2.0 * n_best), size=n_best)
            a0 = np.maximum(a0 + jitter, 0.0)
            a0 = _project_to_simplex(a0, np.sqrt(2.0 * n_best))

            a_opt, opt_score = _subgradient_descent(a0, refine_deadline)
            if opt_score < best_score:
                best_score = opt_score
                best_seq = a_opt.copy()
                print(f"** SUB‑GRAD IMPROVEMENT **  restart={restart}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Final cheap hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a_opt, opt_score = _random_hill_climb(best_seq.copy(), deadline)
        if opt_score < best_score:
            best_score = opt_score
            best_seq = a_opt.copy()
            print(f"** FINAL HILL‑CLIMB IMPROVEMENT **  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        a = np.ones(n, dtype=np.float64)
        a *= np.sqrt(2.0 * n) / n
        best_seq = a

    # Trim trailing zeros for neatness (optional)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    print(f"*** FINAL RESULT ***  n={best_seq.size}  score={best_score:.6f}")
    return best_seq.tolist()
