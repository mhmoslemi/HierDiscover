"""
Propose a candidate sequence that minimises `evaluate_sequence`.

Algorithm
---------

1. **Initialisation**
   * Use an existing construction (`height_sequence_1`) if present,
     otherwise pick a length n ≈ 10 000.
   * Normalise the ℓ₁–norm to `sqrt(2·n)` – the same normalisation used
     in the reference paper.

2. **Parametric sweep**
   * Generate a large collection of analytic families (`linear`,
     `quadratic`, `exponential`, **geometric**) with three possible
     directions (symmetric, decreasing, increasing).  Each family is
     scanned over a grid of shape parameters.
   * Keep the best candidate (lowest convolution maximum).

3. **Gradient descent**
   * Perform projected gradient descent on a *smooth* approximation of
     the maximum (`log‑sum‑exp`).  The gradient is a cross‑correlation
     and is computed by FFT in `smooth_max_step`.
   * After the temperature has been cooled below a threshold we switch
     to the *hard*‑max gradient (gradient of the actual maximal
     convolution entry) via `hard_max_step`.
   * Momentum, learning‑rate decay, occasional random pair transfers
     and jitter are used to escape plateaus.

4. **Polishing (simulated annealing)**
   * Starting from the best sequence found, repeatedly propose small
     mass‑transfer moves.  Moves that improve the objective are always
     accepted; worsening moves are accepted with a decreasing
     temperature (Metropolis criterion).

All operations keep the vector non‑negative and constantly re‑project
onto the ℓ₁‑hyperplane, guaranteeing that the returned list is a valid
input for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures / lambdas)
# ----------------------------------------------------------------------


def fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors a and b using an FFT of power‑of‑2 length."""
    n, m = a.shape[0], b.shape[0]
    L = 1 << ((n + m - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)
    return conv[: n + m - 1]


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (length 2·len(a)-1)."""
    return fft_convolve(a, a)


def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector so that its ℓ₁‑norm equals *target*."""
    s = v.sum()
    if s > 0.0:
        return v * (target / s)
    # fallback – uniform distribution (should never happen)
    return np.full_like(v, target / v.shape[0])


def generate_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
    direction: str = "symmetric",
) -> np.ndarray:
    """
    Produce a deterministic shape of length *n*.

    *kind* – {"linear","quadratic","exponential","geometric"}.
    *direction* – {"symmetric","decreasing","increasing"}.
    *param* controls the strength of the edge/gradient (≥0).
    The shape is finally scaled so that the ℓ₁‑norm equals *target_sum*.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0

    if direction == "symmetric":
        # distance from the centre, normalised to [0,1]
        d = np.abs(idx - mid) / mid
        base = d
    elif direction == "decreasing":
        base = idx / (n - 1)          # 0 … 1
    elif direction == "increasing":
        base = (n - 1 - idx) / (n - 1)  # 1 … 0
    else:
        raise ValueError("unknown direction")

    if kind == "linear":
        raw = 1.0 + param * base
    elif kind == "quadratic":
        raw = 1.0 + param * base ** 2
    elif kind == "exponential":
        raw = np.exp(param * base)
    elif kind == "geometric":
        # geometric decay / growth; param is the exponent of e
        # For decreasing direction, we want large values on the left.
        if direction == "symmetric":
            raw = np.exp(-param * base)            # high centre → low edges
        elif direction == "decreasing":
            raw = np.exp(-param * idx)              # high left, low right
        else:  # increasing
            raw = np.exp(-param * (n - 1 - idx))    # high right, low left
    else:
        raise ValueError("unknown shape kind")

    raw = np.clip(raw, 0.0, None)
    return project_to_sum(raw, target_sum)


def random_pair_transfer(
    a: np.ndarray,
    target_sum: float,
    scale: float = 0.01,
) -> np.ndarray:
    """
    Transfer a tiny fraction of mass from a random source index to a random target.
    The total ℓ₁‑norm is preserved by re‑normalisation.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0.0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_to_sum(a, target_sum)


def jitter(
    a: np.ndarray,
    target_sum: float,
    sigma: float = 1e-3,
) -> np.ndarray:
    """Add Gaussian noise (clipped) and renormalise."""
    a = a + np.random.randn(a.shape[0]) * sigma
    a = np.clip(a, 0.0, None)
    return project_to_sum(a, target_sum)


def compute_score(a: np.ndarray) -> float:
    """Return max entry of the self‑convolution (the exact objective)."""
    return float(conv_self_fft(a).max())


# ----------------------------------------------------------------------
# Gradient‑descent steps
# ----------------------------------------------------------------------


def smooth_max_step(
    a: np.ndarray,
    target_sum: float,
    lr: float,
    tau: float,
    momentum: float,
    v_prev: np.ndarray,
) -> tuple[np.ndarray, np.ndarray]:
    """
    One iteration of gradient descent on the smoothed max (log‑sum‑exp).
    Returns the updated vector and the new momentum term.
    """
    n = a.shape[0]

    # 1) convolution a * a
    conv = conv_self_fft(a)                     # length 2n‑1

    # 2) soft‑max weights (log‑sum‑exp trick for stability)
    max_c = conv.max()
    exp_arg = (conv - max_c) / tau
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:
        # degenerate – fall back to hard max
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    # 3) gradient of the smoothed max:
    #    g_i = 2 * Σ_k w_k * a_{k-i}
    a_rev = a[::-1]
    grad_full = fft_convolve(w, a_rev)          # length 3n‑2
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]    # slice i = 0…n‑1

    # 4) projected direction (preserve the ℓ₁‑norm)
    direction = -(grad - np.mean(grad))

    # 5) momentum + step
    v = momentum * v_prev + direction
    a_new = a + lr * v

    # 6) enforce feasibility
    a_new = np.clip(a_new, 0.0, None)
    a_new = project_to_sum(a_new, target_sum)

    return a_new, v


def hard_max_step(
    a: np.ndarray,
    target_sum: float,
    lr: float,
    momentum: float,
    v_prev: np.ndarray,
) -> tuple[np.ndarray, np.ndarray]:
    """
    Gradient step for the *hard* max (the true objective).
    The gradient is 2·a_{K‑i} on the active index K.
    """
    n = a.shape[0]

    conv = conv_self_fft(a)
    K = int(np.argmax(conv))

    # compute gradient
    grad = np.zeros_like(a)
    i = np.arange(n)
    j = K - i
    mask = (j >= 0) & (j < n)
    grad[mask] = 2.0 * a[j[mask]]

    # zero‑sum direction
    direction = -(grad - np.mean(grad))

    v = momentum * v_prev + direction
    a_new = a + lr * v
    a_new = np.clip(a_new, 0.0, None)
    a_new = project_to_sum(a_new, target_sum)

    return a_new, v

    # Choose a length – reuse a known construction if it exists
    # --------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        use_base = True
    else:
        # a size that is large enough to give a good bound but still cheap
        n = np.random.randint(8000, 12000)
        use_base = False

    target_sum = np.sqrt(2.0 * n)                    # normalisation used in the paper

    # --------------------------------------------------------------
    # Initialise best sequence
    # --------------------------------------------------------------
    if use_base:
        best_seq = np.array(base, dtype=np.float64)
        best_seq = project_to_sum(best_seq, target_sum)
        best_score = compute_score(best_seq)
        print(f"[{time.time():.2f}] Start from supplied construction (n={n}) → score {best_score:.6f}")
    else:
        best_seq = np.full(n, target_sum / n, dtype=np.float64)   # uniform baseline
        best_score = compute_score(best_seq)
        print(f"[{time.time():.2f}] Uniform baseline (n={n}) → score {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – parametric sweep (including geometric families)
    # --------------------------------------------------------------
    families = [
        ("symmetric", "linear"),
        ("symmetric", "quadratic"),
        ("symmetric", "exponential"),
        ("symmetric", "geometric"),
        ("decreasing", "linear"),
        ("decreasing", "quadratic"),
        ("decreasing", "exponential"),
        ("decreasing", "geometric"),
        ("increasing", "linear"),
        ("increasing", "quadratic"),
        ("increasing", "exponential"),
        ("increasing", "geometric"),
    ]

    param_grids = {
        "linear": np.linspace(0.0, 5.0, 26),          # 0 … 5 step 0.2
        "quadratic": np.linspace(0.0, 4.0, 21),      # 0 … 4 step 0.2
        "exponential": np.linspace(0.0, 3.0, 16),    # 0 … 3 step 0.2
        "geometric": np.linspace(0.0, 8.0, 21),      # decay strength
    }

    print(f"[{time.time():.2f}] Starting parametric sweep (n={n})")
    for direction, kind in families:
        for p in param_grids[kind]:
            cand = generate_shape(n, target_sum, kind=kind, param=p, direction=direction)
            score = compute_score(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"  New best {direction}/{kind} p={p:.3f} → {best_score:.6f}")
            if time.time() >= deadline:
                print("Time exhausted during parametric sweep.")
                return best_seq.tolist()

    # a handful of random (dense & sparse) initialisations
    for _ in range(60):
        if np.random.rand() < 0.5:
            # dense random
            cand = np.random.rand(n)
        else:
            # sparse random
            k = np.random.randint(50, 200)
            cand = np.zeros(n)
            idx = np.random.choice(n, k, replace=False)
            cand[idx] = np.random.rand(k) + 0.1
        cand = project_to_sum(cand, target_sum)
        score = compute_score(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"  Random init improved → {best_score:.6f}")
        if time.time() >= deadline:
            return best_seq.tolist()

    # --------------------------------------------------------------
    # Phase 2 – smooth‑max / hard‑max gradient descent with momentum
    # --------------------------------------------------------------
    a = best_seq.copy()
    v = np.zeros_like(a)                     # momentum buffer

    lr = 0.03 * (target_sum / n)              # ≈ 3 % of the average entry
    tau = max(0.2 * a.max(), 0.05)            # start with a moderate temperature
    momentum = 0.9

    iteration = 0
    stagnation = 0

    print(f"[{time.time():.2f}] Entering gradient descent (lr={lr:.5e}, tau={tau:.5e})")
    while time.time() < deadline:
        if tau > 1e-4:
            a, v = smooth_max_step(a, target_sum, lr, tau, momentum, v)
        else:
            a, v = hard_max_step(a, target_sum, lr, momentum, v)

        score = compute_score(a)
        iteration += 1

        if score + 1e-12 < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"Iter {iteration:06d} – improvement: {best_score:.6f}")
            stagnation = 0
        else:
            stagnation += 1

        # schedule adaptations
        if iteration % 300 == 0:
            lr *= 0.96
            tau = max(tau * 0.96, 1e-6)

        # occasional random mass transfer
        if np.random.rand() < 0.025:
            a = random_pair_transfer(a, target_sum, scale=0.009)

        # reset on long stagnation
        if stagnation >= 1500:
            a = jitter(a, target_sum, sigma=5e-4)
            v[:] = 0.0
            stagnation = 0
            print(f"Iter {iteration:06d} – jitter reset")

    # --------------------------------------------------------------
    # Phase 3 – polishing with simulated annealing
    # --------------------------------------------------------------
    print(f"[{time.time():.2f}] Descent finished; best score {best_score:.6f}")

    a = best_seq.copy()
    temperature = 0.05 * best_score  # initial temperature for Metropolis rule

    while time.time() < deadline - 0.2:
        # propose a move: a pair transfer (slightly larger scale)
        cand = random_pair_transfer(a.copy(), target_sum, scale=0.018)

        # occasional extra jitter
        if np.random.rand() < 0.4:
            cand = jitter(cand, target_sum, sigma=2e-4)

        cand_score = compute_score(cand)
        delta = cand_score - best_score

        if delta < 0 or np.random.rand() < np.exp(-delta / max(temperature, 1e-12)):
            a = cand
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"Polish – improvement to {best_score:.6f}")

        # geometric cooling
        temperature *= 0.9997

    print(f"[{time.time():.2f}] Search completed; final best {best_score:.6f}")
    return best_seq.tolist()
