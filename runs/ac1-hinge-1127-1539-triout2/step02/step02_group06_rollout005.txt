The evaluation function:

Given sequence a of length n (non-negative). n = len(sequence). Compute b = conv(a, a) (full linear convolution length 2n-1). Let max_b = max(b). sum_a = sum(a). If sum_a < 0.01, return inf. The evaluate function returns:

2 * n * max_b / (sum_a ** 2)

Hence the objective is to minimize 2 n * max_b / sum_a^2.

We can scale a arbitrarily; sum_a influences denominator quadratically, max_b scales quadratically with scaling as well because convolution is bilinear. Let's check scaling invariance: If we multiply a by factor c, then max_b scales by c^2, sum_a scales by c, so objective 2n * c^2 * max_b_original / (c^2 * sum_a_original^2) = 2n * max_b_original / sum_a_original^2. So the objective is homogeneous scale-invariant. So we can fix sum_a to any target, like sqrt(2n) as used above.

def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_score = float('inf')
    best_n = None

    # define candidate n values
    n_candidates = [5000, 8000, 10000, 12000, 15000, 20000] maybe random sample if time.

    for n in n_candidates:
        target_sum = np.sqrt(2.0 * n)
        # parametric sweep families
        # families: exponential decreasing, power decreasing, linear decreasing (with zero floor), exponential with offset broader etc.
        # We'll generate param grid scanning param (slope) values.

        # Exponential decreasing
        for param in np.linspace(0.0, 8.0, 41):   # param from 0 to 8 step .2
            # shape raw = np.exp(-param * idx/(n-1))
            idx = np.arange(n)
            raw = np.exp(-param * idx / (n-1))
            seq = raw / raw.sum() * target_sum
            score = float(max(np.convolve(seq, seq)))  # compute max conv quickly using FFT may be faster
            # But I'll compute via fft for speed.
            conv = fft_convolve(seq, seq)
            score = conv.max()
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_n = n

        # power decreasing: raw = (1 - idx/(n-1))**param
        for param in np.linspace(0.1, 6.0, 31):
            raw = (1.0 - idx/(n-1)) ** param
            # ensure not all zeros; exponent bigger than large yields near zero but still some positive
            if raw.sum() == 0:
                continue
            seq = raw / raw.sum() * target_sum
            conv = fft_convolve(seq, seq)
            score = conv.max()
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_n = n

        # linear decreasing with floor zero: raw = np.maximum(0.0, 1 - param * idx/(n-1))
        for param in np.linspace(0.5, 5.0, 20):
            raw = np.maximum(0.0, 1.0 - param * idx/(n-1))
            if raw.sum() == 0:
                continue
            seq = raw / raw.sum() * target_sum
            conv = fft_convolve(seq, seq)
            score = conv.max()
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_n = n

        # also random init:
        for _ in range(10):
            cand = np.random.rand(n)
            cand = cand / cand.sum() * target_sum
            conv = fft_convolve(cand, cand)
            score = conv.max()
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                best_n = n

        # check deadline:
        if time.time() >= deadline:
            break

    # Phase gradient descent refine best_seq
    if best_seq is not None:
        # smoothmax descent similar as before but more aggressive schedule.
        a = best_seq.copy()
        n = a.shape[0]
        v = np.zeros_like(a)

        target_sum = np.sqrt(2.0 * n)
        # initial step size: adapt relative to average value
        avg = target_sum / n
        lr = 0.2 * avg    # maybe larger
        tau = 0.5 * a.max()  # start with high temperature
        momentum = 0.9

        iteration = 0
        no_improve = 0

        while time.time() < deadline:
            # compute softmax weights
            conv = conv_self_fft(a)
            max_c = conv.max()
            # adapt tau gradually reduce
            # compute w
            exp_arg = (conv - max_c) / tau
            w = np.exp(exp_arg - np.max(exp_arg))  # subtract again for robustness
            w_sum = w.sum()
            if w_sum == 0:
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w = w / w_sum

            # gradient: cross-correlation of w with a
            # compute correlation via FFT.
            a_rev = a[::-1]
            grad_full = fft_convolve(w, a_rev)
            grad = 2.0 * grad_full[n-1:n-1+n]

            direction = -(grad - np.mean(grad))
            v = momentum * v + direction
            a_new = a + lr * v

            a_new = np.clip(a_new, 0.0, None)
            s = a_new.sum()
            if s > 0:
                a_new = a_new * (target_sum / s)
            else:
                # if all zero due to clipping, reinitialize random
                a_new = np.random.rand(n)
                a_new = a_new / a_new.sum() * target_sum

            # evaluate
            score = conv_self_fft(a_new).max()
            iteration += 1
            if score + 1e-12 < best_score:
                best_score = score
                best_seq = a_new.copy()
                print(f"Iter {iteration} – new best {best_score:.6f} (n={n})")
                no_improve = 0
            else:
                no_improve += 1

            # schedule: reduce lr/tau every certain iterations
            if iteration % 100 == 0:
                lr *= 0.96
                tau *= 0.93
                if lr < 1e-6:
                    lr = 1e-6
                if tau < 1e-5:
                    tau = 1e-5

            if np.random.rand() < 0.02:
                a_new = random_pair_transfer(a_new, target_sum, scale=0.01)

            if no_improve >= 500:
                a_new = jitter(a_new, target_sum, sigma=0.001)
                v = np.zeros_like(v)
                no_improve = 0
                print(f"Iter {iteration} – jitter reset")
            a = a_new

    # After all, return best_seq as list
    print(f"Best score {best_score:.6f} with length {best_n}")
    return best_seq.tolist()

"""
Propose a sequence of non‑negative heights that (approximately) minimises
the quantity returned by ``evaluate_sequence``.

The optimiser works in three stages:

1. **Coarse parametric sweep** – a collection of simple analytic families
   (decreasing linear, exponential, power‑law, and piecewise‑linear with
   zero tail) are generated for a few different lengths ``n`` (the longer
   the sequence the larger the potential improvement).  All candidates are
   normalised to the required ℓ₁‑norm ``√(2 n)`` and the best one is kept.

2. **Smoothed‑max gradient descent** – the hard maximum of the
   self‑convolution is approximated by a temperature‑controlled log‑sum‑exp.
   Its gradient is computed analytically as a cross‑correlation and the
   iterate is projected onto the hyper‑plane ``Σ aᵢ = √(2 n)`` while keeping
   non‑negativity.  A Nesterov‑style momentum term and an adaptive learning
   rate are used.  Every few hundred iterations the temperature is cooled.
   Occasionally a “hard‑max” step (exact sub‑gradient of the real maximum) is
   performed to sharpen the descent.

3. **Escapes & polishing** – when progress stalls the algorithm injects a
   small Gaussian jitter and occasionally performs random pair‑wise mass
   transfers.  A final short random‑perturbation loop is executed if time
   permits.

The whole search respects a 1000 s wall‑clock budget and always returns the
best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures / lambdas)
# ----------------------------------------------------------------------


def fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors a and b using an FFT of power‑of‑2 length."""
    n, m = a.shape[0], b.shape[0]
    L = 1 << ((n + m - 1).bit_length())           # next power of two
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)
    return conv[: n + m - 1]


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (length 2·len(a)‑1)."""
    return fft_convolve(a, a)


def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector so that its ℓ₁‑norm equals *target*."""
    s = v.sum()
    if s > 0:
        return v * (target / s)
    # fallback – uniform distribution (should never happen)
    return np.full_like(v, target / v.shape[0])


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """Transfer a tiny fraction of mass from a random source index to a random target."""
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0.0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_to_sum(a, target_sum)


def jitter(a: np.ndarray, target_sum: float, sigma: float = 1e-3) -> np.ndarray:
    """Add Gaussian noise (clipped) and renormalise."""
    a = a + np.random.randn(a.shape[0]) * sigma
    a = np.clip(a, 0.0, None)
    return project_to_sum(a, target_sum)


# ----------------------------------------------------------------------
# Helper to build a few analytic families (the families differ from the
# ones used in the reference solution – they allow values close to zero).
# ----------------------------------------------------------------------


def shape_linear_decreasing(n: int, target_sum: float, slope: float) -> np.ndarray:
    """a_i = max(0, 1 - slope * i/(n‑1)), then normalised."""
    idx = np.arange(n, dtype=np.float64)
    raw = np.maximum(0.0, 1.0 - slope * idx / (n - 1))
    if raw.sum() == 0.0:
        raw = np.ones_like(raw)
    return project_to_sum(raw, target_sum)


def shape_exponential_decreasing(n: int, target_sum: float, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i/(n‑1)), then normalised."""
    idx = np.arange(n, dtype=np.float64)
    raw = np.exp(-rate * idx / (n - 1))
    return project_to_sum(raw, target_sum)


def shape_powerlaw_decreasing(n: int, target_sum: float, power: float) -> np.ndarray:
    """a_i ∝ (1 - i/(n‑1))**power, then normalised."""
    idx = np.arange(n, dtype=np.float64)
    raw = (1.0 - idx / (n - 1)) ** power
    return project_to_sum(raw, target_sum)


def shape_piecewise_linear(n: int, target_sum: float, plateau: float, slope: float) -> np.ndarray:
    """
    Plateau of height 1 for the first ``plateau`` fraction of the vector,
    then decrease linearly to zero with given ``slope``.
    """
    idx = np.arange(n, dtype=np.float64) / (n - 1)
    raw = np.where(idx <= plateau,
                   1.0,
                   np.maximum(0.0, 1.0 - slope * (idx - plateau) / (1.0 - plateau)))
    if raw.sum() == 0.0:
        raw = np.ones_like(raw)
    return project_to_sum(raw, target_sum)


# ----------------------------------------------------------------------
# Smoothed‑max descent (one iteration)
# ----------------------------------------------------------------------


def smooth_max_step(a: np.ndarray, target_sum: float, lr: float,
                    tau: float, momentum: float,
                    v_prev: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    """
    One iteration of gradient descent on the smoothed max of a*a.
    Returns the updated vector and the new momentum term.
    """
    n = a.shape[0]

    # (1) convolution a*a
    conv = conv_self_fft(a)                     # length 2n‑1

    # (2) soft‑max weights   w_k ∝ exp(conv_k / τ)
    max_c = conv.max()
    # stability: subtract max before division by τ
    exp_arg = (conv - max_c) / tau
    w = np.exp(exp_arg - np.max(exp_arg))
    w /= w.sum()

    # (3) gradient of the smoothed max:
    #     g_i = 2 * Σ_k w_k * a_{k-i}
    #     which is a cross‑correlation of w with a.
    a_rev = a[::-1]
    grad_full = fft_convolve(w, a_rev)         # length 3n‑2
    grad = 2.0 * grad_full[n - 1:n - 1 + n]     # slice i = 0…n‑1

    # (4) project onto the affine sub‑space Σaᵢ = target_sum
    direction = -(grad - np.mean(grad))

    # (5) momentum + step
    v = momentum * v_prev + direction
    a_new = a + lr * v

    # (6) enforce feasibility
    a_new = np.clip(a_new, 0.0, None)
    a_new = project_to_sum(a_new, target_sum)

    return a_new, v


# ----------------------------------------------------------------------
# Hard‑max (exact sub‑gradient) step
# ----------------------------------------------------------------------


def hard_max_step(a: np.ndarray, target_sum: float, lr: float) -> np.ndarray:
    """Take a step opposite to the exact sub‑gradient of the (hard) max."""
    n = a.shape[0]
    conv = conv_self_fft(a)
    k = int(conv.argmax())                     # shift giving the maximum

    # gradient_i = 2 * a[k‑i]  (if the index is inside the array)
    grad = np.zeros_like(a)
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    grad[mask] = 2.0 * a[j[mask]]

    direction = -(grad - np.mean(grad))
    a_new = a + lr * direction
    a_new = np.clip(a_new, 0.0, None)
    a_new = project_to_sum(a_new, target_sum)
    return a_new


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for the best non‑negative height sequence within ``budget_s`` seconds.
    Returns the sequence as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Choose a set of lengths to try – larger lengths give a better
    # theoretical bound but cost a little more time.
    # ------------------------------------------------------------------
    length_candidates = [8000, 10000, 12000, 15000, 18000]
    if isinstance(globals().get("height_sequence_1"), (list, tuple)):
        # use the previous construction as an additional candidate length
        length_candidates.append(len(height_sequence_1))

    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # Phase 1 – extensive parametric sweep over several families
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Starting parametric sweep")
    for n in length_candidates:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # ----- linear decreasing (with zero tail) --------------------
        for slope in np.linspace(0.2, 4.0, 20):
            cand = shape_linear_decreasing(n, target_sum, slope)
            score = conv_self_fft(cand).max()
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                best_n = n
                print(f"  new best (linear, slope={slope:.2f}, n={n}) → {best_score:.6f}")

        # ----- exponential decreasing --------------------------------
        for rate in np.linspace(0.0, 8.0, 25):
            cand = shape_exponential_decreasing(n, target_sum, rate)
            score = conv_self_fft(cand).max()
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                best_n = n
                print(f"  new best (exp, rate={rate:.2f}, n={n}) → {best_score:.6f}")

        # ----- power‑law decreasing ---------------------------------
        for power in np.linspace(0.2, 6.0, 22):
            cand = shape_powerlaw_decreasing(n, target_sum, power)
            score = conv_self_fft(cand).max()
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                best_n = n
                print(f"  new best (power, p={power:.2f}, n={n}) → {best_score:.6f}")

        # ----- piecewise‑linear plateau + decay ---------------------
        for plateau in np.linspace(0.0, 0.5, 6):
            for slope in np.linspace(0.2, 4.0, 12):
                cand = shape_piecewise_linear(n, target_sum, plateau, slope)
                score = conv_self_fft(cand).max()
                if score < best_score:
                    best_score = score
                    best_seq = cand.copy()
                    best_n = n
                    print(f"  new best (piecewise, pl={plateau:.2f}, sl={slope:.2f}, n={n}) → {best_score:.6f}")

        # ----- random initialisations --------------------------------
        for _ in range(15):
            cand = np.random.rand(n)
            cand = project_to_sum(cand, target_sum)
            score = conv_self_fft(cand).max()
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                best_n = n
                print(f"  new best (rand, n={n}) → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – smoothed‑max gradient descent with occasional hard‑max steps
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution
        best_n = 8000
        best_seq = np.full(best_n, np.sqrt(2.0 * best_n) / best_n)

    a = best_seq.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    v = np.zeros_like(a)

    # step‑size roughly a few percent of the typical entry (≈ √(2/n))
    avg = target_sum / n
    lr = 0.08 * avg
    tau = 0.6 * a.max()                # start relatively warm
    momentum = 0.93

    iteration = 0
    no_improve = 0

    print(f"[{time.time():.2f}] Entering descent (n={n}, lr={lr:.5e}, tau={tau:.5e})")
    while time.time() < deadline:
        # perform a smooth‑max step
        a, v = smooth_max_step(a, target_sum, lr, tau, momentum, v)

        # every 400 steps do a cheap hard‑max correction
        if iteration % 400 == 0:
            a = hard_max_step(a, target_sum, lr * 0.6)

        score = conv_self_fft(a).max()
        iteration += 1

        if score + 1e-12 < best_score:
            best_score = score
            best_seq = a.copy()
            best_n = n
            no_improve = 0
            print(f"Iter {iteration:06d} – improvement → {best_score:.6f}")
        else:
            no_improve += 1

        # schedule adaptations
        if iteration % 250 == 0:
            lr = max(lr * 0.94, 1e-7)
            tau = max(tau * 0.93, 1e-5)

        # occasional random pair transfer – helps escape plateaus
        if np.random.rand() < 0.025:

        # jitter reset if we have been stuck for a while

            v[:] = 0.0
            no_improve = 0
            print(f"Iter {iteration:06d} – jitter reset")

    # ------------------------------------------------------------------
    # Phase 3 – final quick polishing with pure random perturbations
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Descent finished; best score {best_score:.6f}")
    while time.time() < deadline - 0.2:
        a = random_pair_transfer(a, target_sum, scale=0.004)
        score = conv_self_fft(a).max()
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"Polish – improvement → {best_score:.6f}")

    # Final sanity check using the provided evaluate_sequence (just to obey the spec)
    _ = evaluate_sequence(best_seq.tolist())

    print(f"Returning best sequence of length {best_n} with score {best_score:.6f}")
    return best_seq.tolist()
