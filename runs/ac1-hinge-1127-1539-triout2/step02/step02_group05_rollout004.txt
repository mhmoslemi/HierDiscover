"""
Propose a non‑negative step function (sequence) with a low value of

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The objective equals the maximum entry of the auto‑convolution when the
sequence is normalized to have L1‑norm √(2 n).  The algorithm works in
two stages:

1. **Coarse optimisation** on a modest size (≈ 1 k) using a projected
   Adam optimiser with a smooth‑max (soft‑max) surrogate.  This stage
   converges quickly and discovers a good global shape.

2. **Upscaling & fine‑tuning** – the coarse shape is interpolated to the
   final length (the length of the supplied ``height_sequence_1`` or a
   default of ≈ 5600).  A second Adam optimisation is performed with a
   more aggressive learning‑rate schedule and a larger soft‑max
   temperature.  Finally a short vanilla gradient descent with a very
   high temperature polishes the solution.

All updates are projected onto the simplex
    { x ≥ 0 , Σ x = √(2 n) }
so the normalisation never changes.  The code respects the 1000 s
budget, prints progress, and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _compute_max_conv(a: np.ndarray) -> float:
    """
    Compute max entry of the auto‑convolution a*a.
    Input is assumed non‑negative.
    """
    n = a.shape[0]
    nfft = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
    return float(conv.max())


def project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of *v* onto the simplex
        { x : x_i ≥ 0 , Σ_i x_i = target_sum }.
    """
    # Duchi et al. (ICML 2008) – O(n log n) sort based.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(len(v)) + 1  # 1‑based
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax_weights(conv: np.ndarray, gamma: float) -> np.ndarray:
    """Soft‑max over the convolution entries with temperature γ."""
    max_c = conv.max()
    w = np.exp(gamma * (conv - max_c))
    s = w.sum()
    if s == 0.0:
        w[:] = 1.0 / w.size
    else:
        w /= s
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2 n‑1)."""
    n = a.shape[0]
    nfft = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
    return conv


def _gradient_wrt_a(a: np.ndarray, gamma: float,
                    nfft_corr: int, nfft_conv: int) -> tuple[np.ndarray, float]:
    """
    Return (gradient, max_conv) for the soft‑max surrogate.
    Gradient = 2 * (w ⊛ a_rev) where w are the soft‑max weights.
    """
    n = a.shape[0]

    # --- convolution a*a -------------------------------------------------
    af = np.fft.rfft(a, n=nfft_conv)
    conv = np.fft.irfft(af * af, n=nfft_conv)[: 2 * n - 1]
    max_conv = conv.max()

    # --- soft‑max weights -------------------------------------------------
    w = _softmax_weights(conv, gamma)

    # --- correlation w ⊛ a_rev -------------------------------------------
    wf = np.fft.rfft(w, n=nfft_corr)
    af_rev = np.fft.rfft(a[::-1], n=nfft_corr)
    corr_full = np.fft.irfft(wf * af_rev, n=nfft_corr)
    # extract the central n‑length slice
    grad = 2.0 * corr_full[n - 1 : n - 1 + n]
    return grad, max_conv


def _interpolate_sequence(seq_small: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a 1‑D sequence to a new length.
    Keeps the shape but rescales the total mass later.
    """
    old_n = seq_small.shape[0]
    if old_n == new_n:
        return seq_small.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, seq_small)

# Optimisation routine (used for both coarse and fine phases)
# ----------------------------------------------------------------------


def _optimise(
    a0: np.ndarray,
    time_limit: float,
    deadline: float,
    phase_name: str = "phase",
    fine: bool = False,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation of the soft‑max surrogate.
    Parameters

    a0 : np.ndarray
        Starting non‑negative vector (already normalised to the simplex).

    deadline : float
        Global deadline (seconds since epoch) – a safety fallback.
    phase_name : str
        Identifier printed in the log.

        If True, a very small constant‑step gradient descent is performed
        after the main Adam loop (polishing).
    Returns
    -------
    best_a : np.ndarray
        Best vector found during the phase.
    best_score : float

    """
    n = a0.shape[0]
    target_sum = np.sqrt(2.0 * n)

    # FFT sizes – pre‑computed once for speed.
    nfft_conv = _next_pow2(2 * n - 1)
    nfft_corr = _next_pow2(3 * n - 2)

    # Adam state.
    m = np.zeros_like(a0)
    v = np.zeros_like(a0)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t_iter = 0

    lr = 0.25
    lr_min = 0.001
    lr_decay = 0.9995
    gamma = 1.0
    gamma_max = 300.0
    gamma_inc = 1.003

    a = a0.copy()
    grad, max_conv = _gradient_wrt_a(a, gamma, nfft_corr, nfft_conv)
    best_score = max_conv
    best_a = a.copy()
    no_improve = 0

    # Main Adam loop
    # ------------------------------------------------------------------
    while time.time() < time_limit and time.time() < deadline:
        t_iter += 1

        # Gradient w.r.t. a for current gamma.
        grad, max_conv = _gradient_wrt_a(a, gamma, nfft_corr, nfft_conv)

        # Record progress.
        if max_conv < best_score:
            best_score = max_conv
            best_a = a.copy()
            no_improve = 0
            elapsed = time.time() - (deadline - (deadline - time_limit))
            print(

                f" (elapsed {elapsed:.1f}s)"
            )
        else:
            no_improve += 1

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t_iter)

        a = a_new

        # Schedule updates.
        gamma = min(gamma * gamma_inc, gamma_max)
        lr = max(lr * lr_decay, lr_min)

        # If we have been stuck for a long while, inject a tiny random
        # perturbation around the current best.
        if no_improve >= 5000:
            a = best_a + 5e-4 * np.random.randn(n)
            a = project_onto_simplex(a, target_sum)
            no_improve = 0

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if fine:

        fine_gamma = gamma_max
        fine_lr = 5e-4

            a = a - fine_lr * grad
            a = project_onto_simplex(a, target_sum)

            if max_conv < best_score:
                best_score = max_conv
                best_a = a.copy()
                elapsed = time.time() - (deadline - (deadline - time_limit))
                print(
                    f"{phase_name:>6s} fine‑tune: new best {best_score:.6f}"
                    f" (elapsed {elapsed:.1f}s)"

    )

    # ------------------------------------------------------------------
    # Phase 2 – upscale to final size and continue optimisation
    # ------------------------------------------------------------------
    # Interpolate the best coarse solution to the final dimension.
    a_fine = _interpolate_sequence(a_coarse_best, final_n)
    a_fine = np.maximum(a_fine, 0.0)
    a_fine = a_fine * np.sqrt(2.0 * final_n) / a_fine.sum()

    # Use the remaining time (≈ 70 % of the budget) for the fine phase.
    phase2_deadline = global_deadline
    a_final, best_score_final = _optimise(
        a_fine,
        time_limit=phase2_deadline,
        deadline=global_deadline,
        phase_name="fine",
        fine=True,
    )

    # ------------------------------------------------------------------
    # Return the best sequence discovered.
    # ------------------------------------------------------------------
    print(
        f"Finished. Best max‑conv = {best_score_final:.6f} "
        f"(time {time.time() - start:.1f}s, length {final_n})"
    )
    return a_final.tolist()
