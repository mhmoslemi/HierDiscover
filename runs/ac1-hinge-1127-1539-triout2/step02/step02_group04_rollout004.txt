    Every candidate is normalised to unit ℓ¹‑norm (the objective is
    scale‑invariant) and evaluated.  The best sequence found is kept as a
    starting point for the second phase.

Phase 2 – **projected Adam‑style gradient descent**  
    The objective depends only on `max(conv(x,x))`.  The sub‑gradient of the
    max‑convolution is obtained from all active (maximising) indices.
    An Adam optimiser (first‑ and second‑moment estimates) is applied to
    this sub‑gradient; after each update the iterate is projected onto the
    probability simplex (non‑negativity and unit sum).  Random jitter is
    added occasionally to escape plateaus and, every few thousand
    iterations, a batch of tiny entries is appended so that the length can
    grow (up to a preset maximum).  If an iterate improves the best score
    it is stored.  The loop stops early as soon as the target bound
    `1.503` is reached or the time budget expires.

All helper functions are top‑level; no closures or filesystem access are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _generate_powerlaw(p: float, sigma: float, n: int) -> np.ndarray:
    """Power‑law: a[i] = (i + sigma)⁻ᵖ for i = 0,…,n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = (i + sigma) ** (-p)
    return a


def _generate_exponential(lam: float, n: int) -> np.ndarray:
    """Exponential decay: a[i] = exp(‑lam·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a


def _generate_mixed(p: float, sigma: float, lam: float, n: int) -> np.ndarray:
    """Mixed shape: (i+σ)⁻ᵖ·exp(‑lam·i)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + sigma) ** (-p) * np.exp(-lam * i)
    return a


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    { x ≥ 0, Σ x = 1 }.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection works only on 1‑D arrays")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient(x: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of g(x) = max_k (conv(x,x))[k] with respect to x.
    The active set (indices where the maximum is attained) is used.
    """
    n = x.shape[0]
    conv = np.convolve(x, x)                # length 2n‑1
    m = conv.max()
    # indices where convex value is within machine tolerance of the max
    active = np.where(conv >= m - 1e-14)[0]
    g = np.zeros_like(x)
    for k in active:
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        i = np.arange(i_min, i_max + 1, dtype=int)
        j = k - i
        # derivative of conv[k] w.r.t x[i] is 2·x[j] (the factor 2 will be absorbed
        # by the step‑size, so we add just x[j] here).
        g[i] += x[j]
    return g


def _evaluate(seq: np.ndarray) -> float:
    """Thin wrapper around the provided evaluate_sequence (list‑oriented)."""
    # The external function is assumed to be already imported.
    try:
        return evaluate_sequence(seq.tolist())
    except Exception:
        return float('inf')


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list:
    """
    Return the best step sequence discovered within the time budget.
    The routine follows the two‑phase strategy described in the module docstring.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0    # small safety margin

    best_score = float('inf')
    best_seq = None

    # ------------------------------------------------------------------
    # 0.  Use a globally supplied construction if it exists.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            sc = _evaluate(np.array(height_sequence_1, dtype=float))
            if np.isfinite(sc):
                best_score = float(sc)
                best_seq = list(height_sequence_1)
                print(f'Initial global sequence: score = {best_score:.6f}')
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Broad parametric sweep over several lengths and families.
    # ------------------------------------------------------------------
    lengths = [1200, 2000, 3000, 4000, 5000, 6000, 8000]
    print('=== Phase 1 – parametric sweep ===')
    # ----- power‑law family -----
    p_vals = np.linspace(0.10, 0.90, 17)           # finer grid
    sigma_vals = np.logspace(-7, 2, 12)           # from 1e‑7 to 1e2
    for n in lengths:
        for p in p_vals:
            for sigma in sigma_vals:
                if time.time() >= deadline:
                    break
                seq = _generate_powerlaw(p, sigma, n)
                if seq.sum() == 0:
                    continue
                seq = seq / seq.sum()
                score = _evaluate(seq)
                if np.isfinite(score) and score < best_score:
                    best_score = float(score)
                    best_seq = seq.tolist()
                    print(f'new best PL  n={n:4d} p={p:.3f} σ={sigma:.2e} score={best_score:.6f}')

    # ----- exponential family -----
    lam_vals = np.logspace(-7, 0, 15)          # from 1e‑7 to 1
    for n in lengths:
        for lam in lam_vals:
            if time.time() >= deadline:
                break
            seq = _generate_exponential(lam, n)
            if seq.sum() == 0:
                continue
            seq = seq / seq.sum()
            score = _evaluate(seq)
            if np.isfinite(score) and score < best_score:
                best_score = float(score)
                best_seq = seq.tolist()
                print(f'new best EXP n={n:4d} λ={lam:.2e} score={best_score:.6f}')

    # ----- mixed power‑law / exponential family -----
    mixed_params = [
        (0.30, 1e-3, 1e-3),
        (0.45, 1e-2, 5e-3),
        (0.55, 5e-2, 1e-2),
        (0.70, 1e-1, 1e-2),
    ]
    for n in lengths:
        for p, sigma, lam in mixed_params:
            if time.time() >= deadline:
                break
            seq = _generate_mixed(p, sigma, lam, n)
            if seq.sum() == 0:
                continue
            seq = seq / seq.sum()
            score = _evaluate(seq)
            if np.isfinite(score) and score < best_score:
                best_score = float(score)
                best_seq = seq.tolist()
                print(f'new best MIX n={n:4d} p={p:.2f} σ={sigma:.2e} λ={lam:.2e} score={best_score:.6f}')

    # Random‑draw families (to catch unexpected shapes)
    rnd_draws = 200
    for _ in range(rnd_draws):
        if time.time() >= deadline:
            break
        n = np.random.choice(lengths)
        choice = np.random.randint(3)
        if choice == 0:      # power‑law
            p = 10 ** np.random.uniform(-1, 0)      # 0.1 – 1.0
            sigma = 10 ** np.random.uniform(-7, 2)
            seq = _generate_powerlaw(p, sigma, n)
        elif choice == 1:    # exponential
            lam = 10 ** np.random.uniform(-7, 0)
            seq = _generate_exponential(lam, n)
        else:                # mixed
            p = 10 ** np.random.uniform(-1, 0)
            sigma = 10 ** np.random.uniform(-7, 2)
            lam = 10 ** np.random.uniform(-7, 0)
            seq = _generate_mixed(p, sigma, lam, n)

        if seq.sum() == 0:
            continue
        seq = seq / seq.sum()
        score = _evaluate(seq)
        if np.isfinite(score) and score < best_score:
            best_score = float(score)
            best_seq = seq.tolist()
            print(f'new best RND n={n:4d} score={best_score:.6f}')

    # ------------------------------------------------------------------
    # 2.  Adam‑style projected sub‑gradient refinement.
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback: uniform distribution of moderate size.
        n0 = 2000
        best_seq = [1.0] * n0
        best_score = _evaluate(np.ones(n0) / n0)
        print('fallback to uniform; initial score =', best_score)

    x = np.array(best_seq, dtype=np.float64)
    x = x / x.sum()                     # ensure we are on the simplex

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.5                           # base learning rate
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    iteration = 0
    grow_interval = 3000                # when to append tiny entries
    max_total_len = 12000               # absolute ceiling (still fast enough)

    print('=== Phase 2 – Adam‑style refinement ===')
    while time.time() < deadline:
        iteration += 1

        # ---- gradient / sub‑gradient ----
        g = _subgradient(x)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step_size = lr0 / np.sqrt(iteration)          # simple decay
        step = step_size * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- projected gradient step ----
        y = x - step
        x = _simplex_projection(y)

        # ---- occasional random jitter (helps escape flat regions) ----
        if np.random.rand() < 0.015:
            idx = np.random.randint(x.size)
            delta = (np.random.rand() - 0.5) * 0.02 * x[idx]  # up to ±1 %
            x[idx] = max(0.0, x[idx] + delta)
            x = x / x.sum()

        # ---- length growth (adds almost‑zero mass) ----
        if iteration % grow_interval == 0 and x.shape[0] < max_total_len:
            extra = np.full(200, 1e-12, dtype=np.float64)
            x = np.concatenate([x, extra])
            x = x / x.sum()
            # extend Adam's moments with zeros for the new coordinates
            m = np.concatenate([m, np.zeros_like(extra)])
            v = np.concatenate([v, np.zeros_like(extra)])
            print(f'growth: length increased to {x.shape[0]} at iter {iteration}')

        # ---- evaluation ----
        cur_score = _evaluate(x)
        if np.isfinite(cur_score) and cur_score < best_score:
            best_score = float(cur_score)
            best_seq = x.tolist()
            print(f'iteration {iteration:6d} → new best score = {best_score:.6f} (len={len(x)})')
            if best_score < 1.503:          # target reached
                print('Target bound achieved – exiting early.')
                break

        # ---- occasional restart to avoid stagnation ----
        if iteration % 7000 == 0:
            n_cur = x.shape[0]
            rnd = np.random.rand(n_cur)
            x = 0.5 * x + 0.5 * rnd
            x = x / x.sum()
            m = np.zeros_like(x)
            v = np.zeros_like(x)
            print(f'restart at iteration {iteration}')

    print('final best score:', best_score, 'length:', len(best_seq))
    return best_seq
