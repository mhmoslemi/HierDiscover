def local_search(a, best_score):
    while time.time() < deadline:
        i, j = rng.integers(n, size=2)
        if i == j: continue
        max_move = min(a[i], 0.01 * a[i]) # small fraction
        if max_move <= 0: continue
        eps = rng.uniform(0, max_move)
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        # no need to project as sum unchanged and similar
        # ensure nonnegatives (a_new[i]>=0)
        score_new = _compute_score(a_new)
        if score_new < best_score:
            best_score = score_new
            a = a_new
            best_a = a_new.copy()

    # Setup
    import time, numpy as np

    start = time.time()
    deadline = start + budget_s - 2.0

    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # Determine n and target sum
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)
        n = init_seq.shape[0]
    except Exception:
        n = 5600  # fallback
        init_seq = None

    target_sum = np.sqrt(2.0 * n)

    # Helper functions defined earlier in top-level: _next_pow2, _compute_score, project_onto_simplex

    # Generate multiple candidates and pick best
    best_a = None
    best_score = np.inf

    # include provided
    if init_seq is not None:
        a0 = np.maximum(init_seq, 0.)
        s = a0.sum()
        if s > 0:
            a0 = a0 * target_sum / s
        else:
            a0 = np.full(n, target_sum/n)
        score = _compute_score(a0)
        if score < best_score:
            best_score = score; best_a = a0.copy()

    # random shapes
    for _ in range(120):  # many random initializations
        if time.time() >= deadline:
            break
        shape_type = rng.choice(['power','exp','piecewise'])
        if shape_type == 'power':
            p = rng.uniform(0.2, 1.0)
            offset = rng.uniform(0.5, 5.0)
            a = 1.0 / np.power(np.arange(n, dtype=np.float64) + offset, p)
        elif shape_type == 'exp':
            lam = rng.uniform(0.0005, 0.02)

            a = np.concatenate([np.full(flat_len, flat_val), tail])

        a = a_start.copy()
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        t = 0
        lr = 0.3
        lr_min = 0.001
        lr_decay = 0.9998
        gamma = gamma_start
        gamma_max = 500.0
        gamma_inc = 1.005
        local_best_a = a.copy()
        local_best_score = _compute_score(a)
        for it in range(iter_budget):
            nonlocal best_a, best_score
            if time.time() >= deadline:

            t += 1
            # conv
            af = np.fft.rfft(a, n=nfft_conv)
            conv = np.fft.irfft(af * af, n=nfft_conv)[:2*n-1]

            corr = np.fft.irfft(wf * a_rev, n=nfft_corr)
            grad = 2.0 * corr[n-1:n-1+n]
            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad

    remaining_iters = 60000

    # set high gamma for a few thousand iterations
    if time.time() < deadline:

        v = np.zeros_like(a)
        t = 0
        lr = 0.01
        gamma = gamma_max
        for it in range(5000):

            t += 1
            af = np.fft.rfft(a, n=nfft_conv)
            conv = np.fft.irfft(af * af, n=nfft_conv)[:2*n-1]
            max_conv = conv.max()

            grad = 2.0 * corr[n-1:n-1+n]

            a = a - lr * grad
            a = project_onto_simplex(a, target_sum)
            # evaluation

            if score < best_score:
                best_score = score
                best_a = a.copy()
                elapsed = time.time() - start
                print(f"Fine {it} new best {best_score:.6f} (elapsed {elapsed:.1f}s)")

    # Final random local search

        a = best_a.copy()

                print(f"Local {it} new best {best_score:.6f} (elapsed {elapsed:.1f}s)")

    # Return best list
    return best_a.tolist()

"""
Step‑function optimisation for the objective

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works in four phases:

1. **Exploratory initialisation** – a batch of simple families
   (power‑law, exponential, piecewise‑constant) is sampled, each candidate
   is normalised to the required ℓ₁‐norm √(2 n), and the best one is kept.
   If the global variable ``height_sequence_1`` is present it is also
   considered.

2. **Projected Adam optimisation** – the max‑convolution is smoothed with a
   soft‑max (temperature 𝛾).  The gradient of the smoothed objective is
   obtained by an FFT‑based correlation and an Adam update is performed.
   After every step the vector is projected onto the simplex
   {x ≥ 0, Σ x = √(2 n)}.

3. **Fine‑tuning with a true sub‑gradient** – 𝛾 is set to a very large value
   and the gradient is taken as the average of the gradients of all
   convolution entries that are within a tiny tolerance of the maximum.
   A small constant step size is used.

4. **Local random search** – a cheap hill‑climbing phase moves a tiny
   amount of mass from one index to another and accepts the move only if
   the exact evaluation function improves.

All heavy operations use FFTs for O(n log n) speed and the whole routine
stops before the supplied time budget (default 1000 s).  The function
returns the best sequence found as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _compute_score(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for non‑negative vectors.
    The ℓ₁‑norm is assumed to be non‑zero.
    """
    n = a.shape[0]
    nfft = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))


def project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of *v* onto the simplex
        { x : x_i ≥ 0 , Σ x_i = target_sum }.
    Implements the algorithm of Duchi et al. (ICML 2008).
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(len(v)) + 1  # 1‑based
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def adam_optimize(
    a0: np.ndarray,
    target_sum: float,
    nfft_conv: int,
    nfft_corr: int,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 10_000_000,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation on the soft‑max approximation of the max‑
    convolution.  Returns the best vector found and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    lr = 0.30          # initial learning rate
    lr_min = 0.001
    lr_decay = 0.9998

    gamma = 1.0
    gamma_max = 500.0
    gamma_inc = 1.005

    best_a = a.copy()
    best_score = _compute_score(a)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        t += 1

        # ----- convolution a*a ------------------------------------------------
        af = np.fft.rfft(a, n=nfft_conv)
        conv = np.fft.irfft(af * af, n=nfft_conv)[: 2 * n - 1]
        max_conv = conv.max()

        # ----- soft‑max weights -------------------------------------------------
        w = np.exp(gamma * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # ----- gradient w.r.t. a (correlation) ---------------------------------
        wf = np.fft.rfft(w, n=nfft_corr)
        a_rev = np.fft.rfft(a[::-1], n=nfft_corr)
        corr_full = np.fft.irfft(wf * a_rev, n=nfft_corr)
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]

        # ----- Adam update -----------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto simplex -----------------------------------------
        a = project_onto_simplex(a, target_sum)

        # ----- occasional jitter (helps escape plateaus) -----------------------
        if it % 103 == 0:
            jitter = 1e-5 * rng.standard_normal(n)
            a = project_onto_simplex(a + jitter, target_sum)

        # ----- exact evaluation -------------------------------------------------
        score = _compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # ----- schedule ---------------------------------------------------------
        gamma = min(gamma * gamma_inc, gamma_max)
        lr = max(lr * lr_decay, lr_min)

    return best_a, best_score


def fine_tune_subgradient(
    a0: np.ndarray,
    target_sum: float,
    nfft_conv: int,
    nfft_corr: int,
    deadline: float,
    max_iters: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Fine‑tuning using a true sub‑gradient of the max‑convolution.
    The gradient is the average of the gradients at all entries that are
    within a tiny relative tolerance of the maximum.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _compute_score(a)

    lr = 0.01                     # tiny constant step size
    # Use a very large γ to make the soft‑max close to a true max;
    # however we explicitly build the sub‑gradient, so γ is unused.
    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # convolution a*a
        af = np.fft.rfft(a, n=nfft_conv)
        conv = np.fft.irfft(af * af, n=nfft_conv)[: 2 * n - 1]
        max_conv = conv.max()

        # indices that attain (or are extremely close to) the maximum
        tol = max_conv * 1e-9
        max_idx = np.where(conv >= max_conv - tol)[0]
        if max_idx.size == 0:
            max_idx = np.array([conv.argmax()], dtype=int)

        w = np.zeros_like(conv)
        w[max_idx] = 1.0 / max_idx.size

        # gradient = correlation of w with reversed a
        wf = np.fft.rfft(w, n=nfft_corr)
        a_rev = np.fft.rfft(a[::-1], n=nfft_corr)
        corr_full = np.fft.irfft(wf * a_rev, n=nfft_corr)
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]

        a = a - lr * grad
        a = project_onto_simplex(a, target_sum)

        score = _compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a, best_score


def local_random_search(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a small amount of mass from one
    index to another and keep the change if the exact objective improves.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _compute_score(a)

    n = a.shape[0]

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        # propose moving a fraction of a[i]
        max_move = min(a[i], a[i] * 0.02)
        if max_move <= 1e-12:
            continue
        eps = rng.uniform(0.0, max_move)

        a_candidate = a.copy()
        a_candidate[i] -= eps
        a_candidate[j] += eps

        score = _compute_score(a_candidate)
        if score < best_score:
            best_score = score
            best_a = a_candidate.copy()
            a = a_candidate  # continue from the improved point

    return best_a, best_score

        )
        elapsed = time.time() - start_time
        print(f"After local search best {best_score:.6f} (elapsed {elapsed:.1f}s)")

    # ------------------------------------------------------------------
    # Return the best sequence as a plain list
    # ------------------------------------------------------------------
    return best_a.tolist()
