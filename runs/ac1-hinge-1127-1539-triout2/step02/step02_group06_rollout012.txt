def downsample(seq, new_n):
    old_n = len(seq)
    # Map each new index i to range [i*old_n/new_n, (i+1)*old_n/new_n)
    # average over that slice.
    seq = np.array(seq)
    # Use np.linspace to get boundaries.
    boundaries = np.linspace(0, old_n, new_n+1, dtype=int)
    new_seq = np.empty(new_n, dtype=float)
    for i in range(new_n):
        start = boundaries[i]
        end = boundaries[i+1]
        if end > start:
            new_seq[i] = seq[start:end].mean()
        else:
            new_seq[i] = 0.0
    return new_seq

"""
search for a step‑function (non‑negative vector) that minimises
the maximal entry of its self‑convolution (the value returned by
``evaluate_sequence``).

Approach
--------
1. Fix a moderate length ``n`` (≈1500–2500).  Larger n give a finer
   discretisation but the linear programme (LP) that drives the search
   becomes cubic in n, so a compromise is used.

2. Initialise the vector from the best known construction
   (`height_sequence_1`) – down‑sampled to length ``n`` – and normalise it
   to the required ℓ₁‑norm ``sqrt(2·n)``.

3. **Alternating linear programming.**
   For the current vector ``f`` we solve the LP

        minimise   M
        subject to  Σ_i f_i·g_{k−i} – M ≤ 0   for all convolution indices k
                    Σ_j g_j = S
                    g_j ≥ 0 ,   M ≥ 0

   where ``S = sqrt(2·n)``.  This finds the distribution ``g`` of the
   same total mass that makes the *cross‑convolution* with ``f`` as flat
   as possible, i.e. with the smallest possible uniform bound ``M``.

   The LP has ``n`` variables for ``g`` plus one extra variable ``M``.
   The constraint matrix is a Toeplitz pattern; we pre‑compute the
   sparsity structure once and only update the numerical data each
   iteration.

   The new candidate is obtained by a convex combination

        f_new = (1‑α)·f + α·g               (α≈0.5)

   which preserves the ℓ₁‑norm.  If the self‑convolution of ``f_new`` is
   lower than the best value so far we keep it; otherwise we reduce
   ``α`` (down to a small floor) to avoid overshooting.

4. The LP is solved with ``scipy.optimize.linprog`` (HiGHS interior‑point
   method).  Each iteration costs O(n²) time to build the sparse matrix
   and a few seconds to solve the LP; we repeat as long as time permits.

5. **Final polishing** – if any time remains we run a short
   smooth‑max gradient descent (the same routine used in the reference
   implementation) and occasional random pair‑wise mass transfers to
   escape tiny plateaus.

The whole routine respects the 1000 s wall‑clock limit and always returns
the best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (LP solving)
# ----------------------------------------------------------------------
try:
    import scipy.optimize
    import scipy.sparse
    _SCIPY_OK = True
except Exception:               # pragma: no cover
    _SCIPY_OK = False

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector so that its ℓ₁‑norm equals *target*."""
    s = v.sum()
    if s > 0.0:
        return v * (target / s)
    # fallback – uniform distribution (should not happen)
    return np.full_like(v, target / v.shape[0])


def downsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Deterministically down‑sample a sequence to *new_n* entries by averaging."""
    old_n = seq.shape[0]
    if new_n >= old_n:
        return np.copy(seq)
    # integer borders for the averaging windows
    borders = np.linspace(0, old_n, new_n + 1, dtype=int)
    out = np.empty(new_n, dtype=float)
    for i in range(new_n):
        lo, hi = borders[i], borders[i + 1]
        if hi > lo:
            out[i] = seq[lo:hi].mean()
        else:                      # very unlikely, but keep safe
            out[i] = seq[lo]
    return out


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """Transfer a tiny fraction of mass from a random source index to a random target."""
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0.0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_to_sum(a, target_sum)


def jitter(a: np.ndarray, target_sum: float, sigma: float = 1e-3) -> np.ndarray:
    """Add small Gaussian noise and re‑normalise."""
    a = a + np.random.randn(a.shape[0]) * sigma
    a = np.clip(a, 0.0, None)
    return project_to_sum(a, target_sum)


# ----------------------------------------------------------------------
# FFT‑based convolution helpers (used for the final polishing stage)
# ----------------------------------------------------------------------


def fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using an FFT of power‑of‑2 length."""
    n, m = a.shape[0], b.shape[0]
    L = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)
    return conv[: n + m - 1]


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full convolution of ``a`` with itself."""
    return fft_convolve(a, a)


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (optional polishing)
# ----------------------------------------------------------------------


def smooth_max_step(
    a: np.ndarray,
    target_sum: float,
    lr: float,
    tau: float,
    momentum: float,
    v_prev: np.ndarray,
):
    """One iteration of the soft‑max gradient descent used in the reference code."""
    n = a.shape[0]
    conv = conv_self_fft(a)                     # length 2n‑1

    max_c = conv.max()
    exp_arg = (conv - max_c) / tau
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    a_rev = a[::-1]
    grad_full = fft_convolve(w, a_rev)          # length 3n‑2
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]

    direction = -(grad - np.mean(grad))
    v = momentum * v_prev + direction
    a_new = a + lr * v
    a_new = np.clip(a_new, 0.0, None)
    a_new = project_to_sum(a_new, target_sum)
    return a_new, v


# ----------------------------------------------------------------------
# LP construction (once per problem size)
# ----------------------------------------------------------------------


def _build_lp_structure(n: int):
    """
    Pre‑compute the sparsity pattern of the Toeplitz convolution matrix.
    Returns three 1‑D integer arrays (row, col, i_index) such that
    the value in row r and column c of the constraint matrix is f[i_index].
    """
    rows = []
    cols = []
    i_idx = []
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            i = k - j
            rows.append(k)
            cols.append(j)
            i_idx.append(i)
    rows = np.asarray(rows, dtype=np.int32)
    cols = np.asarray(cols, dtype=np.int32)
    i_idx = np.asarray(i_idx, dtype=np.int32)
    return rows, cols, i_idx


def _solve_min_M_lp(f_vec: np.ndarray, target_sum: float,
                    rows: np.ndarray, cols: np.ndarray, i_idx: np.ndarray):
    """
    Solve the linear programme
        minimise   M
        s.t.       Σ_i f_i·g_{k‑i} – M ≤ 0      (for all k)
                   Σ_j g_j = target_sum
                   g_j ≥ 0 ,  M ≥ 0

    Returns (M_opt, g_opt) or (None, None) if the solver fails.
    """
    n = f_vec.shape[0]
    # data for the Toeplitz part (depends on the current f)
    toeplitz_data = f_vec[i_idx]                       # length = nnz
    # append the column for the variable M (coefficient –1 in every row)
    data = np.concatenate([toeplitz_data, -np.ones(2 * n - 1, dtype=float)])
    row_ind = np.concatenate([rows, np.arange(2 * n - 1, dtype=np.int32)])
    col_ind = np.concatenate([cols, np.full(2 * n - 1, n, dtype=np.int32)])

    A_ub = scipy.sparse.csr_matrix((data, (row_ind, col_ind)),
                                   shape=(2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1, dtype=float)

    # equality: sum(g) = target_sum (no M in this equation)
    A_eq = np.zeros((1, n + 1), dtype=float)
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum], dtype=float)

    # objective: minimise M → c = (0,…,0,1)
    c = np.zeros(n + 1, dtype=float)
    c[-1] = 1.0

    bounds = [(0.0, None)] * n + [(0.0, None)]          # g_j ≥0 , M ≥0

    res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub,
                                 A_eq=A_eq, b_eq=b_eq,
                                 bounds=bounds,
                                 method='highs',
                                 options={'presolve': True})
    if not res.success:
        return None, None
    opt = res.x
    M_opt = opt[-1]
    g_opt = opt[:n]
    return M_opt, g_opt


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises the
    value returned by ``evaluate_sequence`` within the given wall‑clock budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5               # safety margin

    # --------------------------------------------------------------
    # Choose problem size and initial vector
    # --------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        old_n = len(base)
        # moderate size for the LP – large enough to capture shape, small enough for speed
        n = min(2500, max(1200, old_n // 4))
        a0 = downsample(np.asarray(base, dtype=float), n)
    else:
        # fall back to a simple exponential shape if no reference is available
        n = 2000
        x = np.arange(n, dtype=float)
        a0 = np.exp(-0.001 * x)                           # decaying shape
    target_sum = np.sqrt(2.0 * n)
    a = project_to_sum(a0.astype(float), target_sum)

    # current best
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time():.2f}] init n={n}, score={best_score:.6f}")

    # --------------------------------------------------------------
    # Pre‑compute LP sparsity pattern
    # --------------------------------------------------------------
    if not _SCIPY_OK:
        print("SciPy not available – falling back to pure gradient descent.")
    else:
        rows, cols, i_idx = _build_lp_structure(n)

    # --------------------------------------------------------------
    # Phase 1 – alternating LP updates
    # --------------------------------------------------------------
    iteration = 0
    alpha = 0.5                # mixing coefficient
    no_improve = 0

    while _SCIPY_OK and time.time() < deadline:
        M_cur = evaluate_sequence(a.tolist())
        # solve the LP that finds the “most orthogonal’’ partner g
        M_opt, g = _solve_min_M_lp(a, target_sum, rows, cols, i_idx)
        if g is None:
            print("LP failed – exiting LP phase.")
            break

        # mixture update
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = np.clip(a_new, 0.0, None)               # safeguard
        a_new = project_to_sum(a_new, target_sum)

        new_score = evaluate_sequence(a_new.tolist())
        iteration += 1

        if new_score + 1e-12 < best_score:
            best_score = new_score
            best_seq = a_new.copy()
            a = a_new
            no_improve = 0
            print(f"Iter {iteration:04d} – improvement: {best_score:.6f} (α={alpha:.3f})")
            # successful step – optionally increase α a little
            alpha = min(0.9, alpha * 1.05)
        else:
            # no improvement – be more conservative
            no_improve += 1
            alpha = max(0.05, alpha * 0.7)

        # occasional escape move if we are stuck for a while
        if no_improve >= 30:
            a = random_pair_transfer(a, target_sum, scale=0.006)
            no_improve = 0
            print(f"Iter {iteration:04d} – random escape")

        # stop after a reasonable number of LP iterations
        if iteration >= 80:
            break

    # --------------------------------------------------------------
    # Phase 2 – short smooth‑max gradient descent (polishing)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.2f}] entering polishing gradient descent")
        a = best_seq.copy()
        v = np.zeros_like(a)
        lr = 0.02 * (target_sum / n)
        tau = 0.12 * a.max()
        momentum = 0.9
        grad_iter = 0
        while time.time() < deadline - 0.2:
            a, v = smooth_max_step(a, target_sum, lr, tau,
                                   momentum, v)
            grad_iter += 1
            cur = evaluate_sequence(a.tolist())
            if cur + 1e-12 < best_score:
                best_score = cur
                best_seq = a.copy()
                print(f"Grad {grad_iter:04d} – new best {best_score:.6f}")
            # slow cooling
            if grad_iter % 150 == 0:
                lr *= 0.94
                tau *= 0.93

    # --------------------------------------------------------------
    # Final random jitter (tiny budget left)
    # --------------------------------------------------------------
    while time.time() < deadline - 0.1:
        a = jitter(a, target_sum, sigma=4e-4)
        cur = evaluate_sequence(a.tolist())
        if cur < best_score:
            best_score = cur
            best_seq = a.copy()
            print(f"Jitter improved to {best_score:.6f}")

    print(f"[{time.time():.2f}] finished – best score {best_score:.6f}")
    return best_seq.tolist()
