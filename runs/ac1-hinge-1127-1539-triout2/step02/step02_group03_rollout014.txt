def simplex_projection(v, z=1):
    # v non-negative? but we will project onto simplex sum = z
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def simplex_projection(v, z=1.0):
    # From https://gist.github.com/...
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback: all zeros.
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

Works.

"""
Gradient‑softmax optimiser for the step‑function ∞‑norm problem.

The algorithm works on a simplex (∑a_i = 1, a_i ≥ 0).  Scaling does not
affect the objective, so we simply keep the L1‑norm fixed.  For a given
sequence we evaluate the full self‑convolution using an FFT (O(n log n))
and smooth the max‑operator with a temperature‑controlled soft‑max.
The gradient of the smoothed objective is obtained analytically via a
single correlation (again using an FFT).  An Adam‑style update is applied
and the iterate is projected back onto the probability simplex.  A small
probability of random “mass‑swap’’ moves (simulated‑annealing) is added
to help escape local minima.  Many short restarts with different lengths
are performed within the 1000 s budget; the best sequence (according to
the provided ``evaluate_sequence``) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum(x) = z})
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum x = z}."""
    if z < 0:
        raise ValueError("z must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    # find the largest rho satisfying the KKT condition
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, linear convolution)
# ----------------------------------------------------------------------


def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a power‑of‑2 FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_max(a: np.ndarray) -> tuple[np.ndarray, float, int]:
    """Return (full convolution, its maximum value, index of the maximum)."""
    conv = _conv_full(a)
    idx = int(np.argmax(conv))
    return conv, float(conv[idx]), idx


# ----------------------------------------------------------------------
# Gradient of the soft‑max surrogate of the max‑convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a Σ_k p_k (a*a)_k  where p is a probability vector (soft‑max).

    The formula is:
        g_i = 2 * Σ_j a_j p_{i+j}  –  a_i * p_{2i}
    which is evaluated via a single FFT‑based correlation.
    """
    n = a.shape[0]
    # length needed for linear correlation: n + (2n‑1) - 1 = 3n‑2
    corr_len = 3 * n - 2
    fft_len = 1 << (corr_len - 1).bit_length()

    Fa = np.fft.rfft(a, n=fft_len)
    Fp = np.fft.rfft(p, n=fft_len)

    corr = np.fft.irfft(Fa * np.conj(Fp), n=fft_len)  # linear correlation
    # first n entries correspond to lags 0 … n‑1
    g = corr[:n]

    # diagonal correction (the terms where i = j in the convolution)
    diag = a * p[2 * np.arange(n)]
    grad = 2.0 * g - diag
    return grad


# ----------------------------------------------------------------------
# Simple random initialisation (Dirichlet‑like)
# ----------------------------------------------------------------------


def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """Create a non‑negative length‑n sequence with sum = 1."""
    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        a += rng.normal(scale=a.mean() * 0.001, size=n)  # tiny jitter
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)

    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / total
    return a


# ----------------------------------------------------------------------
# Local optimiser: soft‑max gradient + occasional random swaps
# ----------------------------------------------------------------------


def _optimize_local(a0: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Returns the best sequence found (as a NumPy array) and its *true*
    maximal convolution value (not the soft‑max surrogate).
    """
    n = a0.shape[0]
    a = a0.copy()
    a = a / a.sum()                     # keep on the simplex

    # best seen so far
    conv, cur_max, cur_idx = _conv_max(a)
    best_a = a.copy()
    best_val = cur_max

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # step‑size schedule
    lr = 0.25
    lr_decay = 0.9998

    # temperature for the soft‑max (smooth max)
    tau = cur_max * 0.5 if cur_max > 0 else 1.0
    tau_decay = 0.996

    # simulated‑annealing swap parameters
    move_prob = 0.12
    epsilon_move = 0.04
    temp = cur_max * 0.1 + 1e-12
    temp_decay = 0.9994

    iteration = 0
    while time.time() < deadline:
        # occasional random mass move (annealed)
        if rng.random() < move_prob:
            # indexes participating in the current maximal convolution entry
            start_i = max(0, cur_idx - (n - 1))
            end_i = min(n - 1, cur_idx)
            partner_idxs = np.arange(start_i, end_i + 1)
            non_partner_idxs = np.setdiff1d(np.arange(n), partner_idxs)

            if partner_idxs.size and non_partner_idxs.size:
                i = rng.choice(partner_idxs)
                j = rng.choice(non_partner_idxs)

                delta = epsilon_move * a[i]
                delta = min(delta, a[i])

                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, 1.0)

                cand_conv, cand_max, cand_idx = _conv_max(cand)

                # Metropolis acceptance
                accept = False
                if cand_max <= cur_max:
                    accept = True
                else:
                    prob = np.exp(-(cand_max - cur_max) / max(temp, 1e-12))
                    if rng.random() < prob:
                        accept = True

                if accept:
                    a = cand
                    cur_max, cur_idx = cand_max, cand_idx
                    if cur_max < best_val:
                        best_val = cur_max
                        best_a = a.copy()
                temp *= temp_decay
            continue

        # ----- gradient step (soft‑max surrogate) -----
        iteration += 1

        # full convolution (needed for the soft‑max)
        conv = _conv_full(a)

        # update best with the *true* max (not the surrogate)
        cur_max = float(conv.max())
        cur_idx = int(np.argmax(conv))
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()

        # soft‑max over the convolution entries
        max_c = conv.max()
        exp_shift = np.exp((conv - max_c) / max(tau, 1e-12))
        p = exp_shift / exp_shift.sum()

        # gradient of the surrogate
        grad = _gradient_softmax(a, p)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        a = a - lr * m_hat / (np.sqrt(v_hat) + 1e-12)
        a = _project_simplex(a, 1.0)

        # schedule updates
        lr *= lr_decay
        tau = max(tau * tau_decay, 1e-8)

    return best_a, best_val
