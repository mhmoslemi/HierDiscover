def smooth_grad_step(a, target_sum, eta, tau_factor):
    n = a.shape[0]
    conv = conv_self_fft(a)  # length 2n-1
    max_conv = conv.max()
    # compute smoothing weights w
    tau = max(tau_factor * max_conv, 1e-12)
    w = np.exp((conv - max_conv) / tau)
    w /= w.sum()
    # gradient step: gradient = (2/tau) * conv of w and a_rev
    a_rev = a[::-1]
    L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(W * A_rev, n=L)[:len(w) + len(a_rev) - 1]
    # gradient for each a_i is (2 / tau) * conv_full[n - 1 + i]
    grad = (2.0 / tau) * conv_full[n - 1 : n - 1 + n]
    # update
    a_new = a - eta * grad
    # enforce nonnegativity and project onto simplex
    a_new = project_onto_simplex(np.maximum(a_new, 0.0), target_sum)
    return a_new, max_conv
```

def gradient_step_adam(a, target_sum, m, v, t, lr, tau_factor):
    n = a.shape[0]

    # compute conv and max
    conv = conv_self_fft(a)
    max_conv = conv.max()

    # smoothing weights
    tau = max(tau_factor * max_conv, 1e-12)
    w = np.exp((conv - max_conv)/tau)
    w_sum = w.sum()
    w /= w_sum

    # compute gradient
    a_rev = a[::-1]
    L = 1 << ((len(w)+len(a_rev)-1).bit_length())
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(W * A_rev, n=L)[:len(w)+len(a_rev)-1]
    grad = (2.0 / tau) * conv_full[n-1:n-1+n]

    # Adam update
    m = 0.9 * m + 0.1 * grad
    v = 0.999 * v + 0.001 * (grad**2)
    m_hat = m / (1 - 0.9**t)
    v_hat = v / (1 - 0.999**t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)

    a_new = project_onto_simplex(np.maximum(a_new, 0.0), target_sum)
    return a_new, max_conv, m, v
```

def propose_candidate(seed=123, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1

    # Determine n and target_sum
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base):
        n = len(base)
        init_seq = np.array(base, dtype=np.float64)
    else:
        n = np.random.randint(7500, 8800)
        init_seq = None
    target_sum = np.sqrt(2.0 * n)

    best_seq = np.full(n, target_sum / n, dtype=np.float64)
    best_score = max_conv_of(best_seq)
    print(f"Initial uniform score {best_score}")

    # generate candidates
    candidates = []

    # uniform
    candidates.append(best_seq.copy())

    # monotonic exponential shapes
    alphas = np.linspace(0.0, 5.0, 51)
    for a in alphas:
        cand = generate_monotonic_exponential(n, target_sum, a)
        candidates.append(cand)

    # monotonic linear
    slopes = np.linspace(0.0, 8.0, 41)
    for s in slopes:
        cand = generate_monotonic_linear(n, target_sum, s)
        candidates.append(cand)

    coeffs = np.linspace(0.0, 8.0, 41)
    for c in coeffs:
        cand = generate_monotonic_quadratic(n, target_sum, c)
        candidates.append(cand)

    # random Dirichlet
    for _ in range(20):
        raw = np.random.dirichlet(np.ones(n))
        cand = raw * target_sum
        candidates.append(cand)

    # half-uniform
    half = np.full(n, 0.0)
    half[:n//2] = target_sum / (n//2)
    half = project_onto_simplex(half, target_sum)
    candidates.append(half)

    # evaluate each candidate
    for cand in candidates:
        cur = max_conv_of(cand)
        if cur < best_score:
            best_score = cur
            best_seq = cand.copy()
            print(f"New best from init {best_score:.6f}")

    # Phase 2 gradient descent (Adam)
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 1
    lr = 0.1 * (target_sum / n)  # initial step
    tau_factor = 0.02
    no_improve = 0

    while time.time() < deadline:

            best_score = cur_max
            best_seq = a.copy()

"""
Propose a step‑function (non‑negative coefficient vector) that minimizes
the objective

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works in three stages:

1. **Broad initial sweep** – several families of simple parametric
   sequences (uniform, monotone linear / quadratic / exponential,
   random Dirichlet, half‑uniform) are generated, projected onto the
   ℓ₁‑simplex (∑a = √(2 n)) and the one with the smallest max‑convolution
   is kept.

2. **Smoothed gradient descent with Adam** – the non‑differentiable
   max is replaced by a log‑sum‑exp smoothing.  The gradient of the
   smoothed objective can be written as a convolution of a symmetric
   weight vector with the reversed current sequence.  An Adam step
   (first‑ and second‑moment adaptation) is taken, then the result is
   projected back onto the simplex.  The learning rate and the smoothing
   factor are slowly reduced; occasional random pair‑mass transfers and
   jitter help to escape plateaus.  The descent is restarted from a few
   random Dirichlet starts, each run keeping the best iterate seen.

3. **Final polishing** – a handful of tiny random pair transfers are
   applied to the best sequence found so far; any improvement is kept.

All convolutions are performed with FFTs (O(n log n)), allowing thousands
of iterations within the 1000 s time budget.  The function returns the
best sequence as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑convolution utilities (full linear convolution)
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a with itself using a power‑of‑two FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def max_conv_of(a: np.ndarray) -> float:
    """Return max(conv(a, a))."""
    return conv_self_fft(a).max()


# ----------------------------------------------------------------------
# Projection onto the ℓ₁‑simplex (non‑negative, prescribed sum)
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of v onto {x | x_i ≥ 0, Σx_i = target}."""
    if target <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fall back to uniform distribution
        return np.full_like(v, target / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple families of candidate sequences (all are projected onto the simplex)
# ----------------------------------------------------------------------


def generate_uniform(n: int, target_sum: float) -> np.ndarray:
    return np.full(n, target_sum / n, dtype=np.float64)


def generate_monotonic_linear(n: int, target_sum: float, slope: float) -> np.ndarray:
    """Increasing linear ramp: a_i = 1 + slope * i/(n‑1)."""
    idx = np.arange(n, dtype=np.float64)
    raw = 1.0 + slope * idx / (n - 1)
    return project_onto_simplex(raw, target_sum)


def generate_monotonic_quadratic(n: int, target_sum: float, coeff: float) -> np.ndarray:
    """Increasing quadratic ramp: a_i = 1 + coeff * (i/(n‑1))**2."""
    idx = np.arange(n, dtype=np.float64)
    raw = 1.0 + coeff * (idx / (n - 1)) ** 2
    return project_onto_simplex(raw, target_sum)


def generate_monotonic_exponential(n: int, target_sum: float, alpha: float) -> np.ndarray:
    """Increasing exponential ramp: a_i = exp(alpha * i/(n‑1))."""
    idx = np.arange(n, dtype=np.float64)
    raw = np.exp(alpha * idx / (n - 1))
    return project_onto_simplex(raw, target_sum)


def generate_half_uniform(n: int, target_sum: float) -> np.ndarray:
    """Uniform distribution on the first half of the indices, zero elsewhere."""
    a = np.zeros(n, dtype=np.float64)
    half = n // 2
    a[:half] = target_sum / half
    return project_onto_simplex(a, target_sum)


def generate_random_dirichlet(n: int, target_sum: float) -> np.ndarray:
    """Random point on the simplex via a Dirichlet draw."""
    return np.random.dirichlet(np.ones(n)) * target_sum


# ----------------------------------------------------------------------
# Small random mass transfer (always followed by projection)
# ----------------------------------------------------------------------


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """Move a tiny amount of mass from one random index to another."""
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


# ----------------------------------------------------------------------
# Smoothed gradient step (log‑sum‑exp smoothing) + Adam adaptation
# ----------------------------------------------------------------------


def gradient_step_adam(
    a: np.ndarray,
    target_sum: float,
    m: np.ndarray,
    v: np.ndarray,
    t: int,
    lr: float,
    tau_factor: float,
) -> tuple[np.ndarray, float, np.ndarray, np.ndarray]:
    """
    Perform one Adam‑style gradient step on the smoothed objective.
    Returns (new_a, max_conv, new_m, new_v).
    """
    n = a.shape[0]

    # 1) convolution of a with itself and its maximum
    conv = conv_self_fft(a)
    max_conv = conv.max()

    # 2) smoothed weighting (log‑sum‑exp)
    tau = max(tau_factor * max_conv, 1e-12)
    # subtract max for numerical stability
    w = np.exp((conv - max_conv) / tau)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # 3) gradient of the smoothed max
    a_rev = a[::-1]
    L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]

    # slice that corresponds to ∂/∂a_i
    grad = (2.0 / tau) * conv_full[n - 1 : n - 1 + n]

    # 4) Adam update
    beta1, beta2 = 0.9, 0.999
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad ** 2)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)

    # 5) enforce constraints
    a_new = project_onto_simplex(np.maximum(a_new, 0.0), target_sum)

    return a_new, max_conv, m, v

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------

    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2   # small safety margin

    # ------------------------------------------------------------------
    # Problem size (use existing construction if present)
    # ------------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:

        init_from_base = None

    target_sum = np.sqrt(2.0 * n)             # normalization used by the scorer

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    best_seq = generate_uniform(n, target_sum)

    candidates = [best_seq.copy()]

    # optional baseline

    # evaluate all candidates
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        cur = max_conv_of(cand)
        if cur < best_score:
            best_score = cur
            best_seq = cand.copy()
            print(f"  Init #{idx:03d} improved score to {best_score:.6f}")

    print(f"[{time.time():.2f}] Starting gradient descent, best score {best_score:.6f}")
