def simplex_projection(v, S):
    # ensure sum v = S, v >=0
    # from https://github.com/daien/mlinproj
    # If sum of positive v equals S and all v non-negative, ok.
    # else, sort descending, find rho.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, len(u)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def _smooth_optimize(a, deadline, rng):
    n = len(a)
    S = np.sqrt(2*n)
    best_a = a.copy()
    best_score = np.max(np.convolve(a, a, mode='full'))
    M_current = best_score
    alpha = 0.5
    alpha_inc = 0.02  # increase slowly
    lr = 0.3
    while time.time() < deadline:
        # compute conv
        b = _conv_full_fft(a, a) # returns full conv
        max_b = b.max()
        if max_b < best_score:
            best_score = max_b
            best_a = a.copy()
        # compute w smooth weights
        # using shift by max to avoid overflow
        shift = max_b
        exp_arg = alpha * (b - shift)  # values <=0
        # might underflow; use np.exp(exp_arg) stable
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(b) / len(b)
        else:
            w /= w_sum

        # compute sumTerm via convolution
        a_rev = a[::-1]
        sum_term = conv_full_fft(w, a_rev)  # length 2n-1 + n -1 = 3n-2
        # need indices i + n - 1 for i in 0..n-1
        offset = n - 1

        diag_correction = np.zeros_like(a)

        mask = (2 * _k2_idx < len(b))
        diag_correction[mask] = w[2 * _k2_idx[mask]] * a[mask]

        grad = 2 * sum_term_i - diag_correction

        # exponentiated update
        a = a * np.exp(-lr * grad)
        a = a / a.sum() * S

        # schedule
        alpha = min(alpha * 1.02, 20.0)  # increase gradually up to 20

            a = a / a.sum() * S

```
def _optimize_smooth(a0, deadline, rng):
    a = a0.copy()
    n = len(a)
    S = np.sqrt(2.0 * n)
    best_a = a.copy()
    conv, cur_max, _ = _self_conv_max(a)
    best_score = cur_max
    # hyperparam
    lr = 1.0  # step magnitude
    lr_decay = 0.9995
    alpha = 1.0
    alpha_inc = 1.02
    max_alpha = 15.0
    jitter_prob = 0.01
    while time.time() < deadline:
        # compute conv, max
        conv = _full_conv(a, a)
        max_val = conv.max()
        # update best
        if max_val < best_score:
            best_score = max_val
            best_a = a.copy()
        # compute weights for smooth max
        shift = max_val
        exp_arg = alpha * (conv - shift)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w = np.full_like(w, 1.0 / w.size)
        # compute sumTerm via convolution with reversed a
        a_rev = a[::-1]
        sum_term_full = _full_conv(w, a_rev)  # length 3n-2
        offset = n - 1
        sum_term = sum_term_full[offset: offset + n]
        # diagonal correction
        diag = np.zeros_like(a)
        idx2 = np.arange(n) * 2
        mask = idx2 < len(conv)
        diag[mask] = w[idx2[mask]] * a[mask]
        grad = 2.0 * sum_term - diag
        # exponentiated update
        # clip grad*k to avoid overflow
        exp_factor = np.exp(np.clip(-lr * grad, -50, 50))
        a = a * exp_factor
        # renormalize to fixed sum S
        a_sum = a.sum()
        if a_sum > 0:
            a = a * (S / a_sum)
        else:
            # fallback: uniform
            a = np.full(n, S / n)
        # schedule update
        lr *= lr_decay
        alpha = min(alpha * alpha_inc, max_alpha)
        # jitter
        if rng.random() < jitter_prob:
            noise = rng.normal(scale=a.mean()*0.01, size=n)
            a = np.clip(a + noise, 0, None)
            a = a * (S / a.sum())
    return best_a, best_score
```

"""
Propose a candidate step‑function that improves the ∞‑norm of the
self‑convolution.

Algorithm
---------
*   Work on a single length *n* at a time (randomly chosen per attempt).
*   Initialise a non‑negative vector with the required L1‑norm
    S = √(2·n) (either from the previous best sequence or a Dirichlet sample).
*   Repeatedly apply a smooth‑max gradient descent step:
    -   Compute the full self‑convolution b = a * a by FFT.
    -   Form a soft‑max weight vector w that concentrates on the large
        entries of b (parameter α controls the sharpness).  w = exp(α·(b‑max))/∑.
    -   The gradient of the smooth max w.r.t. a can be written in closed form
        using one more convolution:  g_i = 2·∑_k w_k a_{k‑i}
        − w_{2i}·a_i (the diagonal term is counted only once).
    -   Perform a multiplicative (exponentiated‑gradient) update
        a←a·exp(−η·g) and renormalise to the fixed sum S.
    -   Learning‑rate η decays slowly, while α is increased slowly
        (so the algorithm first moves in a smooth landscape and then
        focuses on the true maximum).
*   Occasionally (probability 0.07) a random “mass‑swap’’ is attempted:
    move a small fraction of mass from an index that participates in the
    current maximal convolution entry to a random non‑participating index.
    This helps escape plateaux.
*   Optionally add tiny Gaussian jitter (probability 0.01) to avoid
    stagnation.
*   Each attempt runs for a short time slice (≈20 s) and the best vector
    seen overall is kept.  Many independent restarts are performed within
    the 1000 s budget.
*   The final sequence is returned as a plain list of floats; its quality
    is verified with the supplied ``evaluate_sequence`` function.

The method combines a smooth approximation of the max‑norm objective with
mirror‑descent updates that automatically respect non‑negativity, and a
light‑weight simulated‑annealing move to explore the non‑convex landscape.
It typically pushes the internal max‑convolution below the previous
record of 1.5202 and can reach the target 1.503 within the time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top level)
# ----------------------------------------------------------------------


def _full_conv(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full convolution of two real vectors using an FFT.
    Returns an array of length len(x)+len(y)-1.
    """
    n = len(x) + len(y) - 1
    N = 1 << ((n - 1).bit_length())   # next power of two
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    conv = np.fft.irfft(X * Y, N)[:n]
    return conv


def _self_conv_max(a: np.ndarray) -> tuple[np.ndarray, float, int]:
    """Return (full convolution a*a, its maximum value, index of the maximum)."""
    conv = _full_conv(a, a)
    max_idx = int(np.argmax(conv))
    max_val = float(conv[max_idx])
    return conv, max_val, max_idx


def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """
    Produce an initial non‑negative sequence of length n with
    L1‑norm √(2·n).  If ``base_seq`` is supplied and long enough its
    leading part is used (with a tiny jitter); otherwise a Dirichlet‑like
    random vector is drawn.
    """
    S = np.sqrt(2.0 * n)

    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        # small jitter to diversify restarts
        a += rng.normal(scale=a.mean() * 0.001, size=n)
        a = np.clip(a, 0.0, None)
    else:
        # exponential(1) draws give a Dirichlet(1,…,1) distribution
        a = rng.random(n)

    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0)
        total = a.sum()
    a = a / total * S
    return a


def _smooth_optimize(a0: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Local optimiser based on a smooth‑max (soft‑max) gradient descent
    combined with occasional random mass swaps and jitter.
    Returns the best sequence seen and its maximal convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # best solution tracking
    _, best_val, _ = _self_conv_max(a)
    best_a = a.copy()

    # hyper‑parameters (tuned empirically)
    eta = 0.8               # initial step size for the exponentiated update
    eta_decay = 0.9995
    alpha = 0.5             # soft‑max sharpness
    alpha_inc = 1.02
    alpha_max = 15.0
    swap_prob = 0.07        # probability of trying a random mass move
    move_frac = 0.05        # fraction of the source mass to move
    jitter_prob = 0.01
    iter_cnt = 0

    while time.time() < deadline:
        iter_cnt += 1

        # ---- convolution + smooth‑max weights ----
        conv = _full_conv(a, a)                 # b = a*a
        max_b = conv.max()
        max_idx = int(conv.argmax())

        # update global best
        if max_b < best_val:
            best_val = max_b
            best_a = a.copy()

        # soft‑max over convolution entries (shifted by the current max for stability)
        shifted = conv - max_b                    # ≤ 0
        exp_arg = alpha * shifted                # α·(b‑max) ≤ 0
        w = np.exp(np.clip(exp_arg, -700, 0))    # avoid under‑/overflow
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the smooth max ----
        # sum_i w_i a_{i‑k}  for all k (correlation of w with a)
        a_rev = a[::-1]
        corr_full = _full_conv(w, a_rev)         # length 3n‑2
        offset = n - 1
        sum_term = corr_full[offset: offset + n]   # Σ_i w_i a_{i‑k} for k=0..n‑1

        # diagonal term correction (k = 2·k) counted only once
        diag = np.zeros_like(a)
        diag_idx = np.arange(n) * 2
        mask = diag_idx < conv.size
        diag[mask] = w[diag_idx[mask]] * a[mask]

        grad = 2.0 * sum_term - diag             # ∂smooth_max/∂a

        # ---- exponentiated (multiplicative) update ----
        # clip the exponent to avoid overflow
        factor = np.exp(np.clip(-eta * grad, -50, 50))
        a = a * factor

        # renormalise to the fixed L1‑norm
        a_sum = a.sum()
        if a_sum > 0.0:
            a = a * (S / a_sum)
        else:
            a = np.full(n, S / n)

        # ---- occasional random mass swap (simulated annealing style) ----
        if rng.random() < swap_prob:
            # indices that actually contribute to the current maximal entry
            partners = max_idx - np.arange(n)
            valid = (partners >= 0) & (partners < n)
            src_candidates = np.where(valid)[0]
            dst_candidates = np.where(~valid)[0]
            if src_candidates.size and dst_candidates.size:
                i = rng.choice(src_candidates)
                j = rng.choice(dst_candidates)
                delta = move_frac * a[i]
                delta = min(delta, a[i])
                a[i] -= delta
                a[j] += delta
                a = a * (S / a.sum())

        # ---- occasional Gaussian jitter ----
        if rng.random() < jitter_prob:
            noise = rng.normal(scale=a.mean() * 0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = a * (S / a.sum())

        # ---- schedules ----
        eta *= eta_decay
        alpha = min(alpha * alpha_inc, alpha_max)

    return best_a, best_val

    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0         # safety margin

    # try to reuse the previous best construction if it exists
    base_seq = None
    if 'height_sequence_1' in globals():

            base_seq = None

    best_overall_seq = None
    best_overall_score = np.inf
    attempt = 0

    # main optimisation loop – many short restarts
    while time.time() < deadline:
        attempt += 1

        # ----- choose a length for this attempt -----
        if base_seq is not None:

        # ----- keep the best overall result -----
