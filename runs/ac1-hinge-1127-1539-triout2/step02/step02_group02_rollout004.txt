def build_conv_matrix(f):
    n = len(f)
    m = 2*n - 1
    H = np.zeros((m, n))
    for i in range(n):
        # This column corresponds to g_i.
        # For each shift where g_i contributes: for row k = i + j for j in 0..n-1.
        # coefficient = f_j (since k-i = j)
        H[i:i+n, i] = f  # because rows from i to i+n-1 correspond to k = i+j, coefficient f_j.
    return H

def minimize_max_convolution(n, max_iters=20, t=0.2):
    # initialize f random as Dirichlet or uniform.
    f = np.full(n, 1.0) (or random). Then normalize to S = sqrt(2n).
    for iter in range(max_iters):
        # compute convolution of f with itself: conv = np.convolve(f,f)
        M = conv.max()
        # Build convolution matrix H_f (size (2n-1) x n)
        H = build_conv_matrix(f)
        # solve LP
        res = linprog(c=-np.ones(n), A_ub=H, b_ub=np.full(2*n-1, M), bounds=(0, None), method='highs', options={'presolve': True, 'time_limit': 5})
        if not res.success:
            break
        g0 = res.x
        # rescale to have sum = S
        sum_g0 = g0.sum()
        if sum_g0 == 0:
            # fallback uniform perhaps
            g = np.full(n, S/n)
        else:
            g = g0 * (S / sum_g0)
        # update f
        f = (1 - t) * f + t * g
        # renorm to sum S again (since g sum ensures S, the mixture also sum S maybe deviates due to rounding)
        f = f / f.sum() * S
        # Evaluate new max M_new; if improvement then keep track.
        # If M_new close to previous within tolerance break.
    return f

def _lp_refine(f, t=0.2, max_iter=5):
    n = f.shape[0]
    S = math.sqrt(2.0 * n)
    for it in range(max_iter):
        conv = np.convolve(f, f)
        M = conv.max()
        H = _build_conv_matrix(f)
        # Use linprog: maximize sum(g) => minimize -sum(g)
        c = -np.ones(n)
        A_ub = H
        b_ub = np.full(2*n-1, M)
        bounds = [(0, None) for _ in range(n)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 3})
        if not res.success:
            break
        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 == 0:
            g = np.full(n, S/n)
        else:
            g = g0 * (S / sum_g0)
        # mix
        f = (1 - t) * f + t * g
        # renormalize to sum S
        f = f / f.sum() * S
    return f

"""
Improved search for a step function that minimises ``evaluate_sequence``.
The core is a fast exponentiated‑gradient descent on the max‑convolution
surrogate (the convolution maximum is exactly the external score once the
vector is normalised to ℓ₁–norm √(2 n)).  The optimiser runs many
independent restarts with different lengths, adaptive top‑K gradients,
momentum and occasional jitter.  When a new global best is found a tiny
linear‑programming refinement (if SciPy is present and the problem size
is moderate) is applied to push the bound further.

The routine respects the 1000 s wall‑clock budget, prints progress and
returns the best sequence as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# optional import – the LP refinement is only used when SciPy is available
try:
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


def _conv_max(a: np.ndarray):
    """
    Linear self‑convolution via FFT.
    Returns (full convolution array, maximal entry, index of the maximum).
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()      # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    max_idx = int(np.argmax(conv))
    return conv, float(conv[max_idx]), max_idx


def _gradient_from_shifts(a: np.ndarray, shifts: np.ndarray):
    """
    Sub‑gradient of the sum of the selected convolution entries.
    For each shift k we add 2·a_{k‑i} to the i‑th component whenever the
    index lies inside the vector.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    idx = np.arange(n)
    for k in shifts:
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g[valid] += 2.0 * a[partner[valid]]
    return g


def _run_optimize_one_start(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            top_k_start: int = 3,
                            eta0: float = 0.8,
                            beta: float = 0.9,
                            decay: float = 0.99995,
                            jitter_prob: float = 0.02,
                            max_iters: int = 200_000):
    """
    Exponentiated‑gradient descent (with momentum) starting from ``a0``.
    The surrogate loss is the maximal entry of the self‑convolution.
    Returns the best vector seen during the run and its surrogate value.
    """
    a = a0.copy()
    S = math.sqrt(2.0 * n)                # ℓ₁‑norm we keep fixed
    best_local_max = float('inf')
    best_local_seq = a.copy()

    v = np.zeros_like(a)                 # momentum buffer
    eta = eta0
    it = 0
    last_improve = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv, cur_max, _ = _conv_max(a)

        # adapt the number of active shifts – increase if we have stalled
        top_k = top_k_start
        if it - last_improve > 2_000 and top_k < 30:
            top_k = min(30, top_k * 2)

        if conv.size <= top_k:
            shifts = np.arange(conv.size)
        else:
            shifts = np.argpartition(conv, -top_k)[-top_k:]

        # sub‑gradient of the sum of the selected shifts
        g = _gradient_from_shifts(a, shifts)

        # momentum (exponential moving average)
        v = beta * v + (1.0 - beta) * g

        # multiplicative (exponentiated) update – guarantees positivity
        a = a * np.exp(-eta * v)

        # renormalise to keep the ℓ₁‑norm fixed
        total = a.sum()
        if total == 0.0:
            a.fill(S / n)
        else:
            a *= (S / total)

        # occasional random jitter to escape tiny plateaus
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * 0.01
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()

        # step‑size decay
        eta *= decay

        # record improvement of the surrogate (which equals the external score)
        if cur_max < best_local_max - 1e-12:
            best_local_max = cur_max
            best_local_seq = a.copy()
            last_improve = it

        # early exit if we are about to run out of time
        if time.time() >= deadline - 0.1:
            break

    return best_local_seq, best_local_max


def _build_conv_matrix(f: np.ndarray):
    """
    Construct the (2n‑1) × n matrix H where (H @ g)[k] = (f * g)[k].
    This routine is used only in the LP refinement and therefore builds
    a dense matrix – the dimensions are kept modest (n ≤ 3000).
    """
    n = f.shape[0]
    m = 2 * n - 1
    H = np.zeros((m, n), dtype=np.float64)
    # each column i contributes the vector f aligned at rows i … i+n‑1
    for i in range(n):
        H[i:i + n, i] = f
    return H


def _lp_refine(f: np.ndarray, deadline: float, t: float = 0.2,
               max_steps: int = 5):
    """
    Linear‑programming refinement (Frank‑Wolfe / cutting‑plane style).

    Given the current vector ``f`` we solve:
        maximise   Σ g_i
        subject to 0 ≤ g_i ,   (f * g)[k] ≤ M   for all k,
    where M = max_k (f * f)[k].
    The optimal ``g`` is then mixed with ``f`` (step size ``t``) and
    renormalised.  A few such iterations often lower the maximal
    convolution entry.

    The method is invoked only when SciPy is available and n is small
    enough (≤ 3000) to keep the LP tractable.
    """
    if not _SCIPY_AVAILABLE:
        return f
    n = f.shape[0]
    if n > 3000:          # avoid gigantic dense LPs
        return f

    S = math.sqrt(2.0 * n)

    for _ in range(max_steps):
        if time.time() >= deadline - 0.5:
            break

        conv = np.convolve(f, f)
        M = conv.max()

        H = _build_conv_matrix(f)
        c = -np.ones(n)                     # maximise Σg  ⇔ minimise –Σg
        A_ub = H
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None) for _ in range(n)]

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 8})
        except Exception:                 # pragma: no cover
            break

        if not res.success:
            break

        g0 = res.x
        s0 = g0.sum()
        if s0 == 0.0:
            g = np.full(n, S / n)
        else:
            g = g0 * (S / s0)

        # convex combination and renormalisation
        f = (1.0 - t) * f + t * g
        f = f / f.sum() * S

    return f

    Main entry point required by the evaluation harness.
    Returns a list of non‑negative heights that (hopefully) yields a score
    lower than the current best bound.
    """

    deadline = time.time() + budget_s - 5.0   # small safety margin

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------

        except Exception:                    # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # 2)  Search over a few lengths.  The surrogate loss equals the
    #     external score once the vector is normalised, so we can work
    #     with the max‑convolution directly.
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000, 12000, 16000]
    run_id = 0

    while time.time() < deadline:

        # ----------------------------------------------------------------

        # One optimisation run.
        # ----------------------------------------------------------------
        seq, score = _run_optimize_one_start(a0, n, deadline)

        # keep the best sequence seen so far (score is the max‑convolution)

        best_seq = a
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] score = {best_score:.6f}")

    return best_seq.tolist()
