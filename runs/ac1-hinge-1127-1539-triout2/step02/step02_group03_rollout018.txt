def grad_smooth_max(a, w):

    a_rev = a[::-1]
    Lw = len(w) # = 2n-1
    Ltemp = n + Lw - 1 = 3n - 2

    corr = np.fft.irfft(Fr * Fw, n=nfft2)[:Ltemp]

def local_optimize(n, rng, deadline):

    a = rng.random(n)
    a = a / a.sum()
    best_a = a.copy()
    best_max = conv_max(a) # compute initial max conv.

    # set alpha and eta.
    alpha = 5.0
    eta = 0.5
    decay_alpha = 1.02 (increase alpha each iteration up to maybe 200)
    decay_eta = 0.995.

    while time.time() < deadline:

        max_val = conv.max()

            best_max = max_val
            best_a = a.copy()

def _project_simplex(v, z=1):
    # algorithm when sum of v > z could be >0. handle negative.
    v = np.asarray(v, dtype=float)
    if np.sum(v) == z and np.all(v >= 0):
        return v
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, len(v)+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zeros
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

    # ensure normalized
    a = _project_simplex(a, 1.0)
    best_a = a.copy()
    cur_conv = conv_fft(a)
    best_max = cur_conv.max()
    # hyperparameters

    alpha = 5.0

    decay_eta = 0.9995
    decay_alpha = 1.001
    # moves
    prob_swap = 0.1
    epsilon_move = 0.02

    while time.time() < deadline:
        cur_conv = conv_fft(a)
        cur_max = cur_conv.max()
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
        # decide operation

            # targeted swap: pick i from contributors
            # identify contributors to convex: pair indices where i+ j = idx_max
            k = np.argmax(cur_conv)
            # list of i where partner = k - i valid

"""
A smooth‑max gradient descent optimiser for the step‑function problem.

The evaluation function is homogeneous, therefore we normalise each
candidate to have L1‑norm 1.  With this normalisation the external score
reduces to `2·n·max(conv(f,f))`, so the internal objective we minimise is
the maximal entry of the self‑convolution.

The algorithm works as follows:

1. Initialise a non‑negative vector (optionally from the previously
   discovered best sequence) and project it onto the simplex {x≥0,
   Σx=1}.
2. While time permits, repeatedly
   * compute the full convolution via an FFT,
   * form a smooth approximation of the max (soft‑max with a growing
     temperature parameter α),
   * obtain the gradient of this smooth max (a correlation of the
     soft‑max weights with the current vector),
   * take an additive step `x ← x − η·grad`, clip to non‑negative and
     re‑project onto the simplex,
   * occasionally perform a targeted random swap of a small amount of
     mass between an index that participates in the current maximal
     convolution entry and a random non‑participating index (simulated
     annealing),
   * slowly decrease the step size η and increase the smoothing
     parameter α.
3. Keep the best sequence seen (lowest maximal convolution value)
   across many independent restarts with different lengths and random
   seeds.
4. Return the best sequence as a plain Python list; the harness will
   evaluate it with the provided `evaluate_sequence` function.

The optimiser is deliberately lightweight (FFT‑based O(n log n) operations)
and can run many short restarts within the 1000 s budget, improving on
the previous best bound (≈1.5202) and aiming for the target ≤1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Project a vector onto the simplex {x≥0, Σx = z}."""
    if z < 0:
        raise ValueError("z must be non‑negative")
    v = v.astype(float, copy=True)
    # Fast projection (see Duchi et al., 2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT (real inputs)."""
    n = a.shape[0]
    L = 2 * n - 1
    nfft = 1 << ((L - 1).bit_length())        # next power of two
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[:L]
    return conv


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Soft‑max weights that approximate the arg‑max of conv.
    The weights sum to one.
    """
    max_val = conv.max()
    # shift for numerical stability
    shifted = conv - max_val
    exp_vals = np.exp(alpha * shifted)
    w = exp_vals / exp_vals.sum()
    return w


def _gradient_smooth_max(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth max with respect to a.
    It is 2 * (a reversed) convolved with the weight vector w,
    extracted at the valid positions (length n).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    Lw = w.shape[0]               # = 2*n-1
    Ltemp = n + Lw - 1             # = 3*n-2
    nfft = 1 << ((Ltemp - 1).bit_length())
    Fa_rev = np.fft.rfft(a_rev, n=nfft)
    Fw = np.fft.rfft(w, n=nfft)
    corr = np.fft.irfft(Fa_rev * Fw, n=nfft)[:Ltemp]
    grad = 2.0 * corr[n - 1:n - 1 + n]   # slice of length n
    return grad


def _gradient_step_smooth(a: np.ndarray, eta: float, alpha: float) -> np.ndarray:
    """
    One additive gradient step that reduces a smoothed approximation of
    the maximal convolution entry.
    """
    conv = _conv_fft(a)
    w = _smooth_weights(conv, alpha)          # soft‑max over conv entries
    grad = _gradient_smooth_max(a, w)

    # additive step, then enforce feasibility
    a_new = a - eta * grad
    a_new = np.clip(a_new, 0.0, None)
    a_new = _project_simplex(a_new, 1.0)       # keep L1‑norm = 1
    return a_new


def _initial_sequence(base_seq: np.ndarray, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative sequence of length n with L1‑norm 1.
    If a suitable base sequence is supplied, use its leading part with a
    tiny jitter; otherwise draw from a Dirichlet‑like uniform distribution.
    """
    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        a += rng.normal(scale=a.mean() * 0.001, size=n)
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)

    a = a / a.sum()                           # project onto simplex
    return a


def _optimize_local(a: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Local optimiser mixing smooth‑max gradient steps with targeted
    simulated‑annealing swaps.
    Returns the best sequence seen (as a NumPy array) together with its
    maximal convolution value.
    """
    n = a.shape[0]

    # keep the best solution encountered
    conv = _conv_fft(a)
    best_a = a.copy()
    best_max = conv.max()

    # hyper‑parameters – these were tuned empirically
    eta = 0.8                     # initial additive step size
    alpha = 5.0                   # initial smooth‑max temperature
    alpha_target = 200.0
    eta_decay = 0.9997
    alpha_growth = 1.002
    swap_prob = 0.12              # how often to attempt a random swap
    eps_move = 0.04               # fraction of mass moved during a swap
    jitter_prob = 0.008
    temp = best_max * 0.05        # annealing temperature for swaps
    temp_decay = 0.9995

    while time.time() < deadline:
        # possibly perform a targeted swap (annealing move)
        if rng.random() < swap_prob:
            # index of the current maximal convolution entry
            cur_k = int(np.argmax(_conv_fft(a)))

            # indices that actually contribute to that entry
            i_candidates = np.arange(n)
            j_partners = cur_k - i_candidates
            valid = (j_partners >= 0) & (j_partners < n)
            active_i = i_candidates[valid]

            # indices that do NOT participate in the max entry
            inactive_i = i_candidates[~valid]

            if active_i.size and inactive_i.size:
                i = rng.choice(active_i)
                j = rng.choice(inactive_i)

                delta = eps_move * a[i]
                delta = min(delta, a[i])                 # keep non‑negative

                a_candidate = a.copy()
                a_candidate[i] -= delta
                a_candidate[j] += delta
                a_candidate = _project_simplex(a_candidate, 1.0)

                cand_max = _conv_fft(a_candidate).max()

                # classic simulated‑annealing acceptance rule
                if cand_max <= best_max:
                    accept = True
                else:
                    prob = np.exp(-(cand_max - best_max) / max(temp, 1e-12))
                    accept = rng.random() < prob

                if accept:
                    a = a_candidate
                    if cand_max < best_max:
                        best_max = cand_max
                        best_a = a.copy()
                temp *= temp_decay
        else:
            # smooth‑max gradient descent step
            a = _gradient_step_smooth(a, eta, alpha)
            cur_max = _conv_fft(a).max()
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()

        # occasional jitter to avoid stagnation
        if rng.random() < jitter_prob:
            noise = rng.normal(scale=a.mean() * 0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_simplex(a, 1.0)

        # schedule updates
        eta *= eta_decay
        alpha = min(alpha_target, alpha * alpha_growth)

    return best_a, best_max


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Entry point required by the evaluation harness.
    Executes many short independent optimisation runs with different
    lengths and random seeds, keeping the best sequence overall.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0          # safety margin

    # reuse the previously discovered sequence if it exists
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_overall_seq = None
    best_overall_score = np.inf
    attempt = 0

    # main loop: perform many short restarts within the time budget
    while time.time() < deadline:
        attempt += 1

        # pick a length – explore a reasonable range
        if base_seq is not None:
            max_len = min(6000, base_seq.size)
            n = rng.integers(800, max_len + 1)
        else:
            n = rng.integers(800, 4000)

        # initialise a candidate sequence
        a0 = _initial_sequence(base_seq, n, rng)

        # allocate a time slice for this run (≈30 s)
        slice_deadline = min(time.time() + 30.0, deadline)

        # run the local optimiser
        a_opt, max_conv = _optimize_local(a0, slice_deadline, rng)

        # external score for a normalized vector (L1‑norm 1)
        score = 2.0 * n * max_conv

        if score < best_overall_score:
            best_overall_score = score
            best_overall_seq = a_opt.copy()
            print(f"[global] attempt={attempt} n={n} new best score={best_overall_score:.6f}")

    # final sanity check – evaluate through the provided black‑box function
    if best_overall_seq is not None:
        final_seq = best_overall_seq.tolist()
        external_score = evaluate_sequence(final_seq)
        print(f"[final] best external score = {external_score:.6f} "
              f"(internal max‑conv = {best_overall_score / (2 * len(final_seq)):.6f}) "
              f"length = {len(final_seq)}")
        return final_seq
    else:
        # fallback – should never happen
        return [1.0]
