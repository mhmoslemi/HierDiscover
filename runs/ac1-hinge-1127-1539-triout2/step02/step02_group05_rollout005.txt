Propose a step‑function (non‑negative vector) that minimizes

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

Because we always keep the ℓ1‑norm fixed to √(2 n), the objective is
exactly the maximum entry of the linear (zero‑padded) autoconvolution
`conv(a, a)`.  The algorithm below performs a projected Adam optimisation
on a smooth Lp‑approximation of the ∞‑norm of the convolution:

* Lp‑loss  L(a) = Σ |conv(a,a) – μ|^p   with  μ = (∑a)²/(2n‑1)   (the
  average convolution value).  For p → ∞  the loss approaches the
  max‑norm.

* The gradient of the loss can be written in closed form using a single
  correlation, which we compute efficiently with FFTs.

* After each Adam step we project the iterate onto the simplex
  {x ≥ 0, ∑x = √(2 n)} to keep the constraints satisfied.

* The exponent p is increased step‑wise (2 → 4 → 8 → … → 64) so that the
  optimisation first flattens the convolution globally (small p) and
  then sharpens the peak (large p).  A short “fine‑tune’’ phase with a
  very large p and a tiny learning rate finishes the run.

The whole routine is fully vectorised, uses only NumPy and runs well
within the 1000 s budget.

import time

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _compute_score(a: np.ndarray) -> float:
    """
    Fast version of `evaluate_sequence` – assumes a is non‑negative.
    Returns 2·n·max(conv(a,a))/sum(a)².
    """
    n = a.shape[0]
    nfft = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))


def project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x≥0, ∑x = target_sum}.
    Implements the algorithm of Duchi et al. (ICML 2008).
    """
    if target_sum <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(len(v)) + 1  # 1‑based
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    # Exponent schedule: p starts at 2 and is doubled a few times.
    # The schedule is time‑based, not iteration‑based.
    p_schedule = [(0.0, 2.0), (0.2, 4.0), (0.4, 8.0),
                  (0.6, 16.0), (0.8, 32.0), (0.9, 64.0)]
    cur_p = 2.0

    # ------------------------------------------------------------------
    # 5. Main optimisation loop (gradient on Lp loss)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ----- 5.1 – current convolution (FFT) -----
        af = np.fft.rfft(a, n=nfft_conv)
        conv = np.fft.irfft(af * af, n=nfft_conv)[: 2 * n - 1]

        # ----- 5.2 – compute Lp‑gradient -----
        diff = conv - avg_conv                     # shape (2n‑1,)
        # clip absolute values to avoid overflow for very large p
        abs_diff = np.abs(diff)
        abs_diff = np.clip(abs_diff, 0.0, 2.0)

        # weight = dL/d conv  = p * sign(diff) * |diff|^{p‑1}
        weight = np.sign(diff) * (abs_diff ** (cur_p - 1.0))
        weight *= cur_p

        # correlation (gradient of the loss w.r.t. a)
        wf = np.fft.rfft(weight, n=nfft_corr)
        af_rev = np.fft.rfft(a[::-1], n=nfft_corr)
        corr_full = np.fft.irfft(wf * af_rev, n=nfft_corr)
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]   # length‑n slice

        # ----- 5.3 – Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- 5.4 – projection onto the simplex -----
        a_new = project_onto_simplex(a_new, target_sum)

        # ----- 5.5 – evaluate and keep the best -----
        score = _compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            elapsed = time.time() - start_time
            print(
                f"Iter {it:6d}, p={cur_p:5.1f}, score {best_score:.6f} "
                f"(elapsed {elapsed:.1f}s)"
            )

        a = a_new

        # ----- 5.6 – schedules (learning‑rate & exponent) -----
        lr = max(lr * lr_decay, lr_min)

        # Update exponent according to elapsed fraction of the budget
        frac = (time.time() - start_time) / budget_s
        for thr, val in reversed(p_schedule):
            if frac >= thr:
                cur_p = val
                break

        # Tiny random jitter every few hundred iterations – helps escape plateaus
        if it % 257 == 0:
            jitter = 1e-5 * np.random.randn(n)
            a = project_onto_simplex(a + jitter, target_sum)

        # Periodic safety check
        if it % 1000 == 0 and time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 6. Fine‑tuning phase – very high p, tiny step size
    # ------------------------------------------------------------------
    fine_iters = 2000
    lr_fine = 5e-4
    cur_p = 128.0                     # practically the max‑norm
    for _ in range(fine_iters):
        if time.time() >= deadline:
            break

        af = np.fft.rfft(a, n=nfft_conv)
        conv = np.fft.irfft(af * af, n=nfft_conv)[: 2 * n - 1]

        diff = conv - avg_conv
        abs_diff = np.abs(diff)
        abs_diff = np.clip(abs_diff, 0.0, 2.0)
        weight = np.sign(diff) * (abs_diff ** (cur_p - 1.0))
        weight *= cur_p

        wf = np.fft.rfft(weight, n=nfft_corr)
        af_rev = np.fft.rfft(a[::-1], n=nfft_corr)
        corr_full = np.fft.irfft(wf * af_rev, n=nfft_corr)
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]

        a = a - lr_fine * grad
        a = project_onto_simplex(a, target_sum)

        score = _compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            elapsed = time.time() - start_time
            print(
                f"Fine‑tune: score {best_score:.6f} (elapsed {elapsed:.1f}s)"
            )

    # ------------------------------------------------------------------
    # 7. Return the best sequence found
    # ------------------------------------------------------------------
    return best_a.tolist()
