def full_convolution(a):
    # compute linear convolution using FFT
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv

def max_convolution(a):
    conv = full_convolution(a)
    max_val = conv.max()
    return max_val, conv

def gradient_topk(a, conv, topk, alpha):
    n = a.shape[0]
    # get topk indices
    if conv.size <= topk:
        shifts = np.arange(conv.size)
        weights = np.ones_like(shifts, dtype=float) / shifts.size
    else:
        # partial argpartition
        idx = np.argpartition(conv, -topk)[-topk:]
        # sort descending?
        shifts = idx
        # compute weights via softmax
        top_vals = conv[shifts]
        # stable softmax
        max_val = top_vals.max()
        exps = np.exp(alpha * (top_vals - max_val))
        weights = exps / exps.sum()
    # compute gradient vector g
    g = np.zeros_like(a)
    for k, wk in zip(shifts, weights):
        # indexes i where 0 <= i < n and 0 <= k-i < n
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start > i_end:
            continue
        i = np.arange(i_start, i_end + 1)
        j = k - i
        g[i] += 2.0 * wk * a[j]
    return g

def optimize_one_start(a0, n, deadline, topk_start=3, topk_max=30, steps=20000):

    beta2 = 0.999

    best_surrogate = np.inf

        iteration += 1

        cur_max = conv.max()

                ext = evaluate_sequence(a.tolist())
                if ext < best_score:
                    best_score = ext
                    best_seq = a.copy()
                    print(f"[improve] iter={iteration} n={n} score={best_score:.6f} cur_max={cur_max:.6f}")
            except Exception:
                pass

        # adapt topk: linearly increase after certain steps
        # compute current topk as function of iteration
        top_k = int(topk_start + (topk_max - topk_start) * min(1.0, iteration / (steps*0.5)))
        # alpha: start low then increase
        alpha = 0.5 + 4.5 * min(1.0, iteration / (steps*0.5))

        # compute gradient
        g = gradient_topk(a, conv, top_k, alpha)

        # Adam update on log_a
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        lr = lr0 * (1 - iteration / steps)  # linear decay

        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # map back to a and renormalise
        a = np.exp(log_a)
        total = a.sum()
        if total == 0.0:
            a.fill(S / n)
        else:
            a *= (S / total)

        # jitter occasionally
        if np.random.rand() < 0.01:
            noise = np.random.randn(n) * a.mean() * 0.01
            a = np.clip(a + noise, 1e-12, None)
            a *= S / a.sum()
    return best_seq, best_score

def lp_refine(a, deadline):
    n = len(a)
    S = np.sqrt(2.0 * n)
    # Build convolution matrix A (2n-1 rows)
    rows = 2 * n - 1
    # Build A as dense matrix; each column j: a padded at start index j
    A = np.zeros((rows, n), dtype=float)
    for j in range(n):
        A[j:j+n, j] = a  # a placed starting at row index j
    # Solve LP
    b = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [cp.sum(b) == S, A @ b <= M]
    prob = cp.Problem(cp.Minimize(M), constraints)
    # Solve with ECOS
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    except Exception as e:
        print("LP solve error:", e)
        return a
    if b.value is None:
        print("LP did not find solution")
        return a
    b_opt = np.maximum(b.value, 0.0)  # ensure nonneg
    # Compute new max conv
    conv = full_convolution(b_opt)
    new_max = conv.max()
    print(f"[LP refine] old_max={full_convolution(a).max():.6f} new_max={new_max:.6f}")
    # If improvement, return b_opt else original a
    if new_max + 1e-12 < full_convolution(a).max():
        return b_opt
    else:
        return a

"""
Hybrid search for the step‑function minimising
    2·n·max(conv(a,a))/sum(a)^2  ==  max(conv(a,a))

Algorithm
---------
1.  Multi‑start multiplicative Adam on the log‑amplitudes.
    * The loss is a smooth approximation of the max‑convolution:
      softmax over the K largest convolution entries.
    * Adam is performed on `log(a)` (ensuring positivity) and the
      vector is re‑normalised after each update to keep
      Σa = √(2·n).
    * Learning‑rate, softmax temperature and K are annealed.

2.  Optional LP refinement (only for moderate lengths ≤ 3000):
    For a fixed vector *f* the convolution *f*b* is linear in *b*.
    We solve the linear program
        minimise   M
        subject to A·b ≤ M ,   Σb = √(2·n) ,   b ≥ 0
    where A is the (2n‑1)×n convolution matrix of the current *f*.
    The optimal *b* yields a strictly smaller max‑convolution.
    A few alternating LP steps are performed after the Adam phase.

The routine spends the allocated wall‑clock time on many independent
restarts with different lengths.  The best sequence found is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------
def _full_conv(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT (full length)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Return the maximal entry of the self‑convolution of a."""
    return _full_conv(a).max()


# ----------------------------------------------------------------------
# Gradient (soft‑max over top‑K shifts)
# ----------------------------------------------------------------------
def _grad_topk(a: np.ndarray,
               conv: np.ndarray,
               topk: int,
               alpha: float) -> np.ndarray:
    """
    Weighted sub‑gradient of the soft‑max of the top‑K convolution entries.
    For each selected shift k the contribution to g_i is
        2·w_k·a_{k‑i}
    where w_k are soft‑max weights (sum to 1).
    """
    n = a.shape[0]
    # select the K largest shifts
    if conv.size <= topk:
        shifts = np.arange(conv.size)
        weights = np.full_like(shifts, 1.0 / conv.size, dtype=float)
    else:
        idx = np.argpartition(conv, -topk)[-topk:]
        shifts = idx
        # soft‑max across the selected values
        vals = conv[shifts]
        vmax = vals.max()
        exps = np.exp(alpha * (vals - vmax))
        weights = exps / exps.sum()

    g = np.zeros_like(a)
    for k, wk in zip(shifts, weights):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start > i_end:
            continue
        i = np.arange(i_start, i_end + 1, dtype=int)
        j = k - i                     # partner index
        # contribution 2·wk·a_j
        g[i] += 2.0 * wk * a[j]
    return g

# Adam optimiser on log‑amplitudes
# ----------------------------------------------------------------------
def _optimize_one_start(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 25000,
                        topk_start: int = 3,
                        topk_end: int = 30) -> tuple[np.ndarray, float]:
    """Adam on log‑a with soft‑max surrogate. Returns best vector & score."""
    a = a0.copy()
    S = np.sqrt(2.0 * n)

    # optimisation state
    log_a = np.log(np.maximum(a, 1e-16))
    m = np.zeros_like(a)          # 1st moment
    v = np.zeros_like(a)          # 2nd moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.3                     # initial learning rate

    # bookkeeping
    best_seq = a.copy()
    best_score = float('inf')
    best_surrogate = float('inf')
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # current convolution and its maximum
        conv = _full_conv(a)
        cur_max = conv.max()

        # monitor improvement of the surrogate (the true max)
        if cur_max < best_surrogate - 1e-12:
            best_surrogate = cur_max
            try:
                ext = evaluate_sequence(a.tolist())
                if ext < best_score:
                    best_score = ext
                    best_seq = a.copy()
                    print(f"[improve] it={it:5d} n={n:5d} score={best_score:.6f}"
                          f" cur_max={cur_max:.6f}")
            except Exception:
                pass

        # schedules
        frac = it / max_iters
        top_k = int(topk_start + (topk_end - topk_start) * min(1.0, 2 * frac))
        alpha = 0.5 + 4.5 * min(1.0, 2 * frac)          # 0.5 → 5.0
        lr = lr0 * (1.0 - frac)                        # linear decay

        # gradient of the soft‑max surrogate
        g = _grad_topk(a, conv, top_k, alpha)

        # Adam update on log‑a
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # back‑to amplitudes and L1‑normalisation
        a = np.exp(log_a)
        total = a.sum()
        if total == 0.0:
            a.fill(S / n)
        else:
            a *= (S / total)

        # occasional jitter to escape tiny plateaus
        if np.random.rand() < 0.012:
            noise = np.random.randn(n) * a.mean() * 0.015
            a = np.clip(a + noise, 1e-16, None)
            a *= S / a.sum()

    return best_seq, best_score

# Linear‑programming refinement (only for modest n)
# ----------------------------------------------------------------------
def _lp_refine(a: np.ndarray, time_left: float) -> np.ndarray:
    """
    Perform a few alternating LP refinements:
        minimise M
        s.t.   A·b ≤ M , Σb = √(2·n) , b ≥ 0
    where A is the convolution matrix of the current vector a.
    Returns the refined vector (or the original one if refinement fails).
    """
    if not _HAS_CVX:
        return a                                         # CVXPY not available

    n = a.shape[0]
    if n > 3000:                                         # keep memory modest
        return a

    S = np.sqrt(2.0 * n)
    rows = 2 * n - 1

    # build the dense convolution matrix A (rows × n)
    # column j contains a shifted by j rows
    A = np.zeros((rows, n), dtype=float)
    for j in range(n):
        A[j:j + n, j] = a

    b = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [cp.sum(b) == S, A @ b <= M]
    prob = cp.Problem(cp.Minimize(M), constraints)

    # we may perform up to three alternating refinements
    for _ in range(3):
        if time.time() + 2.0 > time.time() + time_left:
            break
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
        except Exception:    # pragma: no cover
            break
        if b.value is None:
            break
        b_opt = np.maximum(b.value, 0.0)
        new_max = _max_conv(b_opt)
        old_max = _max_conv(a)
        print(f"[LP] old_max={old_max:.6f} new_max={new_max:.6f}")
        if new_max + 1e-12 < old_max:
            a = b_opt
            # rebuild A for the next refinement
            S = np.sqrt(2.0 * n)          # unchanged
            for j in range(n):
                A[j:j + n, j] = a
        else:
            break
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) yields a score
    lower than the current best bound of 1.503.
    The routine spends the allotted wall‑clock time on many independent
    restarts of the optimiser described above.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5          # tiny safety margin

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # Record any pre‑existing construction (height_sequence_1)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            if cand.ndim == 1 and cand.size > 0:
                sc = evaluate_sequence(cand.tolist())
                best_seq = cand.copy()
                best_score = sc
                print(f"[baseline] existing height_sequence_1 score = {best_score:.6f}"
                      f" length = {cand.size}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Search over several lengths
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 5000, 8000]     # mix of moderate and large sizes
    run_id = 0

    while time.time() < deadline:
        n = int(rng.choice(lengths))

        # initial vector: either a truncation of a known good solution or fresh Dirichlet
        if best_seq is not None and best_seq.size >= n and rng.random() < 0.25:
            a0 = best_seq[:n].copy()
        else:
            a0 = rng.random(n)

        # normalise to Σa = √(2·n)
        a0 = a0 / a0.sum() * np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # 1) Adam optimisation on the soft‑max surrogate
        # --------------------------------------------------------------
        a_opt, score_opt = _optimize_one_start(a0, n, deadline)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[global improve] run={run_id} n={n} score={best_score:.6f}")

        # --------------------------------------------------------------
        # 2) (Optional) LP refinement – cheap because n is moderate
        # --------------------------------------------------------------
        time_remaining = deadline - time.time()
        if time_remaining > 4.0 and best_seq is not None:
            a_ref = _lp_refine(best_seq, time_remaining)
            try:
                sc_ref = evaluate_sequence(a_ref.tolist())
                if sc_ref < best_score:
                    best_score = sc_ref
                    best_seq = a_ref.copy()
                    print(f"[LP improve] run={run_id} refined score={best_score:.6f}")
            except Exception:
                pass

        run_id += 1

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        a = np.full(n, np.sqrt(2.0 * n) / n)
        best_seq = a
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] score={best_score:.6f}")

    print(f"[final] best score = {best_score:.6f}  length = {len(best_seq)}")
    return best_seq.tolist()
