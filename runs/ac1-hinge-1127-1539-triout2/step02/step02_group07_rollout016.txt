"""
Non‑negative step‑function optimizer.

Goal:
    minimise  f(a) = 2 * n * max(conv(a,a)) / (sum(a)**2).

Approach:
* Work with a probability vector p (sum(p)=1).  The objective then becomes
    score(p) = 2 * n * max(conv(p,p)).
* Use an exponentiated‑gradient (mirror‑descent) scheme that directly
  reduces the largest convolution entry.
* In each iteration:
    - Compute the full linear convolution via a fast FFT.
    - Identify the index k where the convolution attains its maximum.
    - The sub‑gradient of the ℓ∞‑norm at this point is
          g_i = 2 * p_{k-i}   for i that actually contribute,
      all other components are zero.
    - Perform a multiplicative update   p_i ← p_i·exp(−η·g_i)
      with a step‑size η that decays like 1/√t.
    - Renormalise p to the probability simplex (simple division by the sum).
    - Occasionally apply a tiny random jitter to escape plateaus.
* A diverse pool of seeds (previous best, power‑law, exponential,
  uniform and Dirichlet draws) is explored.  The very first seed – the
  previously best construction if available – is refined for many more
  iterations, while the remaining seeds get a moderate budget.
* The routine respects a 1000 s wall‑clock limit and returns the best
  sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_conv_fft(p: np.ndarray, L: int) -> np.ndarray:
    """
    Linear convolution of a real vector with itself using FFT.
    Returns a real array of length 2*len(p)-1.
    """
    ft = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(ft * ft, n=L)
    conv = conv[: 2 * p.shape[0] - 1]
    conv[conv < 0] = 0.0                # numerical safety
    return conv


def _subgradient_max(p: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(p,p)) w.r.t. the probability vector p.
    The gradient is taken at the (first) maximal convolution index.
    """
    n = p.shape[0]
    k = int(np.argmax(conv))            # index of the current max

    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    idx = np.arange(i_min, i_max + 1, dtype=int)

    # contribution from the overlapping part
    grad = np.zeros_like(p)
    grad[idx] = 2.0 * p[k - idx]
    return grad


def _normalize(p: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex by scaling."""
    s = p.sum()
    if s == 0.0 or not np.isfinite(s):
        # Re‑initialise to a small random vector if something went wrong
        p = np.abs(np.random.randn(p.shape[0]))
        s = p.sum()
    return p / s


def _random_dirichlet(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet‑like vector (Gamma samples) of length n."""
    x = np.random.gamma(shape=alpha, scale=1.0, size=n)
    return _normalize(x)


def _generate_powerlaw(p_exp: float,
                       sigma: float,
                       max_len: int = 4000,
                       eps: float = 1e-12) -> np.ndarray:
    """
    Truncated power‑law a_i = (i + sigma)^{‑p_exp}.
    The sequence stops when values drop below `eps`.
    """
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p_exp)
    mask = a > eps
    return a[mask]


def _generate_exponential(lam: float,
                          max_len: int = 4000,
                          eps: float = 1e-12) -> np.ndarray:
    """
    Truncated exponential a_i = exp(‑lam·i).
    """
    i = np.arange(max_len, dtype=np.float64)
    a = np.exp(-lam * i)
    mask = a > eps
    return a[mask]


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0):
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises `evaluate_sequence`.  The routine respects the 1000 s
    time limit and returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # a small safety margin

    best_score = float("inf")
    best_seq = None

    # --------------------------------------------------------------
    # 1) Assemble a pool of diverse initial vectors
    # --------------------------------------------------------------
    seeds = []

    # previous best (if present)
    if "height_sequence_1" in globals():
        seeds.append(np.asarray(height_sequence_1, dtype=np.float64))

    # uniform vectors of various lengths
    for n in (500, 1000, 2000, 4000):
        seeds.append(np.ones(n, dtype=np.float64))

    # power‑law families
    for p_exp in np.linspace(0.42, 0.58, 9):
        for sigma in np.logspace(-3.0, 0.3, 12):
            seq = _generate_powerlaw(p_exp, sigma, max_len=4000, eps=1e-12)
            if seq.size >= 200:
                seeds.append(seq)

    # exponential families
    for lam in np.linspace(0.001, 0.006, 10):
        seq = _generate_exponential(lam, max_len=4000, eps=1e-12)
        if seq.size >= 200:
            seeds.append(seq)

    # random Dirichlet‑like seeds with different concentration parameters
    for alpha in (0.2, 0.5, 1.0, 3.0, 10.0):
        seeds.append(_random_dirichlet(4000, alpha))

    # --------------------------------------------------------------
    # 2) Optimize each seed (the first one gets a longer budget)
    # --------------------------------------------------------------
    for seed_idx, base in enumerate(seeds):
        if time.time() >= deadline:
            break

        p = np.asarray(base, dtype=np.float64)
        if p.size == 0:
            continue
        p = np.maximum(p, 0.0)
        if p.sum() == 0.0:
            continue
        p = _normalize(p)

        n = p.shape[0]

        # keep the length modest – the cost of an FFT grows with n
        if n > 4000:
            p = p[:4000]
            n = p.shape[0]
            p = _normalize(p)

        # FFT length (next power of two) – fixed for a given n
        L = _next_pow2(2 * n - 1)

        # maximal number of inner iterations – we stop earlier if out of time
        # First seed (presumably the strongest) receives the biggest budget.
        max_iters = 60_000 if seed_idx == 0 else 12_000

        it = 0
        no_improve = 0
        while time.time() < deadline and it < max_iters:
            it += 1

            # 1) compute convolution and current score
            conv = _full_conv_fft(p, L)
            max_val = conv.max()
            cur_score = 2.0 * n * max_val

            # 2) record improvement
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = p.tolist()
                print(
                    f"Iter {it:5d} | seed {seed_idx} | n={n} | new best {best_score:.6f}"
                )
                no_improve = 0
                # early termination if we manage to beat the target
                if best_score < 1.503:
                    print("Target satisfied – exiting early.")
                    return best_seq
            else:
                no_improve += 1

            # 3) sub‑gradient at the (first) maximal convolution entry
            g = _subgradient_max(p, conv)

            # 4) exponentiated‑gradient update
            eta = 0.30 / np.sqrt(it)          # decreasing step size
            exp_factor = -eta * g
            # avoid overflow/underflow in exp()
            exp_factor = np.clip(exp_factor, -30.0, 30.0)
            p = p * np.exp(exp_factor)

            # 5) re‑normalise onto the simplex (simple ℓ1 scaling)
            p = _normalize(p)

            # 6) occasional random jitter to help escape plateaus
            if np.random.rand() < 0.015:
                idx = np.random.randint(n)
                delta = (np.random.rand() - 0.5) * 0.02 * p[idx]
                p[idx] = max(p[idx] + delta, 0.0)
                p = _normalize(p)

            # 7) stop early if we have stagnated for a while
            if no_improve >= 10_000:
                break

        # end of optimisation for this seed

    # --------------------------------------------------------------
    # 3) Return the best sequence found
    # --------------------------------------------------------------
    print("Final best score:", best_score, "length:", len(best_seq) if best_seq else None)
    return best_seq if best_seq is not None else []  # empty list signals failure
