    a = a_initial.copy()
    best_a = a.copy()
    best_score = _compute_score(a)
    rng = np.random.default_rng(seed)
    n = len(a)
    start = time.time()
    elapsed = 0.0
    iteration = 0
    T0 = start_temp
    # schedule: T = T0 * (end_temp/T0)^(elapsed / max_time)
    while time.time() < start + max_time:
        iteration += 1
        # generate candidate by moving mass

        a_new[j] += delta
        # Ensure non-negative automatically
        # Evaluate
        score_new = _compute_score(a_new)
        # Accept condition
        elapsed = time.time() - start
        T = T0 * (end_temp / T0) ** (elapsed / max_time)

"""
propose_candidate
-----------------
A two‑stage optimiser for the step‑function problem.

Stage 1 – Adam on a smooth‑max surrogate (the same idea as the reference
implementation, but run longer and with a slightly more aggressive
schedule).

Stage 2 – Simulated‑annealing fine‑tuning.  The current vector is
perturbed by moving a small amount of mass from one index to another
while preserving the L1‑norm.  The Metropolis rule with a temperature
that decays to ~1e‑7 accepts occasional uphill moves, allowing us to
escape the basin reached by Adam.

Both stages keep the sequence non‑negative and projected onto the
simplex {x≥0, Σx = √(2 n)}.  The best vector seen during the whole run is
returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _compute_score(a: np.ndarray) -> float:
    """
    Fast computation of the objective 2·n·max(conv(a,a)) / (sum(a)**2).

    The input is assumed to be a non‑negative 1‑D NumPy array.
    """
    n = a.shape[0]
    nfft = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))


def project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex
        { x : x_i ≥ 0 , Σ_i x_i = target_sum }.
    Implementation of Duchi et al. (ICML 2008).
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(len(v)) + 1  # 1‑based indexing
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises the provided
    evaluation function.  The routine runs Adam for ~55 % of the time
    and a simulated‑annealing local search for the remainder.
    The best sequence discovered is returned as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin (a few seconds for the host to collect the result)
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 0. Determine length n and target L1‑norm (√(2 n))
    # ------------------------------------------------------------------
    try:

        init_seq = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        n = init_seq.shape[0]
    except Exception:
        n = 5600                     # fallback size (close to the previous best)
    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # 1. Random initialisation (power‑law + optional supplied start)
    # ------------------------------------------------------------------
    best_a = None
    best_score = np.inf

    # use the supplied construction if it exists
    try:
        a0 = np.maximum(init_seq, 0.0)
        a0 = a0 * target_sum / a0.sum()
        score0 = _compute_score(a0)
        best_a = a0.copy()
        best_score = score0
    except Exception:
        pass

    # a few random power‑law candidates
    for _ in range(40):
        if time.time() >= deadline:
            break
        p = rng.uniform(0.30, 0.80)          # exponent
        offset = rng.uniform(0.5, 3.5)       # shift to avoid division by zero
        a = 1.0 / np.power(np.arange(n, dtype=np.float64) + offset, p)
        a = a * target_sum / a.sum()
        sc = _compute_score(a)
        if sc < best_score:

            best_a = a.copy()

    if best_a is None:
        best_a = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _compute_score(best_a)

    # ------------------------------------------------------------------
    # 2. Adam optimisation on the soft‑max surrogate
    # ------------------------------------------------------------------
    a = best_a.copy()                # current iterate
    nfft_conv = _next_pow2(2 * n - 1)          # length for a*a
    nfft_corr = _next_pow2(3 * n - 2)          # length for w ⊛ a_rev (correlation)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    lr_min = 0.001
    lr_decay = 0.9994

    gamma_max = 400.0          # a bit larger than in the reference code
    gamma_inc = 1.004

    adam_deadline = start_time + 0.55 * budget_s
    iteration = 0

        af = np.fft.rfft(a, n=nfft_conv)

        i = rng.integers(n)
        # ensure j ≠ i

            j += 1

        a_new = a.copy()

        # ---- evaluate candidate -------------------------------------------
        score_new = _compute_score(a_new)

        # ---- temperature according to elapsed SA time --------------------
        elapsed_sa = time.time() - sa_start

        accept = False

        if score_new < best_score:

            a = project_onto_simplex(a + jitter, target_sum)
            cur_score = _compute_score(a)

        if time.time() >= sa_deadline:
            break

    # 4. Very fine gradient polishing (high γ, tiny fixed step) – optional
    # ------------------------------------------------------------------

    fine_iters = 1500

    # ------------------------------------------------------------------
