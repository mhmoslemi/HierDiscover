"""
Propose a step‑function (non‑negative sequence) that minimizes the
`evaluate_sequence` objective.

Algorithm (≈ 1000 s budget)
--------------------------------
1. Several vector lengths are tried (≈1500–6500).  For each length a
   random initial shape is generated:
        * power‑law   a_i ∝ (i+1)^{-p}
        * exponential a_i ∝ exp(-λ·i)
        * combined    a_i ∝ (i+1)^{-p}·exp(-λ·i)
   The L¹‑norm is projected onto the simplex
   Σa_i = √(2·n).

2. **Annealed projected gradient descent** on a smooth soft‑max
   approximation of `max(conv(a,a))`:
        * β (temperature) starts at 1.0 and is multiplied by 1.2 every
          100 iterations up to 1024.
        * Gradient of the soft‑max is computed with FFTs.
        * Momentum (μ=0.9) and an adaptive step size
          `lr ≈ target_sum/(‖g‖·(β+1))` are used.
        * After each iteration the iterate is projected back onto the
          simplex.

   The best iterate seen during this phase is returned.

3. **Stochastic local search** (simulated‑annealing style) refines the
   result while any remaining time is left:
        * Randomly transfer a fraction of mass from one index to one
          or two other indices (preserving the total sum).
        * Moves are accepted if they improve the exact objective or,
          with probability exp(-(Δ)/T), if a temperature T (linearly
          annealed) permits.
        * The best sequence seen is stored.

4. The global best across all lengths/restarts is returned (trailing
   zeros are stripped).  If no candidate is found a uniform sequence
   is used as a safe fallback.

All heavy work (convolution, gradient) uses FFTs, and the code
periodically prints progress and the current best bound.
"""

import time
import numpy as np

# Basic helpers

def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """Fast version of the competition objective (no safety checks)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    { x | x_i ≥ 0, Σ x_i = s }.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Gradient of the soft‑max approximation of max(conv(a,a))
# ------------------------------------------------------------

def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Return (g, b) where
        b = a*a (full convolution, length 2·n‑1)
        g = ∇ (log‑sum‑exp_{β}(b)) = 2·(a ⋆ p)  with p_k ∝ exp(β·b_k)
    Both are computed with FFTs for speed.
    """
    n = a.shape[0]

    # ---- b = a*a -------------------------------------------------------
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ---- soft‑max probabilities p ---------------------------------------
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)                # shape (2·n‑1,)

    # ---- gradient g = 2·(a ⋆ p) ---------------------------------------
    # cross‑correlation a ⋆ p = central part of conv(a_rev, p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1        # = 3·n‑2
    size2 = 1 << ((conv_len).bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1 : n - 1 + n]   # length n
    return g, b

# ------------------------------------------------------------

def _gradient_descent_anneal(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the soft‑max approximation of the
    maximum convolution.  Uses momentum and an annealing schedule for β.
    Returns the best iterate seen before `deadline`.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 1.0
    beta_max = 1024.0
    beta_mul = 1.2

    mu = 0.9                                   # momentum coefficient
    v = np.zeros_like(a)                       # velocity

    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        # gradient of soft‑max at current β
        g, _ = _gradient_softmax(a, beta)

        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            # step size inversely proportional to β and ‖g‖
            lr = (target_sum * 0.3) / (norm_g * (beta + 1.0))
            v = mu * v - lr * g                 # Nesterov‑style momentum
            a = a + v

        # simplex projection (keeps sum and non‑negativity)
        a = _project_to_simplex(a, target_sum)

        # occasional tiny random jitter (helps escape plateaus)
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005 / (beta + 1.0), size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # evaluate exact objective and store the best vector
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] GD improved n={n} β={beta:.2f} val={best_val:.6f}")

        # anneal β
        if it % 100 == 0 and beta < beta_max:
            beta = min(beta * beta_mul, beta_max)

        it += 1

    return best_a

def _local_search(a_start: np.ndarray, deadline: float):
    """
    Improves `a_start` by repeatedly moving a random fraction of mass
    from one index to one or two other indices.  Moves are accepted
    according to a temperature schedule that linearly cools to zero.
    Returns the best sequence found and its score.
    """
    a_cur = a_start.copy()
    cur_val = _score(a_cur)
    best_a = a_cur.copy()
    best_val = cur_val

    n = a_cur.size
    rng = np.random.default_rng()
    start_time = time.time()
    # initial temperature proportional to the current value
    T0 = cur_val * 0.02 + 1e-9

    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        if total <= 0:
            break
        # linear cooling: T → 0 at the end of the interval
        T = T0 * max(0.0, 1.0 - elapsed / total)

        # propose a new candidate by moving mass
        a_new = a_cur.copy()
        i = rng.integers(0, n)
        if a_new[i] < 1e-12:
            continue

        if rng.random() < 0.8:                     # move to a single destination
            j = rng.integers(0, n)
            if i == j:
                continue
            max_frac = 0.15
            frac_move = rng.random() * max_frac
            delta = a_new[i] * frac_move
            a_new[i] -= delta
            a_new[j] += delta
        else:                                      # split to two destinations
            j = rng.integers(0, n)
            k = rng.integers(0, n)
            if len({i, j, k}) < 3:
                continue
            max_frac = 0.12
            frac_move = rng.random() * max_frac
            delta = a_new[i] * frac_move
            split = rng.random()
            a_new[i] -= delta
            a_new[j] += delta * split
            a_new[k] += delta * (1.0 - split)

        # enforce non‑negativity (tiny numerical errors possible)
        a_new = np.maximum(a_new, 0.0)

        new_val = _score(a_new)
        Δ = new_val - cur_val

        if Δ < 0 or rng.random() < np.exp(-Δ / (T + 1e-12)):
            # accept
            a_cur = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[{time.time():.1f}] LS improved = {best_val:.6f}")

    return best_a, best_val

            p = rng.uniform(0.3, 0.9)
            lam = rng.uniform(0.0, 0.01)
            a0 = (np.arange(1, n + 1, dtype=np.float64) ** (-p)) * \
                 np.exp(-lam * np.arange(n, dtype=np.float64))

        # scale to the required ℓ¹‑norm and add a small jitter
        a0 *= target_sum / a0.sum()
        a0 += rng.normal(scale=target_sum * 0.001, size=n)
        a0 = np.maximum(a0, 0.0)
        a0 = _project_to_simplex(a0, target_sum)

        # Decide how much time to spend on this restart
        remaining = deadline - time.time()
        if remaining < 8.0:               # not enough time for another full run
            break
        # give up to 25 s to the gradient phase, but never more than half the remaining budget
        grad_deadline = time.time() + min(25.0, remaining * 0.55)

        # ----- gradient descent phase -----
        a_opt = _gradient_descent_anneal(a0, grad_deadline)

        # ----- local refinement (uses the whole remaining budget) -----
        if time.time() < deadline:
            a_opt, final_val = _local_search(a_opt, deadline)
        else:
            final_val = _score(a_opt)

        # ----- keep global best -----
        if final_val < best_score:
            best_score = final_val
            best_seq = a_opt.copy()
            print(f"** NEW GLOBAL BEST ** n={n} init={init_kind} score={best_score:.6f}")

    # ----- fallback (should never be needed) -----
    if best_seq is None:
        n = 2000
        a = np.ones(n, dtype=np.float64)
        a *= np.sqrt(2.0 * n) / n
        best_seq = a

    # Trim trailing zeros (just in case) and return as a Python list
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    return best_seq.tolist()
