"""
Non‑negative step‑function optimizer (soft‑max weighted mirror descent).

The evaluation metric can be written, after normalising the vector to
a probability simplex (∑a_i = 1), as

    score = 2·n·max(conv(a,a))                     (1)

where `conv` is the full linear convolution.
The algorithm proceeds as follows:

1.  Build a diverse pool of seed vectors (uniform, power‑law,
    exponential, Dirichlet, and the previously found best).
2.  For each seed run a mirror‑descent (exponentiated‑gradient) loop
    on the simplex.
    *  At iteration `t` we compute the convolution `c = conv(p,p)`.
    *  Convert `c` into a probability distribution `w` by a
       soft‑max with temperature `α_t`.  When `α_t` is small the update
       reacts to the *average* of the convolution, and as `α_t` grows
       it focuses on the largest entries – a smooth approximation of the
       ℓ_∞‑sub‑gradient.
    *  The weighted gradient `g` of `∑ w_k·c_k` w.r.t. `p` can be
       written as a correlation and evaluated with two FFTs:
       `g_j = 2·∑_k w_k·p_{k‑j}`.
    *  We perform a multiplicative (exponentiated) update
       `p ← p·exp(‑η_t·g)` and renormalise the vector onto the simplex.
    *  The step size `η_t` decays like `η₀/√t`;  the temperature
       `α_t` is increased geometrically (`α←α·γ`) so that the algorithm
       gradually switches from smoothing to true max‑minimisation.

3.  Throughout the search we keep the best sequence (lowest score) found
    and periodically apply a tiny random jitter to escape plateaus.
4.  After the optimisation of a seed we trim leading / trailing
    entries that are essentially zero (below 1e‑9); this reduces `n`
    without changing the sum and often improves the objective.

The routine respects the 1000 s time budget and finally returns the best
list of heights discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _generate_powerlaw(p: float, sigma: float, max_len: int = 4000,
                       eps: float = 1e-12) -> np.ndarray:
    """
    Truncated power‑law a_i = (i + sigma)^{-p}.
    Stops when values fall below `eps` or `max_len` is reached.
    """
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p)
    mask = a > eps
    return a[mask]


def _full_conv_fft(p: np.ndarray) -> np.ndarray:
    """
    Compute the full (linear) convolution of a real vector with itself
    using an FFT of size a power of two.
    Returns a real array of length 2*len(p)-1.
    """
    n = p.shape[0]
    L = 1 << ((2 * n - 2).bit_length())          # next power of two
    ft = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(ft * ft, n=L)[: 2 * n - 1]
    conv = np.real(conv)
    conv[conv < 0] = 0.0                           # remove tiny negatives
    return conv


def _gradient_weighted(p: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Weighted gradient of  Σ_k w_k·conv_k(p)  w.r.t a probability vector `p`.
    Implements g_j = 2·Σ_k w_k·p_{k‑j} using a single FFT‑based correlation.
    """
    n = p.shape[0]
    # reverse p for correlation
    p_rev = p[::-1]
    L = 1 << ((len(w) + len(p_rev) - 1).bit_length())
    ft_w = np.fft.rfft(w, n=L)
    ft_pr = np.fft.rfft(p_rev, n=L)
    corr = np.fft.irfft(ft_w * ft_pr, n=L)        # length L
    start = n - 1                                 # slice that yields size n
    g = 2.0 * corr[start:start + n]
    # Numerical noise can make a few entries slightly negative; clip
    g[g < 0] = 0.0
    return g


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The routine respects the 1000 s time limit and
    returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0          # safety margin

    best_score = float("inf")
    best_seq = None
    max_len = 4000                                    # limit per vector

    # ------------------------------------------------------------------
    # 1) Build a heterogeneous collection of seeds
    # ------------------------------------------------------------------
    seeds = []

    # previously found best (if present)
    if "height_sequence_1" in globals():
        seeds.append(np.asarray(height_sequence_1, dtype=np.float64))

    # uniform vectors of several sizes
    for n in (500, 1000, 2000, 3000, 4000):
        seeds.append(np.ones(n, dtype=np.float64))

    # random Dirichlet-like vectors
    for n in (500, 1000, 2000, 3000, 4000):
        seeds.append(np.random.dirichlet(np.ones(n)))

    # power‑law families
    for p_exp in np.linspace(0.40, 0.70, 7):
        for sigma in np.logspace(-3.0, 0.5, 8):
            seq = _generate_powerlaw(p_exp, sigma, max_len=max_len, eps=1e-12)
            if seq.size >= 200:

    # exponential decay families
    for lam in np.linspace(5e-4, 5e-3, 8):
        i = np.arange(max_len, dtype=np.float64)
        seq = np.exp(-lam * i)
        seq = seq[seq > 1e-12]
        if seq.size >= 200:

    # ------------------------------------------------------------------
    # 2) Optimise each seed using soft‑max weighted mirror descent
    # ------------------------------------------------------------------

        n = p.shape[0]

        alpha_factor = 1.001     # geometric increase per iteration

            w = np.exp(alpha * (conv - max_val))

            w = w / w_sum

            eta = eta0 / np.sqrt(it)

            sum_p = p.sum()

                p = np.abs(np.random.randn(n))
                sum_p = p.sum()
            p = p / sum_p

            # 7) occasional tiny random perturbation to escape plateaus
            if np.random.rand() < 0.01:
                idx = np.random.randint(n)
                delta = (np.random.rand() - 0.5) * 0.01 * p[idx]
                p[idx] = max(p[idx] + delta, 0.0)
                p = p / p.sum()

            # 8) slowly sharpen the soft‑max (focus on the true maximum)
            alpha *= alpha_factor
            if alpha > 1000.0:               # avoid overflow in exp
                alpha = 1000.0

        # ------------------------------------------------------------------
        # 2b) post‑process: trim negligible tails (they increase n without benefit)
        # ------------------------------------------------------------------
        eps_trim = 1e-9
        nz = np.where(p > eps_trim)[0]
        if nz.size:

            conv_trim = _full_conv_fft(p_trim)
            score_trim = 2.0 * p_trim.size * np.max(conv_trim)
            if score_trim < best_score:

                best_seq = p_trim.tolist()
                print(f"Trimmed improvement | n={p_trim.size} | score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Return the best sequence found
    # ------------------------------------------------------------------
    print("Final best score:", best_score, "length:", len(best_seq) if best_seq else None)
    return best_seq if best_seq is not None else []   # empty list signals failure
