"""
Propose a non‑negative step‑function (a list of heights) that minimises

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

The objective reduces to ``max(conv)`` when we keep the L1‑norm fixed to
√(2 n).  The algorithm follows three ideas:

1. **Large search space** – unless a global ``height_sequence_1`` is
   supplied we use a fairly large length (≈10 000).  More degrees of
   freedom give a lower theoretical bound (the average of the
   convolution is ≈1).

2. **Projected Adam on a smooth‑max surrogate** – we minimise a
   soft‑max approximation of the maximum convolution.  The gradient of
   the surrogate is obtained by a cheap correlation via FFTs.  After each
   Adam step the vector is projected onto the simplex
   {x ≥ 0, ∑x = √(2n)}.  Periodic jitter and optional symmetry enforcement
   keep the iterates from stagnating.

3. **Fine‑tuning** – after the main phase we raise the soft‑max
   temperature to a large value (≈500) and use a tiny constant step
   size to polish the best solution.

Multiple restarts and a simple “no‑improvement” reset are used to
explore the landscape within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ *x*."""
    return 1 << (x - 1).bit_length()


def _compute_score(a: np.ndarray) -> float:
    """
    Fast version of the provided ``evaluate_sequence`` (no safety checks).
    a must be non‑negative.  Returns 2 n·max(conv(a,a))/sum(a)²
    """
    n = a.shape[0]
    nfft = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))


def project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of *v* onto the simplex
        {x_i ≥ 0, ∑_i x_i = target_sum }.
    Implementation of Duchi‑et‑al. (ICML 2008).
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(len(v)) + 1  # 1‑based indexing
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _make_symmetric(v: np.ndarray) -> np.ndarray:
    """Force exact symmetry (v_i = v_{n‑1‑i})."""
    return 0.5 * (v + v[::-1])

# Main optimisation routine
# ----------------------------------------------------------------------

        grad = 2.0 * corr_full[n - 1 : n - 1 + n]   # length‑n slice

        _maybe_update_best(a_new, score_new)

        # ---- keep or replace the current iterate --------------------------------
        a = a_new

        # ---- schedule updates -------------------------------------------------

        lr = max(lr * lr_decay, lr_min)

        # ---- restart if stagnation for many iterations -------------------------
        if iteration - last_improve_iter > 5000:

            rand = np.random.rand(n)
            rand = rand / rand.sum() * target_sum

            a = project_onto_simplex(a, target_sum)

            step_counter = 0
            last_improve_iter = iteration

    # ------------------------------------------------------------------

        a = best_a.copy()

        conv = np.fft.irfft(af * af, n=nfft_conv)[: 2 * n - 1]
