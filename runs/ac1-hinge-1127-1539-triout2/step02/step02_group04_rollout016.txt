"""
Propose a step‑function (non‑negative sequence) with a low value of
evaluate_sequence().  The algorithm consists of three phases:

1️⃣  A dense parametric sweep.  Families of analytic shapes (power‑law,
    exponential, mixed, plus random Dirichlet draws) are sampled for a
    variety of lengths.  Every candidate is normalised to unit ℓ¹‑norm
    (the objective is scale‑free) and evaluated.  The best one so far
    is kept.

2️⃣  Refined optimisation of the best candidate with an Adam‑style
    projected stochastic sub‑gradient method.  The sub‑gradient of
    maxₖ (x * x)[k] is obtained from the active (maximising) convolution
    indices, an adaptive Adam update is applied and the iterate is
    projected onto the probability simplex.  Random jitter and occasional
    restarts help to escape plateaus.

3️⃣  Length growth.  Every few thousand iterations a block of tiny
    entries (total mass ≈ 10⁻¹²) is appended, allowing the total mass to
    spread further and (potentially) lowering n·max conv.

The routine respects a 1000 s wall‑clock budget and returns the best
sequence found before the time limit expires.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = 1}."""
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D vector")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient(x: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of g(x) = maxₖ (x * x)[k] with respect to x (on the simplex).
    All active convolution indices are used.
    """
    n = x.shape[0]
    conv = np.convolve(x, x)          # length 2n‑1
    m = conv.max()
    # indices where the maximum is attained (numerical tolerance)
    active = np.where(conv >= m - 1e-14)[0]
    g = np.zeros_like(x)
    for k in active:
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        i = np.arange(i_min, i_max + 1, dtype=int)
        j = k - i
        # derivative of conv[k] w.r.t. x_i is x_j (the factor 2 is absorbed later)
        g[i] += x[j]
    return g


def _evaluate(seq: np.ndarray) -> float:
    """Thin wrapper around the supplied evaluate_sequence (list‑oriented)."""
    try:
        return evaluate_sequence(seq.tolist())
    except Exception:
        return float('inf')


def _generate_powerlaw(p: float, sigma: float,
                       max_len: int = 5000, eps: float = 1e-12) -> np.ndarray:
    """a_i = (i+σ)^{‑p}, truncated when a_i < eps."""
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p)
    return a[a > eps]


def _generate_exponential(lam: float,
                          max_len: int = 5000, eps: float = 1e-12) -> np.ndarray:
    """a_i = exp(-λ i), truncated when a_i < eps."""
    i = np.arange(max_len, dtype=np.float64)
    a = np.exp(-lam * i)
    return a[a > eps]


def _generate_mixed(p: float, sigma: float, lam: float,
                    max_len: int = 5000, eps: float = 1e-12) -> np.ndarray:
    """a_i = (i+σ)^{‑p}·exp(‑λ i), truncated when a_i < eps."""
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p) * np.exp(-lam * i)
    return a[a > eps]


def _generate_dirichlet(n: int) -> np.ndarray:
    """Sample an i.i.d. Dirichlet(1,…,1) vector of length n."""
    rnd = np.random.exponential(size=n)
    return rnd / rnd.sum()

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list:
    """
    Search for a non‑negative step function that minimises
    evaluate_sequence().  The function respects a 1000 s wall‑clock budget
    and returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    best_score = float('inf')
    best_seq = None

    # ------------------------------------------------------------------
    # 0.  Use the globally supplied construction if present.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            sc = _evaluate(np.array(height_sequence_1, dtype=float))
            if np.isfinite(sc):
                best_score = float(sc)
                best_seq = list(height_sequence_1)
                print(f'Initial global sequence score = {best_score:.6f}')
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Broad parametric sweep (including random Dirichlet samples)
    # ------------------------------------------------------------------
    print('=== Phase 1 – parametric sweep ===')
    max_len = 8000                     # allow a fairly large support
    eps = 1e-12

    # ----- power‑law family -----
    p_vals = np.linspace(0.20, 0.80, 31)          # finer grid around 0.5
    sigma_vals = np.logspace(-6, 2, 50)         # many offsets
    for p in p_vals:
        for sigma in sigma_vals:
            if time.time() >= deadline:
                break
            seq = _generate_powerlaw(p, sigma, max_len, eps)
            if seq.size < 2:
                continue
            seq = seq / seq.sum()
            score = _evaluate(seq)
            if np.isfinite(score) and score < best_score:
                best_score = float(score)
                best_seq = seq.tolist()
                print(f'new best PL   p={p:.3f} σ={sigma:.5g} len={len(seq)} score={best_score:.6f}')
        if time.time() >= deadline:
            break

    # ----- exponential family -----
    lam_vals = np.logspace(-5, 0, 40)   # 1e‑5 … 1
    for lam in lam_vals:
        if time.time() >= deadline:
            break
        seq = _generate_exponential(lam, max_len, eps)
        if seq.size < 2:
            continue
        seq = seq / seq.sum()
        score = _evaluate(seq)
        if np.isfinite(score) and score < best_score:
            best_score = float(score)
            best_seq = seq.tolist()
            print(f'new best EXP  λ={lam:.5e} len={len(seq)} score={best_score:.6f}')

    # ----- mixed power‑law / exponential -----
    for p in (0.35, 0.45, 0.55, 0.65):
        for sigma in (1e-4, 1e-2, 1e-1, 1.0):
            for lam in (1e-4, 5e-4, 1e-3, 5e-3, 1e-2):
                if time.time() >= deadline:
                    break
                seq = _generate_mixed(p, sigma, lam, max_len, eps)
                if seq.size < 2:
                    continue
                seq = seq / seq.sum()
                score = _evaluate(seq)
                if np.isfinite(score) and score < best_score:
                    best_score = float(score)
                    best_seq = seq.tolist()
                    print(f'new best MIX p={p:.2f} σ={sigma:.5g} λ={lam:.5g} len={len(seq)} score={best_score:.6f}')

    # ----- random Dirichlet samples -----
    for n in (2000, 4000, 6000, 8000):
        for _ in range(80):                     # modest number of draws
            if time.time() >= deadline:
                break
            seq = _generate_dirichlet(n)
            score = _evaluate(seq)
            if np.isfinite(score) and score < best_score:
                best_score = float(score)
                best_seq = seq.tolist()
                print(f'new best DIR n={n} score={best_score:.6f}')
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 2.  Refined optimisation – Adam projected sub‑gradient descent
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform vector of a moderate size
        n0 = 2000
        best_seq = [1.0] * n0
        best_score = _evaluate(np.full(n0, 1.0 / n0))
        print('fallback to uniform sequence, score =', best_score)

    x = np.array(best_seq, dtype=np.float64)
    x = x / x.sum()                     # lie on simplex

    # Adam hyper‑parameters
    lr = 0.25
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    iteration = 0
    grow_interval = 3500               # how often we append tiny entries
    max_total_len = 12000              # hard cap on length

    print('=== Phase 2 – Adam projected sub‑gradient descent ===')
    while time.time() < deadline:
        iteration += 1

        # --- sub‑gradient of the max‑convolution term ---
        g = _subgradient(x)

        # --- Adam update (un‑scaled) ---
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # bias‑correction
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # step size per coordinate
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # --- project back onto simplex ---
        y = x - step
        x = _simplex_projection(y)

        # --- occasional random jitter (helps escape plateaus) ---
        if np.random.rand() < 0.015:
            idx = np.random.randint(x.shape[0])
            delta = (np.random.rand() - 0.5) * 0.2 * x[idx]   # up to ±10 %
            x[idx] = max(0.0, x[idx] + delta)
            x = x / x.sum()

        # --- length growth (adds a few 1e‑12‑mass entries) ---
        if iteration % grow_interval == 0 and x.shape[0] < max_total_len:
            extra = np.full(250, 1e-12, dtype=np.float64)
            x = np.concatenate([x, extra])
            x = x / x.sum()
            m = np.concatenate([m, np.zeros_like(extra)])
            v = np.concatenate([v, np.zeros_like(extra)])

        # --- evaluate current candidate ---
        cur_score = _evaluate(x)
        if np.isfinite(cur_score) and cur_score < best_score:
            best_score = float(cur_score)
            best_seq = x.tolist()
            print(f'iteration {iteration:6d} → improved score = {best_score:.6f}')
            # early exit if we already beat the target bound
            if best_score < 1.503:
                print('Target bound reached – terminating early.')
                break

        # --- restart if stagnant for a while ---
        if iteration % 8000 == 0:
            n = x.shape[0]
            rnd = np.random.rand(n)
            x = 0.6 * x + 0.4 * rnd
            x = x / x.sum()
            m = np.zeros_like(x)
            v = np.zeros_like(x)

    print('final best score:', best_score, 'length:', len(best_seq))
    return best_seq
