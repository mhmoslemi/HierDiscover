def _project_simplex(v, s):
    # using algorithm from https://arxiv.org/pdf/1101.6081.pdf
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _gradient_step_smooth(a, lr, alpha):
    conv = _conv_full(a)
    w = _smooth_weights(conv, alpha)  # length 2n-1
    # compute gradient
    n = a.shape[0]
    w_rev = w[::-1]
    # compute conv2 via FFT
    conv_len = n + (2*n-1) -1
    fft_len = 1 << (conv_len-1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    conv2 = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    # gradient slice
    grad = 2.0 * conv2[2*n-2 - np.arange(n)]
    # step
    a_new = a - lr * grad
    # project
    a_new = _project_simplex(a_new, np.sqrt(2.0 * n))
    return a_new, conv.max()

def _optimize_local_smooth(a0, deadline, rng):
    a = a0.copy()
    n = len(a)
    S = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_score = np.max(np.convolve(a,a))
    lr = 0.2
    lr_decay = 0.9995
    alpha = 1.0
    alpha_mul = 1.001
    last_improve = 0
    iter_count = 0
    while time.time() < deadline:
        iter_count += 1
        # gradient step
        a, cur_max = _gradient_step_smooth(a, lr, alpha)
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            last_improve = iter_count
        # sometimes random swap
        if rng.random() < 0.05:
            # random transfer: choose i, j randomly, move epsilon fraction
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0:
                delta = 0.02 * a[i] * rng.random()  # some fraction
                a[i] -= delta
                a[j] += delta
                # reproject to simplex
                a = _project_simplex(a, S)
        # schedule update
        lr *= lr_decay
        alpha *= alpha_mul
        # early stop if no improvement for large number of iters
        if iter_count - last_improve > 2000:
            break
    return best_a, best_score

"""
Propose a candidate step‑function minimizing the max entry of the
self‑convolution (equivalently the evaluation score).

Algorithm
---------
1.   Keep the L¹‑norm fixed to √(2·n) (the optimal normalisation for the
     score) by projecting onto the simplex after each update.
2.   Use a smooth approximation of the infinity‑norm:
       w_i = exp(α·c_i) / Σ_j exp(α·c_j),  where c = a∗a (full convolution).
     For small α the weights are spread, for large α they concentrate on the
     maximal convolution entry.
3.   Compute the gradient of the smoothed objective:
       ∇_j ≈ 2·Σ_i w_{i+j}·a_i .
     This is obtained efficiently via an FFT based convolution.
4.   Perform a projected gradient step with a decaying step size.
5.   Slowly increase α (so the optimisation first sees a smooth landscape,
     then sharpens to the true max‑norm).
6.   Occasionally perform a random “mass‑swap’’ move to help escape plateaus.
7.   Run many short independent restarts with random lengths (≈ 800‑5000)
     and random seeds, optionally seeding from the previously discovered
     best sequence (`height_sequence_1`).  Keep the overall best sequence.

The routine respects the 1000 s time budget, prints progress, and returns the
best sequence as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Geometry helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = s}.
    """
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    # find rho
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based for speed)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using an FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient term g_j = 2 * Σ_i w_{i+j} * a_i.
    Both a (len n) and w (len 2n-1) are real vectors.
    """
    n = a.shape[0]
    # reverse w for correlation
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1  # = 3n-2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    # gradient slice: indices 2n-2 - j
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Smooth max and gradient step
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute soft‑max weights from a convolution vector.
    w_i = exp(alpha*conv_i) / Σ_j exp(alpha*conv_j)
    """
    scaled = alpha * conv
    # numerical stability
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


def _gradient_step(a: np.ndarray, lr: float, alpha: float) -> tuple[np.ndarray, float]:
    """
    Perform one projected gradient step on the smoothed max‑norm.
    Returns the updated sequence and the *exact* maximal convolution entry
    (computed without smoothing).
    """
    n = a.shape[0]
    # 1. full self‑convolution
    conv = _full_convolution(a)

    # 2. smooth weights (soft‑max)
    w = _smooth_weights(conv, alpha)

    # 3. gradient w.r.t. a
    grad = _conv_with_weights(a, w)

    # 4. gradient descent step
    a_new = a - lr * grad

    # 5. project back onto the simplex Σa = √(2n)
    S = np.sqrt(2.0 * n)
    a_new = _project_simplex(a_new, S)

    # 6. evaluate true max for bookkeeping
    true_max = np.max(_full_convolution(a_new))
    return a_new, true_max


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n with L¹‑norm √(2·n).
    If a base_seq is given and long enough, its leading part (with jitter)
    is used; otherwise a Dirichlet/Uniform sample is taken.
    """
    S = np.sqrt(2.0 * n)

    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        a += rng.normal(scale=a.mean() * 0.001, size=n)  # tiny jitter
        a = np.clip(a, 0.0, None)
    else:
        # Dirichlet(1) = uniform over simplex
        a = rng.random(n)

    # normalise to the required sum
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0)
        a_sum = a.sum()
    a = a / a_sum * S
    return a


# ----------------------------------------------------------------------
# Local optimiser
# ----------------------------------------------------------------------


def _optimize_local_smooth(a0: np.ndarray,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Refine a candidate sequence using the smooth‑max gradient descent.
    Returns the best sequence encountered and its maximal self‑convolution
    value (the true objective, not the smoothed surrogate).
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # current best
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    # hyper‑parameters
    lr = 0.3               # initial learning rate
    lr_decay = 0.9995
    alpha = 1.0            # soft‑max temperature (low -> smooth)
    alpha_mul = 1.001      # slow increase to sharpen
    swap_prob = 0.07       # probability of performing a random mass swap
    swap_frac = 0.03       # fraction of source mass moved in a swap

    last_improve_iter = 0
    iter_cnt = 0

    while time.time() < deadline:
        iter_cnt += 1

        # ----- gradient descent step -----
        a, cur_max = _gradient_step(a, lr, alpha)

        # update best if improved
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            last_improve_iter = iter_cnt

        # ----- occasional random mass swap (simulated annealing flavour) -----
        if rng.random() < swap_prob:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0:
                delta = swap_frac * a[i] * rng.random()
                a_candidate = a.copy()
                a_candidate[i] -= delta
                a_candidate[j] += delta
                a_candidate = _project_simplex(a_candidate, S)
                cand_max = np.max(_full_convolution(a_candidate))
                # accept if better or with small probability (temperature = cur_max)
                if cand_max <= cur_max or rng.random() < np.exp(-(cand_max - cur_max) / max(cur_max * 0.05, 1e-12)):
                    a = a_candidate
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                        last_improve_iter = iter_cnt

        # ----- schedule updates -----
        lr *= lr_decay
        alpha *= alpha_mul

        # early stop if stagnated for many iterations
        if iter_cnt - last_improve_iter > 2000:
            break

    return best_a, best_score
