"""
Propose a candidate step‑function using a conditional‑gradient (Frank‑Wolfe)
scheme.

The algorithm works as follows:

1.  Choose a length ``n`` (typically 800‑2500) and initialise a non‑negative
    vector ``f`` with L1‑norm ``sqrt(2·n)``.  If a previously discovered
    sequence is available via the global name ``height_sequence_1`` it is used
    as a warm start.

2.  Repeatedly improve ``f``:
    * Compute the maximal entry ``M`` of the self‑convolution ``f*f``.
    * Build the linear operator ``A`` that maps any vector ``b`` to the
      convolution ``f*b`` (i.e. ``(A·b)_k = Σ_i f_i·b_{k-i}``).
    * Solve the linear programme

          maximise   Σ_j b_j
          subject to A·b ≤ M·1 ,  b ≥ 0 .

      The LP finds a vector ``b`` that packs as much mass as possible under the
      current convolution bound ``M``.
    * Scale the LP solution to the required L1‑norm: ``g = (sqrt(2·n)/Σb)·b``.
    * Replace ``f`` by the midpoint ``(f+g)/2`` (any convex combination would
      preserve the norm).  This step strictly reduces the maximal
      convolution unless a stationary point is reached.
    * Occasionally add a tiny Gaussian jitter to avoid stagnation.

    The loop stops when the time budget for the current restart expires or
    when no progress is observed.

3.  Keep the best sequence (smallest maximal convolution) over many
    independent restarts until the overall time budget (≈1000 s) runs out.
    Finally return the best sequence as a plain ``list[float]`` – this is
    exactly what ``evaluate_sequence`` expects.

The method is entirely deterministic apart from the seeded RNG and
requires only NumPy, SciPy and the provided ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def _conv_max(a: np.ndarray) -> float:
    """
    Return the maximal entry of the (full) self‑convolution of ``a``.
    Uses FFT for speed (O(n log n)).
    """
    n = a.shape[0]
    L = 2 * n - 1
    # size for efficient FFT (next power of two)
    fft_len = 1 << (L - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:L]
    return float(conv.max())


def _build_conv_matrix(f: np.ndarray) -> sparse.csr_matrix:
    """
    Construct the (2n‑1)×n sparse matrix ``A`` such that ``A @ b`` equals the
    convolution ``f * b`` (full, zero‑padded).  Each diagonal of ``A`` holds a
    constant value taken from ``f``.
    """
    n = f.shape[0]
    # For offset k (0 ≤ k < n) we need a diagonal consisting of f[k] repeated n times.
    diag_data = [np.full(n, f[k], dtype=np.float64) for k in range(n)]
    offsets = np.arange(n, dtype=np.int64)  # 0,1,…,n‑1
    A = sparse.diags(diag_data, offsets, shape=(2 * n - 1, n), format="csr")
    return A


def _frank_wolfe_step(f: np.ndarray, M: float, A: sparse.csr_matrix) -> np.ndarray:
    """
    Solve the LP ``max Σ b_j  s.t.  A·b ≤ M,  b≥0`` and return the scaled vector
    ``g`` that has L1‑norm ``sqrt(2·n)``.
    """
    n = f.shape[0]
    # linear objective: maximise Σ b  → minimize -Σ b
    c = -np.ones(n, dtype=np.float64)

    # upper bounds for each convolution entry
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # solve with HiGHS (fast for large sparse problems)
    res = linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=(0, None),
        method="highs",
        options={"presolve": True, "time_limit": 30.0},
    )
    if not res.success:
        # If the solver fails for any reason we simply return the current vector.
        return f.copy()

    b = res.x
    total = b.sum()
    if total == 0.0:
        return f.copy()
    S = np.sqrt(2.0 * n)
    g = b * (S / total)
    return g


def _local_optimize(f0: np.ndarray, deadline: float, rng: np.random.Generator) -> np.ndarray:
    """
    Perform a small number of Frank‑Wolfe iterations starting from ``f0``.
    Stops when the deadline is reached or when progress stalls.
    Returns the best vector encountered (as a NumPy array).
    """
    f = f0.copy()
    n = f.shape[0]
    S = np.sqrt(2.0 * n)

    best_f = f.copy()
    best_val = _conv_max(f)

    # parameters controlling the inner loop
    max_iters = 200
    jitter_prob = 0.03
    progress_tol = 1e-7

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        M = _conv_max(f)
        A = _build_conv_matrix(f)

        # Frank‑Wolfe style update: compute a partner g and move to midpoint
        g = _frank_wolfe_step(f, M, A)

        # convex combination (midpoint) – any λ∈(0,1) works, 0.5 is a good trade‑off
        f_new = 0.5 * (f + g)

        # optional jitter to escape flat regions
        if rng.random() < jitter_prob:
            noise = rng.normal(scale=f_new.mean() * 0.001, size=n)
            f_new = np.clip(f_new + noise, 0.0, None)

        # renormalise (numerical drift may happen)
        f_new = f_new / f_new.sum() * S

        new_val = _conv_max(f_new)

        # keep the best seen so far
        if new_val < best_val - progress_tol:
            best_val = new_val
            best_f = f_new.copy()
        else:
            # no substantial progress – we may still continue to give
            # jitter a chance, but if the value is unchanged for a while we
            # break early.
            if it > 10 and abs(new_val - best_val) < 1e-9:
                break

        f = f_new

    return best_f


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Entry point used by the evaluation harness.

    Executes many short restarts of the Frank‑Wolfe optimiser described above,
    each with a randomly chosen length and a random initial vector.
    The best sequence discovered (smallest value of ``evaluate_sequence``) is
    returned as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0  # safety margin

    # Try to reuse a previously found good sequence, if available.
    base_seq = None
    if "height_sequence_1" in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    attempt = 0

    while time.time() < deadline:
        attempt += 1

        # ------------------------------------------------------------------
        # Choose a problem size.
        # ------------------------------------------------------------------
        if base_seq is not None:
            # keep the size in a sensible range, but allow some variation.
            min_n, max_n = 800, min(3000, int(base_seq.size * 1.5))
        else:
            min_n, max_n = 800, 3000
        n = rng.integers(min_n, max_n + 1)

        # ------------------------------------------------------------------
        # Initialise a feasible vector with the correct L1 norm.
        # ------------------------------------------------------------------
        if base_seq is not None and base_seq.size >= n:
            f0 = base_seq[:n].astype(float).copy()
            # small jitter so that restarts are not identical
            f0 += rng.normal(scale=f0.mean() * 0.001, size=n)
            f0 = np.clip(f0, 0.0, None)
        else:
            f0 = rng.random(n)

        S = np.sqrt(2.0 * n)
        f0 = f0 / f0.sum() * S

        # ------------------------------------------------------------------
        # Run a short Frank‑Wolfe optimisation on this instance.
        # ------------------------------------------------------------------
        slice_deadline = min(time.time() + 30.0, deadline)
        f_opt = _local_optimize(f0, slice_deadline, rng)

        # ------------------------------------------------------------------
        # Evaluate the candidate with the provided black‑box.
        # ------------------------------------------------------------------
        candidate = f_opt.tolist()
        score = evaluate_sequence(candidate)

        if score < best_score:
            best_score = score
            best_seq = candidate
            print(
                f"[global] attempt={attempt} n={n} "
                f"new best external score={score:.6f}"
            )

    # ----------------------------------------------------------------------
    # Safety fallback
    # ----------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0]

    return best_seq
