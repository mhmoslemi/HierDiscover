"""
A fast stochastic‑gradient optimiser for the self‑convolution bound.

Key ideas
---------
* 1️⃣  Broad parametric sweep:  several analytic families are tried –
  linear / quadratic / exponential “U” shapes, power‑law shapes
  a[i] ∝ ((i+δ)*(n‑1‑i+δ))⁻ᵞ and a simple composite
  (1+ p·d)·exp(q·d) with d being the normalised distance from the centre.
  Every candidate is projected onto the ℓ¹‑simplex
  Σaᵢ = √(2·n) (the normalisation used by ``evaluate_sequence``).

* 2️⃣  Adam‑type projected gradient descent on a smoothed max of the
  self‑convolution.  The soft‑max weights wₖ∝exp(bₖ/τ) (b = a*a)
  give a smooth surrogate for the hard max.  The gradient
  ∇ = 2·cross‑correlation(w,a) is computed by a single FFT.
  The direction is orthogonally projected to the plane Σaᵢ = const
  (by subtracting its mean) and an Adam update is applied.
  τ (temperature) and the learning‑rate decay slowly.

* 3️⃣  Escape mechanisms – occasional random pair transfers
  and jitter when the search stalls.

The whole routine respects a 1000 s wall‑clock limit and always returns
the best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures / lambdas)
# ----------------------------------------------------------------------


def fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using an FFT of power‑of‑2 length."""
    n, m = a.shape[0], b.shape[0]
    L = 1 << ((n + m - 1).bit_length())           # next power of two
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)
    return conv[: n + m - 1]


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (length 2·len(a)-1)."""
    return fft_convolve(a, a)


def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector so that its ℓ₁‑norm equals *target*."""
    s = v.sum()
    if s > 0:
        return v * (target / s)
    # fallback – uniform distribution (should never happen)
    return np.full_like(v, target / v.shape[0])


def generate_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
    direction: str = "symmetric",
) -> np.ndarray:
    """
    Produce a simple analytic shape of length *n* and scale it to *target_sum*.

    ``kind``    – {"linear", "quadratic", "exponential"}.
    ``direction`` – {"symmetric", "decreasing", "increasing"}.
    ``param``   – strength of the edge effect (≥0).
    """
    idx = np.arange(n, dtype=np.float64)
    if direction == "symmetric":
        mid = (n - 1) / 2.0
        d = np.abs(idx - mid) / mid                      # ∈[0,1]
        base = d
    elif direction == "decreasing":
        base = 1.0 - idx / (n - 1)                        # ∈[1,0]
    elif direction == "increasing":
        base = idx / (n - 1)                              # ∈[0,1]
    else:
        raise ValueError("unknown direction")

    if kind == "linear":
        raw = 1.0 + param * base
    elif kind == "quadratic":
        raw = 1.0 + param * base ** 2
    elif kind == "exponential":
        raw = np.exp(param * base)
    else:
        raise ValueError("unknown shape kind")

    raw = np.clip(raw, 0.0, None)
    return project_to_sum(raw, target_sum)


def generate_powerlaw_shape(
    n: int,
    target_sum: float,
    gamma: float = 0.25,
    offset: float = 0.5,
) -> np.ndarray:
    """a[i] ∝ ((i+offset)*(n‑1‑i+offset))⁻ᵞ, then ℓ¹‑normalised."""
    idx = np.arange(n, dtype=np.float64)
    denom = (idx + offset) * (n - 1 - idx + offset)
    raw = denom ** (-gamma)
    raw = np.clip(raw, 0.0, None)
    return project_to_sum(raw, target_sum)


def generate_composite_shape(
    n: int,
    target_sum: float,
    p: float = 0.0,
    q: float = 0.0,
) -> np.ndarray:
    """
    (1 + p·d)·exp(q·d) – a flexible “U” shape, d = normalised distance from centre.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                      # ∈[0,1]
    raw = (1.0 + p * d) * np.exp(q * d)
    raw = np.clip(raw, 0.0, None)
    return project_to_sum(raw, target_sum)


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """Transfer a tiny fraction of mass from a random source index to a random target."""
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0.0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_to_sum(a, target_sum)


def jitter(a: np.ndarray, target_sum: float, sigma: float = 1e-3) -> np.ndarray:
    """Add Gaussian noise (clipped) and renormalise."""
    a = a + np.random.randn(a.shape[0]) * sigma
    a = np.clip(a, 0.0, None)
    return project_to_sum(a, target_sum)


# ----------------------------------------------------------------------
# Gradient step (Adam on the smoothed max)
# ----------------------------------------------------------------------


def smoothed_max_grad(
    a: np.ndarray,
    tau: float,
) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max_k (a*a)_k.
    Returns a vector of length len(a) with zero‑mean (i.e. already projected).
    """
    n = a.shape[0]

    # self‑convolution
    conv = conv_self_fft(a)                     # length 2n‑1

    # soft‑max weights
    max_c = conv.max()
    exp_arg = (conv - max_c) / tau
    # subtract max to keep numbers moderate
    w = np.exp(exp_arg - np.max(exp_arg))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w = w / w_sum

    # gradient = 2·cross‑correlation(w, a)
    a_rev = a[::-1]
    grad_full = fft_convolve(w, a_rev)         # length 3n‑2
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]   # slice i = 0…n‑1

    # project to Σaᵢ = const  (removing the mean makes the step stay on the simplex)
    grad -= np.mean(grad)
    return grad


def adam_update(
    a: np.ndarray,
    grad: np.ndarray,
    m: np.ndarray,
    v: np.ndarray,
    t: int,
    lr: float,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, np.ndarray]:
    """One Adam step with projection onto the ℓ¹‑simplex."""
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad ** 2)

    m_hat = m / (1.0 - beta1 ** (t + 1))
    v_hat = v / (1.0 - beta2 ** (t + 1))

    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = a - step                     # note the minus sign (gradient ascent on the smooth max)
    a_new = np.clip(a_new, 0.0, None)    # feasibility
    return a_new, m, v


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.

    The routine runs a quick parametric sweep, then a projected Adam descent
    on a smoothed max of the self‑convolution, with occasional random
    mass‑transfer moves to escape plateaus.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5    # tiny safety margin

    # ------------------------------------------------------------------
    # Choose the length – we stay in the neighbourhood of the known
    # construction (≈ 8k) because the evaluation is essentially length‑invariant.
    # ------------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
    else:
        n = int(np.random.randint(7200, 8800))

    target_sum = np.sqrt(2.0 * n)              # the normalisation used in the paper

    # ------------------------------------------------------------------
    # Phase 1 – broad parametric sweep (including new families)
    # ------------------------------------------------------------------
    best_seq = np.full(n, target_sum / n)      # uniform baseline
    best_score = float("inf")

    print(f"[{time.time():.2f}] Starting parametric sweep (n={n})")
    families = [
        ("symmetric", "linear"),
        ("symmetric", "quadratic"),
        ("symmetric", "exponential"),
        ("decreasing", "linear"),
        ("decreasing", "quadratic"),
        ("decreasing", "exponential"),
        ("increasing", "linear"),
        ("increasing", "quadratic"),
        ("increasing", "exponential"),
    ]

    param_grids = {
        "linear": np.linspace(0.0, 6.0, 31),          # extended range
        "quadratic": np.linspace(0.0, 5.0, 26),
        "exponential": np.linspace(0.0, 5.0, 26),
    }

    # classic families
    for direction, kind in families:
        for p in param_grids[kind]:
            cand = generate_shape(n, target_sum, kind=kind, param=p, direction=direction)
            score = evaluate_sequence(cand.tolist())
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"  New best {direction}/{kind} (p={p:.2f}) → {best_score:.6f}")
            if time.time() >= deadline:
                print("Time exhausted during sweep.")
                return best_seq.tolist()

    # power‑law family
    gammas = np.linspace(0.05, 0.45, 17)
    offsets = [0.5, 1.0, 2.0]
    for gamma in gammas:
        for off in offsets:
            cand = generate_powerlaw_shape(n, target_sum, gamma=gamma, offset=off)
            score = evaluate_sequence(cand.tolist())
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"  Power‑law γ={gamma:.3f}, off={off:.1f} → {best_score:.6f}")
            if time.time() >= deadline:
                print("Time exhausted during power‑law sweep.")
                return best_seq.tolist()

    # composite exponential‑linear family
    ps = np.linspace(0.0, 8.0, 17)
    qs = np.linspace(0.0, 4.0, 17)
    for p in ps:
        for q in qs:
            cand = generate_composite_shape(n, target_sum, p=p, q=q)
            score = evaluate_sequence(cand.tolist())
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"  Composite p={p:.2f}, q={q:.2f} → {best_score:.6f}")
            if time.time() >= deadline:
                print("Time exhausted during composite sweep.")
                return best_seq.tolist()

    # a handful of truly random draws (helps to avoid systematic bias)
    for _ in range(30):
        rand = np.random.rand(n)
        cand = project_to_sum(rand, target_sum)
        score = evaluate_sequence(cand.tolist())
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"  Random init improved → {best_score:.6f}")
        if time.time() >= deadline:
            return best_seq.tolist()

    # ------------------------------------------------------------------
    # Phase 2 – Adam descent on a smoothed max of the convolution
    # ------------------------------------------------------------------
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Adam hyper‑parameters
    lr = 0.03 * (target_sum / n)          # base step ≈ 3 % of the average entry
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # temperature for the soft‑max (start relatively high, cool slowly)
    conv0 = conv_self_fft(a)
    tau = max(0.2 * conv0.max(), 1e-4)

    iteration = 0
    no_improve = 0

    print(f"[{time.time():.2f}] Entering Adam descent (lr={lr:.5e}, tau={tau:.5e})")
    while time.time() < deadline:
        # 1) gradient of the smoothed max
        grad = smoothed_max_grad(a, tau)

        # 2) Adam update (note that ``adam_update`` already projects onto the simplex)
        a_new, m, v = adam_update(a, grad, m, v, iteration, lr,
                                 beta1=beta1, beta2=beta2, eps=eps)

        # 3) enforce the ℓ¹‑norm exactly (tiny numeric drift may appear)
        a_new = project_to_sum(a_new, target_sum)

        a = a_new
        iteration += 1

        # evaluate current candidate
        score = evaluate_sequence(a.tolist())
        if score + 1e-12 < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"Iter {iteration:06d} – improvement: {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # schedule adaptations
        if iteration % 300 == 0:
            lr *= 0.94                 # decay learning rate
            tau *= 0.93                # cool the temperature
            if lr < 1e-7:
                lr = 1e-7
            if tau < 1e-5:
                tau = 1e-5

        # occasional random pair transfer (helps to jump out of flat regions)
        if np.random.rand() < 0.025:
            a = random_pair_transfer(a, target_sum, scale=0.01)

        # if we have been stuck for a while, inject a small jitter
        if no_improve >= 2000:
            a = jitter(a, target_sum, sigma=5e-4)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"Iter {iteration:06d} – jitter reset")

    # ------------------------------------------------------------------
    # Phase 3 – quick polishing with very small random moves
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Descent finished; best score {best_score:.6f}")
    while time.time() < deadline - 0.2:
        a = random_pair_transfer(a, target_sum, scale=0.004)
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"Polish – improvement to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found (plain Python list)
    # ------------------------------------------------------------------
    return best_seq.tolist()
