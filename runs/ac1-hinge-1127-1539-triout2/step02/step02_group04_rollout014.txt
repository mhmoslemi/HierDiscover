def subgradient(x):
    n = x.shape[0]
    conv = np.convolve(x,x)
    maxc = conv.max()
    active = np.where(conv >= maxc - 1e-14)[0]
    g = np.zeros_like(x)
    for k in active:
        i_start = max(0, k-(n-1))
        i_end = min(k, n-1)
        i = np.arange(i_start, i_end+1)
        j = k - i
        g[i] += x[j]
    return g
```

def local_search(x0, max_moves, deadline):
    x = x0.copy()
    n = len(x)
    best_score = compute_score(x)
    best_x = x.copy()
    moves = 0
    while moves < max_moves and time.time() < deadline:
        moves += 1
        i = random.randrange(n)
        j = random.randrange(n)
        if i == j:
            continue
        # amount to move
        delta = 0.001 * x[i] # small fraction
        if delta == 0:
            continue
        # propose move
        x[i] -= delta
        x[j] += delta
        # avoid negative
        if x[i] < 0:
            # revert
            x[i] += delta
            x[j] -= delta
            continue
        cur_score = compute_score(x)
        if cur_score < best_score:
            best_score = cur_score
            best_x = x.copy()
            # keep change
        else:
            # revert with probability to escape
            if random.random() < 0.001:
                # keep
                best_score = cur_score  # allow temporary worse
                best_x = x.copy()
            else:
                # revert
                x[i] += delta
                x[j] -= delta
    return best_x, best_score
```

"""
propose_candidate – optimisation of a non‑negative step sequence
-----------------------------------------------------------------
We minimise    2·n·max(conv(a,a)) / (sum a)²   (scale invariant)
using a three‑stage strategy:

1.  **Broad randomised parametric sweep**  
    Families: power‑law a_i=(i+σ)^{-p}, exponential a_i=e^{-λi},
    and mixed a_i=(i+σ)^{-p}·e^{-λi}.  The ranges for the
    parameters are deliberately wide (p ∈ [0.45,0.85],
    σ ∈ [10⁻³,5·10²], λ ∈ [10⁻⁴,2]) and the generated vectors
    are truncated when the entries drop below 10⁻¹².  For each
    candidate we normalise to ℓ¹‑norm = 1 and keep the one with
    the smallest objective value.

2.  **Projected Adam optimisation on the simplex**  
    Starting from the best candidate we run an Adam optimiser
    (adaptive moment estimation) with Euclidean projection onto the
    probability simplex after each step.  The sub‑gradient of
    max(conv(a,a)) is computed from all active convolution indices.
    Momentum and bias‑correction give a fast, stable descent and
    continuously improve the best solution seen so far.

3.  **Local stochastic mass‑transfer search**  
    When the Adam phase stalls we perform a cheap hill‑climbing
    routine that repeatedly moves a tiny amount of mass from one
    coordinate to another and accepts the move if the objective
    improves (or with a very small probability to escape plateaus).
    This fine‑tunes the distribution on the scale that Adam’s
    continuous updates cannot resolve.

At any moment we respect the 1000 s wall‑clock budget; the routine
returns the best sequence discovered.
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def _generate_powerlaw(p: float, sigma: float,
                       max_len: int = 12000, eps: float = 1e-12) -> np.ndarray:
    """a_i = (i + sigma) ** (-p)  truncated when value < eps."""
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p)
    return a[a > eps]


def _generate_exponential(lam: float,
                          max_len: int = 12000, eps: float = 1e-12) -> np.ndarray:
    """a_i = exp(-lam * i)  truncated when value < eps."""
    i = np.arange(max_len, dtype=np.float64)
    a = np.exp(-lam * i)
    return a[a > eps]


def _generate_mixed(p: float, sigma: float, lam: float,
                    max_len: int = 12000, eps: float = 1e-12) -> np.ndarray:
    """a_i = (i+sigma)^(-p) * exp(-lam*i)  truncated when value < eps."""
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p) * np.exp(-lam * i)
    return a[a > eps]


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if v.ndim != 1:
        raise ValueError("simplex projection works only on 1‑D vectors")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient(x: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of   g(x) = max_k (x*x)_k
    on the simplex.  All indices attaining the maximum are used.
    """
    n = x.shape[0]
    conv = np.convolve(x, x)
    m = conv.max()
    # tolerance to collect the active set (absolute)
    active = np.where(conv >= m - 1e-14)[0]
    g = np.zeros_like(x)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1, dtype=int)
        j = k - i
        g[i] += x[j]
    return g


def _objective(x: np.ndarray) -> float:
    """
    Compute 2·n·max(conv(x,x)) / (sum x)² .
    This is exactly what evaluate_sequence does after the safety
    checks, but is much faster because we already guarantee valid input.
    """
    n = x.shape[0]
    max_conv = np.max(np.convolve(x, x))
    s = x.sum()
    # s is forced to be 1 in our pipelines, but keep the formula general
    return float(2.0 * n * max_conv / (s * s))

# Optimisation phases
# ----------------------------------------------------------------------


def _phase_random_sweep(best_x, best_score, deadline):
    """
    Wide random search over the three families.  Returns possibly improved
    (x,score) pair.
    """
    rng = np.random.RandomState(123456)
    # Helper to try a candidate, update best if improved
    def try_candidate(arr):
        nonlocal best_x, best_score
        if arr.size < 2:
            return
        arr = arr.astype(np.float64)
        arr = arr / arr.sum()
        sc = _objective(arr)
        if sc < best_score:
            best_score = sc
            best_x = arr.copy()
            print(f'   new sweep best → score={best_score:.6f}  len={len(arr)}')
    # --- Power‑law ---------------------------------------------------------
    for _ in range(1200):
        if time.time() >= deadline:
            return best_x, best_score
        p = rng.uniform(0.45, 0.85)
        sigma = 10 ** rng.uniform(-3, 2.7)          # 1e‑3 … ~500
        seq = _generate_powerlaw(p, sigma, max_len=12000, eps=1e-12)
        try_candidate(seq)
    # --- Exponential -------------------------------------------------------
    for _ in range(400):
        if time.time() >= deadline:
            return best_x, best_score
        lam = 10 ** rng.uniform(-5, 0.5)            # 1e‑5 … ~3.2
        seq = _generate_exponential(lam, max_len=12000, eps=1e-12)
        try_candidate(seq)
    # --- Mixed -------------------------------------------------------------
    for _ in range(600):
        if time.time() >= deadline:
            return best_x, best_score
        p = rng.uniform(0.45, 0.8)
        sigma = 10 ** rng.uniform(-3, 2.0)
        lam = 10 ** rng.uniform(-5, -0.2)
        seq = _generate_mixed(p, sigma, lam, max_len=12000, eps=1e-12)
        try_candidate(seq)
    return best_x, best_score


def _phase_adam(initial_x, best_score, deadline):
    """
    Adam optimisation on the simplex.  Returns the best vector seen.
    """
    x = initial_x.copy()
    m = np.zeros_like(x)        # first moment
    v = np.zeros_like(x)        # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    t = 0
    best_x = x.copy()
    while time.time() < deadline:
        t += 1
        g = _subgradient(x)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        # step size that slowly decays – 0.5 works well empirically
        step = 0.5 / (np.sqrt(t) + 1.0)
        update = step * m_hat / (np.sqrt(v_hat) + eps)
        y = x - update
        x = _simplex_projection(y)

        # small jitter every few hundred iterations to avoid stalls
        if t % 400 == 0 and random.random() < 0.07:
            i = random.randrange(len(x))
            delta = (random.random() - 0.5) * 0.05 * x[i]
            x[i] = max(0.0, x[i] + delta)
            x = x / x.sum()

        cur = _objective(x)
        if cur < best_score:
            best_score = cur
            best_x = x.copy()
            print(f'   Adam iter {t:6d} → score={best_score:.6f}')
            # early termination if we beat the target bound
            if best_score < 1.503:
                print('   Target reached inside Adam – stopping early.')
                break
    return best_x, best_score


def _phase_local_search(initial_x, best_score, deadline):
    """
    Simple stochastic hill‑climbing: move a tiny amount of mass from one
    coordinate to another.  Accept only improving moves (with a tiny
    probability to accept equal moves to escape flat plateaus).
    """
    x = initial_x.copy()
    n = len(x)
    cur_score = best_score
    moves = 0
    # magnitude of a single transfer – starts at 0.5 % of the source mass
    base_frac = 0.005
    while time.time() < deadline and moves < 20000:
        moves += 1
        i = random.randrange(n)
        j = random.randrange(n)
        if i == j or x[i] == 0.0:
            continue
        delta = base_frac * x[i] * (0.8 + 0.4 * random.random())   # 40‑%–120 % of base
        x[i] -= delta
        x[j] += delta
        # keep feasibility
        if x[i] < 0.0:
            # revert and skip this move
            x[i] += delta
            x[j] -= delta
            continue
        new_score = _objective(x)
        if new_score < cur_score or (random.random() < 0.001 and new_score <= cur_score):
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_vec = x.copy()
                print(f'   local move {moves:6d} → score={best_score:.6f}')
                if best_score < 1.503:
                    print('   Target reached inside local search – stopping early.')
                    break
        else:
            # reject – revert
            x[i] += delta
            x[j] -= delta
    # return best seen during local search
    return best_vec if best_score < _objective(initial_x) else initial_x, best_score

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list:
    """
    Return a list of non‑negative heights that minimises the supplied
    evaluate_sequence function within the given wall‑clock budget.
    """
    random.seed(seed)
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_score = float('inf')
    best_seq = None

    # ------------------------------------------------------------------
    # 0.  Use a globally supplied construction if available
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            seq0 = np.asarray(height_sequence_1, dtype=np.float64)
            if seq0.size >= 2:
                seq0 = seq0 / seq0.sum()
                sc0 = _objective(seq0)
                if np.isfinite(sc0):
                    best_score = sc0
                    best_seq = seq0.tolist()
                    print(f'Initial global sequence score = {best_score:.6f}')
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Randomised parametric sweep (broad exploration)
    # ------------------------------------------------------------------
    # start from a uniform fallback if we have nothing yet
    if best_seq is None:
        n0 = 1500
        best_seq = (np.full(n0, 1.0 / n0)).tolist()
        best_score = _objective(np.array(best_seq))
        print(f'Fallback uniform start score = {best_score:.6f}')

    cur_vec = np.asarray(best_seq, dtype=np.float64)
    cur_vec = cur_vec / cur_vec.sum()   # ensure we are on the simplex

    cur_vec, best_score = _phase_random_sweep(cur_vec, best_score, deadline)

    # ------------------------------------------------------------------
    # 2.  Projected Adam optimisation (continuous refinement)
    # ------------------------------------------------------------------
    cur_vec, best_score = _phase_adam(cur_vec, best_score, deadline)

    # ------------------------------------------------------------------
    # 3.  Local stochastic mass‑transfer search (fine‑tuning)
    # ------------------------------------------------------------------
    cur_vec, best_score = _phase_local_search(cur_vec, best_score, deadline)

    # ------------------------------------------------------------------
    # 4.  Optional length growth – tiny mass entries can further lower
    #     the objective if the new normalisation reduces the peak.
    # ------------------------------------------------------------------
    while time.time() < deadline and len(cur_vec) < 12000:
        extra_len = 200
        extra = np.full(extra_len, 1e-12, dtype=np.float64)
        new_vec = np.concatenate([cur_vec, extra])
        new_vec = new_vec / new_vec.sum()
        new_score = _objective(new_vec)
        if new_score < best_score:
            cur_vec = new_vec
            best_score = new_score
            print(f'   length grown to {len(cur_vec)} → score={best_score:.6f}')
            if best_score < 1.503:
                break
        else:
            break

    print('final best score:', best_score, 'length:', len(cur_vec))
    # convert to plain python list for the caller
    return cur_vec.tolist()
