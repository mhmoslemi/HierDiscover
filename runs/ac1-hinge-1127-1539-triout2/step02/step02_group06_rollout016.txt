"""
A fast multi‑phase optimiser for the self‑convolution bound.

Phases
------
1. **Broad parametric sweep** – many analytic families (linear,
   quadratic, exponential, symmetric / monotone) are generated,
   scaled to the required ℓ₁‑norm and the best one is kept.

2. **Smooth‑max gradient descent** – the hard maximum of the
   convolution is approximated by a log‑sum‑exp.  Gradients are
   obtained by a single FFT‑convolution, projected onto the plane
   Σaᵢ = target and updated with momentum.  The temperature is
   reduced steadily.

3. **Exact‑max sub‑gradient descent** – after the soft‑max phase the
   current maximum index is identified and the exact sub‑gradient
   of the (hard) max is applied.  This often yields an extra drop
   in the objective because the update is directed precisely at the
   offending shift.  A tiny learning‑rate schedule together with
   occasional random jitter prevents stagnation.

The algorithm respects a 1000 s wall‑clock budget, always returns the
best sequence seen, and works with any starting sequence stored in the
global variable ``height_sequence_1``.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Low‑level utilities
# ------------------------------------------------------------

def fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using an FFT of power‑of‑2 length."""
    n, m = a.shape[0], b.shape[0]
    L = 1 << ((n + m - 1).bit_length())           # next power of two
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)
    return conv[: n + m - 1]


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself."""
    return fft_convolve(a, a)


def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector so that its ℓ₁‑norm equals *target*."""
    s = v.sum()
    if s > 0:
        return v * (target / s)
    # fall back – uniform distribution (should never happen)
    return np.full_like(v, target / v.shape[0])


def generate_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
    direction: str = "symmetric",
) -> np.ndarray:
    """
    Deterministic analytic shape of length *n*.
    ``kind``    – {"linear","quadratic","exponential","power"}.
    ``direction`` – {"symmetric","decreasing","increasing"}.
    ``param``  controls the strength of the edge/gradient (≥0).
    """
    idx = np.arange(n, dtype=np.float64)

    if direction == "symmetric":
        mid = (n - 1) / 2.0
        d = np.abs(idx - mid) / mid                   # ∈[0,1]
        base = d
    elif direction == "decreasing":
        base = 1.0 - idx / (n - 1)                     # ∈[1,0]
    elif direction == "increasing":
        base = idx / (n - 1)                           # ∈[0,1]
    else:
        raise ValueError("unknown direction")

    if kind == "linear":
        raw = 1.0 + param * base
    elif kind == "quadratic":
        raw = 1.0 + param * base ** 2
    elif kind == "exponential":
        raw = np.exp(param * base)
    elif kind == "power":
        # power concentrates more mass at the edges (base∈[0,1])
        raw = (base + 0.001) ** param
    else:
        raise ValueError("unknown shape kind")

    raw = np.clip(raw, 0.0, None)
    return project_to_sum(raw, target_sum)


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """Transfer a tiny fraction of mass from a random source index to a random target."""
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0.0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_to_sum(a, target_sum)


def jitter(a: np.ndarray, target_sum: float, sigma: float = 1e-3) -> np.ndarray:
    """Add tiny Gaussian noise, clip and renormalise."""
    a = a + np.random.randn(a.shape[0]) * sigma
    a = np.clip(a, 0.0, None)
    return project_to_sum(a, target_sum)


# ------------------------------------------------------------
# Smooth‑max gradient descent (identical to the reference)
# ------------------------------------------------------------

def smooth_max_step(
    a: np.ndarray,
    target_sum: float,
    lr: float,
    tau: float,
    momentum: float,
    v_prev: np.ndarray,
) -> tuple[np.ndarray, np.ndarray]:
    """One iteration of gradient descent on the smoothed max of the convolution."""
    n = a.shape[0]

    # 1) self‑convolution
    conv = conv_self_fft(a)                     # length 2n‑1

    # 2) soft‑max weights
    max_c = conv.max()
    exp_arg = (conv - max_c) / tau
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:                           # degenerate – fall back to hard max
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    # 3) gradient of the smooth max (cross‑correlation)
    a_rev = a[::-1]
    grad_full = fft_convolve(w, a_rev)         # length 3n‑2
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]   # slice corresponding to i=0…n‑1

    # 4) enforce ∑aᵢ = target_sum (project onto the hyper‑plane)
    direction = -(grad - np.mean(grad))

    # 5) momentum + step
    v = momentum * v_prev + direction
    a_new = a + lr * v

    # 6) feasibility
    a_new = np.clip(a_new, 0.0, None)
    a_new = project_to_sum(a_new, target_sum)

    return a_new, v


# ------------------------------------------------------------
# Exact‑max (hard) sub‑gradient utilities
# ------------------------------------------------------------

def subgradient_at_max(a: np.ndarray, max_idx: int) -> np.ndarray:
    """
    Sub‑gradient of  maxₖ (a*a)  at the index *max_idx*.
    The derivative w.r.t aⱼ is 2·a[max_idx‑j] when the index is valid,
    otherwise 0.
    """
    n = a.shape[0]
    idx = np.arange(n)
    src = max_idx - idx
    mask = (src >= 0) & (src < n)
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[src[mask]]
    return grad


def subgradient_step(
    a: np.ndarray,
    target_sum: float,
    lr: float,
    momentum: float,
    v_prev: np.ndarray,
) -> tuple[np.ndarray, np.ndarray]:
    """One iteration of hard‑max sub‑gradient descent."""
    conv = conv_self_fft(a)
    max_idx = int(np.argmax(conv))

    grad = subgradient_at_max(a, max_idx)

    # Project to the Σaᵢ = target hyper‑plane
    direction = -(grad - np.mean(grad))

    v = momentum * v_prev + direction
    a_new = a + lr * v

    a_new = np.clip(a_new, 0.0, None)
    a_new = project_to_sum(a_new, target_sum)
    return a_new, v

    print(f"[{time.time():.2f}] Polishing phase, best score {best_score:.6f}")
    while time.time() < deadline - 0.2:
        a = random_pair_transfer(a, target_sum, scale=0.003)
        score = np.max(conv_self_fft(a))
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"Polish – improvement to {best_score:.6f}")

    print(f"[{time.time():.2f}] Search finished, final best score {best_score:.6f}")
    # Return as plain Python list (the evaluator expects a list)
    return best_seq.tolist()
