"""
Improved search for a non‑negative step sequence that minimises the supplied
evaluate_sequence() objective.

The algorithm works in three main phases (all respecting a 1000 s wall‑clock
budget):

1. **Initial candidate** – we start from a previously discovered sequence if
   the global variable ``height_sequence_1`` exists, otherwise a short
   parametric sweep (power‑law / exponential) supplies a reasonable first
   guess.

2. **Linear‑programming refinement** – for the current vector `f` we compute
   `M = max(f*f)`.  Solving the LP

        maximise   Σ b_j
        subject to  (f * b)_k ≤ M   for all k,
                    b_j ≥ 0,

   yields a vector `b` with a larger ℓ¹‑norm while respecting the same
   convolution bound `M`.  Rescaling `b` to unit sum reduces the maximal
   convolution by the factor Σb, i.e. the objective improves by the same
   factor.  We repeat this step a few times, each time mixing the new vector
   into the current one.

3. **Smooth‑max gradient descent** – the non‑smooth objective `max(f*f)`
   is approximated by a soft‑max
        φ_γ(f) = (1/γ)·log Σ_k exp(γ·(f*f)_k) .
   Its gradient can be written as a simple convolution, which allows cheap
   updates.  Using an exponentiated‑gradient (multiplicative) update together
   with an increasing temperature `γ` quickly flattens the auto‑convolution.
   The best sequence seen during this phase (and a final short Euclidean
   sub‑gradient polishing) is returned.

All helper functions are top‑level, there are no closures or file I/O, and
the implementation falls back to the original sub‑gradient method if SciPy
is unavailable.  Progress and intermediate bounds are printed for
monitoring.
"""

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("simplex projection works only on 1‑D arrays")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _subgradient(x: np.ndarray) -> np.ndarray:
    """Sub‑gradient of  g(x) = max_k (x*x)_k  (Euclidean version)."""
    n = x.shape[0]
    conv = np.convolve(x, x)
    m = conv.max()
    active = np.where(conv >= m - 1e-14)[0]
    g = np.zeros_like(x)
    for k in active:
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        i = np.arange(i_min, i_max + 1, dtype=int)
        # derivative of conv[k] w.r.t. x_i is x_{k-i}
        g[i] += x[k - i]
    return g


def _evaluate(seq: np.ndarray) -> float:
    """Thin wrapper around the provided evaluate_sequence (list‑oriented)."""
    try:
        return evaluate_sequence(seq.tolist())
    except Exception:  # pragma: no cover
        return float('inf')


def _generate_powerlaw(p: float, sigma: float,
                       max_len: int = 5000, eps: float = 1e-10) -> np.ndarray:
    """a_i = (i + sigma) ** (-p)   truncated when a_i < eps."""
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p)
    return a[a > eps]


def _generate_exponential(lam: float,
                          max_len: int = 5000, eps: float = 1e-10) -> np.ndarray:
    """a_i = exp(-lam * i)   truncated when a_i < eps."""
    i = np.arange(max_len, dtype=np.float64)
    a = np.exp(-lam * i)
    return a[a > eps]


def _generate_mixed(p: float, sigma: float, lam: float,
                    max_len: int = 5000, eps: float = 1e-10) -> np.ndarray:
    """a_i = (i+sigma)^(-p) * exp(-lam*i)   truncated when a_i < eps."""
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p) * np.exp(-lam * i)
    return a[a > eps]

def _build_constraint_matrix(f: np.ndarray) -> 'sp.csr_matrix':  # type: ignore
    """
    Construct the matrix A (size (2n‑1)×n) such that
        (A @ b)_k = (f * b)_k   (auto‑convolution of f with variable b).
    The matrix is sparse but we build it explicitly as CSR.
    """
    n = f.shape[0]
    size = n * n
    rows = np.empty(size, dtype=np.int32)
    cols = np.empty(size, dtype=np.int32)
    data = np.empty(size, dtype=np.float64)

    idx = 0
    for i in range(n):
        # column j runs 0..n-1, rows are i+j
        rows[idx:idx + n] = np.arange(i, i + n, dtype=np.int32)
        cols[idx:idx + n] = np.arange(n, dtype=np.int32)
        data[idx:idx + n] = f[i]
        idx += n

    return _sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _lp_improve(f: np.ndarray, M: float) -> (np.ndarray, float):
    """
    Solve the LP  maximise Σ b_j   s.t.  (f*b)_k ≤ M,  b ≥ 0.
    Returns the optimal vector b and its ℓ¹‑norm Σb.
    If the LP fails, returns (None, None).
    """
    n = f.shape[0]
    A = _build_constraint_matrix(f)
    c = -np.ones(n)                      # maximise Σb  ⇔ minimise -Σb
    ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None) for _ in range(n)]

    try:
        res = _opt.linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                           method='highs', options={'presolve': True})
    except Exception:  # pragma: no cover
        return None, None

    if not res.success:
        return None, None
    b = res.x
    return b, float(b.sum())

            M = conv.max()
            b, sum_b = _lp_improve(f, M)
            if b is None or sum_b is None:
                print('LP failed – terminating LP phase.')
                break
            if sum_b <= 1.0 + 1e-8:
                # no improvement possible
                print('LP gave no increase in ℓ¹‑norm – stopping LP phase.')
                break
            # scaled vector that respects the bound with a reduced M
            g = b / sum_b
            # mixing parameter – 0.5 found to be reasonably stable
            t = 0.5
            f_new = (1.0 - t) * f + t * g
            # re‑normalise (numerical drift)
            f_new = f_new / f_new.sum()
            cur_score = _evaluate(f_new)
            if np.isfinite(cur_score) and cur_score < best_score:
                best_score = float(cur_score)
                best_seq = f_new.tolist()
                print(f'LP iter {it+1}: improved score → {best_score:.6f}')
            f = f_new
        # store the LP‑improved vector for the next phase
        best_seq = f.tolist()
    else:
        # if SciPy is missing we simply keep the current best_seq
        f = np.array(best_seq, dtype=np.float64) / np.sum(best_seq)

    # --------------------------------------------------------------
    # 3.  Smooth‑max gradient descent (exponentiated update)
    # --------------------------------------------------------------
    print('=== Phase 3 – smooth‑max gradient descent ===')
    a = np.array(best_seq, dtype=np.float64)
    a = a / a.sum()
    n = a.shape[0]
    gamma = 5.0           # start with a modest smoothing temperature
    gamma_max = 3000.0
    eta = 0.2             # step‑size for exponentiated gradient
    iter_counter = 0

    while time.time() < deadline:
        iter_counter += 1
        # auto‑convolution and soft‑max weights
        conv = np.convolve(a, a)                # length 2n‑1
        maxc = conv.max()
        # avoid overflow: shift by maxc before exponentiation
        w = np.exp(gamma * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0.0:               # numerical safety
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth‑max (derived analytically)
        a_rev = a[::-1]
        grad_full = np.convolve(w, a_rev)       # length 3n‑2
        grad = 2.0 * grad_full[n - 1: n - 1 + n]    # slice central part

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        # occasionally evaluate and possibly record improvement
        if iter_counter % 5 == 0:
            cur_score = _evaluate(a)
            if np.isfinite(cur_score) and cur_score < best_score:
                best_score = float(cur_score)
                best_seq = a.tolist()
                print(f'iter {iter_counter:4d} – smooth descent → score {best_score:.6f}')
            # increase smoothing gradually
            gamma = min(gamma * 1.05, gamma_max)

        # stop early if we already beat the target bound
        if best_score < 1.503:
            print('Target bound reached during smooth descent – exiting.')
            break

    # --------------------------------------------------------------
    # 4.  Final short Euclidean sub‑gradient polishing (optional)
    # --------------------------------------------------------------
    print('=== Phase 4 – final Euclidean sub‑gradient polishing ===')
    x = np.array(best_seq, dtype=np.float64)
    x = x / x.sum()
    step0 = 0.6
    beta = 0.85
    momentum = np.zeros_like(x)
    iteration = 0
    grow_interval = 3000
    max_total_len = 8000

    while time.time() < deadline:
        iteration += 1
        g = _subgradient(x)
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / np.sqrt(iteration)
        y = x - step * momentum
        x = _simplex_projection(y)

        # tiny random jitter every now and then
        if np.random.rand() < 0.01:
            j = np.random.randint(x.shape[0])
            delta = (np.random.rand() - 0.5) * 0.1 * x[j]
            x[j] = max(0.0, x[j] + delta)
            x = x / x.sum()

        # occasional length growth (adds 100 near‑zero entries)
        if iteration % grow_interval == 0 and x.shape[0] < max_total_len:
            extra = np.full(100, 1e-12, dtype=np.float64)
            x = np.concatenate([x, extra])
            x = x / x.sum()
            momentum = np.concatenate([momentum, np.zeros_like(extra)])

        cur_score = _evaluate(x)
        if np.isfinite(cur_score) and cur_score < best_score:
            best_score = float(cur_score)
            best_seq = x.tolist()
            print(f'sub‑grad iter {iteration:6d} → improved score = {best_score:.6f}')
            if best_score < 1.503:
                print('Target bound reached – terminating early.')
                break

    print('Final best score:', best_score, 'length:', len(best_seq))
    return best_seq
