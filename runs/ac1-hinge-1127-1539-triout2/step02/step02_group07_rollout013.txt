def propose_candidate(seed=42, budget_s=1000.0):
    import time; import numpy as np

    start_time = time.time()
    deadline = start_time + budget_s - 5.0

    np.random.seed(seed)

    best_score = np.inf
    best_seq = None

    lengths = [500, 1000, 2000, 3000, 4000]   # maybe upto 5000?
    # Additional lengths maybe up to 6000 if time left.

    # Outer loop over lengths
    for n in lengths:
        if time.time() >= deadline: break

        # generate list of seeds for this n
        seeds = []

        # uniform
        seeds.append(np.full(n, 1.0/n))

        # Dirichlet with random concentration
        for _ in range(5):
            alpha = np.random.rand()*5 + 0.1  # 0.1-5.1
            seeds.append(np.random.dirichlet(np.full(n, alpha)))

        # Exponential
        for lam in np.linspace(0.0002, 0.004, 4):
            i = np.arange(n)
            seq = np.exp(-lam * i)
            seq = seq / seq.sum()
            seeds.append(seq)

        # Power law
        for p_exp in np.linspace(0.3, 0.8, 5):
            sigma = 0.0
            i = np.arange(n)
            seq = (i + sigma + 1) ** (-p_exp)
            seq = seq / seq.sum()
            seeds.append(seq)

        # power law * exponential
        for p_exp in [0.4, 0.5, 0.6]:
            for lam in [0.0005, 0.001, 0.002]:
                i = np.arange(n)
                seq = (i + 1) ** (-p_exp) * np.exp(-lam * i)
                seq = seq / seq.sum()
                seeds.append(seq)

        # Beta-like (symmetric)
        for a in np.linspace(-0.5, 2.0, 4):
            for b in np.linspace(-0.5, 2.0, 4):

    # compute Lc, Lw

"""
Improved non‑negative step‑function optimiser.

The score to minimise is

    f(a) = 2 * n * max(conv(a, a)) / (sum(a)**2)

which is invariant under scaling.  By keeping the vector on the
probability simplex (∑a = 1) the score reduces to 2·n·max(conv).

The optimiser works as follows

1.  A diverse collection of seed vectors is created (uniform,
    exponential, power‑law, power‑law·exponential, Beta‑shaped,
    Dirichlet samples and the previous best sequence if present).

2.  Every seed is refined by a mirror‑descent / Adam scheme.
    In each iteration we compute the full linear convolution
    of the current vector with itself (FFT, O(n log n)).
    Instead of the true ℓ∞‑norm we use a *soft* max:
    the gradient is the (normalised) sum of the gradients of the
    `k` largest convolution entries (the set size grows with `n`).
    The gradient of the weighted sum can be written as a single
    convolution; this is also performed with the FFT.
    An exponentiated‑gradient (multiplicative) Adam update keeps the
    iterates non‑negative and a final normalisation projects onto the
    simplex.

3.  The best vector seen over the whole search (across all seeds)
    is returned.  The loop stops as soon as the target value
    `1.503` is reached or the 1000 s budget is exhausted.

The algorithm explores many initial shapes and uses a far richer
gradient information than the previous “top‑5” sub‑gradient, which
allows it to push the maximal convolution constant well below the
previous best of `1.5709`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, fully real)
# ----------------------------------------------------------------------


def _conv_self(p: np.ndarray, Lc: int) -> np.ndarray:
    """Linear convolution of `p` with itself (full length 2*len(p)-1)."""
    ft = np.fft.rfft(p, n=Lc)
    conv = np.fft.irfft(ft * ft, n=Lc)[: 2 * p.size - 1]
    # tiny negative values may appear due to round‑off – clip them
    conv[conv < 0] = 0.0
    return conv


def _conv_weighted(p: np.ndarray, w: np.ndarray, Lw: int) -> np.ndarray:
    """Convolution of weight vector `w` (length 2n‑1) with `p` (length n)."""
    ft_w = np.fft.rfft(w, n=Lw)
    ft_p = np.fft.rfft(p, n=Lw)
    conv = np.fft.irfft(ft_w * ft_p, n=Lw)[: len(w) + p.size - 1]
    conv[conv < 0] = 0.0
    return conv


# ----------------------------------------------------------------------
# Gradient from a set of “active” convolution indices
# ----------------------------------------------------------------------


def _gradient_from_active(p: np.ndarray, conv: np.ndarray, top_k: int,
                         Lw: int) -> np.ndarray:
    """
    Returns the sub‑gradient of the (normalised) sum of the `top_k`
    largest convolution entries.  The gradient can be expressed as

        g = 2 * ( (w * p)[n-1 : n-1+n] + p ),

    where `w` is the normalised indicator of the active indices.
    """
    n = p.size
    # build the weight vector (indicator of the active set)
    w = np.zeros_like(conv)
    if top_k >= conv.size:
        w[:] = 1.0 / conv.size
    else:
        idx = np.argpartition(conv, -top_k)[-top_k:]
        w[idx] = 1.0 / float(top_k)

    # convolution w * p
    wp = _conv_weighted(p, w, Lw)

    # the slice (n‑1 … n‑1+n‑1) contains Σ_k w_k p_{k‑i}
    h = wp[n - 1 : n - 1 + n]

    # full gradient (always non‑negative)
    g = 2.0 * (h + p)
    return g


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def _seed_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _seed_exponential(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _seed_powerlaw(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) ** (-p)
    return a / a.sum()


def _seed_powerlaw_exp(n: int, p: float, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) ** (-p) * np.exp(-lam * i)
    return a / a.sum()


def _seed_beta_like(n: int, a: float, b: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    # shape (i+1)^a * (n-i)^b   – ensure non‑negative
    a_vec = np.where(np.isfinite((i + 1.0) ** a), (i + 1.0) ** a, 0.0)
    b_vec = np.where(np.isfinite((n - i) ** b), (n - i) ** b, 0.0)
    seq = a_vec * b_vec
    if seq.sum() == 0.0:
        seq = np.full(n, 1.0 / n)
    else:
        seq = seq / seq.sum()
    return seq


def _seed_dirichlet(n: int, concentration: float) -> np.ndarray:
    a = np.full(n, concentration, dtype=np.float64)
    return np.random.dirichlet(a)


def _seed_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / (2.0 * n))
    a = np.maximum(a, 0.0)
    return a / a.sum()


# ----------------------------------------------------------------------
# Core optimiser for a single seed
# ----------------------------------------------------------------------


def _optimise_one_seed(p0: np.ndarray, deadline: float,
                       global_best_score: float,
                       global_best_seq: list) -> tuple[float, list]:
    """
    Mirror‑descent / Adam optimisation starting from `p0`
    (already normalised to sum‑1).  Returns the best score and the
    associated sequence encountered while processing this seed.
    """
    p = p0.astype(np.float64, copy=True)
    n = p.size

    # FFT lengths (next power‑of‑two for speed)
    Lc = 1 << ((2 * n - 2).bit_length())          # for self‑convolution
    Lw = 1 << ((3 * n - 3).bit_length())          # for w * p

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(p)
    v = np.zeros_like(p)

    # learning‑rate schedule
    lr0 = 0.5

    # number of active indices used for the gradient – grows slowly
    # with *n* to keep the update meaningful.
    base_top = max(5, int(0.001 * (2 * n - 1)))   # ~0.1 % of convolution size

    best_local_score = float("inf")
    best_local_seq = None

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- 1.  current convolution and score ---------------------------------
        conv = _conv_self(p, Lc)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val               # sum(p)=1 ⇒ denominator = 1

        # ----- 2.  record improvement -------------------------------------------
        if cur_score < best_local_score - 1e-12:
            best_local_score = cur_score
            best_local_seq = p.tolist()
            if cur_score < global_best_score - 1e-12:
                global_best_score = cur_score
                global_best_seq = best_local_seq
                print(f"Iter {it:5d} | n={n} | new global best {global_best_score:.6f}")
                if global_best_score <= 1.503:
                    return global_best_score, global_best_seq

        # ----- 3.  build the active set (top‑k) ---------------------------------
        top_k = base_top + it // 400                # slowly enlarge the set
        grad = _gradient_from_active(p, conv, top_k, Lw)

        # ----- 4.  Adam step (multiplicative) -----------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr_t = lr0 / (np.sqrt(it) + 1e-8)

        factor = np.exp(-lr_t * m_hat / (np.sqrt(v_hat) + eps))
        p = p * factor

        # ----- 5.  project back onto the simplex ---------------------------------
        s = p.sum()
        if s <= 0.0 or not np.isfinite(s):
            # re‑initialise if the vector collapsed
            p = np.abs(np.random.randn(n))
            s = p.sum()
        p = p / s

        # ----- 6.  occasional random jitter (helps escape plateaus) -------------
        if it % 250 == 0 and np.random.rand() < 0.2:
            idx = np.random.randint(n)
            delta = (np.random.rand() - 0.5) * p[idx] * 0.05
            p[idx] = max(p[idx] + delta, 0.0)
            p = p / p.sum()

        # safety break – we never need more than a few thousand iterations
        if it >= 4000:
            break

    return best_local_score, best_local_seq

# Public entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0):
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises `evaluate_sequence`.  The routine respects the 1000 s
    time limit and returns the best sequence discovered.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0           # spare a few seconds for tidy‑up

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # 1)  Define a collection of lengths to explore (larger gives more
    #     freedom but also costs more time – 4000 proved effective)
    # ------------------------------------------------------------------
    lengths = (2000, 3000, 4000)                # a few different problem sizes

    # ------------------------------------------------------------------
    # 2)  Build a rich pool of seed vectors for each length
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        seeds = []

        # previous solution, if present
        if "height_sequence_1" in globals():
            seq = np.asarray(height_sequence_1, dtype=np.float64)
            if seq.size >= n:
                seeds.append(seq[:n] / seq[:n].sum())
            else:
                # pad with tiny mass if the old vector is shorter
                pad = np.full(n - seq.size, 1e-9, dtype=np.float64)
                seeds.append(np.concatenate([seq, pad]))
                seeds[-1] = seeds[-1] / seeds[-1].sum()

        # uniform
        seeds.append(_seed_uniform(n))

        # exponential families
        for lam in np.linspace(0.0003, 0.003, 4):
            seeds.append(_seed_exponential(n, lam))

        # pure power‑law families
        for p_exp in np.linspace(0.35, 0.70, 5):
            seeds.append(_seed_powerlaw(n, p_exp))

        # power‑law × exponential
        for p_exp in (0.45, 0.55, 0.65):
            for lam in (0.0005, 0.001, 0.002):
                seeds.append(_seed_powerlaw_exp(n, p_exp, lam))

        # Beta‑like (symmetric or skewed)
        for a in (-0.3, 0.0, 0.8, 1.5):
            for b in (-0.3, 0.0, 0.8, 1.5):
                seeds.append(_seed_beta_like(n, a, b))

        # Dirichlet samples with diverse concentration parameters
        for conc in (0.1, 0.5, 1.0, 2.0, 5.0):
            seeds.append(_seed_dirichlet(n, conc))

        # sine‑shaped seed
        seeds.append(_seed_sine(n))

        # ------------------------------------------------------------------
        # 3)  Optimise each seed independently (the loops respect the global
        #     deadline).  The best vector ever seen is kept globally.
        # ------------------------------------------------------------------
        for s in seeds:
            if time.time() >= deadline:
                break
            # Normalise to the simplex – defensive against numerical drift
            if s.sum() == 0.0:
                continue
            p0 = s / s.sum()
            best_score, best_seq = _optimise_one_seed(
                p0, deadline, best_score, best_seq
            )

    # ------------------------------------------------------------------
    # 4)  Return the best sequence (as plain Python list)
    # ------------------------------------------------------------------
    print("Final best score:", best_score, "length:", len(best_seq) if best_seq else None)
    return best_seq if best_seq is not None else []   # empty list signals failure
