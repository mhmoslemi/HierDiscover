def propose_candidate(seed=..., budget_s=1000):

    if base_seq is not None:
        score = evaluate_sequence(base_seq)
        if score < best_score:
            best_seq = np.array(base_seq, dtype=np.float64)
            best_score = score
    # Phase 0: parametric sweep
    families easy param loops describing generate_decreasing_exponential etc.

    # Phase random Dirichlet draws
    for i in range(5000):
        if time.time() >= deadline: break
        alpha = np.random.uniform(0.2, 5.0)
        raw = np.random.gamma(shape=alpha, scale=1.0, size=n)
        cand = project_to_sum(raw, target_sum)
        score = evaluate_sequence(cand.tolist())
        if score < best_score:
            best_score = score
            best_seq = cand
            print(f"Random gamma improved: {best_score:.6f}")
    # Phase gradient descent with Adam
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_iter = 0
    lr = 0.5 * (target_sum/n) # step size
    tau = max(0.1 * a.max(), 1e-4) # temperature
    no_improve = 0
    while time.time() < deadline:
        t_iter += 1
        conv = conv_self_fft(a) # length 2n-1
        max_c = conv.max()
        # softmax weights
        w = np.exp((conv - max_c) / tau)
        w_sum = w.sum()
        w /= w_sum
        # gradient via cross-correlation
        a_rev = a[::-1]
        grad_full = fft_convolve(w, a_rev)
        grad = 2.0 * grad_full[n-1:n-1+n]
        # subtract mean to enforce sum invariance
        direction = -(grad - np.mean(grad))
        # Adam updates
        m = beta1 * m + (1 - beta1) * direction
        v = beta2 * v + (1 - beta2) * (direction*direction)
        m_hat = m / (1 - beta1**t_iter)
        v_hat = v / (1 - beta2**t_iter)

        a = a + step
        a = np.clip(a, 0., None)
        a = project_to_sum(a, target_sum)
        # evaluate
        score = evaluate_sequence(a.tolist())
        if score < best_score - 1e-12:

"""
Improved search for a low‑peak self‑convolution.

The optimiser works in four stages:

1. **Initialisation** – use the previously‑found sequence (if any) and a
   large parametric sweep that now includes exponential, power‑law and
   logarithmic families with much wider parameter ranges.

2. **Random Dirichlet samples** – generate many diverse points from a
   Dirichlet‐like distribution (Gamma draws) to discover promising
   non‑convex shapes.

3. **Adam‑accelerated smooth‑max descent** – minimise a soft‑max of the
   self‑convolution while staying on the ℓ₁‑hyperplane (Σaᵢ = √2n).  The
   temperature of the soft‑max is cooled exponentially, the step size is
   decayed, and occasional random pair‑mass transfers are injected to
   escape plateaux.

4. **Final polishing** – cheap random pair transfers and jitter are
   performed until the time budget expires, always keeping the best
   sequence found.

All operations are pure NumPy; the convolution uses FFTs for speed.
The function returns a plain Python list of non‑negative heights that
minimises ``evaluate_sequence`` within the given time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures / lambdas)
# ----------------------------------------------------------------------


def fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using an FFT of the next power‑of‑2 length.
    """
    n, m = a.shape[0], b.shape[0]
    L = 1 << ((n + m - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)
    return conv[: n + m - 1]


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (length 2·len(a)‑1)."""
    return fft_convolve(a, a)


def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """
    Scale a non‑negative vector so that its ℓ₁‑norm equals *target*.
    The vector is assumed to be non‑negative; a zero‑vector is replaced
    by a uniform distribution.
    """
    s = v.sum()
    if s > 0:
        return v * (target / s)
    # fallback – uniform distribution (should never happen)
    return np.full_like(v, target / v.shape[0])


def generate_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
    direction: str = "symmetric",
) -> np.ndarray:
    """
    Produce a deterministic shape of length *n*.

    ``kind``     – {"linear", "quadratic", "exponential"}.
    ``direction`` – {"symmetric", "decreasing", "increasing"}.

    *param* controls the strength of the edge/gradient (≥0) and is
    clipped to a non‑negative raw profile before scaling.
    """
    idx = np.arange(n, dtype=np.float64)
    if direction == "symmetric":
        mid = (n - 1) / 2.0
        d = np.abs(idx - mid) / mid                # ∈[0,1]
        base = d
    elif direction == "decreasing":
        base = 1.0 - idx / (n - 1)                  # ∈[1,0]
    elif direction == "increasing":
        base = idx / (n - 1)                        # ∈[0,1]
    else:
        raise ValueError("unknown direction")

    if kind == "linear":
        raw = 1.0 + param * base
    elif kind == "quadratic":
        raw = 1.0 + param * base ** 2
    elif kind == "exponential":
        raw = np.exp(param * base)
    else:
        raise ValueError("unknown shape kind")

    raw = np.clip(raw, 0.0, None)
    return project_to_sum(raw, target_sum)


def generate_powerlaw(
    n: int,
    target_sum: float,
    exponent: float = 1.0,
    direction: str = "decreasing",
) -> np.ndarray:
    """Power‑law decreasing / increasing shape ( (n‑i)^p or (i+1)^p )."""
    idx = np.arange(n, dtype=np.float64)
    if direction == "decreasing":
        raw = (n - idx) ** exponent
    elif direction == "increasing":
        raw = (idx + 1) ** exponent
    else:
        raise ValueError("unknown direction")
    raw = np.clip(raw, 0.0, None)
    return project_to_sum(raw, target_sum)


def generate_logarithmic(
    n: int,
    target_sum: float,
    param: float = 1.0,
    direction: str = "decreasing",
) -> np.ndarray:
    """Logarithmic shape: log(1 + param·(n‑i)) or log(1 + param·i)."""
    idx = np.arange(n, dtype=np.float64)
    if direction == "decreasing":
        raw = np.log1p(param * (n - idx))
    elif direction == "increasing":
        raw = np.log1p(param * idx)
    else:
        raise ValueError("unknown direction")
    raw = np.clip(raw, 0.0, None)
    return project_to_sum(raw, target_sum)


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """Transfer a tiny fraction of mass from a random source index to a random target."""
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0.0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_to_sum(a, target_sum)


def jitter(a: np.ndarray, target_sum: float, sigma: float = 1e-3) -> np.ndarray:
    """Add Gaussian noise (clipped) and renormalise."""
    a = a + np.random.randn(a.shape[0]) * sigma
    a = np.clip(a, 0.0, None)
    return project_to_sum(a, target_sum)


# ----------------------------------------------------------------------
# Adam‑accelerated smooth‑max gradient descent
# ----------------------------------------------------------------------


def softmax_gradient(a: np.ndarray, tau: float) -> np.ndarray:
    """
    Compute the gradient of the soft‑max of the self‑convolution of *a*.
    The result is the vector 2·(w ⊛ a_rev) where w are the soft‑max
    weights for the convolution entries.
    """
    conv = conv_self_fft(a)                     # length 2n‑1
    max_c = conv.max()
    # stability trick
    exp_arg = (conv - max_c) / tau
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    # cross‑correlation w * a (equivalently convolution with reversed a)
    a_rev = a[::-1]
    grad_full = fft_convolve(w, a_rev)         # length 3n‑2
    n = a.shape[0]
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]   # slice for i = 0…n‑1
    return grad


def adam_step(
    a: np.ndarray,
    target_sum: float,
    lr: float,
    tau: float,
    beta1: float,
    beta2: float,
    eps: float,
    m: np.ndarray,
    v: np.ndarray,
    t: int,
) -> tuple[np.ndarray, np.ndarray]:
    """
    One Adam step on the soft‑max objective, followed by a projection onto
    the ℓ₁‑hyperplane Σaᵢ = *target_sum*.
    Returns the updated vector and the new moment buffers.
    """
    # gradient of the smoothed max
    grad = softmax_gradient(a, tau)
    # project gradient to the hyperplane ΣΔa = 0
    direction = -(grad - np.mean(grad))

    # Adam moment updates
    m = beta1 * m + (1.0 - beta1) * direction
    v = beta2 * v + (1.0 - beta2) * (direction * direction)

    # bias‑corrected estimates
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    # parameter update
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = a + step

    # feasibility: non‑negative and right ℓ₁‑norm
    a_new = np.clip(a_new, 0.0, None)
    a_new = project_to_sum(a_new, target_sum)
    return a_new, m, v

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence`` within the supplied time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # tiny safety margin

    # ------------------------------------------------------------------
    # Choose a length – honour an existing construction if present
    # ------------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
    else:
        # a size that is large enough to give a good bound but still cheap
        n = np.random.randint(7200, 8800)

    target_sum = np.sqrt(2.0 * n)                    # the normalisation used in the paper

    # ------------------------------------------------------------------
    # Phase 0 – initialise with a known good sequence (if any)
    # ------------------------------------------------------------------
    if isinstance(base, (list, tuple)) and len(base) > 0:
        best_seq = np.array(base, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[init] Using supplied sequence – score {best_score:.6f}")
    else:
        best_seq = np.full(n, target_sum / n)        # uniform baseline
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[init] Uniform baseline – score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – extensive parametric sweep (expanded families)
    # ------------------------------------------------------------------
    families = [
        # (direction, kind, param_grid)
        ("decreasing", "linear", np.linspace(0.0, 10.0, 51)),
        ("decreasing", "quadratic", np.linspace(0.0, 6.0, 31)),
        ("decreasing", "exponential", np.linspace(0.0, 12.0, 41)),
        ("decreasing", "powerlaw", np.linspace(0.5, 6.0, 28)),   # special handling below
        ("decreasing", "logarithmic", np.linspace(0.1, 10.0, 50)),
    ]

    print(f"[{time.time():.2f}] Starting parametric sweep (n={n})")
    for direction, kind, param_grid in families:
        for p in param_grid:
            if time.time() >= deadline:
                print("Time exhausted during sweep.")
                return best_seq.tolist()

            if kind in {"linear", "quadratic", "exponential"}:
                cand = generate_shape(n, target_sum, kind=kind, param=p, direction=direction)
            elif kind == "powerlaw":
                cand = generate_powerlaw(n, target_sum, exponent=p, direction=direction)
            elif kind == "logarithmic":
                cand = generate_logarithmic(n, target_sum, param=p, direction=direction)
            else:
                continue

            score = evaluate_sequence(cand.tolist())
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"  New best {direction}/{kind} (p={p:.3f}) → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – random Dirichlet / Gamma draws (explorative)
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Random Dirichlet / Gamma exploration")
    for _ in range(4000):
        if time.time() >= deadline:
            break
        alpha = np.random.uniform(0.2, 5.0)               # concentration parameter
        raw = np.random.gamma(shape=alpha, scale=1.0, size=n)
        cand = project_to_sum(raw, target_sum)
        score = evaluate_sequence(cand.tolist())
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"  Gamma draw improved → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam‑accelerated smooth‑max gradient descent
    # ------------------------------------------------------------------
    a = best_seq.copy()
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    lr = 0.5 * (target_sum / n)          # base step size (≈ half of the average entry)
    tau = max(0.2 * a.max(), 1e-4)       # initial temperature (fraction of peak)
    iteration = 0
    no_improve = 0

    print(f"[{time.time():.2f}] Entering Adam smooth‑max descent (lr={lr:.5e}, tau={tau:.5e})")
    while time.time() < deadline:
        iteration += 1
        a, m, v = adam_step(
            a,
            target_sum,
            lr,
            tau,
            beta1,
            beta2,
            eps,
            m,
            v,
            iteration,
        )
        # occasional hard‑max step (every 500 iter) to sharpen progress
        if iteration % 500 == 0:
            conv = conv_self_fft(a)
            k_max = int(np.argmax(conv))
            # gradient of the hard max is simply 2*a shifted by k_max
            grad_hard = np.zeros_like(a)
            # contribution from a_{k_max-i}
            for i in range(n):
                j = k_max - i
                if 0 <= j < n:
                    grad_hard[i] += 2.0 * a[j]
            # project to hyperplane
            direction = -(grad_hard - np.mean(grad_hard))
            # simple SGD step (no Adam) to inject a brief push
            a = a + lr * direction
            a = np.clip(a, 0.0, None)
            a = project_to_sum(a, target_sum)

        score = evaluate_sequence(a.tolist())
        if score + 1e-12 < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"Iter {iteration:06d} – improvement: {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # schedule adaptations
        if iteration % 300 == 0:
            lr = max(lr * 0.85, 1e-6)
            tau = max(tau * 0.80, 1e-5)

        # occasional random pair transfer – helps escaping plateaus
        if np.random.rand() < 0.025:
            a = random_pair_transfer(a, target_sum, scale=0.01)

        # jitter reset after long stagnation
        if no_improve >= 2500:
            a = jitter(a, target_sum, sigma=5e-4)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"Iter {iteration:06d} – jitter reset")

    # ------------------------------------------------------------------
    # Phase 4 – final quick random perturbations (polishing)
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Descent finished; best score {best_score:.6f}")

    while time.time() < deadline - 0.2:
        # small random pair transfer; accept only if it improves
        a_try = random_pair_transfer(a, target_sum, scale=0.004)
        score = evaluate_sequence(a_try.tolist())
        if score < best_score:
            best_score = score
            best_seq = a_try.copy()
            a = a_try
            print(f"Polish – improvement to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as plain Python list)
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Returning best sequence – final score {best_score:.6f}")
    return best_seq.tolist()
