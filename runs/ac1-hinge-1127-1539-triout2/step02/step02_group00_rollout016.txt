def _subgradient_descent(a, deadline):
    a = a.copy()
    n = len(a)
    target_sum = np.sqrt(2*n)

    best_a = a.copy()
    best_val = _score(a)
    iteration = 0

    # parameters
    initial_step = target_sum * 0.5 # maybe large
    # step size decays as step = initial_step / sqrt(iter+1)

    while time.time()<deadline:
        # compute convolution and max
        size = 1 << ((2*n-1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
        max_b = b.max()
        # find positions where b is max (within small epsilon)
        eps = 1e-12  # just equality
        max_pos = np.where(np.abs(b - max_b) < 1e-12)[0]  # maybe int indices
        # use e vector: uniform weights over max positions
        e = np.zeros_like(b)
        if max_pos.size>0:
            weight = 1.0 / max_pos.size
            e[max_pos] = weight
        else:
            e[np.argmax(b)]=1
        # Compute gradient g = 2 * crosscorr(a_rev, e) sliced.
        a_rev = a[::-1]
        conv_len = n + len(e) - 1  # 3n-2
        size2 = 1 << (conv_len.bit_length())
        Ar_fft = np.fft.rfft(a_rev, n=size2)
        e_fft = np.fft.rfft(e, n=size2)
        conv_rev = np.fft.irfft(Ar_fft * e_fft, n=size2)[:conv_len]
        g = 2.0 * conv_rev[n-1 : n-1+n]  # length n

        # step size
        step = initial_step / np.sqrt(iteration + 1) / (np.linalg.norm(g) + 1e-12)
        # update
        a_new = a - step * g
        # project onto simplex
        a_new = _project_to_simplex(a_new, target_sum)
        a = a_new
        # evaluate and update best
        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] subgrad improved = {best_val:.6f}")
        iteration += 1

    return best_a, best_val

def _adam_optimization(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    target_sum = sqrt(2*n)
    # Adam hyperparameters
    beta1=0.9; beta2=0.999; eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_a = a.copy()
    best_val = _score(a)
    # schedule for beta temperature: start 1, increase gradually.
    beta_temp = 1.0
    max_beta = 4096.0 # high
    while time.time() < deadline:
        t += 1
        # increase beta gradually
        if t % 200 == 0 and beta_temp < max_beta:
            beta_temp = min(beta_temp * 1.5, max_beta)
        # compute softmax gradient
        g, _ = _gradient_softmax(a, beta_temp)
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        # step size maybe set to step0 * target_sum / (sqrt(v_hat) + eps)
        # set step0 small
        step0 = 0.1  # or something
        a = a - step0 * m_hat / (np.sqrt(v_hat) + eps)
        # project onto simplex
        a = _project_to_simplex(a, target_sum)
        # optionally jitter
        if t % 500 == 0:
            jitter = np.random.default_rng().normal(scale=target_sum * 0.001 / (beta_temp+1), size=n)

            a = np.maximum(a, 0)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] adam iter={t} beta={beta_temp:.2f} score={best_val:.6f}")

"""
A multi‑phase optimiser for the non‑negative step‑function bound.

The algorithm works inside the time budget (default 1000 s) as follows:

1.  Randomly initialise a non‑negative vector a of length n, projected onto
    the ℓ¹‑simplex {a≥0, Σa = √(2 n)}.  Several different lengths are tried.

2.  **Annealed soft‑max gradient descent** – minimise a smooth approximation
    of max(conv(a,a)) using the log‑sum‑exp with an increasing temperature β.
    The update is a projected gradient step with a step size inversely
    proportional to β⋅‖∇‖.  β is ramped up to a large value (≈2048).

3.  **Exact sub‑gradient descent** – after the soft‑max phase we switch to the
    true sub‑gradient of the max‑convolution.  The sub‑gradient is
        g = 2·(a ⋆ e) ,  e = uniform weight on all τ where (a*a)[τ] attains its
    maximum.  A diminishing step size  ηₜ = η₀/√t /‖g‖ is used,
    followed by projection onto the simplex.

4.  **Stochastic hill‑climbing** – a cheap local search that repeatedly moves
    a small random fraction of mass from one index to another while keeping the
    ℓ¹‑norm fixed.  Improvements are kept.

All three phases keep the sum constraint, enforce non‑negativity, and update
the global best sequence whenever the objective value (which equals the
maximum convolution entry after normalisation) improves.

The routine runs as many restarts and lengths as the time budget permits,
always returning the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """Fast version of the competition objective (no safety checks)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:  # should never happen
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    { x | x_i ≥ 0, Σ x_i = s } (Duchi et al. 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of the soft‑max approximation of max(conv(a,a))
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b) where
        b = a*a (full linear convolution, length 2·n‑1)
        g = ∇ (log‑sum‑exp_{β}(b)) = 2·(a ⋆ p)
    with p_k ∝ exp(β·b_k).  All quantities are computed via FFTs.
    """
    n = a.shape[0]

    # b = a*a
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)  # shape (2·n‑1,)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1               # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Annealed soft‑max gradient descent
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the log‑sum‑exp approximation of the
    maximum convolution.  The temperature β is increased progressively
    (up to 2048) while the step size adapts to the gradient magnitude.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 2048.0
    step_base = target_sum * 0.30          # larger base step than before
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            lr = step_base / (norm_g * (beta + 1.0))
            a = a - lr * g

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] anneal it={it} β={beta:.2f} score={best_val:.6f}")

        # annealing schedule
        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on max(conv(a,a))
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent directly on max(conv(a,a)).
    The sub‑gradient is  g = 2·(a ⋆ e) where e distributes unit mass over all
    indices attaining the current maximum of the convolution.
    A diminishing step size ηₜ = η₀ / √t / ‖g‖ is used.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    iter_cnt = 0
    step0 = target_sum * 0.50          # initial scaling for the step
    rng = np.random.default_rng()

    while time.time() < deadline:
        # full convolution a*a
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_b = np.max(b)
        # indicator (or uniform weight) on all max positions
        eps = 1e-12
        max_pos = np.where(np.abs(b - max_b) < eps)[0]
        e = np.zeros_like(b)
        if max_pos.size == 0:
            e[np.argmax(b)] = 1.0
        else:
            e[max_pos] = 1.0 / max_pos.size

        # sub‑gradient g = 2·(a ⋆ e)
        a_rev = a[::-1]
        conv_len = n + b.size - 1               # = 3·n‑2
        size2 = 1 << (conv_len.bit_length())
        Ar_fft = np.fft.rfft(a_rev, n=size2)
        Er_fft = np.fft.rfft(e, n=size2)
        conv_rev = np.fft.irfft(Ar_fft * Er_fft, n=size2)[:conv_len]
        g = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # step size: ηₜ = η₀ / √t / ‖g‖
        norm_g = np.linalg.norm(g)
        if norm_g == 0:
            norm_g = 1.0
        step = step0 / np.sqrt(iter_cnt + 1) / norm_g
        a = a - step * g

        a = _project_to_simplex(a, target_sum)

        # evaluate
        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] subgrad it={iter_cnt} score={best_val:.6f}")

        # occasional tiny jitter to prevent stagnation
        if iter_cnt % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        iter_cnt += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Cheap stochastic hill‑climbing (local refinement)
# ----------------------------------------------------------------------


def _random_hill_climb(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Randomly transfers a small fraction of mass between two positions,
    keeping the ℓ¹‑norm fixed, and accepts strictly improving moves.
    """
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    cur_val = best_val

    max_frac = 0.12
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        frac = rng.random() * max_frac
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        new_val = _score(a)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill‑climb improved = {best_val:.6f}")
        else:
            a[j] -= delta
            a[i] += delta

    return best_a, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the supplied
    `evaluate_sequence` objective.  The routine runs for at most `budget_s`
    seconds, exploring several vector lengths and random restarts.
    The best sequence found before the deadline is returned.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # Vector lengths to explore – larger lengths give more flexibility.
    lengths = [1500, 2500, 4000, 6000, 8000]

    # Number of restarts per length (may be reduced if time runs out).
    restarts_per_len = 3

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        for restart in range(restarts_per_len):
            if time.time() >= deadline:
                break

            # ---------- initialise ----------
            # Sample a Dirichlet‑like random vector (exponential draws) and project.
            a0 = rng.exponential(scale=1.0, size=n)
            a0 = _project_to_simplex(a0, target_sum)

            # Small jitter before optimisation
            a0 += rng.normal(scale=target_sum * 0.001, size=n)
            a0 = np.maximum(a0, 0.0)
            a0 = _project_to_simplex(a0, target_sum)

            # Allocate a slice of the remaining budget for this restart.
            remaining = deadline - time.time()
            # Give the first phase up to 30 % of the remaining budget (capped).
            phase1_deadline = time.time() + min(remaining * 0.30, 25.0)

            # -------------------- Phase 1 : annealed soft‑max GD --------------------
            a_mid, score_mid = _annealed_gradient_descent(a0, phase1_deadline)

            # -------------------- Phase 2 : exact sub‑gradient descent --------------------
            if time.time() < deadline:
                a_mid, score_mid = _subgradient_descent(a_mid, deadline)

            # -------------------- Phase 3 : cheap hill‑climbing --------------------
            if time.time() < deadline:
                a_mid, score_mid = _random_hill_climb(a_mid, deadline)

            if score_mid < best_score:
                best_score = score_mid
                best_seq = a_mid.copy()
                print(f"** NEW GLOBAL BEST ** n={n} restart={restart} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # fallback – should never be needed, but guarantees a valid return.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        a = np.ones(n, dtype=np.float64)
        a *= np.sqrt(2.0 * n) / n
        best_seq = a

    # Trim trailing zeros (if any) for a cleaner output.
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Convert to plain Python list (the required return type).
    return best_seq.tolist()
