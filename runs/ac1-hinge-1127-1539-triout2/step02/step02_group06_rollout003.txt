1.  Choose a length n (≈ 8‑12 k) and the normalisation
    target_sum = sqrt(2·n)  (the same normalisation used by the reference code).

2.  Initialise the vector with a simple analytic shape that can be zero at one
    side – e.g. a decreasing power‑law, an exponential decay, or a “U” shape
    that is low in the centre and high at the edges.  Several random
    initialisations are tried; the best current solution is kept.

3.  Perform a projected Adam optimiser on a *soft‑max* approximation of the
    max‑convolution.  For a temperature τ the soft‑max weights are
        w_k ∝ exp(conv_k/τ) .
    The gradient of the smoothed objective is a cross‑correlation that can be
    evaluated by FFTs.  After each step the vector is clipped to non‑negative
    values and projected back to the ℓ₁‑norm target_sum.

4.  The temperature τ is slowly reduced (simulated annealing) and, when the
    optimiser stalls, a small random jitter or a pairwise mass transfer is
    applied to escape local minima.

5.  The best sequence found (lowest value of ``evaluate_sequence``) is
    returned.  The whole search respects a 1000 s wall‑clock budget.

All helpers are top‑level functions; no closures or lambdas are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors a and b via FFT (next power of two)."""
    n, m = a.shape[0], b.shape[0]
    L = 1 << ((n + m - 1).bit_length())           # next power of two
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)
    return conv[: n + m - 1]


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (length 2·len(a)‑1)."""
    return fft_convolve(a, a)


def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector so that its ℓ₁‑norm equals *target*."""
    s = v.sum()
    if s > 0.0:
        return v * (target / s)
    # fallback – uniform vector (should never happen)
    return np.full_like(v, target / v.shape[0])


# ----------------------------------------------------------------------
# Simple analytic families for initialisation
# ----------------------------------------------------------------------


def generate_initial_shape(
    n: int,
    target_sum: float,
    shape_type: str = "decreasing_power",
    param: float = 1.0,
) -> np.ndarray:
    """
    Return a deterministic shape of length *n* and ℓ₁‑norm *target_sum*.

    Supported *shape_type*:
        - "decreasing_power":   (1 - i/(n‑1))**param   (zero at the right end)
        - "decreasing_exp":     exp(-param * i/(n‑1))
        - "symmetric_U":        (1 - |i‑mid|/mid)**param  (low centre, high edges)
    """
    idx = np.arange(n, dtype=np.float64)

    if shape_type == "decreasing_power":
        base = 1.0 - idx / (n - 1)                     # linear decay to 0
        raw = np.power(base, max(param, 0.0))
    elif shape_type == "decreasing_exp":
        raw = np.exp(-param * idx / (n - 1))
    elif shape_type == "symmetric_U":
        mid = (n - 1) / 2.0
        d = np.abs(idx - mid) / mid                    # ∈[0,1]
        raw = np.power(1.0 - d, max(param, 0.0))
    else:
        raise ValueError(f"unknown shape_type: {shape_type}")

    # avoid all‑zero vectors – a tiny epsilon keeps the scaling well‑behaved
    raw = np.clip(raw, 1e-12, None)
    return project_to_sum(raw, target_sum)


def random_initial_shape(n: int, target_sum: float) -> np.ndarray:
    """Pick a random shape type and a random parameter."""
    shape_type = np.random.choice(
        ["decreasing_power", "decreasing_exp", "symmetric_U"]
    )
    # param range chosen to give both gentle and very steep profiles
    param = np.random.uniform(0.2, 8.0)
    return generate_initial_shape(n, target_sum, shape_type, param)


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Move a small fraction of the mass from a random source index to a random
    target index, then re‑normalise to the target ℓ₁‑norm.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0.0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_to_sum(a, target_sum)


def jitter(a: np.ndarray, target_sum: float, sigma: float = 1e-3) -> np.ndarray:
    """Add small Gaussian noise, clip to ≥0 and re‑normalise."""
    a = a + np.random.randn(a.shape[0]) * sigma
    a = np.clip(a, 0.0, None)
    return project_to_sum(a, target_sum)


# ----------------------------------------------------------------------
# Optimisation core (soft‑max + Adam)
# ----------------------------------------------------------------------


def softmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """
    Compute soft‑max weights w_k ∝ exp(conv_k / τ) in a numerically stable way.
    """
    # subtract the maximum to avoid overflow
    max_c = conv.max()
    exp_arg = (conv - max_c) / tau
    # a second subtraction of the now‑max value (which is 0) is harmless
    exp_arg -= exp_arg.max()
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w
    return w / w_sum


def gradient_softmax_max(a: np.ndarray, tau: float) -> np.ndarray:
    """
    Gradient of the smooth‑max (soft‑max) of the self‑convolution.
    Returns a vector of length len(a).
    """
    n = a.shape[0]
    conv = conv_self_fft(a)                     # length 2n‑1
    w = softmax_weights(conv, tau)              # same length
    # cross‑correlation (w * a_rev) gives the gradient; we slice the central part
    a_rev = a[::-1]
    grad_full = fft_convolve(w, a_rev)         # length 3n‑2
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]   # slice i = 0 … n‑1
    return grad


def adam_step(
    a: np.ndarray,
    grad: np.ndarray,
    m: np.ndarray,
    v: np.ndarray,
    iteration: int,
    lr: float,
    beta1: float,
    beta2: float,
    eps: float,
) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """
    Perform one Adam update on the variable vector *a* (which is assumed
    already projected onto the feasible set), returning the updated vector,
    the new first-moment estimate and the new second-moment estimate.
    """
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad ** 2)

    m_hat = m / (1.0 - beta1 ** iteration)
    v_hat = v / (1.0 - beta2 ** iteration)

    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = a - step
    return a_new, m, v


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The routine respects a wall‑clock budget of *budget_s* seconds and always
    returns the best sequence discovered within that time.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5      # small safety margin

    # ------------------------------------------------------------------
    # Choose a length (re‑use a known construction if convenient)
    # ------------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        # ensure we have the same normalisation as the reference code
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.array(base, dtype=np.float64)
        best_seq = project_to_sum(best_seq, target_sum)
        best_score = float("inf")
    else:
        # pick a length that balances expressiveness and speed
        n = np.random.randint(7200, 9000)
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = float("inf")

    # ------------------------------------------------------------------
    # Evaluate the (possibly supplied) starting point
    # ------------------------------------------------------------------
    try:
        score0 = evaluate_sequence(best_seq.tolist())
        if score0 < best_score:
            best_score = score0
            best_seq = best_seq.copy()
            print(f"Initial supplied sequence score: {best_score:.6f}")
    except Exception:
        # evaluation failed – reset
        best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper to (re)initialise the optimisation variable
    # ------------------------------------------------------------------
    def initialise_a() -> np.ndarray:
        # Occasionally start from a fresh random analytic shape
        shape = random_initial_shape(n, target_sum)
        return shape

    # ------------------------------------------------------------------
    # Optimisation hyper‑parameters
    # ------------------------------------------------------------------
    lr_base = 0.12 * target_sum / n          # base learning rate (≈12 % of the average entry)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    tau = 0.18 * target_sum                  # initial soft‑max temperature
    min_tau = 1e-4

    # Adam state
    a = initialise_a()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    iteration = 0
    eval_every = 30          # evaluate only every few iterations (expensive)
    stagnation = 0
    best_improvement_iter = 0
    max_stagnation_iters = 2000

    print(f"[{time.time():.2f}] Starting optimisation n={n}, target_sum={target_sum:.4f}")

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # ----- gradient of smoothed max -----
        grad = gradient_softmax_max(a, tau)

        # ----- Adam update + projection -----
        a, m, v = adam_step(a, grad, m, v, iteration, lr_base, beta1, beta2, eps)
        a = np.clip(a, 0.0, None)
        a = project_to_sum(a, target_sum)

        # ----- temperature schedule -----
        if iteration % 250 == 0:
            tau = max(min_tau, tau * 0.94)

        # ----- occasional random jumps for escaping plateaus -----
        if np.random.rand() < 0.012:
            a = jitter(a, target_sum, sigma=5e-4)
            # reset Adam moments to avoid stale direction after a jump
            m.fill(0.0)
            v.fill(0.0)

        if np.random.rand() < 0.015:
            a = random_pair_transfer(a, target_sum, scale=0.015)

        # ----- evaluate periodically -----
        if iteration % eval_every == 0:
            cur_score = evaluate_sequence(a.tolist())
            if cur_score + 1e-12 < best_score:
                best_score = cur_score
                best_seq = a.copy()
                best_improvement_iter = iteration
                stagnation = 0
                print(f"Iter {iteration:06d} – new best {best_score:.6f}")
            else:
                stagnation += eval_every

            # ----- restart if stuck for a long time -----
            if stagnation >= max_stagnation_iters:
                # keep the best solution seen so far, re‑initialise a fresh start
                a = initialise_a()
                m.fill(0.0)
                v.fill(0.0)
                tau = 0.18 * target_sum
                stagnation = 0
                print(f"Iter {iteration:06d} – stagnation, re‑initialising")

    # ------------------------------------------------------------------
    # Final polishing phase (quick random perturbations)
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Main loop finished. Best score so far: {best_score:.6f}")

    while time.time() < deadline - 0.2:
        # tiny random transfer and immediate evaluation
        a = random_pair_transfer(best_seq, target_sum, scale=0.006)
        cur = evaluate_sequence(a.tolist())
        if cur < best_score:
            best_score = cur
            best_seq = a.copy()
            print(f"Polish – improvement to {best_score:.6f}")

    print(f"[{time.time():.2f}] Returning best sequence with score {best_score:.6f}")
    return best_seq.tolist()
