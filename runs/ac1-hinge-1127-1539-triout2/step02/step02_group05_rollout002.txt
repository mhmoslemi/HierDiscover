"""
Propose a step‑function (non‑negative sequence) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm consists of three phases:

1. **Diversified initialisation** – a few simple analytical families
   (power‑law, exponential, piecewise‑constant) are sampled, the best
   candidate is kept and used as warm start.

2. **Soft‑max Adam phase** – a smooth approximation of the `max`
   (soft‑max with a temperature γ) is optimised with Adam, while the
   vector is projected onto the simplex {x ≥ 0, Σx = √(2 n)}.
   This quickly drives the solution into a good basin.

3. **Hard‑max exponentiated‑gradient phase** – the true max is used.
   The sub‑gradient of the active convolution entries is obtained via
   FFT, a multiplicative (exponentiated) update is applied and the
   vector is renormalised.  A small amount of random jitter is added
   periodically to escape plateaus.

Both phases keep track of the best sequence ever observed and return it
when the time budget (≈ 1000 s) is exhausted.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ *x*."""
    return 1 << (x - 1).bit_length()


def _compute_score(a: np.ndarray) -> float:
    """
    Fast version of `evaluate_sequence` (no security checks).
    Assumes a is a non‑negative 1‑D array.
    """
    n = a.shape[0]
    # convolution length = 2 n‑1
    nfft = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    # the original objective equals 2 n · max_c / s²
    return float(2.0 * n * max_c / (s * s))


def project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of *v* onto the simplex
        {x : x_i ≥ 0 , Σ_i x_i = target_sum }.
    """
    # Duchi et al. (ICML 2008) – O(n log n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, len(v) + 1)  # 1‑based indexing
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Main optimisation routine
# ----------------------------------------------------------------------

    function.  The routine respects the time budget, prints progress,
    and finally returns the best sequence found (as a plain Python list).
    """

    start_time = time.time()

    deadline = start_time + budget_s - 5.0
