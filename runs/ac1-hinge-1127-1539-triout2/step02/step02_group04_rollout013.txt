"""
Propose a candidate sequence minimizing ``evaluate_sequence``.

Algorithm
---------
1. **Quick parametric sweep**  
   Generate a few families of analytically‑defined sequences
   (power‑law, exponential and especially the flexible beta‑distribution
   shape `a_i ∝ x^{α‑1}(1‑x)^{β‑1}` on a regular grid).  Every candidate is
   normalised to sum 1, its (fast) autocorrelation is obtained by an FFT‑based
   convolution and the cheap objective `2·n·max(conv)` is inspected.
   The best candidate seen so far is stored.

2. **Smooth‑max Adam optimisation**  
   Starting from the best candidate, we minimise a *smooth* surrogate of the
   true objective

       f_β(a) = 2·n·log‑sum‑exp(β·(a∗a))/β

   whose gradient is analytically available:
       ∇f_β(a) = 4·n·corr(w, a)  with  w_k ∝ exp(β·(a∗a)_k).

   The correlation `corr(w,a)` is computed by a second FFT‑based convolution.
   An Adam optimiser (adaptive learning rate) together with Euclidean
   projection onto the probability simplex keeps the iterates feasible.
   The smoothing parameter `β` is increased slowly (β≈5 → β≈200) so that
   the algorithm first moves under a smooth landscape and finally sharpens
   to the genuine max‑norm.

3. **Simple local refinement**  
   After the Adam phase a fast random‑swap hill‑climbing (moving a tiny
   amount of mass between two positions) tries to squeeze a few more
   points out of the remaining “flat” region.

All steps respect the 1000 s wall‑clock budget and print progress.
The returned value is a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("simplex projection works only on 1‑D arrays")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _evaluate(seq: np.ndarray) -> float:
    """Thin wrapper around the provided ``evaluate_sequence``."""
    try:
        return evaluate_sequence(seq.tolist())
    except Exception:
        return float('inf')


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (fast for vectors up to ~10 k)."""
    n = a.shape[0] + b.shape[0] - 1
    N = 1 << ((n - 1).bit_length())          # next power of two
    Fa = np.fft.rfft(a, n=N)
    Fb = np.fft.rfft(b, n=N)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=N)[:n]
    return conv


def _fft_correlation(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear correlation of w (length m) with a (length n) producing
    the ``valid`` part of length n:
        corr[j] = Σ_i w[i+j] * a[i]    for j = 0,…,n‑1
    Implemented via a single FFT convolution.
    """
    conv_full = _fft_convolve(w, a[::-1])          # length m+n‑1
    n = a.shape[0]
    start = n - 1
    return conv_full[start:start + n]


def _beta_distribution(n: int, alpha: float, beta: float, eps: float = 1e-12) -> np.ndarray:
    """Return a normalised beta‑like shape on {0,…,n‑1}."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    # avoid division by zero at the exact boundaries
    x = np.clip(x, eps, 1.0 - eps)
    a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    if a.sum() == 0:
        a = np.ones_like(a)
    return a / a.sum()


def _random_simplex(n: int) -> np.ndarray:
    """Draw a random point on the probability simplex (Dirichlet(1,…,1))."""
    v = np.random.rand(n)
    return v / v.sum()

    print('=== Phase 1 – parametric sweep ===')
    # Candidate lengths – larger lengths give a lower theoretical bound.
    lengths = [2000, 4000, 8000]

    # Very cheap families: power‑law, exponential
    for n in lengths:
        # power‑law
        for p in (0.3, 0.4, 0.5, 0.6):
            for sigma in (1e-3, 1e-2, 1e-1, 1.0):
                if time.time() >= deadline:
                    break
                i = np.arange(n, dtype=np.float64)
                seq = (i + sigma) ** (-p)
                seq = seq[seq > 0]
                if seq.size < 2:
                    continue
                seq = seq / seq.sum()
                # cheap estimate (no call to expensive evaluate_sequence)
                conv = _fft_convolve(seq, seq)
                est = 2.0 * n * conv.max()
                if est < best_score:
                    real = _evaluate(seq)
                    if real < best_score:
                        best_score = real
                        best_seq = seq.tolist()
                        print(f'new best PL  n={n} p={p:.2f} σ={sigma:.3g} est={est:.6f} real={real:.6f}')

        # exponential
        for lam in (1e-4, 5e-4, 2e-3, 5e-3, 1e-2, 5e-2, 1e-1):
            if time.time() >= deadline:
                break
            i = np.arange(n, dtype=np.float64)
            seq = np.exp(-lam * i)
            seq = seq[seq > 0]
            seq = seq / seq.sum()
            conv = _fft_convolve(seq, seq)
            est = 2.0 * n * conv.max()
            if est < best_score:
                real = _evaluate(seq)
                if real < best_score:
                    best_score = real
                    best_seq = seq.tolist()
                    print(f'new best EXP n={n} λ={lam:.1e} est={est:.6f} real={real:.6f}')

        # beta‑distribution family (high flexibility)
        alphas = (0.3, 0.5, 0.8, 1.0, 1.5, 2.0)
        betas = (0.3, 0.5, 0.8, 1.0, 1.5, 2.0)
        for a in alphas:
            for b in betas:
                if time.time() >= deadline:
                    break
                seq = _beta_distribution(n, a, b)
                conv = _fft_convolve(seq, seq)
                est = 2.0 * n * conv.max()
                if est < best_score:
                    real = _evaluate(seq)
                    if real < best_score:
                        best_score = real
                        best_seq = seq.tolist()
                        print(f'new best BETA n={n} α={a:.2f} β={b:.2f} est={est:.6f} real={real:.6f}')

    # ------------------------------------------------------------------
    # 2.  Smooth‑max Adam optimisation on the best candidate found
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback uniformly distributed vector
        n0 = 4000
        best_seq = np.ones(n0, dtype=np.float64) / n0
        best_score = _evaluate(best_seq)
        print('fallback to uniform sequence, score =', best_score)

    x = np.array(best_seq, dtype=np.float64)
    n = x.shape[0]
    x = x / x.sum()          # ensure we start on the simplex

    # Adam hyper‑parameters
    lr = 0.3
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    # smooth‑max schedule
    beta_smooth = 5.0
    beta_target = 200.0
    beta_inc_every = 1500    # increase after this many iterations

    iteration = 0
    best_iter = 0
    print('=== Phase 2 – smooth‑max Adam optimisation ===')
    while time.time() < deadline:
        iteration += 1

        # ---- convolution and smooth‑max weights ----
        conv = _fft_convolve(x, x)                 # length 2n‑1
        cmax = conv.max()
        # stable softmax
        exp_part = np.exp(beta_smooth * (conv - cmax))
        w = exp_part / exp_part.sum()              # shape (2n‑1,)

        # ---- gradient (4·n·corr) ----
        corr = _fft_correlation(w, x)               # length n
        grad = 4.0 * n * corr

        # ---- Adam update ----
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        y = x - step
        x = _simplex_projection(y)

        # occasional jitter to avoid tiny plateaus
        if np.random.rand() < 0.02:
            j = np.random.randint(n)
            delta = (np.random.rand() - 0.5) * 0.1 * x[j]
            x[j] = max(0.0, x[j] + delta)
            x = x / x.sum()

        # increase smoothness gradually
        if iteration % beta_inc_every == 0 and beta_smooth < beta_target:
            beta_smooth = min(beta_target, beta_smooth * 1.5)

        # evaluate every 100 iterations (fast estimate + real evaluate)
        if iteration % 100 == 0:
            cur_est = 2.0 * n * conv.max()
            # evaluate the true objective only when promising
            if cur_est < best_score:
                cur_real = _evaluate(x)
                if cur_real < best_score:
                    best_score = cur_real
                    best_seq = x.tolist()
                    best_iter = iteration
                    print(f'iter {iteration:6d} – est={cur_est:.6f} – new best={best_score:.6f}')

        # restart if no improvement for a long while
        if iteration - best_iter > 20000:
            x = _random_simplex(n)
            m = np.zeros_like(x)
            v = np.zeros_like(x)
            beta_smooth = 5.0
            best_iter = iteration
            print('restart after stagnation')

    # ------------------------------------------------------------------
    # 3.  Light local refinement (random mass swaps)
    # ------------------------------------------------------------------
    print('=== Phase 3 – local hill‑climbing ===')
    x = np.array(best_seq, dtype=np.float64)
    n = x.shape[0]
    cur_score = best_score
    swaps = 0
    while time.time() < deadline and swaps < 5000:
        i, j = np.random.choice(n, size=2, replace=False)
        eps = min(x[i], 0.01 * x[i] + 1e-12)  # move at most 1 % of mass from i
        if eps <= 0:
            continue
        y = x.copy()
        y[i] -= eps
        y[j] += eps
        # project not needed – sum stays 1 and no negatives introduced
        conv_y = _fft_convolve(y, y)
        cand_score = 2.0 * n * conv_y.max()
        if cand_score < cur_score:
            # confirm with the official evaluator (cheap since we already have conv)
            real = _evaluate(y)
            if real < best_score:
                best_score = real
                best_seq = y.tolist()
                cur_score = cand_score
                x = y
                print(f'swap {swaps:4d} – improved to {best_score:.6f}')
        swaps += 1

    print('final best score:', best_score, 'length:', len(best_seq))
    return best_seq
