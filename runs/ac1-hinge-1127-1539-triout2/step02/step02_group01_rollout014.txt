def subgradient_step(a, target_sum, eta):
    conv = conv_self_fft(a)
    max_val = conv.max()
    # compute mask for indices where conv approx max
    eps = max_val * 1e-12 + 1e-12?
    mask = (conv >= max_val - 1e-12 * max_val) # tolerance
    w = mask.astype(np.float64)
    # weight is uniform over active constraints
    # number of active entries = cnt
    if cnt > 0:
        w /= cnt
    else:
        # should not happen
        w = np.full_like(conv, 1.0 / conv.size)
    # compute gradient
    a_rev = a[::-1]
    # convolve w and a_rev
    L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
    grad = 2.0 * full[n-1: n-1 + len(a)]   # length n
    a_new = a - eta * grad
    # ensure nonnegative and normalize
    a_new = project_onto_simplex(np.maximum(a_new, 0.0), target_sum)
    return a_new, max_val

"""
Propose a step‑function (non‑negative coefficient vector) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

where `conv` is the linear self‑convolution.  The evaluation is
scale‑invariant, so we fix the ℓ₁‑norm to √(2 n); under this
normalisation the objective equals `max(conv)`.  The algorithm consists of

1. **A cheap parametric sweep** over several symmetric families
   (linear, quadratic, exponential, cosine‑U, power‑U and two‑block
   shapes).  The best candidate becomes the start point.

2. **Projected (sub‑)gradient descent** on the *hard* max‑convolution
   (no smoothing).  The sub‑gradient of `max(conv)` is obtained by a
   convolution of a binary mask (the active constraints) with the
   reversed current sequence.  An Adam‑style momentum term is used,
   after each step the vector is symmetrised and projected back onto
   the ℓ₁‑simplex.

3. **Polishing** with tiny random pair‑mass transfers.

All convolutions use an FFT of length a power‑of‑two, giving an
`O(n log n)` iteration and allowing several 10⁵ steps within the
≈1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a with itself using an FFT of power‑of‑two
    length.  Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex
        {x | x_i ≥ 0, Σ_i x_i = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, target / len(v))

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def generate_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param1: float = 0.0,
    param2: float = 0.0,
) -> np.ndarray:
    """
    Produce a symmetric shape of length n and project it onto the ℓ₁‑simplex.
    Supported `kind`:
        linear      –  a_i = 1 + p·d
        quadratic   –  a_i = 1 + p·d²
        exponential –  a_i = exp(p·d)
        cosine      –  a_i = 1 + p·(1 - cos(π·d))
        power       –  a_i = (1 + p·d)^q   (q = param2)
        block       –  edges of width w = p·n, edge value = 1+q,
                       interior = 1
    Here d = |i - mid| / mid ∈ [0,1] is the normalised distance from the centre.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param1 * d
    elif kind == "quadratic":
        raw = 1.0 + param1 * d ** 2
    elif kind == "exponential":
        raw = np.exp(param1 * d)
    elif kind == "cosine":
        raw = 1.0 + param1 * (1.0 - np.cos(np.pi * d))
    elif kind == "power":
        q = max(0.1, param2)                         # exponent, avoid zero
        raw = (1.0 + param1 * d) ** q
    elif kind == "block":
        # param1 = width fraction (0..0.5), param2 = edge factor (>1)
        width = int(np.clip(param1, 0.0, 0.5) * n)
        width = max(1, width)
        edge_factor = max(1.0, param2)
        raw = np.ones(n, dtype=np.float64)
        raw[:width] = edge_factor
        raw[-width:] = edge_factor
    else:
        raise ValueError(f"unknown shape kind: {kind}")

    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def subgradient_step(
    a: np.ndarray,
    target_sum: float,
    eta: float,
    momentum_vec: np.ndarray,
    beta: float,
) -> tuple[np.ndarray, np.ndarray, float]:
    """
    One iteration of (momentum) sub‑gradient descent on `max(conv(a,a))`.
    Returns (new_a, new_momentum, max_conv_current).
    """
    n = a.shape[0]

    # 1) self‑convolution and its maximum
    conv = conv_self_fft(a)                     # length 2n‑1
    max_conv = conv.max()

    # 2) mask of active constraints (indices attaining the max)
    eps = max(max_conv * 1e-12, 1e-14)
    mask = (conv >= max_conv - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt == 0:
        weight = np.full_like(conv, 1.0 / conv.size)
    else:
        weight = mask / cnt                     # uniform over active indices

    # 3) sub‑gradient = 2 * (weight * a_rev) restricted to the n coordinates
    a_rev = a[::-1]
    L = 1 << ((len(weight) + len(a_rev) - 1).bit_length())
    W = np.fft.rfft(weight, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    full = np.fft.irfft(W * A_rev, n=L)[: len(weight) + len(a_rev) - 1]
    grad = 2.0 * full[n - 1 : n - 1 + n]

    # 4) momentum update (Adam‑style but without bias correction)
    momentum = beta * momentum_vec + (1.0 - beta) * grad

    # 5) gradient step, enforce non‑negativity, symmetry and simplex constraint
    a_new = a - eta * momentum
    a_new = np.maximum(a_new, 0.0)
    a_new = (a_new + a_new[::-1]) * 0.5          # enforce exact symmetry
    a_new = project_onto_simplex(a_new, target_sum)

    return a_new, momentum, max_conv


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a tiny amount of mass from one random index to another,
    re‑project onto the ℓ₁‑simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def evaluate_maxconv(a: np.ndarray) -> float:
    """Return max(conv(a,a)) assuming Σa = √(2 n)."""
    return conv_self_fft(a).max()
