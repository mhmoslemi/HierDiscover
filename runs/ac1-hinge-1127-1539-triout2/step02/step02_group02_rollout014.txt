```
def smooth_max_gradient_descent(a0, n, deadline, tau0=1.0, tau_decay=0.99, lr=0.1, lr_decay=0.995, max_iters=10000):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    tau = tau0
    best_a = a.copy()
    # compute score
    best_score = evaluate_sequence(a.tolist())
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # compute conv
        conv = np.convolve(a, a)
        # compute weights
        w = np.exp(conv / tau)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(conv) / conv.size
        else:
            w /= w_sum
        # compute gradient
        a_rev = a[::-1]
        # FFT conv
        L = 1 << ((len(w) + len(a_rev) - 1) - 1).bit_length()
        # compute via np.fft.rfft
        Fa = np.fft.rfft(a_rev, n=L)
        Fw = np.fft.rfft(w, n=L)
        conv_grad_full = np.fft.irfft(Fa * Fw, n=L)[:len(w) + len(a_rev) - 1]
        grad = 2 * conv_grad_full[n-1:n-1+len(a)]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # enforce non-negativity
        a = np.maximum(a, 0.0)
        # renormalize sum to sqrt(2n)
        S = np.sqrt(2.0 * n)
        sum_a = a.sum()
        if sum_a == 0:
            a.fill(S / n)
        else:
            a *= (S / sum_a)
        # decay parameters
        lr *= lr_decay
        tau *= tau_decay
        # compute score occasionally
        if it % 20 == 0:
            score = evaluate_sequence(a.tolist())
            if score < best_score - 1e-9:
                best_score = score
                best_a = a.copy()
                print(f"[improve] time={time.time():.1f}s score={best_score:.6f} n={n} it={it} tau={tau:.4f}")
    return best_a, best_score
```

```
def smooth_max_optimize(a0, n, deadline, rng, max_iters=2000):
    a = a0.copy()
    S = np.sqrt(2.0 * n)
    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05
    lr_decay = 0.9995
    tau = 1.0  # temperature
    tau_decay = 0.999
    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # compute conv via FFT
        conv = _full_conv_max(a)
        max_conv = conv.max()
        # compute softmax weights
        # subtract max to stabilize
        scaled = (conv - max_conv) / tau
        # clip to avoid overflow
        scaled = np.clip(scaled, -700, 0)  # because np.exp will be <=1
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum
        # gradient
        grad = _grad_from_weights(a, w)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # enforce non-negativity
        a = np.maximum(a, 0.0)
        # renormalize
        sum_a = a.sum()
        if sum_a == 0:
            a.fill(S / n)
        else:
            a *= (S / sum_a)
        # schedule decay
        lr *= lr_decay
        tau = max(tau * tau_decay, 0.05)  # lower bound to avoid extreme sharpness
        # jitter
        if rng.random() < 0.01:
            noise = rng.normal(scale=a.mean()*0.02, size=n)
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()
        # evaluate every few steps
        if it % 20 == 0:
            try:
                score = evaluate_sequence(a.tolist())
                if score < best_score - 1e-9:
                    best_score = score
                    best_a = a.copy()
                    print(f"[local improve] time={time.time():.1f}s score={best_score:.6f} n={n} it={it} tau={tau:.4f}")
            except Exception:
                pass
    return best_a, best_score
```

assistantfinal**Solution Overview**

The goal is to find a non‑negative step‑function `a` that minimises  

```
score(a) = 2·n·max(a∗a) / ( Σ a )²        (evaluate_sequence)
```

The expression is *scale invariant*: multiplying all entries by a constant does not
change the score.  
Consequently we may freely normalise the vector after every update.  
The score is completely determined by the **maximum entry of the self‑convolution**
`c = a∗a`.  Reducing the peak of `c` while keeping the total mass the same is
exactly the optimisation problem.

### 1.  Smooth approximation of the max‑operator  

Instead of handling the non‑smooth “max’’ directly we use a temperature‑controlled
soft‑max

```
Lτ(a) = τ·log( Σi exp( c[i] / τ ) )
```

which converges to `max(c)` when `τ→0`.  
For a fixed `τ` the gradient is

```
∂Lτ/∂a = 2· ( w ⊗ a_rev )[n‑1 : n‑1+n] ,  w_i = exp( (c_i–max(c))/τ ) / Σj exp(...)
```

where `⊗` denotes a linear convolution and `a_rev` is `a` reversed.
All convolutions are performed with FFTs (`O(n log n)`).

### 2.  Adam optimiser with non‑negativity & normalisation  

*   **Adam** provides an adaptive step size and works well for the noisy,
    non‑convex landscape.
*   After each Adam step the vector is **clipped to ≥0** and renormalised to the
    canonical L1‑norm  

    `Σ a = √(2·n)` – this value is arbitrary because of scale invariance, but it
    keeps the numbers nicely bounded.
*   A **temperature schedule** (`τ ← τ·τ_decay`) starts with a smooth landscape
    (`τ≈1.0`) and progressively focuses on the true maximum.
*   Very occasional **Gaussian jitter** helps to escape tiny plateaus.

### 3.  Multi‑restart driver  

Different lengths `n` and many random initialisations are tried until the
budget (≈ 1000 s) expires.  The best vector seen over all runs is returned.
If a previous construction (`height_sequence_1`) is available it is used as an
additional start point.

The implementation stays fully inside the allowed time, uses only NumPy,
no file‑IO, and calls `evaluate_sequence` only when a genuine improvement is
detected (every ~20 inner iterations).

```python
"""
smooth_max_optimize.py

Search for a step‑function with a low self‑convolution peak using a
temperature–controlled soft‑max surrogate and the Adam optimiser.
The algorithm works for any length n, normalises the vector after each
update (scale invariance) and keeps every entry non‑negative.
Multiple independent restarts are performed until the time budget runs out.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a using an FFT.
    Returns the full (2·n‑1) array.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())   # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate with respect to a.
    Implements   g_j = 2· Σ_i w_i·a_{i‑j}
    efficiently via an FFT convolution.
    """
    n = a.shape[0]
    a_rev = a[::-1]                     # reversed a
    # length of the full convolution of w (len 2n‑1) with a_rev (len n)
    m = len(w) + len(a_rev) - 1
    fft_len = 1 << ((m - 1).bit_length())
    Fw = np.fft.rfft(w, n=fft_len)
    Fa = np.fft.rfft(a_rev, n=fft_len)
    full = np.fft.irfft(Fw * Fa, n=fft_len)[:m]
    # extraction of the slice that corresponds to g
    grad = 2.0 * full[n - 1: n - 1 + n]
    return grad


def _softmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """
    Compute the soft‑max weights w_i = exp((conv_i‑max)/tau) / Σ.
    The subtraction of max(conv) guarantees numerical stability.
    """
    max_c = conv.max()
    # Stabilise the exponent; clip to avoid overflow in np.exp
    scaled = (conv - max_c) / tau
    scaled = np.clip(scaled, -700, 0)   # exp() will be ≤1
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum == 0.0:
        # fallback – uniform weights
        return np.full_like(w, 1.0 / w.size)
    return w / w_sum


# ----------------------------------------------------------------------
# Core optimiser for a fixed length n
# ----------------------------------------------------------------------


def _smooth_max_optimize(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the soft‑max surrogate.
    Returns the best vector found and its external score.
    """
    a = a0.copy()
    S = np.sqrt(2.0 * n)           # canonical L1‑norm (any scale works)
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # schedules
    lr = 0.05
    lr_decay = 0.9995
    tau = 1.0
    tau_decay = 0.999
    tau_min = 0.03          # never let tau become too tiny (numerical)

    # initial external evaluation
    best_score = evaluate_sequence(a.tolist())
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # ----- surrogate computation -------------------------------------------------
        conv = _full_convolution(a)                     # a*a
        w = _softmax_weights(conv, tau)                 # soft‑max over conv

        # ----- gradient --------------------------------------------------------------
        grad = _gradient_from_weights(a, w)

        # ----- Adam update -----------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # ----- enforce constraints ----------------------------------------------------
        a = np.maximum(a, 0.0)          # non‑negative
        total = a.sum()
        if total == 0.0:
            a.fill(S / n)
        else:
            a *= (S / total)           # renormalise to the fixed L1‑norm

        # ----- schedules -------------------------------------------------------------
        lr *= lr_decay
        tau = max(tau * tau_decay, tau_min)

        # ----- occasional jitter -----------------------------------------------------
        if rng.random() < 0.015:
            noise = rng.normal(scale=0.02 * a.mean(), size=n)
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()

        # ----- external evaluation every few steps -----------------------------------
        if it % 20 == 0:
            try:
                cur_score = evaluate_sequence(a.tolist())
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[local improve] time={time.time():.1f}s "
                          f"score={best_score:.6f} n={n} iter={it} tau={tau:.4f}")
            except Exception:
                pass

    return best_a, best_score


# ----------------------------------------------------------------------
# Public entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the current
    bound (≈ 1.503).  The routine runs many independent Adam‑optimisation
    instances with the smooth‑max surrogate, each on a possibly different
    length.  The best vector seen before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # small safety margin

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # Use a previously supplied construction if it exists.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            if cand.ndim == 1 and cand.size > 0:
                sc = evaluate_sequence(cand.tolist())
                best_seq = cand.copy()
                best_score = sc
                print(f"[baseline] existing height_sequence_1 score = {best_score:.6f} "
                      f"length = {cand.size}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Search over several lengths (small enough for fast FFTs, large enough
    # to give the optimiser freedom).
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048, 4096]   # diverse set
    run_id = 0

    while time.time() < deadline:
        n = int(rng.choice(lengths))

        # ---- initial vector -------------------------------------------------
        if best_seq is not None and best_seq.size >= n and rng.random() < 0.25:
            # reuse a truncated version of the current global best
            a0 = best_seq[:n].copy()
        else:
            # fresh random start (Dirichlet‑like)
            a0 = rng.random(n) + 1e-3

        # normalise to the canonical L1‑norm √(2·n)
        S = np.sqrt(2.0 * n)
        a0 = a0 / a0.sum() * S

        # ---- run a single optimisation instance -------------------------------
        a_opt, score_opt = _smooth_max_optimize(a0, n, deadline, rng)

        # ---- keep the global best --------------------------------------------
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[global improve] run={run_id} n={n} score={best_score:.6f}")

        run_id += 1

    # ------------------------------------------------------------------
    # Fallback (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 512
        a = np.full(n, np.sqrt(2.0 * n) / n)
        best_seq = a
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] score={best_score:.6f}")

    print(f"[final] best score = {best_score:.6f}  length = {len(best_seq)}")
    return best_seq.tolist()
```
