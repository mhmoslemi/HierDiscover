"""
Propose a non‑negative step‑function (a list of heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where n = len(a) and conv is the full convolution (np.convolve).

The implementation combines a few different ideas:

* **Multiple restarts** – each restart gets a fresh random
  initialization (uniform, Dirichlet, power‑law and the globally supplied
  sequence, if any) and runs an independent optimisation.
* **Adam optimisation on a soft‑max surrogate** – exactly the same
  surrogate used in the reference code, but with a more aggressive
  learning‑rate schedule and a higher temperature (γ) so that the
  smooth‑max approaches the true max more closely.
* **Projection onto the simplex** – after every update the vector is
  projected onto the set `{x ≥ 0, sum(x)=√(2 n)}` to keep the normalisation
  fixed.
* **Fine‑tuning** – after the restarts a long polishing phase with a
  very large γ and a tiny constant step size is performed.

The routine respects the 1000‑second budget; it always returns the best
vector found before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << (x - 1).bit_length()


def project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of *v* onto the simplex
        {x_i ≥ 0, sum_i x_i = target_sum}.
    """
    # Duchi et al. (ICML 2008) algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(len(v)) + 1  # 1‑based
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_score(a: np.ndarray) -> float:
    """
    Fast version of `evaluate_sequence` – assumes a is already non‑negative.
    """
    n = a.shape[0]
    nfft = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))

# Main optimisation routine
# ----------------------------------------------------------------------
