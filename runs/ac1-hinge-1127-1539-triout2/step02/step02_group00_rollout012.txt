def alternating_lp(f, target_sum, deadline, max_iters=5, t=0.2):
    best_f = f.copy()
    best_obj = _score(f)
    for i in range(max_iters):
        if time.time() >= deadline: break
        M = _max_convolution_fft(f)
        # Build sparse matrix A
        n = f.shape[0]
        # Use repeated method as above.
        i_idx = np.repeat(np.arange(n), n)
        j_idx = np.tile(np.arange(n), n)
        row = i_idx + j_idx
        col = j_idx
        data = f[i_idx]  # f[i] repeated n times
        A = csr_matrix((data, (row, col)), shape=(2*n-1, n))
        b = np.full(2*n-1, M + 1e-12)   # ensure little slack

        # Solve LP via linprog
        c = -np.ones(n)
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
        if not res.success:
            break
        g0 = res.x
        sum_g0 = np.sum(g0)
        if sum_g0 <= 0:
            break
        g = g0 * (target_sum / sum_g0)
        # blend
        f_new = (1 - t) * f + t * g
        # Project to simplex to ensure sum fixed exactly (though scaling ensures sum but small numeric errors)
        f_new = _project_to_simplex(f_new, target_sum)
        new_obj = _score(f_new)
        if new_obj < best_obj:
            best_obj = new_obj
            best_f = f_new.copy()
            f = f_new  # continue from improved
            print(f"LP iteration {i}, new score {best_obj:.6f}")
        else:
            # If no improvement, maybe reduce t and try again?
            # Do simple break.
            break
    return best_f, best_obj

def _lp_improve(f, target_sum, deadline, max_iter=5, t=0.3):
    best_f = f.copy()
    best_score = _score(f)
    n = len(f)
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        M = _max_convolution_fft(f)
        # Build matrix
        i_idx = np.repeat(np.arange(n), n)
        j_idx = np.tile(np.arange(n), n)
        row = i_idx + j_idx
        col = j_idx
        data = f[i_idx]   # vector f repeated n times
        A = csr_matrix((data, (row, col)), shape=(2*n-1, n))
        # Solve LP
        c = -np.ones(n)
        b = np.full(2*n-1, M + 1e-12)   # M + slack
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
        if not res.success:
            break
        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            break
        g = g0 * (target_sum / sum_g0)
        # Blend
        f_new = (1 - t) * f + t * g
        f_new = _project_to_simplex(f_new, target_sum)
        new_score = _score(f_new)
        if new_score < best_score:
            best_score = new_score
            best_f = f_new.copy()
            f = f_new   # continue from improved
            print(f"LP blend improved score {best_score:.6f}")
        else:
            # if not improved, reduce t and try again maybe
            t *= 0.5
            if t < 1e-3:
                break
    return best_f, best_score

def _adam_descent(a0, deadline):
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_val = _score(a)
    beta = 1.0
    max_beta = 2048.0
    step_base = target_sum * 0.18
    mu = 0.9
    v = np.zeros_like(a)
    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        # magnitude scaling
        norm_g = np.linalg.norm(g) + 1e-12
        lr = step_base / (norm_g * (beta + 1.0))
        # momentum update
        v = mu * v - lr * g
        a = a + v
        # project to simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] beta={beta:.2f} iter={it} best={best_val:.6f}")

        # beta schedule
        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 1.4, max_beta)

"""
Advanced optimizer for the step‑function bound.

The evaluation metric (after normalising the ℓ¹‑norm to √(2·n))
is exactly the maximum entry of the self‑convolution.
We therefore minimise max(a*a) under the constraints a_i≥0,
∑a_i = √(2·n).

The algorithm mixes three complementary ideas:

1. **Annealed soft‑max gradient descent with momentum** –  
   A smooth approximation of the max (log‑sum‑exp) is differentiated.
   The temperature β is slowly increased while the step size is
   scaled with 1/(β·‖∇‖).  Momentum (velocity) accelerates convergence
   and a tiny stochastic jitter helps to escape plateaux.

2. **Linear‑programming refinement** –  
   Given the current sequence f, we solve  

        maximise    ∑ g_i
        subject to  (f * g)_k ≤ M   for all k,
                    g_i ≥ 0,

   where M = max(f*f).  The LP finds the most “spread’’ sequence g that
   respects the current worst‑case convolution bound.  After rescaling
   g to the required ℓ¹‑norm we blend it with f (t≈0.3) and keep the
   improvement.  Repeating this step further flattens the convolution.

3. **Focused stochastic hill‑climbing** –  
   Randomly transfer a fraction of mass between two positions,
   accepting only improvements.  This cheap local search polishes the
   result obtained from the previous two stages.

The optimiser runs for at most the supplied time budget, tries several
vector lengths and random restarts, and returns the best sequence found.
All numbers are non‑negative and the ℓ¹‑norm is fixed, so the returned
list can be fed directly to `evaluate_sequence`.
"""

import time
import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the full linear convolution a*a (FFT version)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _objective(a: np.ndarray) -> float:
    """Fast version of the competition objective (max convolution)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:            # should never happen because we keep the norm fixed
        return np.inf
    return float(max_b)    # with the fixed ℓ¹‑norm the expression reduces to max_b


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the log‑sum‑exp approximation of max(a*a).

    Returns (g, b) where
        b = a*a (full convolution, length 2·n‑1)
        g = ∇ (log‑sum‑exp_{β}(b)) = 2·(a ⋆ p),
    with p_k ∝ exp(β·b_k) and ⋆ denoting cross‑correlation.
    """
    n = a.shape[0]

    # ---- b = a*a -------------------------------------------------------
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ---- soft‑max probabilities p ---------------------------------------
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)                     # shape (2·n‑1,)

    # ---- gradient g = 2·(a ⋆ p) ---------------------------------------
    # cross‑correlation a ⋆ p = (reverse(a) * p) central part
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1             # = 3·n‑2
    size2 = 1 << ((conv_len).bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]       # take the centre n values
    return g, b


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _annealed_momentum_descent(a0: np.ndarray, deadline: float):
    """
    Projected gradient descent on the soft‑max approximation.
    Momentum (velocity) is used, β (temperature) is annealed.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _objective(a)

    beta = 1.0
    max_beta = 2048.0
    step_base = target_sum * 0.20
    mu = 0.9
    v = np.zeros_like(a)

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        norm_g = np.linalg.norm(g) + 1e-12
        lr = step_base / (norm_g * (beta + 1.0))

        # momentum update
        v = mu * v - lr * g
        a = a + v

        # projection onto the simplex (keeps norm fixed and non‑negative)
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to shake out of plateaus
        if it % 300 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=n
            )
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] beta={beta:.2f} iter={it} best={best_val:.6f}")

        # annealing schedule
        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 1.4, max_beta)

        it += 1

    return best_a, best_val


def _random_hill_climb(a: np.ndarray, deadline: float, max_frac: float = 0.15):
    """
    Very cheap local search preserving the ℓ¹‑norm.
    Randomly transfers a fraction of mass from i to j and keeps
    any improvement of the objective.
    """
    n = a.size
    best_a = a.copy()
    best_val = _objective(a)
    cur_val = best_val

    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        frac = rng.random() * max_frac
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        new_val = _objective(a)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
        else:
            a[j] -= delta
            a[i] += delta

    return best_a, best_val


def _lp_improve(a: np.ndarray,
                target_sum: float,
                deadline: float,
                max_iter: int = 3,
                blend_t: float = 0.30):
    """
    Linear‑programming refinement (see description above).
    Solves a single LP per iteration, rescales the solution to the
    required ℓ¹‑norm and mixes it with the current vector.
    """
    n = a.shape[0]

    # for very large n the dense LP becomes prohibitive – skip it
    if n > 3000:
        return a, _objective(a)

    best_a = a.copy()
    best_val = _objective(a)

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        M = _max_convolution_fft(a)

        # Build the sparse constraint matrix A where
        #   (f * g)_k = Σ_i f_i * g_{k-i}  ≤  M
        #   corresponds to row k = i + j with coefficient f_i.
        # Repetition pattern: each f_i appears in exactly n rows.
        i_idx = np.repeat(np.arange(n, dtype=np.int32), n)
        j_idx = np.tile(np.arange(n, dtype=np.int32), n)
        rows = i_idx + j_idx                     # 0 … 2n‑2
        cols = j_idx
        data = a[i_idx]                          # f_i repeated n times

        A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

        # LP: maximise Σ g_i  ↔  minimise −Σ g_i
        c = -np.ones(n, dtype=np.float64)
        b = np.full(2 * n - 1, M + 1e-12, dtype=np.float64)

        res = linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": max(0.0, deadline - time.time())},
        )
        if not res.success:
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0.0:
            break

        # Rescale to the required ℓ¹‑norm
        g = g0 * (target_sum / sum_g0)

        # Blend and project
        a_new = (1.0 - blend_t) * a + blend_t * g
        a_new = _project_to_simplex(a_new, target_sum)

        new_val = _objective(a_new)
        if new_val < best_val:
            best_val = new_val
            best_a = a_new.copy()
            a = a_new                           # continue from the improved point
            print(f"[{time.time():.1f}] LP blend improved score={best_val:.6f}")
        else:
            # No improvement – reduce the blend factor and try once more
            blend_t *= 0.5
            if blend_t < 1e-3:
                break

    return best_a, best_val

# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising the supplied
    `evaluate_sequence` objective.  A mixture of annealed gradient descent,
    linear‑programming refinement and stochastic hill‑climbing is performed
    under the given time budget.  The best sequence encountered is returned.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    deadline = time.time() + budget_s - 0.2   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Optional: start from the previously discovered best construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        hseq = np.array(globals()["height_sequence_1"], dtype=np.float64)
        n0 = hseq.size
        target0 = np.sqrt(2.0 * n0)
        hseq = _project_to_simplex(hseq, target0)
        val0 = _objective(hseq)
        print(f"Initial known sequence (n={n0}) score={val0:.6f}")
        best_seq, best_score = hseq, val0

    # ------------------------------------------------------------------
    # Search over several vector lengths
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 4000, 5000, 6000]

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)

        # Decide how many restarts for this length (more for smaller n)
        restarts = 4 if n <= 4000 else 2

        for r in range(restarts):
            if time.time() >= deadline:
                break

            # ----- random power‑law start --------------------------------
            p = 0.5 + rng.uniform(-0.09, 0.09)
            a0 = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
            a0 *= target_sum / a0.sum()
            a0 += rng.normal(scale=target_sum * 0.001, size=n)
            a0 = np.maximum(a0, 0.0)
            a0 = _project_to_simplex(a0, target_sum)

            # ----- allocate a chunk of the remaining budget -------------
            remaining = deadline - time.time()
            # give roughly half of what is left to the gradient phase,
            # but never more than 30 s for a single run
            grad_time = min(remaining * 0.5, 30.0)
            grad_deadline = time.time() + grad_time

            # ----- annealed momentum gradient descent -------------------
            a_opt, score = _annealed_momentum_descent(a0, grad_deadline)

            # ----- LP refinement (if time permits) ----------------------
            if time.time() < deadline:
                a_opt, score = _lp_improve(
                    a_opt, target_sum, deadline, max_iter=2, blend_t=0.30
                )

            # ----- final hill‑climbing ----------------------------------
            if time.time() < deadline:
                a_opt, score = _random_hill_climb(a_opt, deadline)

            # ----- keep global best --------------------------------------
            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(
                    f"** GLOBAL BEST UPDATED ** n={n} restart={r} "
                    f"score={best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # Fallback: uniform vector (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        a = np.full(n, np.sqrt(2.0 * n) / n, dtype=np.float64)
        best_seq = a

    # trim trailing zeros for neatness
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    return best_seq.tolist()
