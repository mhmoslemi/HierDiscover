"""
Propose a step‐function sequence that minimizes

        2 * n * max(conv(a, a)) / (sum(a)**2)

by treating the problem as “flatten the self‑convolution”.
The algorithm works with the probability view:
p_i = a_i / sqrt(2 n) and conv(a,a) = (sqrt(2 n))²·(p∗p).
Thus we minimise the maximal entry of the sum distribution p∗p.

We use a smooth approximation of the max (log‑sum‑exp) to obtain a dense
sub‑gradient, then perform an exponentiated (multiplicative) Adam update.
All updates keep the vector non‑negative and are re‑normalised to the
required ℓ₁‑norm √(2 n).  The procedure is repeatedly restarted with
different lengths and random seeds; the best sequence found before the
budget expires is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a * a (full length)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())      # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _smoothmax_gradient(a: np.ndarray, alpha: float) -> tuple[float, np.ndarray]:
    """
    Compute a smooth approximation of the max‑convolution gradient.

    Returns
        cur_max : the current maximal convolution value (for monitoring)
        g       : dense sub‑gradient of the smoothed max with respect to a
    """
    n = a.shape[0]
    # ----- convolution a * a -----
    conv = _full_convolution(a)
    cur_max = float(conv.max())

    # ----- smooth max weights -----
    # w_k ∝ exp(alpha * (conv_k - cur_max)),  ∑ w_k = 1
    diff = conv - cur_max
    # avoid overflow/underflow
    w = np.exp(alpha * diff)
    w_sum = w.sum()
    if w_sum == 0.0:                     # extremely unlikely, fallback to uniform
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # ----- gradient = 2 * correlation(a, w) -----
    # correlation g_j = Σ_i a_i * w_{i+j}   (j = 0 … n‑1)
    # linear correlation can be obtained via FFT:
    L = 1 << ((3 * n - 2 - 1).bit_length())   # ≥ n + (2n‑1) – 1
    Fa = np.fft.rfft(a, n=L)
    Fw = np.fft.rfft(w, n=L)
    corr = np.fft.irfft(Fa * np.conj(Fw), n=L)[:n]   # real part, first n entries
    g = 2.0 * corr
    return cur_max, g


def _run_optimize_one_start_smooth(
    a0: np.ndarray,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    alpha: float = 10.0,
    eta0: float = 0.5,
    decay: float = 0.9999,
    jitter_prob: float = 0.02,
    jitter_scale: float = 0.05,
    max_iters: int = 200_000,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    stagnation_limit: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑Adam optimisation using a smooth‑max surrogate.
    Returns the best vector encountered and its external score.
    """
    a = a0.copy()
    S = math.sqrt(2.0 * n)

    # best seen so far
    try:
        best_score = evaluate_sequence(a.tolist())
    except Exception:
        best_score = float("inf")
    best_seq = a.copy()
    best_surrogate = float("inf")
    last_improve = 0

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    eta = eta0
    it = 0
    while time.time() < deadline and it < max_iters:
        # ----- smooth‑max gradient step -----
        cur_max, g = _smoothmax_gradient(a, alpha)

        # Adam bias‑corrected moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # multiplicative (exponentiated) update
        step = -eta * m_hat / (np.sqrt(v_hat) + eps)
        a = a * np.exp(step)

        # keep strict positivity (avoid exact zeros)
        a = np.clip(a, 1e-12, None)

        # re‑normalise to the required ℓ₁‑norm
        total = a.sum()
        if total == 0.0:
            a.fill(S / n)
        else:
            a *= (S / total)

        # occasional jitter to escape tiny plateaus
        if rng.random() < jitter_prob:
            noise = rng.normal(scale=jitter_scale * a.mean(), size=n)
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()

        # monitor improvement of the surrogate (the true max of conv)
        if cur_max < best_surrogate - 1e-12:
            best_surrogate = cur_max
            last_improve = it
            try:
                ext = evaluate_sequence(a.tolist())
                if ext < best_score:
                    best_score = ext
                    best_seq = a.copy()
                    print(
                        f"[local improve] t={time.time():.1f}s  score={best_score:.6f}"
                        f" n={n} iter={it} maxB={cur_max:.6f}"
                    )
            except Exception:
                pass

        # early stop if no progress for a long time
        if it - last_improve > stagnation_limit:
            break

        # schedule updates
        eta *= decay
        it += 1

    return best_seq, best_score

        a = np.full(n, math.sqrt(2.0 * n) / n)
        best_seq = a
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] score={best_score:.6f}")

    print(f"[final] best score = {best_score:.6f}  length = {len(best_seq)}")
    return best_seq.tolist()
