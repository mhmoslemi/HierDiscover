"""
Implementation of smooth-max gradient descent (Adam) to minimise the
self‑convolution peak of a non‑negative step function.
The algorithm works in log‑space, keeps the L1‑norm fixed (√(2 n))
and uses a soft‑max approximation of the max‑convolution to obtain a
dense gradient. A continuation on the soft‑max temperature (γ) is
employed: start with a low γ for a smooth landscape and increase it
gradually so that the surrogate converges to the true maximum.
Multiple random restarts with different lengths are performed until the
time budget expires.
"""
import time
import math
import numpy as np

# Helper functions
def _next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    N = _next_pow2(conv_len)
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[:conv_len]
    return conv

def _valid_correlation(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation (x ⋆ y) evaluated only on the ‘valid’
    indices where the shift keeps the two vectors overlapping.
    Returns an array of length len(y) corresponding to shifts 0 … len(y)-1.
    """
    Lx = x.shape[0]
    Ly = y.shape[0]
    N = _next_pow2(Lx + Ly - 1)
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    corr = np.fft.irfft(np.conj(X) * Y, n=N)
    start = Ly - 1
    return corr[start:start + Ly]

def _softmax_weights(conv: np.ndarray, gamma: float) -> np.ndarray:
    """
    Returns the soft‑max probabilities of the entries of ``conv`` with
    temperature 1/γ.  For numerical stability we subtract the maximal entry
    before exponentiation.
    """
    max_c = np.max(conv)
    shifted = conv - max_c            # ≤0
    # Clip to avoid overflow when γ is large.
    scaled = gamma * shifted
    scaled = np.clip(scaled, a_min=-700, a_max=None)
    exp_vals = np.exp(scaled)
    total = np.sum(exp_vals)
    if total == 0.0:
        # Should not happen, but fall back to uniform weights.
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / total

def _adam_step(w: np.ndarray,
               grad: np.ndarray,
               m: np.ndarray,
               v: np.ndarray,
               t: int,
               lr: float,
               beta1: float,
               beta2: float,
               eps: float) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """One Adam update on the parameters w."""
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    w_new = w - lr * m_hat / (np.sqrt(v_hat) + eps)
    return w_new, m, v

def _run_one_restart(n: int,
                     deadline: float,
                     rng: np.random.Generator,
                     init_a: np.ndarray | None = None) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length ``n`` using smooth‑max Adam.
    Returns the best sequence (as a NumPy array) found during this run
    together with its external score.
    """
    S = math.sqrt(2.0 * n)          # target L1‑norm

    # Initialise in log‑space.
    if init_a is not None:
        a0 = init_a.copy()
        # Normalise to the required L1‑norm.
        a0 = np.clip(a0, 0.0, None)
        a0_sum = a0.sum()
        if a0_sum == 0.0:
            a0 = np.full(n, S / n)
        else:
            a0 = a0 * (S / a0_sum)
        eps = 1e-12
        w = np.log(np.maximum(a0, eps))
    else:
        # Random Dirichlet‑like start.
        w = rng.normal(loc=0.0, scale=0.5, size=n)

    # Adam state
    m = np.zeros_like(w)
    v = np.zeros_like(w)

    # Hyper‑parameters
    lr = 0.05
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    gamma = 0.8      # initial temperature (low → smooth)
    gamma_max = 120.0
    gamma_factor = 1.003   # multiplicative increase per iteration

    # Track the best external score seen.
    best_seq = None
    best_score = float('inf')
    best_surrogate = float('inf')

    # Evaluate the start point once.
    a = np.exp(w)
    a = a * (S / a.sum())
    try:
        cur_score = evaluate_sequence(a.tolist())
        best_score = cur_score
        best_seq = a.copy()
        best_surrogate = np.max(_conv_self(a))
    except Exception:
        pass

    it = 0
    while time.time() < deadline and it < 20000:
        it += 1

        # ------- forward pass -------
        a = np.exp(w)
        a = a * (S / a.sum())               # keep Σa = S
        conv = _conv_self(a)                # full self‑convolution
        cur_max = float(conv.max())

        # ------- surrogate (smooth max) -------
        if cur_max < best_surrogate - 1e-12:
            best_surrogate = cur_max
            # Evaluate with the expensive external function only when the surrogate improves.
            try:

"""
Smooth‑max Adam optimiser for the self‑convolution peak problem.

The objective to be minimised is

    score(a) = 2·n·max(a∗a) / (sum(a)²) ,

where the sequence a must be non‑negative.  Scaling a does not change the
score, therefore we keep the L1‑norm fixed at √(2 n) (the normalisation used
by the reference implementation).

The algorithm works in log‑space (w = log a) and uses a differentiable
approximation of the max operator:

    softmax_γ(x)_i = exp(γ·x_i) / Σ_j exp(γ·x_j) .

For a given γ the surrogate objective is  Σ_i softmax_γ(conv)_i·conv_i,
whose gradient w.r.t. a is 2·(softmax_γ(conv) ⋆ a) (linear cross‑correlation).
We apply Adam to the parameters w and increase γ gradually (continuation)
so that the surrogate converges to the true max.

Multiple random restarts with several sequence lengths are performed until
the global time budget (≈ 1000 s) is exhausted.  The best sequence found is
returned as a plain Python list.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Low‑level numeric helpers (FFT based, no Python loops)
# --------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    N = _next_pow2(conv_len)
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[:conv_len]
    return conv


def _valid_correlation(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation (x ⋆ y) evaluated on the ‘valid’ shifts.
    Returns an array of length len(y) corresponding to shifts 0 … len(y)-1.
    """
    Lx = x.shape[0]
    Ly = y.shape[0]
    N = _next_pow2(Lx + Ly - 1)
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    corr = np.fft.irfft(np.conj(X) * Y, n=N)
    start = Ly - 1
    return corr[start:start + Ly]


def _softmax_weights(conv: np.ndarray, gamma: float) -> np.ndarray:
    """
    Soft‑max (with temperature 1/γ) of the convolution entries.
    Numerical stability is ensured by subtracting the maximum entry.
    """
    max_c = np.max(conv)
    shifted = conv - max_c               # ≤ 0
    scaled = gamma * shifted
    # Clip to avoid overflow when γ is large (exp(‑700) ≈ 5e‑305).
    scaled = np.clip(scaled, a_min=-700, a_max=None)
    exp_vals = np.exp(scaled)
    total = np.sum(exp_vals)
    if total == 0.0:
        # Degenerate case – fall back to uniform weights.
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / total


def _adam_step(w: np.ndarray,
               grad: np.ndarray,
               m: np.ndarray,
               v: np.ndarray,
               t: int,
               lr: float,
               beta1: float,
               beta2: float,
               eps: float) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """One Adam update of the parameters w."""
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    w_new = w - lr * m_hat / (np.sqrt(v_hat) + eps)
    return w_new, m, v


# --------------------------------------------------------------
# Core optimisation routine (one random restart)
# --------------------------------------------------------------

def _run_one_restart(n: int,
                     deadline: float,
                     rng: np.random.Generator,
                     init_a: np.ndarray | None = None) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length ``n`` using smooth‑max Adam.
    Returns the best sequence discovered and its external score.
    """
    S = math.sqrt(2.0 * n)               # target L1‑norm (Σa = √(2 n))

    # ----- initialise in log‑space ---------------------------------
    if init_a is not None:
        a0 = np.clip(init_a, 0.0, None)
        a0_sum = a0.sum()
        if a0_sum == 0.0:
            a0 = np.full(n, S / n)
        else:
            a0 = a0 * (S / a0_sum)
        eps = 1e-12
        w = np.log(np.maximum(a0, eps))
    else:
        # Occasionally start from a simple exponential shape, otherwise random.
        if rng.random() < 0.25:
            c = rng.uniform(1.5, 2.5)                # shape parameter
            a0 = np.exp(-c * np.arange(n) / n)
            a0 = a0 * (S / a0.sum())
            w = np.log(np.maximum(a0, 1e-12))
        else:
            w = rng.normal(loc=0.0, scale=0.5, size=n)

    # ----- Adam state -----------------------------------------------
    m = np.zeros_like(w)
    v = np.zeros_like(w)

    # ----- hyper‑parameters -----------------------------------------
    lr = 0.05
    beta1 = 0.9
    beta2 = 0.999

    gamma = 0.8                 # start with a smooth surrogate
    gamma_max = 120.0
    gamma_factor = 1.003        # geometric increase per iteration

    best_seq = None
    best_score = float('inf')

    # ----- evaluate the start point once ----------------------------
    a = np.exp(w)
    a = a * (S / a.sum())
    try:

    except Exception:
        pass

    it = 0
    while time.time() < deadline and it < 20000:
        it += 1

        # ---------- forward pass ----------
        a = np.exp(w)

        conv = _conv_self(a)                 # full self‑convolution

        # ---------- surrogate improvement ----------
        if cur_max < best_surrogate - 1e-12:

            try:

            except Exception:

        # ---------- gradient w.r.t. a (smooth max) ----------
        weights = _softmax_weights(conv, gamma)          # shape (2n‑1)
        grad_a = 2.0 * _valid_correlation(weights, a)    # shape (n)
        grad_w = grad_a * a                               # chain rule through a = exp(w)

        # ---------- Adam step ----------
        w, m, v = _adam_step(w, grad_w, m, v, it, lr, beta1, beta2, eps)

        # keep w in a safe numeric range
        np.clip(w, -30.0, 30.0, out=w)

        # ---------- temperature schedule ----------
        gamma = min(gamma_max, gamma * gamma_factor)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if best_seq is None:
