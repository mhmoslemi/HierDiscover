def compute_gradient_soft(p, conv, alpha):
    n = len(p)
    max_c = np.max(conv)
    diff = conv - max_c
    w_raw = np.exp(alpha * diff)  # diff ≤0, so w_raw ≤1
    w_sum = np.sum(w_raw)
    if w_sum == 0 or not np.isfinite(w_sum):
        w = np.ones_like(w_raw) / w_raw.size
    else:
        w = w_raw / w_sum
    
    w_rev = w[::-1]
    # compute convLen = 3*n-2
    conv_len = 3 * n - 2
    L = 1 << ((conv_len).bit_length())
    # compute ft of p, w, w_rev
    ft_p = np.fft.rfft(p, n=L)
    ft_w = np.fft.rfft(w, n=L)
    ft_wr = np.fft.rfft(w_rev, n=L)
    term1 = np.fft.irfft(ft_p * ft_w, n=L)[:conv_len]
    term2 = np.fft.irfft(ft_p * ft_wr, n=L)[:conv_len]
    slice_start = n - 1
    g = 2.0 * (term1[slice_start:slice_start + n] + term2[slice_start:slice_start + n])
    return g

"""
Propose a step‑function that minimises

    f(a) = 2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm works in three phases per seed:

1. **Active‑set exponentiated‑gradient descent** – the sub‑gradient of the
   ℓ∞‑norm of the convolution is taken only over the currently
   active (maximal) shifts.  An exponentiated‑gradient update keeps the
   vector non‑negative; a simplex projection enforces Σa = 1.

2. **Smooth‑max (soft‑max) gradient descent** – the hard max is replaced
   by a temperature‑controlled log‑sum‑exp.  The resulting gradient is
   obtained via two FFT‑based convolutions and a standard exponentiated‑
   gradient step with a gradually decreasing step‑size and an increasing
   temperature (larger α → tighter approximation of the true max).

3. **Stochastic hill‑climbing** – occasional random transfers of a
   small amount of mass between two coordinates are accepted only if
   they strictly improve the objective.  This helps to escape tiny
   plateaus left by the smooth optimisation.

All phases respect a global time budget (default 1000 s) and keep the
best sequence seen so far.  The routine returns the best sequence as a
plain Python list; `evaluate_sequence` can be called externally to verify
the score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _generate_powerlaw(p: float, sigma: float, max_len: int = 4000,
                       eps: float = 1e-9) -> np.ndarray:
    """Truncated power‑law a_i = (i + sigma) ** (‑p)."""
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p)
    return a[a > eps]


def _full_conv_fft(p: np.ndarray) -> np.ndarray:
    """Linear convolution p * p using a power‑of‑2 FFT."""
    n = p.shape[0]
    L = 1 << ((2 * n - 2).bit_length())
    ft = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(ft * ft, n=L)[: 2 * n - 1]
    conv[conv < 0] = 0.0               # numerical safety
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = 1} (the probability simplex)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _subgradient_active(p: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(p, p)) w.r.t. p, using only the active
    (maximal) convolution entries.
    """
    n = p.shape[0]
    max_c = conv.max()
    # tolerance – a few multiples of machine epsilon, absolute
    tol = max(1e-12, max_c * 1e-12)
    active = np.where(conv >= max_c - tol)[0]

    g = np.zeros_like(p)
    for k in active:
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        idx = np.arange(i_min, i_max + 1, dtype=np.int64)
        g[idx] += p[k - idx]
    g *= 2.0
    return g


def _gradient_softmax(p: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smooth approximation
        φ(p) = (1/α) log Σ_k exp(α * conv_k(p))
    using FFT‑based convolutions.
    """
    n = p.shape[0]
    max_c = conv.max()

    # soft‑max weights (already normalised)
    diffs = conv - max_c                     # ≤ 0
    w_raw = np.exp(alpha * diffs)           # may under‑flow → zero
    w_sum = w_raw.sum()
    if not np.isfinite(w_sum) or w_sum == 0.0:
        w = np.full_like(w_raw, 1.0 / w_raw.size)
    else:
        w = w_raw / w_sum

    w_rev = w[::-1]

    # convolution length = n + (2n‑1) – 1 = 3n‑2
    conv_len = 3 * n - 2
    L = 1 << (conv_len.bit_length())     # next power of two

    ft_p = np.fft.rfft(p, n=L)
    ft_w = np.fft.rfft(w, n=L)
    ft_wr = np.fft.rfft(w_rev, n=L)

    term1 = np.fft.irfft(ft_p * ft_w, n=L)[:conv_len]
    term2 = np.fft.irfft(ft_p * ft_wr, n=L)[:conv_len]

    start = n - 1
    grad = 2.0 * (term1[start:start + n] + term2[start:start + n])
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0):
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises `evaluate_sequence`.  The function respects the time budget
    and returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # small safety margin

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # 1) Generate a diverse pool of seeds
    # ------------------------------------------------------------------
    seeds = []

    # previous best (if present)
    if "height_sequence_1" in globals():
        seeds.append(np.asarray(height_sequence_1, dtype=np.float64))

    # power‑law families
    for p_exp in np.linspace(0.40, 0.65, 7):
        for sigma in np.logspace(-3.0, 0.5, 12):
            seq = _generate_powerlaw(p_exp, sigma, max_len=4000, eps=1e-9)
            if seq.size >= 200:
                seeds.append(seq)

    # exponential families
    for lam in np.linspace(0.0008, 0.0042, 7):
        i = np.arange(4000, dtype=np.float64)
        seq = np.exp(-lam * i)
        seq = seq[seq > 1e-9]
        if seq.size >= 200:
            seeds.append(seq)

    # uniform and random Dirichlet‑like seeds
    for n_len in (800, 1600, 3200, 4000):
        seeds.append(np.ones(n_len, dtype=np.float64))
        seeds.append(np.random.rand(n_len))

    # ------------------------------------------------------------------
    # 2) Optimise each seed
    # ------------------------------------------------------------------
    for base in seeds:
        if time.time() >= deadline:
            break

        p = np.asarray(base, dtype=np.float64)
        if p.sum() == 0.0:
            continue
        p = p / p.sum()
        if p.shape[0] > 4000:               # cap length for speed
            p = p[:4000]
            p = p / p.sum()
        n = p.shape[0]

        # initial evaluation
        conv = _full_conv_fft(p)
        cur_score = 2.0 * n * np.max(conv)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = p.tolist()
            print(f"Seed init | n={n} | score={best_score:.6f}")

        # --------------------------------------------------------------
        # Phase 1 – active‑set exponentiated gradient
        # --------------------------------------------------------------
        it = 0
        max_it1 = 3000
        while time.time() < deadline and it < max_it1:
            it += 1
            conv = _full_conv_fft(p)
            max_c = conv.max()
            cur_score = 2.0 * n * max_c

            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = p.tolist()
                print(f"Iter {it:4d} | n={n} | new best {best_score:.6f}")
                if best_score <= 1.503:
                    print("Target reached – terminating.")
                    return best_seq

            # sub‑gradient from active constraints
            g = _subgradient_active(p, conv)

            # exponentiated‑gradient step
            eta = 0.5 / np.sqrt(it)
            p = p * np.exp(-eta * g)
            p = p / p.sum()

            # occasional jitter to leave tiny plateaus
            if np.random.rand() < 0.01:
                idx = np.random.randint(n)
                delta = (np.random.rand() - 0.5) * 0.01 * p[idx]
                p[idx] = max(p[idx] + delta, 0.0)
                p = p / p.sum()

        # --------------------------------------------------------------
        # Phase 2 – smooth‑max (soft‑max) gradient descent
        # --------------------------------------------------------------
        it2 = 0
        max_it2 = 2500
        eta0 = 0.3
        alpha0 = 500.0
        while time.time() < deadline and it2 < max_it2:
            it2 += 1
            conv = _full_conv_fft(p)
            max_c = conv.max()
            cur_score = 2.0 * n * max_c

            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = p.tolist()
                print(f"SoftIter {it2:4d} | n={n} | new best {best_score:.6f}")
                if best_score <= 1.503:
                    print("Target reached – terminating.")
                    return best_seq

            # temperature (α) grows with √iteration
            alpha = alpha0 * np.sqrt(it2)

            # smooth‑max gradient
            g = _gradient_softmax(p, conv, alpha)

            eta = eta0 / np.sqrt(it2)
            p = p * np.exp(-eta * g)
            p = p / p.sum()

            # tiny random perturbation
            if np.random.rand() < 0.01:
                idx = np.random.randint(n)
                delta = (np.random.rand() - 0.5) * 0.005 * p[idx]
                p[idx] = max(p[idx] + delta, 0.0)
                p = p / p.sum()

        # --------------------------------------------------------------
        # Phase 3 – stochastic hill climbing (strict improvements)
        # --------------------------------------------------------------
        it3 = 0
        max_it3 = 5000
        while time.time() < deadline and it3 < max_it3:
            it3 += 1
            # pick two distinct indices
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            # move a small fraction of mass from i to j
            eps = 0.01 * p[i] * np.random.rand()
            if eps <= 0.0:
                continue
            p_new = p.copy()
            p_new[i] -= eps
            p_new[j] += eps
            if p_new[i] < 0.0:
                continue

            conv_new = _full_conv_fft(p_new)
            new_score = 2.0 * n * np.max(conv_new)

            if new_score < cur_score - 1e-12:
                p = p_new
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = p.tolist()
                    print(f"HillIter {it3:4d} | n={n} | new best {best_score:.6f}")
                    if best_score <= 1.503:
                        print("Target reached – terminating.")
                        return best_seq

    # ------------------------------------------------------------------
    # 3) Return the best sequence discovered
    # ------------------------------------------------------------------
    print("Final best score:", best_score, "length:", len(best_seq) if best_seq else None)
    return best_seq if best_seq is not None else []  # empty list signals failure
