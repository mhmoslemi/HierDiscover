def _run_optimize_one_start(a0, n, deadline, top_k=10, eta0=0.6, beta=0.7, decay=0.9999,
                            jitter_prob=0.02, max_iters=20000, method='exp', eps=1e-12):
  a = a0.copy()
  # ensure sum normalized
  a_sum = a.sum()
  if a_sum == 0:
      a = np.full_like(a, 1.0 / n)
  else:
      a = a / a_sum   # sum to 1

  best_score = float('inf')
  best_seq = a.copy()
  best_surrogate = float('inf')
  v = np.zeros_like(a)
  eta = eta0
  it = 0
  while time.time() < deadline and it < max_iters:
      it += 1
      conv, cur_max, max_idx = _conv_max(a)
      # select shifts near max
      if top_k is None:
          shifts = np.where(conv >= cur_max - eps)[0]
      else:
          if conv.size <= top_k:
              shifts = np.arange(conv.size)
          else:
              shifts = np.argpartition(conv, -top_k)[-top_k:]
      g = _gradient_from_shifts(a, shifts)
      v = beta * v + (1 - beta) * g
      if method == 'exp':
          a = a * np.exp(-eta * v)
      else:
          a = a - eta * v
          # ensure non-negative
          a = np.maximum(a, 0.0)
      # Clip to max 1000 to avoid evaluate clamps
      a = np.clip(a, 0.0, 1000.0)
      # Renormalize to sum = 1
      total = a.sum()
      if total == 0.0:
          a.fill(1.0 / n)
      else:
          a = a / total
      # jitter
      if np.random.rand() < jitter_prob:
          noise = np.random.randn(n) * a.mean() * 0.01
          a = np.clip(a + noise, 0.0, None)
          a = a / a.sum()
      eta *= decay
      if cur_max < best_surrogate - 1e-14:
          best_surrogate = cur_max
          try:
              sc = evaluate_sequence(a.tolist())
              if sc < best_score:
                  best_score = sc
                  best_seq = a.copy()
                  print(f"[local improve] time={time.time():.1f}s score={best_score:.6f} n={n} iter={it} cur_max={cur_max:.6f}")
          except Exception:
              pass
  return best_seq, best_score

def _local_random_search(a, deadline, rng, attempts=2000):
    best_seq = a.copy()
    try:
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        best_score = float('inf')
    n = best_seq.shape[0]
    it = 0
    while time.time() < deadline and it < attempts:
        it += 1
        # pick two indices
        i, j = rng.choice(n, size=2, replace=False)
        # propose transfer
        delta = best_seq[i] * rng.uniform(-0.3, 0.3)  # up to +-30% of i's value
        if delta == 0:
            continue
        if delta > 0:
            # move from i to j
            if best_seq[i] < delta: continue
            new_seq = best_seq.copy()
            new_seq[i] -= delta
            new_seq[j] += delta
        else:
            # move from j to i
            delta_abs = -delta
            if best_seq[j] < delta_abs: continue
            new_seq = best_seq.copy()
            new_seq[j] -= delta_abs
            new_seq[i] += delta_abs
        # keep sum unchanged and positivity
        # ensure values non-negative and not exceed 1000
        if np.any(new_seq < 0) or np.any(new_seq > 1000):
            continue
        try:
            sc = evaluate_sequence(new_seq.tolist())
        except Exception:
            continue
        if sc < best_score - 1e-12:
            best_score = sc
            best_seq = new_seq
            print(f"[local search improve] time={time.time():.1f}s score={best_score:.6f} it={it}")
    return best_seq, best_score

"""
Improved multi‑restart exponentiated / additive gradient descent.

The optimizer works on the surrogate objective
    max_k (a * a)[k]      (the maximum entry of the self‑convolution)
which is homogeneous: scaling a does not change the final score.
Consequently a is kept normalised to ℓ₁‑norm = 1 at every step.

For each restart we
  * pick a length n (2000‒12000) and a random start,
  * repeatedly compute the convolution via FFT,
  * collect the K shifts with the largest convolution values (K≈5‑12)
    and form a sub‑gradient of the max‑convolution,
  * apply either exponentiated or additive updates with momentum,
  * renormalise to ℓ₁‑norm = 1,
  * occasionally add a tiny jitter to escape plateaux,
  * call the expensive  evaluate_sequence  only when the surrogate
    maximum improves (the external score can only get better then).

A very light local random‑swap search is performed in the last few seconds
to polish the current best candidate.

The whole procedure respects the 1000‑second wall‑clock budget and returns
the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def _conv_max(a: np.ndarray) -> tuple[np.ndarray, float, int]:
    """
    Full linear self‑convolution a*a via FFT.
    Returns (conv_array, maximal value, index of that maximal value).
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()   # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    max_idx = int(np.argmax(conv))
    return conv, float(conv[max_idx]), max_idx


def _gradient_from_shifts(a: np.ndarray, shifts: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution with respect to a,
    summed over the supplied shift indices.
    For each shift k the contribution to g_j is 2·a_{k−j} whenever the index is valid.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    idx = np.arange(n)
    for k in shifts:
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g[valid] += 2.0 * a[partner[valid]]
    return g


def _run_optimize_one_start(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            top_k: int = 8,
                            eta0: float = 0.6,
                            beta: float = 0.7,
                            decay: float = 0.99995,
                            jitter_prob: float = 0.02,
                            max_iters: int = 20000,
                            method: str = "exp",
                            eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Perform a single optimisation run starting from ``a0``.
    ``method`` can be "exp" (exponentiated update) or "add" (additive update).
    Returns the best vector seen and its external score.
    """
    a = a0.copy()
    # Normalise to ℓ₁ = 1 (scaling does not affect the final score)
    if a.sum() == 0.0:
        a.fill(1.0 / n)
    else:
        a /= a.sum()

    best_score = float('inf')
    best_seq = a.copy()
    best_surrogate = float('inf')

    v = np.zeros_like(a)          # momentum buffer
    eta = eta0
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv, cur_max, _ = _conv_max(a)

        # indices of the K largest convolution entries
        if conv.size <= top_k:
            shifts = np.arange(conv.size)
        else:
            shifts = np.argpartition(conv, -top_k)[-top_k:]

        # sub‑gradient
        g = _gradient_from_shifts(a, shifts)

        # momentum
        v = beta * v + (1.0 - beta) * g

        # update
        if method == "exp":
            a = a * np.exp(-eta * v)
        else:                               # additive
            a = a - eta * v
            a = np.maximum(a, 0.0)

        # keep values inside the allowed range of evaluate_sequence
        a = np.clip(a, 0.0, 1000.0)

        # renormalise to ℓ₁ = 1
        total = a.sum()
        if total == 0.0:
            a.fill(1.0 / n)
        else:
            a /= total

        # occasional jitter
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * 0.01
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()

        eta *= decay

        # record improvements of the surrogate
        if cur_max < best_surrogate - 1e-14:
            best_surrogate = cur_max
            try:
                ext = evaluate_sequence(a.tolist())
                if ext < best_score:
                    best_score = ext
                    best_seq = a.copy()
                    print(f"[local improve] time={time.time():.1f}s  score={best_score:.6f} "
                          f"n={n} iter={it} cur_max={cur_max:.6f}")
            except Exception:
                pass

    return best_seq, best_score


def _local_random_search(a: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         attempts: int = 3000) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: repeatedly move a random small amount of mass
    from one index to another.  Accept only improvements.
    """
    best_seq = a.copy()
    try:
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        best_score = float('inf')

    n = best_seq.shape[0]
    it = 0
    while time.time() < deadline and it < attempts:
        it += 1
        i, j = rng.choice(n, size=2, replace=False)
        # propose a transfer of up to ±30 % of the source amount
        delta = best_seq[i] * rng.uniform(-0.3, 0.3)
        if delta > 0:
            if best_seq[i] < delta:
                continue
            new_seq = best_seq.copy()
            new_seq[i] -= delta
            new_seq[j] += delta
        else:
            d = -delta
            if best_seq[j] < d:
                continue
            new_seq = best_seq.copy()
            new_seq[j] -= d
            new_seq[i] += d

        if np.any(new_seq < 0) or np.any(new_seq > 1000):
            continue
        try:
            sc = evaluate_sequence(new_seq.tolist())
        except Exception:
            continue
        if sc < best_score - 1e-12:
            best_score = sc
            best_seq = new_seq
            print(f"[local search] time={time.time():.1f}s  score={best_score:.6f} it={it}")
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) yields a score
    lower than the current best bound of 1.503.
    The routine spends the allotted wall‑clock time on many independent
    restarts of a gradient‑descent based optimiser.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # Record any pre‑supplied construction
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            if cand.ndim == 1 and cand.size > 0:
                sc = evaluate_sequence(cand.tolist())
                best_seq = cand.copy()
                best_score = sc
                print(f"[baseline] existing height_sequence_1 score = {best_score:.6f} length = {cand.size}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Search loop
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000, 12000]
    run_id = 0

    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 5.0:
            break

        n = int(rng.choice(lengths))

        # initialise – either reuse a prefix of the current best or draw fresh
        if best_seq is not None and best_seq.size >= n and rng.random() < 0.25:
            a0 = best_seq[:n].copy()
        else:
            a0 = rng.random(n)

        a0 = a0 / a0.sum()   # ℓ₁‑normalise

        # randomise hyper‑parameters for diversity
        top_k = int(rng.choice([3, 5, 8, 12]))
        eta0 = rng.uniform(0.3, 0.8)
        beta = rng.uniform(0.5, 0.9)
        method = rng.choice(['exp', 'add'])

        # rough budget for this run (more iterations for larger remaining time)
        max_iters = int(min(20000, remaining * 25))

        a_opt, score_opt = _run_optimize_one_start(
            a0, n, deadline,
            top_k=top_k,
            eta0=eta0,
            beta=beta,
            decay=0.99995,
            jitter_prob=0.02,
            max_iters=max_iters,
            method=method,
            eps=1e-12
        )

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[global improve] run={run_id} n={n} score={best_score:.6f} "
                  f"top_k={top_k} method={method}")

        run_id += 1

    # ------------------------------------------------------------------
    # Final short‑range local search (uses the remaining few seconds)
    # ------------------------------------------------------------------
    if best_seq is not None:
        best_seq, best_score = _local_random_search(best_seq, deadline, rng)

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, 1.0 / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] score={best_score:.6f}")

    print(f"[final] best score = {best_score:.6f}  length = {len(best_seq)}")
    return best_seq.tolist()
