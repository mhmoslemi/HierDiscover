    2 * n * max(conv(a, a)) / (sum(a)**2)

The search consists of three stages:

1. **Broad parametric sweep** – try several lengths (including the provided
   baseline) and a few one‑parameter families (inverse‑power, exponential,
   uniform).  The best sequence so far is kept.

2. **Smoothed Adam optimisation** – minimise the smoothed objective
   𝔼τ[ max(conv) ] using a log‑sum‑exp smoothing.  The gradient of the
   smoothed max can be obtained by a single FFT‑based convolution.
   Adam’s adaptive steps, a slowly decaying learning‑rate and a
   decreasing smoothing temperature τ guide the search.  Random pair
   transfers and occasional jitter help escaping plateaus.

3. **Polishing** – tiny random mass transfers and occasional swaps are
   applied; any improvement is retained.

All operations are O(n log n) and respect the 1000 s time budget.

import time

# Helper utilities
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum x = target}."""
    if target <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, target / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself using an FFT of power‑of‑two size."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of x and y (FFT, power‑of‑two size)."""
    n = len(x) + len(y) - 1
    L = 1 << (n.bit_length())
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)
    return conv[:n]


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """Transfer a tiny fraction of the mass from one index to another and re‑project."""
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def gradient_step_adam(
    a: np.ndarray,
    target_sum: float,
    lr: float,
    beta1: float,
    beta2: float,
    eps: float,
    m: np.ndarray,
    v: np.ndarray,
    t: int,
    tau_factor: float,
) -> tuple[np.ndarray, float, np.ndarray, np.ndarray]:
    """
    Perform one Adam step on the smoothed max‑convolution objective.

    Returns:
        a_new      – updated coefficient vector
        max_conv   – current (unsmoothed) max of a*a
        m, v       – updated Adam moments
    """
    n = a.shape[0]

    # 1) self‑convolution and its maximum
    conv = conv_self_fft(a)
    max_conv = conv.max()

    # 2) smoothing weights (log‑sum‑exp)
    tau = max(tau_factor * max_conv, 1e-12)
    w = np.exp((conv - max_conv) / tau)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # 3) gradient of the smoothed max
    a_rev = a[::-1]
    conv_w_rev = conv_full_fft(w, a_rev)          # length = (2n‑1)+n‑1 = 3n‑2
    grad = (2.0 / tau) * conv_w_rev[n - 1 : n - 1 + n]

    # 4) Adam update
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = project_onto_simplex(np.maximum(a_new, 0.0), target_sum)

    return a_new, max_conv, m, v
