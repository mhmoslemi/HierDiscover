    2·n·max(conv(a,a)) / (sum(a)^2)

We parametrize a via a softmax (u) to keep positivity and unit ℓ¹ norm.
A smooth approximation of the maximum (soft‑max with temperature β) yields a
differentiable surrogate.  An Adam optimiser on u is applied; β is raised
gradually so the surrogate converges to the true max.  Periodic pruning of
tiny entries reduces the length n (which multiplies the objective).  Several
restarts are performed within the 1000 s budget, always keeping the best
sequence found.
"""

def _generate_powerlaw(p: float, sigma: float,
                       max_len: int = 5000, eps: float = 1e-10) -> np.ndarray:
    """Generate a power‑law sequence a_i = (i+σ)^{-p} truncated at eps."""
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p)
    return a[a > eps]


def _simple_softmax(u: np.ndarray) -> np.ndarray:
    """Return soft‑max of u (positives, sums to 1)."""
    # subtract max for numerical stability
    u_shift = u - np.max(u)
    exp_u = np.exp(u_shift)
    return exp_u / exp_u.sum()


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    Fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(Fa * Fa, size)[: 2 * n - 1]
    return conv


def _smooth_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights proportional to exp(β·conv)."""
    # stable formulation
    bc = beta * conv
    max_bc = np.max(bc)
    w = np.exp(bc - max_bc)
    w /= w.sum()
    return w


def _grad_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∂/∂a  ( Σ_k w_k (a*a)_k ) = 2·corr(w,a).
    Returns a vector of length len(a).
    """
    n = a.shape[0]
    len_w = w.shape[0]          # = 2n-1
    # linear correlation (cross‑correlation) of w with a
    size = 1 << ((len_w + n - 1 - 1).bit_length())
    Fa = np.fft.rfft(a, size)
    Fw = np.fft.rfft(w, size)
    corr = np.fft.irfft(Fw * np.conj(Fa), size)[: len_w + n - 1]
    # slice the central part: indices n‑1 … n‑1+n‑1
    g = 2.0 * corr[n - 1 : n - 1 + n]
    return g


def _objective_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact objective given convolution and length (sum == 1)."""
    return float(2.0 * n * conv.max())


def _initial_sequence() -> np.ndarray:

    # Prefer the globally supplied construction if present.
    if "height_sequence_1" in globals():

    if seq.size > 2500:
        seq = seq[:2500]
    return seq / seq.sum()


def _optimize(initial_a: np.ndarray, time_limit: float):
    """
    Optimise a starting probability vector `initial_a` for at most
    `time_limit` seconds.  Returns the best vector found and its objective.
    """
    start = time.time()
    a_len = initial_a.shape[0]
    # soft‑max parametrisation
    u = np.log(initial_a + 1e-12)
    m = np.zeros_like(u)
    v = np.zeros_like(u)

    beta = 5.0          # temperature for the smooth max
    beta_max = 200.0
    beta_factor = 1.03

    lr = 0.05           # base learning‑rate for Adam
    eps = 1e-8
    beta1 = 0.9
    beta2 = 0.999

    best_a = initial_a.copy()
    best_score = _objective_from_conv(_conv_fft(best_a), a_len)

    iteration = 0
    # Main optimisation loop
    while time.time() - start < time_limit:
        iteration += 1

        # ------------------------------------------------------------------
        # 1. obtain current simplex point from u
        # ------------------------------------------------------------------
        a = _simple_softmax(u)            # guarantees non‑negative, sum=1
        n = a.shape[0]

        # ------------------------------------------------------------------
        # 2. convolution and smooth max weights
        # ------------------------------------------------------------------
        conv = _conv_fft(a)
        w = _smooth_weights(conv, beta)

        # ------------------------------------------------------------------
        # 3. gradient w.r.t. a and chain‑rule back to u
        # ------------------------------------------------------------------
        grad_a = _grad_wrt_a(a, w)                # ∂L/∂a
        dot = grad_a @ a
        grad_u = a * (grad_a - dot)               # ∂L/∂u (soft‑max Jacobian)

        # ------------------------------------------------------------------
        # 4. Adam update on u (unconstrained)
        # ------------------------------------------------------------------
        m = beta1 * m + (1 - beta1) * grad_u
        v = beta2 * v + (1 - beta2) * (grad_u * grad_u)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        u = u - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep the scale of u bounded (otherwise exp over/under‑flows)
        u = u - np.mean(u)

        # ------------------------------------------------------------------
        # 5. slowly increase β so the surrogate approaches the true max
        # ------------------------------------------------------------------
        if iteration % 100 == 0 and beta < beta_max:
            beta = min(beta_max, beta * beta_factor)

        # ------------------------------------------------------------------

        # ------------------------------------------------------------------

            cur_score = _objective_from_conv(conv, n)
            if cur_score < best_score:

                best_a = a.copy()
                print(f'Iter {iteration:6d} → new best {best_score:.6f} (n={n})')
                if best_score < 1.503:

        # ------------------------------------------------------------------

                # re‑initialise the optimisation variables for the new size
                u = np.log(a_pruned + 1e-12)
                m = np.zeros_like(u)
                v = np.zeros_like(u)
                # no need to continue this iteration – the loop restarts with
                # the compressed vector
                continue

        # ------------------------------------------------------------------
        # 8. small random jitter to avoid stagnation
        # ------------------------------------------------------------------
        if np.random.rand() < 0.005:
            u += 0.02 * np.random.randn(*u.shape)

def _generate_powerlaw(p: float, sigma: float,
                       max_len: int = 5000, eps: float = 1e-10) -> np.ndarray:
    """Generate a power‑law sequence a_i = (i+σ)^{-p} truncated at eps."""
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p)
    return a[a > eps]


def _simple_softmax(u: np.ndarray) -> np.ndarray:
    """Return soft‑max of u (positive, sums to 1)."""
    u_shift = u - np.max(u)                     # stabilise exp
    exp_u = np.exp(u_shift)
    return exp_u / exp_u.sum()


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    Fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(Fa * Fa, size)[: 2 * n - 1]
    return conv


def _smooth_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights proportional to exp(β·conv)."""
    bc = beta * conv
    max_bc = np.max(bc)
    w = np.exp(bc - max_bc)
    w /= w.sum()
    return w


def _grad_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∂/∂a  ( Σ_k w_k (a*a)_k ) = 2·corr(w,a).
    Returns a vector of length len(a).
    """
    n = a.shape[0]
    len_w = w.shape[0]          # = 2n-1
    size = 1 << ((len_w + n - 1 - 1).bit_length())
    Fa = np.fft.rfft(a, size)
    Fw = np.fft.rfft(w, size)
    corr = np.fft.irfft(Fw * np.conj(Fa), size)[: len_w + n - 1]
    g = 2.0 * corr[n - 1 : n - 1 + n]
    return g


def _objective_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact objective assuming sum(a)=1."""
    return float(2.0 * n * conv.max())


def _initial_sequence() -> np.ndarray:
    """Return a reasonable start vector (unit ℓ¹ norm)."""
    if "height_sequence_1" in globals():
        arr = np.array(height_sequence_1, dtype=float)
        arr = arr[arr > 1e-12]               # drop ultra‑tiny entries
        if arr.size > 2500:
            arr = arr[:2500]                  # cap length
        if arr.size < 2:
            arr = np.ones(100)
        return arr / arr.sum()

    seq = _generate_powerlaw(p=0.5, sigma=1e-3, max_len=4000, eps=1e-12)
    if seq.size > 2500:
        seq = seq[:2500]
    return seq / seq.sum()


def _optimize(initial_a: np.ndarray, time_limit: float):
    """
    Optimise a starting probability vector `initial_a` for at most
    `time_limit` seconds.  Returns the best vector found and its objective.
    """
    start = time.time()
    u = np.log(initial_a + 1e-12)               # soft‑max parametrisation
    m = np.zeros_like(u)                       # Adam first moment
    v = np.zeros_like(u)                       # Adam second moment

    beta = 5.0          # temperature for the smooth max
    beta_max = 200.0
    beta_factor = 1.03

    lr = 0.05
    eps = 1e-8
    beta1 = 0.9
    beta2 = 0.999

    best_a = initial_a.copy()
    best_score = _objective_from_conv(_conv_fft(best_a), best_a.shape[0])

    iteration = 0
    while time.time() - start < time_limit:
        iteration += 1

        # ---- 1. Current point on the simplex --------------------------------
        a = _simple_softmax(u)
        n = a.shape[0]

        # ---- 2. Convolution and smooth‑max weights -------------------------
        conv = _conv_fft(a)
        w = _smooth_weights(conv, beta)

        # ---- 3. Gradient wrt a and back‑propagation to u --------------------
        grad_a = _grad_wrt_a(a, w)          # ∂L/∂a
        dot = grad_a @ a
        grad_u = a * (grad_a - dot)         # Jacobian of soft‑max

        # ---- 4. Adam update -------------------------------------------------
        m = beta1 * m + (1 - beta1) * grad_u
        v = beta2 * v + (1 - beta2) * (grad_u * grad_u)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        u = u - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep u centred to avoid overflow
        u = u - np.mean(u)

        # ---- 5. Increase temperature gradually -------------------------------
        if iteration % 100 == 0 and beta < beta_max:
            beta = min(beta_max, beta * beta_factor)

        # ---- 6. Exact evaluation (every 10 iterations) -----------------------
        if iteration % 10 == 0:
            cur_score = _objective_from_conv(conv, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f'Iter {iteration:6d} → new best {best_score:.6f} (n={n})')
                if best_score < 1.503:
                    break

        # ---- 7. Prune near‑zero entries (every 200 iterations) -------------
        if iteration % 200 == 0:
            mask = a > 1e-6
            if mask.sum() >= 2 and mask.sum() < n:
                a_pruned = a[mask]
                u = np.log(a_pruned + 1e-12)
                m = np.zeros_like(u)
                v = np.zeros_like(v)
                continue

        # ---- 8. Small random jitter to shake off plateaus -------------------
        if np.random.rand() < 0.005:
            u += 0.02 * np.random.randn(*u.shape)

    return best_a, best_score


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list:
    """
    Search for a non‑negative step sequence that minimises the supplied
    `evaluate_sequence`.  The routine respects a 1000 s wall‑clock budget
    and returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Phase 0 – obtain a sensible start vector
    # ------------------------------------------------------------------
    init_seq = _initial_sequence()
    best_seq = init_seq.tolist()
    best_score = _objective_from_conv(_conv_fft(init_seq), init_seq.shape[0])
    print(f'Initial candidate – score {best_score:.6f}, length {len(best_seq)}')

    # ------------------------------------------------------------------
    # Phase 1 – main optimisation (≈80 % of the time budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    primary_time = min(800.0, remaining * 0.8)
    if primary_time > 5.0:
        opt_seq, opt_score = _optimize(init_seq, primary_time)
        if opt_score < best_score:
            best_score, best_seq = opt_score, opt_seq.tolist()
            print(f'After primary optimisation → score {best_score:.6f}')

    # ------------------------------------------------------------------
    # Phase 2 – several short restarts (mix of best & random directions)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 20.0:
            break

        cur_len = len(best_seq)
        rnd = np.random.rand(cur_len)
        rnd /= rnd.sum()
        start_vec = 0.7 * np.array(best_seq) + 0.3 * rnd
        start_vec /= start_vec.sum()

        slice_time = min(30.0, remaining * 0.5)
        opt_seq, opt_score = _optimize(start_vec, slice_time)
        if opt_score < best_score:
            best_score, best_seq = opt_score, opt_seq.tolist()
            print(f'New overall best {best_score:.6f} (len {len(best_seq)})')
            if best_score < 1.503:
                break

    print('Final best score:', best_score, 'length:', len(best_seq))
    return best_seq
