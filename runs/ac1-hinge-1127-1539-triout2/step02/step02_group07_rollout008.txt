def _softmax_gradient(p, conv, beta):
    n = p.shape[0]
    max_c = conv.max()
    # compute softmax weights
    w = np.exp(beta * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # reverse w
    w_rev = w[::-1]

    corr_len = n + w_rev.size - 1
    corr = corr[:corr_len]
    start = w_rev.size - 1
    idx = start - np.arange(n)
    g = 2.0 * corr[idx]
    return g

"""
Non‑negative step‑function optimizer (improved).

The optimizer works in two phases:

1.  A very diverse pool of *parameterised* seeds is generated
    (power‑law, exponential, combined, arcsine‑like, sine‑window,
    uniform, random Dirichlet and the previous best sequence if
    available).  Each seed is normalised to a probability vector
    (∑a_i = 1) and its score 2·n·max(conv(a,a)) is evaluated.

2.  The best seed is refined by a *soft‑max weighted* exponentiated‑
    gradient (mirror‑descent) scheme.  The gradient of the smooth
    approximation ϕβ(conv)= (1/β)·log∑exp(β·conv_i) is computed by
    correlating the current probability vector with the soft‑max
    weights (FFT‑based, O(N log N)).  An adaptive step‑size together
    with a simple line‑search guarantees monotone improvement.
    Small entries are periodically pruned — this reduces the length
    factor n in the objective while keeping the convolution‑peak
    essentially unchanged.

The routine respects a 1000 s budget (a small safety margin is kept)
and returns the best sequence discovered as a plain Python list.
All helper functions are defined at the top level, no closures or
lambda‑expressions are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def _full_conv_fft(p: np.ndarray) -> np.ndarray:
    """Linear convolution of a vector with itself (full length)."""
    n = p.shape[0]
    L = 1 << ((2 * n - 2).bit_length())      # next power of two
    ft = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(ft * ft, n=L)[: 2 * n - 1]
    conv = np.real(conv)
    conv[conv < 0] = 0.0                      # clip tiny negatives
    return conv


def _compute_score(p: np.ndarray) -> float:
    """Exact evaluation of the target function (scale‑invariant)."""
    n = p.shape[0]
    s = p.sum()
    if s < 0.01:               # protection – should never happen for our seeds
        return np.inf
    conv = _full_conv_fft(p)
    max_b = conv.max()
    return 2.0 * n * max_b / (s * s)


def _softmax_gradient(p: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth max (soft‑max) of the convolution.
    Implements   g = 2 * correlation(w, p)   where
    w_i ∝ exp(beta * (conv_i - max(conv))) and sum_i w_i = 1.
    The correlation is carried out via FFT for O(N log N) cost.
    """
    n = p.shape[0]
    max_c = conv.max()
    # soft‑max weights (normalised)
    w = np.exp(beta * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # correlation of w and p:
    # g_k = 2 * Σ_i w_i * p_{i‑k}
    w_rev = w[::-1]                     # reversed weight vector
    L = 1 << ((n + w_rev.size - 1).bit_length())
    ft_p = np.fft.rfft(p, n=L)
    ft_w = np.fft.rfft(w_rev, n=L)
    corr = np.fft.irfft(ft_p * ft_w, n=L)
    corr_len = n + w_rev.size - 1
    corr = corr[:corr_len]               # linear correlation result

    start = w_rev.size - 1               # index that aligns with k = 0
    g = 2.0 * corr[start - np.arange(n)]   # pick the required slice
    return g


def _quick_refine(p: np.ndarray,
                  max_iters: int,
                  deadline: float,
                  init_beta: float = 5.0,
                  init_eta: float = 0.4) -> tuple[np.ndarray, float]:
    """
    A short exponentiated‑gradient run (few hundred iterations) used
    to improve a candidate before the final refinement.
    Returns the best vector found and its score.
    """
    best_p = p.copy()
    best_score = _compute_score(p)
    beta = init_beta
    eta0 = init_eta
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _full_conv_fft(p)
        g = _softmax_gradient(p, conv, beta)
        eta = eta0 / np.sqrt(it)

        # simple back‑tracking line‑search (up to 5 reductions)
        for _ in range(5):
            p_new = p * np.exp(-eta * g)
            s = p_new.sum()
            if s == 0 or not np.isfinite(s):
                break
            p_new /= s
            sc = _compute_score(p_new)
            if sc < best_score:
                best_score = sc
                best_p = p_new.copy()
                p = p_new
                break          # accept and continue
            eta *= 0.5         # shrink step and retry

        # occasional random jitter to escape tiny plateaus
        if np.random.rand() < 0.01:
            idx = np.random.randint(p.shape[0])
            delta = (np.random.rand() - 0.5) * 0.02 * p[idx]
            p[idx] = max(p[idx] + delta, 0.0)
            p /= p.sum()

        # slowly increase the smoothing parameter
        if it % 100 == 0:
            beta = min(beta * 1.1, 200.0)

    return best_p, best_score


def _full_refine(p: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Long‑running refinement that uses the same exponentiated‑gradient
    scheme as `_quick_refine`, but with adaptive pruning of negligible
    entries (reducing the length factor n) and a more aggressive
    increase of the soft‑max temperature.
    """
    best_p = p.copy()
    best_score = _compute_score(p)
    beta = 5.0
    eta0 = 0.5
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _full_conv_fft(p)
        g = _softmax_gradient(p, conv, beta)
        eta = eta0 / np.sqrt(it)

        # line search – at most 4 reductions
        for _ in range(4):
            p_new = p * np.exp(-eta * g)
            s = p_new.sum()
            if s == 0 or not np.isfinite(s):
                break
            p_new /= s

            # prune very small components every few hundred steps
            if it % 300 == 0:
                thr = p_new.max() * 1e-5
                mask = p_new > thr
                if mask.sum() >= 2 and mask.sum() < p_new.shape[0]:
                    p_new = p_new[mask]
                    p_new /= p_new.sum()

            sc = _compute_score(p_new)
            if sc < best_score:
                best_score = sc
                best_p = p_new.copy()
                p = p_new
                break
            eta *= 0.5

        # tiny random perturbation if no progress
        if np.random.rand() < 0.008:
            idx = np.random.randint(p.shape[0])
            delta = (np.random.rand() - 0.5) * 0.015 * p[idx]
            p[idx] = max(p[idx] + delta, 0.0)
            p /= p.sum()

        # increase temperature slowly – larger β focuses more on the
        # true max as optimisation proceeds
        if it % 250 == 0:
            beta = min(beta * 1.15, 300.0)

    return best_p, best_score
