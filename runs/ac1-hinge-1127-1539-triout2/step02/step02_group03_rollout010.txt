    # multiplicative update: a_new = a * exp(-eta * grad)
    a_new = a * np.exp(-eta * grad)

    if total == 0:
        a_new = np.full_like(a, S / len(a))
    else:
        a_new = a_new * (S / total)
    return a_new

def _lp_step(a):
   n = len(a)
   M = max_conv(a)[0]   # max value
   # Variables: b (size n). Objective: maximize sum b -> minimize -np.ones(n). 
   # Constraints: for each k in 0..2n-2: sum_{j} a[j] * b[k-j] <= M
   # Build A_ub shape (2n-1, n) with many zero's.
   # Use dense matrix maybe also okay if n <= 1500.

   A = np.zeros((2*n-1, n))
   for i in range(n):
       A[i:i+n, i] = a  # because for each column i (b_i), coefficient at rows i..i+n-1 is a
   # That fills A.
   b_ub = np.full(2*n-1, M)
   bounds = [(0, None)] * n
   res = linprog(c=-np.ones(n), A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
   if not res.success:
       return a  # fallback
   b_opt = res.x
   # Scale to L1 S
   S = np.sqrt(2.0 * n)
   sum_b = b_opt.sum()
   if sum_b == 0:
       # fallback uniform
       b_opt = np.full(n, S/n)
   else:
       b_opt = b_opt * (S / sum_b)
   # Mix with a: t maybe 0.2
   t = 0.2
   a_new = (1 - t) * a + t * b_opt
   return a_new

def _smooth_gradient_step(a, p, eta):
   n = a.shape[0]
   conv = np.convolve(a, a)
   w = conv ** (p - 1)
   rev_a = a[::-1]
   conv_rev = np.convolve(rev_a, w)  # full
   offset = n - 1
   dot = conv_rev[offset: offset + n]  # length n
   # diag indices 2*i
   diag = w[2 * np.arange(n)]
   grad = 2 * p * dot - p * diag * a
   # clipping
   grad = np.clip(grad, -100, 100)  # maybe not needed
   a_new = a * np.exp(-eta * grad)
   # enforce non-negativity
   a_new = np.maximum(a_new, 0.0)
   # renorm
   S = np.sqrt(2.0 * n)
   total = a_new.sum()
   if total == 0.0:
       a_new = np.full(n, S / n)
   else:
       a_new *= (S / total)
   return a_new

def _smooth_gradient_step_fft(a, p, eta):
    n = a.shape[0]
    # compute convolution a*a via FFT
    conv_len = 2*n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa)[:conv_len]  # shape 2n-1

    w = conv ** (p - 1)
    # compute convolution of a (size n) with w (size 2n-1)
    conv_len2 = n + (2*n-1) - 1  # 3n-2
    fft_len2 = 1 << (conv_len2 - 1).bit_length()
    Fa2 = np.fft.rfft(a, n=fft_len2)
    Fw = np.fft.rfft(w, n=fft_len2)
    conv_aw = np.fft.irfft(Fa2 * Fw)[:conv_len2]  # full conv
    # dot_i = conv_aw[i + n - 1]
    offset = n - 1
    dot = conv_aw[offset:offset + n]  # length n

    diag = conv[2 * np.arange(n)]  # conv[2i]
    grad = 2 * p * dot - p * diag * a
    # clip gradient for safety
    grad = np.clip(grad, -50, 50)  # adjust heavy
    a_new = a * np.exp(-eta * grad)
    # ensure nonneg and renorm
    a_new = np.maximum(a_new, 0.0)
    S = np.sqrt(2.0 * n)
    total = a_new.sum()
    if total == 0.0:
        a_new = np.full(n, S / n)
    else:
        a_new *= (S / total)
    return a_new

"""
Improved search for a step‑function minimizing the ∞‑norm of its
self‑convolution.

The algorithm combines three ideas:

1. **Smooth p‑norm descent** – we minimise  Σ (a*a)_k^p for a
   sequence of increasing exponents p (2 → 10).  The gradient of this
   smooth objective is computed with FFTs, allowing many cheap updates.
   A multiplicative update together with L1‑normalisation keeps the
   vector non‑negative and of the required sum √(2·n).

2. **Linear‑programming projection** – after each p‑stage we solve a
   small LP that maximises the total mass of a new vector b under the
   constraint (a*b)_k ≤ current maximal convolution value M.
   The solution pushes mass into positions where a is small, flattening
   the convolution.

3. **Final local refinement** – a short run of the earlier
   multiplicative sub‑gradient / simulated‑annealing routine cleans up
   any remaining peaks.

Many independent restarts of different lengths are performed within the
time budget; the best sequence found is returned.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper functions (top level)
# ----------------------------------------------------------------------


def _conv_max(a: np.ndarray) -> tuple[np.ndarray, float, int]:
    """
    Compute the full linear convolution a*a using an FFT,
    returning the convolution array, its maximum value and the index
    where the maximum occurs.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa)[:conv_len]
    conv = np.maximum(conv, 0.0)            # guard against tiny negatives
    max_idx = int(np.argmax(conv))
    return conv, float(conv[max_idx]), max_idx


def _gradient_step(a: np.ndarray, eta: float) -> np.ndarray:
    """
    One multiplicative sub‑gradient step that reduces the current
    maximal entry of the self‑convolution.
    """
    n = a.shape[0]
    _, _, k = _conv_max(a)

    idx = np.arange(n)
    partner = k - idx
    valid = (partner >= 0) & (partner < n)

    g = np.zeros_like(a)
    g[valid] = 2.0 * a[partner[valid]]

    if k % 2 == 0:
        d = k // 2
        if 0 <= d < n:
            g[d] = a[d]                     # diagonal contribution

    a_new = a * np.exp(-eta * g)
    total = a_new.sum()
    S = np.sqrt(2.0 * n)
    if total == 0.0:
        a_new = np.full(n, S / n)
    else:
        a_new *= (S / total)
    return a_new


def _initial_sequence(base_seq: np.ndarray,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """
    Produce an initial non‑negative sequence of length n and L1‑norm √(2·n).
    """
    S = np.sqrt(2.0 * n)

    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        a += rng.normal(scale=a.mean() * 0.001, size=n)
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)

    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0)
        a_sum = a.sum()
    a = a / a_sum * S
    return a


def _smooth_gradient_step_fft(a: np.ndarray,
                              p: float,
                              eta: float) -> np.ndarray:
    """
    One multiplicative step for the smooth objective
        L_p(a) = Σ_k ( (a*a)_k )^p .
    The gradient is computed via FFTs for speed.
    """
    n = a.shape[0]

    # ----- convolution a*a -----
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa)[:conv_len]
    conv = np.maximum(conv, 0.0)                 # avoid tiny negatives

    # ----- weight = conv^{p-1} -----
    w = conv ** (p - 1)

    # ----- convolution a * w -----
    conv_len2 = n + (2 * n - 1) - 1              # = 3n‑2
    fft_len2 = 1 << (conv_len2 - 1).bit_length()
    Fa2 = np.fft.rfft(a, n=fft_len2)
    Fw = np.fft.rfft(w, n=fft_len2)
    conv_aw = np.fft.irfft(Fa2 * Fw)[:conv_len2]

    # dot_i = Σ_j a_j w_{i+j} = conv_aw[i + n‑1]
    offset = n - 1
    dot = conv_aw[offset:offset + n]

    # diagonal part (k = 2i)
    diag = conv[2 * np.arange(n)]

    # gradient of Σ conv^p
    grad = 2.0 * p * dot - p * diag * a

    # smooth the update to avoid overflow/underflow
    grad = np.clip(grad, -50.0, 50.0)

    a_new = a * np.exp(-eta * grad)
    a_new = np.maximum(a_new, 0.0)

    # re‑normalise to the required L1‑norm √(2·n)
    S = np.sqrt(2.0 * n)
    total = a_new.sum()
    if total == 0.0:
        a_new = np.full(n, S / n)
    else:
        a_new *= (S / total)
    return a_new


def _lp_step(a: np.ndarray) -> np.ndarray:
    """
    Solve the LP:
        maximise   Σ b_j
        subject to b_j ≥ 0,
                   (a * b)_k ≤ M   for all k,
    where M is the current maximal self‑convolution entry.
    The solution is mixed back into a.
    """
    n = a.shape[0]
    conv, M, _ = _conv_max(a)

    # Build the matrix A where each column i appears on rows i … i+n‑1
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = a

    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    # maximise Σ b ⇒ minimise –Σ b
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=b_ub,
                  bounds=bounds, method='highs')
    if not res.success:
        # fall back to the current vector if LP fails
        return a

    b_opt = res.x
    S = np.sqrt(2.0 * n)
    sum_b = b_opt.sum()
    if sum_b == 0.0:
        b_opt = np.full(n, S / n)
    else:
        b_opt = b_opt * (S / sum_b)

    # mix a little of the LP‑solution back into a
    t = 0.2
    a_new = (1.0 - t) * a + t * b_opt
    a_new = np.maximum(a_new, 0.0)
    a_new = a_new / a_new.sum() * S
    return a_new


def _smooth_optimize(a: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run the smooth‑p optimisation (p = 2,4,6,8,10) followed by a
    single LP projection.  Returns the best sequence seen and its
    maximal convolution value.
    """
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_max = _conv_max(a)[1]

    schedule = [
        (2.0, 0.50, 0.995, 55),
        (4.0, 0.30, 0.995, 55),
        (6.0, 0.20, 0.995, 55),
        (8.0, 0.15, 0.995, 55),
        (10.0, 0.10, 0.995, 55),
    ]

    for p, eta, decay, iters in schedule:
        for _ in range(iters):
            if time.time() >= deadline:
                break

            a = _smooth_gradient_step_fft(a, p, eta)

            # occasional tiny jitter to escape plateaus
            if rng.random() < 0.01:
                noise = rng.normal(scale=a.mean() * 0.01, size=n)
                a = np.clip(a + noise, 0.0, None)
                a = a / a.sum() * S

            cur_max = _conv_max(a)[1]
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()

            eta *= decay

        # one LP projection after each p‑stage
        if time.time() < deadline:
            a = _lp_step(a)
            cur_max = _conv_max(a)[1]
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()

    return best_a, best_max


def _optimize_local(a: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Short local optimiser (the method from the previous state‑of‑the‑art).
    Mixes multiplicative sub‑gradient steps with simulated‑annealing
    random swaps and occasional jitter.
    """
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    conv, cur_max, cur_idx = _conv_max(a)
    best_a = a.copy()
    best_score = cur_max

    eta = 0.6
    decay = 0.9999
    move_prob = 0.15
    epsilon_move = 0.05
    jitter_prob = 0.01
    temp = cur_max * 0.1
    temp_decay = 0.9997

    while time.time() < deadline:
        if rng.random() < move_prob:
            partner = cur_idx - np.arange(n)
            valid = (partner >= 0) & (partner < n)
            i_candidates = np.where(valid)[0]
            j_candidates = np.where(~valid)[0]

            if i_candidates.size and j_candidates.size:
                i = rng.choice(i_candidates)
                j = rng.choice(j_candidates)
                delta = epsilon_move * a[i]
                delta = min(delta, a[i])

                a_candidate = a.copy()
                a_candidate[i] -= delta
                a_candidate[j] += delta

                _, cand_max, cand_idx = _conv_max(a_candidate)

                accept = False
                if cand_max <= cur_max:
                    accept = True
                else:
                    prob = np.exp(-(cand_max - cur_max) / max(temp, 1e-12))
                    if rng.random() < prob:
                        accept = True

                if accept:
                    a = a_candidate
                    cur_max = cand_max
                    cur_idx = cand_idx
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                temp *= temp_decay
        else:
            a = _gradient_step(a, eta)
            _, cur_max, cur_idx = _conv_max(a)
            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()

        if rng.random() < jitter_prob:
            noise = rng.normal(scale=a.mean() * 0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = a / a.sum() * S

        eta *= decay

    return best_a, best_score


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Entry point required by the evaluation harness.
    Performs many independent short searches (different lengths,
    random seeds, and initialisations) within the allotted time and
    returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # small safety margin

    # try to reuse the previously discovered construction
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_overall_seq = None
    best_overall_score = np.inf
    attempt = 0

    # main optimisation loop
    while time.time() < deadline:
        attempt += 1

        # choose a problem size – we explore a reasonable range
        if base_seq is not None:
            max_len = max(800, min(2500, base_seq.size))
            n = rng.integers(800, max_len + 1)
        else:
            n = rng.integers(800, 2500)

        # initialise a candidate
        a0 = _initial_sequence(base_seq, n, rng)

        # give each run a time slice (≈ 25 s)
        slice_deadline = min(time.time() + 25.0, deadline)

        # 1️⃣ smooth p‑norm descent + LP projection
        a_opt, score = _smooth_optimize(a0, slice_deadline, rng)

        # 2️⃣ final polishing with the older local optimiser (if time left)
        if time.time() < slice_deadline:
            a_opt, score_local = _optimize_local(a_opt, slice_deadline, rng)
            if score_local < score:
                score = score_local

        # keep the best overall result
        if score < best_overall_score:
            best_overall_score = score
            best_overall_seq = a_opt.copy()
            print(f"[global] attempt={attempt} n={n} new best score={best_overall_score:.6f}")

    # final sanity check – evaluate through the provided black‑box function
    if best_overall_seq is not None:
        final_seq = best_overall_seq.tolist()
        external_score = evaluate_sequence(final_seq)
        print(f"[final] best external score = {external_score:.6f} "
              f"(internal max = {best_overall_score:.6f}) "
              f"length = {len(final_seq)}")
        return final_seq
    else:
        # fallback – should never happen
        return [1.0]
