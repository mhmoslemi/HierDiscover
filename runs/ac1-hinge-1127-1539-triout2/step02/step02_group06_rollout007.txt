def adam_update(a, grad, m, v, t_iter, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad * grad)
    m_hat = m / (1 - beta1**t_iter)
    v_hat = v / (1 - beta2**t_iter)

    a_new = a - step
    return a_new, m, v

def fft_convolve(a, b):

    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)
    return conv[: n + m - 1]

def conv_self_fft(a):
    return fft_convolve(a, a)

def project_to_sum(v, target):
    s = v.sum()
    if s != 0:
        return v * (target / s)
    else:
        # fallback uniform distribution
        return np.full_like(v, target / v.shape[0])

def generate_shape(n, target_sum, kind='linear', param=0.0, direction='symmetric'):
    idx = np.arange(n, dtype=np.float64)
    if direction == 'symmetric':
        mid = (n - 1) / 2.0
        d = np.abs(idx - mid) / mid
        base = d
    elif direction == 'decreasing':
        base = 1.0 - idx / (n - 1)
    elif direction == 'increasing':
        base = idx / (n - 1)
    elif direction == 'sine':
        base = np.sin(np.pi * idx / (n - 1))
    else:
        raise ValueError('unknown direction')

    if kind == 'linear':
        raw = 1.0 + param * base
    elif kind == 'quadratic':
        raw = 1.0 + param * base ** 2
    elif kind == 'exponential':
        raw = np.exp(param * base)
    elif kind == 'power':
        raw = (1.0 + param * base) ** 2
    else:
        raise ValueError('unknown kind')
    raw = np.clip(raw, 0.0, None)
    return project_to_sum(raw, target_sum)

def random_pair_transfer(a, target_sum, scale=0.01):
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    if a[i] <= 0:
        return a
    delta = scale * a[i]
    a[i] -= delta
    a[j] += delta
    # preserve sum exactly, no need to renorm but renorm fix small floating diff
    return project_to_sum(a, target_sum)

def jitter(a, target_sum, sigma=1e-3):
    a = a + np.random.randn(a.shape[0]) * sigma
    a = np.clip(a, 0.0, None)
    return project_to_sum(a, target_sum)

def smooth_max_and_grad(a, tau):
    conv = conv_self_fft(a)
    max_c = conv.max()
    exp_arg = (conv - max_c) / tau
    # clip for large negative to avoid underflow?
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w = w / w_sum
    # cross-correlation to get gradient
    a_rev = a[::-1]
    # conv of w with a_rev
    grad_full = fft_convolve(w, a_rev)
    n = a.shape[0]
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return max_c, grad

def adam_step(a, direction, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    # direction is the pseudo-gradient (not actual gradient of loss)
    m = beta1 * m + (1.0 - beta1) * direction
    v = beta2 * v + (1.0 - beta2) * (direction * direction)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = a + step
    return a_new, m, v

"""
search for a low‑peak self‑convolution step function.

The algorithm consists of three stages:

1. **Parametric sweep** – a collection of simple analytic families
   (linear / quadratic / exponential / power, symmetric or monotone)
   are generated, scaled to the required ℓ₁‑norm and the best one is kept.

2. **Adam‑projected gradient descent on a smooth‑max** – the
   max‑entry of the convolution is approximated by a log‑sum‑exp.
   Its gradient can be written as a cross‑correlation and is obtained
   with two FFT‑based convolutions per iteration.  The update uses the
   Adam optimiser, the step is projected onto the feasible set
   (non‑negative, fixed ℓ₁‑norm).  The temperature τ of the soft‑max is
   gradually reduced (cooling) and the learning rate is decayed.

3. **Stochastic polishing** – when the descent stops, short simulated‑annealing
   random pair‑wise mass transfers are performed to escape possible plateaus.

The whole routine respects a 1000 s wall‑clock budget and always returns
the best sequence found.
"""

def fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors a and b using a power‑of‑2 FFT."""
    n, m = a.shape[0], b.shape[0]
    L = 1 << ((n + m - 1).bit_length())           # next power of two
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)
    return conv[: n + m - 1]


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (length 2·len(a)-1)."""
    return fft_convolve(a, a)


def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector so that its ℓ₁‑norm equals *target*."""
    s = v.sum()
    if s > 0.0:
        return v * (target / s)
    # fallback – uniform distribution (should never happen)
    return np.full_like(v, target / v.shape[0])


def generate_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
    direction: str = "symmetric",
) -> np.ndarray:
    """
    Produce a deterministic shape of length *n*.

    ``kind``        – {"linear","quadratic","exponential","power"}.
    ``direction``   – {"symmetric","decreasing","increasing","sine"}.

    *param* controls the strength of the edge/gradient (≥0) and is
    interpreted according to the chosen ``kind``.
    The shape is finally scaled so that its ℓ₁‑norm equals *target_sum*.
    """

        mid = (n - 1) / 2.0
        d = np.abs(idx - mid) / mid                     # ∈[0,1]
        base = d
    elif direction == "decreasing":
        base = 1.0 - idx / (n - 1)                       # 1 … 0
    elif direction == "increasing":
        base = idx / (n - 1)                             # 0 … 1
    elif direction == "sine":
        base = np.sin(np.pi * idx / (n - 1))             # 0 … 0 (peaks in middle)
    else:
        raise ValueError("unknown direction")

    if kind == "linear":
        raw = 1.0 + param * base
    elif kind == "quadratic":
        raw = 1.0 + param * base ** 2
    elif kind == "exponential":
        raw = np.exp(param * base)
    elif kind == "power":
        raw = (1.0 + param * base) ** 2
    else:
        raise ValueError("unknown kind")

    raw = np.clip(raw, 0.0, None)
    return project_to_sum(raw, target_sum)


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """Transfer a tiny fraction of mass from a random source index to a random target."""
    n = a.shape[0]

        return a

    a[i] -= delta
    a[j] += delta
    # renormalise to eliminate round‑off drift
    return project_to_sum(a, target_sum)

    a = a + np.random.randn(a.shape[0]) * sigma
    a = np.clip(a, 0.0, None)
    return project_to_sum(a, target_sum)


def smooth_max_and_grad(a: np.ndarray, tau: float):
    """
    Smooth approximation of max(conv(a,a)) and its gradient.
    Returns (max_value, gradient_vector) where the gradient is the
    derivative of the soft‑max w.r.t. *a*.
    """
    conv = conv_self_fft(a)                # length 2n‑1
    max_c = conv.max()
    # log‑sum‑exp trick for stability
    exp_arg = (conv - max_c) / tau
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:                       # degenerate – fall back to hard max
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w = w / w_sum

    # gradient = 2 * (w * a_rev)  (cross‑correlation)
    a_rev = a[::-1]
    grad_full = fft_convolve(w, a_rev)    # length 3n‑2
    n = a.shape[0]
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return max_c, grad


def adam_step(a, direction, m, v, t, lr,
              beta1=0.9, beta2=0.999, eps=1e-8):
    """
    One Adam update on *direction* (already the descent direction).
    The step is added to ``a`` – direction already carries the sign.
    Returns (new_a, new_m, new_v).
    """
    m = beta1 * m + (1.0 - beta1) * direction
    v = beta2 * v + (1.0 - beta2) * (direction * direction)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = a + step
    return a_new, m, v
