"""
Optimisation of a non‑negative step function to minimise
    max(conv(a, a))

The objective is scale‑invariant, so we fix the ℓ¹‑norm to
    Σ a_i = √(2·n)
and minimise the peak of the linear convolution a∗a.

Approach
--------
* Multiple random restarts with varying lengths (n from 1.5k to 10k).
* For each restart:
    – Initialise a with a random power‑law or uniform vector,
      project onto the simplex.
    – Perform a long projected (soft‑max) gradient descent.
      The smooth approximation of max is the log‑sum‑exp with a
      temperature β that is annealed from 1 up to 1024.
      Gradient = 2·(a ⋆ p) where p is the soft‑max distribution over
      the convolution entries.
      A momentum term and a simple adaptive step size keep updates
      stable.  After each step we project back onto the simplex.
    – A short stochastic hill‑climbing phase moves tiny fractions
      of mass between random coordinates while keeping the ℓ¹‑norm.
      Only improving moves are kept.
* The best sequence seen over the whole budget is returned.
* Extensive use of FFTs (via numpy.fft) makes each iteration O(n log n).

The code is deliberately written with no closures, no lambdas and
no external I/O – it can be run as‑is within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """Fast objective (max entry of the convolution)."""
    return _max_convolution_fft(a)


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σ x_i = s}.
    Uses the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Gradient of the soft‑max approximation of the maximum convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Compute (g, b) where
        b = a*a (full convolution, length 2·n‑1)
        g = ∇ (log‑sum‑exp_β(b)) = 2·(a ⋆ p)
    with p_k ∝ exp(β·b_k).  All operations are performed with FFTs.
    """
    n = a.shape[0]

    # ---- b = a*a -------------------------------------------------------
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ---- soft‑max probabilities p ---------------------------------------
    max_b = np.max(b)
    # subtract max for numerical stability
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)                # shape (2·n‑1,)

    # ---- gradient g = 2·(a ⋆ p) ---------------------------------------
    # cross‑correlation a ⋆ p can be obtained by convolving a_rev with p
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1        # = 3·n‑2
    size2 = 1 << ((conv_len).bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    # take the central n entries (indices n‑1 … n‑1+n‑1)
    g = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Stochastic hill‑climbing (local refinement)
# ----------------------------------------------------------------------


def _random_hill_climb(a: np.ndarray, deadline: float):
    """
    Very cheap hill‑climber that preserves the ℓ¹‑norm.
    Randomly transfers a small fraction of mass from i to j and keeps
    only improving moves.  Stops when the deadline is reached.
    """
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    cur_val = best_val

    max_frac = 0.12
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        frac = rng.random() * max_frac
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        new_val = _score(a)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill‑climb improved = {best_val:.6f}")
        else:
            # revert
            a[j] -= delta
            a[i] += delta
    return best_a, best_val


# ----------------------------------------------------------------------
# Projected gradient descent on the soft‑max approximation
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected gradient descent with an annealing schedule for β.
    Uses a momentum term and a simple adaptive step size.
    Returns the best vector encountered before the deadline.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 1024.0
    step_coeff = target_sum * 0.15          # base step magnitude
    momentum = np.zeros_like(a)

    rng = np.random.default_rng()
    it = 0
    last_improve_it = 0

    while time.time() < deadline:
        # compute smoothed gradient
        g, _ = _gradient_softmax(a, beta)

        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            lr = step_coeff / (norm_g * (beta + 1.0))
            # momentum (0.9 decay)
            momentum = 0.9 * momentum + lr * g
            a -= momentum

        # project back onto the simplex (keeps non‑negativity & ℓ¹‑norm)
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow plateaus
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # evaluate current objective
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve_it = it
            print(f"[{time.time():.1f}] beta={beta:.2f}  score={best_val:.6f}")

        # adaptive annealing: increase β if no improvement for a while
        if it - last_improve_it > 2500 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)
            last_improve_it = it            # reset counter after β change

        # regular schedule – slowly increase β
        if it % 800 == 0 and beta < max_beta:
            beta = min(beta * 1.2, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the supplied
    `evaluate_sequence` objective.  The routine runs for at most `budget_s`
    seconds, trying several vector lengths and many random restarts.
    The best sequence found is returned.
    """
    # initialise RNG
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # safety margin so that the harness does not kill us just before deadline
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_score = np.inf

    # lengths we will explore – larger n gives more flexibility but costs time
    lengths = [1500, 2500, 3500, 4500, 6000, 8000, 10000]

    # we will keep looping until time runs out
    run_id = 0
    while time.time() < deadline:
        n = int(rng.choice(lengths))
        target_sum = np.sqrt(2.0 * n)

        # ------------------------------------------------------------------
        # initialise a candidate vector
        # ------------------------------------------------------------------
        if rng.random() < 0.6:
            # power‑law start (the classic starter)
            p_exp = 0.5 + rng.uniform(-0.20, 0.20)   # broader range than before
            a0 = (np.arange(1, n + 1, dtype=np.float64) ** (-p_exp))
        else:
            # completely random start
            a0 = rng.random(n)

        a0 = a0 * target_sum / a0.sum()
        # small random perturbation before the optimisation
        a0 += rng.normal(scale=target_sum * 0.001, size=n)
        a0 = np.maximum(a0, 0.0)
        a0 = _project_to_simplex(a0, target_sum)

        # ------------------------------------------------------------------
        # allocate a time slice for this restart
        # ------------------------------------------------------------------
        now = time.time()
        time_left = deadline - now
        if time_left < 5.0:
            break                         # not enough time for a useful run

        # give larger vectors a bit more time, but cap at ~30 s
        max_run = 30.0 if n >= 6000 else 20.0
        run_deadline = now + min(max_run, time_left * 0.85)

        # ------------------------------------------------------------------
        # gradient descent phase
        # ------------------------------------------------------------------
        a_opt, opt_score = _annealed_gradient_descent(a0, run_deadline)

        # ------------------------------------------------------------------
        # hill‑climbing refinement (uses whatever time is still left)
        # ------------------------------------------------------------------
        if time.time() < deadline:
            a_opt, opt_score = _random_hill_climb(a_opt, deadline)

        # ------------------------------------------------------------------
        # keep global best
        # ------------------------------------------------------------------
        if opt_score < best_score:
            best_score = opt_score
            best_seq = a_opt.copy()
            print(f"** NEW GLOBAL BEST ** n={n}  run={run_id}  score={best_score:.6f}")

        run_id += 1

    # ----------------------------------------------------------------------
    # fallback in the (unlikely) case no sequence was produced
    # ----------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=np.float64)

    # strip trailing zeros for neatness
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    return best_seq.tolist()
