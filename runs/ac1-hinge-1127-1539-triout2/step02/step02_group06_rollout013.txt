def propose_candidate(seed: int = 42, budget_s: int = 1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # Determine n
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)):
        n = len(base)
        base_seq = np.array(base, dtype=np.float64)
    else:
        # choose n in [6000, 12000] maybe random
        n = np.random.randint(6000, 12000)
        base_seq = None

    target_sum = np.sqrt(2.0 * n)

    # Helper to evaluate objective quickly
    def obj(a):
        # a normalized sum target, compute max convolution
        conv = conv_self_fft(a)
        return conv.max()

    # Initial candidates collection
    candidates = []

    # use base seq if exist
    if base_seq is not None:
        # ensure sum normalized
        cand = project_to_sum(base_seq, target_sum)
        candidates.append(cand)

    # uniform baseline
    uniform = np.full(n, target_sum / n)
    candidates.append(uniform)

    # parametric shapes
    directions = ["symmetric", "decreasing", "increasing"]
    kinds = ["linear", "quadratic", "exponential"]
    # Parameter ranges
    param_dict = {
        "linear": np.linspace(0.0, 15.0, 16),
        "quadratic": np.linspace(0.0, 10.0, 11),
        "exponential": np.linspace(0.0, 5.0, 11)
    }
    for direction in directions:
        for kind in kinds:
            for p in param_dict[kind]:
                cand = generate_shape(n, target_sum, kind=kind, param=p, direction=direction)
                candidates.append(cand)

    # random Dirichlet (multiple)
    for _ in range(20):
        rand_vec = np.random.rand(n)
        cand = project_to_sum(rand_vec, target_sum)
        candidates.append(cand)

    # Evaluate candidates and keep best
    best_seq = None
    best_score = float("inf")
    for cand in candidates:
        score = obj(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
    print(f"[{time.time():.2f}] Initial best score = {best_score:.6f}")

    # If time nearly exhausted, return
    if time.time() >= deadline:
        return best_seq.tolist()

    # Adam optimization loop
    a = best_seq.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    lr = 0.1 * (target_sum / n)   # initial learning rate
    tau = 0.2 * obj(a)           # start temperature as fraction of current max
    if tau <= 0:
        tau = 1.0
    tau_min = 1e-5
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    iteration = 0
    no_improve_iter = 0
    # store timestamp for logging improvement events
    while time.time() < deadline:
        iteration += 1

        # compute convolution
        conv = conv_self_fft(a)
        max_conv = conv.max()

        # update best
        if max_conv + 1e-12 < best_score:
            best_score = max_conv
            best_seq = a.copy()
            print(f"Iter {iteration:06d} – new best = {best_score:.6f}")
            no_improve_iter = 0
        else:
            no_improve_iter += 1

        # decide gradient type
        if tau > tau_min:
            # smooth max gradient
            max_c = conv.max()
            exp_arg = (conv - max_c) / tau
            w = np.exp(exp_arg)
            w_sum = w.sum()
            if w_sum == 0:
                # fallback
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w = w / w_sum
            grad_full = fft_convolve(w, a[::-1])
            grad = 2.0 * grad_full[n-1:n-1+n]
        else:
            # hard max gradient
            k = conv.argmax()
            grad = np.zeros_like(a)
            start = max(0, k-(n-1))
            end = min(n-1, k)
            idx = np.arange(start, end+1)
            grad[idx] = 2.0 * a[k - idx]

        # direction - projected to zero-sum hyperplane
        direction = -(grad - np.mean(grad))

        # Adam moments update (bias corrected)
        m = beta1 * m + (1 - beta1) * direction
        v = beta2 * v + (1 - beta2) * (direction * direction)
        bias_correction1 = 1 - beta1**iteration
        bias_correction2 = 1 - beta2**iteration
        m_hat = m / bias_correction1
        v_hat = v / bias_correction2

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a + step

        # enforce nonnegativity and L1 constraint
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total > 0:
            a = a * (target_sum / total)
        else:
            a = np.full_like(a, target_sum / n)

        # schedule adaptation
        if iteration % 500 == 0:
            lr *= 0.9
            tau *= 0.95
            if lr < 1e-7:
                lr = 1e-7
            if tau < tau_min:
                tau = tau_min

        # occasional escape steps
        if np.random.rand() < 0.02:
            a = random_pair_transfer(a, target_sum, scale=0.01)

        if np.random.rand() < 0.01:
            a = jitter(a, target_sum, sigma=5e-4)

        # if stuck for long, reset momentum and jitter heavy
        if no_improve_iter >= 2000:
            a = jitter(a, target_sum, sigma=0.001)
            m.fill(0.0)
            v.fill(0.0)
            no_improve_iter = 0
            print(f"Iter {iteration:06d} – stuck, jitter reset")

        # exit condition to free time for polishing
        if time.time() + 5.0 >= deadline:
            break

    # Final polishing: short random walk
    print(f"[{time.time():.2f}] Optimization finished, best score {best_score:.6f}")
    while time.time() < deadline - 0.2:
        a = random_pair_transfer(a, target_sum, scale=0.004)
        conv = conv_self_fft(a)
        max_conv = conv.max()
        if max_conv < best_score - 1e-9:
            best_score = max_conv
            best_seq = a.copy()
            print(f"Polish – improvement to {best_score:.6f}")

    return best_seq.tolist()

"""
Optimisation of a non‑negative step‑function that minimises the
self‑convolution peak (the value returned by `evaluate_sequence`).

The algorithm proceeds in three stages:

1. **Broad initialisation** – a variety of analytic families
   (U‑shapes, monotone, exponential, quadratic, random Dirichlet draws)
   are generated and normalised to the required ℓ₁‑norm.
   The best among them seeds the optimiser.

2. **Adaptive Adam descent with a smooth‑max surrogate** – the objective
   `max(a*a)` is approximated by a temperature‑controlled log‑sum‑exp.
   Its gradient is computed by a single FFT‑based convolution.
   An Adam optimiser (first‑ and second‑moment adaptivity) updates the
   vector in the hyper‑plane Σaᵢ = target_sum (projection by scaling).
   The temperature is slowly reduced; when it becomes very small the
   algorithm switches to the exact hard‑max sub‑gradient.

3. **Fine‑grained random perturbations** – after the main descent a short
   stochastic polishing phase (pair‑wise mass transfers and Gaussian jitter)
   explores neighbouring points and keeps any improvement.

All heavy work (convolutions, gradients) uses FFTs, so the routine easily
handles sequences of several‑tens of thousands of entries within the
1000 s budget.  The best sequence found is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors a and b using an FFT of power‑of‑2 length."""
    n, m = a.shape[0], b.shape[0]
    L = 1 << ((n + m - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)
    return conv[: n + m - 1]


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (length 2·len(a)‑1)."""
    return fft_convolve(a, a)


def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector so that its ℓ₁‑norm equals *target*."""
    s = v.sum()
    if s > 0.0:
        return v * (target / s)
    # fallback – uniform distribution (should never happen)
    return np.full_like(v, target / v.shape[0])


def generate_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
    direction: str = "symmetric",
) -> np.ndarray:
    """
    Produce a deterministic shape of length *n*.

    ``kind``     – {"linear", "quadratic", "exponential"}.
    ``direction`` – {"symmetric", "decreasing", "increasing"}.

    *param* controls the strength of the edge/gradient (≥0).
    The shape is finally scaled so that the ℓ₁‑norm equals *target_sum*.
    """
    idx = np.arange(n, dtype=np.float64)

    if direction == "symmetric":
        mid = (n - 1) / 2.0
        d = np.abs(idx - mid) / mid          # ∈[0,1]
        base = d
    elif direction == "decreasing":
        base = 1.0 - idx / (n - 1)            # high → low
    elif direction == "increasing":
        base = idx / (n - 1)                  # low → high
    else:
        raise ValueError("unknown direction")

    if kind == "linear":
        raw = 1.0 + param * base
    elif kind == "quadratic":
        raw = 1.0 + param * base ** 2
    elif kind == "exponential":
        raw = np.exp(param * base)
    else:
        raise ValueError("unknown shape kind")

    raw = np.clip(raw, 0.0, None)
    return project_to_sum(raw, target_sum)


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """Transfer a small fraction of mass from a random source index to a random target."""
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0.0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_to_sum(a, target_sum)


def jitter(a: np.ndarray, target_sum: float, sigma: float = 1e-3) -> np.ndarray:
    """Add Gaussian noise (clipped) and renormalise."""
    a = a + np.random.randn(a.shape[0]) * sigma
    a = np.clip(a, 0.0, None)
    return project_to_sum(a, target_sum)


# ----------------------------------------------------------------------
# Optimiser
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a sequence of non‑negative heights that minimises `evaluate_sequence`."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose a length – reuse a known construction if available
    # ------------------------------------------------------------------
    base_seq = globals().get("height_sequence_1")
    if isinstance(base_seq, (list, tuple)) and len(base_seq) > 0:
        n = len(base_seq)
    else:
        # a length that is large enough to give a good bound but still cheap
        n = np.random.randint(7200, 8800)

    target_sum = np.sqrt(2.0 * n)   # the normalisation used in the paper

    # ------------------------------------------------------------------
    # Helper to evaluate the objective quickly (max of the linear convolution)
    # ------------------------------------------------------------------
    def obj(a_vec: np.ndarray) -> float:
        """Return max(a*a) – exactly the value returned by evaluate_sequence after normalisation."""
        conv = conv_self_fft(a_vec)
        return float(conv.max())

    # ------------------------------------------------------------------
    # Phase 1 – generate a diverse set of initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # start from the supplied construction if present
    if isinstance(base_seq, (list, tuple)):
        cand = np.array(base_seq, dtype=np.float64)
        cand = project_to_sum(cand, target_sum)
        candidates.append(cand)

    # uniform baseline
    candidates.append(np.full(n, target_sum / n))

    # parametric families (larger ranges than the previous code)
    directions = ["symmetric", "decreasing", "increasing"]
    kinds = ["linear", "quadratic", "exponential"]
    param_grids = {
        "linear": np.linspace(0.0, 15.0, 16),          # 0 … 15 step 1
        "quadratic": np.linspace(0.0, 10.0, 11),      # 0 … 10 step 1
        "exponential": np.linspace(0.0, 8.0, 17),    # 0 … 8 step 0.5
    }
    for direction in directions:
        for kind in kinds:
            for p in param_grids[kind]:
                cand = generate_shape(n, target_sum, kind=kind, param=p, direction=direction)
                candidates.append(cand)

    # random Dirichlet‑type draws with several concentration parameters
    for alpha in [0.1, 0.3, 0.7, 1.0, 2.0, 5.0]:
        raw = np.random.gamma(alpha, 1.0, size=n)
        candidates.append(project_to_sum(raw, target_sum))

    # a few pure random vectors (uniform Dirichlet)
    for _ in range(12):
        raw = np.random.rand(n)
        candidates.append(project_to_sum(raw, target_sum))

    # pick the best among the initial pool
    best_seq = None
    best_score = float("inf")
    for cand in candidates:
        sc = obj(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
    print(f"[{time.time():.2f}] Phase‑1 best score = {best_score:.6f}")

    if time.time() >= deadline:
        return best_seq.tolist()

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation with smooth‑max (temperature annealing)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    iteration = 0

    # hyper‑parameters
    avg_entry = target_sum / n
    lr = 0.3 * avg_entry          # initial learning rate (a few‑tenths of the average entry)
    tau = 0.4 * best_score        # start temperature as a fraction of the current max‑convolution
    if tau <= 0.0:
        tau = 1.0
    tau_min = 1e-5
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    no_improve = 0

    while time.time() < deadline:
        iteration += 1

        # ----- convolution and current objective -----
        conv = conv_self_fft(a)
        max_conv = float(conv.max())

        # ----- record improvement -----
        if max_conv + 1e-12 < best_score:
            best_score = max_conv
            best_seq = a.copy()
            print(f"Iter {iteration:06d} – new best = {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # ----- gradient (smooth‑max or hard‑max) -----
        if tau > tau_min:
            # soft‑max approximation of the max
            max_c = conv.max()
            exp_arg = (conv - max_c) / tau
            w = np.exp(exp_arg)
            w_sum = w.sum()
            if w_sum == 0.0:                     # fall back to hard max
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w /= w_sum
            grad_full = fft_convolve(w, a[::-1])
            grad = 2.0 * grad_full[n - 1 : n - 1 + n]   # slice i = 0…n‑1
        else:
            # exact sub‑gradient of the hard max
            k = int(conv.argmax())
            grad = np.zeros_like(a)
            start = max(0, k - (n - 1))
            end = min(n - 1, k)
            idx = np.arange(start, end + 1, dtype=int)
            grad[idx] = 2.0 * a[k - idx]

        # ----- project to the zero‑sum hyperplane (preserve ℓ₁‑norm) -----
        direction = -(grad - np.mean(grad))

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * direction
        v = beta2 * v + (1.0 - beta2) * (direction * direction)

        bias_correction1 = 1.0 - beta1 ** iteration
        bias_correction2 = 1.0 - beta2 ** iteration
        m_hat = m / bias_correction1
        v_hat = v / bias_correction2

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a + step

        # ----- enforce feasibility (non‑negative, correct ℓ₁‑norm) -----
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total > 0.0:
            a = a * (target_sum / total)
        else:
            a = np.full_like(a, avg_entry)

        # ----- schedule adaptations -----
        if iteration % 500 == 0:
            lr *= 0.9
            tau = max(tau_min, tau * 0.94)

        # ----- occasional stochastic escape steps -----
        if np.random.rand() < 0.02:
            a = random_pair_transfer(a, target_sum, scale=0.008)

        if np.random.rand() < 0.01:
            a = jitter(a, target_sum, sigma=5e-4)

        # ----- stagnation handling -----
        if no_improve >= 2500:
            a = jitter(a, target_sum, sigma=0.001)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"Iter {iteration:06d} – stagnation reset")

        # ----- keep a little time for the final polishing phase -----
        if time.time() + 4.0 >= deadline:
            break

    # ------------------------------------------------------------------
    # Phase 3 – short stochastic polishing (pair transfers + jitter)
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Descent finished; best score {best_score:.6f}")

    while time.time() < deadline - 0.2:
        a = random_pair_transfer(a, target_sum, scale=0.004)
        cur = obj(a)
        if cur < best_score - 1e-9:
            best_score = cur
            best_seq = a.copy()
            print(f"Polish – improvement to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence discovered
    # ------------------------------------------------------------------
    return best_seq.tolist()
