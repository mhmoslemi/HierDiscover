def conv_fft(a,b):
    # returns full convolution length len(a)+len(b)-1 using FFT
    n = len(a)+len(b)-1
    L = 1 << ((n-1).bit_length())
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa*fb, n=L)[:n]
    conv = np.real(conv)
    conv[conv<0] = 0.0
    return conv

    conv = np.fft.irfft(fx * fy, n=L)[:n]

def _full_conv_fft(p):
    n = p.shape[0]
    L = 1 << ((2*n-2).bit_length())
    ft = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(ft * ft, n=L)[:2*n-1]
    conv[conv < 0] = 0.0
    return conv

def _conv_fft_two(a,b):
    n = len(a)+len(b)-1
    L = 1 << ((n-1).bit_length())
    Fa = np.fft.rfft(a, n=L)
    Fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Fa*Fb, n=L)[:n]
    conv[conv < 0] = 0.0
    return conv

"""
Non‑negative step‑function optimizer (gradient‑descent on a smoothed max).

The evaluation metric for a non‑negative vector a of length n is

    f(a) = 2 * n * max(conv(a,a)) / (sum(a)**2)

Because the objective is scale‑invariant we may restrict ourselves to
probability vectors (sum = 1).  After this normalisation the score becomes

    score(p) = 2 * n * max(conv(p,p))

where `conv` denotes the full linear convolution.

Instead of minimising the non‑smooth maximum directly we minimise a
smooth upper bound obtained from the log‑sum‑exp trick

    M_β(p) = (1/β) * log Σ_k exp(β * conv_k(p,p))   ≥   max(conv(p,p))

Its gradient w.r.t. the probability vector p is
    ∇_p M_β = 2 * (w ⋆ p)   where   w_k = exp(β conv_k) / Σ_j exp(β conv_j)
and “⋆” denotes the (cross‑)correlation between the weight vector w
(length 2 n‑1) and p (length n).  The correlation can be obtained
efficiently with a single FFT‑based convolution.

The algorithm:
    * generate a diverse pool of seed probability vectors
    * for each seed run a mirror‑descent / exponentiated‑gradient loop:
        - compute conv(p,p) and the smooth max M_β
        - form the softmax weights w
        - compute the gradient g = 2 * correlation(w,p)
        - multiplicative update  p ← p * exp(‑η g)   (η decays like 1/√t)
        - project back onto the simplex
        - slowly increase β (temperature) to sharpen the approximation
    * keep the best sequence (lowest true score) found before the time
      budget expires.

All heavy work uses FFTs (O(n log n)) and the routine respects the 1000 s
budget.  The final list of heights is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _logsumexp(x: np.ndarray) -> float:
    """Numerically stable log‑sum‑exp."""
    m = np.max(x)
    return float(m + np.log(np.sum(np.exp(x - m))))


def _full_conv_fft(p: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a vector with itself using FFT.
    Returns a real array of length 2*len(p)-1.
    """
    n = p.shape[0]
    # size for efficient FFT (next power of two)
    L = 1 << ((2 * n - 2).bit_length())
    ft = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(ft * ft, n=L)[: 2 * n - 1]
    conv = np.real(conv)
    conv[conv < 0] = 0.0
    return conv


def _conv_fft_two(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two real vectors a and b using FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    n = a.shape[0] + b.shape[0] - 1
    L = 1 << ((n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)[:n]
    conv = np.real(conv)
    conv[conv < 0] = 0.0
    return conv


def _correlation_w_p(w: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Correlation of a long vector w (length 2n‑1) with a short vector p (length n).
    The result is a vector of length n defined by
        c[k] = Σ_i w[i] * p[i‑k]   (i‑k in range 0…n‑1)
    Implemented via convolution of w with the reversed p and slicing.
    """
    n = p.shape[0]
    # reverse p for convolution → correlation
    p_rev = p[::-1]
    conv = _conv_fft_two(w, p_rev)                # length = (2n‑1)+n‑1 = 3n‑2
    # the needed slice starts at offset n‑1 and has length n
    start = n - 1
    return conv[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Duchi et al. (2008).
    """
    if np.all(v <= 0):
        # all non‑positive → return uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _generate_powerlaw(p: float, sigma: float,
                       max_len: int = 5000, eps: float = 1e-9) -> np.ndarray:
    """
    Truncated power‑law sequence a_i = (i + sigma)⁻ᵖ.
    Stops when values drop below `eps` or when `max_len` is reached.
    """
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p)
    mask = a > eps
    return a[mask]

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The routine respects the 1000 s time limit and
    returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0          # safety margin

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # 1) Build a diverse pool of starting vectors (all normalised to sum=1)
    # ------------------------------------------------------------------
    seeds = []

    # previous best, if supplied by the environment
    if "height_sequence_1" in globals():
        s = np.asarray(height_sequence_1, dtype=np.float64)
        if s.size > 0:
            s = np.maximum(s, 0.0)
            s = s / s.sum()
            seeds.append(s)

    # power‑law families
    for p_exp in np.linspace(0.35, 0.65, 7):
        for sigma in np.logspace(-3.0, 0.5, 12):
            seq = _generate_powerlaw(p_exp, sigma, max_len=5000, eps=1e-10)
            if seq.size >= 250:
                seq = seq / seq.sum()
                seeds.append(seq)

    # exponential families
    for lam in np.linspace(0.0002, 0.006, 12):
        i = np.arange(5000, dtype=np.float64)
        seq = np.exp(-lam * i)
        seq = seq[seq > 1e-10]
        if seq.size >= 250:
            seq = seq / seq.sum()
            seeds.append(seq)

    # uniform vectors of various lengths
    for n in (800, 1500, 2500, 4000, 5000):
        u = np.ones(n, dtype=np.float64)
        u = u / u.sum()
        seeds.append(u)

    # random Dirichlet‑like seeds
    for n in (800, 1500, 2500, 4000, 5000):
        r = np.random.rand(n)
        r = r / r.sum()
        seeds.append(r)

    # ------------------------------------------------------------------
    # 2) Optimisation loop for each seed
    # ------------------------------------------------------------------
    for base in seeds:
        if time.time() > deadline:
            break

        p = base.astype(np.float64).copy()
        # sanity checks
        p[p < 0] = 0.0
        if p.sum() == 0.0:
            continue
        p = p / p.sum()
        n = p.shape[0]

        # limit maximum length to keep per‑iteration cost reasonable
        max_n = 5000
        if n > max_n:
            p = p[:max_n]
            n = p.shape[0]
            p = p / p.sum()

        # parameters for the inner optimisation
        eta0 = 0.6                # initial step‑size (exponentiated gradient)
        beta = 1.0                # initial smoothing temperature
        beta_max = 200.0
        beta_factor = 1.015       # geometric increase each iteration
        max_iters = 5000

        for it in range(1, max_iters + 1):
            if time.time() > deadline:
                break

            # ----- evaluate true objective -----
            conv_self = _full_conv_fft(p)            # length 2n‑1
            cur_max = np.max(conv_self)
            cur_score = 2.0 * n * cur_max

            # ----- keep best observed sequence -----
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = p.tolist()
                print(f"Iter {it:5d} | n={n} | score={best_score:.6f} | beta={beta:.2f}")

                if best_score < 1.503:
                    print("Target reached – early termination.")
                    return best_seq

            # ----- smooth max (log‑sum‑exp) and its gradient -----
            scaled = beta * conv_self
            lse = _logsumexp(scaled)
            w = np.exp(scaled - lse)               # softmax weights, sum = 1

            # gradient = 2 * correlation(w, p)
            g = 2.0 * _correlation_w_p(w, p)

            # ----- exponentiated‑gradient update -----
            eta = eta0 / np.sqrt(it)
            # avoid overflow/underflow in the exponential
            delta = -eta * g
            # clip delta to a reasonable range
            delta = np.clip(delta, -30.0, 30.0)
            p = p * np.exp(delta)

            # ----- project back onto simplex -----
            p_sum = p.sum()
            if p_sum == 0.0 or not np.isfinite(p_sum):
                # re‑initialise if collapse occurs
                p = np.abs(np.random.randn(n))
                p_sum = p.sum()
            p = _project_simplex(p)

            # ----- occasional random jitter to escape plateaus -----
            if np.random.rand() < 0.015:
                idx = np.random.randint(n)
                jitter = (np.random.rand() - 0.5) * 0.02 * p[idx]
                p[idx] = max(p[idx] + jitter, 0.0)
                p = _project_simplex(p)

            # ----- temperature schedule -----

    # 3) Return the best sequence discovered
    # ------------------------------------------------------------------

          "length:", len(best_seq) if best_seq else None)
    return best_seq if best_seq is not None else []  # empty list signals failure
