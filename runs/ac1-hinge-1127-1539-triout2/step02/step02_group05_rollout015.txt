"""
Search for a non‑negative step‑function that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm consists of three stages:
1. **Random family sampling** – generate a few parametric candidates
   (power‑law, exponential, combined power‑law + exponential, logistic),
   normalise them to the required L1‑norm and keep the best one.
2. **Long Adam optimisation** – smooth the max with a soft‑max of temperature
   γ and perform projected Adam steps.  γ is increased slowly so that the
   surrogate converges to the true maximum.  The sequence is kept on the
   simplex `{x ≥ 0, Σx = √(2 n)}` after every update.
3. **Perturb‑restart + exact sub‑gradient phase** – repeatedly perturb the
   currently best sequence, run a short Adam run to explore a new basin,
   and finally switch to a true sub‑gradient descent where the soft‑max is
   replaced by an indicator of the maximal convolution entries.  The step
   size is annealed.

The routine respects a time budget (default 1000 s) and prints progress.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _compute_score(a: np.ndarray) -> float:
    """Fast version of evaluate_sequence (assumes a ≥ 0)."""
    n = a.shape[0]
    nfft = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))


def project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of *v* onto the simplex
        { x : x_i ≥ 0 , Σ_i x_i = target_sum }.
    Standard O(n log n) algorithm (Duchi et al., 2008).
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _make_power_law(n, p, offset):
    """Return a power‑law vector of length n: a_i = 1/(i+offset)^p."""
    i = np.arange(n, dtype=np.float64)
    return 1.0 / np.power(i + offset, p)


def _make_exponential(n, lam):
    """Return an exponential decay vector: a_i = exp(-lam*i)."""
    i = np.arange(n, dtype=np.float64)
    return np.exp(-lam * i)


def _make_power_exp(n, p, offset, lam):
    """Power‑law multiplied by exponential decay."""
    i = np.arange(n, dtype=np.float64)
    return (1.0 / np.power(i + offset, p)) * np.exp(-lam * i)


def _make_logistic(n, mid, scale):
    """Logistic (sigmoid) decreasing vector."""
    i = np.arange(n, dtype=np.float64)
    return 1.0 / (1.0 + np.exp((i - mid) / scale))

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _compute_score(a: np.ndarray) -> float:
    """Fast version of ``evaluate_sequence`` (assumes a ≥ 0)."""
    n = a.shape[0]
    nfft = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))


def project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of *v* onto the simplex
        { x : x_i ≥ 0 , Σ_i x_i = target_sum }.
    Implements the O(n log n) algorithm of Duchi et al. (ICML 2008).
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _make_power_law(n, p, offset):

    """a_i = exp(-lam * i)."""
    i = np.arange(n, dtype=np.float64)
    return np.exp(-lam * i)

    """Power‑law multiplied by exponential decay."""
    i = np.arange(n, dtype=np.float64)

    i = np.arange(n, dtype=np.float64)
