def _solve_lp_max_sum_under_conv_bound(f: np.ndarray, M: float):
    n = len(f)
    # Build A matrix (2n-1) × n
    rows = np.repeat(np.arange(n), n) + np.tile(np.arange(n), n)  # shape (n*n) vector of row indices
    cols = np.tile(np.arange(n), n)                             # each column repeated n times
    data = np.tile(f, n)    # f repeated for each column
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # Upper bounds b = M * np.ones(2*n-1)
    # c = -np.ones(n)
    res = scipy.optimize.linprog(c=-np.ones(n),
                                 A_ub=A,
                                 b_ub=np.full(2*n-1, M),
                                 bounds=(0, None),
                                 method='highs',
                                 options={'presolve': True, 'time_limit': max(0, deadline-time.time()), 'disp': False})
    if not res.success:
        # maybe fallback to L1 solution with zero vector?
        return np.zeros_like(f), 0.0
    g = res.x
    sum_g = g.sum()
    return g, sum_g

"""
propose_candidate

Search for a non-negative step function that minimizes the bound
2·n·max(conv(a,a))/(sum a)^2.

The algorithm proceeds in two stages.

1.  Linear‑programming improvement.
    For a current vector a we compute M = max(conv(a,a)).
    The LP
        maximise   1^T g
        subject to  conv(a,g) ≤ M , g ≥ 0
    is solved (SciPy HiGHS).  If the LP can increase the ℓ¹‑norm,
    the solution g is scaled back to the required norm and mixed with a
    using a small convex combination.  This provably never increases the
    maximal convolution value, and in practice strictly decreases it.
    The step is repeated a few times.

2.  A cheap random hill‑climbing refinement that transfers a random
    fraction of mass between two positions while preserving the ℓ¹‑norm.
    It keeps any improvement in the objective.

The routine tries several vector lengths (1500‑2500), starts from the
previously discovered best construction if available, and keeps the
best sequence found before the time budget expires.
"""

import time
import numpy as np
import scipy.sparse
import scipy.optimize

# ---------- Cached pattern for convolution constraint matrix ----------
_conv_pattern_cache = {}

def _get_conv_pattern(n):
    """Return (row_indices, col_indices) for the Toeplitz matrix that
    maps g → conv(a,g) (full convolution, length 2n‑1)."""
    if n not in _conv_pattern_cache:
        # row where coefficient a[i] appears for variable g[j] is i+j
        rows = np.arange(n, dtype=np.int64).repeat(n) + np.tile(np.arange(n, dtype=np.int64), n)
        cols = np.tile(np.arange(n, dtype=np.int64), n)
        _conv_pattern_cache[n] = (rows, cols)
    return _conv_pattern_cache[n]

# ---------- Scoring utilities ----------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())

def _score(a: np.ndarray) -> float:
    """Fast objective: 2·n·max(conv)/sum(a)^2."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)

# ---------- LP sub‑routine ----------
def _solve_lp_max_sum_under_conv_bound(f: np.ndarray, M: float):
    """Solve   maximise ∑ g_i   s.t. conv(f,g) ≤ M, g ≥ 0.
    Returns (g, sum(g))."""
    n = f.shape[0]
    rows, cols = _get_conv_pattern(n)
    # repeat f for each column
    data = np.tile(f, n)
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)                     # maximise sum -> minimise -sum
    b = np.full(2 * n - 1, M, dtype=np.float64)
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if not res.success:
        # Fall back to the zero vector – no improvement possible.
        return np.zeros_like(f), 0.0
    g = res.x
    return g, float(g.sum())

# ---------- LP‑driven improvement ----------
def _lp_improve(a0: np.ndarray, target_sum: float, deadline: float):
    """Iterative LP mixing that never increases the maximal convolution."""
    a = a0.copy()
    # enforce exact ℓ¹‑norm (numerical drift)
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    iteration = 0
    max_iters = 12                     # safety cap
    while time.time() < deadline and iteration < max_iters:
        # current maximal convolution
        conv_a = np.convolve(a, a)
        M = float(conv_a.max())

        # solve AP: maximise ℓ¹‑norm under the same max‑convolution bound
        g_opt, sum_g_opt = _solve_lp_max_sum_under_conv_bound(a, M)

        # if the LP cannot increase the ℓ¹‑norm we are done
        if sum_g_opt <= target_sum * (1.0 - 1e-12):
            break

        # scale the optimizer to the required norm
        g_scaled = g_opt * (target_sum / sum_g_opt)

        # try to mix a and g_scaled; start with a moderate step and shrink if needed
        t = 0.5
        mixed = False
        while t > 1e-4:
            a_candidate = (1.0 - t) * a + t * g_scaled
            # tiny numeric noise can introduce -0, clip.
            a_candidate = np.maximum(a_candidate, 0.0)
            conv_candidate = np.convolve(a_candidate, a_candidate)
            M_new = float(conv_candidate.max())
            if M_new <= M * (1 + 1e-12):
                a = a_candidate
                mixed = True
                # update best if we saw a reduction
                if M_new < best_val - 1e-12:
                    best_val = M_new
                    best_a = a.copy()
                break
            t *= 0.5

        if not mixed:
            # mixing failed – nothing to gain from the current LP solution
            break

        iteration += 1
    return best_a, best_val

# ---------- Simple random hill‑climber ----------
def _random_hill_climb(a: np.ndarray, deadline: float):
    """Transfer a random fraction of mass between two entries while preserving the ℓ¹‑norm.
    Accept only moves that improve the objective."""
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    cur_val = best_val
    max_frac = 0.12
    rng = np.random.default_rng()
    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        frac = rng.random() * max_frac
        delta = a[i] * frac
        if delta <= 0.0:
            continue
        a[i] -= delta
        a[j] += delta
        new_val = _score(a)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:

"""
Search for a non‑negative step function that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

by a two‑phase approach:

1. Linear‑programming improvement.
   For the current vector a we compute M = max(conv(a,a)).
   We solve the LP
       maximise   Σ g_i
       s.t.       conv(a,g) ≤ M ,   g ≥ 0 .
   If the LP can increase the ℓ¹‑norm, the solution g is scaled back to the
   required norm (√(2·n)) and mixed with a using a small convex combination.
   This never increases the maximal convolution value and usually makes it
   strictly smaller.  The step is repeated a handful of times.

2. A cheap random hill‑climber that transfers a random fraction of mass
   between two positions while preserving the ℓ¹‑norm and keeps any improving
   move.

The routine tries a few vector lengths (1500, 2000, 2500), starts from the
previously discovered construction if the global variable
`height_sequence_1` is present, and keeps the best sequence found before the
time budget expires.
"""

import time
import numpy as np
import scipy.sparse
import scipy.optimize

# ----------------------------------------------------------------------
# Cached pattern for the Toeplitz matrix that maps g → conv(a,g)
# ----------------------------------------------------------------------
_conv_pattern_cache = {}

def _get_conv_pattern(n):
    """Return (row_idx, col_idx) for the sparse matrix A such that
    (A @ g) == conv(a,g) (full convolution of length 2·n‑1)."""
    if n not in _conv_pattern_cache:
        rows = np.arange(n, dtype=np.int64).repeat(n) + np.tile(np.arange(n, dtype=np.int64), n)
        cols = np.tile(np.arange(n, dtype=np.int64), n)
        _conv_pattern_cache[n] = (rows, cols)
    return _conv_pattern_cache[n]

# ----------------------------------------------------------------------
# Fast objective (score) – identical to `evaluate_sequence` when the sum is fixed
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())

def _score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)²)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)

# ----------------------------------------------------------------------
# Linear programme: maximise ℓ¹‑norm under a fixed max‑convolution bound
# ----------------------------------------------------------------------
def _solve_lp_max_sum_under_conv_bound(f: np.ndarray, M: float):
    """
    Solve   maximise Σ g_i   s.t.   conv(f,g) ≤ M , g ≥ 0.
    Returns (g, Σ g_i).  Uses SciPy HiGHS.
    """
    n = f.shape[0]
    rows, cols = _get_conv_pattern(n)
    data = np.tile(f, n)                     # each column contains f
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n)                          # minimise -Σ g_i  == maximise Σ g_i
    b = np.full(2 * n - 1, M, dtype=np.float64)

    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if not res.success:
        # No feasible improvement – return zero vector.
        return np.zeros_like(f), 0.0
    g = res.x
    return g, float(g.sum())

# ----------------------------------------------------------------------
# LP‑driven monotone improvement
# ----------------------------------------------------------------------
def _lp_improve(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Repeatedly:
        * compute M = max(conv(a,a))
        * solve the LP that maximises the ℓ¹‑norm under the same bound M
        * if the optimum has a larger ℓ¹‑norm, scale it back to `target_sum`
          and mix it with the current a.
    The mixing step never increases the maximal convolution value.
    """
    a = a0.copy()
    # enforce the exact ℓ¹‑norm (numerical drift)
    a = a * (target_sum / a.sum())

    best_a = a.copy()
    best_val = _score(a)

    it = 0
    max_iters = 12                     # safety cap
    while time.time() < deadline and it < max_iters:
        conv_a = np.convolve(a, a)
        M = float(conv_a.max())

        g_opt, sum_g_opt = _solve_lp_max_sum_under_conv_bound(a, M)

        # No increase in total mass → cannot improve further.
        if sum_g_opt <= target_sum * (1.0 - 1e-12):
            break

        # Scale the LP solution back to the required ℓ¹‑norm.
        g_scaled = g_opt * (target_sum / sum_g_opt)

        # Try to mix a and g_scaled.  Reduce the mixing weight if the
        # maximal convolution rises.
        t = 0.5
        mixed = False
        while t > 1e-4:
            a_candidate = (1.0 - t) * a + t * g_scaled
            a_candidate = np.maximum(a_candidate, 0.0)          # clip tiny negatives
            conv_cand = np.convolve(a_candidate, a_candidate)
            M_new = float(conv_cand.max())
            if M_new <= M * (1 + 1e-12):                       # non‑increase
                a = a_candidate
                mixed = True
                if M_new < best_val - 1e-12:
                    best_val = M_new
                    best_a = a.copy()
                break
            t *= 0.5

        if not mixed:
            # mixing failed – further LP steps are unlikely to help
            break

        it += 1

    return best_a, best_val

# Very cheap stochastic hill‑climber
# ----------------------------------------------------------------------
def _random_hill_climb(a: np.ndarray, deadline: float):
    """
    Randomly move a fraction of the mass from one position to another.
    Keep any move that improves the objective.
    """
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    cur_val = best_val
    max_frac = 0.12
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        frac = rng.random() * max_frac
        delta = a[i] * frac
        if delta <= 0.0:

        a[i] -= delta
        a[j] += delta

        new_val = _score(a)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill‑climb improved = {best_val:.6f}")

            # revert the change
            a[j] -= delta
            a[i] += delta

    return best_a, best_val
