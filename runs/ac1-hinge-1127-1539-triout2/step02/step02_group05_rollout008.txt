    np.random.seed(seed)
    start=time.time()
    deadline=start + budget_s - 5

    # Determine sequence length and init
    try:
        init = np.array(height_sequence_1, dtype=np.float64)
        n = init.shape[0]
    except Exception:
        n = 6000
        init = None

    target_sum = np.sqrt(2.0*n)

    # Helper: normalize to target sum
    def normalize(x):
        x = np.maximum(x,0.0)
        s = x.sum()
        if s>0:
            return x * target_sum / s
        else:
            return np.full_like(x, target_sum/x.size)

    # best track
    best_a = None
    best_score = np.inf

    # If provided init, evaluate
    if init is not None:
        a = normalize(init)
        score = _compute_score(a)
        best_a, best_score = a.copy(), score

    # Also evaluate a few random pow law vectors
    for _ in range(10):

        a = 1.0/np.power(np.arange(n,dtype=np.float64)+offset,p)

        score = _compute_score(a)
        if score < best_score:

"""
Hybrid optimisation for the step‑function problem.

The objective to minimise is
    2 * n * max(conv(a, a)) / (sum(a) ** 2)

with a ≥ 0 and sum(a) fixed (the simplex constraint).
The algorithm consists of three phases:

1. **Soft‑max Adam** – Optimise a smooth approximation of the max
   (soft‑max with a temperature γ).  The gradient of the surrogate is
   obtained by a convolution of the soft‑max weights with the reversed
   current vector.  An Adam update is performed and the iterate is
   projected onto the simplex {x ≥ 0, sum x = √(2 n)}.

2. **Sub‑gradient Adam** – When γ has grown large the soft‑max focuses
   on the true maximal entry.  From this point on we switch to the exact
   sub‑gradient of the max‑convolution (a simple slice of the reversed
   vector).  Adam is kept for automatic step‑size adaptation.

3. **Fine‑tuning / stochastic jitter** – In the last part we add very
   small random perturbations and occasional random power‑law restarts
   to escape shallow plateaus.

All operations are performed with NumPy and FFTs, and the whole search
stops automatically after the allotted budget (≈ 1000 s).  The best
sequence discovered is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _compute_score(a: np.ndarray, nfft: int) -> float:
    """
    Fast version of ``evaluate_sequence`` for a non‑negative vector.
    The sum of ``a`` is assumed to be > 0.
    """
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * a.shape[0] - 1]
    max_c = conv.max()
    s = a.sum()
    return float(2.0 * a.shape[0] * max_c / (s * s))


def compute_conv(a: np.ndarray, nfft: int) -> np.ndarray:
    """Return the full self‑convolution a*a (length 2n‑1) using FFT."""
    af = np.fft.rfft(a, n=nfft)
    return np.fft.irfft(af * af, n=nfft)[: 2 * a.shape[0] - 1]


def project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of *v* onto the simplex
        { x : x_i ≥ 0 , Σ x_i = target_sum }.
    """
    # Duchi et al. (ICML 2008) – O(n log n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _normalise(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Make the vector non‑negative and enforce Σ v = target_sum."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0:
        v = v * (target_sum / s)
    else:
        v = np.full_like(v, target_sum / v.size)
    return v


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a non‑negative sequence of heights that (approximately) minimises
    the evaluation function described in the statement.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # --------------------------------------------------------------
    # Determine the problem size and initialise the first iterate
    # --------------------------------------------------------------
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        n = init_seq.shape[0]
    except Exception:
        n = 6000
        init_seq = None

    target_sum = np.sqrt(2.0 * n)

    # ---- best‑so‑far container ------------------------------------
    best_a = None
    best_score = np.inf

    # ---- use the supplied construction if available ----------------
    if init_seq is not None:
        a0 = _normalise(init_seq, target_sum)
        score0 = _compute_score(a0, _next_pow2(2 * n - 1))
        best_a, best_score = a0.copy(), score0

    # ---- a few random power‑law starting points -------------------
    for _ in range(20):
        if time.time() >= deadline:
            break
        p = np.random.uniform(0.35, 0.78)
        offset = np.random.uniform(0.5, 3.5)
        a = 1.0 / np.power(np.arange(n, dtype=np.float64) + offset, p)
        a = _normalise(a, target_sum)
        sc = _compute_score(a, _next_pow2(2 * n - 1))
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # fallback to uniform if nothing else worked
    if best_a is None:
        best_a = np.full(n, target_sum / n)
        best_score = _compute_score(best_a, _next_pow2(2 * n - 1))

    # current iterate
    a = best_a.copy()

    # --------------------------------------------------------------
    # Pre‑compute FFT sizes that stay constant throughout the run
    # --------------------------------------------------------------
    nfft_conv = _next_pow2(2 * n - 1)     # for a*a
    nfft_corr = _next_pow2(3 * n - 2)     # for w ⊛ a_rev

    # --------------------------------------------------------------
    # Adam state
    # --------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # learning‑rate schedule
    lr = 0.20
    lr_min = 1e-4
    lr_decay = 0.9995

    # soft‑max temperature schedule
    gamma = 1.0
    gamma_max = 2500.0
    gamma_inc = 1.005

    # --------------------------------------------------------------
    # Initial convolution (used for the first iteration)
    # --------------------------------------------------------------
    conv = compute_conv(a, nfft_conv)      # length 2n‑1
    iteration = 0
    last_improve_iter = 0

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1
        t += 1

        elapsed = time.time() - start_time
        # allocate roughly 60 % of the time to the soft‑max phase,
        # the following 30 % to pure sub‑gradient Adam,
        # the final 10 % to stochastic jitter / random restarts.
        time_frac = elapsed / (budget_s - 5.0)

        # ------------------------------------------------------------------
        # 1) Soft‑max (smooth max) phase
        # ------------------------------------------------------------------
        if time_frac < 0.60:
            max_conv = conv.max()
            # soft‑max weights over the convolution entries
            w = np.exp(gamma * (conv - max_conv))
            w_sum = w.sum()
            if w_sum == 0.0:                     # guard against underflow
                w[:] = 1.0 / w.size
            else:
                w /= w_sum

            # gradient = 2 * (w ⊛ a_rev)
            w_fft = np.fft.rfft(w, n=nfft_corr)
            a_rev_fft = np.fft.rfft(a[::-1], n=nfft_corr)
            corr_full = np.fft.irfft(w_fft * a_rev_fft, n=nfft_corr)
            grad = 2.0 * corr_full[n - 1 : n - 1 + n]

            # temperature increase
            gamma = min(gamma * gamma_inc, gamma_max)

        # ------------------------------------------------------------------
        # 2) Sub‑gradient of the true max (no soft‑max)
        # ------------------------------------------------------------------
        elif time_frac < 0.90:
            # index of the maximal convolution entry
            k = int(conv.argmax())
            grad = np.zeros_like(a)
            if k < n:
                # Slice 0 … k  (inclusive)
                grad[: k + 1] = 2.0 * a[k::-1]
            else:
                offset = k - (n - 1)
                grad[offset:] = 2.0 * a[offset:][::-1]

        # ------------------------------------------------------------------
        # 3) Stochastic jitter / random restarts (pure exploration)
        # ------------------------------------------------------------------
        else:
            # tiny random perturbation of the current best
            jitter = 1e-4 * np.random.randn(n)
            a = project_onto_simplex(a + jitter, target_sum)
            conv = compute_conv(a, nfft_conv)          # recompute after jitter
            continue  # skip the Adam update for this iteration

        # ------------------------------------------------------------------
        # Adam update (common to all three phases)
        # ------------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        # projection back onto the simplex (non‑negative, fixed L1)
        a_new = project_onto_simplex(a_new, target_sum)

        # occasional jitter to avoid stagnation
        if iteration % 97 == 0:
            jitter = 1e-4 * np.random.randn(n)
            a_new = project_onto_simplex(a_new + jitter, target_sum)

        # --------------------------------------------------------------
        # Evaluate the new point
        # --------------------------------------------------------------
        conv_new = compute_conv(a_new, nfft_conv)
        max_new = conv_new.max()
        s_new = a_new.sum()
        score_new = float(2.0 * n * max_new / (s_new * s_new))

        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            last_improve_iter = iteration
            print(
                f"Iter {iteration:6d}: new best {best_score:.6f} "
                f"(elapsed {elapsed:.1f}s)"
            )

        # --------------------------------------------------------------
        # Prepare for next iteration
        # --------------------------------------------------------------
        a = a_new
        conv = conv_new

        # learning‑rate decay
        lr = max(lr * lr_decay, lr_min)

        # optional random power‑law restart if we have been stuck too long
        if iteration % 5000 == 0 and (iteration - last_improve_iter) > 5000:
            p = np.random.uniform(0.35, 0.78)
            offset = np.random.uniform(0.5, 3.5)
            cand = 1.0 / np.power(np.arange(n, dtype=np.float64) + offset, p)
            cand = _normalise(cand, target_sum)
            conv_cand = compute_conv(cand, nfft_conv)
            sc = float(2.0 * n * conv_cand.max() / (cand.sum() ** 2))
            if sc < best_score:
                best_score, best_a = sc, cand.copy()
                a, conv = cand, conv_cand
                print(
                    f"Restart {iteration:6d}: improved to {best_score:.6f} "
                    f"(elapsed {elapsed:.1f}s)"
                )
                last_improve_iter = iteration

        # safety break
        if time.time() >= deadline:
            break

    # ----------------------------------------------------------------------
    # Fine‑tuning phase: a few extra Adam steps with a very high γ
    # ----------------------------------------------------------------------
    fine_iters = 2000
    gamma_fine = gamma_max
    for _ in range(fine_iters):
        if time.time() >= deadline:
            break

        max_conv = conv.max()
        w = np.exp(gamma_fine * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        w_fft = np.fft.rfft(w, n=nfft_corr)
        a_rev_fft = np.fft.rfft(a[::-1], n=nfft_corr)
        corr_full = np.fft.irfft(w_fft * a_rev_fft, n=nfft_corr)
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]

        # tiny gradient step (no Adam here)
        a = a - 5e-4 * grad
        a = project_onto_simplex(a, target_sum)

        conv = compute_conv(a, nfft_conv)
        score = float(2.0 * n * conv.max() / (a.sum() ** 2))
        if score < best_score:
            best_score, best_a = score, a.copy()
            elapsed = time.time() - start_time
            print(
                f"Fine‑tune: new best {best_score:.6f} (elapsed {elapsed:.1f}s)"
            )

    # ----------------------------------------------------------------------
    # Return the best sequence discovered
    # ----------------------------------------------------------------------
    return best_a.tolist()
