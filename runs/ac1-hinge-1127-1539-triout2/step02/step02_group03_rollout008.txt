def _optimize_softmax(p, deadline, rng):
    n=len(p)
    S= np.sqrt(2.0 * n) # scaling factor for a later
    best_score = np.inf
    best_p = p.copy()
    # hyperparameter
    lr = 1.0  # baseline learning rate
    lr_decay = 0.9995
    beta = 0.5
    beta_factor = 1.001 # slowly increase
    # compute fft_len
    conv_len = 2*n - 1
    corr_len = 3*n - 2
    fft_len = 1 << (corr_len - 1).bit_length()  # for correlation
    # Precompute maybe other
    while time.time() < deadline:
        # compute conv with FFT
        Fa = np.fft.rfft(p, n=fft_len)
        conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
        cur_max = float(conv.max())
        if cur_max < best_score:
            best_score = cur_max
            best_p = p.copy()
        # compute softmax weights
        w = np.exp(beta * (conv - cur_max))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            # fallback to uniform weight on max index
            w.fill(0)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum
        # compute correlation gradient
        Fw = np.fft.rfft(w, n=fft_len)
        corr = np.fft.irfft(Fw * np.conj(Fa), n=fft_len)[:corr_len]
        g = 2.0 * corr[n-1:n-1+n]  # length n
        # multiplicative update
        p = p * np.exp(-lr * g)
        # projection onto simplex
        p = _project_simplex(p)
        # update hyperparameters
        lr *= lr_decay
        beta = min(beta * beta_factor, 50.0)  # avoid overflow

"""
Propose a candidate step‑function that minimises the maximal entry of its
self‑convolution.

Algorithm
---------
1. Work with the normalised probability vector `p = a / Σa` (Σp = 1).
   The objective `max(a*a)` equals `2·n·max(p*p)` and therefore only the
   shape of `p` matters.

2. Initialise `p` either from the previously discovered sequence
   (`height_sequence_1`) or from a Dirichlet draw.

3. Optimise `p` with a *soft‑max mirror‑descent* scheme:
   * The convolution `c = p * p` (full linear convolution) is computed by
     FFT (O(n log n)).
   * Instead of using only the current maximal entry, a smooth approximation
     of the max is taken:
       w_i = exp( β·(c_i – max(c)) ) / Σ_j exp( β·(c_j – max(c)) )
     (soft‑max weights).  For β → ∞ this converges to the indicator of the
     maximal entry, for small β the whole profile participates.
   * The gradient of the soft‑max loss wrt `p` can be written as a
     correlation:
       g = 2·corr(w, p)          (corr is the linear correlation,
                                  i.e. the cross‑correlation of `w` and `p`).
     Computed efficiently by another FFT.
   * A multiplicative (exponentiated‑gradient) update preserves non‑negativity:
       p ← p ⊙ exp( –η·g )
     followed by a projection onto the probability simplex (Σp = 1).

   The step size `η` decays geometrically, while the soft‑max temperature
   `β` is slowly increased (continuation) so that the optimisation first
   smooths the whole profile and finally concentrates on the true maximum.

4. The optimisation runs repeatedly for many random restarts (different
   lengths `n` and seeds) within the time budget.  The best vector found
   (lowest maximal convolution) is returned, scaled back to the original
   normalisation `Σa = √(2·n)`.

The whole routine respects the 1000 s limit, prints progress and returns
the final list of heights.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Projects a vector onto the probability simplex {x | x_i ≥ 0, Σx_i = 1}.
    Implements the algorithm of Duchi et al. (2008).  Returns a new array.
    """
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper for FFT length (next power of two)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power‑of‑2 ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
# Initial probability vector (sum = 1)
# ----------------------------------------------------------------------


def _initial_probability(base: np.ndarray,
                         n: int,
                         rng: np.random.Generator) -> np.ndarray:
    """
    Produce an initial probability vector of length n.
    If *base* is provided and long enough, its leading part (with a tiny
    jitter) is used; otherwise a Dirichlet draw (uniform Dirichlet) is used.
    """
    if base is not None and base.size >= n:
        p = base[:n].astype(float).copy()
        p += rng.normal(scale=1e-4 * p.mean(), size=n)  # tiny jitter
        p = np.clip(p, 0.0, None)
        if p.sum() == 0.0:
            p = rng.random(n)
    else:
        p = rng.random(n)  # exponential(1) is also fine; uniform works well

    p = _project_simplex(p)
    return p


# ----------------------------------------------------------------------
# Core optimiser (soft‑max mirror descent)
# ----------------------------------------------------------------------


def _optimize_softmax(p0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Optimise a probability vector `p` so that the L∞‑norm of its
    self‑convolution is reduced.  Returns the best vector found and the
    corresponding maximal convolution value (in the original scale,
    i.e. max(p*p)).
    """
    n = p0.shape[0]
    conv_len = 2 * n - 1
    corr_len = 3 * n - 2
    fft_len = _next_pow2(corr_len)

    # hyper‑parameters
    eta = 0.8               # initial step‑size
    eta_decay = 0.9995      # exponential decay
    beta = 0.4              # start with a gentle soft‑max
    beta_factor = 1.0015    # slowly increase β → sharper max

    best_p = p0.copy()
    # initial evaluation
    Fa = np.fft.rfft(p0, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    best_score = float(conv.max())

    p = p0.copy()
    while time.time() < deadline:
        # ----- convolution -----
        Fa = np.fft.rfft(p, n=fft_len)
        conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
        cur_max = float(conv.max())

        # record best ever seen in this run
        if cur_max < best_score:
            best_score = cur_max
            best_p = p.copy()

        # ----- soft‑max weights -----
        # subtract cur_max for numerical stability
        w = np.exp(beta * (conv - cur_max))
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            # fallback: focus on the true maximum only
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # ----- correlation (gradient) -----
        Fw = np.fft.rfft(w, n=fft_len)
        corr = np.fft.irfft(Fw * np.conj(Fa), n=fft_len)[:corr_len]
        # extract the n relevant entries (shift by n‑1)
        g = 2.0 * corr[n - 1:n - 1 + n]

        # ----- multiplicative (exponentiated) update -----
        p = p * np.exp(-eta * g)

        # ----- projection onto simplex -----
        p = _project_simplex(p)

        # ----- occasional jitter to escape shallow minima -----
        if rng.random() < 0.005:
            noise = rng.normal(scale=0.01 * p.mean(), size=n)
            p = np.clip(p + noise, 0.0, None)
            p = _project_simplex(p)

        # ----- schedule updates -----
        eta *= eta_decay
        beta = min(beta * beta_factor, 50.0)   # avoid overflow

    return best_p, best_score
