"""
Propose a step‑function (non‑negative coefficient vector) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

where `conv` is the linear self‑convolution.

Algorithm
---------
1. Choose a problem size n (≈8192, a power of two for fast FFT).  If a
   previously discovered sequence (`height_sequence_1`) is available,
   reuse its length, otherwise pick n ≈ 8192.

2. **Parametric sweep** – generate a variety of symmetric “U‑shapes’’
   (linear, quadratic, exponential, power, inverse‑power).  Each shape
   is projected onto the ℓ₁‑simplex with sum = √(2 n).  Keep the best
   shape as the start point.

3. **Smoothed gradient descent with momentum** – minimise the
   log‑sum‑exp smooth approximation of the max‑convolution.
   The gradient is obtained by a single convolution in the frequency
   domain.  After each step the vector is projected onto the simplex.
   The smoothing parameter τ is gradually reduced and the learning
   rate is decayed; occasional random transfers and jitter prevent
   stagnation.

4. **Fine‑grained sub‑gradient phase** – once τ is very small we switch
   to a true sub‑gradient of the max‑convolution (average over all
   maximising indices).  A decreasing step size is used.

5. **Polishing** – a short burst of tiny random pair‑transfers; any
   improvement is kept.

All heavy work uses FFT‑based convolutions (O(n log n)) and the routine
always returns the best sequence seen before the 1000 s budget expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a with itself using a power‑of‑two FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex
        {x | x_i ≥ 0, sum_i x_i = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries will be zero; return uniform distribution
        return np.full_like(v, target / len(v))

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a tiny amount of mass from one random index to another,
    then project back onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def generate_shape(n: int, target_sum: float, kind: str, param: float) -> np.ndarray:
    """
    Produce a symmetric “U‑shaped’’ sequence of length n.
    kind ∈ {linear, quadratic, exponential, power, inverse}.
    param ≥ 0 controls the shape (for power/inverse it is the exponent).
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "power":
        # raw = d^p + ε   (ε prevents exact zeros)
        raw = d ** param + 1e-4
    elif kind == "inverse":
        # raw = 1 / (ε + d^p)
        raw = 1.0 / (1e-4 + d ** param)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def compute_max_conv(a: np.ndarray) -> float:
    """Return max(self‑convolution) of a (a is assumed non‑negative)."""
    return float(conv_self_fft(a).max())


def smooth_gradient_step(a: np.ndarray, target_sum: float,
                         eta: float, tau_factor: float,
                         momentum: np.ndarray,
                         beta: float):
    """
    One iteration of smoothed gradient descent with momentum.
    Returns (new_a, new_momentum, max_conv).
    """
    n = a.shape[0]

    # self‑convolution and its maximum
    conv = conv_self_fft(a)
    max_conv = conv.max()

    # smoothing weights (log‑sum‑exp)
    tau = max(tau_factor * max_conv, 1e-12)
    # stability: subtract max before exponentiation
    w = np.exp((conv - max_conv) / tau)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient = (2/τ) * convolution(w, a_rev) restricted to central n entries
    a_rev = a[::-1]
    L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
    grad = (2.0 / tau) * conv_full[n - 1 : n - 1 + n]

    # momentum update
    momentum = beta * momentum - eta * grad
    a_new = a + momentum
    a_new = np.maximum(a_new, 0.0)
    a_new = project_onto_simplex(a_new, target_sum)
    return a_new, momentum, max_conv


def subgradient_step(a: np.ndarray, target_sum: float,
                     step: float) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient step on the true max‑convolution.
    Returns (new_a, max_conv_before_step).
    """
    n = a.shape[0]
    conv = conv_self_fft(a)
    max_conv = conv.max()

    # mask of all (approximately) maximal entries
    eps = max(1e-12, 1e-9 * max_conv)
    mask = (conv >= max_conv - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt == 0:
        mask[conv.argmax()] = 1.0
        cnt = 1.0

    # gradient = average sub‑gradient over all maximal indices
    a_rev = a[::-1]
    L = 1 << ((len(mask) + len(a_rev) - 1).bit_length())
    M = np.fft.rfft(mask, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(M * A_rev, n=L)[: len(mask) + len(a_rev) - 1]
    grad = (2.0 / cnt) * conv_full[n - 1 : n - 1 + n]

    a_new = a - step * grad
    a_new = np.maximum(a_new, 0.0)
    a_new = project_onto_simplex(a_new, target_sum)
    return a_new, max_conv

# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    `evaluate_sequence`.  The algorithm respects a time budget (~1000 s)
    and returns the best sequence observed.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        init_seq = np.array(base, dtype=np.float64)
    else:
        # Choose a power‑of‑two around 8192 for fast FFTs
        n_candidates = [4096, 8192, 16384]
        n = n_candidates[np.random.choice(len(n_candidates))]
        init_seq = None

    target_sum = math.sqrt(2.0 * n)          # normalisation: sum = √(2 n)

    # --------------------------------------------------------------
    # Phase 0 – parametric sweep over simple U‑shapes
    # --------------------------------------------------------------
    best_seq = np.full(n, target_sum / n)    # uniform start
    best_score = compute_max_conv(best_seq)

    shape_kinds = ("linear", "quadratic", "exponential", "power", "inverse")
    param_grids = {
        "linear": np.linspace(0.0, 8.0, 41),          # step 0.2
        "quadratic": np.linspace(0.0, 6.0, 31),      # step 0.2
        "exponential": np.linspace(0.0, 6.0, 31),    # step 0.2
        "power": np.linspace(0.5, 5.0, 10),          # exponent
        "inverse": np.linspace(0.5, 5.0, 10),
    }

    print(f"[{time.time():.2f}] Starting parametric sweep (n={n})")
    for kind in shape_kinds:
        for p in param_grids[kind]:
            cand = generate_shape(n, target_sum, kind, p)
            score = compute_max_conv(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"  New best {kind} (p={p:.3f}) → {best_score:.6f}")
            if time.time() >= deadline:
                print("Time exhausted during sweep.")
                return best_seq.tolist()

    # consider the supplied baseline if it exists
    if init_seq is not None:
        score = compute_max_conv(init_seq)
        if score < best_score:
            best_score = score
            best_seq = init_seq.copy()
            print(f"  Baseline improves initial score → {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – smoothed gradient descent with momentum
    # --------------------------------------------------------------
    a = best_seq.copy()
    momentum = np.zeros_like(a)

    # hyper‑parameters
    eta = 0.4 * (target_sum / n)            # rather aggressive initial step
    beta = 0.9                               # momentum coefficient
    tau_factor = 0.08                        # τ = τ_factor * max_conv
    iteration = 0
    no_improve = 0

    print(f"[{time.time():.2f}] Entering smoothed gradient descent")
    while time.time() < deadline:
        a, momentum, cur_max = smooth_gradient_step(
            a, target_sum, eta, tau_factor, momentum, beta
        )
        iteration += 1

        if cur_max + 1e-12 < best_score:
            best_score = cur_max
            best_seq = a.copy()
            no_improve = 0
            print(f"Iter {iteration:07d} – improved to {best_score:.6f}")
        else:
            no_improve += 1

        # Adaptive schedules
        if iteration % 1000 == 0:
            eta *= 0.97                      # shrink step size slowly
            tau_factor = max(0.001, tau_factor * 0.96)

        # occasional stochastic mass transfer
        if np.random.rand() < 0.012:
            a = random_pair_transfer(a, target_sum, scale=0.008)

        # jitter after long stagnation
        if no_improve >= 4000:
            jitter = np.random.randn(n) * 0.0004
            a = project_onto_simplex(np.maximum(a + jitter, 0.0), target_sum)
            no_improve = 0
            print(f"Iter {iteration:07d} – jitter applied")

    # --------------------------------------------------------------
    # Phase 2 – sub‑gradient refinement
    # --------------------------------------------------------------
    print(f"[{time.time():.2f}] Smoothed descent finished; best score {best_score:.6f}")
    # set a decreasing step size for the sub‑gradient phase
    step = 0.02 * (target_sum / n)          # modest initial step
    sub_iter = 0
    while time.time() < deadline:
        a, cur_max = subgradient_step(a, target_sum, step)
        sub_iter += 1
        if cur_max + 1e-12 < best_score:
            best_score = cur_max
            best_seq = a.copy()
            print(f"Sub‑grad iter {sub_iter:06d} – improved to {best_score:.6f}")

        # decay step size gently
        if sub_iter % 500 == 0:
            step *= 0.97
        # tiny random transfer now and then
        if np.random.rand() < 0.01:
            a = random_pair_transfer(a, target_sum, scale=0.004)

    # --------------------------------------------------------------
    # Phase 3 – final polishing with random pair transfers
    # --------------------------------------------------------------
    print(f"[{time.time():.2f}] Sub‑gradient phase finished; best score {best_score:.6f}")

    for _ in range(40):
        if time.time() >= deadline:
            break
        candidate = random_pair_transfer(best_seq.copy(), target_sum, scale=0.003)
        cand_score = compute_max_conv(candidate)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = candidate.copy()
            print(f"Polish transfer improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found as a plain Python list
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Returning best sequence with score {best_score:.6f}")
    return best_seq.tolist()
