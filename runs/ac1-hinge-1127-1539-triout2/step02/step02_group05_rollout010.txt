"""
Propose a non‑negative step‑function (vector) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works in four stages:

1. **Diverse initialisation** – many simple families (uniform,
   power‑law, exponential, reversed exponential, random Dirichlet) are
   generated, normalised to the required ℓ1‑norm √(2 n) and the best
   one (according to a fast FFT based evaluator) is kept.

2. **Adam on a smooth‑max surrogate** – the true max is approximated by
   a soft‑max with a temperature γ that is slowly increased.
   An Adam optimiser updates the sequence, after each step projecting
   back onto the simplex `{x≥0, Σx = √(2 n)}`.

3. **L‑BFGS refinement** – the current best vector is re‑parameterised
   as `a_i = C·exp(v_i) / Σexp(v)` (automatically satisfies the
   constraints).  A bound‑constrained L‑BFGS optimisation (SciPy)
   minimises the exact max (sub‑gradient is taken from the
   convolution entries attaining the maximum).

4. **Exact‑max fine‑tuning** – a few short gradient‑descent steps with a
   line‑search‑like step‑size adaptation are performed on the true
   objective.

All heavy work (convolution, gradients) uses FFTs, the whole search
fits comfortably into the 1000 s budget.  The function returns the best
sequence found as a plain Python list of floats.
"""

import time
import math
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ *x*."""
    return 1 << (x - 1).bit_length()


def _compute_score(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` when ``a`` is already
    non‑negative and correctly normalised.  Returns max(conv(a,a)).
    """
    n = a.shape[0]
    nfft = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
    max_c = conv.max()
    # the normalisation sum(a)=√(2n) makes the full objective equal to max_c
    return float(max_c)


def project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of *v* onto the simplex
        { x : x_i ≥ 0 , Σ_i x_i = target_sum }.
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative sequence with a low evaluation score.
    ``budget_s`` is the time budget in seconds (a safety margin of a few
    seconds is automatically subtracted).  The function returns the best
    sequence found as a Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # safety margin

    # --------------------------------------------------------------
    # Determine length of the problem
    # --------------------------------------------------------------
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        n = init_seq.shape[0]
    except Exception:
        n = 6000                            # fallback length
        init_seq = None

    target_sum = math.sqrt(2.0 * n)
    nfft_conv = _next_pow2(2 * n - 1)      # size for a*a convolution
    nfft_corr = _next_pow2(3 * n - 2)      # size for correlation during Adam

    # --------------------------------------------------------------
    # 1. Diverse initialisation phase
    # --------------------------------------------------------------
    best_a = None
    best_score = float("inf")

    def _try_candidate(a_raw: np.ndarray) -> None:
        """Normalise *a_raw* to the simplex and keep it if it improves."""
        nonlocal best_a, best_score
        a = np.maximum(a_raw, 0.0)
        s = a.sum()
        if s == 0.0:
            return

        score = _compute_score(a)
        if score < best_score:

    _try_candidate(np.full(n, target_sum / n))

    v = np.zeros_like(a)
    t = 0

    lr_min = 0.001
    lr_decay = 0.9997

    gamma = 1.0
    gamma_max = 500.0
    gamma_inc = 1.004

    while time.time() < deadline:
        iteration += 1
        t += 1

        # ----- convolution a*a -------------------------------------------------

        corr_full = np.fft.irfft(wf * af_rev, n=nfft_corr)
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]

        # ----- Adam update ------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad

            best_a = a_new.copy()
            elapsed = time.time() - start_time
            print(f"Adam iter {iteration:6d}: new best {best_score:.6f} (elapsed {elapsed:.1f}s)")

        a = a_new

        # ----- schedules --------------------------------------------------------
        gamma = min(gamma * gamma_inc, gamma_max)
        lr = max(lr * lr_decay, lr_min)

        if iteration % 500 == 0 and time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # 3. L‑BFGS refinement on the exact max (sub‑gradient)
    # --------------------------------------------------------------
    def _exp_normalise(v_vec: np.ndarray) -> np.ndarray:
        """Map log‑variables to the simplex (positivity + Σ = target_sum)."""
        ev = np.exp(v_vec)
        s = ev.sum()
        return ev * (target_sum / s)

    def _lbfgs_refine(start_a: np.ndarray) -> np.ndarray:
        """Run a short L‑BFGS‑B optimisation starting from *start_a*."""
        v0 = np.log(start_a + 1e-12)
        bounds = [(-30.0, 30.0)] * len(v0)          # prevent overflow in exp

        def fun(v_vec: np.ndarray) -> float:
            a_vec = _exp_normalise(v_vec)
            af_loc = np.fft.rfft(a_vec, n=nfft_conv)
            conv_loc = np.fft.irfft(af_loc * af_loc, n=nfft_conv)[: 2 * n - 1]
            return float(conv_loc.max())

        def jac(v_vec: np.ndarray) -> np.ndarray:
            a_vec = _exp_normalise(v_vec)
            af_loc = np.fft.rfft(a_vec, n=nfft_conv)
            conv_loc = np.fft.irfft(af_loc * af_loc, n=nfft_conv)[: 2 * n - 1]

                grad_a += 2 * np.roll(a_rev, shift)
            grad_a /= len(max_idxs)

            # chain rule: d f / d v = (∂a/∂v)^T ∂f/∂a
            sum_a_grad = np.dot(a_vec, grad_a)
            grad_v = a_vec * (grad_a - sum_a_grad / target_sum)

        # SciPy's L‑BFGS‑B
        res = scipy.optimize.minimize(fun,
                                      v0,

    # 5. Return the best discovered sequence
    # --------------------------------------------------------------
    return best_a.tolist()
