def adam_optimize(a0, deadline):
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8
    lr0 = target_sum * 0.1 # base learning rate
    beta = 1.0
    beta_max = 1024.0
    iteration = 1
    best_a = a.copy()
    best_val = _score(a)
    while time.time() < deadline:
        g,_ = _gradient_softmax(a, beta)
        # Adam update
        m = beta1*m + (1-beta1)*g
        v = beta2*v + (1-beta2)*(g*g)
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        lr = lr0 / (np.sqrt(v_hat) + eps)
        a = a - lr * m_hat
        # Projection to simplex
        a = _project_to_simplex(a, target_sum)
        # evaluate
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] beta={beta:.2f}, iter={iteration}, best={best_val:.6f}")
        # update beta schedule
        if iteration % 500 == 0 and beta < beta_max:
            beta = min(beta * 1.5, beta_max)
        iteration += 1
    return best_a, best_val
```

"""
Propose a step‑function sequence that minimizes
    2·n·max(conv(a,a)) / (sum(a)^2)

The objective is scale‑invariant, so we keep the ℓ¹‑norm fixed to
√(2·n) and minimise the maximum entry of the self‑convolution.
The algorithm mixes several heuristics:

*  Several lengths are tried (up to 8000).  Larger n gives more
   freedom but is slower, so the time budget is split adaptively.
*  For each length a few random restarts are performed.  Initial
   vectors are drawn either from a power‑law or from a Dirichlet
   distribution, then projected onto the simplex.
*  A projected Adam optimiser minimises a smooth approximation of
   the max‑convolution (log‑sum‑exp) while annealing the temperature
   β.  Adam’s per‑coordinate step‑size helps to cope with the
   ill‑conditioned landscape.
*  After Adam a cheap stochastic hill‑climber refines the solution
   by moving small mass fractions between indices.
*  Finally a short simulated‑annealing phase explores the neighbourhood
   with a Metropolis acceptance rule.
*  All candidates are evaluated with a fast FFT‑based score; the best
   sequence seen overall is returned.

The code respects the 1000 s budget, performs only CPU work and uses
only NumPy (no file or network I/O).  A previously computed solution
`height_sequence_1` (if present) is taken as an initial candidate.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """Fast version of the competition objective (no safety checks)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:  # should never happen
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    { x | x_i ≥ 0, Σ x_i = s }.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of the soft‑max approximation of the maximum convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Return (g, b) where
        b = a*a (full convolution, length 2·n‑1)
        g = ∇ (log‑sum‑exp_{β}(b))  = 2·(a ⋆ p)    with p_k ∝ exp(β·b_k)
    Both are computed with FFTs for speed.
    """
    n = a.shape[0]

    # ---- b = a*a -------------------------------------------------------
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ---- soft‑max probabilities p ---------------------------------------
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)  # shape (2·n‑1,)

    # ---- gradient g = 2·(a ⋆ p) ---------------------------------------
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1  # = 3·n‑2
    size2 = 1 << ((conv_len).bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed max‑convolution
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray, deadline: float):
    """
    Projected Adam optimiser of the log‑sum‑exp approximation.
    The temperature β is increased gradually while Adam adapts its
    per‑coordinate step size.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = target_sum * 0.05
    beta = 0.5
    beta_max = 2048.0
    it = 1

    best_a = a.copy()
    best_val = _score(a)

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = base_lr / (np.sqrt(v_hat) + eps)
        a = a - lr * m_hat

        # Project back onto the simplex (keeps ℓ¹‑norm and non‑negativity)
        a = _project_to_simplex(a, target_sum)

        # Evaluate
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] Adam it={it} β={beta:.2f} score={best_val:.6f}")

        # Anneal β
        if it % 300 == 0 and beta < beta_max:
            beta = min(beta * 1.6, beta_max)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Stochastic hill‑climbing (cheap local refinement)
# ----------------------------------------------------------------------


def _random_hill_climb(a: np.ndarray, deadline: float):
    """
    Very cheap hill‑climber that preserves the ℓ¹‑norm.
    Randomly transfers a fraction of mass from i to j and keeps improving moves.
    """
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    cur_val = best_val

    max_frac = 0.12
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        frac = rng.random() * max_frac
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        new_val = _score(a)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill‑climb improved = {best_val:.6f}")
        else:
            a[j] -= delta
            a[i] += delta
    return best_a, best_val

def _simulated_annealing(a0: np.ndarray, deadline: float):
    """
    Metropolis simulated‑annealing that moves a fraction of mass between two
    positions.  Temperature decays linearly with elapsed time.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    start = time.time()
    total = max(deadline - start, 1e-6)

    # initial temperature proportional to the current objective
    T0 = 0.1 * cur_score
    T_min = 1e-4

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        T = max(T_min, T0 * (1.0 - frac))

        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if i == j or a[i] <= 1e-15:
            continue

        # move up to 20 % of the mass at i
        delta = a[i] * np.random.rand() * 0.20

        a[i] -= delta
        a[j] += delta

        new_score = _score(a)
        dE = new_score - cur_score

        if dE < 0 or np.random.rand() < np.exp(-dE / T):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA improved = {best_score:.6f}")
        else:
            # reject move
            a[j] -= delta
            a[i] += delta

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:

    seconds, trying several vector lengths and random restarts.  Each restart

    by a cheap hill‑climber and a short simulated‑annealing phase.
    The best sequence found is returned.
    """

    np.random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.8  # safety margin

    best_seq = None
    best_score = np.inf

        try:
            cand = list(globals()["height_sequence_1"])
            val = evaluate_sequence(cand)
            if val < best_score:
                best_score = val
                best_seq = np.array(cand, dtype=np.float64)
                print(f"[{time.time():.1f}] Starting from height_sequence_1, score={best_score:.6f}")
        except Exception:
            pass

    # Lengths to explore (larger ⇒ more expressive, but slower)
    lengths = [1500, 2500, 3500, 5000, 8000]

        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)

        # Number of restarts per length (fewer for the biggest n)
        if n <= 3500:
            restarts = 5
        elif n <= 5000:
            restarts = 3
        else:
            restarts = 2

        for r in range(restarts):
            if time.time() >= deadline:
                break

            # ---- Initialise a candidate ---------------------------------
            # Randomly choose one of two simple families
            if np.random.rand() < 0.6:
                # Power‑law family
                p = 0.5 + np.random.uniform(-0.15, 0.15)
                a0 = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
            else:
                # Dirichlet‑like random vector (concentrated when α<1)
                alpha = np.random.uniform(0.1, 0.8)
                a0 = np.random.gamma(alpha, 1.0, size=n)

            # Normalise to the ℓ¹‑simplex
            a0 *= target_sum / a0.sum()
            # Small jitter to break symmetry
            a0 += np.random.normal(scale=target_sum * 0.002, size=n)
            a0 = np.maximum(a0, 0.0)
            a0 = _project_to_simplex(a0, target_sum)

            # ------------------ time allocation -------------------------
            remaining = deadline - time.time()
            if remaining <= 5.0:
                # Not enough time for a full run – bail out
                break
            # Allocate at most 30 s for the Adam phase, the rest for refinements
            adam_deadline = min(time.time() + min(30.0, remaining * 0.5), deadline)

            # ------------------ Adam optimisation -----------------------
            a_opt, val_opt = _adam_optimize(a0, adam_deadline)

            # ------------------ Hill‑climbing refinement --------------
            if time.time() < deadline:
                a_opt, val_opt = _random_hill_climb(a_opt, deadline)

            # ------------------ Simulated annealing --------------------
            if time.time() < deadline:
                a_opt, val_opt = _simulated_annealing(a_opt, deadline)

            # ------------------ Record global best --------------------
            if val_opt < best_score:
                best_score = val_opt
                best_seq = a_opt.copy()
                print(f"** NEW GLOBAL BEST **  n={n} restart={r} score={best_score:.6f}")

    # Fallback in the extremely unlikely case that nothing was found
    if best_seq is None:
        n = 1000
        best_seq = np.ones(n, dtype=np.float64) * np.sqrt(2.0 * n) / n
        best_score = _score(best_seq)
        print(f"[{time.time():.1f}] Fallback uniform sequence, score={best_score:.6f}")

    # Trim trailing near‑zero entries (purely cosmetic)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Return as plain Python list (required by the grader)
    print(f"[{time.time():.1f}] Final best score = {best_score:.6f}, length = {len(best_seq)}")
    return best_seq.tolist()
