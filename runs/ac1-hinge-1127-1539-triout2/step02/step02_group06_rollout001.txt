"""
Search for a non‑negative height sequence that minimises `evaluate_sequence`.

The algorithm works in three stages:

1. **Initialisation** – Use the previously discovered sequence
   (`height_sequence_1`) if it exists, otherwise start from a random
   normalised vector of moderate length (≈ 3000). The ℓ₁‑norm is fixed to
   √(2·n) so that the objective becomes simply the maximum entry of the
   self‑convolution.

2. **LP‑guided improvement** – Repeatedly solve a linear programme
   `max Σ g_j` subject to `f * g ≤ M` (where `M` is the current maximal
   convolution value).  The solution `g` uses the available “mass budget”
   most efficiently while keeping the convolution bounded.  The current
   sequence `f` is blended towards `g` with a step size that is reduced
   until the maximal convolution strictly decreases.  This technique is
   described in the statement and typically lowers the objective by a
   few percent.

3. **Smooth‑max gradient polishing** – After the LP phase a
   smooth‑max (log‑sum‑exp) approximation of the maximum is optimised by
   projected gradient descent with momentum.  Occasional random
   pair‑wise transfers and Gaussian jitter are added to escape plateaus.
   The loop stops when the 1000 s wall‑clock budget is exhausted.

The best sequence found is returned as a plain Python list.
"""

import time
import numpy as np

# SciPy is used for the linear programme; if unavailable we fall back to
# gradient descent only.
try:
    from scipy.optimize import linprog
    _SCIPY_OK = True
except Exception:  # pragma: no cover
    _SCIPY_OK = False


def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector to have ℓ₁‑norm equal to *target*."""
    s = v.sum()
    if s > 0.0:
        return v * (target / s)
    # fallback – uniform distribution (should never happen)
    return np.full_like(v, target / v.shape[0])


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """Move a tiny fraction of mass from one random index to another."""
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0.0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_to_sum(a, target_sum)


def jitter(a: np.ndarray, target_sum: float, sigma: float = 1e-3) -> np.ndarray:
    """Add small Gaussian noise and renormalise."""
    a = a + np.random.randn(a.shape[0]) * sigma
    a = np.clip(a, 0.0, None)
    return project_to_sum(a, target_sum)


def fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using an FFT of power‑of‑2 length (real input)."""
    n, m = a.shape[0], b.shape[0]
    L = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)
    return conv[: n + m - 1]


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself."""
    return fft_convolve(a, a)


def smooth_max_step(
    a: np.ndarray,
    target_sum: float,
    lr: float,
    tau: float,
    momentum: float,
    v_prev: np.ndarray,
) -> tuple[np.ndarray, np.ndarray]:
    """
    One iteration of gradient descent on a smoothed maximum of the convolution.

    Returns the updated vector and the new momentum buffer.
    """
    n = a.shape[0]

    # (1) convolution a * a (full length 2·n‑1)
    conv = conv_self_fft(a)

    # (2) soft‑max weights w_k ∝ exp(conv_k / τ)
    max_c = conv.max()
    exp_arg = (conv - max_c) / tau
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:                          # degenerate – fall back to hard max
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    # (3) gradient of the smoothed max:
    #     g_i = 2 * Σ_k w_k * a_{k-i}
    a_rev = a[::-1]
    grad_full = fft_convolve(w, a_rev)        # length 3·n‑2
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]  # slice i = 0 … n‑1

    # (4) projected direction (preserve the ℓ₁‑norm)
    direction = -(grad - np.mean(grad))

    # (5) momentum + step
    v = momentum * v_prev + direction
    a_new = a + lr * v

    # (6) enforce feasibility
    a_new = np.clip(a_new, 0.0, None)
    a_new = project_to_sum(a_new, target_sum)

    return a_new, v


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (shape (2·n‑1, n)) such that (f * g) = A @ g.
    Entry A[k, j] = f[k‑j] if 0 ≤ k‑j < n else 0.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.empty((rows, n), dtype=np.float64)
    # Fill column‑wise: column j is f shifted by j positions.
    for j in range(n):
        A[j : j + n, j] = f
    return A


def evaluate_score(a: np.ndarray) -> float:
    """
    Fast version of the evaluation (the true objective is 2·n·max(conv)/(sum^2)
    which equals max(conv) when sum = √(2·n)).
    """
    conv = np.convolve(a, a)
    M = conv.max()
    return float(M)


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises ``evaluate_sequence``.

    The function respects the wall‑clock budget and always returns the best
    sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5       # small safety margin

    # ------------------------------------------------------------------
    # 1️⃣ Choose a problem size and initialise a starting sequence.
    # ------------------------------------------------------------------
    base_seq = globals().get("height_sequence_1")
    if isinstance(base_seq, (list, tuple)) and len(base_seq) > 0:
        # Use the length of a known good construction if it is not huge.
        n0 = len(base_seq)
        if n0 > 5000:          # too large for the LP stage – downsample
            n = 3000
        else:
            n = n0
    else:
        n = np.random.randint(2600, 3400)   # moderate size, cheap for LP

    target_sum = np.sqrt(2.0 * n)

    if isinstance(base_seq, (list, tuple)) and len(base_seq) >= n:
        a = np.array(base_seq[:n], dtype=np.float64)
        a = project_to_sum(a, target_sum)
    else:
        # random start
        a = np.random.rand(n)
        a = project_to_sum(a, target_sum)

    best_seq = a.copy()
    best_score = evaluate_score(best_seq)
    print(f"[{time.time():.2f}] Start – n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ LP‑guided improvement (only if SciPy is available).
    # ------------------------------------------------------------------
    if _SCIPY_OK:
        max_lp_iters = 12
        blend_t = 0.4                      # initial blending factor
        for it in range(max_lp_iters):
            if time.time() >= deadline:
                break

            # current bound M
            conv_ff = np.convolve(a, a)
            M = conv_ff.max()

            # Build linear constraints A·g ≤ M·1
            A = build_conv_matrix(a)
            b = np.full(A.shape[0], M, dtype=np.float64)

            # Linear programme: maximise Σ g_j  (i.e. minimise -Σ g_j)
            c = -np.ones(n, dtype=np.float64)
            bounds = [(0.0, None)] * n

            # Solve (HiGHS is fast for the size we use)
            res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
            if not res.success:
                print(f"LP iteration {it}: linprog failed ({res.message}) – abort LP stage.")
                break

            g0 = np.asarray(res.x, dtype=np.float64)
            sum_g0 = g0.sum()
            if sum_g0 <= 0.0:
                print(f"LP iteration {it}: zero solution – abort LP stage.")
                break
            g = g0 * (target_sum / sum_g0)   # renormalise to the correct ℓ₁‑norm

            # Try to blend towards g; reduce the step until we get an improvement.
            improved = False
            dt = blend_t
            for _ in range(6):
                cand = (1.0 - dt) * a + dt * g
                cand = np.clip(cand, 0.0, None)
                cand = project_to_sum(cand, target_sum)
                cand_score = evaluate_score(cand)
                if cand_score < best_score - 1e-12:
                    a = cand
                    best_seq = cand.copy()
                    best_score = cand_score
                    improved = True
                    print(f"LP iter {it:02d}, dt={dt:.4f} → new score {best_score:.6f}")
                    break
                dt *= 0.5  # halve step size and retry

            if not improved:
                # No improvement – reduce the blending factor for next iteration
                blend_t *= 0.7
                print(f"LP iter {it:02d} – no improvement (blend_t now {blend_t:.4f})")
            else:
                # Successful step – keep a relatively aggressive blend factor
                blend_t = min(blend_t * 1.1, 0.6)

    else:
        print("SciPy not available – skipping LP phase.")

    # ------------------------------------------------------------------
    # 3️⃣ Smooth‑max gradient descent with occasional random moves.
    # ------------------------------------------------------------------
    v = np.zeros_like(a)                     # momentum buffer
    lr = 0.03 * (target_sum / n)              # base learning rate (≈3 % of avg entry)
    tau = 0.15 * a.max()                      # initial temperature
    momentum = 0.9

    iteration = 0
    no_improve = 0
    while time.time() < deadline:
        a, v = smooth_max_step(a, target_sum, lr, tau, momentum, v)
        cur_score = evaluate_score(a)
        iteration += 1

        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"Iter {iteration:06d} – improvement: {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # schedule adaptations
        if iteration % 250 == 0:
            lr = max(lr * 0.93, 1e-7)
            tau = max(tau * 0.94, 1e-5)

        # occasional random pair‑mass transfer (helps escaping plateaus)
        if np.random.rand() < 0.025:
            a = random_pair_transfer(a, target_sum, scale=0.009)

        # jitter if stuck for a while
        if no_improve >= 1800:
            a = jitter(a, target_sum, sigma=5e-4)
            v[:] = 0.0
            no_improve = 0
            print(f"Iter {iteration:06d} – jitter reset")

    # ------------------------------------------------------------------
    # 4️⃣ Final quick polishing (if time permits)
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Search finished – best score {best_score:.6f}")

    while time.time() < deadline - 0.2:
        a = random_pair_transfer(a, target_sum, scale=0.004)
        cur_score = evaluate_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"Polish – improvement to {best_score:.6f}")

    # Return as plain Python list
    return best_seq.tolist()
