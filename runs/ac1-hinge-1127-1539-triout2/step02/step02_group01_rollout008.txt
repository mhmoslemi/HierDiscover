"""
Advanced multi‑restart Adam‑based smoothing optimisation for the min‑max
self‑convolution problem.

The algorithm:

1. Determine a problem size n (reuse an existing solution if
   `height_sequence_1` is present).  Normalise the ℓ¹‑norm to √(2 n).

2. Build a small pool of diverse initial vectors:
   – the provided baseline,
   – uniform,
   – several parametric “U‑shapes’’ (linear, quadratic, exponential,
     power‑law),
   – a few Dirichlet random draws.
   Keep the few best of them.

3. For each selected start run a projected Adam optimiser.
   The objective is a smoothed approximation of
        max(conv(a,a))
   using the log‑sum‑exp trick (τ‑smoothing).  When τ becomes tiny we
   switch to the exact sub‑gradient (indicator of the maximal entries).
   After each gradient step the iterate is projected onto the
   ℓ¹‑simplex (non‑negative entries with the required sum).

   Hyper‑parameters (learning‑rate, τ‑factor) are decayed during the
   run; occasional random pair‑mass transfers and occasional jitter are
   applied to escape plateaus.

4. Keep the best sequence ever observed (according to the true
   `evaluate_sequence` value).  After the time budget is exhausted a
   final short polishing phase applies many tiny random transfers.

The function returns the best list of floats found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a power‑of‑two FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def conv_cross_fft(w: np.ndarray, a_rev: np.ndarray) -> np.ndarray:
    """Convolution of w (length 2n‑1) with a_rev (length n) via FFT."""
    L = 1 << ((w.shape[0] + a_rev.shape[0] - 1).bit_length())
    fw = np.fft.rfft(w, n=L)
    fa = np.fft.rfft(a_rev, n=L)
    conv = np.fft.irfft(fw * fa, n=L)
    return conv[: w.shape[0] + a_rev.shape[0] - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex { x ≥ 0, Σ x = target }.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)
    # work on a copy
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # degenerate case – return uniform distribution
        return np.full_like(v, target / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def generate_U_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
) -> np.ndarray:
    """Symmetric U‑shaped vector (linear, quadratic, exponential)."""
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid  # distance ∈ [0,1]
    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    else:
        raise ValueError("unknown kind")
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def generate_power_shape(
    n: int,
    target_sum: float,
    exponent: float = 1.0,
) -> np.ndarray:
    """
    Symmetric power‑law shape: a_i ∝ (i+1)^p + (n-i)^p.
    Larger exponent concentrates mass at the ends.
    """
    idx = np.arange(n, dtype=np.float64)
    raw = (idx + 1) ** exponent + (n - idx) ** exponent
    return project_onto_simplex(raw, target_sum)


def random_dirichlet(n: int, target_sum: float) -> np.ndarray:
    """Random vector drawn from Dirichlet(1,…,1) and scaled to target_sum."""
    vec = np.random.dirichlet(np.ones(n))
    return vec * target_sum


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a small amount of mass from a random index i to j,
    then project back onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def indicator_weights(conv: np.ndarray, max_val: float, eps: float = 1e-12) -> np.ndarray:
    """
    Return a weight vector that is uniform on the indices achieving the maximum.
    """
    mask = np.abs(conv - max_val) <= eps
    count = mask.sum()
    if count == 0:
        # fallback to uniform
        return np.full_like(conv, 1.0 / conv.size)
    w = np.where(mask, 1.0 / count, 0.0)
    return w


# ----------------------------------------------------------------------
# Core optimisation step (Adam with smoothing)
# ----------------------------------------------------------------------


def adam_step(
    a: np.ndarray,
    target_sum: float,
    m: np.ndarray,
    v: np.ndarray,
    t: int,
    eta: float,
    beta1: float,
    beta2: float,
    eps: float,
    tau: float,
    use_indicator: bool,
) -> tuple[np.ndarray, np.ndarray, np.ndarray, float]:
    """
    Perform a single Adam optimisation step on the smoothed (or exact) max‑convolution.
    Returns the updated a, the new moments m,v and the current max‑convolution value.
    """
    n = a.shape[0]

    # 1) self‑convolution
    conv = conv_self_fft(a)
    max_conv = conv.max()

    # 2) weight vector for the smoothed max
    if use_indicator:
        w = indicator_weights(conv, max_conv)
    else:
        # log‑sum‑exp smoothing; τ = tau
        shift = conv - max_conv  # non‑positive, prevents overflow
        if tau <= 0:
            w = indicator_weights(conv, max_conv)
        else:
            w_raw = np.exp(shift / tau)
            w_sum = w_raw.sum()
            if w_sum == 0:
                w = np.full_like(w_raw, 1.0 / w_raw.size)
            else:
                w = w_raw / w_sum

    # 3) gradient of the (smoothed) max: grad_i = 2 * Σ_k w_k a_{k-i}
    a_rev = a[::-1]
    conv_full = conv_cross_fft(w, a_rev)          # length = 3n‑2
    grad = 2.0 * conv_full[n - 1 : n - 1 + n]     # slice of length n

    # 4) Adam moment updates
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad ** 2)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    # 5) parameter update + projection
    step = eta * m_hat / (np.sqrt(v_hat) + eps)
    a_new = a - step
    # enforce non‑negativity before simplex projection
    a_new = np.maximum(a_new, 0.0)
    a_new = project_onto_simplex(a_new, target_sum)

    return a_new, m, v, max_conv

    Larger exponent concentrates mass at the ends.
    """
    idx = np.arange(n, dtype=np.float64)
    raw = (idx + 1) ** exponent + (n - idx) ** exponent
    return project_onto_simplex(raw, target_sum)


def random_dirichlet(n: int, target_sum: float) -> np.ndarray:
    """Random Dirichlet(1,…,1) vector scaled to the required ℓ¹‑norm."""
    vec = np.random.dirichlet(np.ones(n))
    return vec * target_sum


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a tiny amount of mass from a random index i to j,
    then project back onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def indicator_weights(conv: np.ndarray, max_val: float, eps: float = 1e-12) -> np.ndarray:
    """Uniform weights on the entries attaining the maximal convolution value."""
    mask = np.abs(conv - max_val) <= eps
    cnt = mask.sum()
    if cnt == 0:
        return np.full_like(conv, 1.0 / conv.size)
    w = np.where(mask, 1.0 / cnt, 0.0)
    return w


# ----------------------------------------------------------------------
# Core optimisation step (Adam with optional smoothing)
# ----------------------------------------------------------------------


def adam_step(
    a: np.ndarray,
    target_sum: float,
    m: np.ndarray,
    v: np.ndarray,
    t: int,
    eta: float,
    beta1: float,
    beta2: float,
    eps: float,
    tau_factor: float,
    use_indicator: bool,
) -> tuple[np.ndarray, np.ndarray, np.ndarray, float]:
    """
    Perform a single Adam optimisation step on the (smoothed) max‑convolution.
    Returns updated a, the new first/second moments, and the current maximal
    convolution value.
    """
    n = a.shape[0]

    # 1) self‑convolution and its maximum
    conv = conv_self_fft(a)
    max_conv = conv.max()

    # 2) compute weights for the smoothed max
    if use_indicator:
        w = indicator_weights(conv, max_conv)
    else:
        tau = max(tau_factor * max_conv, 1e-12)
        shift = conv - max_conv               # non‑positive
        w_raw = np.exp(shift / tau)
        w_sum = w_raw.sum()
        if w_sum == 0:
            w = np.full_like(w_raw, 1.0 / w_raw.size)
        else:
            w = w_raw / w_sum

    # 3) gradient of the (smoothed) maximal entry:
    #    grad_i = 2 * Σ_k w_k a_{k-i}
    a_rev = a[::-1]
    conv_full = conv_cross_fft(w, a_rev)      # length 3n‑2
    grad = 2.0 * conv_full[n - 1 : n - 1 + n]

    # 4) Adam moment updates
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad ** 2)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    # 5) parameter update + projection onto the ℓ¹‑simplex
    step = eta * m_hat / (np.sqrt(v_hat) + eps)
    a_new = a - step
    a_new = np.maximum(a_new, 0.0)
    a_new = project_onto_simplex(a_new, target_sum)

    return a_new, m, v, max_conv


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The search respects a ≈1000 s wall‑clock budget and returns the best
    sequence observed.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.25  # safety margin

    # --------------------------------------------------------------
    # Determine problem size (reuse previous construction if provided)
    # --------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        baseline = np.array(base, dtype=np.float64)
    else:
        n = np.random.randint(8000, 9500)   # a comfortable size
        baseline = None

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Assemble a small pool of diverse initial vectors
    # --------------------------------------------------------------
    candidates = []

    if baseline is not None:
        candidates.append(baseline.copy())

    candidates.append(np.full(n, target_sum / n))          # uniform

    # parametric “U‑shapes’’
    shape_kinds = ("linear", "quadratic", "exponential")
    param_grid = {
        "linear": np.linspace(0.0, 5.0, 26),       # 0 … 5 step 0.2
        "quadratic": np.linspace(0.0, 3.0, 16),   # 0 … 3 step 0.2
        "exponential": np.linspace(0.0, 2.0, 21),
    }
    for kind in shape_kinds:
        for p in param_grid[kind]:
            candidates.append(generate_U_shape(n, target_sum, kind=kind, param=p))

    # power‑law shapes (edge‑enhancing)
    for exp in [0.5, 1.0, 1.5, 2.0, 3.0]:
        candidates.append(generate_power_shape(n, target_sum, exponent=exp))

    # a few Dirichlet random draws
    for _ in range(6):
        candidates.append(random_dirichlet(n, target_sum))

    # --------------------------------------------------------------
    # Quick evaluation of the pool; keep the best few as starts
    # --------------------------------------------------------------
    def fast_score(vec: np.ndarray) -> float:
        return conv_self_fft(vec).max()

    scored = [(fast_score(c), c) for c in candidates]
    scored.sort(key=lambda x: x[0])
    top_k = 5
    start_points = [c.copy() for _, c in scored[:top_k]]

    best_seq = start_points[0].copy()
    best_score = fast_score(best_seq)

    print(f"[{time.time():.2f}] n={n}, target_sum={target_sum:.4f}")
    print(f"  Selected {len(start_points)} start points, best initial score {best_score:.6f}")

    # --------------------------------------------------------------
    # Optimisation loop – Adam with schedule, random jumps, jitter
    # --------------------------------------------------------------
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    start_idx = 0
    global_iter = 0

    while time.time() < deadline:
        # pick the next start point
        a = start_points[start_idx].copy()
        start_idx = (start_idx + 1) % len(start_points)

        m = np.zeros_like(a)
        v = np.zeros_like(a)

        # per‑run hyper‑parameters
        eta = 0.25 * (target_sum / n)          # initial learning rate
        tau_factor = 0.15                      # τ = τ_factor * max_conv
        min_tau_factor = 1e-4
        no_improve = 0
        iter_in_run = 0

        while time.time() < deadline:
            global_iter += 1
            iter_in_run += 1

            use_indicator = tau_factor < 5e-5   # switch to exact sub‑gradient

            a, m, v, cur_max = adam_step(
                a,
                target_sum,
                m,
                v,
                t=iter_in_run,
                eta=eta,
                beta1=beta1,
                beta2=beta2,
                eps=eps,
                tau_factor=tau_factor,
                use_indicator=use_indicator,
            )

            if cur_max + 1e-12 < best_score:
                best_score = cur_max
                best_seq = a.copy()
                print(f"[{time.time():.2f}] Iter {global_iter:07d} – new best {best_score:.6f}")
                no_improve = 0
            else:
                no_improve += 1

            # schedule decay
            if iter_in_run % 600 == 0:
                eta *= 0.94
                tau_factor = max(min_tau_factor, tau_factor * 0.93)

            # occasional random mass transfer
            if np.random.rand() < 0.012:
                a = random_pair_transfer(a, target_sum, scale=0.012)

            # jitter after a long plateau
            if no_improve >= 3000:
                jitter = np.random.randn(n) * 0.0004
                a = project_onto_simplex(np.maximum(a + jitter, 0.0), target_sum)
                no_improve = 0
                print(f"[{time.time():.2f}] Iter {global_iter:07d} – jitter applied")

            # give up on this start after very long stagnation
            if no_improve >= 12000:
                break

    # --------------------------------------------------------------
    # Final polishing – cheap random pair transfers
    # --------------------------------------------------------------
    print(f"[{time.time():.2f}] Optimisation finished, best score {best_score:.6f}")

    polish_iters = 0
    while time.time() < deadline and polish_iters < 5000:
        polish_iters += 1
        cand = random_pair_transfer(best_seq.copy(), target_sum, scale=0.004)
        cand_score = fast_score(cand)
        if cand_score + 1e-12 < best_score:
            # verify with the official evaluator (more expensive but safe)
            true_score = evaluate_sequence(cand.tolist())
            if true_score < best_score:
                best_score = true_score
                best_seq = cand.copy()
                print(f"[{time.time():.2f}] Polish improvement to {best_score:.6f}")

    # --------------------------------------------------------------
    # Return the best sequence found
    # --------------------------------------------------------------
    return best_seq.tolist()
