"""
Propose a non‑negative step‑function (vector of heights) that minimizes  

    2 * n * max(conv(a, a)) / (sum(a)**2)

where `conv` is the linear self‑convolution.
The algorithm works in three stages:

1. **Coarse parametric sweep** – generate a few symmetric “U‑shapes”
   (linear, quadratic, exponential) and keep the best one.
2. **Smoothed gradient descent with momentum** – minimise the
   log‑sum‑exp smoothed version of the max‑convolution while staying on the
   ℓ₁‑simplex (sum = √(2 n)).  A momentum term accelerates convergence and
   the smoothing temperature τ is slowly reduced.
3. **Sub‑gradient refinement & polishing** – after the smoothed phase we
   switch to a pure sub‑gradient step that only follows the current maximum
   entry of the convolution, interleaved with occasional random mass
   transfers.  Finally a short random‑perturbation pass is performed.

The whole search respects a 1000 s time budget and always returns the best
sequence encountered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top‑level, no closures or lambdas
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a vector with itself using FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex
        {x | x_i ≥ 0, sum_i x_i = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, target / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a tiny amount of mass from one random index to another,
    then project back onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def generate_U_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
) -> np.ndarray:
    """
    Produce a symmetric “U‑shaped’’ sequence of length n.
    kind ∈ {linear, quadratic, exponential}.
    param controls the edge enhancement (≥0).  The result is scaled
    to have ℓ₁‑norm `target_sum`.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def _smoothed_gradient(a: np.ndarray, tau_factor: float):
    """
    Compute the gradient of the log‑sum‑exp smoothed max‑convolution.
    Returns (gradient, max_conv).
    """
    n = a.shape[0]
    conv = conv_self_fft(a)                     # length 2n‑1
    max_conv = conv.max()
    # smoothing temperature
    tau = max(tau_factor * max_conv, 1e-12)
    # stable log‑sum‑exp weights
    w = np.exp((conv - max_conv) / tau)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # gradient = (2/τ) * (w * a_rev) restricted to the central n entries
    a_rev = a[::-1]
    L = _next_pow2(len(w) + len(a_rev) - 1)
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
    grad = (2.0 / tau) * conv_full[n - 1 : n - 1 + n]
    return grad, max_conv


def _subgradient_max(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a, a)) (the max entry of the self‑convolution).
    Returns (gradient, max_conv, idx_max).
    """
    n = a.shape[0]
    conv = conv_self_fft(a)
    max_conv = conv.max()
    idx_max = int(conv.argmax())
    # indices i that actually appear in the maximal entry
    low = max(0, idx_max - (n - 1))
    high = min(idx_max, n - 1)
    idx_range = np.arange(low, high + 1, dtype=int)
    grad = np.zeros_like(a)
    grad[idx_range] = 2.0 * a[idx_max - idx_range]
    return grad, max_conv, idx_max

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function minimizing `evaluate_sequence`.
    The search stops after ~budget_s seconds and returns the best vector.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # small safety margin

    # ------------------------------------------------------------------
    # Problem size – use existing construction if present
    # ------------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        base_vec = np.array(base, dtype=np.float64)
    else:
        n = np.random.randint(7500, 8700)    # a comfortable size
        base_vec = None

    target_sum = np.sqrt(2.0 * n)               # normalisation makes objective = max(conv)

    # ------------------------------------------------------------------
    # Phase 1 – coarse parametric sweep (U‑shapes)
    # ------------------------------------------------------------------
    best_seq = np.full(n, target_sum / n)       # uniform – just a placeholder
    best_score = float("inf")

    print(f"[{time.time():.2f}] Starting parametric sweep (n={n})")

    # linear shapes
    for p in np.linspace(0.0, 4.0, 41):
        cand = generate_U_shape(n, target_sum, kind="linear", param=p)

        if score < best_score:

            print(f"  New best linear (p={p:.2f}) → {best_score:.6f}")
        if time.time() >= deadline:
            return best_seq.tolist()

    # quadratic shapes
    for p in np.linspace(0.0, 3.0, 31):

            return best_seq.tolist()

    # exponential shapes
    for p in np.linspace(0.0, 2.0, 21):
        cand = generate_U_shape(n, target_sum, kind="exponential", param=p)
        score = conv_self_fft(cand).max()
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"  New best exponential (p={p:.2f}) → {best_score:.6f}")

            return best_seq.tolist()

    if base_vec is not None:
        score = conv_self_fft(base_vec).max()

        # momentum update

        a_new = a - eta * velocity
        a_new = project_onto_simplex(np.maximum(a_new, 0.0), target_sum)
        a = a_new
        iteration += 1

        if cur_max + 1e-12 < best_score:
            best_score = cur_max
            best_seq = a.copy()
            print(f"Iter {iteration:06d} – improved to {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # adaptive schedules
        if iteration % 600 == 0:

            a = project_onto_simplex(np.maximum(a + jitter, 0.0), target_sum)
            no_improve = 0
            print(f"Iter {iteration:06d} – jitter applied")

    # ------------------------------------------------------------------

    # sub‑gradient refinement (focus only on the maximal convolution entry)
    eta_sub = 0.5 * (target_sum / n)

    while time.time() < deadline and sub_iter < 3000:

            print(f"Sub‑grad iter {sub_iter:04d} – improved to {best_score:.6f}")
            eta_sub = max(eta_sub * 1.02, 0.01 * (target_sum / n))
