import time
import numpy as np

def project_onto_simplex(v, S):
    # Using simple renormalization after clipping.
    v = np.clip(v, 0, None)
    total = v.sum()
    if total == 0.0:
        # uniform fill
        return np.full_like(v, S / v.size)
    v = v * (S / total)
    return v

def softmax_weights(conv, alpha):
    """Return weights w_i = softmax(alpha*conv_i)."""
    # subtract max to prevent overflow
    maxv = conv.max()
    exp_term = np.exp(alpha * (conv - maxv))
    sum_exp = exp_term.sum()
    if sum_exp == 0.0:
        return np.full_like(conv, 1.0/conv.size)
    return exp_term / sum_exp

def compute_self_convolution(a):
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv

def compute_gradient_from_weights(a, w):
    # gradient = 2 * (w * convolve a_rev?) 
    n = a.shape[0]
    a_rev = a[::-1]  # length n
    conv_len = len(w) + n - 1  # = 3n-2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fw = np.fft.rfft(w, n=fft_len)
    Fa_rev = np.fft.rfft(a_rev, n=fft_len)
    conv2 = np.fft.irfft(Fw * Fa_rev, n=fft_len)
    # take slice from n-1 to n-1 + n
    start = n - 1
    grad = 2.0 * conv2[start:start + n]
    return grad

    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    while time.time() < deadline and t < max_iters:

        # update best if improved

        # alpha schedule

"""
Search for a non‑negative step‑function minimizing the max self‑convolution.
The algorithm uses Adam with a soft‑max surrogate of the max‑convolution.
A simple renormalisation (projection onto the simplex Σa = √(2 n))
keeps the L1‑norm fixed, which makes the external score equal to the
maximum entry of the self‑convolution.  Multiple random restarts with
different lengths are performed until the time budget expires.
"""

import time
import numpy as np

# -------------------------------------------------------------
# Helper utilities
# -------------------------------------------------------------


def _self_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using FFT (real‑valued)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Return w_i = softmax(alpha * conv_i)."""
    # shift by maximum for numerical stability
    m = conv.max()
    exp_term = np.exp(alpha * (conv - m))
    s = exp_term.sum()
    if s == 0.0:                     # extremely unlikely, but safe guard
        return np.full_like(conv, 1.0 / conv.size)
    return exp_term / s


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the (soft‑max) surrogate.
    For each shift k we need 2·a_{k‑j}; the formula can be obtained as
    g = 2 * (w ⋆ a_rev) and then taking the centre block of length n.
    """
    n = a.shape[0]
    a_rev = a[::-1]                         # reversed a
    # length of the linear convolution w * a_rev
    conv_len = len(w) + n - 1                # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fw = np.fft.rfft(w, n=fft_len)
    Fa_rev = np.fft.rfft(a_rev, n=fft_len)
    conv2 = np.fft.irfft(Fw * Fa_rev, n=fft_len)
    start = n - 1
    grad = 2.0 * conv2[start:start + n]
    return grad


def _renormalise(a: np.ndarray, target_sum: float) -> np.ndarray:
    """Clamp to >=0 and rescale to achieve the given L1‑norm."""
    a = np.clip(a, 0.0, None)
    total = a.sum()
    if total == 0.0:
        a.fill(target_sum / a.size)
    else:
        a *= (target_sum / total)
    return a


# -------------------------------------------------------------
# Optimisation of a single start
# -------------------------------------------------------------


def _run_one_start(a0: np.ndarray,
                   deadline: float,
                   max_iters: int = 3000,
                   eta0: float = 0.6,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   alpha_start: float = 0.2,
                   alpha_end: float = 8.0,
                   jitter_prob: float = 0.02,
                   jitter_scale: float = 0.01,
                   decay: float = 0.99995) -> tuple[np.ndarray, float]:
    """
    Adam optimisation with a soft‑max surrogate.
    Returns the best vector seen and its associated max‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)           # Σa = √(2 n)

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_max = float('inf')
    best_seq = a.copy()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # full self‑convolution and surrogate weights
        conv = _self_convolution(a)
        cur_max = conv.max()
        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_seq = a.copy()
            print(f"[local improve] iter={it} n={n} max_conv={best_max:.6f}")

        # schedule for the soft‑max temperature
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)

        # soft‑max weights and gradient
        w = _softmax_weights(conv, alpha)
        g = _gradient_from_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # occasional jitter to escape plateaus
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * jitter_scale
            a = a + noise

        # enforce non‑negativity and keep the L1‑norm fixed
        a = _renormalise(a, target_sum)

        # slowly decay the learning‑rate
        eta0 *= decay

    # final evaluation of the best vector found in this run
    return best_seq, best_max


# -------------------------------------------------------------
# Main entry point required by the competition harness
# -------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000):
    """
    Return a list of heights (non‑negative) that minimises the provided
    evaluate_sequence() score.  Multiple restarts of the Adam/soft‑max
    optimiser are executed until the time budget runs out.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # small safety margin

    best_seq = None
    best_score = float('inf')

    # -----------------------------------------------------------------
    # If a previous construction is present, record its score.
    # -----------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            if cand.ndim == 1 and cand.size > 0:
                sc = evaluate_sequence(cand.tolist())
                best_seq = cand.copy()
                best_score = sc
                print(f"[baseline] existing height_sequence_1 score = {best_score:.6f} length = {cand.size}")
        except Exception:
            pass

    # -----------------------------------------------------------------
    # Lengths to explore – longer sequences generally allow lower scores.
    # -----------------------------------------------------------------
    lengths = [4000, 8000, 12000, 16000]

    run_id = 0
    while time.time() < deadline:
        n = int(rng.choice(lengths))

        # ---- initial vector -------------------------------------------------
        # Warm‑start from the current global best (truncated) with some probability:
        if best_seq is not None and best_seq.size >= n and rng.random() < 0.25:
            a0 = best_seq[:n].copy()
        else:
            # Random Dirichlet‑like draw (exponential samples)
            a0 = rng.exponential(scale=1.0, size=n)

        # Normalise to the required L1‑norm √(2 n)
        S = np.sqrt(2.0 * n)
        a0 = a0 / a0.sum() * S

        # ---- run the optimiser --------------------------------------------
        seq_opt, max_conv = _run_one_start(
            a0,
            deadline,
            max_iters=3000,
            eta0=0.6,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            alpha_start=0.2,
            alpha_end=8.0,
            jitter_prob=0.02,
            jitter_scale=0.01,
            decay=0.99995,
        )

        # External evaluation (the true score).  With the L1‑norm fixed,
        # this equals the maximal entry of the self‑convolution.
        try:
            score = evaluate_sequence(seq_opt.tolist())
        except Exception:
            score = float('inf')

        if score < best_score:
            best_score = score
            best_seq = seq_opt.copy()
            print(f"[global improve] run={run_id} n={n} score={best_score:.6f} max_conv={max_conv:.6f}")

        run_id += 1

    # -----------------------------------------------------------------
    # Fallback: uniform sequence (should never be needed)
    # -----------------------------------------------------------------
    if best_seq is None:
        n = 2000
        a = np.full(n, np.sqrt(2.0 * n) / n)
        best_seq = a
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] score={best_score:.6f}")

    print(f"[final] best score = {best_score:.6f}  length = {len(best_seq)}")
    return best_seq.tolist()
