def _subgradient_max(a):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
    M = b.max()
    # mask as float due to numeric issues
    mask = (b >= M - 1e-12).astype(np.float64)
    # Compute gradient: g = 2 * (a_rev ⋆ mask) central part
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1
    size2 = 1 << ((conv_len).bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Mask_fft = np.fft.rfft(mask, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Mask_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n-1:n-1+n]
    return g, b

"""
Optimiser for the step‑function bound.

The objective after fixing the ℓ¹‑norm (∑a = √(2n)) is simply the
maximum entry of the linear convolution a*a.  The algorithm therefore
minimises `max(conv(a,a))` under the constraints a_i ≥ 0, ∑a_i = √(2n).

Approach
--------
1.  Randomly initialise a‑vectors with several parametric families
    (power‑law, exponential, symmetric variants, Dirichlet draws and,
    when available, the previously discovered best construction).

2.  Phase‑A – a smooth approximation of the max (log‑sum‑exp) is
    minimised by projected gradient descent.  The temperature β is
    annealed from 1 up to 2048, giving a progressively tighter
    approximation.

3.  Phase‑B – sub‑gradient descent on the true max.  The sub‑gradient
    is obtained by a convolution with the mask of indices attaining the
    current maximum.  A decreasing step size (∝1/√t) and simplex
    projection keep the iterates feasible.

4.  Phase‑C – a cheap random hill‑climber that moves a small fraction
    of mass between two positions while preserving the ℓ¹‑norm.

The whole procedure is run for several vector lengths and random
restarts until the allotted time (~1000 s) expires.  The best sequence
found is returned as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """Fast version of the competition objective (no security checks)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    # The denominator is 2·n after fixing the ℓ¹‑norm, so the objective
    # equals max_b.  We keep the full expression for clarity.
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Gradient of the soft‑max (log‑sum‑exp) approximation of the max
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b) where
        b = a*a (full convolution, length 2·n‑1)
        g = ∇ (log‑sum‑exp_{β}(b)) = 2·(a ⋆ p)   with p_k ∝ exp(β·b_k)
    Both are computed with FFTs for speed.
    """
    n = a.shape[0]

    # ---- b = a*a -------------------------------------------------------
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ---- soft‑max probabilities p ---------------------------------------
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)                # shape (2·n‑1,)

    # ---- gradient g = 2·(a ⋆ p) ---------------------------------------
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1        # = 3·n‑2
    size2 = 1 << ((conv_len).bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1 : n - 1 + n]

    return g, b


# ----------------------------------------------------------------------
# Sub‑gradient for the true maximum (non‑smooth)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a,a)).
    Returns (g, b) where b is the full convolution and g = 2·(a ⋆ mask)
    with mask indicating the positions attaining the maximum.
    """
    n = a.shape[0]
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    M = b.max()
    # mask of the active indices (exact equality is fine for floating point)
    mask = (b == M).astype(np.float64)

    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1
    size2 = 1 << ((conv_len).bit_length())
    ar_fft = np.fft.rfft(a_rev, n=size2)
    mask_fft = np.fft.rfft(mask, n=size2)
    conv_rev = np.fft.irfft(ar_fft * mask_fft, n=size2)[:conv_len]

    g = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Random hill‑climbing (local refinement)
# ----------------------------------------------------------------------


def _random_hill_climb(a: np.ndarray, deadline: float):
    """
    Very cheap hill‑climber that preserves the ℓ¹‑norm.
    Randomly transfers a fraction of mass from i to j and keeps
    improving moves.
    """
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    cur_val = best_val

    max_frac = 0.12
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        frac = rng.random() * max_frac
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        new_val = _score(a)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill‑climb improved = {best_val:.6f}")
        else:
            a[j] -= delta
            a[i] += delta

    return best_a, best_val


# ----------------------------------------------------------------------
# Core optimisation routine (soft‑max + sub‑gradient + hill‑climb)
# ----------------------------------------------------------------------


def _optimize(a0: np.ndarray, sub_deadline: float):
    """
    Optimise a single initial vector a0 until `sub_deadline`.
    Returns the best vector found and its score.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    # ----- Phase A: annealed soft‑max gradient descent -----------------
    beta = 1.0
    max_beta = 2048.0
    step_base = target_sum * 0.15
    rng = np.random.default_rng()
    it = 0

    while time.time() < sub_deadline:
        # gradient of the smoothed max
        g, _ = _gradient_softmax(a, beta)
        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            lr = step_base / (norm_g * (beta + 1.0))
            a -= lr * g
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}]  beta={beta:.2f}  score={best_val:.6f}")

        # annealing schedule
        if beta < max_beta:
            beta = min(beta * 1.2, max_beta)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

        # preserve a small time slice for the next phase
        if time.time() + 5.0 > sub_deadline:
            break

    # ----- Phase B: sub‑gradient descent on the true max ---------------
    step_base_sub = target_sum * 0.50
    it = 0
    while time.time() < sub_deadline:
        g, _ = _subgradient_max(a)
        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            lr = step_base_sub / (norm_g * np.sqrt(it + 1))
            a -= lr * g
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}]  subgrad it={it}  score={best_val:.6f}")

        it += 1
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # stop early if we have exhausted the time budget
        if time.time() >= sub_deadline:
            break

    # ----- Phase C: final hill‑climbing refinement --------------------
    if time.time() < sub_deadline:
        a, final_val = _random_hill_climb(a, sub_deadline)
        if final_val < best_val:
            best_val = final_val
            best_a = a.copy()
            print(f"[{time.time():.1f}]  final hill‑climb best={best_val:.6f}")

    return best_a, best_val

# Entry point for the competition

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the competition
    objective.  The routine runs for at most ``budget_s`` seconds,
    trying several vector lengths and random initialisations.
    The best sequence found is returned as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    deadline = time.time() + budget_s - 0.5  # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Vector lengths to explore (larger lengths give more freedom)
    # ------------------------------------------------------------------
    lengths = [1500, 2500, 3500, 4500, 6000]

    # Incorporate the previously discovered construction if it exists.
    hs1 = globals().get("height_sequence_1", None)
    if hs1 is not None:
        hs1 = np.asarray(hs1, dtype=np.float64)
        n_hs1 = hs1.size
        if n_hs1 not in lengths:
            lengths.append(n_hs1)
        # ensure it is considered early
        lengths = [n_hs1] + [l for l in lengths if l != n_hs1]

    # ------------------------------------------------------------------
    # Main search loop
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)

        # Rough estimate of how many restarts we can still afford.
        remaining = deadline - time.time()
        max_per_restart = 25.0 if n < 4000 else 40.0
        allowed_restarts = max(1, int(remaining // max_per_restart))
        allowed_restarts = min(allowed_restarts, 8)  # keep a reasonable cap

        for restart in range(allowed_restarts):
            if time.time() >= deadline:
                break

            # ------------------------------------------------------------
            # Random initialisation (several families)
            # ------------------------------------------------------------
            shape_type = rng.integers(0, 6)  # 0‑4 are parametric, 5 = use hs1
            if shape_type == 0:                      # power‑law
                p = rng.uniform(0.30, 0.85)
                off = rng.uniform(0.0, 5.0)
                a0 = (np.arange(1, n + 1, dtype=np.float64) + off) ** (-p)

            elif shape_type == 1:                    # exponential
                lam = rng.uniform(4e-4, 5e-3)
                a0 = np.exp(-lam * np.arange(n, dtype=np.float64))

            elif shape_type == 2:                    # symmetric power‑law
                p = rng.uniform(0.30, 0.85)
                off = rng.uniform(0.0, 5.0)
                idx = np.arange(n, dtype=np.float64)
                a0 = (idx + 1 + off) ** (-p) + (n - idx + off) ** (-p)

            elif shape_type == 3:                    # symmetric exponential
                lam = rng.uniform(4e-4, 5e-3)
                idx = np.arange(n, dtype=np.float64)
                a0 = np.exp(-lam * idx) + np.exp(-lam * (n - 1 - idx))

            elif shape_type == 4:                    # Dirichlet / Gamma draw
                a0 = rng.gamma(shape=1.0, scale=1.0, size=n)

            else:                                    # use previous best construction
                if hs1 is not None and hs1.size == n:
                    a0 = hs1.copy()
                else:
                    # fallback to a simple Dirichlet draw
                    a0 = rng.gamma(shape=1.0, scale=1.0, size=n)

            # small random jitter before optimisation
            a0 += rng.normal(scale=target_sum * 0.001, size=n)
            a0 = np.maximum(a0, 0.0)
            a0 = _project_to_simplex(a0, target_sum)

            # ------------------------------------------------------------
            # Optimisation for this restart
            # ------------------------------------------------------------
            remaining = deadline - time.time()
            sub_deadline = time.time() + min(remaining - 0.5, max_per_restart)

            a_opt, opt_score = _optimize(a0, sub_deadline)

            if opt_score < best_score:
                best_score = opt_score
                best_seq = a_opt.copy()
                print(f"*** NEW GLOBAL BEST *** n={n} type={shape_type} "
                      f"restart={restart} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the unlikely case nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, 1.0, dtype=np.float64)
        best_seq *= np.sqrt(2.0 * n) / n

    # Trim trailing zeros (purely cosmetic)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    return best_seq.tolist()
