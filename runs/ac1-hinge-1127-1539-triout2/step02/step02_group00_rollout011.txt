def _subgradient_descent(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_val = _score(a)
    lr = 0.5 * target_sum   # initial step size in magnitude
    # maybe adapt lr per iteration.
    while time.time() < deadline:
        conv = _full_convolution(a)
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]
        # compute subgradient
        i_low = max(0, max_idx - (n - 1))
        i_high = min(max_idx, n - 1)
        g = np.zeros_like(a)
        idx = np.arange(i_low, i_high + 1)
        g[idx] = a[max_idx - idx]
        norm_g = np.linalg.norm(g)
        if norm_g < 1e-12:
            break
        # try step sizes
        improved = False
        step = lr
        while step > 1e-9:
            a_new = a - step * g
            a_new = _project_to_simplex(a_new, target_sum)
            val = _score(a_new)
            if val < best_val - 1e-12:
                a = a_new
                best_a = a_new.copy()
                best_val = val
                improved = True
                # increase step slightly
                lr = step * 1.2
                break
            else:
                step *= 0.5
        if not improved:
            # reduce step size globally
            lr *= 0.7
            if lr < 1e-9:
                break
    return best_a, best_val

"""
search‑candidate – a multi‑phase optimizer for the step‑function bound.

The algorithm works in three stages:

1.  Randomly generate a low‑dimensional parametric shape (power‑law,
    two‑segment power‑law or exponential) and project it onto the ℓ¹‑simplex
    {a≥0, Σa = √(2 n)} for a chosen length *n*.

2.  Refine the shape with an Adam optimiser on a smooth log‑sum‑exp
    approximation of  max(a∗a).  The temperature β is annealed from 1 up to
    1024, which makes the gradient approach the true sub‑gradient of the
    max‑convolution.

3.  Perform a few iterations of true sub‑gradient descent on the exact
    max‑convolution (using a line‑search) and finish with a tiny swap‑based
    hill‑climber that only re‑orders entries – the ℓ¹‑norm stays unchanged.

The whole routine runs for at most ``budget_s`` seconds, tries several
vector lengths (up to 8000 entries) and restarts, and returns the best
sequence found.  The objective value printed during the run is the
maximum entry of the auto‑convolution, i.e. the exact value returned by
``evaluate_sequence`` after normalisation.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (same as in the reference implementation)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum value of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Fast evaluation of the competition objective (max‑convolution)."""
    return _max_convolution_fft(a)


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    { x | x_i ≥ 0, Σ x_i = s } (Duchi et al. 2008).
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of the soft‑max (log‑sum‑exp) approximation of the max‑convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b) where

        b = a*a  (full convolution, length 2·n‑1)
        g = ∇ (log‑sum‑exp_{β}(b)) = 2·(a ⋆ p)

    with p_k ∝ exp(β·b_k).  Both are computed with FFTs.
    """
    n = a.shape[0]

    # b = a*a
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)                     # shape (2·n‑1,)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1              # = 3·n‑2
    size2 = 1 << ((conv_len).bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]

    g = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray, deadline: float):
    """Projected Adam optimisation of the log‑sum‑exp surrogate."""
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 1024.0
    base_lr = 0.1 * target_sum                     # base learning rate

    while time.time() < deadline:
        t += 1
        # gradient of the smooth surrogate
        g, _ = _gradient_softmax(a, beta)

        # Adam moment updates (β1=0.9, β2=0.999)
        m = 0.9 * m + 0.1 * g
        v = 0.999 * v + 0.001 * (g * g)

        m_hat = m / (1 - 0.9 ** t)
        v_hat = v / (1 - 0.999 ** t)

        step = base_lr / (np.sqrt(v_hat) + 1e-8)
        a = a - step * m_hat
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] Adam iter {t}, β={beta:.2f}, score={best_val:.6f}")

        # anneal temperature
        if t % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Uses a simple back‑tracking line‑search and a decreasing step size.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    # initial step size (scaled to the typical magnitude of a)
    step = 0.5 * target_sum

    while time.time() < deadline:
        conv = _full_convolution(a)
        k = int(np.argmax(conv))                     # index of the current maximum
        # sub‑gradient of the max‑convolution at position k
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        g = np.zeros_like(a)
        idx = np.arange(i_low, i_high + 1)
        g[idx] = a[k - idx]

        norm_g = np.linalg.norm(g)
        if norm_g < 1e-12:
            break

        improved = False
        cur_step = step
        while cur_step > 1e-9:
            a_new = a - cur_step * g
            a_new = _project_to_simplex(a_new, target_sum)
            val = _score(a_new)
            if val < best_val - 1e-12:               # strict improvement
                a = a_new
                best_a = a_new.copy()
                best_val = val
                improved = True
                # grow step a little after a successful move
                step = cur_step * 1.2
                print(f"[{time.time():.1f}] sub‑grad step success, score={best_val:.6f}")
                break
            else:
                cur_step *= 0.5

        if not improved:
            # no improvement – shrink the global step size
            step *= 0.7
            if step < 1e-9:
                break

    return best_a, best_val


# ----------------------------------------------------------------------
# Very cheap permutation hill‑climber
# ----------------------------------------------------------------------


def _swap_hill_climb(a0: np.ndarray, deadline: float, max_swaps: int = 500):
    """
    Swaps two entries (preserving the ℓ¹‑norm) and keeps improving moves.
    Stops after ``max_swaps`` attempts or the deadline.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _score(a)

    n = a.shape[0]
    rng = np.random.default_rng()
    swaps = 0

    while time.time() < deadline and swaps < max_swaps:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # perform swap
        a[i], a[j] = a[j], a[i]
        val = _score(a)
        swaps += 1

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] swap improved to {best_val:.6f}")

        if val >= best_val:
            # revert the swap
            a[i], a[j] = a[j], a[i]

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the supplied
    ``evaluate_sequence`` objective.  The routine respects the time budget
    and returns the best sequence it discovered.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    best_seq = None
    best_score = np.inf

    # we explore three lengths – larger n gives more flexibility
    lengths = [3000, 5000, 8000]

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # a few restarts for each length
        for restart in range(6):
            if time.time() >= deadline:
                break

            # ---------- random parametric initialisation ----------
            shape_type = rng.choice(['power', 'twopow', 'exp'])
            if shape_type == 'power':
                p = rng.uniform(0.2, 0.9)
                a0 = np.arange(1, n + 1, dtype=np.float64) ** (-p)
            elif shape_type == 'twopow':
                p1 = rng.uniform(0.2, 0.9)
                p2 = rng.uniform(0.2, 0.9)
                cut = rng.integers(int(0.2 * n), int(0.8 * n))
                a0 = np.empty(n, dtype=np.float64)
                # left segment
                a0[:cut] = np.arange(1, cut + 1, dtype=np.float64) ** (-p1)
                # continuity factor for the right segment
                left_val = (cut) ** (-p1)
                right_val = (cut + 1) ** (-p2)
                scale = left_val / right_val if right_val > 0 else 1.0
                a0[cut:] = scale * np.arange(cut + 1, n + 1, dtype=np.float64) ** (-p2)
            else:  # exponential
                rate = rng.uniform(0.0005, 0.01)
                a0 = np.exp(-rate * np.arange(n, dtype=np.float64))

            # add tiny random noise and project onto the simplex
            a0 += rng.normal(scale=target_sum * 0.001, size=n)
            a0 = np.maximum(a0, 0.0)
            a0 = _project_to_simplex(a0, target_sum)

            # occasionally reverse the order (large values at the right)
            if rng.random() < 0.5:
                a0 = a0[::-1]

            # ---------- allocated time for this restart ----------
            sub_deadline = min(deadline, time.time() + 30.0)   # up to 30 s per restart

            # 1) Adam optimisation on the smooth surrogate
            a_opt, score_opt = _adam_optimize(a0, sub_deadline * 0.6)

            # 2) Sub‑gradient descent on the exact max‑convolution
            if time.time() < deadline:
                a_opt, score_opt = _subgradient_descent(a_opt, deadline)

            # 3) Final permutation hill‑climb (very cheap)
            if time.time() < deadline:
                a_opt, score_opt = _swap_hill_climb(a_opt, deadline, max_swaps=800)

            # keep the global best
            if score_opt < best_score:
                best_score = score_opt
                best_seq = a_opt.copy()
                print(f"** NEW GLOBAL BEST **  n={n}  restart={restart}  shape={shape_type}"
                      f"  score={best_score:.6f}")

                # early exit if we have already beaten the target
                if best_score <= 1.503:
                    deadline = time.time() + 0.1   # finish almost immediately

    # fallback in the incredibly unlikely case that nothing was produced
    if best_seq is None:
        n = 1000
        a = np.ones(n, dtype=np.float64)
        a *= np.sqrt(2.0 * n) / n
        best_seq = a

    # strip trailing zeros (should never happen, but just in case)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    return best_seq.tolist()
