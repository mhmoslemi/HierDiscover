def propose_candidate(seed=42, budget_s=1000):
    import time, numpy as np
    from scipy.optimize import linprog
    from scipy import sparse

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    # choose n maybe 8000
    n = 8000
    S = np.sqrt(2.0 * n)

    # initial shape
    f = generate_shape(n, S, kind='exponential', param=2.0, direction='decreasing')
    f = np.clip(f, 1e-12, None) # ensure nonzero

    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"Initial score {best_score}")

    t = 0.3   # convex combination step
    iter_num = 0
    while time.time() < deadline:
        # ensure sum normalized
        f = project_to_sum(f, S)

        # convolution and max
        conv_f = conv_self_fft(f)
        M = conv_f.max()
        score = M  # because sum = S => objective = M
        if score < best_score:
            best_score = score
            best_seq = f.copy()
            print(f"Iter {iter_num} new best {best_score}")

        # if score already below target break
        if best_score < 1.503:
            break

        # active set: top K indices
        K = 150
        if conv_f.size <= K:
            active_idx = np.arange(conv_f.size)
        else:
            # get indices of largest K entries
            # using argpartition for efficiency
            idx_part = np.argpartition(-conv_f, K)[:K]
            active_idx = np.sort(idx_part)   # optional sorting
        # optionally add some random indices to guarantee coverage
        # we also ensure each variable appears by adding idx = np.arange(n) maybe not needed.

        # Build A_ub sparse matrix
        m = len(active_idx)
        rows = []
        cols = []
        data = []
        for row_idx, k in enumerate(active_idx):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
            coeffs = f[k - i_range]  # vector length = i_end-i_start+1
            rows.extend([row_idx] * len(i_range))

        res = linprog(c, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs')

    # final polish using smooth max gradient descent (light)

    a = best_seq.copy()
    v = np.zeros_like(a)
    lr = 0.02 * (S / n)
    tau = 0.2 * a.max()
    momentum = 0.9
    for _ in range(2000):

        a, v = smooth_max_step(a, S, lr, tau, momentum, v)
        score = evaluate_sequence(a.tolist())
        if score < best_score:

            best_seq = a.copy()
        # schedule
        lr *= 0.999
        tau = max(tau * 0.999, 1e-5)

    print(f"Finished with best score {best_score}")
    return best_seq.tolist()

"""
Propose a step‑function sequence that (approximately) solves

    minimise  2·n·max(conv(a,a)) / (sum(a)^2)

The implementation combines three ideas:

1. **A good analytic start** – a decreasing exponential shape is generated and
   normalised to the standard ℓ₁‑norm (‖a‖₁ = √(2n)).

2. **Frank‑Wolfe‐style alternating optimisation** – for the current vector `f`
   we solve a linear programme that maximises Σ g_j subject to the linear
   constraints   (f * g)_k ≤ M   (where M = max(f * f)).  The solution `g` is
   scaled to the same ℓ₁‑norm and mixed with `f` (convex combination
   `f←(1‑t)f + t g`).  Repeating this step reduces the peak of the
   self‑convolution.

3. **Final polishing** – a short soft‑max gradient descent (the method used
   previously) refines the best vector found.

All computation respects a wall‑clock budget (default 1000 s).  The best
sequence discovered is returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Low‑level utilities (no closures / lambdas)
# ----------------------------------------------------------------------


def fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors a and b via an FFT of power‑of‑2 length."""
    n, m = a.shape[0], b.shape[0]
    L = 1 << ((n + m - 1).bit_length())           # next power of two
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)
    return conv[: n + m - 1]


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (length 2·len(a)‑1)."""
    return fft_convolve(a, a)


def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector so that its ℓ₁‑norm equals *target*."""
    s = v.sum()
    if s > 0:
        return v * (target / s)
    # fallback – uniform distribution (should never happen)
    return np.full_like(v, target / v.shape[0])


def generate_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
    direction: str = "symmetric",
) -> np.ndarray:
    """
    Produce a deterministic shape of length *n* and scale it to *target_sum*.

    ``kind`` – {"linear", "quadratic", "exponential"}.
    ``direction`` – {"symmetric", "decreasing", "increasing"}.
    *param* controls the strength of the edge/gradient (≥0).
    """
    idx = np.arange(n, dtype=np.float64)
    if direction == "symmetric":
        mid = (n - 1) / 2.0
        d = np.abs(idx - mid) / mid                      # ∈[0,1]
        base = d
    elif direction == "decreasing":
        base = 1.0 - idx / (n - 1)                        # ∈[1,0]
    elif direction == "increasing":
        base = idx / (n - 1)                              # ∈[0,1]
    else:
        raise ValueError("unknown direction")

    if kind == "linear":
        raw = 1.0 + param * base
    elif kind == "quadratic":
        raw = 1.0 + param * base ** 2
    elif kind == "exponential":
        raw = np.exp(param * base)
    else:
        raise ValueError("unknown shape kind")

    raw = np.clip(raw, 0.0, None)
    return project_to_sum(raw, target_sum)


def jitter(a: np.ndarray, target_sum: float, sigma: float = 1e-3) -> np.ndarray:
    """Add Gaussian noise (clipped) and renormalise."""
    a = a + np.random.randn(a.shape[0]) * sigma
    a = np.clip(a, 0.0, None)
    return project_to_sum(a, target_sum)


def smooth_max_step(
    a: np.ndarray,
    target_sum: float,
    lr: float,
    tau: float,
    momentum: float,
    v_prev: np.ndarray,
) -> tuple[np.ndarray, np.ndarray]:
    """
    One iteration of soft‑max gradient descent on the peak of the self‑convolution.
    Returns the updated vector and the new momentum term.
    """
    # convolution a*a
    conv = conv_self_fft(a)                     # length 2n‑1

    # soft‑max weights (log‑sum‑exp trick)
    max_c = conv.max()
    exp_arg = (conv - max_c) / tau
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:                           # degenerate case – fall back to hard max
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    # gradient of the smoothed max:
    #   g_i = 2 * Σ_k w_k * a_{k-i}
    a_rev = a[::-1]
    grad_full = fft_convolve(w, a_rev)         # length 3n‑2
    grad = 2.0 * grad_full[a.shape[0] - 1 : a.shape[0] - 1 + a.shape[0]]

    # projected direction (preserve the ℓ₁‑norm)
    direction = -(grad - np.mean(grad))

    # momentum + step
    v = momentum * v_prev + direction
    a_new = a + lr * v

    # feasibility
    a_new = np.clip(a_new, 0.0, None)
    a_new = project_to_sum(a_new, target_sum)

    return a_new, v


# ----------------------------------------------------------------------
# Core optimiser – alternating Frank‑Wolfe with LP
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The function respects the given time budget and returns the best sequence.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Choose a length and an initial vector
    # ------------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        f = np.array(base, dtype=np.float64)
    else:
        # a length that balances quality and speed
        n = np.random.randint(7500, 9000)
        target_sum = np.sqrt(2.0 * n)
        f = generate_shape(
            n,
            target_sum,
            kind="exponential",
            param=2.5,
            direction="decreasing",
        )

    # Normalise to the standard ℓ₁‑norm (the objective becomes max(conv))
    target_sum = np.sqrt(2.0 * n)
    f = project_to_sum(f, target_sum)

    best_seq = f.copy()
    best_score = float("inf")
    # an early evaluation (helps to see the starting point)
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time():.2f}] start n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Alternating Frank‑Wolfe (LP) phase
    # ------------------------------------------------------------------
    t = 0.30                       # convex‑combination factor
    iteration = 0
    max_active = 150               # number of convolution indices used as constraints

    while time.time() < deadline:
        # normalise each loop (numerical drift)
        f = project_to_sum(f, target_sum)

        # current peak (since sum = target_sum, objective == max convolution)
        conv_f = conv_self_fft(f)
        M = conv_f.max()
        score = M
        if score < best_score:
            best_score = score
            best_seq = f.copy()
            print(f"Iter {iteration:05d} – new best {best_score:.6f}")
            # early exit if we pass the target bound
            if best_score < 1.503:
                break

        # ------------------------------------------------------------------
        # Build a modest active set of convolution indices (largest entries)
        # ------------------------------------------------------------------
        if conv_f.size <= max_active:
            active_idx = np.arange(conv_f.size, dtype=np.int64)
        else:
            # indices of the top‑max_active entries
            part = np.argpartition(-conv_f, max_active)[:max_active]
            active_idx = np.sort(part)                     # optional ordering

        # ------------------------------------------------------------------
        # Assemble the constraint matrix A·g ≤ M   (A_ub = A, b_ub = M)
        # ------------------------------------------------------------------
        rows = []
        cols = []
        data = []
        for row, k in enumerate(active_idx):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
            coeffs = f[k - i_range]                       # f_{k-i}
            rows.extend([row] * len(i_range))
            cols.extend(i_range.tolist())
            data.extend(coeffs.tolist())
        A = sp.csr_matrix((data, (rows, cols)), shape=(len(active_idx), n))
        b_vec = np.full(len(active_idx), M, dtype=np.float64)

        # Upper bound on each variable (prevents unbounded LP)
        f_max = f.max()
        ub = M / f_max if f_max > 0 else None
        bounds = (0, ub) if ub is not None else (0, None)

        # ------------------------------------------------------------------
        # Solve the linear programme: maximise Σ g_j   ⇔ minimise –Σ g_j
        # ------------------------------------------------------------------
        c = -np.ones(n, dtype=np.float64)   # minimise -sum
        res = linprog(c, A_ub=A, b_ub=b_vec, bounds=bounds, method="highs")
        if not res.success:
            # LP failed – fall back to a tiny random perturbation and continue
            f = jitter(f, target_sum, sigma=5e-4)
            iteration += 1
            continue

        g = res.x
        # Scale the LP solution to the required ℓ₁‑norm
        g = project_to_sum(g, target_sum)

        # Convex combination step
        f = (1.0 - t) * f + t * g

        # Occasional jitter helps to escape plateaus
        if np.random.rand() < 0.04:
            f = jitter(f, target_sum, sigma=4e-4)

        iteration += 1

    # ------------------------------------------------------------------
    # Short polishing phase – smooth‑max gradient descent (a few hundred iters)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    v = np.zeros_like(a)
    lr = 0.02 * (target_sum / n)    # modest learning rate
    tau = 0.2 * a.max()             # initial temperature
    momentum = 0.9

    while time.time() < deadline - 0.2:
        a, v = smooth_max_step(a, target_sum, lr, tau, momentum, v)
        sc = evaluate_sequence(a.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"Polish iteration – improvement to {best_score:.6f}")

        # simple cooling schedule
        lr = max(lr * 0.995, 1e-7)
        tau = max(tau * 0.995, 1e-5)

    print(f"[{time.time():.2f}] finished – best score {best_score:.6f}")
    return best_seq.tolist()
