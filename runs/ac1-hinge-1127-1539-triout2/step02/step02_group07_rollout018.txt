```
def improve_f(f, max_iters=100, tol=1e-6):
    n = len(f)
    sum_f = f.sum()
    # ensure sum_f>0
    for it in range(max_iters):
        conv_f_f = np.convolve(f, f)  # use np.convolve directly? Use FFT for speed.
        M = conv_f_f.max()
        # active set
        eps = 1e-8
        active_idx = np.where(conv_f_f >= M - eps)[0]
        # Build constraints
        rows = []
        for k in active_idx:
            i_min = max(0, k-(n-1))
            i_max = min(k, n-1)
            # row vector with f[i] at cols i_min..i_max
            row = np.zeros(n)
            row[i_min:i_max+1] = f[i_min:i_max+1]
            rows.append(row)
        if not rows:
            break
        A_ub = np.vstack(rows)
        b_ub = np.full(A_ub.shape[0], M)
        # Solve LP: maximize sum(g) => minimize -sum(g)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            # maybe fallback to previous best
            break
        g = res.x
        sum_g = g.sum()
        if sum_g == 0:
            break
        # scale g to have same sum as f
        scale = sum_f / sum_g
        g = g * scale
        # mixing parameter t adaptively based on improvement estimate
        t = 0.5
        f_new = (1 - t) * f + t * g
        # Evaluate new M
        conv_f_new = np.convolve(f_new, f_new)
        M_new = conv_f_new.max()
        if M_new < M - tol:
            f = f_new
            # maybe continue
        else:
            # reduce t or break
            t = 0.25
            f_new = (1 - t) * f + t * g
            conv_f_new = np.convolve(f_new, f_new)
            M_new = conv_f_new.max()
            if M_new < M - tol:
                f = f_new
            else:
                # if no improvement, break
                break
        # optional break if improvement saturates
    return f
```

```
def _smooth_obj_and_grad(z, alpha):
    p = np.exp(z)
    sum_p = p.sum()
    if sum_p == 0: p = np.ones_like(p) / p.size; sum_p = 1.0
    p = p / sum_p
    n = p.shape[0]
    # compute conv
    conv = _full_conv_fft(p)
    # smooth max using log-sum-exp
    # compute w = softmax(alpha*conv)
    a = alpha * conv

    dot = np.dot(p, grad_p)
```

```
def _grad_from_weights(p, w):
    n = p.shape[0]
    # w length should be 2n-1
    w_rev = w[::-1]
    L = 1 << ((n + len(w) - 2).bit_length())
    ft_p = np.fft.rfft(p, n=L)
    ft_w = np.fft.rfft(w_rev, n=L)
    conv_full = np.fft.irfft(ft_p * ft_w, n=L)[: n + len(w) - 1]
    idx_start = 2 * n - 2
    grad = 2.0 * conv_full[idx_start - np.arange(n)]
    return grad
```

The evaluation function  

\[
f(a)=\frac{2\,n\displaystyle\max_k (a\star a)_k}{\bigl(\sum_i a_i\bigr)^2}
\]

is *scale‑invariant*.  Hence we may restrict all candidates to the simplex  

\[
p_i\ge 0,\qquad\sum_i p_i = 1 .
\]

For a simplex vector the score simplifies to  

\[
\text{score}=2\,n\; \max_k (p\star p)_k .
\]

The objective is the **∞‑norm** of the self‑convolution of \(p\).  
Directly minimising the maximum is difficult because the function is
non‑smooth.  We therefore minimise a **smooth approximation** of the
maximum – the log‑sum‑exp (soft‑max) – and use its gradient to drive a
large‑scale *L‑BFGS* optimisation.  After each optimisation stage we
increase the smoothing parameter, i.e. we “sharpen’’ the approximation,
which is a classic continuation strategy for min‑max problems.

All operations are performed with FFT‑based convolutions, giving an
\(O(n\log n)\) cost per iteration (with \(n\le 4000\)).  The routine tries
a handful of different lengths (`n = 2000, 3000, 4000`) and a few random
initialisations; the best sequence found before the 1000 s wall‑time is
returned.

The code below follows the description above and satisfies the
competition constraints.

```python
"""
Non‑negative step‑function optimiser (smooth max + L‑BFGS).

We minimise the log‑sum‑exp of the auto‑convolution, a smooth surrogate
for the ∞‑norm, under the simplex constraint.  The variables are
parameterised as z_i with p_i = exp(z_i) / Σ exp(z), guaranteeing
non‑negativity and Σ p_i = 1.  The gradient of the surrogate is obtained
by a single FFT‑based convolution, so each L‑BFGS iteration costs
O(n log n).  A continuation scheme gradually raises the smoothing
parameter (α) to focus on the true maximum.  Several sequence lengths
and random seeds are tried; the best sequence found before the 1000 s
budget is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# FFT helper (linear convolution)
# ----------------------------------------------------------------------


def _linear_convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using FFT.
    Returns a real array of length len(a)+len(b)-1.
    """
    n = a.shape[0]
    m = b.shape[0]
    L = 1 << ((n + m - 2).bit_length())   # next power of two
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(b, n=L)
    c = np.fft.irfft(A * B, n=L)[: n + m - 1]
    return c


def _full_conv_fft(p: np.ndarray) -> np.ndarray:
    """Full linear convolution p * p using FFT (real output)."""
    return _linear_convolve_fft(p, p)


# ----------------------------------------------------------------------
# Gradient of the soft‑max surrogate
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute soft‑max weights w_k = exp(alpha * conv_k) / Σ exp(alpha * conv).
    Numerical stability is achieved by subtracting the maximum.
    """
    a = alpha * conv
    a_max = a.max()
    exp_a = np.exp(a - a_max)
    w = exp_a / exp_a.sum()
    return w


def _grad_wrt_p(p: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k (p*p)_k w.r.t. p.
    Uses the identity  ∂/∂p_i Σ_k w_k (p*p)_k = 2 * Σ_j w_{i+j} p_j .
    The sum Σ_j w_{i+j} p_j is a cross‑correlation and is
    computed with a single FFT‑based convolution.
    """
    n = p.shape[0]               # length of p
    # w has length 2n‑1, we need its reversal for the convolution
    w_rev = w[::-1]

    # linear convolution p ⋆ w_rev  (length = n + (2n‑1) – 1 = 3n‑2)
    full = _linear_convolve_fft(p, w_rev)

    # index of the first needed element: 2n‑2
    start = 2 * n - 2
    # grad_i = 2 * full[start - i] for i = 0 … n‑1
    grad = 2.0 * full[start - np.arange(n)]
    return grad


def _obj_and_grad(z: np.ndarray, alpha: float):
    """
    Returns (objective, gradient) for the smooth surrogate.
    Parameters
    ----------
    z : np.ndarray
        Unconstrained variables (real).  p = softmax(z) lives on the simplex.
    alpha : float
        Smoothing factor for the log‑sum‑exp.
    """
    # simplex vector p
    e_z = np.exp(z)
    sum_ez = e_z.sum()
    if sum_ez == 0.0 or not np.isfinite(sum_ez):
        # fallback to uniform if something went wrong
        p = np.full_like(z, 1.0 / z.size)
    else:
        p = e_z / sum_ez

    # convolution p * p
    conv = _full_conv_fft(p)                # length 2n‑1

    # log‑sum‑exp (smooth max)
    w = _softmax_weights(conv, alpha)
    # objective = (log Σ exp(alpha * conv)) / alpha
    # we already have w = exp(alpha*conv)/Σexp(...), so
    # log Σ exp(alpha*conv) = log(sum) + a_max
    a = alpha * conv
    a_max = a.max()
    log_sum = a_max + np.log(np.exp(a - a_max).sum())
    obj = log_sum / alpha

    # gradient w.r.t. p
    grad_p = _grad_wrt_p(p, w)

    # chain rule for z (softmax)
    dot = np.dot(p, grad_p)
    grad_z = p * (grad_p - dot)

    return obj, grad_z


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: float = 1000.0):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The routine respects a 1000 s wall‑time and
    returns the best vector found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # try a few different lengths; longer vectors give more freedom,
    # but the optimisation cost grows only like O(n log n)
    # ------------------------------------------------------------------
    lengths = (2000, 3000, 4000)
    for n in lengths:
        if time.time() >= deadline:
            break

        # ----------------------------------------------------------------
        # several random starts for each length (including the uniform start)
        # ----------------------------------------------------------------
        for restart in range(4):
            if time.time() >= deadline:
                break

            # initialise z (log‑weights).  Uniform start -> z = 0.
            if restart == 0:
                z0 = np.zeros(n, dtype=np.float64)
            else:

                rand = np.random.rand(n) + 1e-3
                z0 = np.log(rand)

            # continuation on the smoothing parameter α
            alphas = [5.0, 10.0, 20.0, 40.0, 80.0, 160.0]
            z = z0.copy()
            for alpha in alphas:
                if time.time() >= deadline:

                    obj, grad = _obj_and_grad(z_vec, alpha)

                    print("L‑BFGS stage failed (α={})".format(alpha))

                # keep the parameters in a numerically safe range

            e_z = np.exp(z)
            p = e_z / e_z.sum()
            # prune tiny entries (they only waste time in the evaluator)
            p[p < 1e-12] = 0.0
            p /= p.sum()                     # renormalise after pruning
```
