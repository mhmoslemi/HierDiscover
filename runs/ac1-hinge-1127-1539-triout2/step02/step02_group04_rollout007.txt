def redistribute_mass(x, g, epsilon_factor=0.1):
    # epsilon = epsilon_factor * min(x[i_max], 1e-3?) maybe small.
    i_max = np.argmax(g)
    i_min = np.argmin(g)
    if x[i_max] <= 1e-12:
        return x
    delta = epsilon_factor * x[i_max]  # fraction to move
    if delta < 1e-12:
        return x
    x_new = x.copy()
    x_new[i_max] -= delta
    x_new[i_min] += delta
    # ensure non-negative; x_new[i_max] may become negative if delta > x[i_max], but ensure.
    x_new = np.maximum(x_new, 0)
    # renormalize
    total = x_new.sum()
    if total <= 0:
        return x
    x_new /= total
    return x_new

"""
Propose a step‑function (non‑negative list of heights) that minimises
evaluate_sequence().  The method consists of three stages:

1.  **Broad parametric sweep** – many analytic families are sampled,
    including power‑laws, exponentials, mixed power/exp, and *edge‑peaked*
    shapes (mass concentrated near the two ends).  Every candidate is
    normalised to ℓ¹‑norm = 1 and evaluated; the best one is kept.

2.  **Random Dirichlet + hill‑climbing** – a large number of random
    vectors drawn from Dirichlet distributions (with optional edge‑bias)
    are examined.  The best of these vectors is taken as the starting
    point for a projected sub‑gradient optimiser.

3.  **Projected sub‑gradient descent with momentum and targeted mass
    redistribution** – starting from the best candidate we repeatedly
    minimise the quantity n·max(a∗a) (the scale‑free part of the
    objective).  The sub‑gradient of the ℓ∞‑norm of the auto‑convolution
    is used, together with Nesterov‑type momentum and Euclidean projection
    onto the probability simplex.  Every few hundred iterations a tiny
    deterministic mass‑transfer step is applied: mass is moved from the
    entry with largest sub‑gradient (most responsible for the active
    constraints) to the entry with the smallest sub‑gradient, which
    typically pushes probability mass towards the ends of the support.
    Random jitter and occasional restarts help escape plateaus.

The routine respects a 1000 s wall‑clock budget and returns the best
sequence found before the deadline.  All numbers are non‑negative and
the sequence is normalised internally; the evaluator itself clips to the
range [0,1000] and guards against pathological inputs.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level)
# ----------------------------------------------------------------------


def _evaluate(seq: np.ndarray) -> float:
    """Thin wrapper for the supplied evaluate_sequence (expects a list)."""
    try:
        return evaluate_sequence(seq.tolist())
    except Exception:            # safety net
        return float('inf')


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("simplex_projection works on 1‑D arrays")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient(x: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of g(x)=max_k (x*x)_k w.r.t. x (on the simplex).
    All active convolution indices are used.
    """
    n = x.shape[0]
    conv = np.convolve(x, x)               # length 2n‑1
    m = conv.max()
    active = np.where(conv >= m - 1e-14)[0]
    g = np.zeros_like(x)
    for k in active:
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        i = np.arange(i_min, i_max + 1, dtype=int)
        j = k - i
        g[i] += x[j]                        # factor 2 absorbed in step‑size
    return g

# Families of analytic shapes
# ----------------------------------------------------------------------


def _generate_powerlaw(p: float, sigma: float,
                       max_len: int = 5000, eps: float = 1e-12) -> np.ndarray:
    """a_i = (i+σ)^{-p}, truncated when a_i < eps."""
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p)
    return a[a > eps]


def _generate_exponential(lam: float,
                          max_len: int = 5000, eps: float = 1e-12) -> np.ndarray:
    """a_i = exp(-λ i), truncated when a_i < eps."""
    i = np.arange(max_len, dtype=np.float64)
    a = np.exp(-lam * i)
    return a[a > eps]


def _generate_mixed(p: float, sigma: float, lam: float,
                    max_len: int = 5000, eps: float = 1e-12) -> np.ndarray:
    """a_i = (i+σ)^{-p}·exp(-λ i), truncated when a_i < eps."""
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p) * np.exp(-lam * i)
    return a[a > eps]


def _generate_edge_powerlaw(p: float, sigma: float,
                           max_len: int = 5000, eps: float = 1e-12) -> np.ndarray:
    """
    Edge‑peaked power law:
    a_i = (i+σ)^{-p} + (max_len‑1‑i+σ)^{-p},
    truncated when a_i < eps.
    """
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p) + (max_len - 1 - i + sigma) ** (-p)
    return a[a > eps]


def _generate_edge_exponential(lam: float,
                               max_len: int = 5000, eps: float = 1e-12) -> np.ndarray:
    """
    Edge‑peaked exponential:
    a_i = exp(-λ i) + exp(-λ (max_len‑1‑i)),
    truncated when a_i < eps.
    """
    i = np.arange(max_len, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (max_len - 1 - i))
    return a[a > eps]


def _generate_edge_mixed(p: float, sigma: float, lam: float,
                        max_len: int = 5000, eps: float = 1e-12) -> np.ndarray:
    """
    Edge‑peaked mixed power‑law / exponential:
    a_i = (i+σ)^{-p}·exp(-λ i) + (max_len‑1‑i+σ)^{-p}·exp(-λ (max_len‑1‑i)).
    """
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p) * np.exp(-lam * i) \
        + (max_len - 1 - i + sigma) ** (-p) * np.exp(-lam * (max_len - 1 - i))
    return a[a > eps]


def _generate_beta_like(alpha: float, beta: float,
                        max_len: int = 5000, eps: float = 1e-12) -> np.ndarray:
    """
    Beta‑like shape on a discrete grid:
    a_i = (i+1)^{α}·(max_len‑i)^{β}.
    For α,β<0 the mass concentrates near the ends.
    """
    i = np.arange(1, max_len + 1, dtype=np.float64)
    a = (i ** alpha) * ((max_len - i + 1) ** beta)
    return a[a > eps]

# ----------------------------------------------------------------------

            extra = np.full(extra_len, 1e-12, dtype=np.float64)
            x = np.concatenate([x, extra])
            x = x / x.sum()
            momentum = np.concatenate([momentum, np.zeros_like(extra)])
            print(f'iteration {iteration:6d} – grew length to {len(x)}')

        # ----- evaluate the current candidate
        cur_score = _evaluate(x)
        if np.isfinite(cur_score) and cur_score < best_score:

            last_improve = iteration
            print(f'iteration {iteration:6d} – restart performed.')

    print('final best score:', best_score, 'length:', len(best_seq))
    return best_seq
