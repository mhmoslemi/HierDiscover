def compute_softmax_weights(conv, gamma):
    max_val = conv.max()
    w = np.exp(gamma * (conv - max_val))
    w_sum = w.sum()
    if w_sum == 0: w = np.full_like(w, 1.0 / w.size)
    else: w /= w_sum
    return w

def compute_gradient_from_weights(a, w):
    n = a.shape[0]
    nfft_corr = _next_pow2(3*n - 2)
    wf = np.fft.rfft(w, n=nfft_corr)
    af_rev = np.fft.rfft(a[::-1], n=nfft_corr)
    corr_full = np.fft.irfft(wf * af_rev, n=nfft_corr)
    grad = 2.0 * corr_full[n-1:n-1+n]
    return grad

def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0

    # Determine n
    try:
        init = np.array(height_sequence_1, dtype=np.float64)
        n = init.shape[0]
    except Exception:
        n = 6000

    target_sum = np.sqrt(2.0 * n)

    # Helper: function to normalize
    def normalize(v):
        s = v.sum()
        if s == 0:
            v = np.full_like(v, target_sum / v.size)
        else:
            v = v * (target_sum / s)
        return v

    # Random power-law + uniform + triangular initialization candidates
    best_a = None
    best_score = np.inf

    # include init if provided
    if 'init' in locals():
        a0 = normalize(np.maximum(init, 0.0))
        score = _compute_score(a0)
        best_a, best_score = a0, score

    # generate random candidates
    for _ in range(80):
        if time.time() >= deadline:
            break
        # random shape: generate random exponents and other forms
        mode = np.random.choice(['power', 'exp', 'tri'])
        if mode == 'power':
            p = np.random.uniform(0.3, 1.0)
            offset = np.random.uniform(0.1, 5.0)
            a = 1.0 / np.power(np.arange(n, dtype=np.float64) + offset, p)
        elif mode == 'exp':
            lam = np.random.uniform(0.0005, 0.01)
            a = np.exp(-lam * np.arange(n))
        else:  # triangular
            peak = np.random.randint(0, n)
            a = np.minimum(np.arange(n), np.arange(n)[::-1])
            a = np.abs(np.arange(n) - peak)  # not correct; use linear ramp
            # simpler: make symmetric decreasing from ends:
            a = np.minimum(np.arange(n) + 1, n - np.arange(n))
        # enforce non-negative
        a = np.maximum(a, 0.0)
        a = normalize(a)
        s = _compute_score(a)
        if s < best_score:
            best_score = s
            best_a = a.copy()

    if best_a is None:
        best_a = np.full(n, target_sum / n)
        best_score = _compute_score(best_a)

    a = best_a.copy()
    # Precompute FFT lengths
    nfft_conv = _next_pow2(2 * n - 1)
    nfft_corr = _next_pow2(3 * n - 2)

    # Parameters for Adam phase
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    lr = 0.15
    lr_min = 5e-4
    lr_decay = 0.9996

    gamma = 0.8
    gamma_max = 600.0
    gamma_inc = 1.005

    iteration = 0

    # First phase: Adam with smooth max
    while time.time() < deadline:
        iteration += 1
        t += 1

        # compute convolution and softmax weights
        af = np.fft.rfft(a, n=nfft_conv)
        conv = np.fft.irfft(af * af, n=nfft_conv)[:2*n-1]
        max_conv = conv.max()

        # softmax with current gamma
        w = np.exp(gamma * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # gradient
        wf = np.fft.rfft(w, n=nfft_corr)
        af_rev = np.fft.rfft(a[::-1], n=nfft_corr)
        corr = np.fft.irfft(wf * af_rev, n=nfft_corr)
        grad = 2.0 * corr[n-1:n-1+n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_onto_simplex(a_new, target_sum)

        # occasional jitter
        if iteration % 111 == 0:
            jitter = 1e-4 * np.random.randn(n)
            a_new = project_onto_simplex(a_new + jitter, target_sum)

        # evaluate
        score_new = _compute_score(a_new)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            elapsed = time.time() - start
            print(f"Iter {iteration:6d}: new best {best_score:.6f} (t={elapsed:.1f}s)")

        a = a_new

        # schedule
        gamma = min(gamma * gamma_inc, gamma_max)
        lr = max(lr * lr_decay, lr_min)

        # early termination condition for Adam phase: after certain number of iters or time left < some threshold
        # If less than 30% time left, break
        if time.time() > deadline * 0.7:
            break

    # Second phase: exact max subgradient with exponentiated gradient steps
    # We'll continue with direct subgradient (mask) for a while.
    # Use fixed step size eta decreasing
    eta = 0.1
    eta_decay = 0.9998

    while time.time() < deadline:
        # compute convolution
        af = np.fft.rfft(a, n=nfft_conv)
        conv = np.fft.irfft(af * af, n=nfft_conv)[:2*n-1]
        max_conv = conv.max()

        # mask for max entries
        w = (conv == max_conv).astype(np.float64)  # binary mask

        # gradient
        wf = np.fft.rfft(w, n=nfft_corr)
        af_rev = np.fft.rfft(a[::-1], n=nfft_corr)
        corr = np.fft.irfft(wf * af_rev, n=nfft_corr)
        grad = 2.0 * corr[n-1:n-1+n]

        # exponentiated gradient step
        a = a * np.exp(-eta * grad)
        a = project_onto_simplex(a, target_sum)

        # evaluate
        score_new = _compute_score(a)
        if score_new < best_score:
            best_score = score_new
            best_a = a.copy()
            elapsed = time.time() - start
            print(f"Subgrad iter {iteration:6d}: new best {best_score:.6f} (t={elapsed:.1f}s)")

        iteration += 1
        eta = max(eta * eta_decay, 1e-6)

        if time.time() >= deadline:
            break

    # Return best sequence found
    return best_a.tolist()

"""
Non‑negative step function optimizer.

The routine searches for a sequence a = (a0,…,a_{n‑1}) of non‑negative
numbers whose self‑convolution has a small maximum value.
The objective returned by ``evaluate_sequence`` equals

    max(conv(a, a))

when the L1 norm is fixed to √(2 n); therefore the algorithm works on the
simplex {a ≥ 0, Σa = √(2 n)}.

Algorithm (single run, ≈ 1000 s budget)

1.  **Initialisation** – reuse the global construction ``height_sequence_1`
    if available, otherwise generate a few random shapes (power‑law,
    exponential, symmetric triangular).  Keep the best one.

2.  **Adam on a smooth‑max surrogate** – replace the hard maximum of the
    convolution by a soft‑max with temperature γ.  The gradient of the
    surrogate is obtained by a correlation of the soft‑max weights with
    the reversed sequence (FFT‑based, O(n log n)).  An Adam optimiser with
    projection onto the simplex updates the vector while γ is slowly
    increased (more faithful to the true max).

3.  **Exact‑max subgradient phase** – after the smooth phase, switch to
    the true subgradient of the maximum (weights are a 0/1 mask of the
    maximising convolution entries).  A multiplicative “exponentiated‑gradient”
    step (`a ← a·exp(−η ∇)`) maintains non‑negativity and is followed by a
    projection onto the simplex.  η decays during the phase.

4.  **Best‑ever tracking** – the best sequence found (lowest evaluated
    value) is stored and finally returned.

All heavy work uses NumPy FFTs; the memory footprint stays modest
even for several‑thousand‑element vectors.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Helper utilities (top level)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _compute_score(a: np.ndarray) -> float:
    """
    Fast version of the provided ``evaluate_sequence``.
    Assumes ``a`` is non‑negative and already normalised.
    """
    n = a.shape[0]
    nfft = _next_pow2(2 * n - 1)
    af = rfft(a, n=nfft)
    conv = irfft(af * af, n=nfft)[: 2 * n - 1]
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    # The wrapper normalises the sum to √(2 n); we keep the same target
    # sum throughout, so the returned value equals the true objective.
    return float(2.0 * n * max_c / (s * s))


def project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of *v* onto the simplex
        {x : x_i ≥ 0 , Σ_i x_i = target_sum}.
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative sequence that minimises the evaluation
    function.  The routine respects the 1000 s time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin so that we always return before timeout
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Determine problem size (n) and target L1 sum
    # ------------------------------------------------------------------
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        n = init_seq.shape[0]
    except Exception:
        n = 6000                         # fallback length

    target_sum = np.sqrt(2.0 * n)

    def normalise(v: np.ndarray) -> np.ndarray:
        """Scale *v* so that its L1 norm equals ``target_sum``."""
        s = v.sum()
        if s <= 0:
            return np.full_like(v, target_sum / v.size)
        return v * (target_sum / s)

    # ------------------------------------------------------------------
    # 2. Random initialisation + keep the best candidate
    # ------------------------------------------------------------------
    best_a = None
    best_score = np.inf

    # reuse the supplied construction if it exists
    if 'init_seq' in locals():
        a0 = normalise(np.maximum(init_seq, 0.0))
        sc0 = _compute_score(a0)
        best_a, best_score = a0, sc0

    # a handful of random shapes (power‑law, exponential, triangular)
    for _ in range(80):
        if time.time() >= deadline:
            break
        kind = np.random.choice(["power", "exp", "tri"])
        if kind == "power":
            p = np.random.uniform(0.3, 1.0)
            offset = np.random.uniform(0.1, 5.0)
            a = 1.0 / np.power(np.arange(n, dtype=np.float64) + offset, p)
        elif kind == "exp":
            lam = np.random.uniform(5e-4, 1e-2)
            a = np.exp(-lam * np.arange(n, dtype=np.float64))
        else:  # symmetric triangular (peak in the middle)
            a = np.minimum(np.arange(n, dtype=np.float64),
                           (n - 1) - np.arange(n, dtype=np.float64))
        a = normalise(np.maximum(a, 0.0))
        sc = _compute_score(a)
        if sc < best_score:
            best_score, best_a = sc, a.copy()

    if best_a is None:
        # should never happen, but guard against it
        best_a = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _compute_score(best_a)

    a = best_a.copy()                     # current iterate

    # ------------------------------------------------------------------
    # 3. Pre‑compute FFT lengths (used repeatedly)
    # ------------------------------------------------------------------
    nfft_conv = _next_pow2(2 * n - 1)       # for a*a
    nfft_corr = _next_pow2(3 * n - 2)       # for w ⊛ a_rev

    # ------------------------------------------------------------------
    # 4. Adam optimiser on a smooth‑max surrogate
    # ------------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    lr = 0.15
    lr_min = 5e-4
    lr_decay = 0.9996

    gamma = 0.8
    gamma_max = 600.0
    gamma_inc = 1.005

    iteration = 0

    # Allocate roughly 70 % of the budget to this phase
    adam_deadline = start_time + 0.70 * budget_s

    while time.time() < min(deadline, adam_deadline):
        iteration += 1
        t += 1

        # ----- a * a (full convolution) -----
        af = rfft(a, n=nfft_conv)
        conv = irfft(af * af, n=nfft_conv)[: 2 * n - 1]
        max_conv = conv.max()

        # ----- soft‑max weights (smooth approximation of max) -----
        w = np.exp(gamma * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # ----- gradient of the smooth surrogate -----
        wf = rfft(w, n=nfft_corr)
        af_rev = rfft(a[::-1], n=nfft_corr)
        corr_full = irfft(wf * af_rev, n=nfft_corr)
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]

        # ----- Adam update with projection onto the simplex -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_onto_simplex(a_new, target_sum)

        # occasional jitter to escape plateaus
        if iteration % 111 == 0:
            jitter = 1e-4 * np.random.randn(n)
            a_new = project_onto_simplex(a_new + jitter, target_sum)

        # ----- evaluate & keep best -----
        score_new = _compute_score(a_new)
        if score_new < best_score:
            best_score, best_a = score_new, a_new.copy()
            elapsed = time.time() - start_time
            print(f"Adam iter {iteration:6d}: new best {best_score:.6f} (t={elapsed:.1f}s)")

        a = a_new

        # ----- schedule updates -----
        gamma = min(gamma * gamma_inc, gamma_max)
        lr = max(lr * lr_decay, lr_min)

    # ------------------------------------------------------------------
    # 5. Exact‑max subgradient phase (exponentiated‑gradient updates)
    # ------------------------------------------------------------------
    eta = 0.10
    eta_decay = 0.9998

    while time.time() < deadline:
        # ----- a * a (full convolution) -----
        af = rfft(a, n=nfft_conv)
        conv = irfft(af * af, n=nfft_conv)[: 2 * n - 1]
        max_conv = conv.max()

        # ----- binary mask of maximising entries -----
        # use a tiny tolerance to catch numerical ties
        mask = (conv >= max_conv - 1e-12).astype(np.float64)

        # ----- exact subgradient (no smoothing) -----
        wf = rfft(mask, n=nfft_corr)
        af_rev = rfft(a[::-1], n=nfft_corr)
        corr_full = irfft(wf * af_rev, n=nfft_corr)
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]

        # ----- exponentiated‑gradient multiplicative update -----
        a = a * np.exp(-eta * grad)
        a = project_onto_simplex(a, target_sum)

        # ----- evaluate & keep best -----
        score_new = _compute_score(a)
        if score_new < best_score:
            best_score, best_a = score_new, a.copy()
            elapsed = time.time() - start_time
            print(f"Subgrad iter {iteration:6d}: new best {best_score:.6f} (t={elapsed:.1f}s)")

        iteration += 1
        eta = max(eta * eta_decay, 1e-6)

        # safety break if we are within a couple of seconds of the deadline
        if time.time() > deadline - 2.0:
            break

    # ------------------------------------------------------------------
    # 6. Return best sequence discovered
    # ------------------------------------------------------------------
    return best_a.tolist()
