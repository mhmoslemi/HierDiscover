"""
Propose a non‑negative step‑function (height vector) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

where `conv` is the linear self‑convolution.
The algorithm consists of three stages:

1. **Broad parametric sweep** – generate many simple symmetric “U‑shapes’’
   (linear, quadratic, exponential) and a few random Dirichlet vectors.
   The best of them is kept as a starting point.

2. **Smoothed projected gradient descent with momentum**.
   The non‑smooth `max` is approximated by a soft‑max (log‑sum‑exp) with a
   temperature τ = τ_factor·max_conv.  The gradient of the smoothed
   objective is obtained by a second FFT‑based convolution.  After each
   step the vector is projected back onto the ℓ₁‑simplex (non‑negative,
   prescribed ℓ₁‑norm) by a cheap scaling operation.  A momentum term
   accelerates convergence, while η and τ_factor are slowly decayed.
   Random pair‑mass transfers and occasional jitter help the search
   escape plateaus.

3. **Polishing** – a short pass of tiny random mass transfers, keeping
   any improvement that lowers the true evaluation function.

The whole routine respects a user‑supplied time budget (≈1000 s) and
returns the best sequence observed.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a with itself using a power‑of‑two FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    Fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Fa * Fa, n=L)
    return conv[: 2 * n - 1]


def generate_U_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
) -> np.ndarray:
    """
    Produce a symmetric “U‑shaped’’ sequence of length n.
    kind ∈ {linear, quadratic, exponential}.
    param ≥ 0 controls the edge enhancement.
    The result is scaled to have ℓ₁‑norm `target_sum`.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    else:
        raise ValueError("unknown shape kind")

    raw = np.maximum(raw, 0.0)
    s = raw.sum()
    if s == 0:
        return np.full(n, target_sum / n, dtype=np.float64)
    return raw * (target_sum / s)


def simple_simplex_projection(x: np.ndarray, target: float) -> np.ndarray:
    """
    Fast projection onto the simplex {x_i ≥ 0, Σ x_i = target}
    by scaling the non‑negative vector so that its sum equals `target`.
    """
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s > 0:
        return x * (target / s)
    else:
        # fallback to uniform distribution
        return np.full_like(x, target / x.shape[0])


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a small proportion (scale) of mass from one random index
    to another and re‑project onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return simple_simplex_projection(a, target_sum)

    # Phase 1 – parametric sweep + random seeds
    # ------------------------------------------------------------------
    best_seq = np.full(n, target_sum / n, dtype=np.float64)  # start uniform
    best_score = np.inf

    # a few systematic families
    shape_kinds = ("linear", "quadratic", "exponential")
    param_grid = {
        "linear": np.linspace(0.0, 8.0, 81),        # 0 … 8 step 0.1
        "quadratic": np.linspace(0.0, 6.0, 61),    # 0 … 6
        "exponential": np.linspace(0.0, 4.0, 41), # 0 … 4
    }

    print(f"[{time.time():.2f}] Starting parametric sweep (n={n})")
    for kind in shape_kinds:
        for p in param_grid[kind]:
            cand = generate_U_shape(n, target_sum, kind=kind, param=p)
            cur_max = conv_self_fft(cand).max()
            if cur_max < best_score:
                best_score = cur_max
                best_seq = cand.copy()
                print(f"  New best {kind} (p={p:.2f}) → {best_score:.6f}")
            if time.time() >= deadline:
                print("Time exhausted during sweep.")
                return best_seq.tolist()

    # random Dirichlet‑like seeds
    for _ in range(40):
        rnd = np.random.rand(n)
        cand = simple_simplex_projection(rnd, target_sum)
        cur_max = conv_self_fft(cand).max()
        if cur_max < best_score:
            best_score = cur_max
            best_seq = cand.copy()
            print(f"  Dirichlet seed improved → {best_score:.6f}")
        if time.time() >= deadline:
            break

    # include provided baseline if any
    if base_arr is not None:
        cand = simple_simplex_projection(base_arr, target_sum)
        cur_max = conv_self_fft(cand).max()
        if cur_max < best_score:
            best_score = cur_max
            best_seq = cand.copy()
            print(f"  Baseline improved → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – smoothed gradient descent with momentum
    # ------------------------------------------------------------------
    a = best_seq.copy()
    velocity = np.zeros_like(a)
    beta = 0.9                                 # momentum coefficient
    eta = 0.5 * (target_sum / n)               # initial step size
    tau_factor = 0.2                           # τ = τ_factor * max_conv
    iteration = 0
    stagnation = 0

    print(f"[{time.time():.2f}] Entering gradient descent (init score={best_score:.6f})")

    while time.time() < deadline:
        # 1) self‑convolution and its maximum
        conv = conv_self_fft(a)
        max_conv = conv.max()

        # 2) soft‑max weights (log‑sum‑exp smoothing)
        tau = max(tau_factor * max_conv, 1e-12)
        # subtract max to improve stability (max_conv becomes zero)
        w = np.exp((conv - max_conv) / tau)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.size)

        # 3) gradient of the smoothed max
        a_rev = a[::-1]
        L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
        grad = (2.0 / tau) * conv_full[n - 1 : n - 1 + n]

        # 4) momentum update and projected step
        velocity = beta * velocity + (1.0 - beta) * grad
        a_new = a - eta * velocity
        a_new = simple_simplex_projection(a_new, target_sum)

        a = a_new
        iteration += 1

        # improvement test (objective = max_conv because sum is fixed)
        if max_conv + 1e-12 < best_score:
            best_score = max_conv
            best_seq = a.copy()
            stagnation = 0
            print(f"Iter {iteration:06d} – improved to {best_score:.6f}")
        else:
            stagnation += 1

        # adaptive schedules
        if iteration % 800 == 0:
            eta *= 0.96
            tau_factor = max(0.001, tau_factor * 0.94)

        # occasional random mass transfer
        if np.random.rand() < 0.018:
            a = random_pair_transfer(a, target_sum, scale=0.009)

        # heavy jitter after prolonged stagnation
        if stagnation >= 3500:
            jitter = np.random.randn(n) * (target_sum * 5e-4)
            a = simple_simplex_projection(a + jitter, target_sum)
            velocity[:] = 0.0
            stagnation = 0
            print(f"Iter {iteration:06d} – jitter applied")

    # ------------------------------------------------------------------
    # Phase 3 – final polishing with tiny random transfers
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Descent finished. Best score {best_score:.6f}")

    for _ in range(120):
        if time.time() >= deadline:
            break
        cand = random_pair_transfer(best_seq.copy(), target_sum, scale=0.004)
        cand_score = conv_self_fft(cand).max()
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"Polish improved to {best_score:.6f}")

    # Return as plain Python list
    return best_seq.tolist()
