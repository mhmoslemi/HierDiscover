def _gradient_max(a):
    n = a.size
    size1 = 1 << ((2*n-1).bit_length())
    af = np.fft.rfft(a, n=size1)
    b_full = np.fft.irfft(af * af, n=size1)[:2*n-1]   # full conv
    maxb = b_full.max()
    # create p indicator for max entries
    mask = (b_full == maxb).astype(np.float64)
    # if multiple entries, allocate equal weight? For subgradient set p_i = 1 for each active index (the convex hull includes any convex combination). For gradient, we can choose average: mask /= mask.sum()
    mask_sum = mask.sum()
    if mask_sum > 0:
        p = mask / mask_sum
    else:
        p = mask

    # compute a_rev ⋆ p
    a_rev = a[::-1]
    conv_len = n + (2*n-1) -1   # 3n-2
    size2 = 1 << ((conv_len).bit_length())
    ar_fft = np.fft.rfft(a_rev, n=size2)
    p_fft = np.fft.rfft(p, n=size2)
    rev_conv = np.fft.irfft(ar_fft * p_fft, n=size2)[:conv_len]
    grad = 2.0 * rev_conv[n-1:n-1+n]
    return grad, b_full

"""
Propose a step‑function sequence that minimises
    2·n·max(conv(a,a)) / (sum(a)^2)

The objective is scale‑invariant: after projecting the vector onto the
simplex Σa = √(2·n) the score is simply the maximum entry of the full
linear convolution a*a.  The algorithm consists of three phases:

1.  Randomised initialisation (power‑law shapes with jitter) for several
    vector lengths.

2.  A fast sub‑gradient descent that directly minimises the
    max‑convolution (the gradient is the correlation of a with the
    indicator of the maximal convolution entries).  After each step the
    iterate is projected onto the ℓ¹‑simplex.

3.  A cheap stochastic hill‑climber that moves a random fraction of the
    total mass between two positions while keeping the ℓ¹‑norm fixed.

All phases respect the 1000‑second budget, keep the sequence non‑negative
and use only a single CPU.  The best sequence found is returned as a
plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    { x_i ≥ 0 , Σ x_i = s }.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _score(a: np.ndarray) -> float:
    """Fast version of the competition objective (scale‑invariant)."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------


def _gradient_max(a: np.ndarray):
    """
    Return (g, conv) where
        conv = a*a (full convolution, length 2·n‑1)
        g   = ∇ max(conv)  = 2·(a ⋆ p)   with p_i ∝ 1 on the active set
    The active set consists of all indices attaining the maximal value.
    """
    n = a.shape[0]

    # full convolution
    conv = _full_convolution_fft(a)
    maxb = conv.max()

    # probability vector that is uniform on the active set
    mask = (conv == maxb).astype(np.float64)
    cnt = mask.sum()
    if cnt > 0:
        p = mask / cnt
    else:
        p = mask

    # correlation a_rev ⋆ p  (central part has length n)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << ((conv_len).bit_length())
    ar_fft = np.fft.rfft(a_rev, n=size2)
    p_fft = np.fft.rfft(p, n=size2)
    rev = np.fft.irfft(ar_fft * p_fft, n=size2)[:conv_len]
    g = 2.0 * rev[n - 1: n - 1 + n]
    return g, conv


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         max_iter: int = 5000):
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    The step size decays roughly like 1/√t.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _score(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        g, conv = _gradient_max(a)
        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            # step size: base / (√(it+1)·‖g‖)
            lr = (target_sum * 0.5) / (norm_g * np.sqrt(it + 1))
            a = a - lr * g

        a = _project_to_simplex(a, target_sum)

        # occasional tiny jitter to escape plateaus
        if it % 250 == 0:
            jitter = np.random.default_rng().normal(scale=target_sum * 0.001,
                                                    size=a.size)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur_val = conv.max()
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] sub‑grad it={it}  score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Stochastic hill‑climbing (cheap local search)
# ----------------------------------------------------------------------


def _random_hill_climb(a: np.ndarray, deadline: float):
    """
    Very cheap hill‑climber that preserves the ℓ¹‑norm.
    Randomly transfers a fraction of mass from i to j and keeps
    improving moves; occasional moves that do not improve are rejected.
    """
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    cur_val = best_val

    max_frac = 0.12
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        frac = rng.random() * max_frac
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        new_val = _score(a)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}]  hill‑climb improved = {best_val:.6f}")
        else:
            a[j] -= delta
            a[i] += delta

    return best_a, best_val


# ----------------------------------------------------------------------
# Main optimiser
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine respects the time budget,
    tries several vector lengths and random restarts, runs a fast
    sub‑gradient descent followed by a short hill‑climbing phase,
    and returns the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # leave a small safety margin
    deadline = time.time() + budget_s - 2.0

    best_seq = None
    best_score = np.inf

    # a handful of lengths – larger gives more flexibility, smaller is faster
    lengths = [1200, 1500, 1800, 2000, 2300, 2600, 3000]

    # number of restarts per length (adapted dynamically)
    restart_idx = 0
    while time.time() < deadline:
        for n in lengths:
            if time.time() >= deadline:
                break

            target_sum = np.sqrt(2.0 * n)

            # how many restarts for this length?  we try 3 as long as we have time.
            for restart in range(3):
                if time.time() >= deadline:
                    break

                # ------- initialise a random power‑law shape -----------------
                p = rng.uniform(0.3, 0.9)          # exponent
                a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
                a *= target_sum / a.sum()

                # small Gaussian jitter before optimisation
                a += rng.normal(scale=target_sum * 0.005, size=n)
                a = np.maximum(a, 0.0)
                a = _project_to_simplex(a, target_sum)

                # ------- sub‑gradient descent (≈10 s per restart) ----------
                sub_deadline = min(deadline, time.time() + 10.0)
                a_opt, opt_score = _subgradient_descent(a, target_sum, sub_deadline)

                # ------- cheap hill‑climb refinement -----------------------
                if time.time() < deadline:
                    a_opt, opt_score = _random_hill_climb(a_opt, deadline)

                # keep the globally best
                if opt_score < best_score:
                    best_score = opt_score
                    best_seq = a_opt.copy()
                    print(f"** NEW GLOBAL BEST **  n={n}  p={p:.4f}  restart={restart}"
                          f"  score={best_score:.6f}")

        restart_idx += 1
        # If we have exhausted all lengths but still have time, loop again.
        # This gives the algorithm another chance to improve the current best.

    # ------------------------------------------------------------------
    # fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        a = np.ones(n, dtype=np.float64)
        a *= np.sqrt(2.0 * n) / n
        best_seq = a

    # Trim trailing zeroes for cleanliness.
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    return best_seq.tolist()
