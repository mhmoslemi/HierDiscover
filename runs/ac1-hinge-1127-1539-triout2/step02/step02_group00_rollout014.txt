"""
A lightweight search for a step‑function with a very small
max‑convolution value.

The idea is to restrict the search to simple parametric families
(power‑law and geometric sequences), scan a moderate range of
lengths and parameters with a fast FFT based objective, and finally
refine the best candidate with a cheap random hill‑climber.
All operations keep the ℓ¹‑norm fixed to √(2·n) so that the
objective returned by ``evaluate_sequence`` equals the maximum entry
of the convolution.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast evaluation of the competition objective (identical to evaluate_sequence
# but without the safety checks)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """
    Fast version of the competition objective.
    The ℓ¹‑norm of ``a`` must be the target √(2·n); the routine does not
    enforce it – it is the caller’s responsibility.
    """
    n = a.size
    max_b = _max_convolution_fft(a)
    # The normalising factor √(2·n) is exactly the ℓ¹‑norm of the final
    # candidate, therefore the objective equals ``max_b``.
    return max_b


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _make_powerlaw(p: float, n: int) -> np.ndarray:
    """Generate a decreasing power‑law a_i ∝ (i+1)^{-p} with ℓ¹‑norm √(2·n)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    raw = idx ** (-p)
    target = math.sqrt(2.0 * n)
    raw *= target / raw.sum()
    return raw


def _make_geometric(r: float, n: int) -> np.ndarray:
    """Generate a geometric sequence a_i ∝ r^{i} (0<r<1) with ℓ¹‑norm √(2·n)."""
    idx = np.arange(n, dtype=np.float64)
    raw = r ** idx
    target = math.sqrt(2.0 * n)
    raw *= target / raw.sum()
    return raw


# ----------------------------------------------------------------------
# Very cheap random hill‑climbing (preserves the ℓ¹‑norm)
# ----------------------------------------------------------------------


def _random_hill_climb(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Move a tiny random fraction of mass between two positions,
    keeping improvements.  Stops when ``deadline`` is reached."""
    n = a.size
    best_a = a.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()
    max_frac = 0.25                       # how much of a[i] we may move

    while time.time() < deadline:
        i = rng.integers(0, n)
        if best_a[i] <= 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        frac = rng.random() * max_frac
        delta = best_a[i] * frac
        if delta == 0.0:
            continue

        # propose move
        best_a[i] -= delta
        best_a[j] += delta

        val = _score(best_a)
        if val < best_val:
            best_val = val
            # keep the move
        else:
            # revert
            best_a[i] += delta
            best_a[j] -= delta

    return best_a, best_val


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the provided
    ``evaluate_sequence`` objective.  The routine explores a family of
    simple analytic shapes (power‑law and geometric), picks the best
    parameters, and finishes with a short hill‑climbing refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # a small safety margin

    best_seq = None
    best_val = float('inf')

    # ---- 1.  Scan several lengths ------------------------------------
    lengths = [800, 1500, 2500, 3500, 5000, 8000, 12000]
    for n in lengths:
        if time.time() >= deadline:
            break

        target = math.sqrt(2.0 * n)

        # ---- 1a. Power‑law sweep ------------------------------------
        # coarse grid
        p_grid = np.linspace(0.15, 1.30, 50)   # 50 points → cheap
        for p in p_grid:
            if time.time() >= deadline:
                break
            a = _make_powerlaw(p, n)
            val = _score(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[{time.time():.1f}]  Power‑law   n={n} p={p:.4f}  val={val:.6f}")

        # ---- 1b. Refine the best power‑law for this n (local golden‑section)
        # use the best p from the coarse scan as centre
        # keep a narrow interval around it
        # (if we never updated best_seq inside this n we still have a candidate)
        #   – we recompute the exact p that gives the smallest value.
        if best_seq is not None:
            # use a simple golden‑section around the best p of this n
            # retrieve the p that gave current best for this length
            # (this is a quick heuristic; we redo the search anyway)
            lo, hi = 0.15, 1.30
            phi = (math.sqrt(5) - 1) / 2    # 0.618...
            for _ in range(22):            # ~22 iterations → tolerance ~1e‑5
                if time.time() >= deadline:
                    break
                p1 = lo + (1 - phi) * (hi - lo)
                p2 = lo + phi * (hi - lo)
                a1 = _make_powerlaw(p1, n)
                a2 = _make_powerlaw(p2, n)
                v1 = _score(a1)
                v2 = _score(a2)
                if v1 < v2:
                    hi = p2
                    if v1 < best_val:
                        best_val = v1
                        best_seq = a1.copy()
                        print(f"[{time.time():.1f}]  GS‑Power   n={n} p={p1:.5f}  val={v1:.6f}")
                else:
                    lo = p1
                    if v2 < best_val:
                        best_val = v2
                        best_seq = a2.copy()
                        print(f"[{time.time():.1f}]  GS‑Power   n={n} p={p2:.5f}  val={v2:.6f}")

        # ---- 1c. Geometric sweep ------------------------------------
        r_grid = np.linspace(0.55, 0.985, 45)   # r close to 1 gives flatter vectors
        for r in r_grid:
            if time.time() >= deadline:
                break
            a = _make_geometric(r, n)
            val = _score(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[{time.time():.1f}]  Geometric  n={n} r={r:.4f}  val={val:.6f}")

    # ------------------------------------------------------------------
    # 2.  Local refinement – random hill‑climbing on the best candidate
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a_ref, v_ref = _random_hill_climb(best_seq, deadline)
        if v_ref < best_val:
            best_val = v_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.1f}]  hill‑climb finished, new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3.  Post‑processing: trim trailing zeros (should not happen often)
    # ------------------------------------------------------------------
    if best_seq is None:
        # safety fallback – uniform distribution (guaranteed feasible)
        n = 1000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=np.float64)

    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Return as a plain Python list (required by the challenge harness)
    return best_seq.tolist()
