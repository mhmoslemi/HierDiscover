"""
Propose a sequence of non‑negative heights that minimises the evaluation
function defined by ``evaluate_sequence``.

The search proceeds in three stages:

1. **Parametric sweep** – a family of power‑law shapes
   a_i ∝ (i+offset)^{-p} with an optional linear tilt is explored.
   Each candidate is projected onto the ℓ₁‑sphere
   (Σ a_i = √(2 n)) and scored with a fast FFT based
   computation of the objective.  The best candidate is kept.

2. **Smooth‑max gradient descent** – starting from the best candidate,
   we minimise the soft‑max approximation of the self‑convolution.
   The gradient is obtained by a cross‑correlation (FFT),
   projected to the affine hyper‑plane (fixed ℓ₁‑norm) and updated
   with momentum.  The temperature (τ) and step‑size decay
   exponentially.

3. **Final random polishing** – occasional pair‑wise mass transfers
   and jitter are performed while time permits; any improvement
   is recorded.

All helper functions are top‑level, no closures or lambdas are used.
The implementation respects a 1000 s wall‑clock budget and always
returns the best sequence discovered.
"""

import time
import numpy as np
from math import gamma

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using FFT (real input)."""
    n, m = a.size, b.size
    L = 1 << ((n + m - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)
    return conv[: n + m - 1]


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a (length 2·len(a)‑1)."""
    return fft_convolve(a, a)


def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector so that its ℓ₁‑norm equals *target*."""
    s = v.sum()
    if s > 0.0:
        return v * (target / s)
    # fallback – uniform distribution (should never happen)
    return np.full_like(v, target / v.size)


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def generate_power_shape(
    n: int,
    target_sum: float,
    exponent: float = 0.5,
    offset: float = 0.0,
    tilt: float = 0.0,
) -> np.ndarray:
    """
    a_i = (i + offset)^{-exponent} * (1 + tilt * i/(n‑1))

    The result is projected onto the ℓ₁‑sphere of radius *target_sum*.
    """
    idx = np.arange(n, dtype=np.float64)
    # power‑law core
    core = (idx + offset) ** (-exponent)
    # optional linear tilt (negative tilt reduces early entries)
    if abs(tilt) > 1e-12:
        core = core * (1.0 + tilt * idx / (n - 1))
    core = np.clip(core, 0.0, None)
    return project_to_sum(core, target_sum)


def generate_exponential_shape(
    n: int,
    target_sum: float,
    lam: float = 0.0005,
) -> np.ndarray:
    """a_i = exp(-lam * i)  (decreasing exponential)."""
    idx = np.arange(n, dtype=np.float64)
    core = np.exp(-lam * idx)
    return project_to_sum(core, target_sum)


def random_initial_shape(n: int, target_sum: float) -> np.ndarray:
    """Uniform random start, projected to the target ℓ₁‑norm."""
    cand = np.random.rand(n)
    return project_to_sum(cand, target_sum)


# ----------------------------------------------------------------------
# Objective (fast, avoids the clipping performed inside evaluate_sequence)
# ----------------------------------------------------------------------


def fast_score(seq: np.ndarray) -> float:
    """Return the exact value of evaluate_sequence (no clipping part)."""
    n = seq.size
    conv = conv_self_fft(seq)
    max_b = conv.max()
    s = seq.sum()
    # the evaluate_sequence returns exactly 2*n*max_b/(s**2)
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (single‑step helper)
# ----------------------------------------------------------------------


def smooth_max_step(
    a: np.ndarray,
    target_sum: float,
    lr: float,
    tau: float,
    momentum: float,
    v_prev: np.ndarray,
) -> tuple[np.ndarray, np.ndarray]:
    """
    One iteration of projected gradient descent on the smoothed max
    of the self‑convolution.
    """
    n = a.size

    # 1) a*a
    conv = conv_self_fft(a)                     # length 2n‑1

    # 2) soft‑max weights
    max_c = conv.max()
    exp_arg = (conv - max_c) / tau
    w = np.exp(exp_arg, dtype=np.float64)
    w_sum = w.sum()
    if w_sum == 0.0:                           # degenerate case – use hard max
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    # 3) gradient = 2 * (w ⋆ a)  (cross‑correlation)
    a_rev = a[::-1]
    grad_full = fft_convolve(w, a_rev)          # length 3n‑2
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]

    # 4) enforce the ℓ₁‑constraint by removing the mean component
    direction = -(grad - np.mean(grad))

    # 5) momentum + step
    v = momentum * v_prev + direction
    a_new = a + lr * v

    # 6) feasibility (non‑negative, correct sum)
    a_new = np.clip(a_new, 0.0, None)
    a_new = project_to_sum(a_new, target_sum)

    return a_new, v


# ----------------------------------------------------------------------
# Random perturbations used in the polishing phase
# ----------------------------------------------------------------------


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """Move a fraction *scale* of the mass from a random source to a random target."""
    n = a.size
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0.0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_to_sum(a, target_sum)


def jitter(a: np.ndarray, target_sum: float, sigma: float = 1e-3) -> np.ndarray:
    """Add tiny Gaussian noise, clip and renormalise."""
    a = a + np.random.randn(a.size) * sigma
    a = np.clip(a, 0.0, None)
    return project_to_sum(a, target_sum)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------

    """
    Search for a non‑negative height sequence that minimises the
    ``evaluate_sequence`` objective within the provided time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose a length (reuse an existing construction if possible)
    # ------------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
    else:
        # a size that balances quality and speed; 8000–10000 works well
        n = np.random.randint(8000, 10001)

    target_sum = np.sqrt(2.0 * n)            # the normalisation used in the paper

    # ------------------------------------------------------------------
    # Phase 1 – parametric sweep over power‑law families
    # ------------------------------------------------------------------
    best_seq = np.full(n, target_sum / n)    # uniform baseline
    best_score = fast_score(best_seq)

    print(f"[{time.time():.2f}] Starting parametric sweep, n={n}")

    # grids (coarse – cheap enough to stay well within the budget)
    exp_grid = np.linspace(0.35, 0.65, 13)           # exponent p
    offset_grid = np.linspace(0.0, 5.0, 7)           # additive offset
    tilt_grid = np.linspace(-0.4, 0.4, 9)            # linear tilt

    for p in exp_grid:
        for off in offset_grid:
            for tlt in tilt_grid:
                cand = generate_power_shape(n, target_sum,
                                            exponent=p,
                                            offset=off,
                                            tilt=tlt)
                score = fast_score(cand)
                if score < best_score:
                    best_score = score
                    best_seq = cand.copy()
                    print(f"  New best power shape p={p:.3f}, off={off:.2f}, tilt={tlt:.2f} → {best_score:.6f}")
                if time.time() >= deadline:
                    print("Time exhausted during parametric sweep.")
                    return best_seq.tolist()

    # also try a few purely exponential candidates (may act as good seeds)
    for lam in np.linspace(0.0002, 0.0015, 12):
        cand = generate_exponential_shape(n, target_sum, lam=lam)
        score = fast_score(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"  New best exponential λ={lam:.5f} → {best_score:.6f}")
        if time.time() >= deadline:
            return best_seq.tolist()

    # a handful of random starts
    for _ in range(30):
        cand = random_initial_shape(n, target_sum)
        score = fast_score(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"  Random start improvement → {best_score:.6f}")
        if time.time() >= deadline:
            return best_seq.tolist()

    # ------------------------------------------------------------------
    # Phase 2 – smooth‑max gradient descent with momentum
    # ------------------------------------------------------------------
    a = best_seq.copy()
    v = np.zeros_like(a)                     # momentum buffer

    avg = target_sum / n
    lr = 0.04 * avg                          # initial step size
    tau = 0.3 * a.max()                      # initial temperature
    momentum = 0.93

    iteration = 0
    no_improve = 0

    print(f"[{time.time():.2f}] Entering smooth‑max descent (lr={lr:.6f}, tau={tau:.6f})")
    while time.time() < deadline:
        a, v = smooth_max_step(a, target_sum, lr, tau, momentum, v)
        score = fast_score(a)
        iteration += 1

        if score + 1e-12 < best_score:
            best_score = score
            best_seq = a.copy()
            no_improve = 0
            print(f"Iter {iteration:06d} – improvement: {best_score:.6f}")
        else:
            no_improve += 1

        # exponential cooling schedules
        if iteration % 250 == 0:
            lr *= 0.96
            tau *= 0.96
            lr = max(lr, 1e-7)
            tau = max(tau, 1e-5)

        # occasional stochastic escapes
        if np.random.rand() < 0.025:
            a = random_pair_transfer(a, target_sum, scale=0.007)

        # if stuck for a while, jitter a bit
        if no_improve >= 1800:
            a = jitter(a, target_sum, sigma=4e-4)
            v[:] = 0.0
            no_improve = 0
            print(f"Iter {iteration:06d} – jitter reset")

    # ------------------------------------------------------------------
    # Phase 3 – final polishing with random pair transfers
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Descent finished; best score {best_score:.6f}")

    while time.time() < deadline - 0.2:
        a = random_pair_transfer(a, target_sum, scale=0.004)
        score = fast_score(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"Polish – improvement to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence as a plain Python list
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Returning best sequence, final score {best_score:.6f}")
    return best_seq.tolist()
