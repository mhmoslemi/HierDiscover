```
def _subgradient_max(a):
    n = a.shape[0]
    size = 1 << ((2*n-1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
    max_val = b.max()
    k = int(b.argmax())
    # compute gradient
    g = np.zeros_like(a)
    if k < n:
        # indices 0..k inclusive
        g[:k+1] = 2.0 * a[k::-1]
    else:
        start = k - (n-1)
        length = 2*n - 1 - k
        # length = n - start
        g[start:] = 2.0 * a[:length][::-1]
    return g, max_val
```

"""
Step‑function optimiser.

The objective (after fixing the ℓ¹‑norm to √(2 n)) is simply the
maximum entry of the linear convolution a∗a.  The algorithm combines
three complementary heuristics:

1.  **Annealed soft‑max gradient descent** – minimise a smooth
    approximation of max( a∗a ) using the log‑sum‑exp
    (β‑softmax).  β is increased gradually up to a large value,
    so the final stage behaves like a true max‑gradient.

2.  **Exact max‑subgradient descent** – after the annealing phase,
    the index k⋆ where the convolution attains its maximum is
    identified and the sub‑gradient
        ∇ b_{k⋆}[i] = 2·a_{k⋆‑i}
    (zero outside the support) is followed with a diminishing step
    size and projection on the simplex (non‑negativity + fixed ℓ¹‑norm).

3.  **Stochastic hill‑climbing** – cheap random mass transfers
    between two positions that improve the objective are kept,
    providing a final local refinement.

The search runs several random restarts for a range of vector
lengths (up to 12 000).  Each restart is given a time slice of the
overall budget; the best sequence found so far is returned.
All operations are performed with FFTs, giving O(n log n) cost per
iteration, which allows thousands of iterations within the 1000 s
budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """Fast evaluation of the competition objective (scale‑invariant)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:  # should never happen – safety guard
        return np.inf
    return 2.0 * n * max_b / (s * s)   # = max_b when Σa = √(2 n)


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    { x | x_i ≥ 0, Σ x_i = s }.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) gradient of max(convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Return (g, b) where
        b = a*a (full convolution, length 2·n‑1)
        g = ∇ (log‑sum‑exp_{β}(b)) = 2·(a ⋆ p)
    with p_k ∝ exp(β·b_k).  Both are computed with FFTs.
    """
    n = a.shape[0]

    # ---- b = a*a -------------------------------------------------------
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ---- soft‑max probabilities p ---------------------------------------
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))          # subtract max for stability
    p = exps / np.sum(exps)                    # shape (2·n‑1,)

    # ---- gradient g = 2·(a ⋆ p) ---------------------------------------
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1            # = 3·n‑2
    size2 = 1 << ((conv_len).bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    # central part gives the cross‑correlation
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Exact max‑subgradient (gradient of the active convolution entry)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray):
    """
    Compute the sub‑gradient of max(a*a) and the corresponding maximal value.
    Returns (g, max_val) where
        g_i = 2·a_{k⋆‑i}  (zero if the index is out of range)
    and k⋆ is the index where the convolution attains its maximum.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_val = float(b.max())
    k = int(b.argmax())

    g = np.zeros_like(a)

    if k < n:
        # indices 0 … k inclusive contribute
        g[: k + 1] = 2.0 * a[k::-1]
    else:
        start = k - (n - 1)                     # first i with a_{k‑i} in range
        length = 2 * n - 1 - k                  # number of overlapping entries
        # length == n - start
        g[start:] = 2.0 * a[:length][::-1]

    return g, max_val


# ----------------------------------------------------------------------
# Projected gradient descent on the soft‑max approximation
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected gradient descent on the soft‑max (log‑sum‑exp) surrogate.
    The temperature β is increased gradually up to a large value.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 1024.0                     # higher than before – tighter approximation
    step_base = target_sum * 0.15         # slightly smaller steps for stability
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        # gradient at current temperature
        g, _ = _gradient_softmax(a, beta)

        # step size scaled by β and gradient magnitude
        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            lr = step_base / (norm_g * (beta + 1.0))
            a -= lr * g

        # projection onto the simplex (non‑negative + fixed ℓ¹‑norm)
        a = _project_to_simplex(a, target_sum)

        # occasional tiny Gaussian jitter to escape plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # evaluation
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] anneal β={beta:.1f}  score={best_val:.6f}")

        # annealing schedule
        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 1.4, max_beta)

        it += 1

    return best_a, best_val

# Subgradient descent on the true max(convolution)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected sub‑gradient descent on the exact objective max(a*a).
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    it = 0
    base_step = target_sum * 0.08          # conservative step magnitude

    while time.time() < deadline:
        g, cur_max = _subgradient_max(a)

        norm_g = np.linalg.norm(g)
        if norm_g == 0:
            break

            best_a = a.copy()

# ----------------------------------------------------------------------

# ----------------------------------------------------------------------


def _random_hill_climb(a: np.ndarray, deadline: float):
    """
    Very cheap hill‑climber that preserves the ℓ¹‑norm.
    Randomly moves a fraction of mass from i to j and keeps improving moves.
    """
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    cur_val = best_val

    max_frac = 0.12
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        frac = rng.random() * max_frac
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        new_val = _score(a)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill‑climb improved = {best_val:.6f}")
        else:
            # revert
            a[j] -= delta
            a[i] += delta

    return best_a, best_val
