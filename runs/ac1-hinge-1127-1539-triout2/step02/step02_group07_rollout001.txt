"""
Non‑negative step‑function optimizer (new version).

The objective to minimise is

    score = 2 * n * max(conv(p, p)) / (sum(p)**2)

which is invariant to scaling, so we keep the candidate on the probability
simplex (sum(p)=1).  The algorithm consists of two phases:

1.  A smooth approximation of the ℓ∞‑norm – the log‑sum‑exp (soft‑max) –
    is minimised by an Adam‑style projected gradient descent.
    The “temperature’’ α starts small (very smooth) and is multiplied
    each iteration so that the approximation converges to the true
    maximum.

2.  After the smooth phase a few steps of a classic mirror‑descent /
    exponentiated‑gradient scheme (used in the previous state‑of‑the‑art)
    are applied to polish the solution.

Both phases use fast FFT‑based convolution and a Euclidean projection
onto the simplex (Duchi et al. 2008).  The routine respects a 1000 s
budget and returns the best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("simplex projection works on 1‑D arrays")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT helpers (full convolution and correlation)
# ----------------------------------------------------------------------
def fft_len_for_conv(n: int, extra: int = 0) -> int:
    """Return a power‑of‑two length sufficient for convolution of size n."""
    target = 2 * n - 1 + extra
    return 1 << ((target - 1).bit_length())


def full_conv_fft(p: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution of p with itself using a pre‑chosen FFT length L."""
    ft = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(ft * ft, n=L)[: 2 * p.size - 1]
    # eliminate tiny negatives from numerical noise
    conv[conv < 0] = 0.0
    return conv


def correlation_with_weights(p: np.ndarray, w: np.ndarray, L: int) -> np.ndarray:
    """
    Compute the vector t_j = Σ_k w_k * p_{k-j} for j=0,…,n‑1
    (linear correlation of p with w) using FFT of length L.
    """
    ft_p = np.fft.rfft(p, n=L)
    ft_w_rev = np.fft.rfft(w[::-1], n=L)
    corr_full = np.fft.irfft(ft_p * ft_w_rev, n=L)[: p.size + w.size - 1]
    start = w.size - 1
    return corr_full[start : start + p.size]


# ----------------------------------------------------------------------
# Gradient of smoothed max (log‑sum‑exp)
# ----------------------------------------------------------------------
def softmax_gradient(p: np.ndarray, conv: np.ndarray, alpha: float, L: int):
    """
    Return the (approximate) sub‑gradient of max(conv(p,p))
    obtained from the soft‑max with temperature 1/alpha.
    Also returns the soft‑max weights (not needed outside).
    """
    max_val = conv.max()
    # shift for numerical stability: the largest entry becomes 0
    shifted = conv - max_val
    w = np.exp(alpha * shifted)          # max entry has weight 1, others < 1
    w_sum = w.sum()
    w /= w_sum

    # term = Σ_k w_k * p_{k-j}
    term = correlation_with_weights(p, w, L)
    # gradient of (1/α)·log Σ exp(α·conv) = Σ w_k ∂conv_k/∂p
    # ∂conv_k/∂p_j = p_{k-j} + p_j   (see analysis)
    grad = 2.0 * (term + p)
    return grad, w


# ----------------------------------------------------------------------
# Mirror‑descent refinement (same spirit as the previous baseline)
# ----------------------------------------------------------------------
def subgradient(p: np.ndarray, conv: np.ndarray, top_k: int = 5) -> np.ndarray:
    """Sub‑gradient of max(conv(p,p)) using the `top_k` largest entries."""
    n = p.size
    g = np.zeros_like(p)
    if top_k >= conv.size:
        ks = np.arange(conv.size)
    else:
        ks = np.argpartition(conv, -top_k)[-top_k:]
    for k in ks:
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        idx = np.arange(i_min, i_max + 1, dtype=int)
        g[idx] += p[k - idx]
    g *= 2.0
    return g


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0):
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises `evaluate_sequence`.  The routine respects the 1000 s
    time limit and returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    best_score = float("inf")
    best_seq = None

    # --------------------------------------------------------------
    # 1) Seed generation (all seeds are already normalised to sum=1)
    # --------------------------------------------------------------
    seeds = []

    # previous best if available
    if "height_sequence_1" in globals():
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.size > 8000:
            prev = prev[:8000]
        prev = prev / prev.sum()
        seeds.append(prev)

    # uniform of length 8000
    seeds.append(np.ones(8000, dtype=np.float64) / 8000.0)

    # random Dirichlet‑like seeds
    for _ in range(4):
        r = np.random.rand(8000)
        seeds.append(r / r.sum())

    # power‑law families
    for p_exp in (0.45, 0.50, 0.55):
        for sigma in (0.001, 0.01, 0.1):
            i = np.arange(8000, dtype=np.float64)
            seq = (i + sigma) ** (-p_exp)
            seq = seq[seq > 1e-12]
            seq = seq / seq.sum()
            seeds.append(seq)

    # exponential families
    for lam in (0.0003, 0.0006, 0.0012):
        i = np.arange(8000, dtype=np.float64)
        seq = np.exp(-lam * i)
        seq = seq / seq.sum()
        seeds.append(seq)

    # --------------------------------------------------------------
    # 2) Optimisation per seed
    # --------------------------------------------------------------
    for base in seeds:
        if time.time() >= deadline:
            break

        p = base.copy()
        n = p.size

        # pre‑compute FFT length that works for both convolution and correlation
        L_fft = fft_len_for_conv(n, extra=n - 1)   # 3n‑2 is enough

        # evaluate the initial candidate
        conv = full_conv_fft(p, L_fft)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = p.tolist()
            print(f"Seed init | n={n} | score={best_score:.6f}")

        # ----- phase 1: smooth Adam descent ---------------------------------
        max_iters = 15000
        # Adam hyper‑parameters
        m = np.zeros_like(p)
        v = np.zeros_like(p)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        lr = 0.4

        # temperature schedule (α → α_final)
        alpha = 5.0
        alpha_final = 5e4
        alpha_mul = (alpha_final / alpha) ** (1.0 / max_iters)

        for it in range(1, max_iters + 1):
            if time.time() >= deadline:
                break

            # convolution (needed for the soft‑max)
            conv = full_conv_fft(p, L_fft)
            max_val = conv.max()
            cur_score = 2.0 * n * max_val
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = p.tolist()
                print(f"Iter {it:5d} | n={n} | new best {best_score:.6f}")

                if best_score < 1.503:
                    print("Target reached – terminating early.")
                    return best_seq

            # gradient of the smoothed max
            g, _ = softmax_gradient(p, conv, alpha, L_fft)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g * g)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            p = p - step
            p = np.maximum(p, 0.0)          # keep non‑negative
            # project onto simplex
            s = p.sum()
            if s > 0:
                p = p / s
            else:
                # re‑initialise if everything collapsed
                p = np.random.rand(n)
                p = p / p.sum()

            # increase temperature
            alpha *= alpha_mul

            # occasional tiny jitter to avoid stagnation
            if np.random.rand() < 0.01:
                idx = np.random.randint(n)
                delta = (np.random.rand() - 0.5) * 0.02 * p[idx]
                p[idx] = max(p[idx] + delta, 0.0)
                p = p / p.sum()

        # ----- phase 2: classic mirror‑descent (sub‑gradient) --------------
        # use the result of phase 1 as a warm start
        sub_iters = 4000
        eta0 = 0.25
        top_k = 7
        for it in range(1, sub_iters + 1):
            if time.time() >= deadline:
                break
            conv = full_conv_fft(p, L_fft)
            max_val = conv.max()
            cur_score = 2.0 * n * max_val
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = p.tolist()
                print(f"SubIter {it:4d} | n={n} | new best {best_score:.6f}")
                if best_score < 1.503:
                    print("Target reached in sub‑phase – terminating.")
                    return best_seq

            g = subgradient(p, conv, top_k=top_k)
            eta = eta0 / np.sqrt(it)
            p = p * np.exp(-eta * g)          # multiplicative update
            p = project_simplex(p)

            # rare random jump
            if np.random.rand() < 0.008:
                i, j = np.random.choice(n, size=2, replace=False)
                delta = min(p[i], 0.01) * (np.random.rand() - 0.5)
                p[i] = max(p[i] - delta, 0.0)
                p[j] = max(p[j] + delta, 0.0)
                p = project_simplex(p)

    # ------------------------------------------------------------------
    # 3) Return best sequence found
    # ------------------------------------------------------------------
    print("Final best score:", best_score,
          "length:", len(best_seq) if best_seq is not None else None)
    return best_seq if best_seq is not None else []
