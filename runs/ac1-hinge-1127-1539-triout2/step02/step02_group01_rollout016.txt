"""
Enhanced search for a non‑negative step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three stages:

1.  A quick parametric sweep over several “U‑shaped’’ families
    (linear, quadratic, exponential).  The best candidate from this
    sweep is used as the starting point.

2.  Adam‑accelerated smoothed gradient descent on the
    log‑sum‑exp approximation of the max‑convolution, with an
    adaptive learning‑rate schedule.  After a number of iterations we
    switch to a pure sub‑gradient step that uses the exact active set
    (indices attaining the maximal convolution value).  Both phases
    keep the vector on the ℓ₁‑simplex (fixed ℓ₁‑norm and non‑negativity).

3.  A final local‑search stage that repeatedly transfers a tiny
    amount of mass between two random coordinates.  The convolution
    is updated incrementally, so each trial is O(n) instead of a full
    FFT.  Any improvement is kept.

All operations respect the 1000 s time budget and the solution is
returned as a plain list of floats.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures, no lambdas)
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution of ``a`` using an FFT of power‑of‑two size."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    L = 1 << (n.bit_length())
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    return np.fft.irfft(fx * fy, n=L)[:n]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the simplex
        {x | x_i ≥ 0, sum_i x_i = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, target / len(v))

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def generate_U_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
    exponent: float = 1.0,
) -> np.ndarray:
    """
    Produce a symmetric “U‑shaped’’ sequence of length ``n``.
    ``kind`` ∈ {linear, quadratic, exponential, power}.
    ``param`` controls the edge enhancement, ``exponent`` is used only
    for the ``power`` kind.
    The raw shape is projected onto the ℓ₁‑simplex with sum ``target_sum``.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * (d ** 2)
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "power":
        raw = 1.0 + param * (d ** exponent)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a tiny amount of mass from one index to another,
    then re‑project onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0.0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)

def smoothed_gradient(a: np.ndarray, tau_factor: float):
    """
    Compute the log‑sum‑exp smoothed gradient of ``max(conv(a,a))``.
    Returns (gradient, max_conv, tau) where ``gradient`` has shape (n,).
    """
    n = a.shape[0]
    conv = conv_self_fft(a)                # length 2n‑1
    max_conv = conv.max()
    tau = max(tau_factor * max_conv, 1e-12)

    # smoothing weights
    w = np.exp((conv - max_conv) / tau)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient = (2/τ) * (w * a_rev) restricted to indices n‑1 … n‑1+n‑1
    a_rev = a[::-1]
    L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
    grad = (2.0 / tau) * conv_full[n - 1 : n - 1 + n]
    return grad, max_conv, tau


def subgradient_step(a: np.ndarray, target_sum: float, step_size: float):
    """
    One sub‑gradient step based on the exact active set
    (indices attaining the maximal convolution value).
    """
    n = a.shape[0]
    conv = conv_self_fft(a)
    max_conv = conv.max()
    # active mask (binary)
    mask = (conv >= max_conv - 1e-12).astype(float)

    # w = mask * a_rev  (convolution)
    a_rev = a[::-1]
    L = 1 << ((len(mask) + len(a_rev) - 1).bit_length())
    M = np.fft.rfft(mask, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    w_full = np.fft.irfft(M * A_rev, n=L)[: len(mask) + len(a_rev) - 1]
    grad = 2.0 * w_full[n - 1 : n - 1 + n]

    a_new = a - step_size * grad
    a_new = np.maximum(a_new, 0.0)
    a_new = project_onto_simplex(a_new, target_sum)
    return a_new, max_conv


def local_pair_search(
    a: np.ndarray,
    target_sum: float,
    conv: np.ndarray,
    budget: float,
) -> tuple[np.ndarray, float]:
    """
    Tiny random pair‑mass transfers with an **incremental** convolution update.
    ``budget`` is the remaining wall‑clock seconds.
    Returns the best (vector, max_conv) pair found.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_conv = conv.copy()
    best_score = best_conv.max()
    deadline = time.time() + budget

    while time.time() < deadline:
        i, j = np.random.choice(n, 2, replace=False)
        if best_a[i] <= 0.0:
            continue
        delta = 0.02 * best_a[i]               # transfer 2 % of source mass
        # incremental update of the full convolution
        cand_conv = best_conv.copy()
        # the slice ``i:i+n`` corresponds to the contribution of a[i]
        cand_conv[i:i + n] -= 2.0 * delta * best_a
        cand_conv[j:j + n] += 2.0 * delta * best_a
        new_score = cand_conv.max()
        if new_score + 1e-12 < best_score:
            # accept move
            best_a[i] -= delta
            best_a[j] += delta
            best_conv = cand_conv
            best_score = new_score
            # re‑project to the simplex (numerical drift)
            best_a = project_onto_simplex(best_a, target_sum)
            # recompute exact convolution for safety after acceptance
            best_conv = conv_self_fft(best_a)
            best_score = best_conv.max()
        # no‑else: reject silently

    return best_a, best_score

                for exp in exponents:
                    cand = generate_U_shape(n, target_sum, kind=kind,
                                            param=p, exponent=exp)
                    score = evaluate_sequence(cand.tolist())
                    if score < best_score:
                        best_score = score
                        best_seq = cand.copy()
                        print(f"  New best {kind} (p={p:.2f}, exp={exp:.1f}): {best_score:.6f}")
            else:
                cand = generate_U_shape(n, target_sum, kind=kind, param=p)
                score = evaluate_sequence(cand.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = cand.copy()
                    print(f"  New best {kind} (p={p:.2f}): {best_score:.6f}")

            if time.time() >= deadline:
                print("Time exhausted during sweep.")
                return best_seq.tolist()

    # a few random Dirichlet‑like starts
    for _ in range(5):
        raw = np.random.rand(n)
        cand = project_onto_simplex(raw, target_sum)
        score = evaluate_sequence(cand.tolist())
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"  Random start improved to {best_score:.6f}")

    # baseline from previous run (if any)
    if init_seq is not None:
        score = evaluate_sequence(init_seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = init_seq.copy()
            print(f"  Baseline improves initial score: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam smoothed gradient descent → sub‑gradient refinement
    # --------------------------------------------------------------
    a = best_seq.copy()
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t_iter = 0

    eta = 0.4 * (target_sum / n)   # base learning‑rate (≈10 % of average entry)
    tau_factor = 0.08
    no_improve = 0

    print(f"[{time.time():.2f}] Entering Adam‑accelerated smoothed descent")

    while time.time() < deadline:
        # ----- smoothed gradient -----
        grad, cur_max, tau = smoothed_gradient(a, tau_factor)

        # Adam update
        t_iter += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = project_onto_simplex(a_new, target_sum)

        # evaluate
        new_max = conv_self_fft(a_new).max()
        if new_max + 1e-12 < best_score:
            best_score = new_max
            best_seq = a_new.copy()
            print(f"Iter {t_iter:06d} – improved to {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # ---------------- schedule ----------------
        if t_iter % 800 == 0:
            eta *= 0.94                 # slowly decay learning‑rate
            tau_factor = max(0.003, tau_factor * 0.96)   # tighten smoothing

        # occasional random pair transfer (helps escape plateaus)
        if np.random.rand() < 0.012:
            a = random_pair_transfer(a, target_sum, scale=0.012)

        # heavy jitter after prolonged stagnation
        if no_improve >= 2500:
            jitter = np.random.randn(n) * 0.0006
            a = project_onto_simplex(np.maximum(a + jitter, 0.0), target_sum)
            no_improve = 0
            print(f"Iter {t_iter:06d} – jitter applied")

        # ---------------------------------------------------------
        # Switch to pure sub‑gradient for the last ~15 % of the budget
        # ---------------------------------------------------------
        if time.time() > deadline - (budget_s * 0.15):
            break

    # ---- Sub‑gradient refinement (active‑set) ----
    step_size = target_sum / n * 0.5   # start with a modest step
    print(f"[{time.time():.2f}] Sub‑gradient refinement starts (step={step_size:.6f})")
    while time.time() < deadline:
        a, cur_max = subgradient_step(a, target_sum, step_size)
        if cur_max + 1e-12 < best_score:
            best_score = cur_max
            best_seq = a.copy()
            print(f"Sub‑grad iter – improved to {best_score:.6f}")
        # schedule decay
        step_size *= 0.9995
        if step_size < 1e-6:
            step_size = 1e-6
        # occasional jitter to avoid stagnation
        if np.random.rand() < 0.008:
            jitter = np.random.randn(n) * 0.0004
            a = project_onto_simplex(np.maximum(a + jitter, 0.0), target_sum)

    # --------------------------------------------------------------
    # Phase 3 – final local random pair transfers (incremental update)
    # --------------------------------------------------------------
    print(f"[{time.time():.2f}] Final local search, current best {best_score:.6f}")
    remaining = deadline - time.time()
    if remaining > 5.0:   # ensure we have some time left
        try:
            # start from the best vector found so far
            conv_current = conv_self_fft(best_seq)
            best_seq, best_score = local_pair_search(
                best_seq, target_sum, conv_current, remaining * 0.8
            )
        except Exception as exc:   # pragma: no cover
            print("Local search failed:", exc)

    # ------------------------------------------------------------------
    # Polishing with a few explicit evaluate_sequence calls
    # ------------------------------------------------------------------
    for _ in range(30):
        if time.time() >= deadline:
            break
        cand = random_pair_transfer(best_seq.copy(), target_sum, scale=0.005)
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"Polish transfer improved to {best_score:.6f}")

    print(f"[{time.time():.2f}] Finished – best score {best_score:.6f}")
    return best_seq.tolist()
