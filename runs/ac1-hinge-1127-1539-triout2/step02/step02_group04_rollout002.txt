```
def gradient_wrt_x(w, x):
    # w length 2n-1, x length n
    n = x.shape[0]
    xr = x[::-1]
    N = 1 << (int(np.ceil(np.log2(len(w) + len(xr) - 1))))
    W = np.fft.rfft(w, N)
    Xr = np.fft.rfft(xr, N)
    conv_full = np.fft.irfft(W * Xr, N)[:len(w) + len(xr) - 1]
    # valid slice
    start = n - 1
    end = start + n
    return 2 * conv_full[start:end]
```

"""
Propose a step‑function (non‑negative vector) that minimizes
evaluate_sequence().  The search proceeds in two main phases:

1.  A wide parametric sweep over simple analytic families
    (power‑law, exponential, mixed) – every candidate is normalized
    to the ℓ¹‑simplex and evaluated; the best one is kept.

2.  A fast Adam‑style projected gradient descent on a *smooth*
    approximation of the max‑convolution term.  The smooth max is a
    soft‑max with a temperature γ that is slowly increased, so that the
    optimisation follows a well‑behaved landscape and gradually focuses
    on the true max.  After each update the iterate is projected back to
    the probability simplex.  Occasionally tiny entries are appended
    (increasing the length) and small random jitters are applied to
    escape plateaus.  The best sequence seen during the run is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _generate_powerlaw(p: float, sigma: float,
                       max_len: int = 5000, eps: float = 1e-12) -> np.ndarray:
    """a_i = (i+σ)^{-p}, truncated when the value falls below eps."""
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p)
    return a[a > eps]


def _generate_exponential(lam: float,
                          max_len: int = 5000, eps: float = 1e-12) -> np.ndarray:
    """a_i = exp(-λ·i), truncated when the value falls below eps."""
    i = np.arange(max_len, dtype=np.float64)
    a = np.exp(-lam * i)
    return a[a > eps]


def _generate_mixed(p: float, sigma: float, lam: float,
                    max_len: int = 5000, eps: float = 1e-12) -> np.ndarray:
    """a_i = (i+σ)^{-p}·exp(-λ·i), truncated when the value falls below eps."""
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p) * np.exp(-lam * i)
    return a[a > eps]


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("simplex projection works on 1‑d arrays")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full convolution of x with itself using FFT (returns length 2·len(x)-1).
    """
    n = x.shape[0]
    size = 1 << (int(np.ceil(np.log2(2 * n - 1))))
    X = np.fft.rfft(x, n=size)
    conv = np.fft.irfft(X * X, n=size)[: 2 * n - 1]
    # numerical noise can produce tiny negative values – clip them
    np.maximum(conv, 0.0, out=conv)
    return conv


def _softmax_gradient(x: np.ndarray, gamma: float) -> np.ndarray:
    """
    Gradient of the smooth approximation
        φ_γ(x) = (1/γ)·log Σ_k exp(γ·(x∗x)_k)
    w.r.t. x, using FFT for speed.
    The returned gradient has the same shape as x.
    """
    n = x.shape[0]
    # 1) compute full auto‑convolution
    conv = _full_convolution(x)                 # length 2n‑1
    max_val = conv.max()
    # 2) soft‑max weights (stable by subtracting the max)
    weights = np.exp(gamma * (conv - max_val))
    weights /= weights.sum()
    # 3) gradient = 2 * correlation(weights, x)
    xr = x[::-1]
    size = 1 << (int(np.ceil(np.log2(len(weights) + len(xr) - 1))))
    W = np.fft.rfft(weights, n=size)
    Xr = np.fft.rfft(xr, n=size)
    full = np.fft.irfft(W * Xr, n=size)[: len(weights) + len(xr) - 1]
    start = n - 1
    grad = 2.0 * full[start : start + n]
    return grad


def _eval(seq: np.ndarray) -> float:
    """Thin wrapper around the provided evaluate_sequence (list‑oriented)."""
    try:
        return evaluate_sequence(seq.tolist())
    except Exception:
        return float("inf")

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list:
    """
    Search for a non‑negative step‑function that minimises the supplied
    evaluate_sequence().  The function respects a wall‑clock budget and
    returns the best sequence discovered.
    """
    np.random.seed(seed)
    # keep a small safety margin for finalisation
    deadline = time.time() + budget_s - 5.0

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # 0.  Initialise from a global sequence if it exists.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            sc = _eval(np.array(height_sequence_1, dtype=float))
            if np.isfinite(sc):
                best_score = float(sc)
                best_seq = list(height_sequence_1)
                print(f"Initial global sequence score = {best_score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Broad parametric sweep (fast, cheap families)
    # ------------------------------------------------------------------
    max_len = 5000
    eps = 1e-12
    print("=== Phase 1 – parametric sweep ===")

    # ----- power‑law family -----
    for p in np.linspace(0.30, 0.90, 25):
        for sigma in np.logspace(-6, 2, 30):
            if time.time() >= deadline:
                break
            seq = _generate_powerlaw(p, sigma, max_len, eps)
            if seq.size < 2:
                continue
            seq = seq / seq.sum()
            sc = _eval(seq)
            if np.isfinite(sc) and sc < best_score:
                best_score = float(sc)
                best_seq = seq.tolist()
                print(
                    f"new best PL p={p:.3f} σ={sigma:.5g} len={len(seq)} score={best_score:.6f}"
                )
        if time.time() >= deadline:
            break

    # ----- exponential family -----
    for lam in np.logspace(-5, -0.1, 30):
        if time.time() >= deadline:
            break
        seq = _generate_exponential(lam, max_len, eps)
        if seq.size < 2:
            continue
        seq = seq / seq.sum()
        sc = _eval(seq)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_seq = seq.tolist()
            print(
                f"new best EXP λ={lam:.5g} len={len(seq)} score={best_score:.6f}"
            )

    # ----- mixed power‑law / exponential family -----
    for p in (0.4, 0.5, 0.6):
        for sigma in (1e-4, 1e-2, 1.0):
            for lam in (1e-4, 5e-4, 1e-3):
                if time.time() >= deadline:
                    break
                seq = _generate_mixed(p, sigma, lam, max_len, eps)
                if seq.size < 2:
                    continue
                seq = seq / seq.sum()
                sc = _eval(seq)
                if np.isfinite(sc) and sc < best_score:
                    best_score = float(sc)
                    best_seq = seq.tolist()
                    print(
                        f"new best MIX p={p:.2f} σ={sigma:.5g} λ={lam:.5g} len={len(seq)} score={best_score:.6f}"
                    )

    # Fallback in the extremely unlikely case that nothing was found
    if best_seq is None:
        n0 = 800
        best_seq = np.ones(n0, dtype=float) / n0
        best_score = _eval(best_seq)
        print("fallback uniform sequence ‑ score", best_score)

    # ------------------------------------------------------------------
    # 2.  Adam‑style optimisation on a smooth max‑convolution objective
    # ------------------------------------------------------------------
    x = np.array(best_seq, dtype=np.float64)
    x = x / x.sum()                     # ensure we are on the simplex
    n = x.shape[0]

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.05

    # soft‑max temperature (γ) – start low, increase slowly
    gamma = 5.0
    gamma_max = 50.0
    gamma_inc = 1.001

    iteration = 0
    eval_every = 20
    grow_interval = 3000                # how often we append tiny entries
    extra_len = 200
    max_total_len = 8000

    print("=== Phase 2 – Adam optimisation ===")
    while time.time() < deadline:
        iteration += 1

        # --- gradient of smooth max -------------------------------------------------
        grad = _softmax_gradient(x, gamma)

        # --- Adam update ------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # --- descent step + projection onto the simplex -----------------------------
        y = x - step
        x = _simplex_projection(y)

        # --- occasional random jitter (helps escape plateaus) -----------------------
        if np.random.rand() < 0.01:
            idx = np.random.randint(x.shape[0])
            delta = (np.random.rand() - 0.5) * 0.2 * x[idx]
            x[idx] = max(0.0, x[idx] + delta)
            x = x / x.sum()

        # --- length growth (adds a few ultra‑small entries) -------------------------
        if iteration % grow_interval == 0 and x.shape[0] + extra_len <= max_total_len:
            extra = np.full(extra_len, 1e-12, dtype=np.float64)
            x = np.concatenate([x, extra])
            x = x / x.sum()
            # extend Adam moments with zeros for the new coordinates
            m = np.concatenate([m, np.zeros(extra_len, dtype=np.float64)])
            v = np.concatenate([v, np.zeros(extra_len, dtype=np.float64)])

        # --- raise γ slowly so that the approximation becomes sharper ---------------
        gamma = min(gamma * gamma_inc, gamma_max)

        # --- periodic evaluation ----------------------------------------------------
        if iteration % eval_every == 0:
            cur_score = _eval(x)
            if np.isfinite(cur_score) and cur_score < best_score:
                best_score = float(cur_score)
                best_seq = x.tolist()
                print(
                    f"iter {iteration:6d} γ={gamma:.2f} score={best_score:.6f}"
                )
                if best_score < 1.503:
                    print("Target bound achieved – terminating early.")
                    break

        # --- occasional restart if we appear stuck ---------------------------------
        if iteration % 5000 == 0:
            n_cur = x.shape[0]
            rnd = np.random.rand(n_cur)
            # mix current iterate with fresh random mass
            x = 0.6 * x + 0.4 * rnd
            x = x / x.sum()
            m = np.zeros_like(x)
            v = np.zeros_like(x)

    print("final best score:", best_score, "length:", len(best_seq))
    return best_seq
