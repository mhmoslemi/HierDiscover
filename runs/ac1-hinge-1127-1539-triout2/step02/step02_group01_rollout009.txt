    max_conv = conv.max()

def adam_step(a, grad, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    # ensure nonnegative (project onto simplex later)
    return a_new, m, v

"""
Multi‑start Adam optimiser for the smoothed max self‑convolution problem.

The algorithm works as follows:

1.  Build a small pool of diverse initial vectors:
    – the supplied baseline (if any),
    – a uniform distribution,
    – several random Dirichlet draws,
    – a few random “U‑shapes’’ (linear / quadratic / exponential).

2.  For each start run a projected Adam optimiser that minimises a
    smooth approximation of   max_k (a∗a)_k .
    The smooth max is realised with a temperature τ and a softmax weight
    w_k ∝ exp( (conv_k – max_conv)/τ ).  The gradient of the smooth
    objective is analytically obtained as a convolution of w with the
    reversed current vector.  After each Adam step we project back onto
    the ℓ₁‑simplex (sum equals √(2 n) and non‑negative entries).

3.  τ and the plain learning rate are annealed during the run.
    If progress stalls we inject a small Gaussian jitter or a larger
    random pair‑transfer to escape shallow minima.

4.  The best vector seen over the whole time budget (≈1000 s) is
    returned as a plain Python list.

All heavy work (convolutions) is performed with FFTs, giving
O(n log n) per iteration, which comfortably fits the allowed budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a with itself using an FFT of power‑of‑two size.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def conv_fft(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of u and v (both 1‑D) using an FFT of appropriate size.
    """
    L = 1 << ((len(u) + len(v) - 1).bit_length())
    fu = np.fft.rfft(u, n=L)
    fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fu * fv, n=L)
    return conv[: len(u) + len(v) - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex
        { x | x_i ≥ 0, sum_i x_i = target }.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fall back to uniform distribution
        return np.full_like(v, target / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def generate_U_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
) -> np.ndarray:
    """
    Produce a symmetric “U‑shaped’’ sequence of length n.
    kind ∈ {linear, quadratic, exponential}.
    param ≥ 0 controls the edge enhancement.
    The result is projected onto the ℓ₁‑simplex with total mass = target_sum.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a tiny amount of mass from one random index to another,
    then re‑project onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


# ----------------------------------------------------------------------
# Smoothed‑max gradient utilities
# ----------------------------------------------------------------------


def smoothed_gradient_and_max(
    a: np.ndarray,
    a_rev: np.ndarray,
    tau_factor: float,
) -> tuple[np.ndarray, float]:
    """
    Return (gradient, max_conv) where gradient is the derivative of the
    τ‑smoothed max of the self‑convolution.
    """
    # 1) self‑convolution
    conv = conv_self_fft(a)               # length 2n‑1
    max_conv = conv.max()

    # 2) softmax weights (log‑sum‑exp smoothing)
    tau = max(tau_factor * max_conv, 1e-12)
    # subtract max for numerical stability
    w = np.exp((conv - max_conv) / tau)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # 3) gradient = (2/τ) * (w * a_rev) convolution
    conv_wa = conv_fft(w, a_rev)          # length 3n‑2
    n = a.shape[0]
    start = n - 1
    end = start + n
    grad = (2.0 / tau) * conv_wa[start:end]
    return grad, max_conv


# ----------------------------------------------------------------------
# Adam optimiser
# ----------------------------------------------------------------------


def adam_update(
    a: np.ndarray,
    grad: np.ndarray,
    m: np.ndarray,
    v: np.ndarray,
    t: int,
    lr: float,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """
    One step of the Adam optimiser (no projection).  Returns
    (new_params, new_m, new_v).
    """
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)

    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    return a_new, m, v


# ----------------------------------------------------------------------
# Core optimiser for a single start
# ----------------------------------------------------------------------


def run_optimizer(
    init_a: np.ndarray,
    target_sum: float,
    deadline: float,
    global_best_score: float,
    global_best_seq: np.ndarray,
) -> tuple[np.ndarray, float]:
    """
    Optimise a single start vector until `deadline` is reached.
    Returns the best vector found in this run and its score.
    """
    n = init_a.shape[0]
    a = init_a.copy()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # hyper‑parameters – randomised a bit for each run
    lr = np.random.uniform(0.3, 0.9) * target_sum / n          # base learning rate
    tau_factor = np.random.uniform(0.07, 0.25)                # initial temperature factor

    # local best
    best_local_seq = a.copy()
    best_local_score = float("inf")
    no_improve = 0

    while time.time() < deadline:
        t += 1
        a_rev = a[::-1]

        # gradient of smoothed max
        grad, cur_max = smoothed_gradient_and_max(a, a_rev, tau_factor)

        # Adam step
        a, m, v = adam_update(a, grad, m, v, t, lr)

        # enforce non‑negativity and L1‑norm
        a = project_onto_simplex(np.maximum(a, 0.0), target_sum)

        # annealing schedules
        if t % 500 == 0:
            lr *= 0.96
            tau_factor = max(1e-5, tau_factor * 0.965)

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.008:
            jitter = np.random.randn(n) * (target_sum / n) * 0.0015
            a = project_onto_simplex(np.maximum(a + jitter, 0.0), target_sum)

        # optional larger random pair transfer after long stagnation
        if no_improve >= 2500:
            a = random_pair_transfer(a, target_sum, scale=0.018)
            no_improve = 0
            print(f"[{time.time():.2f}] large transfer applied after stagnation")

        # evaluate current objective (equals cur_max because sum == target_sum)
        cur_score = cur_max

        if cur_score + 1e-12 < best_local_score:
            best_local_score = cur_score
            best_local_seq = a.copy()
            no_improve = 0
            # also update global best if improved
            if cur_score < global_best_score:
                global_best_score = cur_score
                global_best_seq = a.copy()
                print(f"[{time.time():.2f}]  NEW GLOBAL BEST {global_best_score:.6f}")
        else:
            no_improve += 1

    return best_local_seq, best_local_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises the
    `evaluate_sequence` objective.  The routine uses several random
    restarts of a projected Adam optimiser with a τ‑smoothed max
    objective, injects occasional jitter and a final polishing step.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.25   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        baseline = np.array(base, dtype=np.float64)
    else:
        # choose a length similar to the previous best run (≈8‑k)
        n = np.random.randint(7500, 8800)
        baseline = None

    target_sum = np.sqrt(2.0 * n)   # normalisation that makes the objective = max(conv)

    # ------------------------------------------------------------------
    # Build a small pool of start points
    # ------------------------------------------------------------------
    starts = []

    # 1) supplied baseline (if any)
    if baseline is not None:
        starts.append(baseline)

    # 2) uniform distribution
    starts.append(np.full(n, target_sum / n))

    # 3) a few random Dirichlet draws
    for _ in range(3):
        rnd = np.random.exponential(scale=1.0, size=n)
        rnd = rnd / rnd.sum() * target_sum
        starts.append(rnd)

    # 4) random U‑shapes (linear / quadratic / exponential) with random parameters
    shapes = ("linear", "quadratic", "exponential")
    for kind in shapes:
        for _ in range(2):
            p = np.random.uniform(0.0, 5.0)
            starts.append(generate_U_shape(n, target_sum, kind=kind, param=p))

    # ------------------------------------------------------------------
    # Global best bookkeeping
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Run optimisation for each start, respecting the overall time budget.
    # ------------------------------------------------------------------
    start_idx = 0
    while time.time() < deadline and start_idx < len(starts):
        remaining = deadline - time.time()
        # allocate at most 30 s per start, but shrink if time is low
        run_time = min(30.0, remaining)
        run_deadline = time.time() + run_time

        init_vec = starts[start_idx]
        # initialise best_seq / best_score if still empty
        if best_seq is None:
            # evaluate the initial vector directly to have a sensible global bound
            try:
                init_score = evaluate_sequence(init_vec.tolist())
                if init_score < best_score:
                    best_score = init_score
                    best_seq = init_vec.copy()
                    print(f"[{time.time():.2f}] Initial candidate score {best_score:.6f}")
            except Exception:
                pass

        # run the optimiser for this start
        best_local_seq, best_local_score = run_optimizer(
            init_vec,
            target_sum,
            run_deadline,
            best_score,
            best_seq if best_seq is not None else init_vec,
        )

        # update global best if the run discovered a better vector
        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local_seq.copy()
            print(f"[{time.time():.2f}] End of run #{start_idx} improved global to {best_score:.6f}")

        start_idx += 1

    # ------------------------------------------------------------------
    # Final polishing: short burst of tiny random pair transfers
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Polishing phase – current best {best_score:.6f}")
    if best_seq is None:
        # fallback – should not happen
        best_seq = np.full(n, target_sum / n)

    while time.time() < deadline:
        cand = random_pair_transfer(best_seq.copy(), target_sum, scale=0.006)
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] Polishing improved to {best_score:.6f}")

    print(f"[{time.time():.2f}] Finished – best score {best_score:.6f}")
    return best_seq.tolist()
