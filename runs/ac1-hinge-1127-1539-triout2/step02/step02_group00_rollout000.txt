def _simulated_annealing(a, deadline, start_temp=0.01, cooling_rate=0.9999):
    best = a.copy()
    best_score = _score(a)
    cur = a.copy()
    cur_score = best_score
    T = start_temp
    rng = np.random.default_rng()
    n = a.shape[0]
    while time.time() < deadline:
        # propose move
        i = rng.integers(0, n)
        if cur[i] <= 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        # choose delta fraction
        frac = rng.random() * 0.2   # up to 20% of a[i]
        delta = cur[i] * frac
        # apply
        cur[i] -= delta
        cur[j] += delta
        # compute score
        new_score = _score(cur)
        dE = new_score - cur_score
        accept = False
        if dE <= 0:
            accept = True
        else:
            if T > 0:
                accept = rng.random() < np.exp(-dE / T)
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
                print(f"[{time.time():.1f}] SA improved: {best_score:.6f}")
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta
        # cooling
        T *= cooling_rate
        # periodically add gaussian jitter to escape stagnation
        if rng.random() < 0.001:
            jitter = rng.normal(scale=cur.sum() * 0.001, size=n)
            cur = cur + jitter
            cur = np.maximum(cur, 0.0)
            cur = _project_to_simplex(cur, cur.sum())
            cur_score = _score(cur)
            if cur_score < best_score:
                best_score = cur_score
                best = cur.copy()
```

"""
Propose a step‑function (non‑negative sequence) that minimises
evaluate_sequence(...).  The objective reduces, after ℓ¹‑normalisation,
to the maximum entry of the self‑convolution a*a.  The algorithm
combines

* a fast random‑parameter search (power‑law, exponential, mixed
  families and Dirichlet draws),
* a projected gradient descent on a smooth log‑sum‑exp approximation of
  the maximum (annealed β, momentum, adaptive step size),
* a simulated‑annealing refinement that performs random mass transfers
  while accepting occasional uphill moves.

The whole search runs for at most ``budget_s`` seconds (default 1000) on a
single CPU.  The best sequence found is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """Objective = max(conv(a,a)) after ℓ¹‑normalisation."""
    # a is already projected onto the simplex, thus sum(a)=sqrt(2·n)
    max_b = _max_convolution_fft(a)
    return max_b


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    { x | x_i ≥ 0, Σ x_i = s }.
    (Duchi et al. 2008)
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp approximation of the maximum convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Return (g, b) where
        b = a*a (full convolution, length 2·n‑1)
        g = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a ⋆ p)  with p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]

    # ----- b = a*a ----------------------------------------------------
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ----- soft‑max probabilities p ------------------------------------
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))               # stable
    p = exps / np.sum(exps)

    # ----- gradient g = 2·(a ⋆ p) --------------------------------------
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << ((conv_len).bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Projected gradient descent on the smoothed maximum
# ----------------------------------------------------------------------


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Annealed projected gradient descent on the log‑sum‑exp smooth
    approximation of the maximum convolution.  Uses momentum and an
    adaptive step size.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    max_beta = 1024.0
    step_base = target_sum * 0.18          # base magnitude for step
    momentum = np.zeros_like(a)
    rho = 0.9                               # momentum decay

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        # projection back onto simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # annealing schedule for β
        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 1.4, max_beta)

        it += 1
    return a


# ----------------------------------------------------------------------
# Simulated annealing refinement
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simple SA that moves a random fraction of mass between two indices.
    Accepts uphill moves with probability exp(-Δ/T).  Temperature decays
    multiplicatively.  Returns the best sequence seen.
    """
    a = a0.copy()
    best = a.copy()
    cur_score = _score(a)
    best_score = cur_score

    # Initialise temperature as a small fraction of current score
    T = cur_score * 0.02 + 1e-6
    cooling = 0.9996

    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        # pick distinct indices
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # transfer a random fraction of mass from i to j
        frac = rng.random() * 0.15          # up to 15 % of a[i]
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        new_score = _score(a)
        dE = new_score - cur_score
        accept = False
        if dE <= 0.0:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-dE / T):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA improved: {best_score:.6f}")
        else:
            # revert move
            a[j] -= delta
            a[i] += delta

        # geometric cooling
        T *= cooling

        # occasional global jitter to shake out of deep minima
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _score(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA jitter improved: {best_score:.6f}")

    return best


# ----------------------------------------------------------------------
# Random initialiser (several families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Generate a random non‑negative sequence on the ℓ¹‑simplex."""
    target_sum = np.sqrt(2.0 * n)
    choice = rng.integers(0, 5)
    idx = np.arange(n, dtype=float)

    if choice == 0:                     # pure power‑law
        p = rng.uniform(0.15, 1.0)
        a = (idx + 1.0) ** (-p)

    elif choice == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif choice == 2:                   # mixed power‑law + exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif choice == 3:                   # Dirichlet sample (uniform simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target_sum

    else:                               # uniform plus jitter
        a = np.full(n, target_sum / n)

    # add a little Gaussian noise and re‑project
    a += rng.normal(scale=target_sum * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target_sum)
    return a

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative sequence) that minimises the
    competition objective.  The routine runs for at most ``budget_s``
    seconds, trying several vector lengths, random restarts, a short
    projected‑gradient phase and a simulated‑annealing refinement.
    The best sequence found is returned as a plain Python list.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # small safety margin before the hard deadline
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # If a previous construction is available, use it as a starting point
    # ------------------------------------------------------------------
    init_from_global = globals().get("height_sequence_1")
    if isinstance(init_from_global, (list, tuple, np.ndarray)):
        a = np.asarray(init_from_global, dtype=float)
        n0 = a.size
        target0 = np.sqrt(2.0 * n0)
        a = _project_to_simplex(a, target0)
        score0 = _score(a)
        best_seq = a.copy()
        best_score = score0
        print(f"[{time.time():.1f}] Starting from global height_sequence_1 (n={n0}) "
              f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Choose a set of lengths to explore – larger n gives more flexibility
    # ------------------------------------------------------------------
    lengths = [1500, 2500, 3500, 4500, 6000, 8000]

    # keep a small pool of promising candidates for deeper refinement
    top_candidates = []

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)

        # ------------------------------------------------------------------
        # Phase A – random sampling on the simplex
        # ------------------------------------------------------------------
        # allocate a fixed amount of time for the random sampling phase
        phase_end = min(deadline, time.time() + 12.0)
        while time.time() < phase_end:
            a0 = _random_initial_sequence(n, rng)
            sc = _score(a0)
            if sc < best_score:
                best_score = sc
                best_seq = a0.copy()
                print(f"[{time.time():.1f}] New global best (rand) n={n} score={best_score:.6f}")

            # keep a few of the best random seeds for later refinement
            if len(top_candidates) < 5 or sc < top_candidates[-1][0]:
                top_candidates.append((sc, a0.copy()))
                top_candidates.sort(key=lambda x: x[0])
                top_candidates = top_candidates[:5]

        # ------------------------------------------------------------------
        # Phase B – gradient descent on the best seeds for this length
        # ------------------------------------------------------------------
        for _, a_seed in top_candidates:
            if time.time() >= deadline:
                break
            # reserve at most 20 s for this refinement (gradient + SA)
            sub_dead = min(deadline, time.time() + 20.0)

            # gradient descent
            a_gd = _gradient_descent_opt(a_seed, sub_dead)

            # simulated annealing refinement (uses whatever time is left)
            a_sa = _simulated_annealing(a_gd, deadline)

            sc = _score(a_sa)
            if sc < best_score:
                best_score = sc
                best_seq = a_sa.copy()
                print(f"[{time.time():.1f}] NEW GLOBAL BEST (refined) n={n} "
                      f"score={best_score:.6f}")

        # clear the temporary pool to focus on next length
        top_candidates.clear()

    # ------------------------------------------------------------------
    # Fallback: uniform sequence (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        a = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)
        best_seq = a
        best_score = _score(a)

    # Trim trailing zeros (tiny values) for neatness
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    print(f"[{time.time():.1f}] Final best score = {best_score:.6f} "
          f"length = {best_seq.size}")
    return best_seq.tolist()
