"""
Improved search for a low‑score step‑function.

The objective is 2·n·max(conv(a,a)) / (sum(a)**2).  This is
scale‑invariant, so we fix sum(a)=sqrt(2·n) (the “L1‑norm” used by the
reference implementation) and minimise the maximal entry of the
self‑convolution.

Approach
--------
* Optimise the probability vector p = a / sum(a) on the simplex.
* At each iteration we compute the full convolution via FFT, locate the
  shift(s) attaining the current maximum and form a (sub‑)gradient of the
  max‑convolution w.r.t. a.
* An Adam‑style multiplicative update (mirror‑descent with an adaptive
  learning‑rate) is applied, followed by a projection onto the simplex
  (renormalisation to the required L1‑norm).
* Multiple random restarts with different lengths (2000 … 20000) are run
  until the time budget (≈1000 s) is exhausted.
* After each improvement of the surrogate max‐convolution the expensive
  ``evaluate_sequence`` is called and the best external score is kept.
* A cheap post‑processing step re‑orders the final vector in an
  interleaved “high‑low‑high‑low …” pattern, which often reduces the
  max‑convolution further.  Random cyclic shifts are also tried at the
  very end.

The method is intentionally lightweight (FFT‑based convolution, NumPy
vectorised operations) and fits comfortably in the 1 k‑second budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (FFT convolution, gradient from selected shifts)
# ----------------------------------------------------------------------


def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """
    Compute the full linear self‑convolution a*a using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    # next power of two for an efficient FFT
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _gradient_from_shifts(a: np.ndarray, shifts: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution with respect to ``a``.
    For each shift k the contribution to g[j] is
        2·a[k‑j]  (when the index lies inside the vector).
    The gradient is summed over all supplied shift indices.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    idx = np.arange(n)
    for k in shifts:
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g[valid] += 2.0 * a[partner[valid]]
    return g


def _interleaved_reorder(a: np.ndarray) -> np.ndarray:
    """
    Place the values in descending order and interleave them from the
    left and right ends of a new array (high‑low‑high‑low …).  The sum
    of the vector is unchanged.
    """
    order = np.argsort(-a)          # descending indices
    sorted_vals = a[order]
    n = a.shape[0]
    out = np.empty_like(a)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if (i & 1) == 0:            # even → left side
            out[left] = v
            left += 1
        else:                       # odd → right side
            out[right] = v
            right -= 1
    return out


# ----------------------------------------------------------------------
# Optimiser using Adam‑style multiplicative updates
# ----------------------------------------------------------------------


def _run_optimize_adam(a0: np.ndarray,
                       n: int,
                       deadline: float,
                       max_iters: int = 20000,
                       eta: float = 0.5,
                       beta1: float = 0.9,
                       beta2: float = 0.999,
                       eps: float = 1e-8,
                       top_k: int = 1,
                       jitter_prob: float = 0.02,
                       jitter_scale: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Adam‑style mirror descent on the probability simplex.
    Returns the best vector seen (scaled to L1 = sqrt(2·n)) and its
    external score from ``evaluate_sequence``.
    """
    a = a0.copy()
    S = np.sqrt(2.0 * n)                # required L1‑norm
    # initialise momentum buffers
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_surrogate = float('inf')
    best_score = float('inf')
    best_seq = a.copy()

    # keep the external score of the start point
    try:
        cur_score = evaluate_sequence(a.tolist())
        best_score = cur_score
        best_seq = a.copy()
    except Exception:
        pass

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # full convolution via FFT
        conv = _conv_full_fft(a)
        cur_max = float(conv.max())
        # indices of the K largest convolution entries
        if top_k >= conv.size:
            shifts = np.arange(conv.size)
        else:
            shifts = np.argpartition(conv, -top_k)[-top_k:]

        # sub‑gradient (summed over chosen shifts)
        g = _gradient_from_shifts(a, shifts)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # multiplicative Adam update
        a = a * np.exp(-eta * m_hat / (np.sqrt(v_hat) + eps))

        # renormalise to keep Σa = S
        total = a.sum()
        if total == 0.0:
            a.fill(S / n)
        else:
            a *= (S / total)

        # occasional jitter to escape plateaus
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * jitter_scale
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()

        # if the surrogate improves, query the expensive evaluator
        if cur_max < best_surrogate - 1e-12:
            best_surrogate = cur_max
            try:
                ext = evaluate_sequence(a.tolist())
                if ext < best_score:
                    best_score = ext
                    best_seq = a.copy()
                    print(f"[local improve] time={time.time():.1f}s "
                          f"score={best_score:.6f} n={n} iter={it} max={cur_max:.6f}")
            except Exception:
                pass

    return best_seq, best_score
