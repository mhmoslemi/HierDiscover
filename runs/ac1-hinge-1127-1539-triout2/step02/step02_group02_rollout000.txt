def _local_hill_climb(a, n, deadline, rng, max_iters=20000):
   best_max = conv_max(a)
   best_a = a.copy()
   temp = 0.1 * best_max # initial temperature maybe 10% of best_max
   it = 0
   while time.time() < deadline and it < max_iters:
       it += 1
       # Propose move:
       i = rng.integers(n)
       if a[i] < 1e-12:
           continue
       j = rng.integers(n)
       if i == j:
           continue
       # amount
       delta = a[i] * rng.uniform(0.05, 0.2)  # move fraction up to 20%

       a_new = a.copy()
       a_new[i] -= delta
       a_new[j] += delta

       if cur_max < best_max:
           best_max = cur_max
           best_a = a_new.copy()
           a = a_new
           # reduce temperature maybe
           temp *= 0.999  # slowly cool
           print(f"[hill climb improve] iter={it} max={best_max:.6f}")

"""
Propose a candidate step‑function (list of non‑negative heights) that minimises
the evaluation function

    2·n·max(a*a) / (sum(a)^2)

The algorithm combines three ideas:

1. **Multiplicative (exponentiated) gradient descent** on the smooth surrogate
   "max of the self‑convolution".  At each iteration we compute the full linear
   convolution via an FFT, pick the K largest entries (K≈10) and form a
   sub‑gradient of the maximum.  An exponential update with momentum keeps the
   vector non‑negative; after each step we renormalise it to the required L1‑norm
   √(2·n) (this makes the external score equal to the current maximum entry).

2. **Random restarts** with Dirichlet initialisations of varying concentration.
   Short, medium and long lengths are tried (500 … 8000).  Hyper‑parameters such
   as learning‐rate and K are varied between runs.

3. **Local stochastic hill‑climbing / simulated annealing** started from the
   best vector found so far.  A small amount of mass is moved from one index to
   another; the move is accepted if it improves the maximum convolution or
   with a probability that depends on a temperature that slowly cools.

Both phases run in parallel (one after the other) until the time budget is
exhausted.  The best sequence discovered is returned.  All operations respect
the non‑negativity and clipping constraints of `evaluate_sequence`; the final
score is printed for verification.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def conv_max(a: np.ndarray) -> float:
    """
    Return the maximum entry of the linear self‑convolution a*a.
    Computed with an FFT for O(n log n) speed.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()   # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return float(np.max(conv))


def _gradient_from_shifts(a: np.ndarray, shifts: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the (hard) max‑convolution with respect to ``a``,
    summed over the supplied shift indices.
    For a shift k the contribution to g_j is 2·a_{k−j} whenever the index is valid.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    idx = np.arange(n)
    for k in shifts:
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g[valid] += 2.0 * a[partner[valid]]
    return g

# ----------------------------------------------------------------------

                             eta0: float = 0.6,
                             beta: float = 0.9,
                             decay: float = 0.99993,
                             jitter_prob: float = 0.015,
                             max_iters: int = 30000) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated) gradient descent with momentum.
    Returns the best vector found and its *max‑convolution* value
    (which equals the external score when L1‑norm = √(2·n)).
    """
    a = a0.copy()
    S = np.sqrt(2.0 * n)

    # current best
    cur_best = conv_max(a)
    best_a = a.copy()

    v = np.zeros_like(a)          # momentum buffer
    eta = eta0
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # full convolution and its maximum
        conv, cur_max = None, None
        conv_len = 2 * n - 1
        # FFT length (recomputed only when n changes, cheap)
        fft_len = 1 << (conv_len - 1).bit_length()
        Fa = np.fft.rfft(a, n=fft_len)
        conv_full = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
        cur_max = float(np.max(conv_full))

        # choose K shifts with the largest convolution values
        if conv_full.size <= top_k:
            shifts = np.arange(conv_full.size)
        else:
            # np.argpartition gives unsorted top‑K indices efficiently
            shifts = np.argpartition(conv_full, -top_k)[-top_k:]

        # sub‑gradient from those shifts
        g = _gradient_from_shifts(a, shifts)

        # momentum step
        v = beta * v + (1.0 - beta) * g

        # exponentiated update (guarantees non‑negativity)
        a = a * np.exp(-eta * v)

        # renormalise to keep Σa = √(2·n)
        total = a.sum()
        if total == 0.0:
            a.fill(S / n)
        else:
            a *= (S / total)

        # occasional random jitter to escape flat regions
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * 0.01
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()

        # learning‑rate decay
        eta *= decay

        # record improvement
        if cur_max < cur_best - 1e-12:
            cur_best = cur_max
            best_a = a.copy()
            print(f"[exp descent improve] time={time.time():.1f}s "
                  f"n={n} iter={it} max={cur_max:.6f}")

    return best_a, cur_best


# ----------------------------------------------------------------------
# Local stochastic hill‑climbing / simulated annealing
# ----------------------------------------------------------------------


def _local_hill_climb(a_start: np.ndarray,
                      n: int,
                      deadline: float,
                      rng: np.random.Generator,
                      max_iters: int = 25000) -> tuple[np.ndarray, float]:
    """
    Perform a simple simulated‑annealing style hill‑climbing around ``a_start``.
    Moves a random fraction of mass from one index to another and accepts
    worsening moves with a temperature that decays.
    Returns the best vector encountered and its max‑convolution value.
    """
    a = a_start.copy()
    cur_max = conv_max(a)
    best_a = a.copy()
    best_max = cur_max

    # initialise temperature as a fraction of the current max
    temperature = max(0.05 * cur_max, 1e-4)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # pick donor and recipient
        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(n)
        if i == j:
            continue

        # move a random fraction (5‑20%) of the donor's mass
        frac = rng.uniform(0.05, 0.20)
        delta = a[i] * frac

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # compute new max
        new_max = conv_max(a_new)

        # accept if better, otherwise with probability exp(-Δ/temperature)
        if new_max < cur_max - 1e-12:
            a = a_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a_new.copy()
                print(f"[hill climb improve] time={time.time():.1f}s "
                      f"n={n} iter={it} max={new_max:.6f}")
        else:
            prob = np.exp(-(new_max - cur_max) / max(temperature, 1e-12))
            if rng.random() < prob:
                a = a_new
                cur_max = new_max

        # geometric cooling
        temperature *= 0.9995

        # occasional random jitter to diversify
        if rng.random() < 0.001:
            noise = rng.normal(scale=a.mean() * 0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a *= np.sqrt(2.0 * n) / a.sum()
            cur_max = conv_max(a)
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[jitter] time={time.time():.1f}s n={n} iter={it} max={cur_max:.6f}")

    return best_a, best_max

# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that yields a low score.
    The routine respects the 1000 s budget, runs many independent
    restarts of exponentiated gradient descent and finishes with a
    local hill‑climbing refinement.
    Returns a Python list of heights.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 3.0   # safety margin

    best_seq = None          # best numpy vector found so far
    best_score = float("inf")  # its external score (max‑conv)

    # ------------------------------------------------------------------
    # If a construction already exists in the global namespace, record it.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            if cand.ndim == 1 and cand.size > 0:
                sc = evaluate_sequence(cand.tolist())
                best_seq = cand.copy()
                best_score = sc
                print(f"[baseline] existing height_sequence_1 score = {sc:.6f} "
                      f"length = {cand.size}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Parameter space
    # ------------------------------------------------------------------
    lengths = [500, 1000, 2000, 4000, 8000]   # lengths we will try
    run_id = 0

    # Main search loop
    while time.time() < deadline:
        # pick a length randomly, favouring larger ones a bit
        n = int(rng.choice(lengths, p=np.array([0.2, 0.2, 0.25, 0.25, 0.1])))

        # --------------------------------------------------------------
        # Phase 1: multiplicative gradient descent (most of the time)
        # --------------------------------------------------------------
        if rng.random() < 0.68:
            # Dirichlet concentration parameter (controls sparsity)
            conc = rng.uniform(0.05, 1.5)
            a0 = rng.dirichlet(np.full(n, conc))
            a0 = a0 * np.sqrt(2.0 * n)          # enforce Σa = √(2·n)

            a_opt, max_conv = _run_exponential_descent(
                a0, n, deadline,
                top_k=rng.integers(8, 14),         # K varies between 8 and 13
                eta0=rng.uniform(0.4, 0.8),
                beta=0.9,
                decay=0.9999,
                jitter_prob=0.02,
                max_iters=25000
            )
            score = max_conv                     # because Σa = √(2·n)

        # --------------------------------------------------------------
        # Phase 2: local hill‑climbing starting from the current best
        # --------------------------------------------------------------
        else:
            # Use the best vector found so far if it matches the length;
            # otherwise fall back to a fresh random start.
            if best_seq is not None and best_seq.size == n:
                start_vec = best_seq.copy()
            else:
                start_vec = rng.dirichlet(np.full(n, 1.0))
                start_vec = start_vec * np.sqrt(2.0 * n)

            a_opt, max_conv = _local_hill_climb(
                start_vec, n, deadline, rng,
                max_iters=20000
            )
            score = max_conv

        # --------------------------------------------------------------
        # Record global improvement
        # --------------------------------------------------------------
        if score < best_score - 1e-12:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[global improve] run={run_id} n={n} score={best_score:.6f}")

        run_id += 1

    # ------------------------------------------------------------------
    # Fallback in the very unlikely event we never produced a sequence.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        a = np.full(n, np.sqrt(2.0 * n) / n)
        best_seq = a
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final verification (the platform will also evaluate)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] best score = {final_score:.6f}  length = {len(best_seq)}")
    return best_seq.tolist()
