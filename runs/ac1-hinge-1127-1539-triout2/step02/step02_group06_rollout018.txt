def generate_shape(n, target_sum, kind='linear', param=0.0, direction='symmetric'):
    idx = np.arange(n, dtype=np.float64)
    # compute base as in earlier
    if direction == 'symmetric':
        mid = (n - 1) / 2.0
        d = np.abs(idx - mid) / mid
        base = d
    elif direction == 'decreasing':
        base = idx / (n - 1)   # 0..1
    elif direction == 'increasing':
        base = (n - 1 - idx) / (n - 1)   # reversed decreasing
    else:
        raise ...

    if kind == 'linear':
        raw = 1.0 + param * base
    elif kind == 'quadratic':
        raw = 1.0 + param * base**2
    elif kind == 'exponential':
        raw = np.exp(param * base)
    elif kind == 'logistic':
        # logistic decreasing from 1 to 0 as base goes 0->1
        raw = 1.0 / (1.0 + np.exp(param * (base - 0.5)))  # shift
    elif kind == 'cosine':
        # cosine variation, amplitude controlled by param
        raw = 1.0 + param * np.cos(np.pi * idx / (n - 1))
    else:
        raise ...

    raw = np.clip(raw, 0.0, None)
    # ensure not all zero
    if raw.max() == 0:
        raw[0] = 1.0
    return project_to_sum(raw, target_sum)
```

"""
Improved search for the minimal self‑convolution peak.

The algorithm consists of three stages:

1. **Broad parametric sweep** – a large collection of analytic families
   (linear / quadratic / exponential / logistic / cosine, both symmetric
   and monotone) are instantiated, scaled to the required ℓ₁‑norm and the
   best one is retained.  Random sparse starts are also examined.

2. **Adam‑based smooth‑max descent** – the soft‑maximum of the convolution
   is approximated by a soft‑max (log‑sum‑exp).  Its gradient can be written
   as a cross‑correlation and is computed with an FFT.  An Adam optimiser
   (adaptive step‑size, momentum) is used, the iterate is projected onto the
   probability simplex (non‑negativity + prescribed ℓ₁‑norm) after each
   step and a temperature schedule slowly tightens the soft‑max.

3. **Escapes & polishing** – when progress stalls the iterate is jittered
   (small Gaussian noise) and occasional random pairwise mass transfers are
   performed; a final short random‑pair‑transfer loop tries to improve the
   best solution found.

All timing respects a 1000 s wall‑clock budget; the function always returns
the best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures / lambdas)
# ----------------------------------------------------------------------


def fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors a and b using an FFT of power‑of‑2 length."""
    n, m = a.shape[0], b.shape[0]
    L = 1 << ((n + m - 1).bit_length())           # next power of two
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)
    return conv[: n + m - 1]


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (length 2·len(a)‑1)."""
    return fft_convolve(a, a)


def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector so that its ℓ₁‑norm equals *target*."""
    s = v.sum()
    if s > 0.0:
        return v * (target / s)
    # fallback – uniform distribution (should never happen)
    return np.full_like(v, target / v.shape[0])


def generate_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
    direction: str = "symmetric",
) -> np.ndarray:
    """
    Produce a deterministic shape of length *n* and scale it to *target_sum*.

    ``kind`` – {"linear","quadratic","exponential","logistic","cosine"}.
    ``direction`` – {"symmetric","decreasing","increasing"}.
    ``param`` controls the strength of the gradient (≥0).
    """
    idx = np.arange(n, dtype=np.float64)

    if direction == "symmetric":
        mid = (n - 1) / 2.0
        base = np.abs(idx - mid) / mid                     # ∈[0,1]
    elif direction == "decreasing":
        base = idx / (n - 1)                               # ∈[0,1]
    elif direction == "increasing":
        base = (n - 1 - idx) / (n - 1)                     # ∈[0,1] (reverse of decreasing)
    else:
        raise ValueError("unknown direction")

    if kind == "linear":
        raw = 1.0 + param * base
    elif kind == "quadratic":
        raw = 1.0 + param * base ** 2
    elif kind == "exponential":
        raw = np.exp(param * base)
    elif kind == "logistic":
        # logistic decreasing from 1 → 0 as base goes 0 → 1
        raw = 1.0 / (1.0 + np.exp(param * (base - 0.5)))
    elif kind == "cosine":
        # cosine wiggle; param controls amplitude
        raw = 1.0 + param * np.cos(np.pi * idx / (n - 1))
    else:
        raise ValueError("unknown kind")

    raw = np.clip(raw, 0.0, None)
    # guard against the degenerate all‑zero case
    if raw.max() == 0.0:
        raw[0] = 1.0
    return project_to_sum(raw, target_sum)


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """Transfer a tiny fraction of mass from a random source index to a random target."""
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0.0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_to_sum(a, target_sum)


def jitter(a: np.ndarray, target_sum: float, sigma: float = 1e-3) -> np.ndarray:
    """Add Gaussian noise (clipped) and renormalise."""
    a = a + np.random.randn(a.shape[0]) * sigma
    a = np.clip(a, 0.0, None)
    return project_to_sum(a, target_sum)


def compute_score(a: np.ndarray) -> float:
    """
    Compute the same quantity that ``evaluate_sequence`` returns,
    but much faster (uses FFT‑based convolution).
    """
    n = a.shape[0]
    conv = conv_self_fft(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Adam‑based smooth‑max gradient descent
# ----------------------------------------------------------------------


def smooth_max_gradient(a: np.ndarray, tau: float) -> np.ndarray:
    """
    Gradient of the soft‑max of the self‑convolution.
    Returns a vector of length ``len(a)``.
    """
    n = a.shape[0]
    conv = conv_self_fft(a)               # length 2n‑1

    # soft‑max weights
    max_c = conv.max()
    exp_arg = (conv - max_c) / tau
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:                       # numerical fallback
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    # gradient = 2 * (w ⋆ a_rev)
    a_rev = a[::-1]
    grad_full = fft_convolve(w, a_rev)    # length 3n‑2
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Choose a length – reuse a known construction if it exists
    # ------------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
    else:
        n = np.random.randint(7200, 8800)

    target_sum = np.sqrt(2.0 * n)               # normalisation of the problem

    # ------------------------------------------------------------------
    # Phase 1 – extensive parametric sweep (many analytic families)
    # ------------------------------------------------------------------
    best_seq = np.full(n, target_sum / n)        # uniform baseline
    best_score = compute_score(best_seq)

    families = [
        ("symmetric", "linear"),
        ("symmetric", "quadratic"),
        ("symmetric", "exponential"),
        ("symmetric", "cosine"),
        ("decreasing", "linear"),
        ("decreasing", "quadratic"),
        ("decreasing", "exponential"),
        ("decreasing", "logistic"),
        ("increasing", "linear"),
        ("increasing", "quadratic"),
        ("increasing", "exponential"),
        ("increasing", "logistic"),
    ]

    param_grids = {
        "linear": np.linspace(0.0, 5.0, 26),          # 0 … 5  step 0.2
        "quadratic": np.linspace(0.0, 4.0, 21),      # 0 … 4  step 0.2
        "exponential": np.linspace(0.0, 3.0, 16),    # 0 … 3  step 0.2
        "logistic": np.linspace(0.5, 8.0, 15),      # 0.5 … 8
        "cosine": np.linspace(0.0, 5.0, 21),        # 0 … 5
    }

    print(f"[{time.time():.2f}] Starting parametric sweep (n={n})")
    for direction, kind in families:
        for p in param_grids[kind]:
            cand = generate_shape(n, target_sum, kind=kind, param=p, direction=direction)
            score = compute_score(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"  New best {direction}/{kind} (p={p:.2f}) → {best_score:.6f}")
            if time.time() >= deadline:
                print("Time exhausted during sweep.")
                return best_seq.tolist()

    # a few random sparse initialisations (helps escaping the dense families)
    for _ in range(40):
        # choose a random support size between 5 % and 30 % of n
        support = np.random.randint(int(0.05 * n), int(0.30 * n))
        idx = np.random.choice(n, support, replace=False)
        vec = np.zeros(n, dtype=np.float64)
        vec[idx] = np.random.rand(support)
        vec = project_to_sum(vec, target_sum)
        score = compute_score(vec)
        if score < best_score:
            best_score = score
            best_seq = vec.copy()
            print(f"  Random sparse init improved → {best_score:.6f}")
        if time.time() >= deadline:
            return best_seq.tolist()

    # ------------------------------------------------------------------
    # Phase 2 – Adam‑based smooth‑max descent
    # ------------------------------------------------------------------
    a = best_seq.copy()
    m = np.zeros_like(a)            # first‑moment buffer
    v = np.zeros_like(a)            # second‑moment buffer
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t_iter = 0
    no_improve = 0

    # hyper‑parameters
    avg_entry = target_sum / n
    lr = 0.5 * avg_entry            # initial step size (≈ half the mean entry)
    # temperature – start fairly smooth, then tighten
    tau = max(conv_self_fft(a)) * 0.4
    min_tau = 1e-5

    print(f"[{time.time():.2f}] Entering Adam descent (lr={lr:.5e}, tau={tau:.5e})")
    while time.time() < deadline:
        t_iter += 1
        # gradient of the soft‑max
        grad = smooth_max_gradient(a, tau)

        # Adam update (minimisation → move opposite to gradient)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = np.clip(a, 0.0, None)
        a = project_to_sum(a, target_sum)

        score = compute_score(a)

        if score + 1e-12 < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"Iter {t_iter:06d} – improvement: {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # decay the temperature slowly
        if t_iter % 500 == 0:
            tau = max(tau * 0.94, min_tau)

        # occasional jitter to shake the iterate out of plateaus
        if no_improve >= 3000:
            a = jitter(a, target_sum, sigma=5e-4)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"Iter {t_iter:06d} – jitter reset")

    # ------------------------------------------------------------------
    # Phase 3 – final quick random perturbations (pair transfers)
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Descent finished; best score {best_score:.6f}")

    while time.time() < deadline - 0.2:
        a = random_pair_transfer(best_seq, target_sum, scale=0.003)
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"Polish – improvement to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence discovered (plain Python list)
    # ------------------------------------------------------------------
    return best_seq.tolist()
