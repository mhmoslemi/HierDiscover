def _adam_projected_gradient_descent(a0, deadline):
    n = len(a0)
    target_sum = np.sqrt(2.0 * n)
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = target_sum * 0.05  # base learning rate, tunable

    best_a = a.copy()
    best_val = _max_conv_fft(a)

    it = 0
    beta_temp = 1.0
    max_beta = 3000.0
    beta_factor = 1.05 # increase per iteration
    while time.time() < deadline:
        # compute gradient at current beta
        g, b = _gradient_log_sum_exp(a, beta_temp)
        # Adam update
        it += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = - lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a = a + step
        # project onto simplex
        a = _project_to_simplex(a, target_sum)

        # compute score
        cur_val = np.max(b) if b is not None else _max_conv_fft(a)  # b from gradient step might be current conv before projection; but after projection conv changes a bit; we recompute.
        # Actually after projection, we need to compute new conv.
        cur_val = _max_conv_fft(a)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam iter {it}, beta {beta_temp:.2f}, score {best_val:.6f}")

        # update temperature
        beta_temp = min(beta_temp * beta_factor, max_beta)

        # optional early break if beta reached max and gradient small
    return best_a, best_val

"""
search algorithm for the “minimum maximum self‑convolution” problem.

The objective after fixing the ℓ¹‑norm to √(2 n) equals the
maximum entry of the linear convolution a*a.
We minimise this value by a two‑stage stochastic optimiser:

1.  Projected Adam optimiser on a smooth approximation of the max:
    M_β(a)= (1/β)·log Σ_k exp(β·(a*a)_k).  The gradient of M_β is
    ∇M_β = 2·(a ⋆ p) with p the soft‑max probabilities of the
    convolution entries.  The temperature β is increased
    gradually (annealing) while Adam adapts the step sizes.
    After each projection onto the simplex Σa=√(2 n) we keep the
    best vector seen.

2.  Simulated‑annealing hill‑climbing that moves a random fraction
    of mass between two positions, accepting worsening moves
    with a temperature that decays to zero.  This refines the
    solution found by Adam.

The whole routine runs for at most ``budget_s`` seconds,
trying many random restarts and a few different lengths.
The best sequence encountered is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(af * af, n=size)[: 2 * n - 1]
    return float(conv.max())


def _gradient_log_sum_exp(a: np.ndarray, beta: float):
    """
    Gradient of the smooth maximum M_β(a) = (1/β)·log Σ exp(β·(a*a))
    Returns (grad, conv) where conv = a*a (full convolution).
    """
    n = a.shape[0]

    # ---- 1. convolution a*a (full length 2n‑1) --------------------
    size1 = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=size1)
    conv = np.fft.irfft(af * af, n=size1)[: 2 * n - 1]     # shape (2n‑1,)

    # ---- 2. soft‑max probabilities --------------------------------
    max_c = conv.max()
    # shift for numerical stability
    shifted = np.exp(beta * (conv - max_c))
    p = shifted / shifted.sum()                          # p_k ≥ 0, Σ p_k = 1

    # ---- 3. gradient  g = 2·(a ⋆ p) -------------------------------
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = 1 << ((conv_len).bit_length())
    arf = np.fft.rfft(a_rev, n=size2)
    prf = np.fft.rfft(p, n=size2)
    tmp = np.fft.irfft(arf * prf, n=size2)[:conv_len]
    # central part of length n
    grad = 2.0 * tmp[n - 1 : n - 1 + n]

    return grad, conv


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max (projected onto the simplex)
# ----------------------------------------------------------------------
def _adam_projected_opt(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises the smooth maximum of the self‑convolution.
    The ℓ¹‑norm is fixed to sqrt(2·n) by projection after each step.
    """
    n = a0.shape[0]
    target_sum = np.sqrt(2.0 * n)

    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    it = 0
    beta_temp = 1.0                # temperature for the soft‑max
    max_beta = 3000.0
    beta_mul = 1.04                # multiplicative increase per 20‑it block
    lr_base = target_sum * 0.02    # base learning rate (scaled later by temperature)

    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    while time.time() < deadline:
        it += 1

        # gradient of the smooth max
        g, _ = _gradient_log_sum_exp(a, beta_temp)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning rate adapts to the current temperature
        lr = lr_base / (beta_temp + 1.0)

        # Adam step
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex (keeps non‑negativity & ℓ¹‑norm)
        a = _project_to_simplex(a, target_sum)

        # evaluate every few iterations (cheap relative to gradient)
        if it % 10 == 0:
            cur_val = _max_convolution_fft(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Adam it {it:5d} β={beta_temp:6.2f} score={best_val:.6f}")

        # anneal the temperature (slowly increase)
        if it % 20 == 0:
            beta_temp = min(beta_temp * beta_mul, max_beta)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climbing (mass‑transfer moves)
# ----------------------------------------------------------------------
def _simulated_annealing_hc(a_start: np.ndarray, deadline: float,
                           start_temp: float = 0.02,
                           max_frac: float = 0.18) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climber that moves a random fraction of mass
    from one index to another, accepting worsening moves with a
    linearly decaying temperature.
    """
    n = a_start.size
    a = a_start.copy()
    cur_val = _max_convolution_fft(a)
    best_a = a.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    total_time = max(deadline - time.time(), 0.0)
    if total_time <= 0.0:
        return best_a, best_val

    t0 = time.time()
    while time.time() < deadline:
        # annealing temperature (linear decay)
        elapsed = time.time() - t0
        frac_time = elapsed / total_time
        T = start_temp * max(0.0, 1.0 - frac_time)

        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        delta = a[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        new_val = _max_convolution_fft(a)

        if new_val < cur_val:
            # improvement – always keep
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA improved = {best_val:.6f}")
        else:
            # accept with probability exp(-(Δ)/T)
            if T > 0.0 and rng.random() < np.exp(-(new_val - cur_val) / T):
                cur_val = new_val
            else:
                # revert move
                a[j] -= delta
                a[i] += delta

    return best_a, best_val


# ----------------------------------------------------------------------
# One optimisation run for a given length
# ----------------------------------------------------------------------
def _optimise_length(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length n within the time budget ``deadline``.
    The routine splits the budget between Adam optimisation and
    simulated‑annealing hill‑climbing.
    """
    now = time.time()
    time_left = deadline - now
    if time_left < 1.0:
        return None, np.inf

    # split roughly 60 % for Adam, 40 % for hill‑climbing
    adam_deadline = now + 0.60 * time_left
    hc_deadline = deadline

    # ----- random initialisation (exponential decay + tiny noise) -----
    rng = np.random.default_rng()
    lam = rng.uniform(0.001, 0.018)          # controls decay speed
    a0 = np.exp(-lam * np.arange(n, dtype=np.float64))
    a0 += rng.normal(scale=0.001, size=n) * np.sqrt(2.0 * n)
    a0 = np.maximum(a0, 0.0)
    target_sum = np.sqrt(2.0 * n)
    a0 = _project_to_simplex(a0, target_sum)

    # ----- Adam optimisation -----
    a_opt, best_val = _adam_projected_opt(a0, adam_deadline)

    # ----- Simulated‑annealing refinement -----
    if time.time() < hc_deadline:
        a_opt, best_val = _simulated_annealing_hc(a_opt, hc_deadline,
                                                  start_temp=0.02,
                                                  max_frac=0.15)

    return a_opt, best_val

    # ------------------------------------------------------------------
    # fallback – uniform sequence (should never be used)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1500
        a = np.ones(n, dtype=np.float64)
        a *= np.sqrt(2.0 * n) / n
        best_seq = a

    # trim trailing zeros (just in case numerical rounding created them)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    return best_seq.tolist()
