    conv = conv_self_fft(a)
    max_val = conv.max()

def subgradient_max(a, max_idxs):
    n = len(a)
    g = np.zeros_like(a)
    for k in max_idxs:
        i_start = max(0, k - (n-1))
        i_end = min(k, n-1)
        sub = a[i_start:i_end+1]
        g[i_start:i_end+1] += sub[::-1]
    num = len(max_idxs)
    g /= num  # average
    return g

"""
Subgradient optimizer for the self‑convolution max problem.

The evaluation function (with sum normalised to √(2 n)) equals the
maximum entry of the linear self‑convolution a∗a.  The algorithm
searches for a non‑negative vector a of length *n* that minimises this
maximum.  It proceeds in three phases:

1. **Parametric sweep** – a collection of simple symmetric “U‑shapes’’
   (linear, quadratic, exponential and cosine) is generated, projected
   onto the ℓ₁‑simplex with Σa = √(2 n), and the best one is kept.

2. **Projected sub‑gradient descent on the max‑convolution** –  
   the sub‑gradient of the max‑convolution is the (average) gradient of
   those lag(s) attaining the maximum.  At each iteration a step is taken
   in the negative sub‑gradient direction, the vector is projected back
   onto the simplex, and then forced to be symmetric (mirrored).  An
   adaptive line‑search on the step size guarantees monotone decrease,
   while a simple schedule reduces the step size on stagnation and adds
   occasional random jitter.

3. **Polishing** – a short burst of random pair‑mass transfers is
   performed; any improvement is kept.

The whole search respects a wall‑clock budget (≈ 1000 s).  The best
sequence seen so far is returned as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a with itself using an FFT of size a power of two.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())   # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex
        {x | x_i ≥ 0, Σ_i x_i = target }.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, target / len(v))

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def generate_U_shape(n: int, target_sum: float,
                     kind: str = "linear", param: float = 0.0) -> np.ndarray:
    """
    Produce a symmetric “U‑shaped’’ sequence of length n.
    `kind` ∈ {linear, quadratic, exponential, cosine}.
    `param` controls the edge enhancement (≥0).
    The result is projected onto the ℓ₁‑simplex of total mass `target_sum`.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                      # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "cosine":
        # cos(π·d) = 1 at the edges, -1 in the centre.
        raw = 1.0 + param * np.cos(np.pi * d)
    else:
        raise ValueError(f"unknown shape kind: {kind}")

    raw = np.clip(raw, 0.0, None)                     # enforce non‑negativity
    return project_onto_simplex(raw, target_sum)


def max_convolution_info(a: np.ndarray) -> tuple[float, np.ndarray, np.ndarray]:
    """
    Compute the full self‑convolution of a, its maximum value, and the
    integer indices where that maximum is attained (within a tiny tolerance).
    """
    conv = conv_self_fft(a)
    max_val = conv.max()
    # tolerance to capture all (potentially several) maximising indices
    tol = 1e-12
    max_idxs = np.where(np.abs(conv - max_val) <= tol)[0]
    return max_val, max_idxs, conv


def subgradient_max(a: np.ndarray, max_idxs: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution at the current maximising lags.
    If several lags attain the maximum, the gradients are averaged.
    """
    n = a.shape[0]
    grad = np.zeros_like(a, dtype=np.float64)

    for k in max_idxs:
        # i ranges where both i and k-i lie in [0, n-1]
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        sub = a[i_start:i_end + 1]          # length L = i_end-i_start+1
        # derivative contribution is the reversed sub‑vector
        grad[i_start:i_end + 1] += sub[::-1]

    if len(max_idxs) > 1:
        grad /= len(max_idxs)               # average – still a valid sub‑gradient
    return grad


def random_pair_transfer(a: np.ndarray, target_sum: float,
                         scale: float = 0.01) -> np.ndarray:
    """
    Transfer a tiny amount of mass from a random index i to j,
    then re‑project onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(np.maximum(a, 0.0), target_sum)
