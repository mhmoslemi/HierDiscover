def _lp_step(f, M, rng, t):
    n = len(f)
    # Build A
    k = np.arange(2*n-1)[:,None]
    i = np.arange(n)[None,:]
    diff = k - i
    mask = (diff >= 0) & (diff < n)
    A = np.zeros((2*n-1, n), dtype=float)
    A[mask] = f[diff[mask]]
    # solve
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                 bounds=[(0, None)]*n,
                                 method='highs',
                                 options={'presolve': True})
    if not res.success:
        # fallback: return current f
        return f
    b = res.x
    sum_b = b.sum()
    if sum_b <= 0:
        return f
    S = np.sqrt(2.0 * n)
    g = b * (S / sum_b)
    # combine
    new_f = (1-t) * f + t * g
    # ensure nonneg and sum approx S
    new_f = np.maximum(new_f, 0.0)
    new_f = new_f / new_f.sum() * S
    return new_f

def propose_candidate(seed=42, budget_s=1000, **kwargs):

    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5

    best_seq = None; best_score = np.inf

    attempt = 0

    while time.time() < deadline:

        attempt += 1

        # choose n randomly (500-2500)
        n = rng.integers(600, 2500)  # gauge

        # initialize f: from previous best_seq if exists and size matches else Dirichlet random
        if best_seq is not None and len(best_seq) >= n:
            f = np.array(best_seq[:n])
            # small jitter to diversify
            f += rng.normal(scale=f.mean()*0.001, size=n)
            f = np.clip(f, 0, None)
        else:
            f = rng.random(n)

        # normalize
        S = np.sqrt(2.0 * n)
        f = f / f.sum() * S

        # allocate slice time for this run
        slice_deadline = min(time.time() + 40.0, deadline)

        # iterative LP improvement loop
        t = 0.5  # combination factor

        while time.time() < slice_deadline:
            # compute max conv
            _, cur_max, cur_idx = _conv_max(f)

            # Solve LP to maximize sum(g) subject to convolution <= cur_max
            g = _lp_optimize(f, cur_max, rng)

            if g is None:  # LP failed
                break

            # compute new combination
            f_new = (1 - t) * f + t * g
            # renorm
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum() * S

            # compute new conv max
            _, new_max, _ = _conv_max(f_new)

            if new_max < cur_max:
                f = f_new
                # maybe increase t for faster convergence? keep same
            else:
                # if not improving, maybe reduce t and try again
                t *= 0.7
                if t < 0.05:
                    break

            # occasionally jitter
            if rng.random() < 0.05:
                noise = rng.normal(scale=f.mean()*0.01, size=n)
                f = np.clip(f + noise, 0, None)
                f = f / f.sum() * S

        # After LP loop, run local optimizer to fine tune
        # allocate remaining local time
        remaining_time = min(deadline, slice_deadline) - time.time()
        if remaining_time > 5.0:

"""
Improved construction for minimizing the ∞‑norm of the self‑convolution.

The algorithm combines two ideas:

1. **Linear‑programming step** – given a current step function `f`,
   solve a linear program that maximises the L₁‑norm of a new function
   `g` under the constraints `(f * g)_k ≤ M`, where `M = max(f*f)`.
   The solution pushes mass away from positions that contribute to the
   current maximum.  After scaling `g` to the required L₁‑norm the new
   vector is obtained by a convex combination `f ← (1‑t)f + t·g`.

2. **Projected sub‑gradient refinement** – after a few LP updates a fast
   multiplicative sub‑gradient descent (with occasional random swaps
   and jitter) is used to polish the solution.

Both phases respect the non‑negativity constraints, keep the L₁‑norm
fixed to `√(2·n)` (the quantity that cancels out of the official score),
and are run repeatedly within the allotted 1000 s time budget.
The best sequence found is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top level)
# ----------------------------------------------------------------------


def _conv_max(a: np.ndarray) -> tuple[np.ndarray, float, int]:
    """
    Compute the full linear convolution `a*a` using FFT,
    returning the convolution array, its maximum value and the index of that maximum.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    max_idx = int(np.argmax(conv))
    return conv, float(conv[max_idx]), max_idx


def _gradient_step(a: np.ndarray, eta: float) -> np.ndarray:
    """
    Multiplicative sub‑gradient step that reduces the maximal entry of the
    self‑convolution while keeping the L₁‑norm equal to √(2·n).
    """
    n = a.shape[0]
    _, _, k = _conv_max(a)

    idx = np.arange(n)
    partner = k - idx
    valid = (partner >= 0) & (partner < n)

    g = np.zeros_like(a)
    g[valid] = 2.0 * a[partner[valid]]
    if k % 2 == 0:
        d = k // 2
        if 0 <= d < n:
            g[d] = a[d]

    a_new = a * np.exp(-eta * g)
    S = np.sqrt(2.0 * n)
    total = a_new.sum()
    if total == 0.0:                     # should not happen, but guard anyway
        a_new = np.full(n, S / n)
    else:
        a_new *= (S / total)
    return a_new


def _initial_sequence(base_seq: np.ndarray,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """
    Initialise a non‑negative sequence of length `n` with L₁‑norm √(2·n).
    If `base_seq` is provided and long enough its leading part is used
    (with a tiny jitter); otherwise a Dirichlet‑like random vector is drawn.
    """
    S = np.sqrt(2.0 * n)

    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        a += rng.normal(scale=a.mean() * 0.001, size=n)
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)

    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0)
        a_sum = a.sum()
    a = a / a_sum * S
    return a


def _lp_optimize(f: np.ndarray,
                 M: float,
                 rng: np.random.Generator) -> np.ndarray | None:
    """
    Solve the linear program:
        maximise   Σ_j g_j
        subject to  (f * g)_k ≤ M   for all k,
                    g_j ≥ 0.
    Returns a feasible `g` with the correct L₁‑norm, or ``None`` on failure.
    The implementation prefers SciPy’s `linprog`; if unavailable it falls back
    to `cvxpy`.
    """
    n = f.shape[0]

    # Build the convolution matrix A of shape (2n‑1, n)
    k = np.arange(2 * n - 1)[:, None]      # (2n‑1,1)
    i = np.arange(n)[None, :]              # (1,n)
    diff = k - i                           # broadcasted differences
    mask = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n), dtype=float)
    A[mask] = f[diff[mask]]

    # ---------- Try SciPy ----------
    try:
        from scipy.optimize import linprog
        c = -np.ones(n)                     # minimise -Σ g_j  → maximise Σ g_j
        b_ub = np.full(2 * n - 1, M)
        res = linprog(c,
                      A_ub=A,
                      b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
        if not res.success or np.isnan(res.fun):
            return None
        g = res.x
    except Exception:                       # SciPy not available or failed
        # ---------- Fallback to CVXPY ----------
        try:
            import cvxpy as cp
        except Exception:
            return None
        g_var = cp.Variable(n, nonneg=True)
        constraints = [A @ g_var <= M * np.ones(2 * n - 1)]
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=False, max_iters=2000)
        except Exception:
            prob.solve(solver=cp.SCS, max_iters=2000)
        if g_var.value is None:
            return None
        g = np.array(g_var.value).flatten()

    sum_g = g.sum()
    if sum_g <= 0.0:
        return None
    # Scale to the required L₁‑norm √(2·n)
    S = np.sqrt(2.0 * n)
    g = g * (S / sum_g)
    return g


def _optimize_local(a: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Local optimiser that mixes multiplicative sub‑gradient steps,
    simulated‑annealing random swaps, and occasional jitter.
    Returns the best vector seen together with its maximal convolution value.
    """
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    conv, cur_max, cur_idx = _conv_max(a)
    best_a = a.copy()
    best_score = cur_max

    # hyper‑parameters (tuned empirically)
    eta = 0.6
    decay = 0.9999
    move_prob = 0.15
    epsilon_move = 0.05
    jitter_prob = 0.01
    temp = cur_max * 0.1
    temp_decay = 0.9997

    while time.time() < deadline:
        if rng.random() < move_prob:
            # ---------- simulated‑annealing random move ----------
            partner = cur_idx - np.arange(n)
            valid = (partner >= 0) & (partner < n)
            partner_idx = np.where(valid)[0]
            non_partner_idx = np.where(~valid)[0]

            if partner_idx.size and non_partner_idx.size:
                i = rng.choice(partner_idx)
                j = rng.choice(non_partner_idx)

                delta = epsilon_move * a[i]
                delta = min(delta, a[i])

                a_candidate = a.copy()
                a_candidate[i] -= delta
                a_candidate[j] += delta

                _, cand_max, cand_idx = _conv_max(a_candidate)

                accept = False
                if cand_max <= cur_max:
                    accept = True
                else:
                    prob = np.exp(-(cand_max - cur_max) / max(temp, 1e-12))
                    if rng.random() < prob:
                        accept = True

                if accept:
                    a = a_candidate
                    cur_max = cand_max
                    cur_idx = cand_idx
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                temp *= temp_decay
        else:
            # ---------- multiplicative gradient step ----------
            a = _gradient_step(a, eta)
            _, cur_max, cur_idx = _conv_max(a)
            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()

        # occasional jitter
        if rng.random() < jitter_prob:
            noise = rng.normal(scale=a.mean() * 0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = a / a.sum() * S

        eta *= decay

    return best_a, best_score
