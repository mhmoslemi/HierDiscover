"""
Propose a step‑function with a low value of
    2 * n * max(conv(a, a)) / (sum(a)**2)

The routine works as follows

1. **Initialisation** – read the globally supplied `height_sequence_1`
   (if any) and generate a small pool of random power‑law / exponential
   candidates.  All candidates are projected onto the simplex
   `{x ≥ 0,  Σx = √(2 n)}` and the best one (lowest max‑convolution)
   becomes the starting point.

2. **Projected sub‑gradient descent** – the true objective is the
   maximum entry of the auto‑convolution.  At the current iterate we
   form a hard “soft‑max” weight vector that puts equal weight on all
   indices attaining the maximum.  The corresponding sub‑gradient is
   `2 * (w ⊛ a_rev)`, computed by FFTs.  The iterate is updated with a
   momentum term (classical heavy‑ball) and projected back onto the
   simplex.  The step‑size decays slowly; every few hundred steps a tiny
   random jitter is added to help escape plateaus.

3. **Fine‑tuning** – after the main loop a short phase with a tiny
   constant step size finishes the optimisation.

The whole search honour‑s the 1000 s time budget (a small safety margin
is kept) and returns the best sequence found.

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ *x*."""
    return 1 << (x - 1).bit_length()


def _compute_score(a: np.ndarray) -> float:
    """
    Fast version of the provided `evaluate_sequence` that assumes
    a is already non‑negative and respects the sum‑normalisation.
    Returns max(conv(a, a)).
    """
    n = a.shape[0]
    nfft = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
    return float(conv.max())


def project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of *v* onto the simplex
        { x : x_i ≥ 0 , Σ_i x_i = target_sum }.
    """
    # Implementation of Duchi et al. (ICML 2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(len(v)) + 1
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _random_powerlaw(n: int, p: float, offset: float) -> np.ndarray:
    """a_i = 1/(i+offset)^p ,  then normalised to the target sum."""
    a = 1.0 / np.power(np.arange(n, dtype=np.float64) + offset, p)
    return a


def _random_exponential(n: int, r: float) -> np.ndarray:
    """a_i = r**i (geometric), then renormalised."""
    a = np.power(r, np.arange(n, dtype=np.float64))
    return a

    # ------------------------------------------------------------------
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        n = init_seq.shape[0]
    except Exception:
        n = 5600                     # a reasonable default size

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # 2. Generate a few diverse candidates and keep the best
    # ------------------------------------------------------------------
    best_a = None
    best_score = np.inf

    # candidate: the supplied sequence (if present)
    if 'init_seq' in locals():
        a = np.maximum(init_seq, 0.0)
        s = a.sum()
        if s > 0:
            a = a * target_sum / s
        else:
            a = np.full(n, target_sum / n)
        sc = _compute_score(a)
        best_a = a.copy()
        best_score = sc

    # uniform candidate (always feasible)
    a_uni = np.full(n, target_sum / n)
    sc = _compute_score(a_uni)
    if sc < best_score:
        best_score = sc
        best_a = a_uni.copy()

    # a handful of random power‑law / exponential candidates
    for _ in range(30):
        if time.time() >= deadline:
            break
        if np.random.rand() < 0.5:
            p = np.random.uniform(0.3, 1.2)
            off = np.random.uniform(0.5, 4.0)
            a = _random_powerlaw(n, p, off)
        else:
            r = np.random.uniform(0.90, 0.999)   # very slow decay
            a = _random_exponential(n, r)
        # normalise to the required L1‑norm
        a = a * target_sum / a.sum()
        sc = _compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # fallback – in the extremely unlikely case everything failed
    if best_a is None:
        best_a = np.full(n, target_sum / n)
        best_score = _compute_score(best_a)

    # ------------------------------------------------------------------
    # 3. Pre‑compute FFT sizes used throughout the optimisation
    # ------------------------------------------------------------------
    nfft_conv = _next_pow2(2 * n - 1)      # for a * a
    nfft_corr = _next_pow2(3 * n - 2)      # for w ⊛ a_rev

    # ------------------------------------------------------------------
    # 4. Sub‑gradient descent with momentum
    # ------------------------------------------------------------------
    a = best_a.copy()                     # current iterate
    m = np.zeros_like(a)                  # momentum term

    # hyper‑parameters (tuned empirically)
    step_size = 0.30
    step_decay = 0.9997
    min_step = 0.001
    mu = 0.9                               # momentum coefficient

    best_iter = 0
    last_improve = 0
    iteration = 0

    # Tolerance used to decide which convolution entries are “active”
    atol_active = 1e-12

    while time.time() < deadline:
        iteration += 1

        # ---- convolution a * a ------------------------------------------------
        af = np.fft.rfft(a, n=nfft_conv)
        conv = np.fft.irfft(af * af, n=nfft_conv)[: 2 * n - 1]
        max_c = conv.max()

        # ---- identify active indices (hard max) -------------------------------
        active_mask = np.abs(conv - max_c) <= atol_active
        # In the very unlikely case of an empty mask (numerical issues)
        if not np.any(active_mask):
            active_mask = np.isclose(conv, max_c, atol=1e-12)
        w = active_mask.astype(np.float64)
        w /= w.sum()                       # uniform weight over the active set

        # ---- gradient of the (hard) max – correlation -------------------------
        wf = np.fft.rfft(w, n=nfft_corr)
        a_rev = a[::-1]
        af_rev = np.fft.rfft(a_rev, n=nfft_corr)
        corr_full = np.fft.irfft(wf * af_rev, n=nfft_corr)
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]

        # ---- momentum update -------------------------------------------------
        m = mu * m + (1.0 - mu) * grad

        # ---- gradient step ---------------------------------------------------
        a_new = a - step_size * m

        # ---- project back onto the simplex (non‑negative, fixed L1) ----------
        a_new = project_onto_simplex(a_new, target_sum)

        # ---- evaluate ---------------------------------------------------------
        score_new = _compute_score(a_new)

        # keep the best ever seen
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            best_iter = iteration
            last_improve = iteration
            elapsed = time.time() - start_time
            print(
                f"Iter {iteration:6d}: new best {best_score:.6f} "
                f"(elapsed {elapsed:.1f}s)"
            )

        # ---- acceptance -------------------------------------------------------
        # We always move to the new point (the momentum already provides
        # exploration).  If the step is detrimental we keep the old a
        # because the objective is non‑increasing in the momentum scheme.
        if score_new <= _compute_score(a):
            a = a_new
        else:
            # occasionally accept a worse move to escape flat regions
            if np.random.rand() < 0.001:
                a = a_new

        # ---- occasional jitter to shake the iterate --------------------------- 
        if iteration % 73 == 0:
            jitter = 1e-5 * np.random.randn(n)
            a = project_onto_simplex(a + jitter, target_sum)

        # ---- schedule updates -------------------------------------------------
        step_size = max(step_size * step_decay, min_step)

        # ---- simple restart heuristic -----------------------------------------
        if iteration - last_improve > 5000:
            # re‑initialise from a fresh random power‑law blended with the best
            p = np.random.uniform(0.3, 1.0)
            off = np.random.uniform(0.5, 3.0)
            a_rand = _random_powerlaw(n, p, off)
            a_rand = a_rand * target_sum / a_rand.sum()
            # blend a little with the current best to keep momentum meaningful
            a = 0.7 * best_a + 0.3 * a_rand
            a = project_onto_simplex(a, target_sum)
            m.fill(0.0)
            last_improve = iteration
            print(
                f"Iter {iteration:6d}: restart – mixed new candidate "
                f"(best score still {best_score:.6f})"
            )

    # ------------------------------------------------------------------
    # 5. Fine‑tuning phase – tiny constant step without momentum
    # ------------------------------------------------------------------
    fine_step = 5e-4
    fine_iters = 2000
    for _ in range(fine_iters):
        if time.time() >= deadline:
            break
        af = np.fft.rfft(a, n=nfft_conv)
        conv = np.fft.irfft(af * af, n=nfft_conv)[: 2 * n - 1]
        max_c = conv.max()

        active_mask = np.abs(conv - max_c) <= atol_active
        if not np.any(active_mask):
            active_mask = np.isclose(conv, max_c, atol=1e-12)
        w = active_mask.astype(np.float64)
        w /= w.sum()

        wf = np.fft.rfft(w, n=nfft_corr)
        a_rev = a[::-1]
        af_rev = np.fft.rfft(a_rev, n=nfft_corr)
        corr_full = np.fft.irfft(wf * af_rev, n=nfft_corr)
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]

        a = a - fine_step * grad
        a = project_onto_simplex(a, target_sum)

        score_new = _compute_score(a)
        if score_new < best_score:
            best_score = score_new
            best_a = a.copy()
            elapsed = time.time() - start_time
            print(
                f"Fine‑tune: new best {best_score:.6f} "
                f"(elapsed {elapsed:.1f}s)"
            )

    # ------------------------------------------------------------------
    # 6. Return the best sequence discovered
    # ------------------------------------------------------------------
    return best_a.tolist()
