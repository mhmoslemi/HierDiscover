def adam_optimize(a0, deadline):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta = 1.0
    max_beta = 2048.0
    best = a.copy()
    best_val = _score(a)
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        t += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        # step size: adapt relative to gradient magnitude
        lr = 0.1 / (np.sqrt(v_hat) + eps)  # elementwise? Actually we want vector step.
        # We can use global step size: lr = base / (np.linalg.norm(g) + eps)

        base = target_sum * 0.2 / (beta + 1)

        # simple using m_hat may be inaccurate
        a = a - lr_scalar * m_hat  # incorporate momentum

        val = _score(a)

"""
Propose a step‑function (non‑negative vector) that minimises
    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three stages:

1. **Fast parametric sweep** –  for a few modest lengths (≈1500‑2000)
   we scan simple families (power‑law with optional offset and
   exponential decay).  Each candidate is normalised to
   Σa = √(2·n) so that the objective coincides with max(convolution).
   The best‑scoring sequence is kept.

2. **Annealed soft‑max gradient descent** –  starting from the best
   candidate (and several random restarts) we minimise a smooth
   approximation of the max using the log‑sum‑exp (soft‑max) of the
   convolution entries.  The temperature β is increased (annealed) up
   to 2048, while a projected gradient step (Euclidean projection onto
   the simplex) is taken at each iteration.  The step size is scaled
   with 1/(β·‖∇‖) and a small amount of jitter is added periodically.

3. **Stochastic hill‑climbing** –  a cheap local search that moves a
   random fraction of mass between two positions while preserving the
   ℓ¹‑norm.  Improvements are kept; the search runs until the time
   budget expires.

The whole routine respects the given time budget (default 1000 s) and
returns the best sequence found.  All numbers are non‑negative,
automatically clipped to the interval [0,1000] by the evaluator.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """Fast version of the competition objective (no safety checks)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    # Because we always keep Σa = √(2·n), the factor reduces to max_b,
    # but we keep the full expression for safety.
    return 2.0 * n * max_b / (s * s)


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of the soft‑max approximation of the maximum convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Return (g, b) where
        b = a*a (full convolution, length 2·n‑1)
        g = ∇ (log‑sum‑exp_{β}(b)) = 2·(a ⋆ p)   with p_k ∝ exp(β·b_k)
    Both are computed with FFTs for speed.
    """
    n = a.shape[0]

    # ---- b = a*a -------------------------------------------------------
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ---- soft‑max probabilities p ---------------------------------------
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)  # shape (2·n‑1,)

    # ---- gradient g = 2·(a ⋆ p) ---------------------------------------
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1  # = 3·n‑2
    size2 = 1 << ((conv_len).bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Stochastic hill‑climbing (cheap local search)
# ----------------------------------------------------------------------


def _random_hill_climb(a: np.ndarray, deadline: float):
    """
    Randomly moves a fraction of mass from one position to another while
    preserving the ℓ¹‑norm.  Keeps improving moves until the deadline.
    """
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    cur_val = best_val

    max_frac = 0.12
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        frac = rng.random() * max_frac
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        new_val = _score(a)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill‑climb improved = {best_val:.6f}")
        else:
            a[j] -= delta
            a[i] += delta

    return best_a, best_val


# ----------------------------------------------------------------------
# Annealed projected gradient descent on the soft‑max approximation
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected gradient descent on the log‑sum‑exp surrogate of max(conv).
    The temperature β is increased (annealed) up to 2048 while the step
    size adapts to the gradient magnitude.  The iterate is always
    projected onto the simplex {a ≥ 0, Σa = √(2·n)}.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    max_beta = 2048.0
    # a somewhat larger base step to explore more aggressively
    step_base = target_sum * 0.30
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        # gradient of the smoothed max
        g, _ = _gradient_softmax(a, beta)

        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            lr = step_base / (norm_g * (beta + 1.0))
            a = a - lr * g

        # projection onto the simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] beta={beta:.2f} score={best_val:.6f}")

        # annealing schedule
        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 1.6, max_beta)

        it += 1

    return best_a, best_val

        target_sum = np.sqrt(2.0 * n)

        # Initialise a0 – either a perturbed copy of the global best or a fresh random power‑law.
        if best_seq is not None and best_seq.size == n and rng.random() < 0.7:
            a0 = best_seq.copy()
            a0 += rng.normal(scale=target_sum * 0.001, size=n)
            a0 = np.maximum(a0, 0.0)
            a0 = _project_to_simplex(a0, target_sum)
        else:
            # random power‑law with optional offset
            p = rng.uniform(0.35, 0.80)
            off = rng.choice(offset_vals)
            idx = np.arange(1, n + 1, dtype=np.float64) + off
            a0 = idx ** (-p)
            a0 = a0 * (target_sum / a0.sum())
            a0 += rng.normal(scale=target_sum * 0.001, size=n)
            a0 = np.maximum(a0, 0.0)
            a0 = _project_to_simplex(a0, target_sum)

        # Allocate a slice of the remaining budget for this restart.
        # 30 s per restart is a good compromise between depth and breadth.
        sub_deadline = min(deadline, time.time() + 30.0)

        # Gradient‑descent phase
        a_opt, opt_score = _annealed_gradient_descent(a0, sub_deadline)

        # Short hill‑climbing refinement (uses the overall deadline)
        if time.time() < deadline:
            a_opt, opt_score = _random_hill_climb(a_opt, deadline)

        if opt_score < best_score:
            best_score = opt_score
            best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] ** GLOBAL BEST UPDATED ** n={n} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing – remove trailing zeros and return result
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback: uniform distribution (should never happen)
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)

    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    return best_seq.tolist()
