"""
Non‑negative step‑function optimizer (dual‑phase).

Phase 1    – minimise the L₂ distance between the self‑convolution and the
            perfectly uniform distribution (∥a*a – 1/(2n‑1)∥²).  The gradient
            of this loss can be written as a single FFT‑based convolution,
            allowing fast projected gradient descent on the probability simplex.

Phase 2    – directly attack the ℓ∞‑objective.  The current convolution is
            turned into a soft‑max weight vector w (the temperature is
            slowly lowered).  The gradient of the smooth‑max is again a
            convolution (w * reverse(p)).  An exponentiated‑gradient update
            keeps the variables non‑negative; a Euclidean projection restores
            the simplex constraint.

Both phases are applied to a diverse pool of seeds (uniform, random,
power‑law, and the previously‑found best).  The best sequence found before
the time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two real vectors using FFT.
    Returns a real array of length len(a)+len(b)-1.
    """
    n = a.shape[0]
    m = b.shape[0]
    # next power of two for efficient FFT
    L = 1 << ((n + m - 2).bit_length())
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(b, n=L)
    C = np.fft.irfft(A * B, n=L)[: n + m - 1]
    # eliminate tiny negative artefacts
    C[C < 0] = 0.0
    return C


def _full_conv_fft(p: np.ndarray) -> np.ndarray:
    """Convolution of a vector with itself (full linear convolution)."""
    return _conv_fft(p, p)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Duchi et al. (2008) algorithm.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1000.0):

    `evaluate_sequence`.  The routine respects the 1000 s time limit and
    returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # safety margin for I/O

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    seeds = []

    # previous best (if present)

    for n in (2000, 3000, 4000):
        seeds.append(np.random.rand(n))

    # power‑law families (p exponent ~0.4‑0.6)
    for n in (2000, 3000, 4000):
        for p_exp in (0.4, 0.5, 0.6):
            sigma = 0.1

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    for base in seeds:

        p = np.asarray(base, dtype=np.float64)

        # discard pathological seeds
        if p.sum() == 0.0 or p.size < 2:
            continue

        # limit length to keep per‑iteration cost reasonable
        max_len = 4000
        if p.size > max_len:

        # normalise to a probability vector
        p = p / p.sum()
        n = p.shape[0]

        # ------------------------------------------------------------------

        # ------------------------------------------------------------------

            p = _project_simplex(p)

            # occasional bookkeeping

            w_raw = np.exp((conv - max_val) / T)   # max‑weight = 1, others ≤ 1
            w = w_raw / w_raw.sum()

            # gradient of the smooth‑max (same convolution pattern as phase 1)
            grad_full = _conv_fft(w, p[::-1])
            g = 2.0 * grad_full[n - 1 : n - 1 + n]

    # ------------------------------------------------------------------
    # 3) Return the best sequence discovered
    # ------------------------------------------------------------------

        print("No valid sequence found within the time limit.")
        return []          # empty list signals failure

    print("Final best score:", best_score, "length:", len(best_seq))
    return best_seq
