"""
Propose a step‑function (non‑negative sequence) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)^2)

where `conv` is the ordinary discrete convolution.
The code follows three ideas that complement the previous approach:

1. **Rich initial pool** – power‑law, exponential, mixed families and a few
   random Dirichlet draws are generated up to a fairly large length
   (≈ 8000).  The best few are kept as seeds.

2. **Projected Adam optimisation** – a variant of Adam is applied to the
   sub‑gradient of the `max‑convolution` term (the only part that influences
   the objective).  After each update the vector is projected onto the
   probability simplex (∑aᵢ = 1, aᵢ≥0).  Momentum, adaptive step sizes and
   occasional “mix‑with‑random” restarts help to escape plateaus.

3. **Time‑aware loop** – the whole routine respects the 1000 s wall‑clock
   budget.  Progress is printed and the best sequence found so far is
   returned.

The implementation avoids any file or network access, uses only NumPy
(and the supplied `evaluate_sequence`) and keeps all helper functions at the
module level as required.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _generate_powerlaw(p: float, sigma: float,
                       max_len: int = 8000, eps: float = 1e-12) -> np.ndarray:
    """Return a[i] = (i+σ)^(-p) truncated when the value falls below eps."""
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p)
    return a[a > eps]


def _generate_exponential(lam: float,
                          max_len: int = 8000, eps: float = 1e-12) -> np.ndarray:
    """Return a[i] = exp(-λ·i) truncated when the value falls below eps."""
    i = np.arange(max_len, dtype=np.float64)
    a = np.exp(-lam * i)
    return a[a > eps]


def _generate_mixed(p: float, sigma: float, lam: float,
                    max_len: int = 8000, eps: float = 1e-12) -> np.ndarray:
    """Return a[i] = (i+σ)^(-p)·exp(-λ·i) truncated when the value falls below eps."""
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p) * np.exp(-lam * i)
    return a[a > eps]


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection works only on 1‑D arrays")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient(x: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of g(x) = max_k (x∗x)_k.
    For every active index (where the maximum is attained) we add the
    contribution ∂(x∗x)_k/∂x_i = 2·x_{k‑i} (when the index is in range).
    """
    n = x.shape[0]
    conv = np.convolve(x, x)          # length 2n‑1
    max_val = conv.max()
    # indices whose value is within 1e‑14 of the maximum
    active = np.where(conv >= max_val - 1e-14)[0]
    g = np.zeros_like(x)
    for k in active:
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        i = np.arange(i_min, i_max + 1, dtype=int)
        j = k - i
        # each term contributes 2·x_j, the factor 2 is absorbed into the step‑size
        g[i] += x[j]
    return g


def _objective(x: np.ndarray) -> float:
    """
    Compute the evaluation function for a *probability* vector x
    (i.e. Σx = 1).  The same value is returned by `evaluate_sequence`.
    """
    if x.sum() < 1e-12:
        return np.inf
    conv = np.convolve(x, x)
    return float(2.0 * len(x) * conv.max() / (x.sum() ** 2))

# ----------------------------------------------------------------------


def _adam_optimize(x0: np.ndarray,
                   deadline: float,
                   max_iter: int = 20000,
                   lr: float = 0.20,
                   beta1: float = 0.90,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   report_every: int = 500) -> tuple[np.ndarray, float]:
    """
    Perform projected Adam optimisation starting from `x0`.
    Returns the best vector and its objective value encountered.
    """
    x = x0.copy()
    x = x / x.sum()                     # ensure we start on the simplex
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_x = x.copy()
    best_val = _objective(x)

    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        g = _subgradient(x)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # parameter update before projection
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = x - step
        x = _simplex_projection(y)

        # occasional random jitter to avoid stagnation
        if np.random.rand() < 0.015:
            jitter = (np.random.rand(*x.shape) - 0.5) * 0.02 * x
            x = np.maximum(x + jitter, 0.0)
            x = x / x.sum()

        # evaluate every `report_every` iterations
        if t % report_every == 0:
            cur = _objective(x)
            if cur < best_val:
                best_val = cur
                best_x = x.copy()
                print(f'      Adam it {t:6d} – new local best = {best_val:.6f}')
            # simple restart if stuck for long
            if t % (report_every * 10) == 0:
                # mix the current point with a fresh random Dirichlet
                rnd = np.random.rand(x.size)
                rnd = rnd / rnd.sum()
                x = 0.6 * x + 0.4 * rnd
                x = x / x.sum()
                m.fill(0.0)
                v.fill(0.0)

    # final check
    final_val = _objective(x)
    if final_val < best_val:
        best_val = final_val
        best_x = x.copy()
    return best_x, best_val

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list:
    """
    Search for a non‑negative step sequence minimizing `evaluate_sequence`.
    The routine respects a 1000 s wall‑clock limit and returns the best
    sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 4.0   # a few seconds margin for final I/O

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # 0.  Use the globally supplied construction if it exists.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            seq = np.asarray(height_sequence_1, dtype=float)
            if seq.size >= 2:
                sc = _objective(seq / seq.sum())
                if sc < best_score:
                    best_score = sc
                    best_seq = seq / seq.sum()
                    print(f'Global seed score = {best_score:.6f}')
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Assemble a pool of diverse seed sequences.
    # ------------------------------------------------------------------
    seed_pool = []

    # power‑law family (several exponents & offsets)
    for p in np.linspace(0.35, 0.65, 13):
        for sigma in np.logspace(-5, 2, 12):
            seq = _generate_powerlaw(p, sigma, max_len=8000, eps=1e-12)
            if seq.size < 2:
                continue
            seed_pool.append(seq / seq.sum())

    # exponential family
    for lam in np.logspace(-4, 0, 12):
        seq = _generate_exponential(lam, max_len=8000, eps=1e-12)
        if seq.size < 2:
            continue
        seed_pool.append(seq / seq.sum())

    # mixed family (few hand‑picked parameters)
    for p, sigma, lam in [(0.45, 1e-2, 1e-3),
                          (0.50, 1e-1, 5e-3),
                          (0.55, 1.0, 1e-2)]:
        seq = _generate_mixed(p, sigma, lam, max_len=8000, eps=1e-12)
        if seq.size >= 2:
            seed_pool.append(seq / seq.sum())

    # random Dirichlet seeds of various lengths
    for n in [500, 1000, 2000, 4000, 6000]:
        rnd = np.random.rand(n)
        rnd = rnd / rnd.sum()
        seed_pool.append(rnd)

    # ------------------------------------------------------------------
    # 2.  Quick evaluation of seeds and selection of the most promising.
    # ------------------------------------------------------------------
    scored_seeds = []
    for s in seed_pool:
        if time.time() >= deadline:
            break
        sc = _objective(s)
        if np.isfinite(sc):
            scored_seeds.append((sc, s))
    scored_seeds.sort(key=lambda tup: tup[0])
    top_seeds = [s for (_, s) in scored_seeds[:10]]   # keep the 10 best

    print(f'Generated {len(seed_pool)} seeds, kept {len(top_seeds)} for optimisation.')

    # ------------------------------------------------------------------
    # 3.  Optimise each seed with projected Adam (time‑aware).
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds, start=1):
        if time.time() >= deadline:
            break
        print(f'--- Optimising seed {idx}/{len(top_seeds)} (len={len(seed_vec)}) ---')
        # allocate a slice of the remaining budget (rough estimate)
        remaining = deadline - time.time()
        # assume ~30 iterations per second for n≈4000 → set a generous limit
        max_iter = int(min(25000, remaining * 30))
        opt_seq, opt_score = _adam_optimize(seed_vec,
                                            deadline,
                                            max_iter=max_iter,
                                            lr=0.20,
                                            report_every=500)

        if opt_score < best_score:
            best_score = opt_score
            best_seq = opt_seq
            print(f'>>> New global best: score = {best_score:.6f}')

        # early exit if we have already crossed the target bound
        if best_score < 1.503:
            print('Target bound achieved – exiting early.')
            break

    # ------------------------------------------------------------------
    # 4.  Final safety evaluation with the provided function.
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback to a uniform vector of moderate size
        n0 = 2000
        best_seq = np.full(n0, 1.0 / n0, dtype=float)
        best_score = _objective(best_seq)

    final_score = evaluate_sequence(best_seq.tolist())
    print(f'Final sequence length = {len(best_seq)}  final score = {final_score:.6f}')
    return best_seq.tolist()
