def project_onto_simplex(v, S):
    # v is array of shape n (nonneg may be negative).
    # Use sorting to find threshold.
    n = len(v)
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - S
    ind = np.arange(1, n+1)
    # find rho = max i such that u_i > cssv_i / i
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all nonpositive, return zero vector (should not happen)
        return np.maximum(v, 0)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def full_convolution(a):
    n = len(a)
    conv_len = 2*n - 1
    fft_len = 1 << (conv_len-1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv

def _project_simplex(v, S):
    """Project vector v onto the simplex {x>=0, sum x = S}."""
    # Implementation from Duchi et al.
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u > cssv / np.arange(1, n+1))[0]
    if rho.size == 0:
        # all entries after projection are zero
        theta = cssv[-1] / n
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _compute_gradient(a, weights):

    S = np.sqrt(2.0 * n)
    # compute conv
    conv = _full_convolution(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_score = cur_max

    # hyperparameters
    tau = cur_max * 0.5  # smoothing parameter
    tau_min = cur_max * 1e-6 + 1e-12
    tau_decay = 0.9995
    eta = 0.5          # step size for gradient part
    eta_decay = 0.9997
    move_prob = 0.07
    epsilon_move = 0.05
    jitter_prob = 0.01

    while time.time() < deadline:
        # compute conv & max
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
        # smoothing weights
        if tau > 0:
            # stable softmax
            # Subtract cur_max to avoid overflow
            exp_vals = np.exp((conv - cur_max) / tau)
            weights = exp_vals / exp_vals.sum()
        else:
            # extreme
            weights = np.zeros_like(conv)
            weights[np.argmax(conv)] = 1.0

        # compute gradient
        g = _compute_gradient(a, weights)

        # gradient descent step + projection
        a = _project_simplex(a - eta * g, S)

        # random move
        if rng.random() < move_prob:
            # choose source based on gradient magnitude (largest)
            # select i proportionally to squared gradient (or simply max)
            grad_abs = np.abs(g)
            # sample i with probability proportional to gradient magnitude
            probs = grad_abs / (grad_abs.sum() + 1e-12)
            i = rng.choice(n, p=probs)
            # choose a destination index uniformly at random, not equal i
            j = rng.integers(n-1)
            if j >= i:
                j += 1
            delta = epsilon_move * a[i]
            if delta > 0:
                a_candidate = a.copy()
                a_candidate[i] -= delta
                a_candidate[j] += delta
                # evaluate candidate's max conv
                conv_c = _full_convolution(a_candidate)
                max_c = conv_c.max()
                # acceptance rule similar to simulated annealing
                if max_c <= cur_max or rng.random() < np.exp(-(max_c - cur_max) / (cur_max * 0.05 + 1e-12)):

def _full_convolution(a):
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()

    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv

"""
Projected smooth‑max gradient descent with occasional mass swaps.

The evaluation function only depends on max(self‑convolution) when the
L1‑norm is fixed (∑aᵢ = √(2·n)).  This code repeatedly starts from a
random (or up‑sampled) non‑negative vector of the required norm,
optimises it by a projected gradient step on a smoothed approximation of
the max‑norm, adds a few simulated‑annealing moves and jitter, and keeps
the best sequence found before the time budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()      # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _project_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """
    Euclidean projection of `v` onto the simplex
        {x | x_i ≥ 0, Σ x_i = S}.
    Implementation of Duchi–Shalev‑Shwartz et al. (2008).
    """
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    ind = np.arange(1, v.shape[0] + 1)
    # find the largest rho satisfying the condition
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be projected to zero (should not happen)
        theta = cssv[-1] / v.shape[0]
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative length‑n sequence with L1‑norm √(2·n).

    If `base_seq` is supplied it is either sliced (if long enough) or
    interpolated to the desired size; otherwise a Dirichlet‑style random
    vector is used.
    """
    S = np.sqrt(2.0 * n)

    if base_seq is not None:
        if base_seq.size >= n:
            a = base_seq[:n].astype(float).copy()
        else:
            # linear interpolation to enlarge the vector
            x_old = np.linspace(0, 1, base_seq.size)
            x_new = np.linspace(0, 1, n)
            a = np.interp(x_new, x_old, base_seq)
        # tiny jitter to diversify restarts
        a += rng.normal(scale=a.mean() * 0.001, size=n)
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)

    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0)
        total = a.sum()
    a = a / total * S
    return a


def _smooth_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """
    Soft‑max weights that approximate the max operator.
    When `tau` → 0 the weight concentrates on the maximal entry.
    """
    if tau <= 0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w
    vmax = conv.max()
    exp_vals = np.exp((conv - vmax) / tau)
    w = exp_vals / exp_vals.sum()
    return w


def _gradient(a: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """
    Compute the sub‑gradient of the (smoothed) max‑convolution:
        g_i = 2 * Σ_k w_k * a_{k-i}
    using FFT‑based correlation.
    """
    n = a.shape[0]
    rev_w = weights[::-1]
    conv_len = n + len(weights) - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(rev_w, n=fft_len)
    corr = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    offset = len(weights) - 1                     # = 2n‑2
    g = 2.0 * corr[offset - np.arange(n)]
    return g


def _optimize_new(a: np.ndarray,
                  deadline: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Local optimiser based on projected smooth‑max gradient descent
    mixed with occasional random swaps and jitter.

    Returns the best vector found and its maximal self‑convolution entry.
    """
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # initialise best record
    conv = _full_convolution(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_score = cur_max

    # hyper‑parameters
    tau = cur_max * 0.5                # initial smoothing
    tau_min = max(cur_max * 1e-6, 1e-12)
    tau_decay = 0.9995
    eta = 0.5                          # base step size for projected GD
    eta_decay = 0.9997
    move_prob = 0.07
    eps_move = 0.05                    # fraction of mass moved
    jitter_prob = 0.01

    while time.time() < deadline:
        # recompute convolution and max
        conv = _full_convolution(a)
        cur_max = conv.max()

        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # smooth approximation of the max‑operator
        weights = _smooth_weights(conv, tau)

        # gradient of the smoothed objective
        g = _gradient(a, weights)

        # projected gradient step (maintain non‑negativity and L1‑norm)
        a = _project_simplex(a - eta * g, S)

        # --- simulated‑annealing random transfer ---
        if rng.random() < move_prob:
            # choose source proportional to gradient magnitude (large contribution)
            prob_src = np.abs(g)
            prob_src /= prob_src.sum() + 1e-12
            i = rng.choice(n, p=prob_src)

            # choose a distinct destination uniformly
            j = rng.integers(n - 1)
            if j >= i:
                j += 1

            delta = eps_move * a[i]
            if delta > 0:
                a_candidate = a.copy()
                a_candidate[i] -= delta
                a_candidate[j] += delta

                cand_max = _full_convolution(a_candidate).max()
                # Metropolis acceptance
                if cand_max <= cur_max or rng.random() < np.exp(
                        -(cand_max - cur_max) / (cur_max * 0.05 + 1e-12)):
                    a = a_candidate
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()

        # occasional jitter to avoid stagnation
        if rng.random() < jitter_prob:
            noise = rng.normal(scale=a.mean() * 0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_simplex(a, S)

        # decay parameters
        tau = max(tau * tau_decay, tau_min)
        eta *= eta_decay

    return best_a, best_score

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising the ∞‑norm of its
    self‑convolution.  The routine runs many short restarts of the
    gradient‑descent optimiser defined above, re‑using the best sequence
    found so far as a warm start for subsequent attempts.  The best
    sequence discovered before the time budget expires is returned as a
    plain list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0          # safety margin

    # try to reuse the previously discovered sequence if it exists
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    attempt = 0
    slice_time = 18.0                                 # seconds per restart

    # ------------------------------------------------------------------
    # many short restarts
    # ------------------------------------------------------------------
    while time.time() < deadline - slice_time:
        attempt += 1

        # choose a length – vary around the current best or pick random
        if best_seq is not None and rng.random() < 0.6:
            n = int(len(best_seq) * (1.0 + rng.uniform(-0.1, 0.1)))
            n = max(800, min(6000, n))
        else:
            n = rng.integers(800, 5000)

        # initialise a candidate (prefer the global base if we have one)
        a0 = _initial_sequence(base_seq if base_seq is not None else best_seq,
                               n, rng)

        slice_deadline = min(time.time() + slice_time, deadline)
        a_opt, score = _optimize_new(a0, slice_deadline, rng)

        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[global] attempt={attempt} n={n}  new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # final refinement (use remaining time)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        # optionally up‑sample a little before the last run
        up_n = int(len(best_seq) * 1.3)
        up_n = max(800, min(6000, up_n))
        a_start = _initial_sequence(best_seq, up_n, rng)
        a_refined, score_ref = _optimize_new(a_start, deadline, rng)
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_refined.copy()
            print(f"[final] refined up to n={up_n}  best={best_score:.6f}")

    # sanity check – evaluate through the external black‑box function
    if best_seq is None:
        best_seq = np.full(1, np.sqrt(2.0))
    final_seq = best_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} "
          f"(internal max = {best_score:.6f}) length={len(final_seq)}")
    return final_seq
