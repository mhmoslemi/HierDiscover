def _smooth_max_and_grad(a, beta):
    n = len(a)
    # compute convolution via FFT
    conv_len = 2*n-1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    # compute logsumexp and weights
    # note conv may have negative values? a_i >=0 so conv >=0, yes.
    exponent = beta * conv
    max_exponent = np.max(exponent)  # to use stable
    # compute w = exp(exponent) / sum(exp(exponent))
    # Use subtraction of max for stability
    exp_shifted = np.exp(exponent - max_exponent)
    Z = np.sum(exp_shifted)
    w = exp_shifted / Z
    # smooth objective value
    f = (np.log(Z) + max_exponent) / beta
    # gradient: g = 2 * correlate(w, a)
    g = np.correlate(w, a, mode='valid')
    g = 2.0 * g
    return f, g, conv, max(conv)  # maybe also return conv and max for tracking

"""
A fast “smooth‐max” optimiser for the self‑convolution∞‑norm.

The algorithm works with a differentiable approximation of the
max‐operator:

    f_β(a) = (1/β)·log Σ_k exp(β·(a∗a)[k]).

Its gradient is
    ∇f_β(a) = 2·correlate(softmax_β(a∗a), a),

so it can be minimised with a standard Adam optimiser.
The optimiser is wrapped in a short‑run, random‑restart scheme.
Occasional simulated‑annealing swaps and tiny Gaussian jitter are
added to help escape local minima.  The best sequence found is
returned as a plain list; the external `evaluate_sequence`
function is called once at the end for the final score.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Helper utilities (top level)
# ----------------------------------------------------------------------


def _conv_max(a: np.ndarray) -> tuple[np.ndarray, float, int]:
    """
    Full linear convolution a*a, its maximal value and the index of that value.
    Uses FFT for speed.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    max_idx = int(np.argmax(conv))
    max_val = float(conv[max_idx])
    return conv, max_val, max_idx


def _logsumexp(x: np.ndarray) -> float:
    """Numerically stable log‑sum‑exp."""
    m = np.max(x)
    return float(m + np.log(np.sum(np.exp(x - m))))


def _smooth_max_and_grad(
    a: np.ndarray, beta: float
) -> tuple[float, np.ndarray, float]:
    """
    Compute the smooth approximation of max(a*a) and its gradient.
    Returns (smooth_max, gradient, true_max).
    """
    n = a.shape[0]

    # convolution via FFT (real‑to‑real)
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]

    # smooth max (log‑sum‑exp)
    exp_term = beta * conv
    max_exp = np.max(exp_term)
    shifted = np.exp(exp_term - max_exp)               # safe scaling
    Z = np.sum(shifted)
    weights = shifted / Z                               # softmax weights
    smooth_val = (max_exp + np.log(Z)) / beta          # (1/β)·log Σ exp
    true_max = float(conv[np.argmax(conv)])

    # gradient = 2·correlate(weights, a)   (valid mode gives length n)
    grad = 2.0 * np.correlate(weights, a, mode="valid")

    return smooth_val, grad, true_max


def _adam_optimize(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth max objective.
    The vector is kept non‑negative and renormalised to L1 = √(2·n)
    after each update.
    Returns the best (lowest true max) vector seen and its max value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.12
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # schedule for the smoothness parameter β
    beta = 5.0          # start smooth
    beta_max = 200.0
    inc_every = 150    # increase β every this many iterations
    beta_factor = 1.02

    # bookkeeping
    conv, cur_max, cur_idx = _conv_max(a)
    best_a = a.copy()
    best_score = cur_max

    # probabilities for auxiliary moves
    swap_prob = 0.07
    jitter_prob = 0.012
    temp = cur_max * 0.08
    temp_decay = 0.9995

    it = 0
    while time.time() < deadline:
        it += 1

        # ---- smooth‑max Adam step ----
        smooth_val, grad, _ = _smooth_max_and_grad(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection to non‑negative orthant
        a = np.maximum(a, 0.0)

        # renormalise L1‑norm
        total = a.sum()
        if total == 0.0:
            a = np.full(n, S / n)
        else:
            a = a * (S / total)

        # ---- occasional random swap (simulated‑annealing) ----
        if rng.random() < swap_prob:
            # active set = indices that appear in the current maximal convolution entry
            partner = cur_idx - np.arange(n)
            valid = (partner >= 0) & (partner < n)
            idx_active = np.where(valid)[0]
            idx_inactive = np.where(~valid)[0]

            if idx_active.size and idx_inactive.size:
                i = rng.choice(idx_active)
                j = rng.choice(idx_inactive)

                delta = 0.08 * a[i]                # move ~8 % of source mass
                delta = min(delta, a[i])           # stay non‑negative

                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta

                # evaluate the true max for the candidate
                _, cand_max, cand_idx = _conv_max(cand)

                # accept if better or with SA probability
                accept = False
                if cand_max <= cur_max:
                    accept = True
                else:
                    prob = np.exp(-(cand_max - cur_max) / max(temp, 1e-12))
                    if rng.random() < prob:
                        accept = True

                if accept:
                    a = cand
                    cur_max, cur_idx = cand_max, cand_idx
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                temp *= temp_decay

        # ---- occasional jitter to shake the system ----
        if rng.random() < jitter_prob:
            noise = rng.normal(scale=a.mean() * 0.005, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = a * (S / a.sum())

        # ---- monitor true max every few iterations ----
        if it % 70 == 0:
            _, cur_max, cur_idx = _conv_max(a)
            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()

        # ---- increase smoothness β gradually ----
        if it % inc_every == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return best_a, best_score


def _initial_sequence(base_seq: np.ndarray | None,
                     n: int,
                     rng: np.random.Generator) -> np.ndarray:
    """
    Produce an L1‑normalised non‑negative start vector of length n.
    Uses a truncated base sequence if available, otherwise a random Dirichlet.
    """
    S = np.sqrt(2.0 * n)
    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        a += rng.normal(scale=a.mean() * 0.001, size=n)   # tiny jitter
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)

    sm = a.sum()
    if sm == 0.0:
        a = np.full(n, 1.0)
        sm = a.sum()
    a = a / sm * S
    return a

            n = rng.integers(800, 6000)

        # initialise
        a0 = _initial_sequence(base_seq, n, rng)

        # allocate a time slice for this run (shorter for very large n)
        slice_time = 25.0 if n > 4000 else 35.0
        slice_deadline = min(time.time() + slice_time, deadline)

        # optimise
        a_opt, score = _adam_optimize(a0, slice_deadline, rng)

        if score < best_overall_score:
            best_overall_score = score
            best_overall_seq = a_opt.copy()
            print(
                f"[global] attempt={attempt} n={n} "
                f"new best score={best_overall_score:.6f}"
            )

    # ------------------------------------------------------------------
    # final sanity check – evaluate via the provided black‑box function
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        # should never happen, but return a trivial feasible point
        return [1.0]

    final_seq = best_overall_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(
        f"[final] best external score = {external_score:.6f} "
        f"(internal max = {best_overall_score:.6f}) "
        f"length = {len(final_seq)}"
    )
    return final_seq
