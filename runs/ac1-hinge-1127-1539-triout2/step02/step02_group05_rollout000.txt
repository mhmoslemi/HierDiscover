"""
Propose a non‑negative step function that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The sum constraint is enforced by projecting onto the simplex
    {x_i ≥ 0, Σ x_i = √(2 n)}.
The algorithm consists of three stages:

1. **Warm‑start** – generate several simple candidates (uniform,
   power‑law, exponential) and keep the one with the smallest
   `max(conv)`.

2. **Adam optimisation with a gradually sharpening surrogate** –
   we start from a soft‑max approximation of the max‑convolution
   (temperature γ) and increase γ until the surrogate becomes the
   true max (weights are 1 on the maximal entries only).  Adam
   updates are projected onto the simplex.

3. **Fine‑tuning** – a small‑step projected gradient descent that uses
   the exact sub‑gradient of the max‑convolution (weights = uniform
   over the arg‑max positions).  Occasional random jitter helps to
   escape plateaus.

All convolutions are performed with FFTs for speed.  The best
sequence found before the time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the metric (identical to `evaluate_sequence`
    after normalisation).  Assumes `a` is non‑negative.
    """
    n = a.shape[0]
    nfft = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    # With Σa = √(2 n) the denominator equals 2 n, therefore score = max(conv).
    return float(max_c)


def project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of *v* onto the simplex
        { x : x_i ≥ 0 , Σ_i x_i = target_sum }.
    """
    # Implementation of Duchi et al. (ICML 2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """Convolution a*a using FFT of size nfft."""
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * a.shape[0] - 1]
    return conv


def _gradient_wrt_weights(a: np.ndarray, w: np.ndarray, nfft_corr: int) -> np.ndarray:
    """
    Compute ∇_a  Σ_k w_k (a*a)[k]   =   2 * (w ⊛ a_rev)
    where a_rev = a reversed.
    """
    wf = np.fft.rfft(w, n=nfft_corr)
    af_rev = np.fft.rfft(a[::-1], n=nfft_corr)
    corr_full = np.fft.irfft(wf * af_rev, n=nfft_corr)
    n = a.shape[0]
    # slice the part that corresponds to the original length
    grad = 2.0 * corr_full[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises the evaluation
    function.  Returns the best sequence found before the time budget
    expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a 5‑second safety margin
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Determine length of the sequence
    # ------------------------------------------------------------------
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        n = int(init_seq.shape[0])
        use_provided = True
    except Exception:
        # No pre‑supplied sequence – choose a moderately large size.
        n = 8000
        use_provided = False

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # 1. Warm‑start: generate several simple candidates
    # ------------------------------------------------------------------
    best_a = None
    best_score = np.inf

    # a) reuse the provided sequence (if any)
    if use_provided:
        a = np.maximum(init_seq, 0.0)
        s = a.sum()
        if s > 0:
            a = a * (target_sum / s)
        else:
            a = np.full(n, target_sum / n)
        sc = _compute_score(a)
        best_a = a.copy()
        best_score = sc

    # b) uniform vector (always feasible)
    a = np.full(n, target_sum / n)
    sc = _compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    # c) a handful of power‑law candidates
    for _ in range(30):
        if time.time() >= deadline:
            break
        p = np.random.uniform(0.3, 0.9)          # exponent
        offset = np.random.uniform(0.5, 3.0)
        a = 1.0 / np.power(np.arange(n, dtype=np.float64) + offset, p)
        a = a * (target_sum / a.sum())
        sc = _compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # d) exponential decay candidates
    for _ in range(10):
        if time.time() >= deadline:
            break
        lam = np.random.uniform(0.0005, 0.005)
        a = np.exp(-lam * np.arange(n, dtype=np.float64))
        a = a * (target_sum / a.sum())
        sc = _compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # fallback
    if best_a is None:
        best_a = np.full(n, target_sum / n)
        best_score = _compute_score(best_a)

    # ------------------------------------------------------------------
    # Pre‑compute FFT lengths used throughout the optimisation
    # ------------------------------------------------------------------
    nfft_conv = _next_pow2(2 * n - 1)      # for a * a
    nfft_corr = _next_pow2(3 * n - 2)      # for w ⊛ a_rev

    # ------------------------------------------------------------------
    # Adam optimiser initialisation
    # ------------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(best_a)
    v = np.zeros_like(best_a)
    t = 0

    # learning‑rate schedule
    lr = 0.30
    lr_min = 5e-4
    lr_decay = 0.9997

    # soft‑max temperature schedule
    gamma = 0.5
    gamma_max = 2000.0      # very high – effectively exact max
    gamma_inc = 1.005
    # switch to exact max when gamma exceeds this value
    switch_gamma = 30.0

    # ------------------------------------------------------------------
    # Main optimisation loop (Adam + gradually sharpening surrogate)
    # ------------------------------------------------------------------
    a = best_a.copy()
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        t += 1

        # ---- convolution a*a ------------------------------------------------
        af = np.fft.rfft(a, n=nfft_conv)
        conv = np.fft.irfft(af * af, n=nfft_conv)[: 2 * n - 1]
        max_conv = conv.max()

        # ---- compute surrogate weights --------------------------------------
        if gamma < switch_gamma:
            # smooth soft‑max
            w = np.exp(gamma * (conv - max_conv))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w /= w_sum
        else:
            # exact max: uniform weight over all indices attaining the maximum
            mask = (np.abs(conv - max_conv) <= 1e-12 * max(1.0, max_conv))
            cnt = mask.sum()
            if cnt == 0:
                # fallback – should not happen
                mask[conv.argmax()] = True
                cnt = 1
            w = mask.astype(np.float64) / cnt

        # ---- gradient of the surrogate --------------------------------------
        grad = _gradient_wrt_weights(a, w, nfft_corr)

        # ---- Adam update ----------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- projection back onto the simplex --------------------------------
        a_new = project_onto_simplex(a_new, target_sum)

        # ---- occasional jitter to escape flat regions -----------------------
        if iteration % 73 == 0:
            jitter = 1e-4 * np.random.randn(n) * target_sum
            a_new = project_onto_simplex(a_new + jitter, target_sum)

        # ---- evaluate --------------------------------------------------------
        score_new = _compute_score(a_new)

        # keep the best ever seen
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            elapsed = time.time() - start_time
            print(
                f"Iter {iteration:6d}: new best score {best_score:.6f} "
                f"(elapsed {elapsed:.1f}s, n={n})"
            )

        # accept the new point (Adam already performs stochastic exploration)
        a = a_new

        # ---- schedule updates ------------------------------------------------
        gamma = min(gamma * gamma_inc, gamma_max)
        lr = max(lr * lr_decay, lr_min)

        # safety break
        if iteration % 1000 == 0 and time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 2. Fine‑tuning: exact max sub‑gradient with a tiny constant step
    # ------------------------------------------------------------------
    fine_iters = 3000
    fine_lr = 5e-4
    for _ in range(fine_iters):
        if time.time() >= deadline:
            break

        af = np.fft.rfft(a, n=nfft_conv)
        conv = np.fft.irfft(af * af, n=nfft_conv)[: 2 * n - 1]
        max_conv = conv.max()

        # exact max weight
        mask = (np.abs(conv - max_conv) <= 1e-12 * max(1.0, max_conv))
        cnt = mask.sum()
        if cnt == 0:
            mask[conv.argmax()] = True
            cnt = 1
        w = mask.astype(np.float64) / cnt

        grad = _gradient_wrt_weights(a, w, nfft_corr)

        a = a - fine_lr * grad
        a = project_onto_simplex(a, target_sum)

        score_new = _compute_score(a)
        if score_new < best_score:
            best_score = score_new
            best_a = a.copy()
            elapsed = time.time() - start_time
            print(
                f"Fine‑tune: new best score {best_score:.6f} "
                f"(elapsed {elapsed:.1f}s)"
            )

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as a plain Python list)
    # ------------------------------------------------------------------
    print(f"Finished optimisation: best score {best_score:.6f}")
    return best_a.tolist()
