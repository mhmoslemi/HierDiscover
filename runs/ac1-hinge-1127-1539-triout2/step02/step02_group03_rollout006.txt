def proj_simplex(v, s):
    # from Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (s - cssv) / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = (cssv[rho] - s) / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

def _project_simplex(v, s):
    """
    Euclidean projection of v onto the simplex {x >= 0, sum(x)=s}
    Returns projected vector.
    """
    # From Duchi et al.
    n = v.shape[0]
    if s < 0:
        raise ValueError("s must be nonnegative")
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    # find rho
    rho = np.nonzero(u + (s - cssv) / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        # all entries zero after projection
        theta = (cssv[-1] - s) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - s) / (rho + 1)
    w = np.maximum(v - theta, 0)
    # ensure sum exactly s (due to floating precision)
    # adjust via scaling if needed
    diff = w.sum() - s
    if abs(diff) > 1e-9:
        # distribute diff proportionally? We'll just re-normalize small.
        w = np.clip(w, 0.0, None)
        if w.sum() > 0:
            w = w / w.sum() * s
        else:
            w = np.full_like(w, s / n)
    return w

compute the full convolution c = a*a by FFT, then replace the max
operator by the smooth approximation

    φβ(c) = (1/β)·log Σ exp(β·cₖ) ,

whose gradient can be written in closed form using a linear correlation.
A simple additive step is taken, after which the vector is projected back
onto the simplex {x ≥ 0, Σx = S}.  Several independent restarts of this
process (different lengths, random seeds and hyper‑parameters) are run
within the time budget; the best sequence seen is finally returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (s - cssv) / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        # all entries will be zero
        theta = (cssv[-1] - s) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - s) / (rho + 1)
    w = np.maximum(v - theta, 0.0)

    # Numerical fix – enforce the exact sum (helps later rounding)
    diff = w.sum() - s
    if abs(diff) > 1e-9:
        # Distribute the tiny error uniformly over the non‑zero entries
        nz = w > 0
        if nz.any():
            w[nz] -= diff / nz.sum()
            w = np.clip(w, 0.0, None)
        else:  # fallback – uniform distribution
            w = np.full_like(w, s / n)
    return w


def _fft_len_for_n(n: int) -> int:
    """Length of the FFT that can accommodate a linear correlation of
    vectors of lengths 2·n‑1 and n (i.e. 3·n‑2)."""
    size = 3 * n - 2
    # next power of two
    return 1 << ((size - 1).bit_length())


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT of length fft_len."""
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear correlation (w ⋆ a) of lengths (2·n‑1) and n.
    Returned array has length 3·n‑2; index k corresponds to shift k‑(n‑1).
    """
    Fw = np.fft.rfft(w, n=fft_len)
    Fa = np.fft.rfft(a, n=fft_len)
    corr_full = np.fft.irfft(Fw * np.conj(Fa), n=fft_len)[: 3 * a.shape[0] - 2]
    return corr_full


def _softmax_step(a: np.ndarray,
                  fft_len: int,
                  beta: float,
                  lr: float,
                  S: float) -> tuple[np.ndarray, float]:
    """
    One projected gradient step on the soft‑max surrogate.
    Returns the updated vector and the current max convolution entry.
    """
    # ----- convolution -----
    conv = _full_convolution(a, fft_len)
    cur_max = float(conv.max())

    # ----- soft‑max weights -----
    # shift by cur_max for numerical stability (max weight = 1)
    w = np.exp(beta * (conv - cur_max))

    # ----- gradient (via correlation) -----
    corr = _correlation(w, a, fft_len)
    # correlation index i + (n‑1) gives Σ w_{i+j} a_j
    n = a.shape[0]
    g = 2.0 * corr[n - 1 : n - 1 + n]          # shape (n,)

    # ----- additive step + projection -----
    a_new = a - lr * g
    a_new = _project_simplex(a_new, S)

    return a_new, cur_max


def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative sequence of length n with L¹‑norm √(2·n).
    If a base_seq is supplied and sufficiently long, its leading part
    (with a tiny jitter) is used; otherwise a Dirichlet sample is drawn.
    """
    S = np.sqrt(2.0 * n)
    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        # a small jitter helps to diversify restarts
        a += rng.normal(scale=a.mean() * 0.001, size=n)
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)                     # uniform Dirichlet (α=1)
    tot = a.sum()
    if tot == 0.0:
        a = np.full(n, 1.0)
        tot = a.sum()
    a = a * (S / tot)
    return a


def _optimize_local(a0: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Local optimiser based on the smooth surrogate.
    Returns the best vector and its corresponding max‑convolution value.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)
    fft_len = _fft_len_for_n(n)

    a = a0.copy()
    # initialise best
    conv = _full_convolution(a, fft_len)
    best_score = float(conv.max())
    best_a = a.copy()

    # hyper‑parameters (tuned empirically)
    beta = 5.0                # initial smoothness
    beta_max = 80.0
    beta_inc = 0.12           # increase per iteration
    lr = 0.03 * S / n         # initial step size (scaled to problem size)
    lr_decay = 0.9995

    move_prob = 0.12          # probability of attempting a random mass move
    epsilon_move = 0.07       # fraction of source mass moved in a swap
    jitter_prob = 0.015       # occasional Gaussian jitter

    while time.time() < deadline:
        # ----- stochastic mass move (simulated annealing style) -----
        if rng.random() < move_prob:
            # pick an index that actually contributes to the current max
            conv = _full_convolution(a, fft_len)
            max_idx = int(np.argmax(conv))

            partner = max_idx - np.arange(n)
            valid = (partner >= 0) & (partner < n)
            contrib_idx = np.where(valid)[0]
            non_contrib_idx = np.where(~valid)[0]

            if contrib_idx.size and non_contrib_idx.size:
                i = rng.choice(contrib_idx)
                j = rng.choice(non_contrib_idx)

                delta = epsilon_move * a[i]
                delta = min(delta, a[i])
                a_candidate = a.copy()
                a_candidate[i] -= delta
                a_candidate[j] += delta

                # quick accept/reject based on the max after the move
                conv_cand = _full_convolution(a_candidate, fft_len)
                cand_max = float(conv_cand.max())
                cur_max = float(conv.max())
                if cand_max <= cur_max or rng.random() < np.exp(-(cand_max - cur_max) / (cur_max * 0.05)):
                    a = a_candidate
                    if cand_max < best_score:
                        best_score = cand_max
                        best_a = a.copy()
        else:
            # ----- smooth‑surrogate gradient step -----
            a, cur_max = _softmax_step(a, fft_len, beta, lr, S)
            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()

        # ----- schedule updates -----
        beta = min(beta + beta_inc, beta_max)
        lr *= lr_decay

        # ----- occasional jitter to avoid stagnation -----
        if rng.random() < jitter_prob:
            a += rng.normal(scale=0.01 * a.mean(), size=n)
            a = _project_simplex(a, S)

    return best_a, best_score

    """

    # --------------------------------------------------------------
