Now about projection onto simplex. Implement function simplex_projection(v, S). Implementation from Duchi et al.

Implementation details:

```
def _project_simplex(v, s):
    # ensure non-negative sum s
    n = v.shape[0]
    u = np.sort(v)[::-1]   # descending
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def _optimize_smooth(a, deadline, rng):
    n = len(a)
    S = np.sqrt(2.0 * n)
    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1  # can adjust
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # smoothing schedule
    alpha = 1.0
    alpha_inc = 1.02
    max_alpha = 200.0

    # initial best
    conv, cur_max, cur_idx = _conv_max(a)
    best_a = a.copy()
    best_score = cur_max

    iter = 0

    while time.time() < deadline:
        iter += 1

        # compute conv
        conv = np.convolve(a, a)
        # compute weights
        # stable compute: subtract max before exp
        max_conv = conv.max()
        # Note: using scale: w = exp(alpha*(conv - max_conv)) to avoid overflow
        w = np.exp(alpha * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(conv) / conv.size
        else:
            w = w / w_sum

        # compute gradient via convolution w * a_rev
        a_rev = a[::-1]
        conv_len = len(w) + len(a_rev) - 1
        fft_len = 1 << (conv_len - 1).bit_length()
        W_fft = np.fft.rfft(w, n=fft_len)
        A_rev_fft = np.fft.rfft(a_rev, n=fft_len)
        grad_full = np.fft.irfft(W_fft * A_rev_fft, n=fft_len)[:conv_len]
        # slice indices n-1 to n-1+n
        grad_slice = grad_full[n-1: n-1 + n]
        g = 2.0 * grad_slice

        # Adam updates
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** iter)
        v_hat = v / (1 - beta2 ** iter)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex [nonneg sum S]
        a = _project_simplex(a, S)

        # compute actual max to update best
        conv_f, cur_max_f, _ = _conv_max(a)
        if cur_max_f < best_score:
            best_score = cur_max_f
            best_a = a.copy()

        # optionally adjust learning rate

"""
Propose a candidate step‑function that minimizes the ∞‑norm of its
self‑convolution.

Algorithm
---------
1.  Randomly choose a length n (800 ≤ n ≤ ~5000) and generate an
    initial non‑negative sequence a with ℓ₁‑norm √(2 n).  Several
    families of initial shapes are tried (random Dirichlet, half‑sine,
    half‑cosine, triangular “tent”, etc.) as well as a jittered copy of
    a previously found best sequence (if available).

2.  Local optimisation – a smooth‑max gradient descent with Adam:
    *   At each iteration compute the full self‑convolution c = a * a.
    *   Form a soft‑max weight w_k = exp(α·(c_k − max(c))) / Σ.
        The temperature α is increased slowly from ≈0.5 to ≈200,
        gradually sharpening the focus on the true maximum.
    *   The gradient of the smooth objective Σ w_k·c_k is
        g_i = 2·Σ_j w_{i+j}·a_j, obtained efficiently by a single FFT‑based
        convolution (w * a_rev).
    *   Apply an Adam update, then project the result onto the simplex
        {x ≥ 0, Σ x = √(2 n)} (non‑negativity + fixed ℓ₁‑norm).
    *   Occasionally add a tiny Gaussian jitter to avoid stagnation.

    The best vector seen (according to the exact max of a * a) is kept.

3.  Run many independent short restarts within the overall time budget,
    keeping the globally best sequence.

The code respects all constraints (no file I/O, only numpy, no closures)
and prints progress information.  After the time limit the best
sequence is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top level)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection of v onto the simplex {x >= 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.zeros_like(v)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_conv(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·n‑1) using an FFT.
    Returns a real ndarray.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_max(a: np.ndarray) -> tuple[np.ndarray, float, int]:
    """
    Compute the full self‑convolution, its maximum value and the index of that maximum.
    """
    conv = _full_conv(a)
    max_idx = int(np.argmax(conv))
    max_val = float(conv[max_idx])
    return conv, max_val, max_idx


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Given a (length n) and weight array w (length 2·n‑1) corresponding
    to the smooth‑max, compute the gradient g_i = 2 Σ_j w_{i+j} a_j.
    This is performed via a single FFT‑based convolution of w with a reversed.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    conv_len = w.shape[0] + a_rev.shape[0] - 1    # = 3·n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    W_fft = np.fft.rfft(w, n=fft_len)
    A_rev_fft = np.fft.rfft(a_rev, n=fft_len)
    full = np.fft.irfft(W_fft * A_rev_fft, n=fft_len)[:conv_len]
    # slice indices n‑1 … n‑1+n‑1  (inclusive) -> length n
    grad_slice = full[n - 1:n - 1 + n]
    return 2.0 * grad_slice


def _initial_sequence_random(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random non‑negative initial sequence of length n,
    normalised to ℓ₁‑norm √(2·n).  Several families of shapes are tried
    randomly to increase diversity.
    """
    S = np.sqrt(2.0 * n)
    choice = rng.integers(0, 5)
    if choice == 0:
        # pure Dirichlet (uniform)
        a = rng.random(n)
    elif choice == 1:
        # half‑sine
        x = np.linspace(0.0, np.pi, n)
        a = np.sin(x)
    elif choice == 2:
        # half‑cosine (raised)
        x = np.linspace(0.0, np.pi, n)
        a = 1.0 - np.cos(x)
    elif choice == 3:
        # triangular (tent) shape
        half = n // 2
        left = np.linspace(0.0, 1.0, half, endpoint=False)
        right = np.linspace(1.0, 0.0, n - half)
        a = np.concatenate([left, right])
    else:
        # Gamma‑Dirichlet with concentration < 1 (more spiky)
        a = rng.gamma(shape=0.5, scale=1.0, size=n)
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0)
        a_sum = a.sum()
    a = a / a_sum * S
    return a


def _initial_sequence_from_base(base_seq: np.ndarray,
                               n: int,
                               rng: np.random.Generator) -> np.ndarray:
    """
    Initialise from a previously found sequence (global variable).  The
    leading n entries are taken, a tiny Gaussian jitter is added and the
    result is re‑normalised.
    """
    S = np.sqrt(2.0 * n)
    a = base_seq[:n].astype(float).copy()
    a += rng.normal(scale=max(a.mean() * 0.001, 1e-12), size=n)
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0)
        a_sum = a.sum()
    a = a / a_sum * S
    return a


def _optimize_local_smooth(a0: np.ndarray,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Smooth‑max Adam optimiser for a single restart.
    Returns the best sequence found and its maximal convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.2

    # smoothing (soft‑max) temperature
    alpha = 0.5
    alpha_inc = 1.02
    max_alpha = 200.0

    # first evaluation
    conv, cur_max, cur_idx = _conv_max(a)
    best_a = a.copy()
    best_score = cur_max

    # moments for Adam
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # ----- compute smooth objective and its gradient -----
        conv = _full_conv(a)                     # length 2·n‑1
        max_conv = conv.max()
        # stable soft‑max
        exp_term = np.exp(alpha * (conv - max_conv))
        w = exp_term / exp_term.sum()

        # gradient of Σ w_k·(a*a)_k
        g = _gradient_from_weights(a, w)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex (non‑negative + fixed ℓ₁‑norm)
        a = _project_simplex(a, S)

        # ----- occasional jitter to escape flat regions -----
        if rng.random() < 0.02:
            jitter = rng.normal(scale=max(a.mean() * 0.005, 1e-12), size=n)
            a = np.clip(a + jitter, 0.0, None)
            a = _project_simplex(a, S)

        # ----- evaluate exact maximum (for bookkeeping) -----
        _, cur_max, _ = _conv_max(a)
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # ----- schedule updates -----
        alpha = min(alpha * alpha_inc, max_alpha)
        if iteration % 200 == 0:
            lr *= 0.92  # gentle decay

    return best_a, best_score


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Entry point required by the evaluation harness.
    Runs many independent short optimisation attempts and returns the
    best step‑function sequence discovered within the allotted time.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0  # small safety margin

    # try to reuse a previously found sequence if it exists
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_overall_seq = None
    best_overall_score = np.inf
    attempt = 0

    # main outer loop: keep launching short restarts until time runs out
    while time.time() < deadline:
        attempt += 1

        # -------------------------------------------------------------
        # Choose a length for this attempt.
        # -------------------------------------------------------------
        if base_seq is not None:
            max_len = min(6000, max(800, base_seq.size))
            n = rng.integers(800, max_len + 1)
        else:
            n = rng.integers(800, 4000)

        # -------------------------------------------------------------
        # Initialise a candidate.
        # -------------------------------------------------------------
        use_base = (base_seq is not None) and (rng.random() < 0.4)
        if use_base:
            a0 = _initial_sequence_from_base(base_seq, n, rng)
        else:
            a0 = _initial_sequence_random(n, rng)

        # -------------------------------------------------------------
        # Allocate a slice of remaining time for this run.
        # -------------------------------------------------------------
        slice_deadline = min(time.time() + 30.0, deadline)

        # -------------------------------------------------------------
        # Local optimisation.
        # -------------------------------------------------------------
        a_opt, score = _optimize_local_smooth(a0, slice_deadline, rng)

        # -------------------------------------------------------------
        # Keep the best overall result.
        # -------------------------------------------------------------
        if score < best_overall_score:
            best_overall_score = score
            best_overall_seq = a_opt.copy()
            print(f"[global] attempt={attempt} n={n} new best score={best_overall_score:.6f}")

    # -----------------------------------------------------------------
    # Final sanity check – evaluate through the provided black‑box function
    # -----------------------------------------------------------------
    if best_overall_seq is not None:
        final_seq = best_overall_seq.tolist()
        external_score = evaluate_sequence(final_seq)
        print(f"[final] best external score = {external_score:.6f} "
              f"(internal max = {best_overall_score:.6f}) "
              f"length = {len(final_seq)}")
        return final_seq
    else:
        # fallback – should never happen
        return [1.0]
