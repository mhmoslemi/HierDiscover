def _project_onto_simplex(v, S):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _project_onto_simplex(v, S):
    # Returns x >= 0, sum x = S.
    # Implementation based on algorithm from Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0]
    if len(rho) == 0:
        # all zeros
        return np.zeros_like(v)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _additive_step(a, eta, S):
    n = a.shape[0]
    # compute subgradient
    _, _, k = _conv_max(a)
    idx = np.arange(n)
    partner = k - idx
    valid = (partner >= 0) & (partner < n)
    g = np.zeros_like(a)
    g[valid] = 2.0 * a[partner[valid]]
    if k % 2 == 0:
        d = k // 2
        if 0 <= d < n:
            g[d] = a[d]
    # additive update
    a_new = a - eta * g
    # project onto simplex with sum S
    a_proj = _project_onto_simplex(a_new, S)
    return a_proj

def _optimize_local(a, deadline, rng):
    n = a.shape[0]
    S = np.sqrt(2.0*n)
    # compute initial conv
    conv, cur_max, cur_idx = _conv_max(a)
    best_a = a.copy()
    best_score = cur_max

    # hyperparameters
    eta_mult = 0.6
    decay_mult = 0.9995

    eta_add = 0.1
    decay_add = 0.9995

    move_prob = 0.2
    epsilon_move = 0.1

    jitter_prob = 0.02
    temp = cur_max * 0.1
    temp_decay = 0.999

    while time.time() < deadline:
        r = rng.random()
        if r < move_prob:
            # random swap (simulated annealing)
            # similar to before
            partner_idx = cur_idx - np.arange(n)
            valid = (partner_idx >= 0) & (partner_idx < n)
            i_candidates = np.where(valid)[0]
            j_candidates = np.where(~valid)[0]
            if i_candidates.size and j_candidates.size:
                i = rng.choice(i_candidates)
                j = rng.choice(j_candidates)
                delta = epsilon_move * a[i]
                a_candidate = a.copy()
                a_candidate[i] -= delta
                a_candidate[j] += delta
                # project onto simplex? Not needed as sum unchanged
                # evaluate
                _, cand_max, cand_idx = _conv_max(a_candidate)
                accept = False
                if cand_max <= cur_max:
                    accept = True
                else:
                    prob_accept = np.exp(-(cand_max - cur_max) / max(temp, 1e-12))
                    if rng.random() < prob_accept:
                        accept = True
                if accept:
                    a = a_candidate
                    cur_max, cur_idx = cand_max, cand_idx
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                temp *= temp_decay
        else:
            # choose step type
            if r < move_prob + 0.5:  # multiplicative half
                a = _gradient_step(a, eta_mult)
                eta_mult *= decay_mult
            else:
                a = _additive_step(a, eta_add, np.sqrt(2.0*n))
                eta_add *= decay_add

            _, cur_max, cur_idx = _conv_max(a)
            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()

        if rng.random() < jitter_prob:
            noise = rng.normal(scale=a.mean() * 0.01, size=n)

            # renorm to sum S
            total = a.sum()
            if total > 0:
                a = a * (S / total)

def _initial_sequence(shape, n, rng, base_seq=None):
    S = np.sqrt(2.0 * n)
    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        a += rng.normal(scale=a.mean() * 0.001, size=n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    # Choose shape
    if shape == 'uniform':
        a = np.full(n, S / n)
    elif shape == 'linear_inc':
        raw = np.arange(1, n+1, dtype=float)
        a = raw / raw.sum() * S

        raw = rng.random(n)
        a = raw / raw.sum() * S
    else:
        raw = rng.random(n)
        a = raw / raw.sum() * S

"""
Optimiser for the ‖f∗f‖∞⁄‖f‖₁² benchmark.

The algorithm works in two phases:

* Phase‑1 – many short restarts.  Randomly chosen lengths (`n≈800..4000`)
  and a variety of simple initial shapes (uniform, linear, exponential,
  random) are used.  Each restart runs a fast local optimiser that mixes
  three moves:

    – multiplicative exponentiated‑gradient step (keeps non‑negativity);
    – additive projected‑gradient step (explicit L₁‑simplex projection);
    – simulated‑annealing “mass‑swap’’ moves that transfer a fraction of
      mass from indices contributing to the current maximal convolution
      entry to unrelated indices.

  The best sequence found so far is stored.

* Phase‑2 – long refinement of the best sequence.  The same optimiser
  runs until the global time budget expires, allowing many more
  iterations and a slower decay of step‑sizes, which usually yields a
  lower maximal convolution value.

All sequences are kept non‑negative and renormalised to
∑aᵢ = √(2·n), so the evaluation score returned by ``evaluate_sequence``
equals the maximal convolution entry.  The optimiser prints progress
and finally returns the best sequence as a list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top level)
# ----------------------------------------------------------------------


def _conv_max(a: np.ndarray) -> tuple[np.ndarray, float, int]:
    """
    Compute the full linear convolution a*a using FFT.
    Returns the convolution array, its maximum, and the index of the maximum.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    max_idx = int(np.argmax(conv))
    return conv, float(conv[max_idx]), max_idx


def _gradient_step(a: np.ndarray, eta: float) -> np.ndarray:
    """
    Exponentiated‑gradient (multiplicative) sub‑gradient step.
    The L₁‑norm is re‑normalised to √(2·n) after the update.
    """
    n = a.shape[0]
    _, _, k = _conv_max(a)

    idx = np.arange(n)
    partner = k - idx
    valid = (partner >= 0) & (partner < n)

    g = np.zeros_like(a)
    g[valid] = 2.0 * a[partner[valid]]
    if k % 2 == 0:
        d = k // 2
        if 0 <= d < n:
            g[d] = a[d]

    a_new = a * np.exp(-eta * g)

    S = np.sqrt(2.0 * n)
    total = a_new.sum()
    if total == 0.0:
        a_new = np.full(n, S / n)
    else:
        a_new *= (S / total)
    return a_new


def _project_onto_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex {x ≥ 0, Σx = S}.
    Implementation of the algorithm by Duchi et al.
    """
    if S < 0:
        raise ValueError("Simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho_arr = u - cssv / np.arange(1, len(v) + 1)
    rho = np.where(rho_arr > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _additive_step(a: np.ndarray, eta: float, S: float) -> np.ndarray:
    """
    Additive projected‑gradient sub‑gradient step.
    After moving a ← a − η g the result is projected onto the L₁ simplex.
    """
    n = a.shape[0]
    _, _, k = _conv_max(a)

    idx = np.arange(n)
    partner = k - idx
    valid = (partner >= 0) & (partner < n)

    g = np.zeros_like(a)
    g[valid] = 2.0 * a[partner[valid]]
    if k % 2 == 0:
        d = k // 2
        if 0 <= d < n:
            g[d] = a[d]

    a_tmp = a - eta * g
    a_proj = _project_onto_simplex(a_tmp, S)
    return a_proj


def _initial_sequence(shape: str,
                      n: int,
                      rng: np.random.Generator,
                      base_seq: np.ndarray | None = None) -> np.ndarray:
    """
    Produce a non‑negative sequence of length n with L₁‑norm √(2·n).
    Several deterministic shapes are available; otherwise a random Dirichlet
    vector is used.  If ``base_seq`` is supplied and long enough its leading
    part (with tiny jitter) is taken.
    """
    S = np.sqrt(2.0 * n)

    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        a += rng.normal(scale=a.mean() * 0.001, size=n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    if shape == "uniform":
        a = np.full(n, S / n)
    elif shape == "linear_inc":
        raw = np.arange(1, n + 1, dtype=float)
        a = raw / raw.sum() * S
    elif shape == "linear_dec":
        raw = np.arange(n, 0, -1, dtype=float)
        a = raw / raw.sum() * S
    elif shape == "exp_dec":
        alpha = -rng.uniform(0.3, 2.0)
        raw = np.exp(alpha * np.arange(n))
        a = raw / raw.sum() * S
    elif shape == "exp_inc":
        alpha = rng.uniform(0.3, 2.0)
        raw = np.exp(alpha * np.arange(n))
        a = raw / raw.sum() * S
    elif shape == "random":
        raw = rng.random(n)
        a = raw / raw.sum() * S
    else:
        raw = rng.random(n)
        a = raw / raw.sum() * S
    return a


def _optimize_local(a: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Local optimiser mixing three ideas:
      * multiplicative exponentiated‑gradient step,
      * additive projected‑gradient step,
      * simulated‑annealing mass swaps,
    with occasional Gaussian jitter.
    Returns the best sequence seen and its maximal convolution value.
    """
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    conv, cur_max, cur_idx = _conv_max(a)
    best_a = a.copy()
    best_score = cur_max

    # hyper‑parameters
    eta_mult = 0.6
    decay_mult = 0.9995

    eta_add = 0.1
    decay_add = 0.9995

    move_prob = 0.2          # how often we try a random swap
    epsilon_move = 0.10      # fraction of mass moved in a swap
    jitter_prob = 0.02
    temp = cur_max * 0.1
    temp_decay = 0.999

    while time.time() < deadline:
        r = rng.random()

        if r < move_prob:
            # ----- simulated‑annealing random swap -----
            partner = cur_idx - np.arange(n)
            valid = (partner >= 0) & (partner < n)
            i_candidates = np.where(valid)[0]
            j_candidates = np.where(~valid)[0]

            if i_candidates.size and j_candidates.size:
                i = rng.choice(i_candidates)
                j = rng.choice(j_candidates)

                delta = epsilon_move * a[i]
                delta = min(delta, a[i])

                a_cand = a.copy()
                a_cand[i] -= delta
                a_cand[j] += delta

                _, cand_max, cand_idx = _conv_max(a_cand)

                accept = False
                if cand_max <= cur_max:
                    accept = True
                else:
                    prob = np.exp(-(cand_max - cur_max) / max(temp, 1e-12))
                    if rng.random() < prob:
                        accept = True

                if accept:
                    a = a_cand
                    cur_max, cur_idx = cand_max, cand_idx
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                temp *= temp_decay
        else:
            # ----- pick one of the gradient steps -----
            if r < move_prob + 0.5:          # multiplicative half the time
                a = _gradient_step(a, eta_mult)
                eta_mult *= decay_mult
            else:                            # additive half the time
                a = _additive_step(a, eta_add, S)
                eta_add *= decay_add

            _, cur_max, cur_idx = _conv_max(a)
            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()

        # occasional jitter to escape plateaus
        if rng.random() < jitter_prob:
            noise = rng.normal(scale=a.mean() * 0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            total = a.sum()
            if total > 0:
                a = a * (S / total)

    return best_a, best_score


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,

    deadline = start + budget_s - 5.0        # small safety margin

    # ------------------------------------------------------------------
    # try to reuse a previously discovered sequence if available
    # ------------------------------------------------------------------
    base_seq = None
    if "height_sequence_1" in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    attempts = 0

    # initialise from base_seq if present (provides a solid starting point)
    if base_seq is not None:
        n0 = base_seq.size
        S0 = np.sqrt(2.0 * n0)
        if base_seq.sum() > 0:
            base_seq = base_seq * (S0 / base_seq.sum())
        _, score0, _ = _conv_max(base_seq)
        best_seq = base_seq.copy()
        best_score = score0
        print(f"[init] using previous sequence len={n0} score={best_score:.6f}")

    # possible shape prototypes for fresh restarts
    shape_options = ["uniform", "linear_inc", "linear_dec",
                     "exp_dec", "exp_inc", "random"]

    # --------------------------------------------------------------
    # Phase‑1 : many short restarts (reserve time for a final refinement)
    # --------------------------------------------------------------
    reserve = 200.0                         # seconds kept for phase‑2
    phase1_deadline = deadline - reserve

    while time.time() < phase1_deadline:
        attempts += 1

        # pick a length – slightly broader than the previous runs
        if base_seq is not None:
            max_len = max(800, min(6000, base_seq.size))
            n = rng.integers(800, max_len + 1)
        else:
            n = rng.integers(800, 4000)

        shape = rng.choice(shape_options)

        a0 = _initial_sequence(shape, n, rng, base_seq)

        # give each short run about 12‑15 s
        slice_deadline = min(time.time() + 13.0, phase1_deadline)

        a_opt, score = _optimize_local(a0, slice_deadline, rng)

        if score < best_score:

            print(f"[phase1] attempt={attempts} n={n} shape={shape} "

    # --------------------------------------------------------------
    # Phase‑2 : long refinement of the best candidate found so far
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 10.0:
        a_refined, refined_score = _optimize_local(best_seq, deadline, rng)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = a_refined.copy()

              f"(internal max = {best_score:.6f}) length = {len(final_seq)}")
        return final_seq
    else:
        # should never happen – fallback to a trivial feasible vector
        return [np.sqrt(2.0).item()]  # L₁‑norm = √2 for n=1
