"""
Non‑negative step‑function optimizer (smooth max‑convolution).

The evaluation function is

    f(a) = 2 * n * max(conv(a,a)) / (sum(a)**2)

which is invariant under scaling, so we may restrict ourselves to the
probability simplex Σ a_i = 1.  The goal is therefore to minimise

    F(p) = 2 * n * max_k (p * p)[k]      (p ≥ 0, Σp = 1)

where “*” denotes the ordinary linear convolution.
The maximum of a vector can be approximated by a smooth log‑sum‑exp
(soft‑max).  For a fixed smoothing parameter β the smoothed objective is

    G(p) = (1/β)·log Σ_k exp(β·(p*p)[k]).

Its gradient is

    ∂G/∂p_i = 2·Σ_k w_k·p_{k‑i},
    w_k = exp(β·(p*p)[k]) / Σ_j exp(β·(p*p)[j]).

The sum on the right‑hand side is a (cross‑)correlation of the current
vector p with the weight vector w and can be computed by a single FFT
pair.  We perform a projected (multiplicative) gradient step

    p ← p·exp(‑η·∂G/∂p)   ,   p ← p / Σp

which keeps the iterate non‑negative and on the simplex.
β is increased slowly (continuation) so that the optimisation first
flattens the whole convolution and later concentrates on the true
maximum.  Several diverse seeds (uniform, random Dirichlet, power‑law,
exponential) are refined; the best sequence found before the time budget
expires is returned.

The implementation respects a 1000 s wall‑clock limit, prints progress
and requires only NumPy (no external solvers).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _full_conv_fft(p: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a real vector with itself using FFT.
    Returns a real array of length 2*len(p)-1.
    """
    n = p.shape[0]
    L = 1 << ((2 * n - 2).bit_length())          # next power of two
    ft = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(ft * ft, n=L)[: 2 * n - 1]
    conv = np.real(conv)
    conv[conv < 0] = 0.0          # clip tiny negatives caused by round‑off
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w
