def _corr_fft(p: np.ndarray, q: np.ndarray) -> np.ndarray:
    n = p.shape[0]
    m = q.shape[0]
    L = 1 << ((n + m - 1).bit_length())
    ft_p = np.fft.rfft(p, n=L)
    ft_q = np.fft.rfft(q, n=L)
    corr = np.fft.irfft(ft_p * np.conj(ft_q), n=L)
    # The linear correlation result length = n + m - 1, indices [-m+1, n-1].
    # We need indices k from 0..n-1: that's corr[m-1 : m-1 + n]
    start = m - 1
    return corr[start : start + n]
```

"""
Non‑negative step‑function optimizer (gradient‑descent on a smoothed max).

The code searches for a probability vector `p` (non‑negative, sum = 1) that
minimises  f(p) = 2·n·max(conv(p,p)).  The scale‑invariance of the problem
allows us to work with the simplex.  We use a homotopy on the smooth
approximation

        F_q(p) = ( Σ_k conv(p,p)[k]^q )^{1/q}      (q → ∞ ⇒ max)

Its gradient can be written in closed form using a single correlation,
which we evaluate by FFT for speed.  An additive projected‑gradient
descent with a simple back‑tracking line search is performed; the
exponent `q` is increased gradually (2 → 4 → … → 256) so that the
optimization first sees a smooth landscape and then focuses on the true
ℓ∞‑objective.  Many diverse seeds (Dirichlet, power‑law, exponential,
sinusoidal, the previous best) are tried.  The routine respects a
1000 s budget and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def _full_conv_fft(p: np.ndarray) -> np.ndarray:
    """Linear convolution p * p using an FFT of length a power of two."""
    n = p.shape[0]
    L = 1 << ((2 * n - 2).bit_length())          # next power of two ≥ 2n‑1
    ft = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(ft * ft, n=L)[: 2 * n - 1]
    conv = np.real(conv)
    conv[conv < 0] = 0.0                          # numerical safety
    return conv


def _corr_fft(p: np.ndarray, q: np.ndarray) -> np.ndarray:
    """
    Linear correlation of p with q (full, then return the part for lag ≥0).
    Result length = len(p).  Implemented via FFT to stay O(N log N).
    """
    n = p.shape[0]
    m = q.shape[0]
    L = 1 << ((n + m - 1).bit_length())          # power of two ≥ n+m‑1
    ft_p = np.fft.rfft(p, n=L)
    ft_q = np.fft.rfft(q, n=L)
    corr_full = np.fft.irfft(ft_p * np.conj(ft_q), n=L)
    start = m - 1                                 # index of lag = 0
    return corr_full[start:start + n]


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – Duchi et al. (2008)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast score (no security checks)
# ----------------------------------------------------------------------


def _fast_score(p: np.ndarray) -> float:
    """Return 2·n·max(conv(p,p)) for a probability vector p."""
    n = p.shape[0]
    conv = _full_conv_fft(p)
    return 2.0 * n * np.max(conv)

# Main optimisation routine
# ----------------------------------------------------------------------
