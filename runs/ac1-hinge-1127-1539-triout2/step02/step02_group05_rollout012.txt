"""
Propose a non‑negative step‑function that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm consists of three stages:

1. **Initialisation** – build a few diverse candidates (uniform,
   random Dirichlet, and an up‑sampled copy of the globally supplied
   ``height_sequence_1`` if available).  The best of these seeds becomes
   the starting point.

2. **L2‑flattening** – minimise the squared deviation of the
   self‑convolution from the uniform distribution.
   The objective  ∑ₖ ( (a*a)[k] – μ )²  is differentiable;
   its gradient is 4·((a*a) – μ) ⊛ a_rev.
   An Adam optimiser with a decreasing learning‑rate is used,
   together with a Euclidean projection onto the simplex
   {x ≥ 0, ∑x = 1}.  This phase quickly spreads the mass so that the
   convolution becomes almost flat.

3. **Maximum‑reduction** – starting from the point produced by stage 2
   we directly minimise the true objective.
   The non‑smooth “max” is approximated by a soft‑max with an
   increasing temperature γ; the gradient is
   2·(w ⊛ a_rev) where w are the soft‑max weights.
   Again Adam is used, with occasional jitter to escape plateaus
   and a final fine‑tuning pass with a very large γ and a tiny step
   size.

All heavy lifting (convolutions and gradients) is performed with
FFT for O(n log n) cost.  The routine respects the 1000 s time budget,
keeps the best sequence seen so far and returns it as a plain Python
list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of *v* onto the simplex
        { x : x_i ≥ 0 , sum_i x_i = target_sum }.
    """
    # Based on the algorithm of Duchi et al. (ICML 2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_score_from_a(a: np.ndarray) -> float:
    """
    Fast computation of the evaluation metric for a vector whose
    L1‑norm is (approximately) 1.0.  The returned value is exactly
    ``evaluate_sequence(a)`` up to the trivial scaling invariance.
    """
    n = a.shape[0]
    nfft = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative sequence) with a low evaluation
    score.  The routine uses a combination of L2‑flattening and a
    smoothed‑max Adam optimiser.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the caller
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Choose the length of the sequence
    # ------------------------------------------------------------------
    try:
        base_seq = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        base_n = base_seq.shape[0]
    except Exception:
        base_seq = None
        base_n = 0

    # We aim for a longer sequence than the supplied one – more degrees of
    # freedom tend to allow a flatter convolution.
    n = max(8000, int(base_n * 1.5))          # a safe “large enough” size
    # Upper bound to keep the FFT manageable within the time budget
    n = min(n, 20000)

    target_sum = 1.0  # we keep the L1‑norm = 1 (scale‑invariant)

    # ------------------------------------------------------------------
    # 2. Build a few diverse starting candidates
    # ------------------------------------------------------------------
    candidates = []

    # (a) Uniform vector (the classic baseline)
    candidates.append(np.full(n, target_sum / n))

    # (b) Random Dirichlet (exponential) distribution
    candidates.append(np.random.rand(n))
    candidates[-1] /= candidates[-1].sum()

    # (c) Power‑law shapes – a useful family for this problem
    for _ in range(5):
        p = np.random.uniform(0.3, 1.2)
        offset = np.random.uniform(0.2, 3.0)
        a = 1.0 / np.power(np.arange(n, dtype=np.float64) + offset, p)
        a /= a.sum()
        candidates.append(a)

    # (d) Upsampled copy of the global solution if it exists
    if base_seq is not None and base_n > 0:
        # linear interpolation to the larger grid
        x_old = np.arange(base_n, dtype=np.float64)
        x_new = np.linspace(0, base_n - 1, n)
        a = np.interp(x_new, x_old, base_seq)
        a = np.maximum(a, 0.0)
        a /= a.sum()
        candidates.append(a)

    # ------------------------------------------------------------------
    # 3. Pick the best seed according to the cheap score
    # ------------------------------------------------------------------
    best_a = None
    best_score = np.inf
    for a in candidates:
        # local fast score (sum = 1)
        sc = _compute_score_from_a(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # safety fallback
    if best_a is None:
        best_a = np.full(n, target_sum / n)
        best_score = _compute_score_from_a(best_a)

    # ------------------------------------------------------------------
    # 4. Pre‑compute FFT sizes (used repeatedly)
    # ------------------------------------------------------------------
    nfft_conv = _next_pow2(2 * n - 1)          # for a * a
    nfft_corr = _next_pow2(3 * n - 2)          # for w ⊛ a_rev (weight length 2n‑1)

    # ------------------------------------------------------------------
    # 5. Optimisation helpers
    # ------------------------------------------------------------------
    def convolution_self(a_vec):
        af = np.fft.rfft(a_vec, n=nfft_conv)
        conv = np.fft.irfft(af * af, n=nfft_conv)[: 2 * n - 1]
        return conv

    def gradient_from_weights(weights, a_vec):
        """
        Given a weight vector w of length 2n‑1, return the gradient
        of Σ w_k (a * a)[k] w.r.t. a, which is 2·(w ⊛ a_rev).
        """
        wf = np.fft.rfft(weights, n=nfft_corr)
        af_rev = np.fft.rfft(a_vec[::-1], n=nfft_corr)
        corr_full = np.fft.irfft(wf * af_rev, n=nfft_corr)
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]
        return grad

    # ------------------------------------------------------------------
    # 6. Phase‑1 : L2‑flattening  (minimise ||a*a – μ·1||²)
    # ------------------------------------------------------------------
    a = best_a.copy()
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    lr = 0.6               # relatively aggressive
    lr_min = 1e-3
    lr_decay = 0.9997

    mu = 1.0 / (2 * n - 1)   # target average of the convolution (since sum(a)=1)

    phase1_end = start_time + 0.55 * budget_s   # spend about 55 % of the budget

    iteration = 0
    while time.time() < phase1_end:
        iteration += 1
        # --- convolution a*a -------------------------------------------------
        conv = convolution_self(a)

        # --- weights for L2 error (w = 2·(conv – μ)) --------------------------
        w = 2.0 * (conv - mu)

        # --- gradient --------------------------------------------------------
        grad = gradient_from_weights(w, a)          # = 4·(conv‑μ) ⊛ a_rev

        # --- Adam update -----------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        t = iteration
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # --- projection onto the simplex (maintain sum = 1) -------------------
        a_new = project_onto_simplex(a_new, target_sum)

        # --- evaluate ---------------------------------------------------------
        score_new = _compute_score_from_a(a_new)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            elapsed = time.time() - start_time
            print(
                f"[L2] Iter {iteration:6d}: best score {best_score:.6f} (elapsed {elapsed:.1f}s)"
            )

        a = a_new

        # --- schedule updates -------------------------------------------------
        lr = max(lr * lr_decay, lr_min)

        # occasional jitter to avoid stagnation
        if iteration % 97 == 0:
            jitter = 1e-4 * np.random.randn(n)
            a = project_onto_simplex(a + jitter, target_sum)

    # ------------------------------------------------------------------
    # 7. Phase‑2 : Direct max‑reduction (soft‑max Adam)
    # ------------------------------------------------------------------
    # Re‑initialise Adam moments for the second phase
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.25
    lr_min = 1e-4
    lr_decay = 0.9995

    gamma = 1.0
    gamma_max = 500.0
    gamma_inc = 1.003

    phase2_end = deadline

    iteration = 0
    while time.time() < phase2_end:
        iteration += 1
        # (a) convolution
        conv = convolution_self(a)
        max_conv = conv.max()

        # (b) soft‑max weights approximating the max
        #    w = exp(gamma * (conv - max_conv))  (stable version)
        shift = conv - max_conv
        w = np.exp(gamma * shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # (c) gradient of the smoothed objective
        grad = gradient_from_weights(w, a)

        # (d) Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        t = iteration
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # (e) projection onto the simplex
        a_new = project_onto_simplex(a_new, target_sum)

        # (f) evaluate
        score_new = _compute_score_from_a(a_new)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            elapsed = time.time() - start_time
            print(
                f"[MAX] Iter {iteration:6d}: new best {best_score:.6f} (elapsed {elapsed:.1f}s)"
            )

        a = a_new

        # schedule updates
        gamma = min(gamma * gamma_inc, gamma_max)
        lr = max(lr * lr_decay, lr_min)

        # jitter to shake out of shallow plateaus
        if iteration % 101 == 0:
            jitter = 5e-5 * np.random.randn(n)
            a = project_onto_simplex(a + jitter, target_sum)

        # safety break
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 8. Fine‑tuning : very high γ, tiny constant step (no Adam)
    # ------------------------------------------------------------------
    fine_iters = 1500
    gamma_fine = gamma_max
    step_fine = 3e-4
    for _ in range(fine_iters):
        if time.time() >= deadline:
            break
        conv = convolution_self(a)
        max_conv = conv.max()
        w = np.exp(gamma_fine * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum
        grad = gradient_from_weights(w, a)

        a = a - step_fine * grad
        a = project_onto_simplex(a, target_sum)

        score_new = _compute_score_from_a(a)
        if score_new < best_score:
            best_score = score_new
            best_a = a.copy()
            elapsed = time.time() - start_time
            print(
                f"[FINE] new best {best_score:.6f} (elapsed {elapsed:.1f}s)"
            )

    # ------------------------------------------------------------------
    # 9. Return the best sequence found (as a plain list)
    # ------------------------------------------------------------------
    print(f"Finished. Best score = {best_score:.6f}")
    return best_a.tolist()
