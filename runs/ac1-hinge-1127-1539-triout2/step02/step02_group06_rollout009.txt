"""
Propose a height sequence that (approximately) minimizes
evaluate_sequence(sequence).  The algorithm consists of three stages:

1.  A broad parametric sweep over many analytical families (linear,
    quadratic, exponential, “inverse‑U”, and cosine‑powered) with a
    very wide range of parameters.  The best candidate from this sweep
    seeds the local optimisation.

2.  A first‑order optimiser that directly minimises a smooth approximation
    of the ℓ∞‑norm of the self‑convolution.  The soft‑max temperature τ is
    annealed, the update uses the Adam adaptive‑learning‑rate scheme,
    and after each step the vector is projected onto the simplex
    (ℓ₁‑norm = √(2 n) and non‑negativity).  Random pair‑wise mass transfers
    and small Gaussian jitter are injected periodically to escape plateaus.

3.  A short polishing phase that performs many cheap random pair transfers
    while constantly checking the true objective (via the analytical formula
    for the metric, which equals the value returned by evaluate_sequence).

The whole search respects a 1000 s wall‑clock budget and always returns the
best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures / lambdas)
# ----------------------------------------------------------------------


def fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors a and b using an FFT of power‑of‑2 length."""
    n, m = a.shape[0], b.shape[0]
    L = 1 << ((n + m - 1).bit_length())           # next power of two
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)
    return conv[: n + m - 1]


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (length 2·len(a)-1)."""
    return fft_convolve(a, a)


def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector so that its ℓ₁‑norm equals *target*.
    Zero‑vector fallback returns a uniform distribution."""
    s = v.sum()
    if s > 0.0:
        return v * (target / s)
    return np.full_like(v, target / v.shape[0])


def generate_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
    direction: str = "symmetric",
) -> np.ndarray:
    """
    Produce a deterministic shape of length *n*.

    *kind*      – {"linear", "quadratic", "exponential", "cosine"}.
    *direction* – {"symmetric", "inverseU", "decreasing", "increasing"}.

    *param* controls the strength of the edge/gradient (≥0).  For the
    “inverseU’’ direction the base shape is 1−param·(1‑base) so that the
    centre can become smaller than the edges.  For the “cosine’’ kind the
    shape is |cos(π·i/(n‑1))|**param.
    The returned vector is finally scaled so that its ℓ₁‑norm equals *target_sum*.
    """
    idx = np.arange(n, dtype=np.float64)

    # base distance from centre, normalised to [0,1]
    mid = (n - 1) / 2.0
    base = np.abs(idx - mid) / mid                     # ∈[0,1]

    # ------------------------------------------------------------------
    # build raw (unscaled) profile
    # ------------------------------------------------------------------
    if direction == "inverseU":
        # centre lower, edges at level 1
        raw = 1.0 - param * (1.0 - base)
    elif direction == "symmetric":
        # edges higher, centre = 1
        raw = 1.0 + param * base
    elif direction == "decreasing":
        raw = 1.0 - idx / (n - 1)
    elif direction == "increasing":
        raw = idx / (n - 1)
    else:
        raise ValueError(f"unknown direction {direction}")

    if kind == "linear":
        pass                                          # raw already linear in base
    elif kind == "quadratic":
        raw = 1.0 + param * base ** 2
        if direction == "inverseU":
            raw = 1.0 - param * (1.0 - base ** 2)
    elif kind == "exponential":
        raw = np.exp(param * base)
        if direction == "inverseU":
            raw = np.exp(param * (1.0 - base))
    elif kind == "cosine":
        # ignore param as base – we use param as exponent
        raw = np.abs(np.cos(np.pi * idx / (n - 1))) ** param
    else:
        raise ValueError(f"unknown kind {kind}")

    raw = np.clip(raw, 0.0, None)
    return project_to_sum(raw, target_sum)


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """Transfer a tiny fraction of mass from a random source index to a random target."""
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0.0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_to_sum(a, target_sum)


def jitter(a: np.ndarray, target_sum: float, sigma: float = 1e-3) -> np.ndarray:
    """Add Gaussian noise (clipped) and renormalise."""
    a = a + np.random.randn(a.shape[0]) * sigma
    a = np.clip(a, 0.0, None)
    return project_to_sum(a, target_sum)


# ----------------------------------------------------------------------
# Gradient / Adam utilities
# ----------------------------------------------------------------------


def compute_softmax_gradient(a: np.ndarray, tau: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Compute the gradient of the smooth max (log‑sum‑exp) of the self‑convolution.
    Returns (grad, conv) where conv = a * a (full linear convolution) and
    grad is ∂/∂a softmax_max(conv).
    """
    n = a.shape[0]
    conv = conv_self_fft(a)               # length 2n‑1

    # stable soft‑max weights
    max_c = conv.max()
    exp_arg = (conv - max_c) / tau
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:                      # degenerate – fall back to hard max
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    # gradient = 2 * (w ⊛ a_rev)  (where ⊛ is linear convolution)
    a_rev = a[::-1]
    grad_full = fft_convolve(w, a_rev)    # length 3n‑2
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad, conv


def adam_update(
    a: np.ndarray,
    grad: np.ndarray,
    m: np.ndarray,
    v: np.ndarray,
    t: int,
    lr: float,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """
    Perform a single Adam step (followed by non‑negativity and ℓ₁‑projection).
    """
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad ** 2)

    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = a - step                     # minimise -> move opposite to grad
    a_new = np.clip(a_new, 0.0, None)

    # projection onto the ℓ₁‑simplex (maintains the prescribed sum)
    return a_new, m, v


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:

    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence`` within the given time budget.
    """

    # --------------------------------------------------------------
    base = globals().get("height_sequence_1")

    target_sum = np.sqrt(2.0 * n)                   # ℓ₁‑norm used in the literature

    # --------------------------------------------------------------
    best_seq = np.full(n, target_sum / n)           # uniform baseline
    best_score = float("inf")

    # families: (direction, kind, param_grid)
    families = [
        ("symmetric", "linear", np.linspace(0.0, 15.0, 31)),
