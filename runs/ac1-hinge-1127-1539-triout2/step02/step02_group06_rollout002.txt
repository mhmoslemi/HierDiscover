"""
Propose a candidate step‑function that minimizes the maximum entry of the
self‑convolution (the quantity returned by `evaluate_sequence`).  The
algorithm follows three stages:

1. **Parametric sweep** – generate a few analytic families (power‑U,
   exponential, quadratic, linear) for the chosen length and keep the
   best one.  The sweep also tries a few random starts.

2. **Adam‑based smooth‑max descent** – minimise a smooth approximation of
   the max using the log‑sum‑exp trick.  The gradient is obtained by a
   convolution (FFT).  An Adam optimiser with bias‑corrected moments is
   used; after each step the vector is projected back to the simplex
   (fixed ℓ₁‑norm) and symmetrised to exploit the natural symmetry of the
   problem.  The temperature of the soft‑max is exponentially cooled,
   eventually approximating the true max.

3. **Polishing** – occasional random pair‑wise mass transfers and tiny
   jitter are applied when progress stalls, followed by a short final
   random search.

All operations are performed with NumPy and FFTs, keeping the runtime
well below the 1000 s budget.  The code works for any existing
`height_sequence_1` (used as a warm start) and always returns the best
sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures / lambdas)
# ----------------------------------------------------------------------


def fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors a and b using a power‑of‑2 FFT."""
    n, m = a.shape[0], b.shape[0]
    L = 1 << ((n + m - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)
    return conv[: n + m - 1]


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (length 2·len(a)‑1)."""
    return fft_convolve(a, a)


def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector so that its ℓ₁‑norm equals *target*."""
    s = v.sum()
    if s > 0:
        return v * (target / s)
    # fallback – uniform distribution (should never happen)
    return np.full_like(v, target / v.shape[0])


def generate_shape(
    n: int,
    target_sum: float,
    kind: str = "power",
    param: float = 0.0,
    direction: str = "symmetric",
) -> np.ndarray:
    """
    Produce a deterministic shape of length *n* and scale it to `target_sum`.

    ``kind`` – {"power", "linear", "quadratic", "exponential"}.
    ``direction`` – {"symmetric", "decreasing", "increasing"}.

    *param* controls the shape (exponent, gradient, etc.) and must be ≥0.
    """
    idx = np.arange(n, dtype=np.float64)

    if direction == "symmetric":
        # U‑shape centred in the middle
        if kind == "power":
            # (i/(n‑1))**p + ((n‑1‑i)/(n‑1))**p  → U‑shape, p≥0
            base = (idx / (n - 1)) ** param + ((n - 1 - idx) / (n - 1)) ** param
        elif kind == "linear":
            mid = (n - 1) / 2.0
            d = np.abs(idx - mid) / mid  # ∈[0,1]
            base = 1.0 + param * d
        elif kind == "quadratic":
            mid = (n - 1) / 2.0
            d = np.abs(idx - mid) / mid
            base = 1.0 + param * d ** 2
        elif kind == "exponential":
            mid = (n - 1) / 2.0
            d = np.abs(idx - mid) / mid
            base = np.exp(param * d)
        else:
            raise ValueError("unknown shape kind")
    elif direction == "decreasing":
        base = 1.0 - idx / (n - 1)
        if kind != "linear":
            # apply param as a multiplier for the base shape
            base = 1.0 + param * base
    elif direction == "increasing":
        base = idx / (n - 1)
        if kind != "linear":
            base = 1.0 + param * base
    else:
        raise ValueError("unknown direction")

    raw = np.clip(base, 0.0, None)
    return project_to_sum(raw, target_sum)


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """Transfer a tiny fraction of mass from a random source index to a random target."""
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0.0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_to_sum(a, target_sum)


def jitter(a: np.ndarray, target_sum: float, sigma: float = 1e-3) -> np.ndarray:
    """Add small Gaussian noise and renormalise."""
    a = a + np.random.randn(a.shape[0]) * sigma
    a = np.clip(a, 0.0, None)
    return project_to_sum(a, target_sum)


# ----------------------------------------------------------------------
# Adam‑based smooth‑max descent
# ----------------------------------------------------------------------


def smooth_max_step_adam(
    a: np.ndarray,
    target_sum: float,
    lr: float,
    tau: float,
    beta1: float,
    beta2: float,
    eps: float,
    t: int,
    m: np.ndarray,
    v: np.ndarray,
) -> tuple[np.ndarray, np.ndarray, np.ndarray, int, float]:
    """
    One iteration of Adam optimisation on the smoothed max of the
    self‑convolution.

    Returns updated (a, m, v, t, tau).
    """
    n = a.shape[0]

    # 1) self‑convolution (used for soft‑max)
    conv = conv_self_fft(a)          # length 2n‑1
    max_c = conv.max()

    # 2) soft‑max weights
    exp_arg = (conv - max_c) / tau
    # prevent overflow / underflow
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    # 3) gradient of the smooth‑max (cross‑correlation)
    a_rev = a[::-1]
    grad_full = fft_convolve(w, a_rev)            # length 3n‑2
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]      # slice to length n

    # 4) projected direction (preserve ℓ₁‑norm)
    direction = -(grad - np.mean(grad))

    # 5) Adam moments
    t += 1
    m = beta1 * m + (1.0 - beta1) * direction
    v = beta2 * v + (1.0 - beta2) * (direction ** 2)

    bias_correction1 = 1.0 - beta1 ** t
    bias_correction2 = 1.0 - beta2 ** t
    m_hat = m / bias_correction1
    v_hat = v / bias_correction2

    step = lr * m_hat / (np.sqrt(v_hat) + eps)

    # 6) update, keep feasibility
    a_new = a + step
    a_new = np.clip(a_new, 0.0, None)

    # optional symmetry enforcement (helps a lot for this problem)
    a_new = (a_new + a_new[::-1]) * 0.5

    a_new = project_to_sum(a_new, target_sum)

    # 7) cool the temperature a little (outside the caller we may also decay)
    return a_new, m, v, t, tau

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence`` within the allocated wall‑clock budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # Choose a length – reuse a known construction if available
    # --------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
    else:
        # a size large enough to give a good bound but still cheap enough
        n = np.random.randint(8000, 12000)

    target_sum = np.sqrt(2.0 * n)  # normalisation used in the literature

    # --------------------------------------------------------------
    # Phase 1 – broad parametric sweep (few analytic families)
    # --------------------------------------------------------------
    best_seq = np.full(n, target_sum / n)  # uniform baseline
    best_score = np.max(conv_self_fft(best_seq))

    families = [
        ("symmetric", "power"),
        ("symmetric", "exponential"),
        ("symmetric", "quadratic"),
        ("symmetric", "linear"),
        ("decreasing", "linear"),
        ("increasing", "linear"),
    ]

    # parameter grids
    param_grids = {
        "power": np.linspace(0.0, 10.0, 21),          # 0 … 10 step 0.5
        "exponential": np.linspace(0.0, 3.0, 16),    # 0 … 3 step 0.2
        "quadratic": np.linspace(0.0, 4.0, 21),      # 0 … 4 step 0.2
        "linear": np.linspace(0.0, 5.0, 26),         # 0 … 5 step 0.2
    }

    print(f"[{time.time():.2f}] Parametric sweep (n={n})")
    for direction, kind in families:
        for p in param_grids[kind]:
            cand = generate_shape(
                n, target_sum, kind=kind, param=p, direction=direction
            )
            score = np.max(conv_self_fft(cand))
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(
                    f"  New best {direction}/{kind} (p={p:.2f}) → {best_score:.6f}"
                )
            if time.time() >= deadline:
                print("Time exhausted during sweep.")
                return best_seq.tolist()

    # a few random initialisations as well
    for _ in range(30):
        cand = np.random.rand(n)
        cand = project_to_sum(cand, target_sum)
        score = np.max(conv_self_fft(cand))
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"  Random init improved → {best_score:.6f}")
        if time.time() >= deadline:
            return best_seq.tolist()

    # --------------------------------------------------------------
    # Phase 2 – Adam smooth‑max descent
    # --------------------------------------------------------------
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0

    lr = 0.1                     # base learning rate for Adam
    # initialise temperature from the current max‑convolution value
    cur_conv = conv_self_fft(a)
    tau = max(0.2 * cur_conv.max(), 1e-6)
    tau_decay = 0.9995           # exponential cooling factor
    min_tau = 1e-12

    no_improve = 0
    iteration = 0

    print(f"[{time.time():.2f}] Starting Adam descent (lr={lr:.3g}, τ={tau:.3g})")
    while time.time() < deadline:
        # evaluate current sequence (conv already calculated inside step)
        cur_conv = conv_self_fft(a)
        cur_score = cur_conv.max()
        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"Iter {iteration:06d} – new best {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # one Adam step
        a, m, v, t, tau = smooth_max_step_adam(
            a,
            target_sum,
            lr,
            tau,
            beta1,
            beta2,
            eps,
            t,
            m,
            v,
        )

        iteration += 1

        # occasional exploratory moves
        if np.random.rand() < 0.02:
            a = random_pair_transfer(a, target_sum, scale=0.008)

        # cool the temperature a little each iteration
        tau = max(tau * tau_decay, min_tau)

        # decay learning rate slowly
        if iteration % 3000 == 0:
            lr *= 0.9
            if lr < 1e-4:
                lr = 1e-4

        # jitter if stuck for a while
        if no_improve >= 5000:
            a = jitter(a, target_sum, sigma=5e-4)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0
            print(f"Iter {iteration:06d} – jitter reset")

        # safety check on remaining time
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # Phase 3 – final quick random polishing
    # --------------------------------------------------------------
    print(f"[{time.time():.2f}] Descent finished; best score {best_score:.6f}")
    while time.time() < deadline - 0.2:
        a_try = random_pair_transfer(best_seq, target_sum, scale=0.004)
        score_try = np.max(conv_self_fft(a_try))
        if score_try < best_score:
            best_score = score_try
            best_seq = a_try.copy()
            print(f"Polish – improvement to {best_score:.6f}")

    return best_seq.tolist()
