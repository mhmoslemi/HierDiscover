"""
Optimise a non‑negative step‑function (a list of heights) for the
provided evaluate_sequence() objective.

The optimiser works in three stages inside the 1000 s wall‑clock limit:

1. **Parametric sweep** – families `a_i = (i+σ)^{-p}`, pure exponential,
   and mixed power/exponential are sampled with a broad grid of parameters.
   Every candidate is normalised to ℓ¹‑norm = 1 (the objective is scale‑free)
   and evaluated.  The best candidate is kept.

2. **Projected additive sub‑gradient descent** – starting from the best
   candidate we repeatedly move opposite to a sub‑gradient of the
   max‑auto‑convolution term, project back onto the probability simplex,
   and use a decreasing step‑size with Nesterov‑type momentum.
   Random jitter and occasional random restarts help to escape plateaus.

3. **Simple simulated‑annealing hill‑climber** – using the current best
   vector we repeatedly transfer a tiny amount of mass from a random
   index to another.  Moves that improve the objective are kept;
   occasionally a worse move is accepted with a very low temperature to
   explore new basins.  After the time budget expires the best sequence
   (optionally trimmed of near‑zero trailing entries) is returned.

All helper functions are defined at the top level, the code never uses
closures or lambda expressions, and only `numpy`, `time` and the already
available `evaluate_sequence` are imported.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level)
# ----------------------------------------------------------------------


def _generate_powerlaw(p: float, sigma: float,
                       max_len: int = 3000, eps: float = 1e-12) -> np.ndarray:
    """a_i = (i + sigma) ** (-p)   truncated when a_i < eps."""
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p)
    return a[a > eps]


def _generate_exponential(lam: float,
                          max_len: int = 3000, eps: float = 1e-12) -> np.ndarray:
    """a_i = exp(-lam * i)   truncated when a_i < eps."""
    i = np.arange(max_len, dtype=np.float64)
    a = np.exp(-lam * i)
    return a[a > eps]


def _generate_mixed(p: float, sigma: float, lam: float,
                    max_len: int = 3000, eps: float = 1e-12) -> np.ndarray:
    """a_i = (i+sigma)^(-p) * exp(-lam*i)   truncated when a_i < eps."""
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p) * np.exp(-lam * i)
    return a[a > eps]


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("simplex projection only works on 1‑D arrays")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient(x: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of g(x) = max_k (x*x)_k.
    All active convolution indices (where the maximum is attained) are used.
    """
    n = x.shape[0]
    conv = np.convolve(x, x)                # length 2n‑1
    m = conv.max()
    active = np.where(conv >= m - 1e-14)[0]
    g = np.zeros_like(x)
    for k in active:
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        i = np.arange(i_min, i_max + 1, dtype=int)
        j = k - i
        # derivative of conv[k] w.r.t. x_i is x_j  (the factor 2 is absorbed in step‑size)
        g[i] += x[j]
    return g


def _max_conv_fft(x: np.ndarray) -> float:
    """Maximum entry of the auto‑convolution using an FFT (O(n log n))."""
    n = x.shape[0]
    # next power of two >= 2n‑1
    m = 1 << ((2 * n - 1).bit_length())
    X = np.fft.rfft(x, m)
    conv = np.fft.ifft(X * X, m).real[: 2 * n - 1]
    return float(conv.max())


def _objective(x: np.ndarray) -> float:
    """
    Fast version of evaluate_sequence for a vector already satisfying
    the security checks (non‑negative, not all zero, values << 1000).
    """
    sum_a = x.sum()
    if sum_a < 0.01:
        return float('inf')
    n = x.shape[0]
    max_b = _max_conv_fft(x)
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _prune_trailing_zeros(seq: np.ndarray, thresh: float = 1e-12) -> np.ndarray:
    """Remove trailing entries that are ≤ thresh, then renormalise."""
    idx = np.where(seq > thresh)[0]
    if idx.size == 0:
        return seq.copy()   # nothing to keep – return as‑is
    last = idx[-1] + 1
    pruned = seq[:last].copy()
    if pruned.sum() > 0:
        pruned /= pruned.sum()
    return pruned

# Main search routine

    best_score = float('inf')

    n = x.shape[0]
    temperature = 1e-4                  # very low temperature – mostly greedy
    iter_hc = 0
    print('=== Phase 3 – hill‑climber (simulated annealing) ===')
    while time.time() < deadline:

    # ------------------------------------------------------------------
    print('final best score:', best_score, 'length:', len(best_seq))
    return best_seq.tolist()
