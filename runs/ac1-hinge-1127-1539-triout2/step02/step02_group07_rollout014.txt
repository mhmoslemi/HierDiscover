                g = _subgradient_active(p, conv, tol_factor=1e-6)

                # momentum blending
                g = gamma * g_prev + (1.0 - gamma) * g
                g_prev = g

                # decreasing step size
                eta = eta0 / np.sqrt(it)

                # multiplicative (exponentiated‑gradient) update
                update = -eta * g
                update = np.clip(update, -50.0, 50.0)   # avoid overflow
                p = p * np.exp(update)

                # project back onto the simplex
                sum_p = p.sum()
                if sum_p == 0.0 or not np.isfinite(sum_p):
                    p = rng.random(n)
                    sum_p = p.sum()
                p = p / sum_p

                # tiny random jitter to escape flat regions
                if rng.random() < 0.01:
                    idx = rng.integers(0, n)
                    delta = (rng.random() - 0.5) * 0.01 * p[idx]
                    p[idx] = max(p[idx] + delta, 0.0)
                    p = p / p.sum()

                # early exit if we are within 1 % of the information‑theoretic lower bound
                lower_bound = 2.0 * n / (2 * n - 1)
                if cur_score <= lower_bound * 1.01:
                    break

    print("Final best score:", best_score,
          "length:", len(best_seq) if best_seq else None)
    return best_seq if best_seq is not None else []
