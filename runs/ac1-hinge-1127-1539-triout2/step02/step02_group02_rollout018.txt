def _run_ipgd(a0, n, deadline, beta=10.0, eta0=0.5, decay=0.9999):
    a = a0.copy()
    S = np.sqrt(2.0*n)
    # Ensure sum normalized
    a *= S / a.sum()
    best_score = np.inf
    best_seq = a.copy()
    best_max = np.inf

    L = 1 << ((3*n - 2 - 1).bit_length())
    it = 0

    eta = eta0
    while time.time() < deadline and it < max_iters:
        it += 1
        # FFT of a
        Fa = np.fft.rfft(a, n=L)
        # conv = a * a
        conv = np.fft.irfft(Fa * Fa, n=L)[:2*n - 1]
        max_conv = conv.max()
        # compute softmax weights
        # Optionally adjust beta schedule: start small for smoothing; increase.
        exp_w = np.exp(beta * (conv - max_conv))
        w = exp_w / exp_w.sum()
        # Compute gradient via correlation
        Fw = np.fft.rfft(w, n=L)
        corr = np.fft.irfft(np.conj(Fa) * Fw, n=L)[:n]  # length n
        grad = 2.0 * corr
        # exponentiated update
        a = a * np.exp(-eta * grad)
        # renorm
        a = np.maximum(a, 0.0)
        total = a.sum()
        if total == 0:
            a.fill(S/n)
        else:
            a *= (S / total)
        # apply jitter occasionally
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * 0.01
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()
        # decay eta
        eta *= decay
        # monitor improvement
        if max_conv < best_max - 1e-12:
            best_max = max_conv
            # evaluate objective (same as max_conv) using evaluate_sequence maybe to be safe
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[improve] time={time.time():.1f}s score={best_score:.6f} n={n} iter={it} max={max_conv:.6f}")
    return best_seq, best_score

def _run_optimize_one_start(a0, n, deadline,
                            eta0=0.5, beta_mom=0.9, beta_softmax=5.0,
                            decay=0.9998, jitter_prob=0.02,
                            max_iters=20000):
    a = a0.copy()
    S = np.sqrt(2.0 * n)
    best_score = float('inf')
    best_seq = a.copy()
    best_max = float('inf')

    v = np.zeros_like(a)
    eta = eta0

    # Precompute FFT length
    conv_len = 2*n - 1
    L = 1 << ((3*n - 2 - 1).bit_length())  # next pow2 >= 3n-2

    while time.time() < deadline and max_iters > 0:
        # compute FFT once
        Fa = np.fft.rfft(a, n=L)
        # convolution
        conv_full = np.fft.irfft(Fa * Fa, n=L)[:conv_len]
        cur_max = conv_full.max()
        # compute softmax weights
        exp_w = np.exp(beta_softmax * (conv_full - cur_max))
        w = exp_w / exp_w.sum()
        # compute gradient via correlation
        Fw = np.fft.rfft(w, n=L)
        corr_full = np.fft.irfft(np.conj(Fa) * Fw, n=L)[:n]  # lags 0..n-1
        grad = 2.0 * corr_full
        # momentum
        v = beta_mom * v + (1.0 - beta_mom) * grad
        # exponentiated update
        a = a * np.exp(-eta * v)
        # clip nonnegative
        a = np.maximum(a, 0.0)
        # renormalize
        total = a.sum()
        if total == 0:
            a.fill(S / n)
        else:
            a *= (S / total)

        # jitter
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * 0.01
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()

        eta *= decay
        max_iters -= 1

        # monitor improvement
        if cur_max < best_max - 1e-12:
            best_max = cur_max
            # external evaluation
            try:
                ext = evaluate_sequence(a.tolist())
                if ext < best_score:
                    best_score = ext
                    best_seq = a.copy()
                    print(f"[local improve] time={time.time():.1f}s score={best_score:.6f} n={n} iter={max_iters} cur_max={cur_max:.6f}")
            except Exception:
                pass

    return best_seq, best_score

"""
Improved search for a step‑function with low evaluation score.

The algorithm is a smooth‑max (soft‑max) gradient descent with
multiplicative updates (exponentiated gradient) and momentum.
For a vector a of length n we minimise the quantity

    max_k (a * a)[k]                                 (1)

under the linear constraint Σ a_i = √(2 n) and a_i ≥ 0.
Because the external evaluation returns
    2 n · max(a*a) / (Σa)²  = max(a*a)  (after the normalisation)
the internal surrogate (1) is exactly the score we want to minimise.

The surrogate max is approximated by a smooth soft‑max:
w_k ∝ exp(β·(conv_k‑max_conv)).  The gradient of the
soft‑max objective is a cross‑correlation between a and the weight vector w.
All convolutions and correlations are performed with FFTs, so each iteration
costs only O(n log n).

Multiple independent restarts with different lengths and random
initialisations are performed until the 1000 s budget is exhausted.
The best sequence found is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Return a probability vector w with the same length as conv,
    obtained by applying a temperature‑scaled softmax to conv.
    """
    max_val = conv.max()
    # subtract max for numerical stability
    exp_w = np.exp(beta * (conv - max_val))
    # at least one entry (the max) is 1, so sum > 0
    return exp_w / exp_w.sum()


def _gradient_full(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the gradient of the soft‑max objective w.r.t. a.
    Gradient_j = 2 * Σ_i a_i * w_{i+j}  (linear cross‑correlation).

    The computation is performed via FFTs:
        grad = 2 * corr(a, w)
    where corr is the (non‑reversed) cross‑correlation.
    """
    # FFT of a and w (both zero‑padded to fft_len)
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w, n=fft_len)

    # cross‑correlation via inverse FFT of conj(Fa) * Fw
    corr = np.fft.irfft(np.conj(Fa) * Fw, n=fft_len)[: a.shape[0]]
    return 2.0 * corr


def _run_optimize_one_start(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            eta0: float = 0.5,
                            beta_mom: float = 0.9,
                            beta_soft: float = 5.0,
                            decay: float = 0.9998,
                            jitter_prob: float = 0.02,
                            max_iters: int = 15000) -> tuple[np.ndarray, float]:
    """
    Optimise a single start vector a0.
    Returns the best vector seen and its external score.
    """
    a = a0.copy()
    S = np.sqrt(2.0 * n)               # required L1‑norm

    best_seq = a.copy()
    best_score = float('inf')
    best_max = float('inf')

    v = np.zeros_like(a)               # momentum buffer
    eta = eta0
    it = 0

    conv_len = 2 * n - 1
    fft_len = 1 << ((3 * n - 2 - 1).bit_length())   # >= 3n‑2

    while time.time() < deadline and it < max_iters:
        it += 1

        # ---- self‑convolution (full) ----
        Fa = np.fft.rfft(a, n=fft_len)
        conv_full = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
        cur_max = conv_full.max()

        # ---- soft‑max weights over the convolution entries ----
        w = _softmax_weights(conv_full, beta_soft)

        # ---- gradient of the soft‑max objective ----
        g = _gradient_full(a, w, fft_len)

        # ---- momentum ----
        v = beta_mom * v + (1.0 - beta_mom) * g

        # ---- exponentiated (multiplicative) update ----
        a = a * np.exp(-eta * v)

        # protect against tiny negatives caused by numerical error
        a = np.maximum(a, 0.0)

        # ---- re‑normalise to keep Σa = √(2 n) ----
        total = a.sum()
        if total == 0.0:
            a.fill(S / n)
        else:
            a *= (S / total)

        # ---- occasional jitter to escape plateaus ----
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * 0.01
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()

        # ---- step‑size decay ----
        eta *= decay

        # ---- monitor improvement ----
        if cur_max < best_max - 1e-12:
            best_max = cur_max
            try:
                ext = evaluate_sequence(a.tolist())
                if ext < best_score:
                    best_score = ext
                    best_seq = a.copy()
                    print(f"[local improve] time={time.time():.1f}s  "
                          f"score={best_score:.6f}  n={n}  iter={it}  max={cur_max:.6f}")
            except Exception:
                pass

    return best_seq, best_score
