def hill_climb(seq, target_sum, deadline):
    a = seq.copy()
    best = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng()
    n = a.size
    max_frac = 0.02
    # Determine top K based on sorted order (e.g., top 500)
    # We'll recompute donors and receivers every 2000 iteration.
    it = 0
    while time.time() < deadline:
        # recompute donor/receiver lists periodically
        if it % 1000 == 0:
            # donors: indices with large contributions to current max convolution
            conv = _conv_fft(a)  # compute whole conv maybe heavy each 1000 iteration
            k_max = int(conv.argmax())
            # Determine donors based on contributions weight = a_i * a_{k_max-i}
            lo = max(0, k_max-(n-1))
            hi = min(k_max, n-1)
            idx = np.arange(lo, hi+1)
            weight = a[idx] * a[k_max - idx]
            # sort to find top weight donors and low weight receivers
            sorted_idx = idx[np.argsort(weight)]
            donors = sorted_idx[-min(200, len(sorted_idx)):]  # top
            receivers = sorted_idx[:min(200, len(sorted_idx))]  # low
        donor = rng.choice(donors)
        receiver = rng.choice(receivers)
        if donor == receiver:
            continue
        delta = a[donor] * rng.random() * max_frac
        if delta <= 0:
            continue
        a[donor] -= delta
        a[receiver] += delta
        # compute new score
        new_score = _score(a)
        # Accept if better
        if new_score < best_score:
            best_score = new_score
            best = a.copy()
            # optionally update donors/receivers earlier? ignore
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
        it += 1

    return best, best_score

"""
Improved search for a low‑score step function.

Algorithm Overview
------------------
1. **Warm‑start** – use the previously best sequence (if any) resized to the
   current problem size, otherwise a simple power‑law.
2. **Random parametric search** – generate many two‑component mixtures of
   power‑law × exponential shapes, keep the best (≈10 % of the time budget).
3. **Projected sub‑gradient descent** on the exact max‑convolution (first pass).
4. **Deterministic mass‑transfer with hill‑climbing** – repeatedly move a
   fraction of mass from indices that contribute heavily to the current
   maximal convolution entry to indices that contribute little.  The move is
   accepted only if it improves the objective, otherwise it is reverted.
   (≈25 % of the remaining budget.)
5. **Second pass of sub‑gradient descent** – fine‑tuning with the remaining
   time.
6. Trim trailing zeros, evaluate, store for warm‑starts and return.

All steps keep the ℓ¹‑norm fixed to `sqrt(2·n)` and maintain non‑negativity,
so the final candidate can be fed directly into the provided
`evaluate_sequence` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution / scoring utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  With sum(a)=√(2n) the score equals max(a*a)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))

# Simple shape generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Sequence a_i = (i+1)^{-p} normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _random_param_sequence(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random mixture of up to two power‑law × exponential components.
    The result is normalised to the required ℓ¹‑norm.
    """
    i = np.arange(n, dtype=np.float64)

    # decide whether to use one or two components
    if rng.random() < 0.55:
        # single component
        p = rng.uniform(0.1, 2.0)
        lam = rng.exponential(scale=1.0) * 4.0   # typical λ in [0,≈12]
        off = rng.integers(0, 30)
        seq = (i + off + 1.0) ** (-p) * np.exp(-lam * i / n)
    else:
        # two components
        w = rng.random()
        p1 = rng.uniform(0.1, 2.0)
        lam1 = rng.exponential(scale=1.0) * 4.0
        off1 = rng.integers(0, 30)
        p2 = rng.uniform(0.1, 2.0)
        lam2 = rng.exponential(scale=1.0) * 4.0
        off2 = rng.integers(0, 30)
        comp1 = (i + off1 + 1.0) ** (-p1) * np.exp(-lam1 * i / n)
        comp2 = (i + off2 + 1.0) ** (-p2) * np.exp(-lam2 * i / n)
        seq = w * comp1 + (1.0 - w) * comp2

    # normalise to the required ℓ¹‑norm
    s = seq.sum()
    if s == 0.0:
        # fallback – power law
        seq = _power_law_sequence(n, 0.55, target_sum)
    else:
        seq = seq * (target_sum / s)
    return seq

# ----------------------------------------------------------------------


def _subgradient_max_descent(a0: np.ndarray,
                             target_sum: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on max(conv) with a simple diminishing step‑size."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())

        # gradient of the active convolution entry:
        grad = np.zeros_like(a)
        lo = max(0, k_max - (a.shape[0] - 1))
        hi = min(k_max, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        grad[idx] = 2.0 * a[partner]

        norm = np.linalg.norm(grad)
        if norm == 0:
            break

        # safe step that does not break non‑negativity before projection
        safe_eta = np.min(np.where(grad > 0, a / grad, np.inf))
        eta = min(0.5 * safe_eta, 0.5 / np.sqrt(it + 1))

        a = a - eta * grad
        # Euclidean projection onto the simplex {x ≥ 0, Σx = target_sum}
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:4d} score={best_val:.6f}")

        # occasional jitter to escape flat plateaus
        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        it += 1

    return best_a, best_val


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _hill_climb_refine(a0: np.ndarray,
                       target_sum: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a modest fraction of mass from indices that heavily
    contribute to the current maximal convolution entry to indices that
    contribute little.  Only improving moves are kept.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    n = a.shape[0]

    # parameters
    max_frac = 0.04               # at most 4 % of donor mass per move
    update_interval = 500         # recompute donor/receiver sets every 500 steps
    it = 0

    # placeholder donor/receiver sets
    donors = np.arange(n)
    receivers = np.arange(n)

    while time.time() < deadline:
        # recompute the sets from the current maximal entry
        if it % update_interval == 0:
            conv = _conv_fft(a)
            k_max = int(conv.argmax())
            lo = max(0, k_max - (n - 1))
            hi = min(k_max, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            weight = a[idx] * a[k_max - idx]          # contribution a_i·a_{k_max-i}

            # pick top‑5 % as donors, bottom‑5 % as receivers (at least one each)
            num = max(1, int(0.05 * len(idx)))
            order = np.argsort(weight)                # ascending
            receivers = idx[order[:num]]
            donors = idx[order[-num:]]
            # if for some reason a set becomes empty, fall back to whole index set
            if donors.size == 0:
                donors = idx
            if receivers.size == 0:
                receivers = idx

        donor = rng.choice(donors)
        receiver = rng.choice(receivers)
        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * max_frac
        if max_delta <= 0.0:
            it += 1
            continue
        delta = rng.random() * max_delta

        # propose move
        a[donor] -= delta
        a[receiver] += delta

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            # keep the move
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta

        it += 1

    return best_a, best_val
