"""
Step‑function optimiser – extended search.

The evaluator is scale‑invariant; after fixing the ℓ1‑norm to
√(2·n) the score is exactly the peak of the self‑convolution.
The algorithm therefore tries to minimise that peak.

The search consists of
  1. a fast, very diverse random exploration over many lengths and
     several parametric families (including a sparse “Sidon‑like”
     starter);
  2. scaling the current best candidate to a handful of longer
     lengths;
  3. several independent Adam runs (mirrored‑descent on the simplex)
     with a τ‑soft‑max of the convolution;
  4. a sub‑gradient descent that follows the hard maximum;
  5. a short simulated‑annealing / hill‑climb phase;
  6. final sanity check with the provided ``evaluate_sequence``.

All heavy work (convolutions, gradients) is performed with an
FFT‑based implementation, which makes the code fast enough to explore
vectors with up to ≈ 200 00 entries inside the 1000 s budget.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft
from numpy.random import default_rng

# ----------------------------------------------------------------------
# low‑level convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a∗a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = rfft(a, n=f_len)
    conv = irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a∗a."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation resize."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# τ‑soft‑max gradient (used by Adam)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # grad = 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = rfft(mask, n=f_len)
    Fa_rev = rfft(a_rev, n=f_len)
    conv = irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser (mirrored descent on the simplex)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex (log‑softmax parametrisation).
    Returns the best sequence found and its (hard) peak value.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                     # target ℓ1‑norm

    # initialise logits from a feasible a0 (entropic parametrisation)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                         # logits (log‑softmax)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.45

    # temperature starts at a few times the current peak
    tau = max(_full_convolution_fft(a0).max(), 5.0)

    best_a = a0.copy()
    best_peak = _full_convolution_fft(a0).max()
    it = 0

    while time.time() < deadline:
        it += 1

        # recover a from logits
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # gradient (τ‑soft‑max)
        g, peak = _softmax_gradient(a, tau)

        # keep best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # chain rule to logits
        mu = np.dot(a, g) / S                 # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 6.0 * S:
            grad_p = grad_p * (6.0 * S / clip_val)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # schedules (very slow decay)
        tau *= 0.99995
        lr *= 0.99995

        # occasional jitter to escape plateaus
        if it % 800 == 0:
            p += rng.normal(scale=0.006, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Sub‑gradient descent on the hard max
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple sub‑gradient descent that always follows the entry of the
    convolution attaining the current maximum.  After each step the
    vector is projected back onto the simplex (positivity + fixed ℓ1‑norm).
    Returns the best sequence seen and its hard peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    a = a0.copy()
    best_a = a.copy()
    best_val = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1

        b = _full_convolution_fft(a)
        k = int(b.argmax())
        val = b[k]

        if val < best_val:
            best_val = val
            best_a = a.copy()

        # sub‑gradient of (a*a)_k
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        jdx = k - idx
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[jdx]

        step = 0.8 * S / (np.sqrt(it) + 1.0)

        a = a - step * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, S / n)
        else:
            a = a * (S / s)

        # occasional random jitter
        if it % 500 == 0:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                frac = rng.uniform(0.02, 0.12)
                amount = a[i] * frac
                a[i] -= amount
                a[j] += amount
                a = a * (S / a.sum())

    return best_a, best_val

# Parametric families for random initialisation
# ----------------------------------------------------------------------
def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    raw = mix * edge + (1.0 - mix) * np.ones(n)
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    raw = mix * np.exp(-lam * d) + (1.0 - mix) * np.ones(n)
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    raw = mix * lin + (1.0 - mix) * rng.uniform(0.0, 0.25, size=n)
    return raw


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    raw = mix * prod.astype(float) ** (-p) + (1.0 - mix) * np.ones(n)
    return raw


def _init_cosine(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    raw = 1.0 - np.cos(np.pi * i / (n - 1))
    raw *= rng.uniform(0.5, 2.0)
    mix = rng.uniform(0.0, 1.0)
    raw = mix * raw + (1.0 - mix) * np.ones(n)
    return raw


def _init_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    """Bell‑shaped centred Gaussian."""
    centre = (n - 1) / 2.0
    sigma = rng.uniform(0.1, 0.5) * n
    i = np.arange(n)
    raw = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    mix = rng.uniform(0.0, 1.0)
    raw = mix * raw + (1.0 - mix) * np.ones(n) * 0.1
    return raw


def _init_sparse(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Very sparse start: a few random positions with equal weight.
    The number of non‑zero entries is chosen around √n.
    """
    s_min = max(3, int(np.sqrt(n)))
    s_max = max(s_min + 1, int(2.0 * np.sqrt(n)))
    s = int(rng.integers(s_min, s_max + 1))
    indices = rng.choice(n, size=s, replace=False)
    raw = np.zeros(n)
    raw[indices] = 1.0
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.ones(n)


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    With probability ≈0.2 we warm‑start from ``warm_seq`` (scaled).
    Otherwise we draw from a randomly chosen parametric family.
    The returned vector satisfies sum = √(2·n) and all entries ≥0.
    """
    S = np.sqrt(2.0 * n)

    if warm_seq is not None and rng.random() < 0.23:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
        return a

    families = [
        _init_powerlaw,
        _init_exponential,
        _init_linear,
        _init_ushape,
        _init_cosine,
        _init_gaussian,
        _init_sparse,
        _init_uniform,
    ]
    init_fun = rng.choice(families)
    raw = init_fun(n, rng)
    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# cheap stochastic hill‑climb (simulated‑annealing style)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                rng: np.random.Generator,
                start_temp: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Small simulated‑annealing walk that moves a random fraction of mass
    between two positions.  The temperature decays linearly to zero.
    Returns the best vector found and its peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    a = a0.copy()
    best_a = a.copy()
    best_val = _max_convolution(a)

    it = 0
    total_iters = int(1e5)  # just a safety upper bound
    while time.time() < deadline and it < total_iters:
        it += 1
        # linear temperature schedule
        T = start_temp * max(0.0, (deadline - time.time()) / (deadline - (deadline - 5.0)))
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        frac = rng.uniform(0.02, 0.12)
        delta = a[i] * frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = a_new * (S / a_new.sum())  # renormalise (keeps sum fixed)

        val = _max_convolution(a_new)

        if val < best_val:
            best_val = val
            best_a = a_new.copy()

        if val < _max_convolution(a) or rng.random() < np.exp(-(val - _max_convolution(a)) / (T + 1e-12)):
            a = a_new

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence as a plain Python list and stores it in the
    global variable ``height_sequence_1`` for warm‑starts in future runs.
    """
    rng = default_rng(seed)
    end_time = time.time() + budget_s - 4.0   # small safety margin

    # ------------------------------------------------------------------
    # warm‑start handling
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # ------------------------------------------------------------------
    # Phase 1 – fast random exploration (≈ 25 % of budget)
    # ------------------------------------------------------------------
    explore_deadline = end_time - 350.0   # leave ~350 s for refinement
    best_seq = None
    best_score = np.inf
    attempts = 0

    while time.time() < explore_deadline:
        attempts += 1
        # favour a few‑thousand entries but allow some variety
        if rng.random() < 0.70:
            n = int(rng.integers(1800, 8000))
        else:
            n = int(rng.integers(8000, 20000))

        a0 = _initial_vector(n, rng, warm_seq)
        val = _max_convolution(a0)

        if val < best_score:
            best_score = val
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempts:4d} n={n:5d}  new best={best_score:.6f}")

    if best_seq is None:
        # fallback – uniform of moderate size
        best_n = 4000
        best_seq = np.full(best_n, np.sqrt(2.0 * best_n) / best_n)
        best_score = _max_convolution(best_seq)

    # ------------------------------------------------------------------
    # Phase 2 – upscale to a few larger lengths (no optimisation)
    # ------------------------------------------------------------------
    cand_lengths = [best_n]
    for factor in [1.25, 1.5, 2.0, 3.0, 4.0]:
        L = int(best_n * factor)
        if 2000 <= L <= 25000:
            cand_lengths.append(L)
    cand_lengths = sorted(set(cand_lengths))

    cur_seq = best_seq
    cur_score = best_score
    cur_n = best_n

    for L in cand_lengths:
        if time.time() > end_time - 260.0:
            break
        a_scaled = _scale_sequence(cur_seq, L)
        a_scaled = np.clip(a_scaled, 0.0, None)
        S = np.sqrt(2.0 * L)
        a_scaled = a_scaled * (S / a_scaled.sum())
        val = _max_convolution(a_scaled)
        if val < cur_score:
            cur_score = val
            cur_seq = a_scaled.copy()
            cur_n = L
            print(f"[scale] length={L:5d}  improved max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – long Adam optimisation on the current best length
    # ------------------------------------------------------------------
    remaining = end_time - time.time() - 150.0   # reserve time for later phases
    if remaining > 20.0:
        # run several independent Adam runs, each gets a slice of the time
        n_restart = max(1, int(remaining // 70))   # ~1 min per restart
        per_run = remaining / n_restart
        for r in range(n_restart):
            if time.time() > end_time - 150.0:
                break
            # small perturbation of the current best
            noise = rng.normal(scale=0.015 * np.sqrt(2.0 * cur_n) / cur_n, size=cur_n)
            a_start = np.maximum(cur_seq + noise, 0.0)
            a_start = a_start * (np.sqrt(2.0 * cur_n) / a_start.sum())
            a_opt, peak = _adam_optimize(a_start,
                                          time.time() + per_run,
                                          rng)
            if peak < cur_score:
                cur_score = peak
                cur_seq = a_opt.copy()
                print(f"[adam] restart={r} new best max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – sub‑gradient descent (final polishing)
    # ------------------------------------------------------------------
    remaining = end_time - time.time() - 50.0
    if remaining > 5.0:
        a_opt, peak = _subgradient_descent(cur_seq,
                                           time.time() + remaining,
                                           rng)
        if peak < cur_score:
            cur_score = peak
            cur_seq = a_opt.copy()
            print(f"[subgrad] final max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – cheap random hill‑climb / simulated annealing
    # ------------------------------------------------------------------
    while time.time() < end_time - 3.0:
        a_opt, peak = _hill_climb(cur_seq,
                                  time.time() + 0.8,
                                  rng,
                                  start_temp=0.025)
        if peak + 1e-9 < cur_score:
            cur_score = peak
            cur_seq = a_opt.copy()
            print(f"[hill] improved max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    final_seq = cur_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f}  internal max={cur_score:.6f}  length={len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
