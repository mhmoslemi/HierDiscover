It follows the same multi‑stage philosophy as the reference code but replaces the final sub‑gradient phase by an **Adam‑projected gradient descent on a soft‑max surrogate of the max‑convolution**, which provides much larger effective steps while still respecting the simplex constraint.  The algorithm also keeps the earlier cheap parametric warm‑start and a deterministic mass‑transfer polishing step.

```python
"""
Propose a step‑function (non‑negative sequence) with a low max‑convolution
score.  The pipeline:

1. Warm‑start with a two‑parameter power‑law / exponential family (grid
   search on a reduced size, then resize to the full problem size).
2. Refine the shape using Adam‑projected gradient descent on the
   soft‑max approximation of the max‑convolution.  The temperature β is
   increased slowly so the surrogate becomes tighter.
3. Apply a deterministic “mass‑transfer’’ polishing step that moves a
   small fraction of mass from indices that contribute most to the
   current maximal convolution entry toward indices that contribute the
   least.
4. Return the best sequence (trim trailing zeros) and store it globally
   for warm‑starts in future calls.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution / scoring utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score_from_array(a: np.ndarray) -> float:
    """Exact objective value (max‑convolution) for a normalized array."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Simple helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i = (i+1)^{-p} normalised to Σa_i = target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max surrogate of max‑convolution and its gradient
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    Returns (grad, conv) where conv is the full convolution.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_c = np.max(conv)
    # stable soft‑max
    exps = np.exp(beta * (conv - max_c))
    p = exps / np.sum(exps)                     # shape (2n‑1,)

    # grad = 2 * (rev(a) * p) (convolution)
    a_rev = a[::-1]
    rev_fft = np.fft.rfft(a_rev, n=size)
    p_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(rev_fft * p_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1: n - 1 + n]     # keep only n entries
    return grad, conv


# ----------------------------------------------------------------------
# Adam optimiser (projected onto the simplex)
# ----------------------------------------------------------------------


def _adam_softmax_optimize(a0: np.ndarray,
                           target_sum: float,
                           deadline: float):
    """
    Adam projected gradient descent on the soft‑max surrogate.
    Returns the best sequence and its exact max‑convolution value.
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    n = a.shape[0]
    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5                     # base learning rate (absolute)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_val = _score_from_array(a)

    # temperature schedule
    beta = 1.0
    max_beta = 16384.0
    beta_mul = 1.02
    beta_update_interval = 200          # iterations

    rng = np.random.default_rng()
    it = 1
    while time.time() < deadline:
        # gradient of the smooth surrogate
        g, _ = _softmax_gradient(a, beta)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # bias‑correction
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        # evaluate exact objective
        cur_val = _score_from_array(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.2f} score={best_val:.6f}")

        # temperature schedule
        if it % beta_update_interval == 0 and beta < max_beta:
            beta = min(beta * beta_mul, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer polishing
# ----------------------------------------------------------------------


def _mass_transfer_refine(a0: np.ndarray,
                         target_sum: float,
                         deadline: float):
    """
    Repeatedly move a small fraction of mass from indices that
    contribute most to the current maximal convolution entry towards
    indices that contribute the least.
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)
    n = a.shape[0]

    best_a = a.copy()
    best_val = _score_from_array(a)
    rng = np.random.default_rng()

    conv = _conv_fft(a)

    while time.time() < deadline:
        k_max = int(conv.argmax())

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contributions a[i] * a[k_max-i]
        contrib = a[idx] * a[k_max - idx]

        donor = idx[np.argmax(contrib)]

        # pick a receiver among the 10‑percentile low contributors
        low_thr = np.percentile(contrib, 10)
        low_cands = idx[contrib <= low_thr]
        if low_cands.size:
            receiver = low_cands[np.argmin(contrib[contrib <= low_thr])]
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            # fallback to a random move
            donor = rng.choice(idx)
            receiver = rng.integers(n)
            if donor == receiver:
                continue

        max_frac = 0.04
        delta = a[donor] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta

        conv = _conv_fft(a)
        cur_val = _score_from_array(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Transfer improve score={best_val:.6f}")
        else:
            # revert unsuccessful move
            a[donor] += delta
            a[receiver] -= delta
            conv = _conv_fft(a)

        # occasional renormalisation (numeric drift)
        cur_sum = a.sum()
        if abs(cur_sum - target_sum) > 1e-9:
            a *= target_sum / cur_sum

    return best_a, best_val

# Cheap parametric warm‑start on a reduced size
# ----------------------------------------------------------------------


def _grid_param_search(small_n: int,
                       target_sum: float,
                       deadline: float):
    """
    Search a two‑parameter family a_i = (i+1)^(-p)·exp(-λ·i/small_n).
    Returns the best sequence found (still on the small size).
    """
    p_vals = np.linspace(0.2, 1.2, 17)
    lam_vals = np.linspace(0.0, 6.0, 13)

    best_seq = None
    best_score = np.inf
    i = np.arange(small_n, dtype=np.float64)

    for p in p_vals:
        for lam in lam_vals:
            if time.time() > deadline:
                break
            seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
            seq = seq * (target_sum / seq.sum())
            val = _score_from_array(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()
                print(f"[{time.time():.2f}] Grid best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")
        else:
            continue
        break

    rng = np.random.default_rng()
    while time.time() < deadline:
        p = rng.uniform(0.2, 1.2)
        lam = rng.exponential(1.0) * 3.0
        seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
        seq = seq * (target_sum / seq.sum())
        val = _score_from_array(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Random best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")

    return best_seq, best_score

    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")
```
