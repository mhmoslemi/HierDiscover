    b = _full_convolution(a)
    best_a = a.copy()
    best_score = b.max()
    cur_a = a.copy()
    cur_b = b.copy()
    cur_score = best_score

    rng = np.random.default_rng()

        elapsed = time.time() - start
        total = deadline - start

        # choose move
        # directed half time
        if rng.random() < 0.6:
            M = int(np.argmax(cur_b))
            i_low = max(0, M - (n-1))

            if contrib.size == 0:

                continue

            j_min = max(0, M - (n-1))

            cand_sorted = np.argsort(counterpart)

                    j = cand
                    break

            i = rng.integers(0, n)
            if cur_a[i] < 1e-12:
                continue

def _adam_optimize(a0, target_sum, deadline):
    a = a0.copy()
    n = a.size
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    base_lr = 0.15 * target_sum
    beta = 2.0
    beta_max = 1e6
    beta_factor = 2.0
    t=0
    rng = np.random.default_rng()
    while time.time() < deadline:
        t+=1
        g,_ = _gradient_softmax(a, beta)
        # Adam step
        m = beta1*m + (1-beta1)*g
        v = beta2*v + (1-beta2)*(g*g)
        m_hat = m/(1 - beta1**t)
        v_hat = v/(1 - beta2**t)
        step = base_lr * m_hat/(np.sqrt(v_hat)+eps)
        a -= step
        # project onto simplex
        a = _project_to_simplex(a, target_sum)
        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4/(beta+1), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
        if t % 300 == 0 and beta < beta_max:
            beta = min(beta*beta_factor, beta_max)
    return a

"""
Hybrid simulated‑annealing + projected Adam for the max‑convolution problem.

The metric to minimise is
    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)
which, after the standard normalisation Σ seq = √(2n), equals the
maximal entry of the (linear) convolution of the sequence with itself.
The algorithm works as follows:

1.  Load a previously found solution (``height_sequence_1``) if it exists.
2.  Upsample that solution to larger lengths according to a schedule.
    For each length we first run a simple simulated‑annealing that moves
    mass between indices, accepting downhill moves and occasional uphill
    moves according to an exponential temperature schedule.
3.  Polish the result with a few iterations of projected Adam on a
    smooth log‑sum‑exp surrogate of the maximal convolution entry.
4.  Trim trailing zeros and return the best sequence found before the
    wall‑clock deadline (default 1000 s).

The search is fully vectorised, uses FFTs for the Adam gradient and
updates the convolution exactly in O(n) per SA move.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT (returns length 2·n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a*a (assumes a already has the correct ℓ¹ norm)."""
    return float(np.max(_full_convolution(a)))


# ----------------------------------------------------------------------
# Random initialisation & up‑sampling
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    # simple Dirichlet (almost uniform) plus tiny jitter
    a = rng.dirichlet(alpha=np.ones(n))
    a = a * target
    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length len(a) to new_n, then projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with g = 2·(a ⋆ p) and p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float,
                    deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max surrogate.
    Runs until ``deadline`` (wall‑clock time).
    """
    a = a0.copy()
    n = a.size

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.15 * target_sum
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4 / (beta + 1.0),
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Simulated‑annealing local search (exact convolution updates)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple SA that moves a (large) fraction of mass from a source index
    to a destination.  The convolution is updated in O(n) per move.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    n = a.size
    b = _full_convolution(a)

    best_a = a.copy()
    best_score = b.max()
    cur_a = a.copy()
    cur_b = b.copy()
    cur_score = best_score

    start = time.time()
    rng = np.random.default_rng()
    T0 = best_score * 0.3 + 1e-3                 # initial temperature

    while time.time() < deadline:
        elapsed = time.time() - start
        total = deadline - start
        # exponential cooling: halve the temperature by the end
        T = T0 * 0.5 ** (elapsed / total)

        # ---------- choose a move ----------
        if rng.random() < 0.65:
            # directed move: attack the current worst convolution entry
            M = int(np.argmax(cur_b))
            i_low = max(0, M - (n - 1))
            i_high = min(n - 1, M)
            idx = np.arange(i_low, i_high + 1, dtype=int)
            if idx.size == 0:
                continue
            contrib = cur_a[idx] * cur_a[M - idx]
            i = idx[np.argmax(contrib)]

            # destination that contributes least to the same entry
            j_min = max(0, M - (n - 1))
            j_max = min(n - 1, M)
            candidates = np.arange(j_min, j_max + 1, dtype=int)
            counterpart = np.full(n, np.inf)
            counterpart[candidates] = cur_a[M - candidates]
            order = np.argsort(counterpart)
            for cand in order:
                if cand != i:
                    j = cand
                    break
        else:
            i = rng.integers(0, n)
            if cur_a[i] < 1e-14:
                continue
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)

        max_delta = cur_a[i]
        if max_delta <= 1e-14:
            continue
        # move a random fraction of the source mass
        fraction = rng.random() * 0.5 + 0.05      # between 0.05 and 0.55
        delta = max_delta * fraction

        # ---------- update convolution ----------
        b_new = cur_b.copy()
        b_new[i:i + n] -= 2.0 * delta * cur_a
        b_new[j:j + n] += 2.0 * delta * cur_a
        if 2 * i < b_new.size:
            b_new[2 * i] += delta * delta
        if 2 * j < b_new.size:
            b_new[2 * j] += delta * delta
        kij = i + j
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta
        new_score = b_new.max()

        # ---------- accept / reject ----------
        if new_score < cur_score or rng.random() < math.exp((cur_score - new_score) / (T + 1e-12)):
            cur_a[i] -= delta
            cur_a[j] += delta
            cur_b = b_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cur_a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence`` while respecting the wall‑clock budget.
    The routine returns the best sequence (as a Python list) found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 1)  Seed from a previous run if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            n0 = prev.size
            target0 = math.sqrt(2.0 * n0)
            prev = _project_to_simplex(prev, target0)
            best_seq = prev.copy()
            best_score = _max_convolution(best_seq)
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 "
                  f"(n={n0}) score={best_score:.6f}")
        except Exception as e:
            print(f"Seeding error: {e}")

    # ------------------------------------------------------------------
    # 2)  Fallback initialisation (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        n0 = 2000
        best_seq = _random_initial_sequence(n0, rng)
        target0 = math.sqrt(2.0 * n0)
        best_seq = _project_to_simplex(best_seq, target0)
        best_score = _max_convolution(best_seq)
        print(f"[{time.time():.1f}] Random start n={n0} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3)  Length schedule – larger dimensions are tried later
    # ------------------------------------------------------------------
    schedule = [2000, 4000, 8000, 20000, 50000]

    for n in schedule:
        if time.time() >= deadline - 2.0:
            break

        # initialise a candidate for this length
        if best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = best_seq.copy()
        target_sum = math.sqrt(2.0 * n)
        a_init = _project_to_simplex(a_init, target_sum)

        # allocate a time slice for this dimension
        remaining = deadline - time.time()
        if remaining < 10.0:
            break

        # ---- Phase A: Simulated annealing ----
        sa_time = remaining * 0.45
        sa_deadline = time.time() + sa_time
        a_sa, score_sa = _simulated_annealing(a_init, sa_deadline)
        if score_sa < best_score - 1e-12:
            best_seq = a_sa.copy()
            best_score = score_sa
            print(f"[{time.time():.1f}] SA improvement for n={n} score={best_score:.6f}")

        # ---- Phase B: Projected Adam polishing ----
        remaining = deadline - time.time()
        if remaining > 4.0:
            adam_time = remaining * 0.35
            adam_deadline = time.time() + adam_time
            a_adam = _adam_optimize(best_seq, target_sum, adam_deadline)
            score_adam = _max_convolution(a_adam)
            if score_adam < best_score - 1e-12:
                best_seq = a_adam.copy()
                best_score = score_adam
                print(f"[{time.time():.1f}] Adam improvement for n={n} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Final polishing on the current length
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        n = best_seq.size
        target = math.sqrt(2.0 * n)

        # longer Adam run on the remaining budget
        final_adam_deadline = deadline - 0.5
        best_seq = _adam_optimize(best_seq, target, final_adam_deadline)

        # a short final SA run
        final_sa_deadline = deadline - 0.2
        best_seq, final_score = _simulated_annealing(best_seq, final_sa_deadline)
        if final_score < best_score - 1e-12:
            best_score = final_score
            print(f"[{time.time():.1f}] Final SA improvement: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Trim trailing near‑zeros
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # 6) Report the official evaluation
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:          # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
