        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi+1)
        partner = k_max - idx
        weights = a[idx] * a[partner]
        donor = idx[np.argmax(weights)]
        # choose receiver: low weight or low a
        # compute low percentile threshold
        low_thr = np.percentile(a, 10)
        low_idx = np.where(a <= low_thr)[0]

"""
Propose a candidate step‑function (non‑negative sequence) that 
minimises the benchmark score 2·n·max(a*a)/sum(a)^2.
The algorithm is a simple large‑scale local search that starts from the 
best known sequence (height_sequence_1 if available) and repeatedly performs 
targeted mass–transfer moves which directly reduce the maximal convolution 
entry.  The search runs until the given time budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (sum(a)²).
    For the current normalisation sum(a) ≈ sqrt(2·n) this is exactly
    the maximal convolution entry, but we keep the full expression
    for clarity.
    """
    n = a.size
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple families for a warm‑start (only used if no historic solution)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) shape, normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Local mass‑transfer refinement
# ----------------------------------------------------------------------


def _local_refine(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climbing that repeatedly moves a small fraction of mass from the
    index contributing most to the current maximal convolution entry to a weak
    index.  The move is kept only if it lowers the objective.
    """
    a = a0.copy()
    n = a.shape[0]

    # make sure the linear constraint is satisfied initially
    cur_sum = a.sum()
    if cur_sum != target_sum:
        a = a * (target_sum / cur_sum)

    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0
    # start with a generous step size and decay it over time
    step_frac = 0.08
    min_frac = 5e-4
    decay = 0.9995

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) identify the current maximal convolution entry and its
        #    most responsible component (donor)
        # --------------------------------------------------------------
        conv = _conv_fft(a)
        k_max = int(conv.argmax())
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        weights = a[idx] * a[partner]               # contribution a_i·a_{k_max-i}
        donor = idx[np.argmax(weights)]

        # --------------------------------------------------------------
        # 2) choose a receiver – an index with a small amount of mass.
        #    Using a low percentile makes the move efficient.
        # --------------------------------------------------------------
        low_thr = np.percentile(a, 10)
        low_candidates = np.where(a <= low_thr)[0]
        if low_candidates.size:
            receiver = rng.choice(low_candidates)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            # fallback to a random receiver if the deterministic choice fails
            receiver = rng.integers(n)
            if donor == receiver:
                it += 1
                continue

        # --------------------------------------------------------------
        # 3) compute a modest transfer amount
        # --------------------------------------------------------------
        delta = a[donor] * step_frac * rng.random()
        if delta <= 0.0:
            it += 1
            continue

        # --------------------------------------------------------------
        # 4) try the move
        # --------------------------------------------------------------
        a[donor] -= delta
        a[receiver] += delta

        # enforce non‑negativity (numerical safety)
        a = np.maximum(a, 0.0)

        # occasional renormalisation to keep the ℓ¹‑norm exact
        if it % 500 == 0:
            cur_sum = a.sum()
            if abs(cur_sum - target_sum) > 1e-12:
                a = a * (target_sum / cur_sum)

        val = _score(a)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] refine it={it:07d} score={best_val:.6f}")
        else:
            # revert – the move did not help
            a[donor] += delta
            a[receiver] -= delta

        # --------------------------------------------------------------
        # 5) schedule updates
        # --------------------------------------------------------------
        it += 1
        # slowly shrink the step fraction (geometric decay)
        if it % 1000 == 0:
            step_frac = max(step_frac * decay, min_frac)

        # occasional jitter to escape flat plateaus
        if it % 5000 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # tiny safety margin

    # ------------------------------------------------------------------
    # Warm‑start: use the previously found sequence if available,
    # otherwise fall back to a simple power‑law shape.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.array(globals()["height_sequence_1"], dtype=np.float64)
    else:
        n = 12000
        a = _power_law_sequence(n, 0.55, np.sqrt(2.0 * n))

    # ------------------------------------------------------------------
    # Normalise the ℓ¹‑norm to the canonical value sqrt(2·n)
    # (the objective is invariant under this scaling).
    # ------------------------------------------------------------------
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    a = a * (target_sum / a.sum())

    # ------------------------------------------------------------------
    # Intensive local refinement – the core of the search.
    # ------------------------------------------------------------------
    a_best, best_val = _local_refine(a, target_sum, deadline)

    # ------------------------------------------------------------------
    # Final clean‑up: trim trailing near‑zero entries.
    # ------------------------------------------------------------------
    nz = np.where(a_best > 1e-12)[0]
    if nz.size:
        a_best = a_best[: nz[-1] + 1]

    final_seq = a_best.tolist()

    # ------------------------------------------------------------------

        val = evaluate_sequence(final_seq)          # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for warm‑starts in a subsequent call
    globals()["height_sequence_1"] = final_seq

    return final_seq
