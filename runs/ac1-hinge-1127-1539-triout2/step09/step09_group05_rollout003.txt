def _smooth_descent(a0, deadline, rng):
    a = a0.copy()
    n = len(a)
    S = sqrt(2n)
    conv = _full_convolution_fft(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_val = cur_max
    # Define parameters
    initial_mu_factor = 0.2 # start with mu = mu_factor * cur_max
    mu = max(cur_max * initial_mu_factor, 1e-6)
    mu_decay = 0.995 # per iteration
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        conv = _full_convolution_fft(a)
        max_val = conv.max()
        # maybe recompute mu as decreasing factor
        # Keep mu as max(curr_mu * decay, epsilon)
        mu = max(mu * mu_decay, max_val * 0.01)  # ensure not too small
        # compute weight using softmax
        # subtract max for stability
        exp_scaled = np.exp((conv - max_val) / mu)
        sum_exp = exp_scaled.sum()
        w = exp_scaled / sum_exp
        # gradient via conv of w and a_rev
        a_rev = a[::-1]
        # convolution length conv_len = len(w) + n - 1
        conv_len = len(w) + n - 1
        f_len = _next_pow2(conv_len)
        Fw = np.fft.rfft(w, n=f_len)
        Fra = np.fft.rfft(a_rev, n=f_len)
        conv_w_rev = np.fft.irfft(Fw * Fra, n=f_len)[:conv_len]
        start = n - 1
        grad = 2.0 * conv_w_rev[start:start + n]
        # step size: try polyak-like using (max_val - best_val) / norm_sq
        norm_sq = np.dot(grad, grad)
        if norm_sq > 0:
            step = (max_val - best_val) / (norm_sq + 1e-12)
            # limit step
            step = np.clip(step, 1e-7, 0.1)
        else:
            step = 0.0

        # backtracking
        # compute candidate step with simple line search
        a_candidate = _project_onto_simplex(a - step * grad, S)
        cand_max = _max_convolution(a_candidate)
        # accept if better, else reduce step
        ls_iter = 0
        while cand_max > max_val and step > 1e-8 and ls_iter < 10:
            step *= 0.5
            a_candidate = _project_onto_simplex(a - step * grad, S)
            cand_max = _max_convolution(a_candidate)
            ls_iter += 1

        a = a_candidate
        max_val = cand_max  # update after step
        if max_val < best_val - 1e-12:
            best_val = max_val
            best_a = a.copy()
        # occasional jitter
        if iteration % 200 == 0:
            jitter = rng.normal(scale=0.001 * S, size=n)
            a = _project_onto_simplex(a + jitter, S)
    return best_a, best_val

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5

    warm_seq = globals().get('height_sequence_1', None)
    if warm_seq is not None:
        warm_seq = np.asarray(warm_seq, dtype=float)

    best_seq = None
    best_score = np.inf

    # Exploration loop
    while time.time() < end_time - 30: # reserve 30 sec for final
        # pick length n
        # Probability distribution: 0.6 for 4000-12000, 0.2 for 12000-20000, 0.2 for 2000-4000
        r = rng.random()
        if r < 0.6:
            n = int(rng.integers(4000, 12000))
        elif r < 0.8:
            n = int(rng.integers(12000, 20000))
        else:
            n = int(rng.integers(2000, 4000))
        # generate initial vector
        a0 = _initial_vector(n, rng, warm_seq)
        # run smoothing descent with slice approx 10 seconds or less
        slice_time = min(12.0, end_time - time.time() - 30)
        if slice_time <= 0:
            break
        a_opt, val = _smooth_descent(a0, time.time() + slice_time, rng)
        if val < best_score:
            best_score = val
            best_seq = a_opt.copy()

    # After exploration, run final refinement using subgradient to ensure monotonic improvements
    if best_seq is None:
        best_seq = np.full(5000, np.sqrt(2*5000)/5000)

    # Final subgradient refinement
    remaining = end_time - time.time() - 5

def _init_semicircle(n, rng):
    i = np.arange(n)
    mid = (n-1)/2
    radius = mid + 1
    vals = np.sqrt(np.maximum(0, 1 - ((i - mid)/radius)**2))
    mix = rng.uniform(0,1)
    base = np.ones(n)

"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

The search works in three stages:

1. **Broad random sampling** – generate candidates from many parametric
   families (power‑law, exponential, linear, U‑shape, sine, semi‑circle,
   raised‑cosine, uniform, Dirichlet).  Lengths are drawn between 2 000 and
   20 000 with a bias toward larger sizes.

2. **Smooth‑approximation gradient descent** – for each sampled start we
   minimise a smoothed version of the max‑convolution objective:
       f_μ(a) = μ·log(∑_k exp((a∗a)[k]/μ)).
   The gradient is obtained analytically via FFTs, after which the
   iterate is projected onto the simplex {a≥0, ∑a = √(2 n)}.
   The smoothing parameter μ is slowly decreased, turning the method
   into a sub‑gradient descent for the original objective.

3. **Final refinement** – the best sequence found is polished with a
   classic projected sub‑gradient method and a short simulated‑annealing
   local search.

All heavy work is performed with O(n log n) FFT convolutions; the
routine respects the 1000 s budget and stores the final construction in the
global variable ``height_sequence_1``.
"""

# ----------------------------------------------------------------------
# Basic FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution a∗a using a real FFT
    (length 2·len(a)‑1).
    """
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a∗a."""
    return _full_convolution_fft(a).max()


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – Duchi et al. O(n)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """
    Project vector v onto the simplex {w_i ≥ 0, Σ w_i = S}.
    """
    if S < 0:
        raise ValueError("Simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Parametric families for random initialisation

def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    return mix * edge + (1.0 - mix) * base


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    return mix * exp_part + (1.0 - mix) * base


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    return mix * lin + (1.0 - mix) * base


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    return mix * ush + (1.0 - mix) * base


def _init_sine(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    vals = np.sin(np.pi * (i + 1) / (n + 1))
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * vals + (1.0 - mix) * base


def _init_semicircle(n: int, rng: np.random.Generator) -> np.ndarray:
    """Semi‑circular bump, centred and non‑negative."""
    i = np.arange(n, dtype=float)
    mid = (n - 1) / 2.0
    radius = mid + 1.0
    vals = np.sqrt(np.maximum(0.0, 1.0 - ((i - mid) / radius) ** 2))
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * vals + (1.0 - mix) * base


def _init_cosine(n: int, rng: np.random.Generator) -> np.ndarray:
    """Raised cosine (half‑wave, non‑negative)."""
    i = np.arange(n, dtype=float)
    mid = (n - 1) / 2.0
    vals = np.cos(np.pi * (i - mid) / n)
    vals = np.maximum(0.0, vals)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * vals + (1.0 - mix) * base


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    return rng.random(n)


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    With probability ≈0.2 warm‑starts from the globally best sequence;
    otherwise draws from a random parametric family.
    The vector is scaled to satisfy Σ a_i = √(2 n).
    """
    S = np.sqrt(2.0 * n)

    # Warm start occasionally
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
        return a

    # Choose a family
    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'ushape', 'sine', 'semicircle', 'cosine',
                        'uniform', 'dirichlet'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'sine':
        raw = _init_sine(n, rng)
    elif style == 'semicircle':
        raw = _init_semicircle(n, rng)
    elif style == 'cosine':
        raw = _init_cosine(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Smooth‑approximation gradient descent
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Minimise a smoothed version of the objective using projected gradient
    descent with a decreasing smoothing factor μ.
    Returns the best vector seen and its hard max value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Initial hard max and best record
    cur_max = _max_convolution(a)
    best_a = a.copy()
    best_val = cur_max

    # Parameters for the schedule
    mu_factor = 0.25            # start μ = mu_factor * cur_max
    mu = max(cur_max * mu_factor, 1e-6)
    mu_decay = 0.997            # decay per iteration
    min_mu = 1e-6 * cur_max

    it = 0
    while time.time() < deadline:
        it += 1

        # Full self‑convolution (needed for both gradient and μ‑update)
        conv = _full_convolution_fft(a)
        cur_max = conv.max()

        # Update smoothing parameter (never let it vanish)
        mu = max(mu * mu_decay, max(cur_max * 0.01, min_mu))

        # Softmax weights over the convolution entries
        # Stabilise by subtracting the maximum before exponentiation
        exp_scaled = np.exp((conv - cur_max) / mu)
        sum_exp = exp_scaled.sum()
        weight = exp_scaled / sum_exp  # shape (2n‑1,)

        # Gradient = 2 * (weight ∗ reverse(a)) evaluated on central window
        a_rev = a[::-1]
        conv_len = len(weight) + n - 1
        f_len = _next_pow2(conv_len)
        Fw = np.fft.rfft(weight, n=f_len)
        Fa = np.fft.rfft(a_rev, n=f_len)
        tmp = np.fft.irfft(Fw * Fa, n=f_len)[:conv_len]
        start = n - 1
        grad = 2.0 * tmp[start:start + n]

        # Polyak‑style step size (guaranteed non‑negative)
        norm_sq = np.dot(grad, grad)
        if norm_sq > 0:
            step = (cur_max - best_val) / (norm_sq + 1e-12)
            step = np.clip(step, 1e-7, 0.15)
        else:
            step = 0.0

        # Simple back‑tracking line search
        for _ in range(8):
            a_candidate = _project_onto_simplex(a - step * grad, S)
            cand_max = _max_convolution(a_candidate)
            if cand_max <= cur_max:
                break
            step *= 0.5

        a = a_candidate
        cur_max = cand_max

        if cur_max < best_val - 1e-12:
            best_val = cur_max
            best_a = a.copy()

        # Occasionally inject a tiny random jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=0.001 * S, size=n)
            a = _project_onto_simplex(a + jitter, S)

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (hard max) – kept from the previous code
# ----------------------------------------------------------------------
def _max_convolution_and_gradient(a: np.ndarray):
    """
    Returns (grad, max_val) where grad is a sub‑gradient of
    maxₖ(a∗a)[k] with respect to a and max_val is that maximum.
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_val = b.max()
    mask = (b == max_val).astype(float)

    a_rev = a[::-1]
    conv_len = len(mask) + n - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad, max_val


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         global_best: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on f(a)=max(a∗a)."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_val = _max_convolution_and_gradient(a)

        norm_sq = np.dot(grad, grad)
        if norm_sq > 0:
            step = (cur_val - best_val) / (norm_sq + 1e-12)
            step = np.clip(step, 1e-7, 0.2)
        else:
            step = 0.0

        a = _project_onto_simplex(a - step * grad, S)

        cur_val = _max_convolution(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()

        if it % 300 == 0:
            jitter = rng.normal(scale=0.001 * np.mean(a), size=n)
            a = _project_onto_simplex(a + jitter, S)

    return best_a, best_val

# Simple simulated‑annealing hill‑climb (kept from the previous code)
# ----------------------------------------------------------------------
def _anneal_local_search(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Mass‑transfer hill‑climb with exponential temperature decay."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    cur_val = _max_convolution(a)
    best_a = a.copy()
    best_val = cur_val

    T = 0.05 * cur_val
    decay = 0.9999
    it = 0
    while time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        frac = rng.uniform(0.02, 0.18)
        delta = a[i] * frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _max_convolution(a_new)
        delta_val = new_val - cur_val
        if delta_val < 0 or rng.random() < np.exp(-delta_val / (T + 1e-12)):
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()

        T *= decay
        if it % 500 == 0:
            jitter = rng.normal(scale=0.001 * S, size=n)
            a = _project_onto_simplex(a + jitter, S)
            cur_val = _max_convolution(a)

    return best_a, best_val
