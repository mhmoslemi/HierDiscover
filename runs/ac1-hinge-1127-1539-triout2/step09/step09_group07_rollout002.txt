"""
Propose a step‑function sequence that minimises ``evaluate_sequence``.

Algorithm
---------

1. **Warm‑start** – if a previous construction exists (global
   ``height_sequence_1``) it is used as a seed.
2. **Random parametric search** – a set of simple families (exponential,
   power‑law, beta, Gaussian, half‑sine, sparse bumps) is sampled
   with random parameters for several lengths.  The best few candidates
   (by the internal bound 2·n·max(b)/sum(a)²) are kept.
3. **Multi‑scale Adam refinement** – each shortlisted candidate is
   interpolated to a geometric schedule of larger lengths (up to ~12×
   the original size, capped at 80 000).  For every length a
   log‑softmax Adam optimiser is run; the temperature for the
   soft‑max approximation of the max‑convolution is annealed to
   almost‑hard and the learning‑rate decays.
4. **Fine stochastic optimisation** – the best sequence found so far is
   further improved by a simulated‑annealing mass‑transfer walk and a
   cheap greedy hill‑climber that keep the ℓ₁‑norm fixed.
5. The final sequence is returned and stored as ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Peak value of a*a (full convolution)."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Gradient of the (soft) max‑convolution
# ----------------------------------------------------------------------
def _max_gradient(a: np.ndarray, tau: float):
    """
    Returns (grad, max_b) where grad = ∂ max(b) / ∂a
    and max_b = max(b) with b = a*a (full convolution).

    If tau is very small a hard‑max active set is used, otherwise a
    τ‑soft‑max weighting is employed.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        eps = max_b * 1e-9 + 1e-12
        mask = (b >= max_b - eps).astype(float)
        s = mask.sum()
        if s == 0:
            mask = np.ones_like(b) / b.size
        else:
            mask /= s
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / exps.sum()

    a_rev = a[::-1]
    conv_len = mask.size + a.size - 1
    f_len = _next_pow2(conv_len)

    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)

    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    grad = 2.0 * conv[a.size - 1:a.size - 1 + a.size]
    return grad, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   lr0: float = 0.5,
                   tau_decay: float = 0.9995,
                   lr_decay: float = 0.99985,
                   jitter_every: int = 400) -> tuple[np.ndarray, float]:
    """
    Minimises the peak of the self‑convolution under the ℓ₁‑norm
    constraint (sum = sqrt(2 n)). Returns the best vector seen and its
    hard peak value.
    """
    n = a0.shape[0]
    S = math.sqrt(2.0 * n)                     # required ℓ₁‑norm

    # initialise logits from the feasible a0 (log‑softmax)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                           # logits

    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr0

    tau = max(_full_convolution_fft(a0).max(), 0.5)

    best_a = a0.copy()
    best_peak = _max_convolution(a0)
    it = 0

    while time.time() < deadline:
        it += 1

        # --- recover a from logits (log‑softmax) ---
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)               # shift for stability
        soft = exp_p / exp_p.sum()
        a = soft * S

        # --- gradient of (soft‑)max convolution ---
        g, peak = _max_gradient(a, tau)

        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # --- gradient w.r.t. logits (chain rule) ---
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # clip for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # --- Adam update ---
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # --- schedules ---
        tau *= tau_decay
        lr *= lr_decay

        # occasional jitter to escape plateaus
        if jitter_every and (it % jitter_every) == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak

# Simulated annealing / greedy hill‑climbing fine‑tuner
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Mass‑transfer walk on the simplex with exponential acceptance."""
    n = len(seq)
    S = math.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score
    temperature = 0.15 * cur_score

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] == 0.0:
            continue
        delta = rng.uniform(0.0, 0.3) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())
        new_score = _max_convolution(new)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

        temperature *= 0.99995
        if temperature < 1e-6:
            temperature = 1e-6

    return best, best_score


def _hill_climb(seq: np.ndarray,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Cheap greedy hill‑climber (downhill moves only)."""
    n = len(seq)
    S = math.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] < 1e-12:
            continue
        delta = rng.uniform(0.0, 0.15) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())
        new_score = _max_convolution(new)
        if new_score < cur_score:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score
    return best, best_score

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The algorithm consists of a quick parametric random search,
    a multi‑scale Adam refinement and a final stochastic fine‑tuner.
    The best construction is stored in the global variable
    ``height_sequence_1`` and also returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    safety_margin = 5.0
    end_time = time.time() + budget_s - safety_margin
    if end_time <= time.time():
        raise RuntimeError("Budget too small after safety margin.")

    # ------------------------------------------------------------------
    # Warm‑start from a previously stored construction, if any
    # ------------------------------------------------------------------
    warm_seq = None
    if "height_sequence_1" in globals():
        try:
            warm_seq = np.asarray(globals()["height_sequence_1"], dtype=float)
        except Exception:
            warm_seq = None

    # ------------------------------------------------------------------
    # Utility to normalise a raw vector to the required ℓ₁‑norm
    # ------------------------------------------------------------------
    def _normalise(raw: np.ndarray) -> np.ndarray:
        n = len(raw)
        S = math.sqrt(2.0 * n)
        total = raw.sum()
        if total == 0.0:
            return np.full(n, S / n)
        return raw * (S / total)

    # ------------------------------------------------------------------
    # Phase 0 – fast random parametric exploration (≈10 % of budget)
    # ------------------------------------------------------------------
    param_deadline = time.time() + 0.10 * budget_s
    top_k = 12
    top_candidates = []                     # list of (score, seq)
    best_seq = None
    best_score = np.inf

    # determine a sensible set of lengths
    if warm_seq is not None:
        base_len = len(warm_seq)
    else:
        base_len = 4000
    # geometric schedule up to about 12× base, capped at 80 000
    lengths = [int(base_len * (1.5 ** i)) for i in range(5)]
    lengths = [min(max(2000, l), 80000) for l in lengths]
    lengths = sorted(set(lengths))

    attempt = 0
    while time.time() < param_deadline:
        attempt += 1
        n = rng.choice(lengths)
        raw = _random_family_vector(n, rng)
        seq = _normalise(raw)
        score = _internal_score(seq)

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[param] attempt={attempt:5d} n={n:5d}  new best={best_score:.6f}")

        if len(top_candidates) < top_k:
            top_candidates.append((score, seq.copy()))
            top_candidates.sort(key=lambda x: x[0])
        elif score < top_candidates[-1][0]:
            top_candidates[-1] = (score, seq.copy())
            top_candidates.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # Safety net – ensure we have at least one candidate
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 4000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n)
        best_score = _internal_score(best_seq)
        top_candidates = [(best_score, best_seq.copy())]

    # ------------------------------------------------------------------
    # Phase 1 – multi‑scale Adam refinement (≈75 % of budget)
    # ------------------------------------------------------------------
    max_len = min(80000, len(best_seq) * 12)
    # Build a schedule of increasing lengths (geometric growth)
    schedule = [len(best_seq)]
    cur = len(best_seq)
    while cur * 1.5 < max_len:
        cur = int(cur * 1.5)
        schedule.append(cur)
    schedule.append(max_len)
    schedule = sorted(set(schedule))

    # Iterate over the best candidates (most promising first)
    for init_idx, (_, init_seq) in enumerate(top_candidates):
        cur_seq = init_seq.copy()
        cur_len = len(cur_seq)

        for target_n in schedule:
            if time.time() >= end_time - 30.0:
                break

            # interpolate, clip, renormalise
            a_scaled = _scale_sequence(cur_seq, target_n)
            a_scaled = np.clip(a_scaled, 0.0, None)
            a_scaled = _normalise(a_scaled)

            # allocate a slice of the remaining time (max 180 s)
            remaining = end_time - time.time()
            slice_budget = min(remaining * 0.45, 180.0)

            # randomise a few hyper‑parameters for diversity
            lr0 = rng.choice([0.42, 0.48, 0.55, 0.60])
            tau_decay = rng.choice([0.9992, 0.9994, 0.9996, 0.9998])

            refined, peak = _adam_optimize(
                a_scaled,
                time.time() + slice_budget,
                rng,
                lr0=lr0,
                tau_decay=tau_decay,
                lr_decay=0.99985,
                jitter_every=400,
            )

            if peak < best_score:
                best_score = peak
                best_seq = refined.copy()
                best_len = target_n
                print(
                    f"[refine] init#{init_idx+1} len={target_n:5d}  improved score={best_score:.6f}"
                )

            # next scale starts from the refined vector
            cur_seq = refined
            cur_len = target_n

        if time.time() >= end_time - 30.0:
            break

    # ------------------------------------------------------------------
    # Phase 2 – cheap stochastic fine‑tuning (≈15 % of budget)
    # ------------------------------------------------------------------
    remaining = end_time - time.time()
    if remaining > 20.0:
        best_seq, best_score = _anneal(
            best_seq, time.time() + 0.6 * remaining, rng
        )
        remaining = end_time - time.time()
        if remaining > 5.0:
            best_seq, best_score = _hill_climb(
                best_seq, time.time() + remaining, rng
            )

    # ------------------------------------------------------------------
    # Final sanity check via the supplied black‑box evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(
        f"[final] external score = {external:.6f}  internal max = {best_score:.6f}"
        f"  length = {len(final_seq)}"
    )
    globals()["height_sequence_1"] = final_seq
    return final_seq
