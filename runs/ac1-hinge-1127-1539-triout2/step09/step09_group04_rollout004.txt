def _subgradient_max_descent_improved(a0, target_sum, deadline):
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    it = 0
    momentum = np.zeros_like(a)
    gamma = 0.9

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())
        # gradient: 2*a[partner]
        lo = max(0, k_max - (a.shape[0] - 1))
        hi = min(k_max, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1)
        partner = k_max - idx
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[partner]  # gradient of max conv at this k

        # momentum
        momentum = gamma * momentum + grad

        # step size: safe step considering non-negativity via projected onto simplex after
        # compute maximal step size to keep non-negative before projection:
        safe_eta = np.min(np.where(momentum > 0, a / momentum, np.inf))
        # decaying factor:
        eta = min(0.9 * safe_eta, 0.5 / np.sqrt(it + 1))

        a = a - eta * momentum
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] Subgrad+momentum it={it} score={best_val:.6f}")

        # occasional random jitter
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        it += 1

    return best, best_val

"""
Propose a low‑score step‑function.

Algorithm
---------
1. Choose a problem dimension `n` (12000 for a full 1000 s budget) and fix
   the ℓ¹‑norm `target_sum = sqrt(2·n)` (the objective is scale‑invariant).

2. Build a small pool of warm‑starts:
   * the previous best sequence (`height_sequence_1`);
   * a simple power‑law;
   * a few random power‑law × exponential shapes.

3. Keep the best of the pool (cheap evaluation).

4. Perform a short *exponential mirror descent* on a soft‑max
   approximation of the maximal convolution entry – this quickly
   lowers the max‑convolution while preserving the ℓ¹‑norm.

5. Apply a deterministic *mass‑transfer* that moves a modest fraction
   of mass from the most harmful index (largest contribution to the
   current peak) to a weak one – a cheap greedy refinement.

6. Run an *accelerated sub‑gradient descent* on the exact max‑convolution.
   The sub‑gradient is `2·a[k_max–i]` for the active index `k_max`.
   A momentum term and an adaptive step size (limited by positivity) are
   used.  Periodic jitter helps escape flat regions.

7. While time permits, sprinkle tiny random perturbations (jitter) and
   keep any improvement.

8. Trim trailing zeros, store the result globally for warm‑starts and
   return it as a plain list.

All operations respect the simplex constraint (`a_i ≥ 0`, Σa = target_sum)
by Euclidean projection.  The routine obeys the global time budget
(`budget_s` seconds) and prints progress messages.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution / scoring utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) shape, normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where conv is the full convolution a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2·(reverse(a) convolve p)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Stage 2 – exponential (multiplicative) descent
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Mirror‑descent with exponential updates using a soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 16384.0
    base_eta = 0.04
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        # occasional jitter to avoid stagnation
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 5 – deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _mass_transfer_refine(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Move a modest fraction of mass from high‑contributing indices
    to low‑contributing ones (greedy improvement of max‑convolution)."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    while time.time() < deadline:
        k_max = int(np.argmax(conv))

        # indices that actually participate in the maximal entry
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = a[k_max - idx]                       # a_{k_max-i}
        weight = a[idx] * partner                       # contribution a_i·a_{k_max-i}

        donor = idx[np.argmax(weight)]

        # receiver = index with low contribution (10‑percentile)
        low_thr = np.percentile(weight, 10)
        low_idx = idx[weight <= low_thr]
        if low_idx.size:
            receiver = low_idx[np.argmin(weight[weight <= low_thr])]
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            donor = rng.choice(idx)
            receiver = rng.integers(n)
            if donor == receiver:
                continue

        max_frac = 0.08          # a bit larger than before
        delta = a[donor] * rng.random() * max_frac
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta

        conv = _conv_fft(a)
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Transfer improve score={best_val:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            conv = _conv_fft(a)

        # renormalise occasionally (numerical drift)
        cur_sum = a.sum()
        if abs(cur_sum - target_sum) > 1e-9:
            a = a * (target_sum / cur_sum)

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 6 – accelerated sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_max_descent_improved(a0: np.ndarray,
                                      target_sum: float,
                                      deadline: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with momentum on the exact max‑convolution."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0
    momentum = np.zeros_like(a)
    gamma = 0.95  # momentum coefficient

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())

        # sub‑gradient of the active convolution entry
        lo = max(0, k_max - (a.shape[0] - 1))
        hi = min(k_max, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[partner]

        # momentum update
        momentum = gamma * momentum + grad

        # safe step size respecting non‑negativity before projection
        safe_eta = np.min(np.where(momentum > 0, a / momentum, np.inf))
        # a diminishing factor + a 1/√t term
        eta = min(0.9 * safe_eta, 0.5 / np.sqrt(it + 1))

        a = a - eta * momentum
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} score={best_val:.6f}")

        # jitter for escaping plateaus
        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:

    """
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Choose problem size
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Build an initial pool of candidates
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    pool = []

    # warm‑start from previous run if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        pool.append(_resize_sequence(prev, n))

    # a simple power‑law
    pool.append(_power_law_sequence(n, 0.55, target_sum))

    # a few random power‑law × exponential shapes
    for _ in range(4):
        p = rng.uniform(0.2, 1.2)
        lam = rng.uniform(0.0, 6.0)
        i = np.arange(1, n + 1, dtype=np.float64)
        seq = (i ** (-p)) * np.exp(-lam * (i - 1) / n)
        seq = seq * (target_sum / seq.sum())
        pool.append(seq)

    # ------------------------------------------------------------------
    # Keep the best cheap candidate
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    for cand in pool:
        cand = _project_to_simplex(cand, target_sum)
        val = _score(cand)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
    print(f"[{time.time():.2f}] Initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – exponential (multiplicative) descent (≈10 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:

        a_exp, val_exp = _exponential_descent(best_seq, target_sum, exp_deadline)

        print(f"[{time.time():.2f}] After sub‑gradient best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final tiny random jitter sweep (use any leftover time)

    while time.time() < deadline - 0.1:
