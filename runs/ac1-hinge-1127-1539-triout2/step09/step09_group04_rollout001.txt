def propose_candidate(...):

  else: init = powerlaw.

  # then run Adam descent.

  best_score = _score(a)

  # Adam states
  m = np.zeros_like(a)
  v = np.zeros_like(a)
  beta1 = 0.9
  beta2 = 0.999
  lr0 = 0.10 (base learning rate)
  it = 1

  beta = 1.0; beta_target = 16384.0; beta_inc_step = 100
  # schedule: increase beta every so often to focus on max.

  while time.time() < deadline:
      # decreasing learning schedule: lr = lr0 / sqrt(it)
      lr = lr0 / (np.sqrt(it))

      # optionally also scale by 1/(beta+1) as before
      lr = lr / (beta + 1.0)

      # compute gradient of softmax surrogate
      g, conv = _gradient_softmax(a, beta)

      # update moments
      m = beta1 * m + (1 - beta1) * g
      v = beta2 * v + (1 - beta2) * (g * g)

      m_hat = m / (1 - beta1**it)
      v_hat = v / (1 - beta2**it)

      step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
      a = a - step

      # after step, we may get negative values; project to simplex
      a = _project_to_simplex(a, target_sum)

      # evaluate
      val = _score(a)
      if val < best_score:
          best_score = val
          best_a = a.copy()
          print(f"[{time.time():.2f}] Adam it={it} beta={beta:.1f} score={best_score:.6f}")

"""
Propose a step‑function with a low benchmark score.

Strategy
--------
1. Warm‑start from the previously best sequence (`height_sequence_1`) if it
   exists, otherwise from a simple power‑law.
2. Perform a long Adam‑style optimisation on the *soft‑max* approximation
   of the maximal convolution entry.  The parameter β of the soft‑max is
   doubled periodically so that the surrogate becomes tighter.
3. After the Adam phase run a fast deterministic “mass‑transfer” local
   search that moves a small fraction of the mass from indices that
   contribute heavily to the current maximal convolution entry to indices
   that contribute little.
4. Trim trailing zeros, store the result for warm‑starts, evaluate it and
   return it.

Both phases use FFT‑based convolution (O(n log n)) which is fast enough
to perform many thousands of iterations within the 1000 s budget.
The Adam optimiser provides an adaptive learning rate per coordinate and
generally reaches lower values than the simple gradient descent used
previously.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution / scoring utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) / (sum(a)²)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Sequence a_i = (i+1)^(-p) normalised to have ℓ¹‑norm = target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max surrogate utilities
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).

    Returns
    -------
    grad : np.ndarray
        Gradient w.r.t. a (same shape as a).
    conv : np.ndarray
        Full convolution a*a (length 2·n‑1) – useful for diagnostics.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(conv)
    # soft‑max weights
    exps = np.exp(beta * (conv - max_b))
    p = exps / np.sum(exps)

    # grad = 2 * (reverse(a) convolve p)
    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    p_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, conv


# ----------------------------------------------------------------------
# Adam optimisation on the soft‑max surrogate
# ----------------------------------------------------------------------


def _adam_softmax_descent(a0: np.ndarray,
                          target_sum: float,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max surrogate of the max‑convolution.
    The β parameter (soft‑max sharpness) is doubled every 2000 iterations.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # learning‑rate schedule
    base_lr = 0.12          # a generous initial step size
    it = 1
    beta = 1.0
    beta_target = 16384.0

    rng = np.random.default_rng()

    while time.time() < deadline:
        # adapt learning rate (√t decay) and use β in the denominator
        lr = base_lr / (np.sqrt(it) * (beta + 1.0))

        # gradient of the smooth surrogate
        g, _ = _gradient_softmax(a, beta)

        # Adam updates (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # enforce non‑negativity and ℓ¹‑norm
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat plateaus
        if it % 2500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        # increase β (sharpen soft‑max) periodically
        if it % 2000 == 0 and beta < beta_target:
            beta = min(beta * 2.0, beta_target)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer local refinement
# ----------------------------------------------------------------------


def _local_mass_transfer(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a small fraction of mass from a donor that
    contributes heavily to the current maximal convolution entry to a
    receiver that contributes little.  Accept only improving moves.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.size
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    conv = _conv_fft(a)

    while time.time() < deadline:
        k_max = int(np.argmax(conv))

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = k_max - idx
        contrib = a[idx] * a[partner]

        # donor = index with the largest contribution
        donor = idx[np.argmax(contrib)]

        # receiver = index with a small contribution (below 20‑th percentile)
        thresh = np.percentile(contrib, 20)
        low_candidates = idx[contrib <= thresh]
        if low_candidates.size:
            receiver = low_candidates[np.argmin(contrib[contrib <= thresh])]
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            # fallback random move
            donor = rng.choice(idx)
            receiver = rng.integers(n)
            if donor == receiver:
                continue

        max_frac = 0.05
        delta = a[donor] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta

        # renormalise (numeric drift)
        a = a * (target_sum / a.sum())

        # evaluate
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            conv = _conv_fft(a)          # recompute because the convolution changed
            print(f"[{time.time():.2f}] Mass‑transfer improved score={best_val:.6f}")
        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta
            a = a * (target_sum / a.sum())

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5      # small safety margin

    # --------------------------------------------------------------
    # Choose problem size (large enough to give a fine resolution)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)          # ℓ¹‑norm we keep fixed

    # --------------------------------------------------------------
    # Warm‑start (use previously best sequence if available)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"],
                                         dtype=np.float64), n)
        init = init * (target_sum / init.sum())
        print(f"[{time.time():.2f}] Warm‑started from saved sequence (n={n})")
    else:
        init = _power_law_sequence(n, 0.55, target_sum)
        print(f"[{time.time():.2f}] Started from power‑law (n={n})")

    best_seq = init.copy()
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] Initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Adam optimisation on soft‑max surrogate
    # --------------------------------------------------------------
    if time.time() < deadline:
        adam_deadline = time.time() + (deadline - time.time()) * 0.70
        a_adam, val_adam = _adam_softmax_descent(best_seq,
                                                 target_sum,
                                                 adam_deadline)
        if val_adam < best_score:
            best_seq = a_adam
            best_score = val_adam
        print(f"[{time.time():.2f}] After Adam best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – deterministic mass‑transfer refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        mt_deadline = time.time() + (deadline - time.time()) * 0.30
        a_mt, val_mt = _local_mass_transfer(best_seq,
                                            target_sum,
                                            mt_deadline)
        if val_mt < best_score:
            best_seq = a_mt
            best_score = val_mt
        print(f"[{time.time():.2f}] After mass‑transfer best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Clean up: trim trailing zeros and final evaluation
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in a subsequent call
    globals()["height_sequence_1"] = final_seq

    return final_seq
