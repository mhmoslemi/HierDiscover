def _grad_from_weights(weights, a):
    n = a.shape[0]
    a_rev = a[::-1]
    conv_len = len(weights) + n - 1
    f_len = _next_pow2(conv_len)
    Fw = np.fft.rfft(weights, n=f_len)

    conv = np.fft.irfft(Fw * Fa_rev, n=f_len)[:conv_len]
    start = n - 1
    grad = 2.0 * conv[start:start + n]

"""
Improved step‑function optimizer.

The algorithm proceeds in three phases:

1. **Wide random exploration** – generate many candidates of varying
   lengths (4 000 – 30 000) from several parametric families.
   Keep the best according to the hard convolution peak.

2. **Smoothed‑max gradient descent** – for the current best sequence
   we minimise the smooth approximation
        f_ε(a)=ε·log∑ₖexp((a∗a)[k]/ε)
   using a projected gradient step (FFT for the gradient) and a
   Polyak‑type step size.  ε is decreased geometrically so the method
   tracks the true max while enjoying a much richer gradient signal.
   The descent is run for the bulk of the time budget, repeatedly
   scaling the sequence to larger lengths (≈ ×1.5) to give the optimiser
   more freedom.

3. **Final polishing** – a short sub‑gradient descent followed by a
   cheap simulated‑annealing mass‑transfer hill‑climb.

All variables are projected onto the ℓ₁‑simplex
   {a_i≥0, ∑a_i=√(2·n)} after each move, guaranteeing feasibility.
The routine stores the final construction in the global variable
``height_sequence_1`` and returns it as a plain list.
"""

import time
import numpy as np

# Low‑level utilities (FFT, convolution, gradient, projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a∗a."""
    return _full_convolution_fft(a).max()


def _max_convolution_and_gradient(a: np.ndarray):
    """
    Sub‑gradient of maxₖ (a∗a)[k].
    Returns (grad, max_val) where grad has shape (n,).
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_val = b.max()
    mask = (b == max_val).astype(float)

    a_rev = a[::-1]
    conv_len = mask.shape[0] + n - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad, max_val


def _gradient_from_weights(weights: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed objective built from arbitrary non‑negative
    weights that sum to one (i.e. a soft mask).  Equivalent to the
    sub‑gradient routine with ``mask`` replaced by ``weights``.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    conv_len = weights.shape[0] + n - 1
    f_len = _next_pow2(conv_len)
    Fw = np.fft.rfft(weights, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fw * Fa_rev, n=f_len)[:conv_len]
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


def _project_onto_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """
    Euclidean projection of v onto {w | w_i≥0, ∑w_i=S}.
    Implements the O(n) algorithm of Duchi et al.
    """
    if S < 0:
        raise ValueError("Simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]

def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.1, 5.0)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    return mix * edge + (1.0 - mix) * base


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0001, 0.04)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    return mix * exp_part + (1.0 - mix) * base


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.3, 8.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.3))
    return mix * lin + (1.0 - mix) * base


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.2, 3.0)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    return mix * ush + (1.0 - mix) * base


def _init_sine(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    vals = np.sin(np.pi * (i + 1) / (n + 1))
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * vals + (1.0 - mix) * base


def _init_beta_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Shape ∝ x^α (1‑x)^β on [0,1]."""
    x = np.linspace(0.0, 1.0, n)
    a = rng.uniform(0.1, 5.0)
    b = rng.uniform(0.1, 5.0)
    raw = (x ** a) * ((1 - x) ** b)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * raw + (1.0 - mix) * base


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    return rng.random(n)


def _initial_vector(
    n: int,
    rng: np.random.Generator,
    warm_seq: np.ndarray | None,
) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    With probability ≈0.2 we warm‑start from the globally best sequence,
    otherwise we draw from a random parametric family.
    The vector is finally projected onto the simplex
    {a≥0, Σa = √(2 n)}.
    """
    S = np.sqrt(2.0 * n)

    # Warm start
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        return a * (S / total)

    # Pick a family
    style = rng.choice(
        [
            "powerlaw",
            "exponential",
            "linear",
            "ushape",
            "sine",
            "beta",
            "uniform",
            "dirichlet",
        ]
    )
    if style == "powerlaw":
        raw = _init_powerlaw(n, rng)
    elif style == "exponential":
        raw = _init_exponential(n, rng)
    elif style == "linear":
        raw = _init_linear(n, rng)
    elif style == "ushape":
        raw = _init_ushape(n, rng)
    elif style == "sine":
        raw = _init_sine(n, rng)
    elif style == "beta":
        raw = _init_beta_shape(n, rng)
    elif style == "uniform":
        raw = _init_uniform(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    return raw * (S / total)

def _smooth_descent(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    current_best: float,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the smooth approximation
        f_ε(a)=ε·log∑ₖexp((a∗a)[k]/ε)
    using a Polyak‑type step size and projection onto the simplex.
    The smoothing parameter ε is decreased geometrically.
    Returns the best vector seen and its hard‑max value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # initialise best values
    best_a = a.copy()
    best_val = _max_convolution(a)
    # initialise smoothing
    eps = max(0.2 * best_val, 0.01)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a)
        mx = conv.max()

        # soft‑max weights (stable)
        w = np.exp((conv - mx) / eps)
        w_sum = w.sum()
        w = w / w_sum

        # smooth objective value (optional, used for step size)
        smooth_val = eps * np.log(w_sum) + mx

        # gradient of the smooth objective
        grad = _gradient_from_weights(w, a)

        # Polyak step size w.r.t. the *hard* max (guarantees descent)
        norm_sq = np.dot(grad, grad) + 1e-12
        step = (smooth_val - best_val) / norm_sq
        # keep step in a sensible range
        step = np.clip(step, 1e-7, 0.15)

        # projected gradient step
        a = _project_onto_simplex(a - step * grad, S)

        # evaluate hard‑max
        cur_val = _max_convolution(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()
            # tighten smoothing when we improve
            eps = max(eps * 0.9, 1e-6)

        # slowly decay smoothing regardless of improvement
        eps = max(eps * 0.9995, 1e-6)

        # occasional jitter to escape tiny plateaus
        if it % 400 == 0:
            jitter = rng.normal(scale=0.001 * S, size=n)
            a = _project_onto_simplex(a + jitter, S)

    return best_a, best_val


def _subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    global_best: float,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=maxₖ(a∗a)[k].
    Uses a Polyak step size and O(n) simplex projection.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_val = _max_convolution_and_gradient(a)

        norm_sq = np.dot(grad, grad)
        if norm_sq > 0:
            step = (cur_val - best_val) / (norm_sq + 1e-12)
            step = np.clip(step, 1e-7, 0.2)
        else:
            step = 0.0

        a = _project_onto_simplex(a - step * grad, S)

        cur_val = _max_convolution(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()

        if it % 300 == 0:
            jitter = rng.normal(scale=0.001 * np.mean(a), size=n)
            a = _project_onto_simplex(a + jitter, S)

    return best_a, best_val


def _anneal_local_search(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing hill‑climb that moves a random fraction
    of mass between two positions and accepts worsening moves with a
    temperature that decays exponentially.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    cur_val = _max_convolution(a)
    best_a = a.copy()
    best_val = cur_val

    T = 0.05 * cur_val
    decay = 0.9999

    it = 0
    while time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        frac = rng.uniform(0.02, 0.18)
        delta = a[i] * frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _max_convolution(a_new)
        delta_val = new_val - cur_val
        if delta_val < 0 or rng.random() < np.exp(-delta_val / (T + 1e-12)):
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()

        T *= decay
        if it % 500 == 0:
            jitter = rng.normal(scale=0.001 * S, size=n)
            a = _project_onto_simplex(a + jitter, S)
            cur_val = _max_convolution(a)

    return best_a, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    warm_seq = None

        except Exception:
            warm_seq = None

    best_score = np.inf
    best_n = None
    attempts = 0

    # Phase 1 – broad random exploration (≈15 % of the budget)
    # ------------------------------------------------------------------
    explore_deadline = start_time + 0.15 * budget_s

        a0 = _initial_vector(n, rng, warm_seq)
        val = _max_convolution(a0)
        if val < best_score:
            best_score = val
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempts:4d} n={n:5d}  new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – smoothed‑max descent with progressive scaling
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform of moderate size
        n = 6000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = _max_convolution(best_seq)
        best_n = n

    current_seq = best_seq
    while True:
        now = time.time()
        if now >= end_time - 30.0:
            break

        # increase length by a factor ≈1.5 (but cap at 50000)
        target_n = int(len(current_seq) * 1.5)
        if target_n > 50000:
            target_n = 50000
        if target_n <= len(current_seq):

        a_scaled = _scale_sequence(current_seq, target_n)
        S = np.sqrt(2.0 * target_n)
        a_scaled = np.clip(a_scaled, 0.0, None)
        a_scaled = a_scaled * (S / a_scaled.sum())

        # allocate a time slice (≈ 30 % of remaining budget)
        remaining = end_time - now - 30.0
        slice_time = min(remaining * 0.30, 90.0)

        a_opt, val = _smooth_descent(
            a_scaled, time.time() + slice_time, rng, best_score
        )
        if val < best_score:
            best_score = val
            best_seq = a_opt.copy()
            best_n = target_n
            print(f"[smooth] length={target_n:5d}  improved={best_score:.6f}")

        current_seq = a_opt

    # ------------------------------------------------------------------
    # Phase 3 – final sub‑gradient refinement (≈20 % of remaining time)
    # ------------------------------------------------------------------
    if time.time() < end_time - 20.0:
        remaining = end_time - time.time() - 5.0
        a_opt, val = _subgradient_descent(
            best_seq, time.time() + remaining, rng, best_score
        )
        if val < best_score:
            best_score = val
            best_seq = a_opt.copy()
            print(f"[final subgrad] length={len(best_seq)}  best={best_score:.6f}")

        remaining = end_time - time.time() - 1.0
        a_opt, val = _anneal_local_search(
            best_seq, time.time() + remaining, rng
        )
        if val < best_score:
            best_score = val
            best_seq = a_opt.copy()
            print(f"[anneal] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(
        f"[final] external={external:.6f}  internal={best_score:.6f}  length={len(final_seq)}"
    )
    globals()["height_sequence_1"] = final_seq
    return final_seq
