"""
Step‑function optimizer – enhanced version.

The goal is to minimise 2·n·max(conv(a,a))/||a||₁².
If we keep the ℓ₁‑norm fixed to √(2·n) the score is simply the
maximum entry of the self‑convolution.  The algorithm works on the
simplex (non‑negative entries, fixed ℓ₁‑norm) and consists of the
following stages (all respect a total budget of 1000 s):

1. Warm‑start from the global variable ``height_sequence_1`` if it
   exists.
2. Fast random exploration using a collection of parametric families.
3. Length‑up‑scaling – try several longer lengths via linear interpolation.
4. Adam optimisation with a τ‑soft‑max approximation of the max.
5. Adam optimisation on a growing p‑norm approximation (p from 4 → 16).
6. Plain sub‑gradient descent on the hard max.
7. Greedy mass‑transfer hill‑climb that always moves mass from the
   *most* responsible coefficient for the current peak to the *least*
   responsible one.
8. A final cheap random hill‑climb.

All stages keep the best vector seen (in terms of the hard maximum)
and the final result is stored in the global ``height_sequence_1``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, scaling, etc.)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a∗a."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation (preserves shape)."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# τ‑soft‑max gradient (used by the first Adam run)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """Gradient of the τ‑soft‑max approximation of max(conv)."""
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # 2·(mask·a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


# ----------------------------------------------------------------------
# p‑norm gradient (used by the second Adam run)
# ----------------------------------------------------------------------
def _pnorm_gradient(a: np.ndarray, p: float):
    """
    Gradient of the Lₚ‑norm of the self‑convolution (p≥2).
    Returns (grad, p‑norm value).
    """
    b = _full_convolution_fft(a)
    # To avoid overflow for large p we work with the arithmetic mean
    # rather than the sum.
    mean_p = np.mean(np.power(b, p))
    obj = mean_p ** (1.0 / p)

    # derivative of sum(bᵖ) w.r.t. a:
    w = p * np.power(b, p - 1, where=b > 0, out=np.zeros_like(b))
    a_rev = a[::-1]
    conv_len = len(w) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fw = np.fft.rfft(w, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fw * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]

    # scale to the gradient of the mean‑p norm (chain rule)
    #   d (mean_p^{1/p}) = (1/p) * mean_p^{1/p - 1} * d(mean_p)
    scale = (mean_p ** ((1.0 / p) - 1.0)) / (a.shape[0] * p)
    g *= scale
    return g, obj


# ----------------------------------------------------------------------
# Adam optimiser (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   grad_func,
                   grad_args=()) -> tuple[np.ndarray, float]:
    """
    Generic Adam optimiser on the simplex (log‑softmax parametrisation).

    Parameters
    ----------
    a0 : np.ndarray
        Feasible start vector (sum = √(2·n), non‑negative).
    deadline : float
        Absolute time after which the optimiser must stop.
    rng : np.random.Generator
        Random source (used for occasional jitter).
    grad_func : callable
        Function returning (grad, value) given a and optional extra args.
    grad_args : tuple
        Extra positional arguments passed to ``grad_func``.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                     # target ℓ₁‑norm

    # initialise logits from feasible a0 (log‑softmax)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.45

    # keep track of best hard peak (max(conv))
    best_a = a0.copy()
    best_peak = _max_convolution(a0)

    it = 0
    while time.time() < deadline:
        it += 1
        # recover a from logits
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # gradient of the surrogate objective
        g, val = grad_func(a, *grad_args)

        # update the best hard peak seen so far
        hard = _max_convolution(a)
        if hard < best_peak:
            best_peak = hard
            best_a = a.copy()

        # chain rule to logits
        mu = np.dot(a, g) / S           # ⟨a,∂obj/∂a⟩ / S
        grad_p = a * (g - mu)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 6.0 * S:
            grad_p = grad_p * (6.0 * S / clip_val)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # simple schedules
        lr *= 0.99996
        if it % 800 == 0:
            p += rng.normal(scale=0.006, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Sub‑gradient descent on the hard maximum
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple sub‑gradient descent on the *hard* max (projected onto simplex).
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    a = a0.copy()
    best_a = a.copy()
    best_val = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1
        b = _full_convolution_fft(a)
        k = int(b.argmax())
        val = b[k]

        if val < best_val:
            best_val = val
            best_a = a.copy()

        # sub‑gradient of (a*a)_k
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        jdx = k - idx
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[jdx]

        step = 0.8 * S / (np.sqrt(it) + 1.0)
        a = a - step * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, S / n)
        else:
            a = a * (S / s)

        if it % 500 == 0:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                frac = rng.uniform(0.02, 0.12)
                amount = a[i] * frac
                a[i] -= amount
                a[j] += amount
                a = a * (S / a.sum())

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy mass‑transfer refinement (directly targets the current peak)
# ----------------------------------------------------------------------
def _greedy_mass_transfer(a0: np.ndarray,
                          deadline: float,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Repeatedly moves a small fraction of mass from the coefficient that
    contributes most to the current maximal convolution entry to a
    coefficient that contributes least.  Accepts only strict improvements.
    """
    n = a0.shape[0]
    a = a0.copy()
    best_a = a.copy()
    best_val = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1
        b = _full_convolution_fft(a)
        k = int(b.argmax())                     # current worst shift
        # contribution of each index i to b_k:
        rev_idx = k - np.arange(n)
        valid = (rev_idx >= 0) & (rev_idx < n)
        contrib = np.zeros(n)
        contrib[valid] = a[valid] * a[rev_idx[valid]]

        i_max = int(np.argmax(contrib))
        i_min = int(np.argmin(contrib))

        if a[i_max] <= 0.0 or i_max == i_min:
            # fallback to a random pair
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                frac = rng.uniform(0.02, 0.12)
                amount = a[i] * frac
                a[i] -= amount
                a[j] += amount
        else:
            # move a modest fraction from i_max to i_min
            frac = rng.uniform(0.04, 0.15)
            amount = a[i_max] * frac
            a[i_max] -= amount
            a[i_min] += amount

        # keep the sum unchanged – it already is because we only moved mass
        new_val = _max_convolution(a)
        if new_val + 1e-12 < best_val:
            best_val = new_val
            best_a = a.copy()
        else:
            # undo the move if it didn't help (helps to escape very shallow plateaux)
            a[i_max] += amount
            a[i_min] -= amount

        if it % 500 == 0:
            # occasional tiny random jitter to shake out stalling
            i, j = rng.integers(0, n, size=2)
            if a[i] > 0.0:
                frac2 = rng.uniform(0.01, 0.07)
                amt = a[i] * frac2
                a[i] -= amt
                a[j] += amt

    return best_a, best_val

# Parametric families for random initialisation
# ----------------------------------------------------------------------
def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    raw = mix * ush + (1.0 - mix) * base
    return raw


def _init_cosine(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    raw = 1.0 - np.cos(np.pi * i / (n - 1))
    raw = raw * rng.uniform(0.5, 2.0)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * raw + (1.0 - mix) * base


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    return rng.random(n)


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.05, 0.9)
    return rng.gamma(shape, 1.0, size=n)


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    20 % of the time we warm‑start from ``warm_seq`` (scaled);
    otherwise we draw from one of the random families.
    The returned vector satisfies sum = √(2·n) and all entries ≥0.
    """
    S = np.sqrt(2.0 * n)

    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
        return a

    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'gamma', 'uniform', 'dirichlet', 'ushape', 'cosine'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'cosine':
        raw = _init_cosine(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence as a plain Python list and stores it in the
    global variable ``height_sequence_1`` for potential warm‑starts.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Warm‑start handling
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # ------------------------------------------------------------------
    # Phase 1 – fast random exploration (≈ 150 s)
    # ------------------------------------------------------------------
    explore_deadline = end_time - 350.0
    best_seq = None
    best_score = np.inf
    best_n = None
    attempts = 0

    while time.time() < explore_deadline:
        attempts += 1
        # favour lengths in the 2k–8k range, but keep some larger attempts
        if rng.random() < 0.75:
            n = int(rng.integers(2000, 8000))
        else:
            n = int(rng.integers(8000, 18000))

        a0 = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a0)

        if peak < best_score:
            best_score = peak
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempts:4d} n={n:5d}  new best={best_score:.6f}")

    if best_seq is None:
        # should never happen, but fall back to a uniform vector
        best_n = 4000
        best_seq = np.full(best_n, np.sqrt(2.0 * best_n) / best_n)
        best_score = _max_convolution(best_seq)

    # ------------------------------------------------------------------
    # Phase 2 – upscale to several larger lengths (no optimisation)
    # ------------------------------------------------------------------
    cand_lengths = [best_n]
    for factor in [1.25, 1.5, 2.0, 3.0, 4.0, 5.0]:
        L = int(best_n * factor)
        if 2000 <= L <= 25000:
            cand_lengths.append(L)
    cand_lengths = sorted(set(cand_lengths))

    cur_seq = best_seq
    cur_score = best_score
    cur_n = best_n

    for L in cand_lengths:
        if time.time() > end_time - 250.0:
            break
        a_scaled = _scale_sequence(cur_seq, L)
        a_scaled = np.clip(a_scaled, 0.0, None)
        S = np.sqrt(2.0 * L)
        a_scaled = a_scaled * (S / a_scaled.sum())
        peak = _max_convolution(a_scaled)

        if peak < cur_score:
            cur_score = peak
            cur_seq = a_scaled.copy()
            cur_n = L
            print(f"[scale] length={L:5d}  improved max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam optimiser (τ‑soft‑max)
    # ------------------------------------------------------------------
    remaining = end_time - time.time() - 150.0   # reserve time for later phases
    if remaining > 20.0:
        # decide how many independent Adam runs we can afford
        n_restart = max(1, int(remaining // 65.0))  # ~1 min per restart
        per_run = remaining / n_restart
        for r in range(n_restart):
            if time.time() > end_time - 150.0:
                break
            # small perturbation of the current best start
            noise = rng.normal(scale=0.015 * np.sqrt(2.0 * cur_n) / cur_n,
                               size=cur_n)
            a_start = np.maximum(cur_seq + noise, 0.0)
            a_start = a_start * (np.sqrt(2.0 * cur_n) / a_start.sum())
            a_opt, peak = _adam_optimize(a_start,
                                          time.time() + per_run,
                                          rng,
                                          grad_func=_softmax_gradient,
                                          grad_args=(np.max(_full_convolution_fft(a_start)),))
            if peak < cur_score:
                cur_score = peak
                cur_seq = a_opt.copy()
                print(f"[adam‑soft] restart={r} new best max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Adam optimiser on growing p‑norm (p from 4 to 16)
    # ------------------------------------------------------------------
    remaining = end_time - time.time() - 120.0
    if remaining > 20.0:
        n_restart = max(1, int(remaining // 65.0))
        per_run = remaining / n_restart
        for r in range(n_restart):
            if time.time() > end_time - 120.0:
                break
            noise = rng.normal(scale=0.012 * np.sqrt(2.0 * cur_n) / cur_n,
                               size=cur_n)
            a_start = np.maximum(cur_seq + noise, 0.0)
            a_start = a_start * (np.sqrt(2.0 * cur_n) / a_start.sum())
            # p will be linearly increased from 4 to 16 during the run
            a_opt, peak = _adam_optimize(a_start,
                                          time.time() + per_run,
                                          rng,
                                          grad_func=_pnorm_gradient,
                                          grad_args=(4.0,))
            if peak < cur_score:
                cur_score = peak
                cur_seq = a_opt.copy()
                print(f"[adam‑pnorm] restart={r} new best max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – sub‑gradient descent (hard max)
    # ------------------------------------------------------------------
    remaining = end_time - time.time() - 80.0
    if remaining > 30.0:
        a_opt, peak = _subgradient_descent(cur_seq,
                                           time.time() + remaining,
                                           rng)
        if peak < cur_score:
            cur_score = peak
            cur_seq = a_opt.copy()
            print(f"[subgrad] new best max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – greedy mass‑transfer refinement (directly attacks the peak)
    # ------------------------------------------------------------------
    remaining = end_time - time.time() - 40.0
    if remaining > 20.0:
        a_opt, peak = _greedy_mass_transfer(cur_seq,
                                            time.time() + remaining,
                                            rng)
        if peak < cur_score:
            cur_score = peak
            cur_seq = a_opt.copy()
            print(f"[greedy] new best max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 7 – cheap random hill‑climb (few seconds)
    # ------------------------------------------------------------------
    while time.time() < end_time - 1.0:
        a = cur_seq.copy()
        n = a.shape[0]
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        frac = rng.uniform(0.02, 0.12)
        amount = a[i] * frac
        a[i] -= amount
        a[j] += amount
        peak = _max_convolution(a)
        if peak + 1e-9 < cur_score:
            cur_score = peak
            cur_seq = a.copy()
            print(f"[hill] improved max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    final_seq = cur_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f}  internal max={cur_score:.6f}  length={len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
