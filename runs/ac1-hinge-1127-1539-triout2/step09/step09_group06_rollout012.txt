def frank_wolfe_optimize(a_initial, deadline, rng):
    a = a_initial.copy()
    n = len(a)
    best_a = a.copy()
    best_max = _max_convolution(a)
    # Keep sum = 1
    max_iter = 1000
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        # Compute M
        M = _max_convolution(a)
        # Solve LP
        A_ub = build_A_ub(a)
        b_ub = np.full(2*n-1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None)]*n
        try:
            res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
        except Exception as e:
            print('LP error', e)
            break
        if not res.success:
            print('LP failed', res.message)
            break
        g = res.x
        total = g.sum()
        if total <= 0:
            break
        g_norm = g / total  # sum =1
        # update with mixing coefficient
        alpha = 0.5  # could adapt based on iteration or progress
        a = (1 - alpha) * a + alpha * g_norm
        # we could ensure positivity and sum=1 (already)
        M_new = _max_convolution(a)
        if M_new < best_max:
            best_max = M_new
            best_a = a.copy()
            print(f"[FW] it={it} max={best_max:.6f}")
        # optional early break if improvement minimal
        if it % 20 == 0 and time.time() > deadline - 5:
            break
    return best_a, best_max

"""
Optimisation of a non‑negative step function for the provided
``evaluate_sequence`` (minimise the maximal entry of the self‑convolution).
The method is a hybrid of

* fast random sampling from several parametric families,
* a Frank‑Wolfe style linear‑programming refinement
  (solve a LP: maximise total mass of a new vector g under the
  current convolution bound, then mix the current vector with g),
* a tiny Adam run for polishing,
* a cheap stochastic hill‑climb.

The LP uses SciPy's HiGHS solver; the convolution is computed by a
real FFT for speed.  All vectors are normalised to unit ℓ¹‑norm, which
makes the objective simply ``max(a*a)`` (the external score is
``2·n·max`` and is invariant under scaling).  The routine respects a
1000 s budget and stores the best found sequence in the global variable
``height_sequence_1``.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a∗a."""
    return _full_convolution_fft(a).max()


# ----------------------------------------------------------------------
# Simple families for random initialisation (all produce non‑negative
# vectors; a small epsilon is added later to guarantee strict positivity).
# ----------------------------------------------------------------------
def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    return mix * edge + (1.0 - mix) * base


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    return mix * exp_part + (1.0 - mix) * base


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    return mix * lin + (1.0 - mix) * base


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    return mix * ush + (1.0 - mix) * base


def _init_cosine(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    raw = 1.0 - np.cos(np.pi * i / (n - 1))
    raw = raw * rng.uniform(0.5, 2.0)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * raw + (1.0 - mix) * base


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    return rng.random(n)


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.05, 0.9)
    return rng.gamma(shape, 1.0, size=n)


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Return a feasible start vector of length n with ℓ¹‑norm = 1.
    With probability ≈0.2 a scaled warm‑start is used,
    otherwise a random parametric family is sampled.
    The returned vector has strictly positive entries (tiny epsilon added).
    """
    eps = 1e-8
    if warm_seq is not None and rng.random() < 0.23:
        a = np.interp(np.linspace(0, len(warm_seq) - 1, n),
                      np.arange(len(warm_seq)), warm_seq)
        a = np.clip(a, 0.0, None)
        if a.sum() == 0.0:
            a = np.full(n, 1.0)
        a = a / a.sum() + eps
        a = a / a.sum()
        return a

    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'gamma', 'uniform', 'dirichlet', 'ushape', 'cosine'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'cosine':
        raw = _init_cosine(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    if raw.sum() == 0.0:
        raw = np.full(n, 1.0)
    a = raw / raw.sum() + eps
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Build the constraint matrix for the LP: (a * g)_k ≤ M  (for all k)
# ----------------------------------------------------------------------
def _build_A_ub(a: np.ndarray) -> np.ndarray:
    """Return dense matrix A where A[k,i] = a[k-i] (zero if out of range)."""
    n = a.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # For each i, the coefficient a[j] appears in rows i … i+n‑1
    for i in range(n):
        A[i:i + n, i] = a
    return A


# ----------------------------------------------------------------------
# Frank‑Wolfe style LP refinement
# ----------------------------------------------------------------------
def _frank_wolfe_optimize(a0: np.ndarray,
                          deadline: float,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve the LP:
        maximise 1ᵀg   s.t.   a * g ≤ M ,   g ≥ 0
    (where M = max(a*a)).
    The solution g is scaled to unit ℓ¹‑norm and mixed with the current a.
    Returns the best vector seen and its peak convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_max = _max_convolution(a)
    it = 0

    while time.time() < deadline:
        it += 1
        M = _max_convolution(a)

        # ----- build LP -----
        A_ub = _build_A_ub(a)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)          # maximise sum(g)
        bounds = [(0.0, None)] * n

        # ----- solve -----
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})
        if not res.success:
            # LP failed – stop the refinement
            print(f"[FW] LP failed at iter {it}: {res.message}")
            break

        g = res.x
        if g.sum() == 0.0:
            break
        g = g / g.sum()      # enforce unit ℓ¹‑norm

        # ----- mixing -----
        # a simple convex combination; a fixed step seems to work well
        alpha = 0.5
        a = (1.0 - alpha) * a + alpha * g

        # ----- keep best -----
        cur_max = _max_convolution(a)
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
            print(f"[FW] iter={it:3d}  max={best_max:.6f}")

        # stop if the improvement becomes tiny
        if it % 10 == 0 and time.time() > deadline - 5.0:
            break

    return best_a, best_max


# ----------------------------------------------------------------------
# Tiny Adam optimiser (mirroring the earlier approach, but run only a few
# iterations – mainly to polish the LP output).
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    n = a0.shape[0]
    # we keep the ℓ¹‑norm equal to 1 throughout
    eps = 1e-12
    soft = np.maximum(a0, eps)
    soft = soft / soft.sum()
    p = np.log(soft)

    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.4

    best_a = a0.copy()
    best_max = _max_convolution(a0)
    it = 0
    tau = max(_full_convolution_fft(a0).max(), 5.0)

    while time.time() < deadline:
        it += 1
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft                     # already ℓ¹‑norm = 1

        # soft‑max gradient of max convolution
        b = _full_convolution_fft(a)
        max_b = b.max()
        if tau < 1e-12:
            mask = (b == max_b).astype(float)
        else:
            diff = b - max_b
            exps = np.exp(diff / tau)
            mask = exps / exps.sum()
        # gradient w.r.t. a
        a_rev = a[::-1]
        conv_len = len(mask) + n - 1
        f_len = _next_pow2(conv_len)
        Fmask = np.fft.rfft(mask, n=f_len)
        Fa_rev = np.fft.rfft(a_rev, n=f_len)
        conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
        g = 2.0 * conv[n - 1:n - 1 + n]

        # keep best
        if max_b < best_max:
            best_max = max_b
            best_a = a.copy()

        # gradient w.r.t logits
        mu = np.dot(a, g)
        grad_p = a * (g - mu)

        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        tau *= 0.9999
        lr *= 0.9999

        if it % 200 == 0 and time.time() > deadline - 5.0:
            break

    return best_a, best_max


# ----------------------------------------------------------------------
# Very cheap hill‑climb (random mass transfer)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_max = _max_convolution(a)
    n = len(a)

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        frac = rng.uniform(0.02, 0.12)
        amount = a[i] * frac
        a[i] -= amount
        a[j] += amount
        cur_max = _max_convolution(a)
        if cur_max + 1e-12 < best_max:
            best_max = cur_max
            best_a = a.copy()
            print(f"[hill] improved max={best_max:.6f}")

    return best_a, best_max

        except Exception:
            warm_seq = None

    # ------------------------------------------------------------------
    # Phase 1 – random exploration (≈ 15 % of the budget)
    # ------------------------------------------------------------------
    explore_deadline = end_time - 350.0
    best_seq = None
    best_max = np.inf
    attempts = 0

        n = int(rng.integers(1200, 3000))
        a0 = _initial_vector(n, rng, warm_seq)
        cur_max = _max_convolution(a0)
        if cur_max < best_max:
            best_max = cur_max
            best_seq = a0.copy()
            print(f"[explore] attempt={attempts:4d} n={n:4d}  max={best_max:.6f}")

    if best_seq is None:                      # very unlikely
        n = 2000
        best_seq = np.full(n, 1.0 / n)
        best_max = _max_convolution(best_seq)

    # Phase 2 – Frank‑Wolfe LP refinement (the core improvement)
    # ------------------------------------------------------------------

    if max_fw < best_max:
        best_max = max_fw
        best_seq = a_fw.copy()

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    adam_deadline = end_time - 120.0
    if time.time() < adam_deadline:

            best_seq = a_hill.copy()

    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f}  internal max={best_max:.6f}  length={len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
