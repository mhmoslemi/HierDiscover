```python
def _hill_climb(a, target_sum, deadline):
    best = a.copy()
    best_score = _score(a)
    cur_score = best_score
    rng = np.random.default_rng()
    n = a.size
    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi+1)
        partner_idx = k_max - idx
        contrib = a[idx] * a[partner_idx]
        donor = idx[np.argmax(contrib)]

        # Choose receiver:
        # get indices sorted by contribution ascend; pick from lower 20% random
        # compute threshold percentile:
        low_thr = np.percentile(contrib, 20)
        low_idxs = idx[contrib <= low_thr]
        if low_idxs.size > 0:
            receiver = rng.choice(low_idxs)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        max_frac = 0.05
        delta = a[donor] * rng.random() * max_frac
        if delta <= 0:
            continue

        # apply
        a[donor] -= delta
        a[receiver] += delta

        # Projection not needed because sum preserved.

        new_score = _score(a)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time():.2f}] Hill climb improved best={best_score:.6f}")
        else:
            # reject
            a[donor] += delta
            a[receiver] -= delta
    return best, best_score
```

```python
def _piecewise_power_law_sequence(n, k, p1, p2, target_sum):
    i = np.arange(n, dtype=np.float64)
    seq = np.where(i < k, (i+1)**(-p1), (i+1)**(-p2))
    seq *= target_sum / seq.sum()
    return seq
```

"""
Improved step‑function search.

The algorithm builds on the previous multi‑stage pipeline and adds
two new ingredients:

1. A cheap piecewise‑power‑law warm‑start on the full problem size.
2. A random hill‑climbing refinement that repeatedly transfers a
   small amount of mass from the index contributing most to the
   current maximal convolution entry toward an index with a small
   partner contribution.  Only improving moves are kept.

All stages respect the ℓ¹‑simplex Σa = √(2n), a ≥ 0 and use fast FFT‑based
convolution for the objective 2·n·max(a*a)/Σa².
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Convolution / scoring utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact objective value 2·n·max(a*a) / (sum(a)²)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i = (i+1)^(-p) normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient utilities (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, full_convolution) where grad has the same size as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # grad = 2 * (reverse(a) convolve p)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b

# Stage 2 – exponential (multiplicative) descent
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Mirror‑descent with exponential updates using a soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 16384.0
    base_eta = 0.04
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

# Stage 3 – projected gradient descent with restarts (reuse)
# ----------------------------------------------------------------------


def _annealed_gradient_descent_v2(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the soft‑max surrogate."""
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 16384.0
    step_base = target_sum * 0.5
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.95
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.002 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GDv2 it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several short GD restarts with random jitter; keep the best."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_v2(start,
                                               time.time() + run_budget,
                                               target_sum)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val

        seq = seq * (target_sum / seq.sum())
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Random best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stage 1b – piecewise power‑law warm‑start on the full size
# ----------------------------------------------------------------------


def _piecewise_power_law_sequence(n: int,
                                  k: int,
                                  p1: float,
                                  p2: float,
                                  target_sum: float) -> np.ndarray:
    """Two‑segment power law: (i+1)^(-p1) for i<k, (i+1)^(-p2) otherwise."""
    i = np.arange(n, dtype=np.float64)
    seq = np.where(i < k, (i + 1) ** (-p1), (i + 1) ** (-p2))
    seq *= target_sum / seq.sum()
    return seq


def _search_piecewise_power_law(n: int,
                                target_sum: float,
                                deadline: float) -> tuple[np.ndarray, float]:
    """Coarse grid + random sampling for the two‑segment power‑law family."""
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    # --- coarse grid -----------------------------------------------------------------
    k_vals = np.linspace(int(0.1 * n), int(0.9 * n), 9, dtype=int)
    p_vals = np.linspace(0.3, 1.2, 10)

    for k in k_vals:
        for p1 in p_vals:
            for p2 in p_vals:
                if time.time() > deadline:
                    break
                seq = _piecewise_power_law_sequence(n, k, p1, p2, target_sum)
                val = _score(seq)
                if val < best_score:
                    best_score = val
                    best_seq = seq.copy()
                    print(f"[{time.time():.2f}] PW grid k={k} p1={p1:.3f} p2={p2:.3f} score={best_score:.6f}")

    # --- random sampling -------------------------------------------------------------
    while time.time() < deadline:
        p1 = rng.uniform(0.2, 1.2)
        p2 = rng.uniform(0.2, 1.2)
        k = rng.integers(int(0.05 * n), int(0.95 * n))
        seq = _piecewise_power_law_sequence(n, k, p1, p2, target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] PW rand k={k} p1={p1:.3f} p2={p2:.3f} score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stage 4 – random hill‑climbing refinement
# ----------------------------------------------------------------------


def _random_hill_climb(a0: np.ndarray,
                       target_sum: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """Move a small fraction of mass from a high‑contributing index to a low‑partner one."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    cur_val = best_val
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        contrib = a[idx] * a[partner]

        donor = idx[np.argmax(contrib)]

        # pick a receiver among indices whose partner contribution is small
        low_thr = np.percentile(contrib, 15)
        low_idxs = idx[contrib <= low_thr]
        if low_idxs.size:
            receiver = rng.choice(low_idxs)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # transfer a modest fraction of the donor's mass
        max_frac = 0.07          # a bit larger than the previous 0.04
        delta = a[donor] * rng.random() * max_frac
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta

        new_val = _score(a)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Hill‑climb improved best={best_val:.6f}")
        else:
            # revert the move
            a[donor] += delta
            a[receiver] -= delta

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 5 – exact sub‑gradient descent on the max‑convolution
# ----------------------------------------------------------------------


def _subgradient_max_descent(a0: np.ndarray,
                             target_sum: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true maximal convolution entry."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())

        # sub‑gradient of the maximal entry
        grad = np.zeros_like(a)
        lo = max(0, k_max - (a.shape[0] - 1))
        hi = min(k_max, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        grad[idx] = 2.0 * a[partner]

        norm = np.linalg.norm(grad)
        if norm == 0:
            break

        # safe step size that keeps all entries non‑negative before projection
        safe_eta = np.min(np.where(grad > 0, a / grad, np.inf))
        eta = min(0.5 * safe_eta, 0.5 / np.sqrt(it + 1))

        a = a - eta * grad
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it} score={best_val:.6f}")

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """Search for a non‑negative step sequence with a low benchmark score."""
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Choose problem size
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Initial sequence (warm‑start from previous run if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"],
                                        dtype=np.float64), n)
        init = init * (target_sum / init.sum())
    else:
        init = _power_law_sequence(n, 0.55, target_sum)

    # ------------------------------------------------------------------
    # Stage 1 – cheap parametric search on a reduced scale
    # ------------------------------------------------------------------
    small_n = max(2000, n // 4)
    small_target = math.sqrt(2.0 * small_n)
    param_deadline = time.time() + min(20.0,
                                      (deadline - time.time()) * 0.10)
    best_small_seq, _ = _grid_param_search(small_n, small_target, param_deadline)

    if best_small_seq is not None:
        init = _resize_sequence(best_small_seq, n)
        init = init * (target_sum / init.sum())
        print(f"[{time.time():.2f}] Parametric warm‑start ready (n={n})")

    # ------------------------------------------------------------------
    # Stage 1b – piecewise power‑law warm‑start on the full size
    # ------------------------------------------------------------------
    if time.time() < deadline:
        piece_deadline = time.time() + min(15.0,
                                          (deadline - time.time()) * 0.05)
        pw_seq, pw_score = _search_piecewise_power_law(n, target_sum, piece_deadline)
        if pw_seq is not None and pw_score < _score(init):
            init = pw_seq
            print(f"[{time.time():.2f}] Piecewise warm‑start improved init, score={pw_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – exponential (multiplicative) descent (~15 % of remaining)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        exp_deadline = time.time() + (deadline - time.time()) * 0.15
        a_exp, val_exp = _exponential_descent(init, target_sum, exp_deadline)
        best_seq = a_exp
        best_score = val_exp
        print(f"[{time.time():.2f}] After ExpGD best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – projected gradient descent with restarts (~35 % of remaining)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        gd_deadline = time.time() + (deadline - time.time()) * 0.35
        a_gd, val_gd = _gradient_descent_with_restarts(best_seq,
                                                       target_sum,
                                                       gd_deadline)
        if val_gd < best_score:
            best_seq = a_gd
            best_score = val_gd
        print(f"[{time.time():.2f}] After GD best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – random hill‑climbing refinement (~20 % of remaining)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        hill_deadline = time.time() + (deadline - time.time()) * 0.20
        a_hill, val_hill = _random_hill_climb(best_seq, target_sum, hill_deadline)
        if val_hill < best_score:
            best_seq = a_hill
            best_score = val_hill
        print(f"[{time.time():.2f}] After hill‑climb best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – sub‑gradient descent on exact max‑convolution (remaining)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_sub, val_sub = _subgradient_max_descent(best_seq,
                                                  target_sum,
                                                  deadline)
        if val_sub < best_score:
            best_seq = a_sub
            best_score = val_sub
        print(f"[{time.time():.2f}] After sub‑gradient best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up, final evaluation and store for warm‑starts
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
