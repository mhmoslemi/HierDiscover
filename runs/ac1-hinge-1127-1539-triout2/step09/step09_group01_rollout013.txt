def _lp_optimize_g(f, M, target_sum, eps_active=0.02):
    # f: np.ndarray shape (n,)
    # M: scalar (max conv)
    # returns g (np.ndarray shape (n,))
    n = f.shape[0]
    conv_f = _conv_fft(f)
    thr = M * (1 - eps_active)
    active_idx = np.where(conv_f >= thr)[0]
    if active_idx.size == 0:
        active_idx = np.array([np.argmax(conv_f)])
    m = active_idx.shape[0]

    # Build matrix A (m,n)
    A = np.zeros((m, n), dtype=np.float64)
    for row_i, k in enumerate(active_idx):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # slice of f: reversed part
        segment = f[k - i_end : k - i_start + 1][::-1]
        A[row_i, i_start:i_end+1] = segment

    b = np.full(m, M)
    c = -np.ones(n)  # maximize sum

    max_f = f.max()
    if max_f > 0:
        ub = M / max_f
    else:
        ub = np.inf
    bounds = [(0, ub) for _ in range(n)]

    # solve
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        # fallback: return f
        return f.copy()
    g0 = res.x
    # scale to target_sum
    sum_g0 = g0.sum()
    if sum_g0 <= 0:
        # fallback
        return f.copy()
    g = g0 * (target_sum / sum_g0)
    return g

def _lp_refinement(f, target_sum, deadline, t=0.2, eps_active=0.02, max_iters=10):
    best = f.copy()
    best_score = _score(f)
    while time.time() < deadline and max_iters > 0:
        conv_f = _conv_fft(f)
        M = conv_f.max()
        # solve LP to get g
        g = _lp_optimize_g(f, M, target_sum, eps_active)
        # update:
        f_new = (1 - t) * f + t * g
        # project onto simplex
        f_new = _project_to_simplex(f_new, target_sum)
        # evaluate
        new_score = _score(f_new)
        if new_score < best_score:
            best_score = new_score
            best = f_new.copy()
            print(f"[{time.time():.2f}] LP refine improved score={best_score:.6f}")
        # reduce t maybe
        t *= 0.9
        f = f_new
        max_iters -= 1
    return best, best_score

2·n·max(convolution) / (sum)^2 benchmark.

The algorithm proceeds in stages:
1) initialise with a power‑law shape (or a previous best);
2) a short projected gradient descent on a soft‑max smoothed max‑convolution;
3) an LP‑based refinement that exploits the classic “f –> g” update from
   the editorial: maximise Σg subject to f∗g ≤ M;
4) a final high‑β projected gradient descent (polish);
5) optional re‑ordering heuristics.

All stages respect the 1000 s wall‑clock budget.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective when Σa = √(2n). Returns max(conv)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Projected gradient descent (momentum, annealed soft‑max)
# ----------------------------------------------------------------------


def _annealed_gradient_descent_v2_ext(a0: np.ndarray,
                                     deadline: float,
                                     target_sum: float,
                                     max_beta: float = 1e5,
                                     step_factor: float = 0.7,
                                     gamma: float = 0.99) -> tuple[np.ndarray, float]:
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD_ext it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_v2_ext(
            start,
            time.time() + run_budget,
            target_sum,
            max_beta=1e5,
            step_factor=0.8,
            gamma=0.99,
        )
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# LP‑based “f → g” refinement
# ----------------------------------------------------------------------


def _lp_optimize_g(f: np.ndarray, M: float, target_sum: float,
                   top_k: int = 200) -> np.ndarray:
    """Return a feasible g (scaled to target_sum) obtained by solving
    max Σ g_j s.t. f∗g ≤ M on a selected set of constraints."""
    n = f.shape[0]

    # compute convolution once
    conv_f = _conv_fft(f)

    # pick indices with largest values (most binding constraints)
    k = min(top_k, conv_f.shape[0])
    top_idx = np.argpartition(-conv_f, k - 1)[:k]

    # build dense matrix A of shape (k, n)
    A = np.zeros((k, n), dtype=np.float64)
    for row, kk in enumerate(top_idx):
        i_start = max(0, kk - (n - 1))
        i_end = min(kk, n - 1)
        segment = f[kk - i_end: kk - i_start + 1][::-1]  # reversed slice
        A[row, i_start:i_end + 1] = segment

    b = np.full(k, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # maximise sum → minimise -sum

    # simple finite upper bounds keep the LP bounded
    ub = target_sum
    bounds = [(0, ub) for _ in range(n)]

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True, 'disp': False})

    if not res.success:
        # fallback: return the input vector
        print(f"[{time.time():.2f}] LP failed (status {res.status}); using f")
        return f.copy()

    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 <= 0:
        return f.copy()
    g = g0 * (target_sum / sum_g0)
    return g


def _lp_refinement(f: np.ndarray, target_sum: float,
                   deadline: float, t: float = 0.3,
                   top_k: int = 200, max_iters: int = 8) -> tuple[np.ndarray, float]:
    """Iteratively improve f using the linear‑programmed direction."""
    best = f.copy()
    best_score = _score(best)
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _conv_fft(f)
        M = conv.max()
        g = _lp_optimize_g(f, M, target_sum, top_k=top_k)

        # convex combination
        f_new = (1.0 - t) * f + t * g
        f_new = _project_to_simplex(f_new, target_sum)
        new_score = _score(f_new)

        if new_score < best_score:
            best_score = new_score
            best = f_new.copy()
            print(f"[{time.time():.2f}] LP iteration {it} improved score={best_score:.6f}")

        # prepare next iteration
        f = f_new
        t = max(t * 0.7, 0.02)  # slowly shrink step size
        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Simple re‑ordering helper (optional)
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left = 0
    right = n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out

    top_idx = np.argpartition(-conv_f, k - 1)[:k]

    # Build dense matrix A (k × n) with rows = selected convolution constraints
    A = np.zeros((k, n), dtype=np.float64)
    for row, kk in enumerate(top_idx):
        i_start = max(0, kk - (n - 1))
        i_end = min(kk, n - 1)
        # coefficients are f[kk-i] for i in [i_start, i_end]
        segment = f[kk - i_end: kk - i_start + 1][::-1]   # reversed slice
        A[row, i_start:i_end + 1] = segment

    b = np.full(k, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # minimising –Σg  ⇔  maximising Σg

    # simple finite upper bounds keep the LP bounded
    ub = target_sum
    bounds = [(0, ub) for _ in range(n)]

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True, 'disp': False})

    if not res.success:
        # fallback: keep the current vector
        print(f"[{time.time():.2f}] LP failed (status {res.status}); using f")
        return f.copy()

    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 <= 0:
        return f.copy()
    g = g0 * (target_sum / sum_g0)
    return g


def _lp_refinement(f: np.ndarray, target_sum: float,
                   deadline: float,
                   t: float = 0.3,
                   top_k: int = 200,
                   max_iters: int = 8) -> tuple[np.ndarray, float]:
    """
    Iteratively improve f using the linear‑programmed direction.
    Returns the best sequence encountered and its score.
    """
    best = f.copy()
    best_score = _score(best)
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _conv_fft(f)
        M = conv.max()
        g = _lp_optimize_g(f, M, target_sum, top_k=top_k)

        # convex combination
        f_new = (1.0 - t) * f + t * g
        f_new = _project_to_simplex(f_new, target_sum)
        new_score = _score(f_new)

        if new_score < best_score:
            best_score = new_score
            best = f_new.copy()
            print(f"[{time.time():.2f}] LP iteration {it} improved score={best_score:.6f}")

        # prepare next iteration
        f = f_new
        t = max(t * 0.7, 0.02)   # gradually shrink step size
        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Simple re‑ordering helper (optional)
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """
    Reorder a sequence by placing the largest value at the leftmost,
    the second largest at the rightmost, the third largest next to the left,
    and so on – an alternating “big–small’’ pattern.
    """
    sorted_vals = np.sort(seq)[::-1]          # descending
    n = seq.shape[0]
    out = np.empty_like(seq)
    left = 0
    right = n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The algorithm proceeds through gradient descent, LP‑based refinement,
    optional re‑ordering, and a final polishing stage.
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n gives a tighter theoretical bound)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Build initial candidate
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if init.shape[0] != n:
            init = _resize_sequence(init, n)
    else:
        init = _power_law_sequence(n, 0.55, target_sum)

    init = _project_to_simplex(init, target_sum)
    best_seq = init.copy()
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] Initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Stage A – short projected gradient descent (exploration)
    # --------------------------------------------------------------
    if time.time() < deadline:
        gd_deadline = time.time() + (deadline - time.time()) * 0.25
        a_gd, score_gd = _gradient_descent_with_restarts(best_seq,
                                                         target_sum,
                                                         gd_deadline)
        if score_gd < best_score:
            best_seq = a_gd.copy()
            best_score = score_gd
            print(f"[{time.time():.2f}] After GD best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Stage B – LP‑based refinement (theoretically motivated)
    # --------------------------------------------------------------
    if time.time() < deadline:
        lp_deadline = time.time() + (deadline - time.time()) * 0.30
        a_lp, score_lp = _lp_refinement(best_seq,
                                         target_sum,
                                         lp_deadline,
                                         t=0.4,
                                         top_k=250,
                                         max_iters=12)
        if score_lp < best_score:
            best_seq = a_lp.copy()
            best_score = score_lp
            print(f"[{time.time():.2f}] After LP refinement best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Stage C – cheap re‑ordering attempt
    # --------------------------------------------------------------
    if time.time() < deadline:
        cand = _alternating_order(best_seq)
        cand_score = _max_convolution_fft(cand)
        if cand_score < best_score:
            best_seq = cand.copy()
            best_score = cand_score
            print(f"[{time.time():.2f}] After ordering best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Stage D – final high‑β projected gradient polishing
    # --------------------------------------------------------------
    if time.time() + 3.0 < deadline:
        a_fin, score_fin = _annealed_gradient_descent_v2_ext(
            best_seq,
            deadline,
            target_sum,
            max_beta=3e5,
            step_factor=0.6,
            gamma=0.995,
        )
        if score_fin < best_score:
            best_seq = a_fin.copy()
            best_score = score_fin
            print(f"[{time.time():.2f}] Final polished best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Trim trailing zeros and final evaluation
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as exc:
        print(f"Final evaluation error: {exc}")

    # expose the sequence for possible later runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
