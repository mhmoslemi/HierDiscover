"""
Propose a step‑function sequence that minimizes ``evaluate_sequence``.

The algorithm consists of three stages:

1️⃣ **Diverse random sampling** – draw many candidate vectors of
   various lengths (2 000‑12 000) from several parametric families
   (exponential, power‑law, half‑sine, Gaussian, Dirichlet, sparse,
   uniform).  Each candidate is normalised to the required ℓ₁‑norm
   (sum = √(2 n)).  The best ~20 candidates are kept.

2️⃣ **Exact sub‑gradient Adam on the simplex** – for each shortlisted
   candidate we run an Adam optimiser on the log‑softmax parametrisation
   of the simplex, using the exact sub‑gradient of the hard maximum
   (τ‑soft‑max with τ → 0).  The optimiser respects the ℓ₁‑norm by
   re‑projecting through the log‑softmax trick.  A short but aggressive
   optimisation (≈ 30 s per candidate) refines the vectors.

3️⃣ **Final stochastic fine‑tuning** – a cheap simulated‑annealing mass‑transfer
   walk followed by a strictly downhill hill‑climber extracts any remaining
   tiny improvements.

All work is bounded by the 1000 s budget; the best construction is stored
in the global ``height_sequence_1`` and also returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution (a * a) using a real FFT."""
    n = a.shape[0]
    tot = 2 * n - 1
    f_len = _next_pow2(tot)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:tot]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Peak value of a*a (full convolution)."""
    return _full_convolution_fft(a).max()


# ----------------------------------------------------------------------
# Sub‑gradient of the exact hard‑max (τ‑soft‑max for stability)
# ----------------------------------------------------------------------
def _max_gradient(a: np.ndarray, tau: float):
    """
    Returns (grad, max_val) where ``grad = ∂ max(a*a) / ∂a``.
    ``tau`` controls the soft‑max temperature; very small values
    switch to an exact active‑set average.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:                # hard max → average over all active indices
        eps = max_b * 1e-9 + 1e-12
        mask = (b >= max_b - eps).astype(float)
        s = mask.sum()
        if s == 0:
            mask = np.ones_like(b) / b.size
        else:
            mask /= s
    else:                           # τ‑soft‑max
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / exps.sum()

    # gradient: 2 * (mask * a_rev) convolved with a_rev
    a_rev = a[::-1]
    conv_len = mask.size + a.size - 1
    f_len = _next_pow2(conv_len)

    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)

    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    grad = 2.0 * conv[a.size - 1:a.size - 1 + a.size]
    return grad, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   lr0: float = 0.5,
                   tau_decay: float = 0.9995,
                   lr_decay: float = 0.99985,
                   jitter_every: int = 400) -> tuple[np.ndarray, float]:
    """
    Minimise the peak of the self‑convolution under the ℓ₁‑norm
    constraint (∑a = √(2 n)).  Returns the best vector seen and its
    hard peak (max convolution value).
    """
    n = a0.shape[0]
    S = math.sqrt(2.0 * n)                     # required ℓ₁‑norm

    # initialise logits from a feasible a0 (log‑softmax)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                           # logits = log‑softmax parameters

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr0

    # initialise temperature for τ‑soft‑max
    tau = max(_full_convolution_fft(a0).max(), 0.5)

    best_a = a0.copy()
    best_peak = _max_convolution(a0)
    it = 0

    while time.time() < deadline:
        it += 1

        # ------- reconstruct a from logits (log‑softmax) -------
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)               # stability shift
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ------- gradient of (soft‑)max convolution -------
        g, peak = _max_gradient(a, tau)

        # keep best hard peak seen so far
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ------- chain rule: gradient w.r.t. logits -------
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ------- Adam update -------
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ------- annealing schedules -------
        tau *= tau_decay
        lr *= lr_decay

        # occasional jitter to escape plateaus
        if jitter_every and (it % jitter_every) == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Random families for the exploration phase
# ----------------------------------------------------------------------
def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotone decreasing exponential a_i = exp(-λ·i)."""
    lam = rng.uniform(1e-4, 5e-3)             # λ ≈ 10⁻⁴ … 5·10⁻³
    return np.exp(-lam * np.arange(n))


def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """Power‑law decay a_i = (i + offset)^(-p)."""
    p = rng.uniform(0.2, 3.0)
    offset = rng.uniform(0.5, 10.0)
    idx = np.arange(n)
    return 1.0 / ((idx + offset) ** p)


def _init_half_sine(n: int, rng: np.random.Generator) -> np.ndarray:
    """Half‑sine shape (zero at the ends, peak in the middle)."""
    x = np.arange(n)
    raw = np.sin(np.pi * x / (n - 1))
    raw = raw * rng.uniform(0.5, 2.0) + rng.uniform(0.0, 0.2)
    return raw


def _init_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    """Centered Gaussian bump."""
    mu = (n - 1) / 2.0
    sigma = rng.uniform(0.05, 0.30) * n
    x = np.arange(n)
    return np.exp(-0.5 * ((x - mu) / sigma) ** 2)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet‑like vector (Gamma draws)."""
    alpha = rng.uniform(0.2, 5.0)
    return rng.gamma(alpha, 1.0, size=n)


def _init_sparse(n: int, rng: np.random.Generator) -> np.ndarray:
    """Few random rectangular bumps on a small background."""
    raw = np.full(n, rng.uniform(0.0, 0.12))
    nbumps = rng.integers(1, max(2, n // 250))
    for _ in range(nbumps):
        centre = rng.integers(0, n)
        width = rng.integers(3, max(5, n // 100))
        height = rng.uniform(0.8, 4.0)
        lo = max(0, centre - width // 2)
        hi = min(n, lo + width)
        raw[lo:hi] = np.maximum(raw[lo:hi], height)
    return raw


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Simulated‑annealing / greedy hill‑climbing fine‑tuner
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Mass‑transfer walk with exponential acceptance."""
    n = len(seq)
    S = math.sqrt(2.0 * n)

    cur = seq.copy()
    cur_val = _max_convolution(cur)
    best = cur.copy()
    best_val = cur_val
    temperature = 0.15 * cur_val

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] <= 0.0:
            continue
        delta = rng.uniform(0.0, 0.3) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())
        new_val = _max_convolution(new)

        if new_val < cur_val:
            accept = True
        else:
            prob = math.exp(-(new_val - cur_val) / max(temperature, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best = new.copy()
                best_val = new_val

        temperature *= 0.99995
        if temperature < 1e-6:
            temperature = 1e-6

    return best, best_val


def _hill_climb(seq: np.ndarray,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Very cheap greedy hill‑climber (strictly downhill moves only)."""
    n = len(seq)
    S = math.sqrt(2.0 * n)

    cur = seq.copy()
    cur_val = _max_convolution(cur)
    best = cur.copy()
    best_val = cur_val

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] <= 0.0:
            continue
        delta = rng.uniform(0.0, 0.15) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())
        new_val = _max_convolution(new)
        if new_val < cur_val:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best = new.copy()
                best_val = new_val
    return best, best_val


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The algorithm proceeds in three stages: random exploration,
    Adam‑based sub‑gradient refinement, and a final stochastic fine‑tuning.
    The best construction is stored in the global ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    safety_margin = 5.0
    end_time = time.time() + budget_s - safety_margin
    if end_time <= time.time():
        raise RuntimeError("Budget too small after safety margin.")

    # ------------------------------------------------------------------
    # Warm‑start from a previously stored construction, if any
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # ------------------------------------------------------------------
    # Phase 1 – fast random exploration (≈30 % of budget)
    # ------------------------------------------------------------------
    explore_deadline = time.time() + 0.30 * budget_s
    top_k = 20
    top_candidates = []            # list of (score, seq)
    best_seq = None
    best_score = np.inf
    attempts = 0

    families = ['uniform', 'exponential', 'powerlaw',
                'half_sine', 'gaussian', 'dirichlet', 'sparse']

    while time.time() < explore_deadline:
        attempts += 1
        n = int(rng.integers(2000, 12000))

        # Occasionally warm‑start from the global best
        if warm_seq is not None and rng.random() < 0.15:
            a = _scale_sequence(warm_seq, n)
        else:
            style = rng.choice(families)
            if style == 'uniform':
                a = np.full(n, 1.0)
            elif style == 'exponential':
                a = _init_exponential(n, rng)
            elif style == 'powerlaw':
                a = _init_powerlaw(n, rng)
            elif style == 'half_sine':
                a = _init_half_sine(n, rng)
            elif style == 'gaussian':
                a = _init_gaussian(n, rng)
            elif style == 'dirichlet':
                a = _init_dirichlet(n, rng)
            else:                      # sparse
                a = _init_sparse(n, rng)

        a = np.clip(a, 0.0, None)
        a = a * (math.sqrt(2.0 * n) / a.sum())
        score = _max_convolution(a)           # internal objective (same as evaluate_sequence)

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[explore] attempt={attempts:5d} n={n:5d} new best={best_score:.6f}")

        # maintain top‑k list
        if len(top_candidates) < top_k:
            top_candidates.append((score, a.copy()))
            top_candidates.sort(key=lambda x: x[0])
        elif score < top_candidates[-1][0]:
            top_candidates[-1] = (score, a.copy())
            top_candidates.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # Safety net – guarantee at least one candidate
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 4000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n)
        best_score = _max_convolution(best_seq)
        top_candidates = [(best_score, best_seq.copy())]

    # ------------------------------------------------------------------
    # Phase 2 – Adam refinement (≈55 % of budget)
    # ------------------------------------------------------------------
    # allocate a modest slice of remaining time per candidate
    for idx, (init_score, init_seq) in enumerate(top_candidates):
        if time.time() >= end_time - 30.0:
            break

        # run Adam on the original length (no up‑sampling – length already
        # sufficiently large for the candidate)
        slice_budget = min((end_time - time.time()) * 0.45, 180.0)

        refined, peak = _adam_optimize(init_seq,
                                       time.time() + slice_budget,
                                       rng,
                                       lr0=0.52,
                                       tau_decay=0.9994,
                                       lr_decay=0.99985,
                                       jitter_every=300)

        if peak < best_score:
            best_score = peak
            best_seq = refined.copy()
            print(f"[refine] init#{idx+1} len={len(refined):5d} improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final stochastic fine‑tuning (≈15 % of budget)
    # ------------------------------------------------------------------
    remaining = end_time - time.time()
    if remaining > 20.0:
        best_seq, best_score = _anneal(best_seq,
                                       time.time() + 0.6 * remaining,
                                       rng)
        remaining = end_time - time.time()
        if remaining > 5.0:
            best_seq, best_score = _hill_climb(best_seq,
                                               time.time() + remaining,
                                               rng)

    # ------------------------------------------------------------------
    # Final sanity check via the supplied black‑box evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f} internal max = {best_score:.6f} "
          f"length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
