def _subgradient_descent(a0, target_sum, deadline, base_step_factor=0.3, momentum_gamma=0.9, max_backtrack=5):

    a = _project_to_simplex(a0.copy(), target_sum)

    cur_score = _score(a)

    best_a = a.copy(); best_score = cur_score

    momentum = np.zeros_like(a)

    step_base = target_sum * base_step_factor

    iteration = 0

    while time.time() < deadline:

        conv = _conv_fft(a)

        M = conv.max()

        # tolerance for active set
        tol = max(M * 1e-12, 1e-12)  # small

        idx = np.where(conv >= M - tol)[0]

        p = np.zeros_like(conv)

        # distribute weight equally among active indices (subgradient)
        if idx.size > 0:
            p[idx] = 1.0 / idx.size

        # compute gradient via weighted convolution
        # we reuse _gradient_weighted
        grad = _gradient_weighted(a, p)

        # momentum update
        momentum = momentum_gamma * momentum + grad

        norm = np.linalg.norm(momentum)

        if norm == 0:
            # cannot progress
            break

        # step size scaling: decreasing.
        lr = step_base / (norm * (np.sqrt(iteration + 1) + 1))

        # backtracking line search
        improved = False
        a_candidate = a.copy()
        for _ in range(max_backtrack):
            a_new = a_candidate - lr * momentum
            a_new = _project_to_simplex(a_new, target_sum)
            new_score = _score(a_new)
            if new_score < cur_score - 1e-12:
                a = a_new
                cur_score = new_score
                improved = True
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Subgrad it={iteration} lr={lr:.4e} score={best_score:.6f}")
                break
            else:
                lr *= 0.5

        if not improved:
            # no improvement, maybe jitter and continue
            # optionally occasional random jitter
            pass

        iteration += 1

    return best_a, best_score

def _gradient_weighted(a, p):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    p_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[:2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad

"""
Optimiser for the step‑function benchmark.

The algorithm combines several ideas:

* 0️⃣  Fast parametric seeds (power‑law, exponential tail, Gaussian,
  Beta‑distribution) are generated and the best few are kept.
* 1️⃣  A sub‑gradient descent that directly uses the exact max‑convolution
  (instead of a soft‑max approximation).  The active set of indices is
  taken as a convex combination, a Nesterov‑style momentum term is used,
  and a back‑tracking line‑search guarantees improvement.
* 2️⃣  Deterministic mass‑transfer moves (as in the reference solution) that
  attack the current maximal convolution entry.
* 3️⃣  Global random mass‑transfer (simulated‑annealing style) for extra
  exploration.
* 4️⃣  A final polishing phase that runs the soft‑max gradient descent
  with a very large temperature (β≈1e6) to fine‑tune the solution.

All phases respect the overall time budget (default 1000 s) and the
sequence is projected onto the simplex Σa = sqrt(2 n) after each update.
The routine returns a plain Python list of floats and prints progress
information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Benchmark objective 2·n·max(a*a) / (∑a)²."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families of parametric seeds
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) scaled to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) * exp(-q * i / n) scaled to target_sum."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, mu_frac: float, sigma_frac: float, target_sum: float) -> np.ndarray:
    """Discretised Gaussian, mu = mu_frac*(n-1), sigma = sigma_frac*(n-1)."""
    i = np.arange(n, dtype=np.float64)
    mu = mu_frac * (n - 1)
    sigma = max(sigma_frac * (n - 1), 1e-6)
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _beta_sequence(n: int, alpha: float, beta: float, target_sum: float) -> np.ndarray:
    """Beta distribution shape on {0,…,n‑1}, scaled to target_sum."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    a += 1e-12                # avoid all zeros
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Gradient of a weighted max‑convolution (exact subgradient)
# ----------------------------------------------------------------------


def _gradient_weighted(a: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k weight_k (a*a)[k] with respect to a.
    weight must have length 2·n‑1 and be non‑negative.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    w_fft = np.fft.rfft(weight, n=size)
    conv_rev = np.fft.irfft(a_rev_fft * w_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         base_step_factor: float = 0.6,
                         momentum_gamma: float = 0.9,
                         max_backtrack: int = 5) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the max‑convolution (exact active set).
    Returns the best vector found and its score.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    momentum = np.zeros_like(a)
    step_base = target_sum * base_step_factor
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        M = conv.max()
        # active set – all indices within a relative tolerance of the max
        eps = max(M * 1e-12, 1e-12)
        active = np.where(conv >= M - eps)[0]
        weight = np.zeros_like(conv)
        if active.size:
            weight[active] = 1.0 / active.size   # convex combination

        grad = _gradient_weighted(a, weight)

        # momentum update
        momentum = momentum_gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm == 0.0:
            break

        lr = step_base / (norm * (np.sqrt(it + 1) + 1.0))

        improved = False
        a_candidate = a.copy()
        for _ in range(max_backtrack):
            a_new = a_candidate - lr * momentum
            a_new = _project_to_simplex(a_new, target_sum)
            new_score = _score(a_new)
            if new_score < cur_score - 1e-12:
                a = a_new
                cur_score = new_score
                improved = True
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Subgrad it={it:5d} lr={lr:.2e} score={best_score:.6f}")
                break
            else:
                lr *= 0.5

        if not improved:
            # occasional jitter to escape flat areas
            if it % 200 == 0:
                rng = np.random.default_rng()
                jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
                a = _project_to_simplex(a + jitter, target_sum)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Soft‑max gradient descent (used for final polishing)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, full_convolution) where grad is the gradient vector.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 2e5,
                               step_factor: float = 0.6,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max approximation.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    step_base = target_sum * step_factor
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 300 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (Phase B)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-5,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy moves that directly reduce the current maximal convolution entry.
    Uses an exponentially decaying temperature for occasional uphill steps.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0.0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0.0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Global random mass‑transfer (exploration)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing moves: shift a small amount of mass from the
    largest entries to the smallest ones, accepting only improving moves.
    """
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random move new best={best_score:.6f}")

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:

            candidates.append(_exp_power_law_sequence(n, p, q, target_sum))

    # (c) Gaussian family

    # (d) Beta distribution family
    for alpha in (0.5, 1.0, 2.0):
        for beta in (0.5, 1.0, 2.0):

    # (e) previous best, if available
    if "height_sequence_1" in globals():

    # jittered variants of everything above

    # evaluate a manageable subset
