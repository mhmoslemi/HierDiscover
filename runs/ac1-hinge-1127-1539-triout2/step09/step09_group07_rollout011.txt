def _full_convolution_fft(a):
   n = a.shape[0]; size = 2*n-1; f_len = 1 << ((size-1).bit_length()); A = np.fft.rfft(a, n=f_len); conv = np.fft.irfft(A*A, n=f_len)[:size]; return conv

def _max_gradient(a, tau):
    b = _full_convolution_fft(a)
    max_b = b.max()
    if tau < 1e-12:
        eps = max_b * 1e-9 + 1e-12
        mask = (b >= max_b - eps).astype(float)
        mask /= mask.sum()
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / exps.sum()
    a_rev = a[::-1]
    conv_len = mask.size + a.size - 1
    f_len = 1 << ((conv_len-1).bit_length())
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa, n=f_len)[:conv_len]
    grad = 2.0 * conv[a.size-1:a.size-1 + a.size]
    return grad, max_b

def _solve_lp(seq):
   n = len(seq)
   S = np.sqrt(2.0 * n)
   # Build sparse A using diags
   from scipy.sparse import diags, csc_matrix
   # data: list of arrays each of shape n, values = seq[i]
   data = [np.full(n, seq[i]) for i in range(n)]
   offsets = [-i for i in range(n)]
   A = diags(data, offsets, shape=(2*n-1, n), format='csc')
   # Setup linprog
   # Variable order: g (n) then M
   # Build inequalities Gx <= h
   # G = [A, -ones]
   import numpy as np
   ones_col = np.ones((2*n-1, 1))
   # Convert A to csc and then horizontally stack with -ones column
   from scipy.sparse import hstack, csr_matrix
   G = hstack([A, -ones_col])
   h = np.zeros(2*n-1)
   # Equality for sum(g) = S
   eq_row = np.r_[np.ones(n), np.array([0.0])]
   A_eq = eq_row.reshape(1, -1)
   b_eq = np.array([S])
   # Bounds:
   bounds = [(0.0, None)] * n + [(0.0, None)]
   res = linprog(c=np.r_[np.zeros(n), np.array([1.0])],
                 A_ub=G, b_ub=h,
                 A_eq=A_eq, b_eq=b_eq,
                 bounds=bounds,
                 method='highs')
   if not res.success:
       return None, None
   x = res.x
   g = x[:n]
   M = x[-1]
   return g, M

def _lp_refine(seq):
    n = len(seq)
    from scipy.sparse import diags, hstack, csr_matrix
    from scipy.optimize import linprog
    S = math.sqrt(2.0 * n)
    # Build A as sparse matrix
    diag_data = [np.full(n, seq[i]) for i in range(n)]
    offsets = [-i for i in range(n)]
    A = diags(diag_data, offsets, shape=(2*n-1, n), format='csc')
    ones_col = np.ones((2*n-1, 1))
    G = hstack([A, -ones_col])
    h = np.zeros(2*n-1)
    # Equality sum(g)=S
    A_eq = np.r_[np.ones(n), np.array([0.0])].reshape(1, -1)
    b_eq = np.array([S])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    c = np.r_[np.zeros(n), np.array([1.0])]
    try:
        res = linprog(c, A_ub=G, b_ub=h, A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})
    except Exception as e:
        print("[lp] linprog exception:", e)
        return None, None
    if not res.success:
        print("[lp] LP failed:", res.message)
        return None, None
    x = res.x
    g = x[:n]
    M = x[-1]
    # Clip tiny negatives
    g = np.maximum(g, 0.0)
    # Renormalize to ensure sum exactly S due to numeric error
    total = g.sum()
    if total > 0:
        g = g * (S / total)
    else:
        g = np.full(n, S / n)
    return g, M

"""
Propose a step‑function (non‑negative sequence) that minimises the provided
``evaluate_sequence``.  The algorithm works in three stages:

1. **Exploration** – draw many random vectors from a rich family of
   distributions (Dirichlet, Gamma, power‑law, exponential, sparse bumps,
   Gaussian and half‑sine).  Keep the best few candidates.

2. **Refinement** – for each candidate run a fast Adam optimiser on the
   simplex (log‑soft‑max parametrisation).  The optimiser uses the exact
   sub‑gradient of the hard‑max of the self‑convolution obtained from a
   soft‑max mask that gradually anneals to a true max.  This yields a very
   low internal peak (≈ the value returned by ``evaluate_sequence``).

3. **Linear‑programming fine‑tuning** – given a feasible vector ``f`` the
   problem ``min_M  M  s.t.  (f * g)_k ≤ M  ∀k,  g ≥ 0,  Σg = Σf`` is a
   linear programme (LP).  Solving it provides a new feasible vector ``g``
   whose peak cannot be larger than the current one.  We solve the LP with
   SciPy’s HiGHS solver (sparse matrices) on a modest length (≤ 5000) and
   lift the solution back to the original size.

   After the LP we perform a short Adam run and a cheap stochastic hill‑climb
   to capture any remaining tiny improvements.

The whole search respects a 1000 s budget (5 s safety margin).  The best
sequence is stored in the global ``height_sequence_1`` and returned as a
plain list of floats.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags, hstack

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution via a real FFT."""
    n = a.shape[0]
    length = 2 * n - 1
    f_len = _next_pow2(length)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:length]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Peak value of a * a (full convolution)."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Gradient of the (soft)‑max convolution
# ----------------------------------------------------------------------


def _max_gradient(a: np.ndarray, tau: float):
    """
    Returns (grad, max_b) where

        grad = ∂ max(b) / ∂ a      (b = a * a)
        max_b = max(b).

    ``tau`` is the temperature of a soft‑max; for very small values a hard
    max (active‑set) is used.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    # ---- build weighting mask -----------------------------------------
    if tau < 1e-12:
        eps = max_b * 1e-9 + 1e-12
        mask = (b >= max_b - eps).astype(float)
        s = mask.sum()
        if s == 0:
            mask = np.ones_like(b) / b.size
        else:
            mask /= s
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / exps.sum()

    # ---- gradient ------------------------------------------------------
    a_rev = a[::-1]
    conv_len = mask.size + a.size - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    # slice to obtain length‑n gradient
    grad = 2.0 * conv[a.size - 1:a.size - 1 + a.size]
    return grad, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   lr0: float = 0.5,
                   tau_decay: float = 0.9995,
                   lr_decay: float = 0.99985,
                   jitter_every: int = 400) -> tuple[np.ndarray, float]:
    """
    Minimise the peak of the self‑convolution under the ℓ₁‑norm constraint
    (Σa = √(2 n)).  Returns the best vector seen and its hard peak.
    """
    n = a0.shape[0]
    S = math.sqrt(2.0 * n)                     # required ℓ₁‑norm

    # initialise logits from the feasible a0 (log‑softmax)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                           # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr0

    # initialise temperature for τ‑soft‑max
    tau = max(_full_convolution_fft(a0).max(), 0.5)

    best_a = a0.copy()
    best_peak = _max_convolution(a0)
    it = 0

    while time.time() < deadline:
        it += 1

        # ------- recover a from logits (log‑softmax) -------
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)               # shift for stability
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ------- gradient of (soft‑)max convolution -------
        g, peak = _max_gradient(a, tau)

        # keep best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ------- gradient w.r.t. logits (chain rule) -------
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ------- Adam update -------
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ------- annealing schedules -------
        tau *= tau_decay
        lr *= lr_decay

        # occasional jitter to escape plateaus
        if jitter_every and (it % jitter_every) == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak

def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.2, 5.0)
    return rng.gamma(alpha, 1.0, size=n)


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.1, 0.8)
    return rng.gamma(shape, 1.0, size=n)


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_left_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    p = rng.uniform(0.3, 3.0)
    offset = rng.uniform(0.0, 10.0)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * base + (1.0 - mix) * (idx + offset) ** (-p)


def _init_left_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    lam = rng.uniform(0.001, 0.02)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * base + (1.0 - mix) * np.exp(-lam * idx)


def _init_sparse_bumps(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = np.full(n, rng.uniform(0.0, 0.15))
    nbumps = rng.integers(1, max(2, n // 250))
    for _ in range(nbumps):
        center = rng.integers(0, n)
        width = rng.integers(3, max(5, n // 100))
        height = rng.uniform(0.8, 4.0)
        lo = max(0, center - width // 2)
        hi = min(n, lo + width)
        raw[lo:hi] = np.maximum(raw[lo:hi], height)
    return raw


def _init_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    mu = (n - 1) / 2.0
    sigma = rng.uniform(0.05, 0.30) * n
    x = np.arange(n)
    return np.exp(-0.5 * ((x - mu) / sigma) ** 2)


def _init_half_sine(n: int, rng: np.random.Generator) -> np.ndarray:
    x = np.arange(n)
    raw = np.sin(np.pi * x / (n - 1))
    raw = raw * rng.uniform(0.5, 2.0) + rng.uniform(0.0, 0.2)
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.  Occasionally warm‑start
    from the global best, otherwise draw from a random parametric family.
    """
    S = math.sqrt(2.0 * n)

    # Warm start occasionally
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    style = rng.choice(
        ['dirichlet', 'gamma', 'uniform',
         'left_powerlaw', 'left_exponential',
         'sparse', 'gaussian', 'half_sine']
    )
    if style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'left_powerlaw':
        raw = _init_left_powerlaw(n, rng)
    elif style == 'left_exponential':
        raw = _init_left_exponential(n, rng)
    elif style == 'sparse':
        raw = _init_sparse_bumps(n, rng)
    elif style == 'gaussian':
        raw = _init_gaussian(n, rng)
    else:  # half_sine
        raw = _init_half_sine(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    return raw * (S / total)

# Simulated‑annealing / greedy hill‑climbing fine‑tuner
# ----------------------------------------------------------------------


def _anneal(seq: np.ndarray,
            deadline: float,
            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Mass‑transfer walk on the simplex with exponential acceptance."""
    n = len(seq)
    S = math.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score
    temperature = 0.15 * cur_score  # initial temperature

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] == 0.0:
            continue
        delta = rng.uniform(0.0, 0.3) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())
        new_score = _max_convolution(new)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

        temperature *= 0.99995
        if temperature < 1e-6:
            temperature = 1e-6

    return best, best_score


def _hill_climb(seq: np.ndarray,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Very cheap greedy hill‑climber (strictly downhill moves only)."""
    n = len(seq)
    S = math.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] < 1e-12:
            continue
        delta = rng.uniform(0.0, 0.15) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())
        new_score = _max_convolution(new)
        if new_score < cur_score:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score
    return best, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (solve for optimal g given f)
# ----------------------------------------------------------------------


def _lp_refine(seq: np.ndarray) -> tuple[np.ndarray | None, float | None]:
    """
    Solve the LP:
        minimise M
        s.t.   (seq * g)_k ≤ M   for all k,
               g ≥ 0,
               Σ g = Σ seq.
    Returns the refined vector g and the optimal M, or (None, None) on failure.
    """
    n = len(seq)
    S = math.sqrt(2.0 * n)

    # Build sparse matrix A where A[k, j] = seq[k - j] if 0 ≤ k-j < n else 0.
    # Each column j corresponds to a diagonal with offset -j.
    diag_data = [np.full(n, seq[i]) for i in range(n)]
    offsets = [-i for i in range(n)]
    A = diags(diag_data, offsets, shape=(2 * n - 1, n), format='csc')

    # Inequalities: A g - M ≤ 0
    ones_col = np.ones((2 * n - 1, 1))
    G = hstack([A, -ones_col])
    h = np.zeros(2 * n - 1)

    # Equality: Σ g = S
    A_eq = np.r_[np.ones(n), 0.0].reshape(1, -1)
    b_eq = np.array([S])

    # Variable bounds: g_i ≥ 0,   M ≥ 0
    bounds = [(0.0, None)] * n + [(0.0, None)]

    # Objective: minimise M
    c = np.r_[np.zeros(n), 1.0]

    try:
        res = linprog(c, A_ub=G, b_ub=h,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': 30})
    except Exception as exc:        # pragma: no cover
        print("[lp] exception during linprog:", exc)
        return None, None

    if not res.success:            # pragma: no cover
        print("[lp] linprog failed:", res.message)
        return None, None

    x = res.x
    g = x[:n]
    M = x[-1]

    # Numerical clean‑up
    g = np.maximum(g, 0.0)
    total = g.sum()
    if total > 0:
        g = g * (S / total)
    else:
        g = np.full(n, S / n)

    return g, M

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The algorithm follows the three‑stage scheme described in the module
    docstring and stores the best construction in ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    safety_margin = 5.0
    end_time = time.time() + budget_s - safety_margin
    if end_time <= time.time():      # pragma: no cover
        raise RuntimeError("Budget too small after safety margin.")

    # --------------------------------------------------------------
    # Warm‑start from a previously stored construction, if any
    # --------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # --------------------------------------------------------------
    # Phase 1 – fast random exploration (≈15 % of budget)
    # --------------------------------------------------------------
    explore_deadline = time.time() + 0.15 * budget_s
    top_k = 10
    top_candidates = []          # list of (score, seq)
    best_seq = None
    best_score = np.inf
    attempt = 0

    while time.time() < explore_deadline:
        attempt += 1
        # length distribution – a mix of moderate and larger sizes
        if rng.random() < 0.75:
            n = int(rng.integers(2500, 7000))
        else:
            n = int(rng.integers(7000, 15000))

        a0 = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a0)

        if peak < best_score:
            best_score = peak
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempt:5d} n={n:5d}  new best={best_score:.6f}")

        # maintain top‑k list
        if len(top_candidates) < top_k:
            top_candidates.append((peak, a0.copy()))
            top_candidates.sort(key=lambda x: x[0])
        elif peak < top_candidates[-1][0]:
            top_candidates[-1] = (peak, a0.copy())
            top_candidates.sort(key=lambda x: x[0])

    # --------------------------------------------------------------
    # Safety net – ensure at least one candidate exists
    # --------------------------------------------------------------
    if best_seq is None:
        n = 4000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n)
        best_score = _max_convolution(best_seq)
        best_n = n
        top_candidates = [(best_score, best_seq.copy())]

    # --------------------------------------------------------------
    # Phase 2 – multi‑scale Adam refinement (≈70 % of budget)
    # --------------------------------------------------------------
    # Choose a generous upper bound for the length (≤ 8·10⁴)
    max_len = min(80000, best_n * 12)
    lengths = [best_n]
    cur_len = best_n
    while cur_len * 1.5 < max_len:
        cur_len = int(cur_len * 1.5)
        lengths.append(cur_len)
    lengths.append(max_len)
    lengths = sorted(set(lengths))

    for init_idx, (init_score, init_seq) in enumerate(top_candidates):
        cur_seq = init_seq.copy()
        cur_len = len(cur_seq)

        for target_n in lengths:
            if time.time() >= end_time - 30.0:
                break

            # interpolate to the new length and re‑normalise
            a_scaled = _scale_sequence(cur_seq, target_n)
            a_scaled = np.clip(a_scaled, 0.0, None)
            a_scaled = a_scaled * (math.sqrt(2.0 * target_n) / a_scaled.sum())

            # allocate a slice of the remaining time (capped at 150 s)
            remaining = end_time - time.time()
            slice_budget = min(remaining * 0.45, 150.0)

            # randomise a few hyper‑parameters for diversity
            lr0 = rng.choice([0.42, 0.48, 0.55, 0.60])
            tau_decay = rng.choice([0.9992, 0.9994, 0.9996, 0.9998])
            refined, peak = _adam_optimize(a_scaled,
                                           time.time() + slice_budget,
                                           rng,
                                           lr0=lr0,
                                           tau_decay=tau_decay,
                                           lr_decay=0.99985,
                                           jitter_every=400)

            if peak < best_score:
                best_score = peak
                best_seq = refined.copy()
                best_n = target_n
                print(f"[refine] init#{init_idx+1} len={target_n:5d}  improved score={best_score:.6f}")

            cur_seq = refined
            cur_len = target_n

        if time.time() >= end_time - 30.0:
            break

    # --------------------------------------------------------------
    # Phase 3 – Linear‑programming fine‑tuning (≈10 % of budget)
    # --------------------------------------------------------------
    if time.time() < end_time - 20.0:
        # Possibly down‑scale before solving the LP to keep the problem
        # size manageable.
        lp_target_n = best_n
        seq_for_lp = best_seq
        if best_n > 5000:
            lp_target_n = 5000
            seq_for_lp = _scale_sequence(best_seq, lp_target_n)

        print("[lp] solving linear program...")
        g, M = _lp_refine(seq_for_lp)

        if g is not None:
            # Upscale back to the original length if we had down‑scaled.
            if lp_target_n != best_n:
                g_up = _scale_sequence(g, best_n)
                g_up = g_up * (math.sqrt(2.0 * best_n) / g_up.sum())
            else:
                g_up = g

            new_score = _max_convolution(g_up)
            print(f"[lp] LP optimum M={M:.6f}, post‑scale max={new_score:.6f}")

            if new_score < best_score:
                best_score = new_score
                best_seq = g_up.copy()
                print("[lp] accepted LP‑improved sequence")

                # a quick Adam run to polish the LP result
                remaining = end_time - time.time()
                if remaining > 5.0:
                    refined, peak = _adam_optimize(best_seq,
                                                   time.time() + min(remaining, 30.0),
                                                   rng,
                                                   lr0=0.45,
                                                   tau_decay=0.9995,
                                                   lr_decay=0.99985,
                                                   jitter_every=800)
                    if peak < best_score:
                        best_score = peak
                        best_seq = refined.copy()
                        print(f"[lp‑adam] further improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – cheap stochastic fine‑tuning (remaining budget)
    # --------------------------------------------------------------
    remaining = end_time - time.time()
    if remaining > 15.0:
        # simulated annealing first
        best_seq, best_score = _anneal(best_seq,
                                       time.time() + 0.6 * remaining,
                                       rng)
        remaining = end_time - time.time()
        if remaining > 5.0:
            best_seq, best_score = _hill_climb(best_seq,
                                               time.time() + remaining,
                                               rng)

    # --------------------------------------------------------------
    # Final sanity check via the supplied black‑box evaluator
    # --------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  internal max = {best_score:.6f}"
          f"  length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
