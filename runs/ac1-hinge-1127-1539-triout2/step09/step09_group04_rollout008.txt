"""
Improved step‑function construction.

The goal is to minimise  

    score = 2·n·max(conv(a,a)) / (sum(a)²)

where `conv(a,a)` is the linear convolution of `a` with itself.
If we enforce Σa = √(2·n) then the score equals max(conv(a,a)).
The optimiser works in several phases:

1.  A cheap two‑parameter family (power‑law × exponential) is searched
    on a reduced size and up‑scaled to the target length.
2.  Mirror‑descent with exponential updates (soft‑max surrogate of the
    max‑convolution) quickly reaches a good region.
3.  Adam‑type adaptive gradient descent on the same surrogate further
    refines the shape.
4.  Projected gradient descent with random restarts provides extra
    escape from shallow local minima.
5.  A deterministic mass‑transfer step moves a little mass from the
    most harmful index to a less harmful one.
6.  A simple simulated‑annealing local search performs occasional
    pairwise mass moves with a temperature schedule.
7.  The best sequence found is trimmed, re‑normalised and returned.
All updates keep the ℓ¹‑norm exactly (projection onto the simplex) so
the final sequence can be fed directly to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact score = 2·n·max(conv)/sum(a)²."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Monotone decreasing shape a_i = (i+1)^{-p}, normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max surrogate utilities
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the smooth approximation
        f(a) = (1/β)·log Σ_k exp(β·conv_k(a,a))

    Returns (grad, conv) where conv is the full convolution a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_c = np.max(conv)
    exps = np.exp(beta * (conv - max_c))
    probs = exps / np.sum(exps)          # soft‑max probabilities

    # grad = 2 * reverse(a) convolve probs
    a_rev = a[::-1]
    ar_fft = np.fft.rfft(a_rev, n=size)
    pr_fft = np.fft.rfft(probs, n=size)
    conv_rev = np.fft.irfft(ar_fft * pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]  # keep n entries
    return grad, conv

def _grid_param_search(small_n: int,
                       target_sum: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """Two‑parameter family (power‑law × exponential) on a small size."""
    p_vals = np.linspace(0.2, 1.2, 17)
    lam_vals = np.linspace(0.0, 6.0, 13)

    best_seq = None
    best_score = np.inf
    i = np.arange(small_n, dtype=np.float64)

    for p in p_vals:
        for lam in lam_vals:
            if time.time() > deadline:
                break
            seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
            seq = seq * (target_sum / seq.sum())
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()
                print(f"[{time.time():.2f}] Grid best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")
        if time.time() > deadline:
            break

    rng = np.random.default_rng()
    while time.time() < deadline:
        p = rng.uniform(0.2, 1.2)
        lam = rng.exponential(1.0) * 3.0
        seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
        seq = seq * (target_sum / seq.sum())
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Random best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")

    return best_seq, best_score

def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Mirror‑descent with exponential updates using a soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 16384.0
    base_eta = 0.04
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * grad)
        a = a * (target_sum / a.sum())

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

def _adam_descent(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """Adam optimiser on the soft‑max surrogate, with simplex projection."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 16384.0
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size tuned to the typical magnitude of a
        step = 0.05 * target_sum / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # project back onto the simplex (keeps sum = target_sum)
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} score={best_val:.6f}")

    return best_a, best_val

def _annealed_gradient_descent_v2(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the soft‑max surrogate (momentum version)."""
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 16384.0
    step_base = target_sum * 0.5
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.95
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.002 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GDv2 it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Several short GD restarts with random jitter; keep the best."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_v2(start,
                                               time.time() + run_budget,
                                               target_sum)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val

def _mass_transfer_refine(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Move a small fraction of mass from heavily weighted indices to lighter ones."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    while time.time() < deadline:
        k_max = int(np.argmax(conv))

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k_max - idx]          # a_{k_max-i}
        weight = a[idx] * partner         # contribution a_i·a_{k_max-i}

        donor = idx[np.argmax(weight)]

        low_thr = np.percentile(weight, 10)
        low_idx = idx[weight <= low_thr]
        if low_idx.size:
            receiver = low_idx[np.argmin(weight[weight <= low_thr])]
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            donor = rng.choice(idx)
            receiver = rng.integers(n)
            if donor == receiver:
                continue

        max_frac = 0.04
        delta = a[donor] * rng.random() * max_frac
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta

        conv = _conv_fft(a)
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Transfer improve score={best_val:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            conv = _conv_fft(a)

        # keep the total mass exact (numerical drift)
        cur_sum = a.sum()
        if abs(cur_sum - target_sum) > 1e-9:
            a = a * (target_sum / cur_sum)

    return best_a, best_val

def _simulated_annealing_refine(a0: np.ndarray,
                               target_sum: float,
                               deadline: float) -> tuple[np.ndarray, float]:
    """Pairwise mass moves accepted with a temperature schedule."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.shape[0]
    best_a = a.copy()
    cur_val = best_val = _score(a)

    rng = np.random.default_rng()
    t0 = time.time()
    total_time = deadline - t0
    if total_time <= 0:
        return best_a, best_val

    # initial temperature – a small fraction of current score
    T0 = max(0.01 * best_val, 1e-3)
    Tf = 1e-6
    max_iters = 8000

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        donor = rng.integers(n)
        receiver = rng.integers(n)
        if donor == receiver:
            continue

        max_frac = 0.05
        delta = a[donor] * rng.random() * max_frac
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        a_new = a_new * (target_sum / a_new.sum())

        new_val = _score(a_new)

        elapsed = (time.time() - t0) / total_time
        T = T0 * (1.0 - elapsed) + Tf * elapsed

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / T):
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA improved score={best_val:.6f}")

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5    # leave a small safety margin

    # ------------------------------------------------------------------
    # Choose problem size (larger n gives more freedom, but also heavier ops)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Warm‑start: either a previously saved sequence or a simple power‑law
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"],
                                        dtype=np.float64), n)
        init = init * (target_sum / init.sum())
    else:
        init = _power_law_sequence(n, 0.55, target_sum)

    # ------------------------------------------------------------------
    # Phase 1 – cheap parametric search on a reduced scale
    # ------------------------------------------------------------------
    small_n = max(2000, n // 4)
    small_target = np.sqrt(2.0 * small_n)
    param_deadline = time.time() + min(12.0,
                                      (deadline - time.time()) * 0.07)
    best_small_seq, _ = _grid_param_search(small_n, small_target,
                                           param_deadline)

    if best_small_seq is not None:
        init = _resize_sequence(best_small_seq, n)
        init = init * (target_sum / init.sum())
        print(f"[{time.time():.2f}] Parametric warm‑start ready (n={n})")

    # ------------------------------------------------------------------
    # Phase 2 – exponential (multiplicative) descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        exp_deadline = time.time() + (deadline - time.time()) * 0.12
        a_exp, val_exp = _exponential_descent(init, target_sum, exp_deadline)
        best_seq = a_exp
        best_score = val_exp
        print(f"[{time.time():.2f}] After ExpGD best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam adaptive descent on soft‑max surrogate
    # ------------------------------------------------------------------
    if time.time() < deadline:
        adam_deadline = time.time() + (deadline - time.time()) * 0.12
        a_adam, val_adam = _adam_descent(best_seq, target_sum, adam_deadline)
        if val_adam < best_score:
            best_seq = a_adam
            best_score = val_adam
        print(f"[{time.time():.2f}] After Adam best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – projected gradient descent with restarts
    # ------------------------------------------------------------------
    if time.time() < deadline:
        gd_deadline = time.time() + (deadline - time.time()) * 0.30
        a_gd, val_gd = _gradient_descent_with_restarts(best_seq,
                                                       target_sum,
                                                       gd_deadline)
        if val_gd < best_score:
            best_seq = a_gd
            best_score = val_gd
        print(f"[{time.time():.2f}] After GD restarts best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – deterministic mass‑transfer refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        mt_deadline = time.time() + (deadline - time.time()) * 0.20
        a_mt, val_mt = _mass_transfer_refine(best_seq, target_sum, mt_deadline)
        if val_mt < best_score:
            best_seq = a_mt
            best_score = val_mt
        print(f"[{time.time():.2f}] After mass‑transfer best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – simulated‑annealing local search
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sa_deadline = deadline
        a_sa, val_sa = _simulated_annealing_refine(best_seq,
                                                   target_sum,
                                                   sa_deadline)
        if val_sa < best_score:
            best_seq = a_sa
            best_score = val_sa
        print(f"[{time.time():.2f}] After SA best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final clean‑up, trimming trailing zeros
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Final evaluation (for logging)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # keep a copy for possible warm‑starts in a subsequent call
    globals()["height_sequence_1"] = final_seq

    return final_seq
