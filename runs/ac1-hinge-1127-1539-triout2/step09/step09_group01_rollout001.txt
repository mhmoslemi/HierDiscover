def _subgradient_max_convolution(a):
    n = a.shape[0]
    size = 1 << ((2*n-1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    # compute convolution a*a
    conv = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
    max_val = conv.max()
    # mask active indices, tolerance 1e-12 maybe
    mask = np.isclose(conv, max_val, atol=1e-12, rtol=0)
    p = mask.astype(np.float64)
    p_sum = p.sum()
    if p_sum == 0:
        p = np.zeros_like(conv)
        p[conv.argmax()] = 1.0
        p_sum = 1
    p = p / p_sum
    # compute gradient
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[:2*n-1]
    grad = 2.0 * conv_rev[n-1:n-1+n]
    return grad, conv

def _swap_hill_climb(a, target_sum, deadline, max_attempts=None):
    cur_seq = a.copy()
    cur_score = _max_convolution_fft(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    n = cur_seq.shape[0]
    attempts = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # swap
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        new_score = _max_convolution_fft(cur_seq)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
                print("[time] swap new best", best_score)
        else:
            # revert
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        attempts += 1
        if max_attempts and attempts >= max_attempts:
            break
    return best_seq, best_score

def _subgradient_descent(a0, target_sum, deadline, step_factor=0.2, gamma=0.9):
   a = a0.copy()
   a = _project_to_simplex(a, target_sum)
   best_a = a.copy()
   best_val = _max_convolution_fft(a)
   momentum = np.zeros_like(a)
   rng = np.random.default_rng()
   it = 0
   while time.time() < deadline:
       grad, conv = _subgradient_max_convolution(a)
       momentum = gamma * momentum + grad
       norm = np.linalg.norm(momentum) + 1e-12
       lr = step_factor * target_sum / norm  # adjust scale
       a -= lr * momentum
       a = _project_to_simplex(a, target_sum)
       
       # maybe jitter
       if it % 5000 == 0:
           jitter = rng.normal(scale=target_sum*1e-3, size=a.shape)
           a += jitter
           a = _project_to_simplex(a, target_sum)
       
       cur_val = _max_convolution_fft(a)
       if cur_val < best_val:
           best_val = cur_val
           best_a = a.copy()
           print(f"[{time.time():.2f}] subgrad it={it} score={best_val:.6f}")
       
       it += 1
   return best_a, best_val

"""
Improved step‑function optimizer.

The core idea:
  * The evaluation after normalisation is exactly the maximal entry of
    the self‑convolution, so we directly minimise ``max(a*a)``.
  * We use a *sub‑gradient* method for the max‑convolution (no soft‑max
    smoothing) together with a Euclidean projection onto the ℓ¹‑simplex.
  * The bulk of the time budget (≈70 %) is spent on this first‑order
    optimisation, which is cheap (one FFT per iteration).
  * Afterwards a short deterministic mass‑transfer refinement and a
    permutation‑annealing (swap‑based simulated annealing) are applied.
  * A final high‑β soft‑max gradient polish cleans any remaining
    irregularities.

The algorithm works for any sequence length; a larger size gives more
degrees of freedom but also larger FFTs.  For a 1000 s budget we use
~18000 entries, which fits easily into memory and runs fast enough.

All helper functions are defined at module level – no closures or lambdas.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the ℓ¹‑simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple sequence generators
# ----------------------------------------------------------------------
def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i = (i+1)^(-p) normalised to have Σa_i = target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------
def _subgradient_max_convolution(a: np.ndarray):
    """
    Returns (grad, max_val) where grad is a sub‑gradient of
    f(a) = max_k (a*a)_k.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    # full convolution a*a
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_val = float(conv.max())
    # active set (allow a tiny tolerance)
    mask = np.isclose(conv, max_val, atol=1e-12, rtol=0.0)
    if not mask.any():
        mask = np.zeros_like(conv, dtype=bool)
        mask[conv.argmax()] = True

    p = mask.astype(np.float64)
    p /= p.sum()                      # uniform over the active indices

    # grad = 2 * (a_rev * p)   (restricted to length n)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]

    return grad, max_val


# ----------------------------------------------------------------------
# Sub‑gradient descent (majority of the time budget)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.9) -> tuple[np.ndarray, float]:
    """Projected (momentum) sub‑gradient descent on max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        grad, cur_max = _subgradient_max_convolution(a)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum) + 1e-12
        # learning rate decays like 1/√it
        lr = step_factor * target_sum / (norm * (np.sqrt(it + 1) + 1.0))
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional tiny jitter to escape flat plateaus
        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 1e-3, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] subgrad it={it:6d} max={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (as in the previous run)
# ----------------------------------------------------------------------
def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy transfer of mass from a “donor’’ (large contribution to the
    maximal convolution entry) to a “receiver’’ (small contribution).
    Moves are accepted with a temperature that decays exponentially.
    """
    a = a0.copy()
    n = a.shape[0]

    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        # temperature schedule
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]
        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:6d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter
        if it % 3000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 1e-3, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple re‑ordering heuristic (alternating big‑small)
# ----------------------------------------------------------------------
def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place largest, then smallest, then second‑largest, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    left_turn = True
    for v in sorted_vals:
        if left_turn:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        left_turn = not left_turn
    return out


# ----------------------------------------------------------------------
# Permutation annealing (swap‑based simulated annealing)
# ----------------------------------------------------------------------
def _permutation_annealing(a: np.ndarray,
                           target_sum: float,
                           deadline: float,
                           init_score: float | None = None) -> tuple[np.ndarray, float]:
    """
    Simulated annealing over swaps of two positions; ordering only.
    """
    cur_seq = a.copy()
    cur_score = init_score if init_score is not None else _max_convolution_fft(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score

    start_time = time.time()
    n = cur_seq.shape[0]
    rng = np.random.default_rng()
    start_temp = 0.2
    end_temp = 1e-5

    while time.time() < deadline:
        elapsed = time.time() - start_time
        ratio = elapsed / (deadline - start_time) if deadline > start_time else 0.0
        temp = start_temp * (end_temp / start_temp) ** ratio

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # swap
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        new_score = _max_convolution_fft(cur_seq)
        delta = new_score - cur_score

        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / max(temp, 1e-12)):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
                print(f"[{time.time():.2f}] Perm anneal new best = {best_score:.6f}")
        else:
            # revert
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        # occasional reshuffle near the end to kick‑start
        if ratio > 0.9 and rng.random() < 0.01:
            rng.shuffle(cur_seq)
            cur_score = _max_convolution_fft(cur_seq)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Final polishing with high‑β soft‑max gradient (as in the reference)
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 3e5,
                               step_factor: float = 0.6,
                               gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """
    Momentum gradient descent on the soft‑max approximation.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum) + 1e-12
        lr = step_base / (norm * (beta + 1.0))
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] polisher it={it:5d} β={beta:6.1f} max={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal maximal self‑convolution.
    The routine combines:
        1. sub‑gradient descent (majority of time),
        2. deterministic mass‑transfer,
        3. ordering heuristics,
        4. permutation annealing,
        5. high‑β soft‑max polishing.
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # small safety margin

    # --------------------------------------------------------------
    # Choose a problem size (bigger = more freedom)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 400:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)

    # Build an initial sequence (reuse previous run if possible)
    # --------------------------------------------------------------
    rng = np.random.default_rng()

    # --------------------------------------------------------------

    subgrad_deadline = time.time() + (deadline - time.time()) * 0.70

    # --------------------------------------------------------------
