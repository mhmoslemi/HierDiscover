def solve_linear_program(f, M):

   n = len(f)
   # Build CSR matrix A
   # Use scipy.sparse.lil_matrix for convenient assignment, then convert to csr.
   rows = []
   cols = []
   data = []
   # Loop over i from 0 to 2n-2
   for i in range(2*n-1):
       j_start = max(0, i-(n-1))
       j_end = min(n-1, i)
       # iterate j range
       for j in range(j_start, j_end+1):
           rows.append(i)
           cols.append(j)
           data.append(f[j])
   A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

   c = -np.ones(n)   # maximize sum => minimize -sum
   bounds = [(0, None) for _ in range(n)]

   # b_ub = M for all rows
   b_ub = np.full(2*n-1, M)

   res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
   if not res.success:
       return None
   return res.x

Now define LP solver:

def solve_lp(f, M):
    n = f.shape[0]
    rows = np.arange(n)[:, None] + np.arange(n)[None, :]  # shape (n,n)
    rows = rows.ravel()
    cols = np.repeat(np.arange(n), n)
    data = np.repeat(f, n)
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    c = -np.ones(n)  # maximize sum(g)
    bounds = [(0, None)] * n
    b_ub = np.full(2*n-1, M)
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        return None

"""
Improved search for a step‑function that minimises
    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

The algorithm proceeds in three stages:

1. **Random restarts** – generate a few positive sequences on the ℓ¹‑simplex.
2. **Linear‑programming refinement** – for a fixed sequence f,
   solve   max Σ g_j   s.t.  f⋆g ≤ M  (entry‑wise),
   where M = max(f⋆f).  The solution g is then mixed with f.
   Repeating this step quickly drives the maximal convolution down.
3. **Exact deterministic hill‑climb** – after the LP phase a cheap O(n)
   greedy move reduces the peak further.

The whole routine respects the wall‑clock budget (default 1000 s) and
returns the best sequence it found.
"""

import time
import math
import numpy as np
from scipy import sparse
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Return max entry of a⋆a using an FFT (O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a⋆a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Random initial sequence on the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Draw a positive vector on the simplex Σa = √(2n)."""
    target = math.sqrt(2.0 * n)

    # a Dirichlet draw (guarantees positivity) + tiny jitter
    a = rng.dirichlet(alpha=np.ones(n)) * target
    a += rng.normal(scale=target * 1e-4, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _ensure_strictly_positive(a: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """Make sure every entry is ≥ eps and re‑project onto the simplex."""
    a = np.where(a < eps, eps, a)
    target = a.sum()
    # after forcing eps‑positivity the sum may have increased a little;
    # re‑project to keep the original total.
    a = _project_to_simplex(a, target)
    return a


# ----------------------------------------------------------------------
# Linear program: maximise Σg subject to f⋆g ≤ M
# ----------------------------------------------------------------------


def _solve_lp(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max Σ g_j   s.t.   (f⋆g)_k ≤ M  for all k,
    with g_j ≥ 0.  Returns the optimal g (may be dense) or None.
    """
    n = f.shape[0]

    # Build the (2n‑1)×n convolution matrix A (row i, column j has coeff f_j
    # if 0 ≤ i‑j < n, otherwise 0).  This can be assembled with a simple
    # outer‑sum of indices.
    idx = np.arange(n, dtype=np.int64)
    rows = (idx[:, None] + idx[None, :]).ravel()          # shape (n²,)
    cols = np.repeat(idx, n)                               # shape (n²,)
    data = np.repeat(f, n)                                 # shape (n²,)

    A = sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    # Objective – maximize Σg ⇒ minimise -Σg.
    c = -np.ones(n, dtype=np.float64)

    # Upper‑bound RHS.
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Bounds for each variable.
    bounds = [(0, None)] * n

    # Use HiGHS (fast and accepts sparse matrices).
    res = linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
# Deterministic hill‑climbing (exact max)
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray,
                              b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that moves mass from the entry that currently attains
    the maximal convolution value to a position that reduces that entry.
    Returns the best vector seen and its maximal convolution entry.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()
    rng = np.random.default_rng()

    while time.time() < deadline:
        M_idx = int(np.argmax(b))

        i_low = max(0, M_idx - (n - 1))
        i_high = min(n - 1, M_idx)
        idx = np.arange(i_low, i_high + 1, dtype=int)

        contrib = a[idx] * a[M_idx - idx]
        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        # choose a destination with the smallest counterpart contribution
        counterpart = np.zeros(n, dtype=float)
        counterpart[i_low:i_high + 1] = a[M_idx - np.arange(i_low, i_high + 1)]

        cand_order = np.argsort(counterpart[i_low:i_high + 1]) + i_low
        for j in cand_order:
            if j != i:
                break
        else:
            break

        delta = 0.8 * a[i]
        if delta <= 0.0:
            break

        # linear update of the full convolution (O(n))
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()
        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # try a smaller move (binary search)
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                # tiny jitter to escape possible plateaus
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                a = _project_to_simplex(a, math.sqrt(2.0 * n))
                b = _full_convolution(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  The routine obeys the wall‑clock budget
    (default 1000 s) and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # a small safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 1) Choose a length.  2000 works well with the LP approach (memory ≈ 100 MiB).
    #    If the sandbox provides a previous solution, reuse its length.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            n = prev.size
            target = math.sqrt(2.0 * n)
            f0 = _project_to_simplex(prev, target)
        except Exception:
            n = 2000
            f0 = None
    else:
        n = 2000
        f0 = None

    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # 2) Perform a few random restarts and improve each with LP + hill‑climb.
    # ------------------------------------------------------------------
    # allocate ~70 % of the time budget to the LP phase, remaining for polishing
    lp_budget = (deadline - time.time()) * 0.70
    remaining_budget = deadline - time.time() - lp_budget

    # number of restarts – each gets an equal slice of the LP budget
    n_restarts = max(1, int(lp_budget // 30.0))   # aim for ~30 s per restart
    per_restart_budget = lp_budget / n_restarts

    for restart in range(n_restarts):
        if time.time() >= deadline:
            break

        # ----- initialise -------------------------------------------------
        if restart == 0 and f0 is not None:
            a = f0.copy()
        else:
            a = _random_initial_sequence(n, rng)

        a = _ensure_strictly_positive(a)   # avoid zero columns in the LP

        # ----- LP‑driven refinement --------------------------------------
        stop_time = min(deadline - 0.5, time.time() + per_restart_budget)
        t_mix = 0.5               # start with a moderate mixing factor
        best_local = a.copy()
        best_local_score = _max_convolution_fft(a)

        while time.time() < stop_time:
            M = _max_convolution_fft(a)

            # solve the linear programme
            g0 = _solve_lp(a, M)
            if g0 is None:
                break
            # scale g0 to the required ℓ¹‑norm
            sum_g0 = g0.sum()
            if sum_g0 <= 0:
                break
            g = g0 * (target_sum / sum_g0)

            # convex combination
            a_new = (1.0 - t_mix) * a + t_mix * g
            a_new = _project_to_simplex(a_new, target_sum)

            M_new = _max_convolution_fft(a_new)

            if M_new < M - 1e-12:
                a = a_new
                if M_new < best_local_score - 1e-12:
                    best_local_score = M_new
                    best_local = a.copy()
                # gently increase the mixing factor (helps speed)
                t_mix = min(t_mix * 1.05, 0.9)
            else:
                # if we did not improve, shrink the step size
                t_mix *= 0.5
                if t_mix < 1e-3:
                    break   # nothing more can be done

        # ----- deterministic hill‑climb (exact) ------------------------
        if time.time() < deadline - 2.0:
            b = _full_convolution(best_local)
            hc_deadline = min(deadline - 0.5,
                               time.time() + remaining_budget * 0.30)
            a_hc, score_hc = _hill_climb_deterministic(best_local, b,
                                                       hc_deadline)
            if score_hc < best_local_score - 1e-12:
                best_local = a_hc
                best_local_score = score_hc

        # ----- keep global best -----------------------------------------
        if best_local_score < best_score - 1e-12:
            best_seq = best_local.copy()
            best_score = best_local_score
            print(f"[{time.time():.1f}] Restart {restart + 1}/{n_restarts} → "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Final polishing (remaining wall‑clock time)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        # a short extra hill‑climb if time permits
        b = _full_convolution(best_seq)
        final_deadline = deadline - 0.5
        best_seq, final_score = _hill_climb_deterministic(best_seq,
                                                          b,
                                                          final_deadline)
        if final_score < best_score - 1e-12:
            best_score = final_score
            print(f"[{time.time():.1f}] Final hill‑climb improvement → "
                  f"{best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Trim trailing near‑zeros and report
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        best_seq = np.full(n, target_sum / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence => {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:      # pragma: no cover
        print(f"Error while calling evaluate_sequence: {exc}")

    return best_seq.tolist()
