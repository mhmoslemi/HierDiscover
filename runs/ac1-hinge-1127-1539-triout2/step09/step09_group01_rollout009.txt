"""
Multi‑phase optimizer for the step‑function benchmark.

The algorithm tries to minimise
    score = 2·n·max(conv(a,a)) / (sum(a))²

under the constraints a_i ≥ 0 and sum(a)=√(2 n).
Because the denominator is fixed, the objective reduces to
the peak of the self‑convolution.

Phases (all time‑limited):
1️⃣  Warm‑up by projected gradient descent on a smooth soft‑max
    approximation of the peak (FFT‑based gradients, momentum,
    increasing temperature β).
2️⃣  Fine‑grained deterministic mass‑transfer that directly equalises
    the contributions a_i·a_{k‑i} at the current maximal lag k.
    A Metropolis acceptance with an exponential cooling schedule
    prevents getting stuck.
3️⃣  Simple re‑ordering heuristics (alternating large/small pattern).
4️⃣  Simulated‑annealing over swaps of two positions (permutation
    annealing) to exploit the combinatorial freedom of the ordering.
5️⃣  Final polishing with a high‑β gradient descent.

The whole search respects the 1000 s budget and returns the best
sequence found.  The intermediate best is stored in the global
variable ``height_sequence_1`` for possible reuse in later runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Scoring (the benchmark objective)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Return the benchmark score (max of the self‑convolution)."""
    return float(_max_convolution_fft(a))


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution (used in GD phases)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where conv is the full linear convolution.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(conv)
    exps = np.exp(beta * (conv - max_b))
    p = exps / np.sum(exps)

    # gradient: 2 * (reversed a) ⋆ p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, conv


# ----------------------------------------------------------------------
# Phase A – Projected gradient descent (momentum, annealed β)
# ----------------------------------------------------------------------


def _annealed_gradient_descent_ext(a0: np.ndarray,
                                   deadline: float,
                                   target_sum: float,
                                   max_beta: float = 1e5,
                                   step_factor: float = 0.7,
                                   gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max approximation.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD_ext it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Several short gradient‑descent runs with random jitter.
    Keeps the best result found before the given deadline.
    """
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_ext(
            start,
            time.time() + run_budget,
            target_sum,
            max_beta=1e5,
            step_factor=0.8,
            gamma=0.99,
        )
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Deterministic product‑based mass‑transfer refinement
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_product(a0: np.ndarray,
                                        deadline: float,
                                        target_sum: float,
                                        start_temp: float = 0.5,
                                        end_temp: float = 1e-4) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer moves that directly target the contributions
    a[i]·a[kmax‑i] at the current maximal convolution entry.
    Moves are accepted with a Metropolis rule whose temperature decays
    exponentially.
    """
    a = a0.copy()
    n = a.shape[0]

    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()

    while time.time() < deadline:
        # temperature schedule (linear in log‑time)
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # compute full convolution and the index of the maximal entry
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # valid index range where both i and kmax-i lie inside [0,n)
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # counterpart values b_i = a[kmax-i]
        b = a[kmax - idx]

        # contribution of each i to the peak
        contrib = a[idx] * b

        # donor = index with largest contribution, receiver with smallest
        donor = idx[np.argmax(contrib)]
        receiver = idx[np.argmin(contrib)]

        if donor == receiver:
            # fallback to a random pair inside the valid window
            if idx.size < 2:
                break
            donor, receiver = rng.choice(idx, size=2, replace=False)

        # compute a step that equalises the two contributions (if possible)
        a_d, a_r = a[donor], a[receiver]
        b_d, b_r = a[kmax - donor], a[kmax - receiver]

        denom = b_d + b_r
        if denom <= 0.0:
            # no effect – perform a tiny random move instead
            delta = a_d * 0.01 * rng.random()
        else:
            delta_opt = (a_d * b_d - a_r * b_r) / denom
            # keep delta non‑negative and bounded
            delta = max(0.0, min(delta_opt, a_d * 0.5))

        if delta < 1e-12:
            # if the optimal step is negligible, try a small random transfer
            delta = a_d * 0.01 * rng.random()
            if delta < 1e-12:
                continue

        # apply move
        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        new_score = _score(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MassTransferProd it={int(elapsed):5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter to avoid dead‑ends
        if int(elapsed) % 2000 == 0 and elapsed > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            cur_score = _score(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Re‑ordering utilities
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """
    Reorder a sequence by placing the largest value leftmost,
    the second largest rightmost, the third next to the left, etc.
    """
    sorted_vals = np.sort(seq)[::-1]          # descending
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


# ----------------------------------------------------------------------
# Phase D – Permutation annealing (swap‑based simulated annealing)
# ----------------------------------------------------------------------


def _permutation_annealing(a: np.ndarray,
                           target_sum: float,
                           deadline: float,
                           init_score: float | None = None) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing over swaps of two indices.
    The sum is invariant, only the ordering changes.
    """
    cur_seq = a.copy()
    cur_score = init_score if init_score is not None else _max_convolution_fft(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score

    start_time = time.time()
    n = cur_seq.shape[0]
    rng = np.random.default_rng()

    start_temp = 0.2
    end_temp = 1e-5

    while time.time() < deadline:
        elapsed = time.time() - start_time
        ratio = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** ratio

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # swap
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        new_score = _max_convolution_fft(cur_seq)
        delta = new_score - cur_score

        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / max(temp, 1e-12)):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
                print(f"[{time.time():.2f}] Perm anneal new best={best_score:.6f}")
        else:
            # revert swap
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        # occasional full reshuffle near the end
        if ratio > 0.9 and rng.random() < 0.01:
            rng.shuffle(cur_seq)
            cur_score = _max_convolution_fft(cur_seq)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The algorithm runs several optimisation phases inside the time budget.
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # Choose problem size (more points give a richer search space)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 15000
    elif budget_s >= 400:
        n = 10000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Build a diverse pool of initial candidates
    # --------------------------------------------------------------
    rng = np.random.default_rng()
    init_seqs = []

    # power‑law families
    for p in (0.45, 0.5, 0.55, 0.6):
        init_seqs.append(_power_law_sequence(n, p, target_sum))

    # if a previous best exists, reuse it (resized)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        init_seqs.append(_resize_sequence(prev, n))

    # jittered copies of the above
    for seq in list(init_seqs):
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        init_seqs.append(seq_j)

    # a couple of random permutations of a power‑law shape
    for p in (0.5, 0.55):
        seq = _power_law_sequence(n, p, target_sum)
        rng.shuffle(seq)
        init_seqs.append(seq)

    # --------------------------------------------------------------
    # Phase A – Projected gradient descent (multiple restarts)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    gd_deadline = time.time() + remaining * 0.30

    best_seq = None
    best_score = np.inf

    for init in init_seqs:
        if time.time() >= gd_deadline:
            break
        cand, val = _gradient_descent_with_restarts(init, target_sum, gd_deadline)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] Global best after GD phase = {best_score:.6f}")

    if best_seq is None:  # fallback
        best_seq = _power_law_sequence(n, 0.55, target_sum)
        best_score = _score(best_seq)

    # --------------------------------------------------------------
    # Phase B – Deterministic product‑based mass transfer
    # --------------------------------------------------------------
    if time.time() < deadline:
        mt_deadline = time.time() + (deadline - time.time()) * 0.35
        a_mt, val_mt = _deterministic_mass_transfer_product(
            best_seq, mt_deadline, target_sum,
            start_temp=0.6, end_temp=1e-4)
        if val_mt < best_score:
            best_score = val_mt
            best_seq = a_mt.copy()
            print(f"[{time.time():.2f}] After product mass‑transfer best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase C – Simple re‑ordering heuristics
    # --------------------------------------------------------------
    reorder_candidates = [best_seq, _alternating_order(best_seq)]
    for _ in range(2):
        tmp = best_seq.copy()
        rng.shuffle(tmp)
        reorder_candidates.append(tmp)

    for cand in reorder_candidates:
        cand_score = _max_convolution_fft(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] Ordering improvement -> score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase D – Permutation annealing
    # --------------------------------------------------------------
    if time.time() < deadline:
        perm_deadline = time.time() + (deadline - time.time()) * 0.20
        a_perm, val_perm = _permutation_annealing(
            best_seq, target_sum, perm_deadline, init_score=best_score)
        if val_perm < best_score:
            best_score = val_perm
            best_seq = a_perm.copy()
            print(f"[{time.time():.2f}] After permutation annealing best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase E – Final polishing (high‑β gradient descent)
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _annealed_gradient_descent_ext(
            best_seq,
            deadline,
            target_sum,
            max_beta=3e5,
            step_factor=0.6,
            gamma=0.995,
        )
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polishing best={best_score:.6f}")

    # --------------------------------------------------------------
    # Post‑processing: remove trailing zeros and final evaluation
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    globals()["height_sequence_1"] = final_seq
    return final_seq
