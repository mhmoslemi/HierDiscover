def _subgradient_descent(a, deadline):
    n = a.size
    target = np.sum(a)  # already sum
    best_a = a.copy()
    b = _full_convolution(a)
    best_score = b.max()
    t = 0
    while time.time() < deadline:
        M = int(np.argmax(b))
        # compute gradient g as 2*a_rev offset?
        # g[j] = 2 * a[M - j] if 0 <= M-j < n else 0
        g = np.zeros_like(a)
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        # this vector corresponds to a[i] where i in range [i_low, i_high].
        # for j = i, M-j = M-i in range ...
        # So we can set g[i_low:i_high+1] = 2 * a[M - np.arange(i_low,i_high+1)]
        idx = np.arange(i_low, i_high+1)
        g[idx] = 2.0 * a[M - idx]
        # step size schedule: we can adopt decreasing lr = alpha / sqrt(t+1)
        t += 1
        alpha = 0.5 * target / n  # base step order sum*? maybe choose adaptively.
        lr = alpha / np.sqrt(t)
        # Ensure step size not too big: limit max change per component to maybe 0.2 * a[i] + epsilon
        step = lr * g
        # Optionally clip step to not exceed a[i], else may negative after subtract
        # Actually we do a - step. If step > a[i], then a[i] would become negative; after projection that mass redistributed possibly beneficial but we can allow.
        a_new = a - step
        # enforce non-negative by clipping to zero then project to simplex
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_to_simplex(a_new, target)
        # update
        a = a_new
        b = _full_convolution(a)  # O(n log n) via FFT.
        cur_score = b.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
        # Optional early stop if not improving for a while? skip.
    return best_a, best_score

def _subgradient_descent(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    n = a.size
    target = np.sum(a)  # Ideally sqrt(2n)
    best_a = a.copy()
    b = _full_convolution(a)
    best_score = b.max()
    t = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        M = int(np.argmax(b))
        # compute gradient g
        # i from max(0, M-(n-1)) to min(n-1, M)
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1)
        # g[idx] = 2 * a[M - idx]
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[M - idx]
        # step size schedule
        t += 1
        # base step = target / n maybe; ensure not too large.
        base_step = target / n
        lr = base_step / np.sqrt(t)
        # optionally random perturbation for exploration
        # also we might skip step if gradient norm zero
        if np.linalg.norm(g) == 0:
            break
        step = lr * g
        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_to_simplex(a_new, target)
        a = a_new
        b = _full_convolution(a)
        cur_score = b.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

   (8000, 20000, 50000).  For each length we run a short Adam phase,
   a dedicated projected sub‑gradient descent on the exact max‑convolution,
   and a final hill‑climb.

3. **Polishing** – with the remaining budget we run a long Adam optimisation
   and another sub‑gradient pass on the final vector.

All steps respect the ℓ¹‑simplex constraint Σa = √(2 n) via Euclidean projection.
The algorithm never calls ``evaluate_sequence`` except for the final report,
instead it uses fast FFT‑based convolutions to compute the objective
``max(conv(a,a))``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a⋆a using FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a⋆a (returns a length‑2n‑1 array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – used by Adam
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a⋆a.
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution b = a⋆a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float,
                   deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max surrogate.
    Runs until the wall‑clock ``deadline``.
    """
    a = a0.copy()
    n = a.size

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # learning‑rate base (scaled with the simplex radius)
    base_lr = 0.15 * target_sum

    # smoothing schedule
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1

        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if t % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 1e-4 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # increase smoothing
        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 7)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    elif family == 5:                   # sparse spikes
        a = np.zeros(n)
        k = rng.integers(5, 15)                     # number of spikes
        positions = rng.choice(n, size=k, replace=False)
        values = rng.random(k) + 0.1
        a[positions] = values

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Deterministic hill‑climbing (exact max)
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that always attacks the current worst convolution entry.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()

    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))

        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]

        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        # find destination with minimal counterpart
        counterpart = np.zeros(n, dtype=float)
        j_low = i_low
        j_high = i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        order = np.argsort(counterpart[j_range])
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        else:
            break

        delta = a[i] * 0.8
        if delta <= 0.0:
            break

        # update convolution (O(n))
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
            continue

        # try a smaller move (binary search)
        improved = False
        delta2 = delta * 0.5
        while delta2 > 1e-14:
            b_test = b.copy()
            b_test[i:i + n] -= 2.0 * delta2 * a
            b_test[j:j + n] += 2.0 * delta2 * a
            if k2i < b_test.size:
                b_test[k2i] += delta2 * delta2
            if k2j < b_test.size:
                b_test[k2j] += delta2 * delta2
            if kij < b_test.size:
                b_test[kij] -= 2.0 * delta2 * delta2
            if b_test.max() < b.max() - 1e-12:
                a[i] -= delta2
                a[j] += delta2
                b = b_test
                if b.max() < best_score - 1e-12:
                    best_score = b.max()
                    best_a = a.copy()
                improved = True
                break
            delta2 *= 0.5

        if not improved:
            # jitter to escape plateau
            jitter = rng.normal(scale=1e-6, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            target = math.sqrt(2.0 * n)
            a = _project_to_simplex(a, target)
            b = _full_convolution(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient method on   max_k (a⋆a)_k .
    Returns the best vector encountered and its max‑convolution value.
    """
    a = a0.copy()
    n = a.size
    target = np.sum(a)                     # ≈ sqrt(2n)
    best_a = a.copy()
    b = _full_convolution(a)
    best_score = b.max()
    rng = np.random.default_rng()
    t = 0

    # heuristic base step size (a few times target/n)
    base_lr = 5.0 * target / n

    while time.time() < deadline:
        M = int(np.argmax(b))

        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)

        # sub‑gradient g_j = 2·a_{M‑j}  for feasible j, else 0
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[M - idx]

        if np.all(g == 0):
            break

        t += 1
        lr = base_lr / np.sqrt(t)

        step = lr * g
        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_to_simplex(a_new, target)

        a = a_new
        b = _full_convolution(a)
        cur_score = b.max()

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()

        # occasional tiny jitter to avoid stalling
        if t % 200 == 0:
            jitter = rng.normal(scale=target * 1e-5, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target)
            b = _full_convolution(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    The routine respects a wall‑clock ``budget_s`` (default 1000 s) and
    returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 0)  Seed from a previously found solution if present
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            n0 = prev.size
            target0 = math.sqrt(2.0 * n0)
            prev = _project_to_simplex(prev, target0)
            best_seq = prev.copy()
            best_score = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 "
                  f"(n={n0}) score={best_score:.6f}")
        except Exception as e:
            print(f"Seeding error: {e}")

    # ------------------------------------------------------------------
    # 1)  Heavy exploration on a modest size (n≈2000)
    # ------------------------------------------------------------------
    n0 = 2000
    target0 = math.sqrt(2.0 * n0)
    if best_seq is None or best_seq.size != n0:
        a0 = _random_initial_sequence(n0, rng)
    else:
        a0 = best_seq.copy()
    a0 = _project_to_simplex(a0, target0)

    # allocate roughly half of the total budget for this stage
    stage_deadline = min(deadline, time.time() + (deadline - time.time()) * 0.5)

    # ---- Adam restarts -------------------------------------------------
    min_restart_time = 5.0
    remaining_stage = stage_deadline - time.time()
    n_restarts = max(1, int(remaining_stage // min_restart_time))
    per_restart = remaining_stage / n_restarts
    local_best = None
    local_best_score = np.inf

    for r in range(n_restarts):
        if time.time() >= stage_deadline:
            break
        start_a = a0.copy() if r == 0 else _random_initial_sequence(n0, rng)
        a_opt = _adam_optimize(start_a, target0, time.time() + per_restart)
        score = _max_convolution_fft(a_opt)
        if score < local_best_score:
            local_best_score = score
            local_best = a_opt.copy()

    # ---- deterministic hill‑climb ------------------------------------
    if time.time() < stage_deadline - 1.0:
        b0 = _full_convolution(local_best)
        a_hc, score_hc = _hill_climb_deterministic(
            local_best, b0, stage_deadline)
        if score_hc < local_best_score:
            local_best, local_best_score = a_hc, score_hc

    # update global best
    if local_best_score < best_score:
        best_seq = local_best.copy()
        best_score = local_best_score
        print(f"[{time.time():.1f}] Initial stage improvement score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2)  Upscale to larger dimensions
    # ------------------------------------------------------------------
    schedule = [8000, 20000, 50000]
    fractions = [0.25, 0.15, 0.10]   # portions of the *remaining* budget per size

    for n, frac in zip(schedule, fractions):
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        if remaining < 8.0:
            break
        stage_deadline = time.time() + remaining * frac

        target = math.sqrt(2.0 * n)

        # initialise (upsample previous best)
        a_init = _upsample_sequence(best_seq, n)

        # ---- Adam phase -------------------------------------------------
        remaining_stage = stage_deadline - time.time()
        min_restart = 6.0
        n_restarts = max(1, int(remaining_stage // min_restart))
        per_restart = remaining_stage / n_restarts
        local_best_a = None
        local_best_score = np.inf

        for r in range(n_restarts):
            if time.time() >= stage_deadline:
                break
            start_a = a_init.copy() if r == 0 else _random_initial_sequence(n, rng)
            a_opt = _adam_optimize(start_a, target, time.time() + per_restart)
            score = _max_convolution_fft(a_opt)
            if score < local_best_score:
                local_best_score = score
                local_best_a = a_opt.copy()

        # ---- Sub‑gradient descent ---------------------------------------
        if time.time() < stage_deadline - 1.0:
            sub_dead = min(stage_deadline, time.time() + (stage_deadline - time.time()) * 0.4)
            a_sub, score_sub = _subgradient_descent(
                _project_to_simplex(local_best_a, target), sub_dead)
            if score_sub < local_best_score:
                local_best_score = score_sub
                local_best_a = a_sub.copy()

        # ---- Hill‑climb (optional, cheap) -------------------------------
        if time.time() < stage_deadline - 1.0:
            b_cur = _full_convolution(local_best_a)
            a_hc, score_hc = _hill_climb_deterministic(
                local_best_a, b_cur, stage_deadline)
            if score_hc < local_best_score:
                local_best_score = score_hc
                local_best_a = a_hc.copy()

        # update global best
        if local_best_score < best_score:
            best_seq = local_best_a.copy()
            best_score = local_best_score
            print(f"[{time.time():.1f}] Improved at n={n} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3)  Final polishing with remaining budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        n = best_seq.size
        target = math.sqrt(2.0 * n)

        # a long Adam run
        a_adam = _adam_optimize(best_seq, target, deadline - 1.0)
        best_seq = a_adam.copy()
        best_score = _max_convolution_fft(best_seq)
        print(f"[{time.time():.1f}] Final Adam polishing score={best_score:.6f}")

        # final sub‑gradient pass
        if time.time() < deadline - 0.5:
            a_final, score_final = _subgradient_descent(best_seq, deadline - 0.5)
            if score_final < best_score:
                best_seq = a_final.copy()
                best_score = score_final
                print(f"[{time.time():.1f}] Final sub‑gradient polishing score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4)  Remove trailing near‑zeros
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # 5)  Report official evaluation
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
```

This implementation respects the time budget, explores several sequence lengths, mixes stochastic Adam optimisation, deterministic hill‑climbing and a dedicated projected sub‑gradient method on the exact max‑convolution, and returns the best step‑function found.
