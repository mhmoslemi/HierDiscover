    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a, best_score = a.copy(), _score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        # compute gradient of max conv
        b = _conv_fft(a)   # full conv
        kmax = int(np.argmax(b))
        # compute gradient for that offset
        grad = np.zeros_like(a)
        # indices that contribute
        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(a.shape[0] - 1, kmax)
        # i ranges lo..hi inclusive
        # grad[i] = a[kmax - i]
        idx = np.arange(lo, hi+1)
        grad[idx] = a[kmax - idx]
        # adam updates
        it1 = it + 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it1)
        v_hat = v / (1 - beta2 ** it1)
        # step size decays with time
        lr = initial_lr / (1 + 0.01 * it1)  # decaying
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)
        # project to simplex
        a = _project_to_simplex(a, target_sum)
        # evaluate
        val = _score(a)
        if val < best_score:
            best_score = val
            best_a = a.copy()
            print(...)

        # occasional jitter
        if it % 500 == 0:
            jitter = np.random.default_rng().normal(scale=target_sum*0.001, size=a.shape)
            a = _project_to_simplex(np.maximum(a + jitter,0.0), target_sum)
        it += 1
    return best_a, best_score

def _targeted_swap_annealing(a, target_sum, deadline):
    cur = a.copy()
    cur_max = _max_convolution_fft(cur)
    best = cur.copy()
    best_val = cur_max
    start = time.time()
    n = len(cur)
    rng = np.random.default_rng()
    while time.time() < deadline:
        conv = _conv_fft(cur)
        k = int(np.argmax(conv))
        # compute products array w
        lo = max(0, k - (n-1))
        hi = min(n-1, k)
        idx = np.arange(lo, hi+1)
        partner = k - idx
        w = cur[idx] * cur[partner]
        # find i with max w, j with min w
        imax = idx[np.argmax(w)]
        imin = idx[np.argmin(w)]
        if imax == imin:
            # random swap if nothing
            i, j = rng.choice(n, size=2, replace=False)
        else:
            i, j = imax, imin
        # swap
        cur[i], cur[j] = cur[j], cur[i]
        new_max = _max_convolution_fft(cur)
        if new_max < cur_max or rng.random() < 0.01:
            cur_max = new_max
            if new_max < best_val:
                best_val = new_max
                best = cur.copy()
        else:
            # revert
            cur[i], cur[j] = cur[j], cur[i]
    return best, best_val

    a = _project_to_simplex(a, target_sum)
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    best_a = a.copy()
    best_val = _score_vec(a)
    it = 0
    while time.time() < deadline:
        if max_iters is not None and it >= max_iters:
            break
        # gradient of max conv
        # compute conv only to get max index and gradient
        conv = _conv_fft(a)  # compute full conv
        k = int(np.argmax(conv))
        # gradient: a[k - i] for i within overlap
        grad = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad[idx] = a[k - idx]
        it1 = it + 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it1)

        step = lr / (1 + 0.001 * it)  # decay slowly

        a = _project_to_simplex(a, target_sum)

            best_a = a.copy()

            jitter = np.random.default_rng().normal(scale=target_sum * 0.001, size=n)
            a = _project_to_simplex(np.maximum(a + jitter, 0.0), target_sum)

"""
Propose a low‑score step function.

The search works in three complementary phases:

1️⃣  **Exact sub‑gradient descent (Adam)** –  
    The objective is 2 n·max(a∗a)/‖a‖₁².  The gradient of the
    max‑convolution entry is obtained analytically (no soft‑max
    approximation).  An Adam optimiser with Euclidean ℓ¹‑simplex
    projection drives the peak down.  Many restarts and jitter are used.

2️⃣  **Targeted swap annealing** –  
    For the current maximal convolution offset ``k`` the contributions
    ``a[i]·a[k−i]`` are examined.  Swapping the index with the largest
    product with the one having the smallest product is often beneficial.
    A simulated‑annealing schedule accepts occasional worsening moves.

3️⃣  **Final polishing** –  
    A short Adam run with a smaller learning‑rate cleans up any remaining
    irregularities.

All heavy work uses FFT‑based convolutions (O(n log n)).  The algorithm
adapts the problem size to the available time budget (≈25000 entries
for a full 1000 s run).  The best sequence found is stored in the global
``height_sequence_1`` variable for possible warm‑starts in later runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Scoring utilities
# ----------------------------------------------------------------------


def _score_vec(a: np.ndarray, max_conv: float | None = None) -> float:
    """Exact objective 2·n·max(a*a)/(sum a)²."""
    n = a.shape[0]
    if max_conv is None:
        max_conv = _max_convolution_fft(a)
    s = a.sum()
    if s < 1e-8:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i = (i+1)^(-p) normalized to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i = exp(-lam·i) normalized to target_sum."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to length ``new_len``."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (Adam) on the max‑convolution
# ----------------------------------------------------------------------


def _adam_subgradient_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int | None = None,
    lr: float = 1.0,
    beta1: float = 0.9,
    beta2: float = 0.999,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on f(a)=max_k (a*a)[k]  (projected onto the ℓ¹‑simplex).
    Returns the best vector encountered and its objective value.
    """
    a = _project_to_simplex(a0.astype(np.float64), target_sum)
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    best_a = a.copy()
    best_val = _score_vec(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        if max_iters is not None and it >= max_iters:
            break

        # --- convolution & max offset -------------------------------------------------
        conv = _conv_fft(a)               # full convolution
        k = int(np.argmax(conv))          # offset of current peak
        max_b = float(conv.max())

        # --- exact sub‑gradient of the max entry ------------------------------------
        # gradient_i = a_{k-i} for all i where 0 ≤ k-i < n
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad = np.zeros_like(a)
        grad[idx] = a[k - idx]

        # --- Adam update -------------------------------------------------------------
        it1 = it + 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it1)
        v_hat = v / (1.0 - beta2 ** it1)
        step = lr / (1.0 + 0.001 * it)          # slowly decaying step
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # projection back onto the simplex (keeps sum = target_sum)
        a = _project_to_simplex(a, target_sum)

        # --- evaluation ---------------------------------------------------------------
        val = _score_vec(a, max_conv=max_b)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam GD iter={it:5d} score={val:.6f}")

        # occasional jitter to escape flat plateaus
        if it % 500 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = _project_to_simplex(np.maximum(a + jitter, 0.0), target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Targeted swap annealing (simulated annealing using the max‑offset)
# ----------------------------------------------------------------------


def _targeted_swap_annealing(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Simulated annealing that repeatedly swaps the index participating in the
    largest product a[i]*a[k−i] at the current maximal convolution offset
    with an index participating in the smallest product.
    """
    cur = a0.copy()
    n = cur.shape[0]
    cur_max = _max_convolution_fft(cur)
    best = cur.copy()
    best_val = cur_max
    start_time = time.time()
    rng = np.random.default_rng()

    while time.time() < deadline:
        # recompute full convolution (needed for new k)
        conv = _conv_fft(cur)
        k = int(np.argmax(conv))

        # contributions w_i = a[i] * a[k-i] for all valid i
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k - idx
        w = cur[idx] * cur[partner]

        # choose indices to swap
        imax = idx[np.argmax(w)]
        imin = idx[np.argmin(w)]
        if imax == imin:
            i, j = rng.choice(n, size=2, replace=False)
        else:
            i, j = imax, imin

        # perform swap
        cur[i], cur[j] = cur[j], cur[i]

        new_max = _max_convolution_fft(cur)

        # annealing schedule (exponential decay)
        elapsed = time.time() - start_time
        ratio = elapsed / (deadline - start_time) if deadline > start_time else 0.0
        temp = 0.2 * (1e-5 / 0.2) ** ratio   # from 0.2 to 1e-5

        accept = False
        if new_max < cur_max:
            accept = True
        else:
            if rng.random() < np.exp(-(new_max - cur_max) / max(temp, 1e-12)):
                accept = True

        if accept:
            cur_max = new_max
            if new_max < best_val:
                best_val = new_max
                best = cur.copy()
                print(f"[{time.time():.2f}] Swap anneal new best max={best_val:.6f}")
        else:
            # revert swap
            cur[i], cur[j] = cur[j], cur[i]

        # occasional random reshuffle near the end to avoid stagnation
        if ratio > 0.95 and rng.random() < 0.02:
            rng.shuffle(cur)
            cur_max = _max_convolution_fft(cur)

    return best, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (optional, kept from previous code)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    start_temp: float = 0.5,
    end_temp: float = 1e-4,
    low_frac: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer moves that directly attack the current maximal
    convolution entry.  Accepts moves with a Metropolis rule whose temperature
    decays exponentially.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _score_vec(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()

    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        # temperature schedule
        tau = (time.time() - start_time) / max(1e-12, deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # index of current maximal convolution entry
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # combined contributions w_i = a_i + a_{kmax-i}
        w = a[idx] + a[kmax - idx]
        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score_vec(a)

        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] DMT (accepted worsening) it={it:5d}")
            else:
                # revert
                a[donor] += delta
                a[receiver] -= delta
                a = _project_to_simplex(a, target_sum)
                conv = _conv_fft(a)

        if it % 2000 == 0 and it > 0:
            # global jitter to escape possible plateaus
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score_vec(a)

        it += 1

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The pipeline consists of:
        * diverse initial shapes,
        * many Adam sub‑gradient descents (exact max‑gradient),
        * targeted swap annealing,
        * optional deterministic mass‑transfer,
        * final short Adam polishing,
        * evaluation & storage.
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem size (larger sizes give a better trade‑off, still cheap)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 25000
    elif budget_s >= 500:
        n = 18000
    else:
        n = 12000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Build a rich pool of initial candidates
    # --------------------------------------------------------------
    init_seqs = []

    # power‑law families
    for p in np.linspace(0.40, 0.80, 9):
        init_seqs.append(_power_law_sequence(n, p, target_sum))

    # exponential families
    for lam in np.linspace(4e-4, 5e-3, 5):
        init_seqs.append(_exponential_sequence(n, lam, target_sum))

    # previous best (warm‑start)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        init_seqs.append(_resize_sequence(prev, n))

    # random vectors on the simplex
    rng = np.random.default_rng()
    for _ in range(3):
        rnd = rng.random(n)
        rnd = _project_to_simplex(rnd, target_sum)
        init_seqs.append(rnd)

    # jittered copies of all the above (adds diversity)
    for seq in list(init_seqs):
        jitter = rng.normal(scale=target_sum * 0.02, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        init_seqs.append(seq_j)

    # --------------------------------------------------------------
    # Phase A – Exact sub‑gradient descent (Adam) with restarts
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    for init in init_seqs:
        if time.time() >= deadline:
            break
        a, val = _adam_subgradient_descent(init, target_sum, deadline)
        if val < best_score:
            best_score = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Global best after Adam = {best_score:.6f}")

    # fallback (should never happen)
    if best_seq is None:
        best_seq = _power_law_sequence(n, 0.55, target_sum)
        best_score = _score_vec(best_seq)

    # --------------------------------------------------------------
    # Phase B – Targeted swap annealing (fast combinatorial refinement)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_sw, val_sw = _targeted_swap_annealing(best_seq, target_sum, deadline)
        if val_sw < best_score:
            best_score = val_sw
            best_seq = a_sw.copy()
            print(f"[{time.time():.2f}] After swap anneal best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase C – Optional deterministic mass‑transfer refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_mt, val_mt = _deterministic_mass_transfer(best_seq, deadline, target_sum)
        if val_mt < best_score:
            best_score = val_mt
            best_seq = a_mt.copy()
            print(f"[{time.time():.2f}] After deterministic mass transfer best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase D – Final polishing Adam run (high precision)
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_subgradient_descent(
            best_seq,
            target_sum,
            deadline,
            max_iters=3000,
            lr=0.5,
        )
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polishing best={best_score:.6f}")

    # --------------------------------------------------------------
    # Post‑processing: convert to plain Python list
    # --------------------------------------------------------------
    final_seq = best_seq.tolist()

    # ------------------------------------------------------------------
    # Final evaluation (for logging – not used for the score)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose the sequence for possible later runs (warm‑start)
    globals()["height_sequence_1"] = final_seq

    return final_seq
