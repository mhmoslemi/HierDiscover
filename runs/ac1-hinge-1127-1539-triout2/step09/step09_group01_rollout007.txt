def propose_candidate(budget_s=1000):
    start = time.time()
    deadline = start + budget_s - 0.1  # safety.

    if budget_s >= 800:
        n = 20000
    elif budget_s >= 400:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)

    # Precompute helper vector for projecting using scaling.

    # Build candidates
    init_candidates = []
    # power law
    for p in np.linspace(0.45, 0.7, 6):
        init_candidates.append(_power_law_sequence(n, p, target_sum))
    # exponential
    for lam in np.linspace(0.0005, 0.003, 5):
        seq = np.exp(-lam * np.arange(n))
        seq *= target_sum / seq.sum()
        init_candidates.append(seq)
    # gaussian
    for sigma_frac in [0.1, 0.15, 0.2, 0.25]:
        sigma = sigma_frac * n
        x = np.arange(n) - (n-1)/2
        seq = np.exp(-0.5 * (x / sigma)**2)
        seq *= target_sum / seq.sum()
        init_candidates.append(seq)
    # uniform
    seq = np.full(n, target_sum / n)
    init_candidates.append(seq)

    # include previous best scaled if exists
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        init_candidates.append(_resize_sequence(prev, n))

    # jitter each small per candidate to produce diversity
    rng = np.random.default_rng()
    diversified = []
    for seq in init_candidates:
        diversified.append(seq)
        jitter = rng.normal(scale=target_sum*0.02, size=n)
        seq_jit = np.maximum(seq + jitter, 0)
        seq_jit = seq_jit * target_sum / seq_jit.sum()
        diversified.append(seq_jit)
    init_candidates = diversified

    best_seq = None
    best_score = np.inf

    # Phase: Subgradient Adam descent (fast)
    # We'll allocate ~30% time for this stage.
    phase_deadline = time.time() + (deadline - time.time()) * 0.30
    for cand in init_candidates:

        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))
        # Compute subgradient g

        left = max(0, kmax - (n - 1))
        right = min(n - 1, kmax)
        idx = np.arange(left, right + 1)
        # g[idx] = a[idx] + a[kmax - idx]
        g = np.zeros_like(a)
        g[idx] = a[idx] + a[kmax - idx]
        # For indices where kmax - idx == idx (i.e., idx = kmax/2), we added a_i twice, that's correct.
        # Adam update

"""
Propose a low‑score step‑function.

Algorithm outline
-----------------
1. **Problem size** – choose n≈20 000 when the full 1000 s budget is available.
2. **Initial pool** – a diverse set of shapes (power‑law, exponential,
   Gaussian, uniform and the previous best) all projected onto the ℓ¹‑simplex.
3. **Fast sub‑gradient Adam descent** – minimise the peak of the self‑convolution
   directly (using the sub‑gradient of the max entry).  The iteration uses the
   Adam optimiser and a cheap scaling projection (clip ≥0 then renormalise).
4. **Deterministic mass‑transfer** – greedy moves that transfer a little mass
   from a “donor’’ index (large contribution to the current peak) to a
   “receiver’’ index (small contribution).  Acceptance follows a temperature
   schedule.
5. **Re‑ordering heuristics** – try the alternating large/small ordering and a
   few random shuffles.
6. **Permutation annealing** – simulated‑annealing over swaps of two positions
   to improve the ordering further.
7. **Final polishing** – a high‑β projected‑gradient descent (soft‑max
   approximation) as in the reference implementation.
8. Return the best sequence (trailing zeros are stripped) and store it in the
   global variable ``height_sequence_1`` for future runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Score = 2·n·max(a*a)/(Σa)² (with Σa=√(2n) it equals max(a*a))."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))

# Simple utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """
    Reorder a sequence by alternating large / small values.
    Largest goes leftmost, second largest rightmost, third next to left, …
    """
    sorted_vals = np.sort(seq)[::-1]          # descending
    n = seq.shape[0]
    out = np.empty_like(seq)
    left = 0
    right = n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out

# Phase A – Adam sub‑gradient descent on the max‑convolution
# ----------------------------------------------------------------------


def _adam_subgradient_descent(a0: np.ndarray,
                              target_sum: float,
                              deadline: float,
                              max_iter: int = 5000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the (non‑smooth) objective max(a*a).
    The sub‑gradient at the current peak is   g_i = a_i + a_{kmax-i}
    for indices that participate in the peak.
    After each step we project by clipping to non‑negative and renormalising
    to the ℓ¹‑simplex (exact sum = target_sum).
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s > 0:
        a = a * target_sum / s
    else:
        a = np.full_like(a, target_sum / a.shape[0])

    n = a.shape[0]
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    rng = np.random.default_rng()

    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    while time.time() < deadline and t < max_iter:
        # convolution and index of the current maximum
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # sub‑gradient of the max entry
        left = max(0, kmax - (n - 1))
        right = min(n - 1, kmax)
        idx = np.arange(left, right + 1, dtype=np.int64)

        g = np.zeros_like(a)
        g[idx] = a[idx] + a[kmax - idx]          # a_i + a_{kmax-i}

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate schedule (aggressive early, then decay)
        lr = 0.8 * target_sum / (np.sqrt(t) + 1.0)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # cheap projection: clip & renormalise
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a * target_sum / s
        else:
            a = np.full_like(a, target_sum / n)

        # occasional jitter to escape plateaus
        if t % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0:
                a = a * target_sum / s

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={t:5d} best={best_val:.6f}")

    return best_a, best_val

def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer moves that directly target the current maximal
    convolution entry.  Moves are accepted with a Metropolis rule whose
    temperature decays exponentially with the remaining time.
    """
    a = a0.copy()
    n = a.shape[0]

    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()

    # initial convolution
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        # temperature schedule
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # index of the current maximal convolution entry
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contributions w_i = a_i + a_{kmax-i}
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        # transfer amount (bounded)
        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        # recompute convolution and score
        conv = _conv_fft(a)
        new_score = _score(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter to escape plateaus
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

# ----------------------------------------------------------------------


def _permutation_annealing(a: np.ndarray,
                           target_sum: float,
                           deadline: float,
                           init_score: float | None = None) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing over swaps of two indices.  The sum is invariant,
    therefore only the ordering is altered.  The current temperature decays
    exponentially from `start_temp` to `end_temp`.  The best ordering found
    is returned.
    """
    cur_seq = a.copy()
    cur_score = init_score if init_score is not None else _max_convolution_fft(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score

    start_time = time.time()
    n = cur_seq.shape[0]
    rng = np.random.default_rng()

    # annealing schedule parameters
    start_temp = 0.2
    end_temp = 1e-5

    while time.time() < deadline:
        elapsed = time.time() - start_time
        ratio = elapsed / (deadline - start_time) if deadline > start_time else 0.0
        temp = start_temp * (end_temp / start_temp) ** ratio

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # perform swap
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        new_score = _max_convolution_fft(cur_seq)
        delta = new_score - cur_score

        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / max(temp, 1e-12)):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
                print(f"[{time.time():.2f}] Perm anneal new best = {best_score:.6f}")
        else:
            # revert swap
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        # occasional full reshuffle to avoid stagnation
        if ratio > 0.9 and rng.random() < 0.01:
            rng.shuffle(cur_seq)
            cur_score = _max_convolution_fft(cur_seq)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Phase E – Final polishing (soft‑max projected gradient descent)
# (identical to the reference implementation)
# ----------------------------------------------------------------------


def _annealed_gradient_descent_v2_ext(a0: np.ndarray,
                                     deadline: float,
                                     target_sum: float,
                                     max_beta: float = 1e5,
                                     step_factor: float = 0.7,
                                     gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max approximation of the max.
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        # ------------------------------------------------------------------
        # soft‑max gradient
        # ------------------------------------------------------------------
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_b = np.max(b)
        exps = np.exp(beta * (b - max_b))
        p = exps / np.sum(exps)

        # gradient: 2 * (a reversed)⋆p
        a_rev = a[::-1]
        Ar_fft = np.fft.rfft(a_rev, n=size)
        Pr_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        # ------------------------------------------------------------------
        # momentum update + projection
        # ------------------------------------------------------------------
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD_ext it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Several short gradient‑descent runs with random jitter.
    Keeps the best result found before the given deadline.
    (kept for compatibility – the new Adam implementation is usually faster)
    """
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_v2_ext(
            start,
            time.time() + run_budget,
            target_sum,
            max_beta=1e5,
            step_factor=0.8,
            gamma=0.99,
        )
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The algorithm follows the stages described at the top of the file.
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # small safety margin

    # --------------------------------------------------------------
    # Choose problem size
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 400:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Build initial pool of diverse candidates
    # --------------------------------------------------------------
    rng = np.random.default_rng()
    init_seqs = []

    # power‑law families
    for p in np.linspace(0.45, 0.7, 6):
        init_seqs.append(_power_law_sequence(n, p, target_sum))

    # exponential decays
    for lam in np.linspace(0.0005, 0.003, 5):
        seq = np.exp(-lam * np.arange(n, dtype=np.float64))
        seq *= target_sum / seq.sum()
        init_seqs.append(seq)

    # Gaussian‑like bumps (different widths)
    for sigma_frac in (0.08, 0.12, 0.16, 0.20):
        sigma = sigma_frac * n
        x = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
        seq = np.exp(-0.5 * (x / sigma) ** 2)
        seq *= target_sum / seq.sum()
        init_seqs.append(seq)

    # uniform reference
    init_seqs.append(np.full(n, target_sum / n, dtype=np.float64))

    # previous best, if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        init_seqs.append(_resize_sequence(prev, n))

    # jittered copies for diversity
    diversified = []
    for seq in init_seqs:
        diversified.append(seq)
        jitter = rng.normal(scale=target_sum * 0.02, size=n)
        seq_jit = np.maximum(seq + jitter, 0.0)
        seq_jit = seq_jit * target_sum / seq_jit.sum()
        diversified.append(seq_jit)
    init_seqs = diversified

    # --------------------------------------------------------------
    # Phase A – Adam sub‑gradient descent (fast)
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    phase_deadline = time.time() + (deadline - time.time()) * 0.30
    for cand in init_seqs:
        if time.time() >= phase_deadline:
            break
        a_opt, val_opt = _adam_subgradient_descent(cand, target_sum, phase_deadline)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] New global best after Adam = {best_score:.6f}")

    # fallback (should never happen)
    if best_seq is None:
        best_seq = _power_law_sequence(n, 0.55, target_sum)
        best_score = _max_convolution_fft(best_seq)

    # --------------------------------------------------------------
    # Phase B – Deterministic mass‑transfer refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        mt_deadline = time.time() + (deadline - time.time()) * 0.30
        a_ref, val_ref = _deterministic_mass_transfer(
            best_seq, mt_deadline, target_sum,
            start_temp=0.8, end_temp=1e-5, low_frac=0.04
        )
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] After mass‑transfer best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase C – Simple re‑ordering heuristics
    # --------------------------------------------------------------
    if time.time() < deadline:
        reorder_deadline = time.time() + (deadline - time.time()) * 0.05
        reorder_cands = [best_seq, _alternating_order(best_seq)]
        for _ in range(3):
            tmp = best_seq.copy()
            rng.shuffle(tmp)
            reorder_cands.append(tmp)

        for cand in reorder_cands:
            if time.time() >= reorder_deadline:
                break
            val = _max_convolution_fft(cand)
            if val < best_score:
                best_score = val
                best_seq = cand.copy()
                print(f"[{time.time():.2f}] Ordering improvement -> score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase D – Permutation annealing (swap‑based SA)
    # --------------------------------------------------------------
    if time.time() < deadline:
        perm_deadline = time.time() + (deadline - time.time()) * 0.20
        a_perm, val_perm = _permutation_annealing(
            best_seq, target_sum, perm_deadline, init_score=best_score
        )
        if val_perm < best_score:
            best_score = val_perm
            best_seq = a_perm.copy()
            print(f"[{time.time():.2f}] After permutation annealing best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase E – Final polishing (high‑β projected gradient descent)
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _annealed_gradient_descent_v2_ext(
            best_seq,
            deadline,
            target_sum,
            max_beta=3e5,
            step_factor=0.5,
            gamma=0.99,
        )
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best={best_score:.6f}")

    # --------------------------------------------------------------
    # Trim trailing zeros, evaluate and return
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        val = evaluate_sequence(final_seq)      # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for possible later runs
    globals()["height_sequence_1"] = final_seq

    return final_seq
