"""
Propose a step‑function that is (almost) optimal for the objective

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)^2)

The continuous problem is minimized by a symmetric triangular function g on
[-1,1] with ∫g = 1.  Its self‑convolution attains the value
∫g² = 2/3, giving an objective value 2·(2/3)=4/3≈1.33333, well below the
target 1.503.  We discretise this triangle, scale it so that the ℓ¹‑norm is
√(2 n) (the same normalisation used in the reference code) and test a few
lengths.  The best discrete triangle found is returned.

The algorithm is completely deterministic, runs in a fraction of the
1000 s budget and needs no external optimisation machinery.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: discrete symmetric triangle with exact ℓ¹‑norm √(2 n)
# ----------------------------------------------------------------------


def _triangle_sequence(n: int) -> np.ndarray:
    """
    Return a non‑negative array of length n that follows a symmetric
    triangle (height 1 at the centre, zero at the ends) and is scaled so
    that its sum equals sqrt(2·n).
    """
    # indices 0 … n‑1
    idx = np.arange(n, dtype=float)

    # centre of the array (may be half‑integer)
    centre = (n - 1) / 2.0

    # distance from the centre, normalised to [0, 1]
    dist = np.abs(idx - centre) / centre

    # raw triangular shape, values in [0,1]
    shape = np.maximum(0.0, 1.0 - dist)

    # scale to the required ℓ¹‑norm  √(2·n)
    target = math.sqrt(2.0 * n)
    scale = target / shape.sum()
    return shape * scale


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Construct a triangular step‑function, try a few lengths and return the
    sequence with the smallest evaluate_sequence value found within the
    allocated wall‑clock budget.
    """
    # the random seed is not needed for this deterministic method,
    # but we keep the signature identical to the required one.
    rng = np.random.default_rng(seed)

    deadline = time.time() + budget_s - 0.5          # small safety margin

    # candidate lengths – larger lengths improve the discrete approximation
    # but also cost more time for the O(n²) evaluation.
    lengths = [2000, 4000, 8000, 12000, 20000]

    best_seq = None
    best_val = np.inf

    for n in lengths:
        if time.time() >= deadline:
            break

        # build the triangle for this length
        a = _triangle_sequence(n)

        # quick sanity check (evaluate_sequence will also clip values)
        if np.any(np.isnan(a)) or np.any(np.isinf(a)):
            continue

        try:
            val = evaluate_sequence(a.tolist())
        except Exception as exc:
            # if the external evaluator crashes we simply skip this length
            print(f"[{time.time():.1f}] length {n}: evaluation error – {exc}")
            continue

        print(f"[{time.time():.1f}] length {n}: evaluate_sequence = {val:.6f}")

        if val < best_val:
            best_val = val
            best_seq = a.copy()

    # ------------------------------------------------------------------
    # Fallback: if for any reason no sequence was produced (should never
    # happen), fall back to a uniform sequence of the smallest length.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)
        best_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] fallback uniform seq, value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Final reporting
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] BEST FOUND: length={best_seq.size}, value={best_val:.6f}")

    # convert to plain Python list of floats as required by the framework
    return best_seq.tolist()
