def solve_lp_max_sum(f, M, target_sum):
    n = len(f)
    # Build A_ub
    rows = 2*n-1
    A = np.zeros((rows, n), dtype=float)
    for i in range(n):
        # row indices k = i + j for j in range(n)
        # for each j, set column j row k = f[i]
        A[i:i+n, :]  # Not correct: row indices range i..i+n-1 where column index j = row - i
        # Actually for fixed i, for each j: row = i + j, col = j, set A[row, j] = f[i].
        # So we can fill column j by adding f[i] to rows i+j.
        # vectorized per i: rows = np.arange(i, i+n)
        # columns = np.arange(n)
    for i in range(n):
        rows_idx = np.arange(i, i + n)
        A[rows_idx, np.arange(n)] = f[i]
    # A shape (2n-1 , n). This loops O(n^2) but okay.
    # Now constraints: A @ g <= M (vector)
    # Bound: g >=0
    # Objective: minimize - sum(g) -> maximize sum(g)
    # Use linprog
    from scipy.optimize import linprog
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(rows, M), bounds=[(0, None)]*n, method='highs')
    if not res.success:
        print('LP failed', res.message)
        return None
    g = res.x
    # Scale to target sum

def _lp_max_sum(f, M, target_sum):
    n = f.shape[0]
    rows = 2*n - 1
    # Build A_ub matrix
    A = np.zeros((rows, n), dtype=float)
    # vector f (size n)
    for j in range(n):
        A[j:j+n, j] = f
    # bound vector
    b_ub = np.full(rows, M, dtype=float)
    # objective to maximize sum(g) => minimize -sum(g)
    c = -np.ones(n, dtype=float)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if not res.success:
        # fallback: return None
        # print error
        return None
    g = res.x
    sum_g = g.sum()
    if sum_g <= 0:
        return None
    scale = target_sum / sum_g
    g_scaled = g * scale
    # Project minor numerical errors onto simplex
    g_scaled = _project_to_simplex(g_scaled, target_sum)
    return g_scaled

            best_score = min(best_score, final_score)

    # Clean trailing zeros
    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]

"""
Hybrid optimization for the step‑function problem.

The algorithm starts from the previous best sequence (if available) and
executes three phases:

1.  Many short restarts of projected Adam on a smooth (log‑sum‑exp)
    surrogate of the maximum auto‑convolution.
2.  A deterministic exact hill‑climb that moves mass away from the
    currently worst convolution entry.
3.  A linear‑programming refinement: given the current bound M
    (=max (a∗a)), we maximise the ℓ¹‑norm of a new vector g under the
    linear constraints (a∗g)≤M, rescale g to the simplex and combine it
    with the current solution (grid search on the mixing weight).

If time permits the sequence is up‑sampled to larger lengths and the
above phases are repeated.  The whole routine respects a wall‑clock
budget (default 1000 s) and returns the best sequence found.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using an FFT (fast, O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float,
                   deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max surrogate.
    Runs until the wall‑clock deadline.
    """
    a = a0.copy()
    n = a.size

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # learning‑rate base (scaled with the simplex radius)
    base_lr = 0.15 * target_sum

    # smoothed‑max schedule
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1

        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4 / (beta + 1.0),
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # increase smoothing
        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a

# Random starts / up‑sampling
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Deterministic hill‑climbing (exact max)
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that always attacks the current worst convolution entry.
    Moves a large fraction of the responsible mass to a position that reduces it.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()

    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))

        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]

        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        # counterpart values for possible destinations
        counterpart = np.zeros(n, dtype=float)
        j_low = i_low
        j_high = i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        order = np.argsort(counterpart[j_range])
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        else:
            break

        delta = a[i] * 0.8
        if delta <= 0.0:
            break

        b_new = b.copy()
        # linear update of convolution (O(n))
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # try a smaller move
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                target = math.sqrt(2.0 * n)
                a = _project_to_simplex(a, target)
                b = _full_convolution(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------


def _lp_max_sum(f: np.ndarray, M: float, target_sum: float) -> np.ndarray | None:
    """
    Solve  max Σ g_j   s.t.   f*g ≤ M, g ≥ 0.
    Returns the feasible vector g scaled to the simplex Σg = target_sum,
    or None if the LP fails.
    """
    n = f.shape[0]
    rows = 2 * n - 1

    # Build the (2n‑1)×n constraint matrix A where
    #   (A g)_k = Σ_i f_i g_{k-i}  (linear in g)
    A = np.zeros((rows, n), dtype=float)
    for j in range(n):
        A[j:j + n, j] = f  # each column j contributes f_i to rows i+j

    b_ub = np.full(rows, M, dtype=float)
    c = -np.ones(n, dtype=float)          # minimise –Σg_j  ⇔ maximise Σg_j
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs')
    except Exception as exc:
        print(f"[{time.time():.1f}] linprog exception: {exc}")
        return None

    if not res.success:
        print(f"[{time.time():.1f}] linprog failed: {res.message}")
        return None

    g = res.x
    s = g.sum()
    if s <= 0:
        return None
    scale = target_sum / s
    g_scaled = g * scale
    # Numerical clean‑up (projection onto the simplex)
    g_scaled = _project_to_simplex(g_scaled, target_sum)
    return g_scaled


def _best_combo(a: np.ndarray, g: np.ndarray, target_sum: float,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Search a few mixing weights α ∈ [0,1] for a convex combination
    (1‑α)·a + α·g (both already on the simplex).  Returns the best
    sequence and its max‑convolution value.
    """
    best_seq = a.copy()
    best_score = _max_convolution_fft(best_seq)

    alphas = [0.0, 0.25, 0.5, 0.75, 1.0]

    for _ in range(3):                     # refine the grid a few times
        new_alphas = []
        for i in range(len(alphas) - 1):
            new_alphas.append(0.5 * (alphas[i] + alphas[i + 1]))
        alphas = sorted(set(alphas + new_alphas))

        for alpha in alphas:
            if time.time() >= deadline:
                break
            seq = (1.0 - alpha) * a + alpha * g
            seq = np.maximum(seq, 0.0)
            seq = _project_to_simplex(seq, target_sum)
            score = _max_convolution_fft(seq)
            if score < best_score - 1e-12:
                best_score = score
                best_seq = seq.copy()

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """

    ``evaluate_sequence``.  The routine respects the wall‑clock budget
    (default 1000 s) and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1)  Initialise from a previous run if possible
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():

            n0 = prev.size
            target0 = math.sqrt(2.0 * n0)
            prev = _project_to_simplex(prev, target0)
            best_seq = prev.copy()
            best_score = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 "
                  f"(n={n0}) score={best_score:.6f}")
        except Exception as e:
            print(f"Seeding error: {e}")
            best_seq = None
            best_score = np.inf
    else:
        best_seq = None
        best_score = np.inf

    if best_seq is None:
        # fallback – start from a random 2000‑element vector
        n0 = 2000
        target0 = math.sqrt(2.0 * n0)
        best_seq = _random_initial_sequence(n0, rng)
        best_seq = _project_to_simplex(best_seq, target0)
        best_score = _max_convolution_fft(best_seq)
        print(f"[{time.time():.1f}] Random start (n={n0}) score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2)  Main optimisation on the current length
    # ------------------------------------------------------------------
    n = best_seq.size
    target_sum = math.sqrt(2.0 * n)

    remaining = deadline - time.time()
    if remaining < 10.0:
        # not enough time left – just return the seed
        print("[early exit] Insufficient time for optimisation.")
        return best_seq.tolist()

    # allocate time slices
    adam_budget = remaining * 0.45
    hc_budget = remaining * 0.30
    lp_budget = remaining * 0.20
    extra_budget = remaining - (adam_budget + hc_budget + lp_budget)

    # ------------------- Phase A – Adam restarts --------------------
    if adam_budget > 5.0:
        min_restart = 4.0

            if time.time() >= deadline:
                break

            start_a = best_seq.copy() if r == 0 else _random_initial_sequence(n, rng)
            a_opt = _adam_optimize(start_a, target_sum, this_deadline)
            sc = _max_convolution_fft(a_opt)

                local_best_score = sc

        if local_best_a is not None and local_best_score < best_score - 1e-12:
            best_seq = local_best_a
            best_score = local_best_score
            print(f"[{time.time():.1f}] Adam improvement score={best_score:.6f}")

    # ------------------- Phase B – deterministic hill‑climb ----------
    if time.time() < deadline - 1.0 and hc_budget > 2.0:
        a_cur = best_seq.copy()
        a_cur = _project_to_simplex(a_cur, target_sum)

        a_hc, sc_hc = _hill_climb_deterministic(a_cur, b_cur, hc_deadline)
        if sc_hc < best_score - 1e-12:
            best_seq = a_hc
            best_score = sc_hc
            print(f"[{time.time():.1f}] Hill‑climb improvement score={best_score:.6f}")

    if time.time() < deadline - 1.0 and lp_budget > 2.0:
        lp_deadline = min(deadline - 0.5, time.time() + lp_budget)
        while time.time() < lp_deadline:

            g = _lp_max_sum(best_seq, M, target_sum)
            if g is None:

            combo_seq, combo_sc = _best_combo(best_seq, g, target_sum, lp_deadline)
            if combo_sc < best_score - 1e-12:
                best_seq = combo_seq
                best_score = combo_sc
                print(f"[{time.time():.1f}] LP+combo improvement score={best_score:.6f}")
            else:

    # ------------------------------------------------------------------

        if time.time() >= deadline - 2.0:
            break
        # up‑sample current best to the new size
        a_big = _upsample_sequence(best_seq, n_big)
        target_big = math.sqrt(2.0 * n_big)

        # a short Adam run (≈10 s)
        sub_deadline = min(deadline - 0.5, time.time() + 10.0)
        a_big_opt = _adam_optimize(a_big, target_big, sub_deadline)
        sc_big = _max_convolution_fft(a_big_opt)
        if sc_big < best_score - 1e-12:
            best_seq = a_big_opt
            best_score = sc_big
            print(f"[{time.time():.1f}] Upsampled n={n_big} Adam improved score={best_score:.6f}")

        # brief hill‑climb
        if time.time() < deadline - 1.0:
            b_big = _full_convolution(best_seq)
            hc_deadline = min(deadline - 0.2, time.time() + 5.0)
            a_big_hc, sc_big_hc = _hill_climb_deterministic(best_seq, b_big,
                                                            hc_deadline)
            if sc_big_hc < best_score - 1e-12:
                best_seq = a_big_hc
                best_score = sc_big_hc
                print(f"[{time.time():.1f}] Upsampled n={n_big} hill‑climb improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4)  Final polish with any remaining time
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        n_fin = best_seq.size
        target_fin = math.sqrt(2.0 * n_fin)
        final_deadline = deadline - 1.0
        best_seq = _adam_optimize(best_seq, target_fin, final_deadline)

        if time.time() < deadline - 0.5:
            b_fin = _full_convolution(best_seq)
            best_seq, fin_sc = _hill_climb_deterministic(best_seq, b_fin,
                                                         deadline - 0.5)
            best_score = min(best_score, fin_sc)

    # ------------------------------------------------------------------
    # 5)  Clean‑up: drop trailing near‑zeros
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # 6)  Official evaluation and reporting
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
