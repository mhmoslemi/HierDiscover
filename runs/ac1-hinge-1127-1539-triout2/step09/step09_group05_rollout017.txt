    idx_order = np.argsort(a)  # ascending a
    for j in idx_order:

        sl = slack[j : j+n]  # length n
        # compute ratios where a_i>0
        mask = a > 1e-12
        if not np.any(mask):

        g[j] = delta
        # update slack
        slack[j : j+n] -= a * delta
        # ensure nonnegative slack (clip small negative)
        slack = np.maximum(slack, 0.0)
    return g

def smooth_descent(a0, deadline, rng):
    a = a0.copy()
    n = len(a)
    S = np.sqrt(2*n)
    best_a = a.copy()
    best_val = _max_convolution(a)
    tau = 0.05
    while time.time() < deadline:
        b = _full_convolution_fft(a)
        # compute softmax
        maxb = b.max()
        expb = np.exp(tau * (b - maxb))
        w = expb / expb.sum()
        # gradient
        a_rev = a[::-1]
        # conv of w and a_rev
        conv_len = len(w) + len(a_rev) -1
        f_len = _next_pow2(conv_len)
        Fw = np.fft.rfft(w, n=f_len)
        Fa_rev = np.fft.rfft(a_rev, n=f_len)
        conv = np.fft.irfft(Fw * Fa_rev, n=f_len)[:conv_len]
        # extract central portion corresponding to a length
        start = n - 1
        grad = 2.0 * conv[start:start+n]
        # step size: adapt like Polyak using best
        norm_sq = np.dot(grad, grad)
        cur_val = b.max()
        step = (cur_val - best_val) / (norm_sq + 1e-12)
        step = np.clip(step, 1e-6, 0.3)  # maybe larger
        # update with projection
        a = _project_onto_simplex(a - step * grad, S)
        # evaluate
        cur_val = _max_convolution(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        # increase tau
        tau = min(tau * 1.02, 5.0)
    return best_a, best_val
```

def greedy_lp_step(a, M):
    n = len(a)
    b = _full_convolution_fft(a)
    slack = M - b  # should be non-negative
    # tolerance
    g = np.zeros_like(a)
    # order indices by a ascending (so that small a's allow large delta)
    idx_order = np.argsort(a)
    for j in idx_order:
        # slice slack
        sl = slack[j:j+n]
        # compute feasible increase
        denom = a
        # avoid division by zero
        mask = denom > 1e-12
        if not np.any(mask):
            continue
        ratios = sl[mask] / denom[mask]
        delta = ratios.min()
        if delta <= 0:
            continue
        # assign delta
        g[j] = delta
        # update slack
        slack[j:j+n] -= a * delta
        # set small negatives to zero
        slack = np.maximum(slack, 0.0)
    return g
```

        except Exception:
            warm_seq = None

    # Parameters
    best_a = None
    best_val = np.inf

    # Phase 1 random exploration
    explore_until = end_time - 300
    attempts = 0
    while time.time() < explore_until:
        attempts += 1
        # choose length
        n = int(rng.integers(3000, 20000))
        # random family
        style = rng.choice(['powerlaw','exponential','linear','ushape','sine','uniform','dirichlet','sparse'])
        if style == 'powerlaw':
            raw = _init_powerlaw(n, rng)
        elif style == 'exponential':
            raw = _init_exponential(n, rng)
        elif style == 'linear':
            raw = _init_linear(n, rng)
        elif style == 'ushape':
            raw = _init_ushape(n, rng)
        elif style == 'sine':
            raw = _init_sine(n, rng)
        elif style == 'uniform':
            raw = _init_uniform(n, rng)
        elif style == 'dirichlet':
            raw = _init_dirichlet(n, rng)
        else:
            raw = _init_sparse(n, rng)
        raw = np.clip(raw, 0, None)
        # scale to sum sqrt(2n)
        S = np.sqrt(2.0 * n)
        total = raw.sum()
        if total <= 0:
            raw = np.full(n, 1.0)
            total = n
        a = raw * (S / total)
        val = _max_convolution(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()

    # Ensure we have a best

"""
Improved step‑function optimizer.

The objective returned by ``evaluate_sequence`` is
    2·n·max_k (a∗a)[k] / (∑a)^2 .
Because the expression is homogeneous we may (and do) keep the
ℓ1‑norm of the vector fixed to S=√(2·n) by Euclidean projection onto the
simplex {a_i≥0,∑a_i=S}.  The optimiser consists of five stages:

1. Broad random exploration of several parametric families (including a
   sparse‑random family) for lengths 3 000–20 000.
2. Projected sub‑gradient descent on the exact max‑convolution (hard
   max) with a Polyak step‑size.
3. A smooth‑max descent where the max is approximated by a softmax
   (temperature τ).  This yields a smoother gradient and often escapes
   plateaus.
4. A cheap greedy “LP‑step’’ that solves approximately
   maximise ∑g_j   s.t.   a∗g ≤ max(a∗a) , g≥0.
   The resulting direction g is mixed into the current solution with a
   small step and projected back to the simplex.
5. A short simulated‑annealing hill‑climb for final polishing.

All stages respect the 1000 s budget.  The final sequence is stored in the
global variable ``height_sequence_1`` and returned as a plain list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT, scaling, projection, gradient)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a∗a via real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a∗a."""
    return _full_convolution_fft(a).max()


def _max_convolution_and_gradient(a: np.ndarray):
    """
    Returns (grad, max_val) where grad is a sub‑gradient of
    maxₖ(a∗a)[k] w.r.t. a and max_val the current maximum.
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_val = b.max()
    mask = (b == max_val).astype(float)

    # gradient = 2·(mask ∗ reverse(a))
    a_rev = a[::-1]
    conv_len = len(mask) + n - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad, max_val


def _smooth_max_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the smooth approximation
        fτ(a) = (1/τ)·log(∑_k exp(τ·(a∗a)[k]))
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = b.max()
    # softmax weights (numerically stable)
    expb = np.exp(tau * (b - max_b))
    w = expb / expb.sum()

    # grad = 2·(w ∗ reverse(a))
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    f_len = _next_pow2(conv_len)
    Fw = np.fft.rfft(w, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fw * Fa_rev, n=f_len)[:conv_len]
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    smooth_val = (np.log(expb.sum()) / tau) + max_b  # actual fτ value (not used)
    return grad, smooth_val


def _project_onto_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {w | w_i≥0, ∑w_i = S}.
    Implementation of Duchi et al. (O(n)).
    """
    if S < 0:
        raise ValueError("Simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]

def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    off = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + off) ** (-p)
    raw = mix * edge + (1.0 - mix) * np.ones(n)
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    raw = mix * np.exp(-lam * d) + (1.0 - mix) * np.ones(n)
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    raw = mix * ush + (1.0 - mix) * np.ones(n)
    return raw


def _init_sine(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    vals = np.sin(np.pi * (i + 1) / (n + 1))
    mix = rng.uniform(0.0, 1.0)
    raw = mix * vals + (1.0 - mix) * np.ones(n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    return rng.random(n)


def _init_sparse(n: int, rng: np.random.Generator) -> np.ndarray:
    """Very sparse vector – only a few positions carry mass."""
    density = rng.uniform(0.01, 0.15)
    k = max(1, int(density * n))
    raw = np.zeros(n)
    idx = rng.choice(n, size=k, replace=False)
    raw[idx] = rng.uniform(0.1, 10.0, size=k)
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    With probability ≈0.2 a warm‑start from the globally best sequence is used;
    otherwise a random parametric family (including the sparse family) is sampled.
    The vector is finally projected onto the simplex of radius √(2·n).
    """
    S = np.sqrt(2.0 * n)

    # warm start
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        tot = a.sum()
        if tot > 0:
            a = a * (S / tot)
            return a

    families = [
        _init_powerlaw, _init_exponential, _init_linear,
        _init_ushape, _init_sine, _init_uniform,
        _init_dirichlet, _init_sparse
    ]
    raw = rng.choice(families)(n, rng)
    raw = np.clip(raw, 0.0, None)
    tot = raw.sum()
    if tot == 0.0:
        raw = np.full(n, 1.0)
        tot = n
    a = raw * (S / tot)
    return a

def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         global_best: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=max(a∗a).
    Polyak step size (cur‑best)/‖g‖² is used.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_val = _max_convolution_and_gradient(a)

        norm_sq = np.dot(grad, grad)
        if norm_sq > 0:
            step = (cur_val - best_val) / (norm_sq + 1e-12)
            step = np.clip(step, 1e-7, 0.3)
        else:
            step = 0.0

        a = _project_onto_simplex(a - step * grad, S)

        cur_val = _max_convolution(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to avoid stagnation
        if it % 400 == 0:
            jitter = rng.normal(scale=0.001 * np.mean(a), size=n)
            a = _project_onto_simplex(a + jitter, S)

    return best_a, best_val


def _smooth_descent(a0: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Descent on a smooth approximation of the max using a softmax with
    temperature τ.  τ is increased gradually (annealing) to approach the true
    max.  Projection onto the simplex is applied after each step.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _max_convolution(a)

    tau = 0.08  # start with a low temperature
    while time.time() < deadline:
        grad, _ = _smooth_max_gradient(a, tau)

        norm_sq = np.dot(grad, grad)
        cur_val = _max_convolution(a)
        if norm_sq > 0:
            step = (cur_val - best_val) / (norm_sq + 1e-12)
            step = np.clip(step, 1e-7, 0.25)
        else:
            step = 0.0

        a = _project_onto_simplex(a - step * grad, S)

        cur_val = _max_convolution(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        tau = min(tau * 1.03, 5.0)      # raise temperature slowly
        if rng.random() < 0.005:        # tiny random jitter
            a = _project_onto_simplex(a + rng.normal(scale=0.0005 * S, size=n), S)

    return best_a, best_val


def _anneal_local_search(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing hill‑climb moving a random fraction of mass
    between two positions.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    cur_val = _max_convolution(a)
    best_a = a.copy()
    best_val = cur_val

    T = 0.05 * cur_val
    decay = 0.9999

    it = 0
    while time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        frac = rng.uniform(0.02, 0.18)
        delta = a[i] * frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _max_convolution(a_new)
        dval = new_val - cur_val
        if dval < 0 or rng.random() < np.exp(-dval / (T + 1e-12)):
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()

        T *= decay
        if it % 500 == 0:
            jitter = rng.normal(scale=0.001 * S, size=n)
            a = _project_onto_simplex(a + jitter, S)
            cur_val = _max_convolution(a)

    return best_a, best_val


def _greedy_lp_step(a: np.ndarray, M: float) -> np.ndarray:
    """
    Approximate the LP
        maximise ∑ g_j
        s.t. a∗g ≤ M , g≥0 .
    A simple greedy construction is used: for each position j we compute
    the maximal admissible increase (limited by all constraints) and set
    g_j to that amount.  The result is feasible but not necessarily optimal.
    """
    n = len(a)
    b = _full_convolution_fft(a)
    slack = M - b                     # non‑negative vector, length 2n‑1
    g = np.zeros_like(a)

    # indices with smallest a first – they allow the largest deltas
    order = np.argsort(a)
    for j in order:
        sl = slack[j:j + n]           # constraints that involve g_j
        mask = a > 1e-12
        if not np.any(mask):
            continue
        ratios = sl[mask] / a[mask]
        delta = ratios.min()
        if delta <= 0.0:
            continue
        g[j] = delta
        slack[j:j + n] -= a * delta
        # numerical safety
        slack = np.maximum(slack, 0.0)
    return g


def _mix_with_direction(a: np.ndarray,
                       d: np.ndarray,
                       t: float,
                       S: float) -> np.ndarray:
    """
    Mix current vector a with direction d using mixing parameter t
    (0 < t ≤ 1) and re‑project onto the simplex of radius S.
    """
    mixed = (1.0 - t) * a + t * d
    return _project_onto_simplex(mixed, S)
