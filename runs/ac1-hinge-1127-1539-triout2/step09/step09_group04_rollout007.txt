        max_x = np.max(x)

        # Adam updates
        step += 1
        m = beta1 * m + (1 - beta1) * g_x
        v = beta2 * v + (1 - beta2) * (g_x * g_x)
        m_hat = m / (1 - beta1**step)

"""
Proposed improvement:

* Warm‑start from a previous run or a power‑law shape.
* Very cheap parametric search on a quarter‑size problem (two parameters).
* Adam‑style mirror descent on the simplex using a soft‑max approximation of
  the maximal convolution entry.  The update is performed in log‑space so
  the ℓ¹‑norm is kept exactly by the soft‑max normalisation.  The temperature
  β is increased gradually, letting the optimiser first see a smooth
  landscape and finally focus on the true max‑convolution.
* A deterministic mass‑transfer step that moves a small fraction of mass
  from the largest contributing pair to a low‑contributing one.
* Exact sub‑gradient descent on the real max‑convolution for a final polish.
* All phases respect the global time budget (default 1000 s) and keep the
  best sequence seen so far.

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution / scoring utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).

    Returns
    -------
    grad : np.ndarray
        ∂ (soft‑max) / ∂a  (shape n)
    conv : np.ndarray
        Full convolution a*a (length 2·n‑1)
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    # stable soft‑max
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * (reverse(a) convolve p)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b

def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) shape, normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _grid_param_search(small_n: int,
                       target_sum: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Search a two‑parameter family a_i = (i+1)^(-p)·exp(-λ·i/small_n).
    Returns the best sequence found (scaled to target_sum) and its score.
    """
    p_vals = np.linspace(0.2, 1.2, 17)
    lam_vals = np.linspace(0.0, 6.0, 13)
    best_seq = None
    best_score = np.inf
    i = np.arange(small_n, dtype=np.float64)

    for p in p_vals:
        for lam in lam_vals:
            if time.time() > deadline:
                break
            seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
            seq = seq * (target_sum / seq.sum())
            max_b = _max_convolution_fft(seq)
            score = 2.0 * small_n * max_b / (target_sum * target_sum)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"[{time.time():.2f}] Grid best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")
        else:
            continue
        break

    rng = np.random.default_rng()
    while time.time() < deadline:
        p = rng.uniform(0.2, 1.2)
        lam = rng.exponential(1.0) * 3.0
        seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
        seq = seq * (target_sum / seq.sum())
        max_b = _max_convolution_fft(seq)
        score = 2.0 * small_n * max_b / (target_sum * target_sum)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Random best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")

    return best_seq, best_score

def _adam_softmax_descent(a0: np.ndarray,
                          target_sum: float,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using Adam updates in log‑space.
    The objective is a soft‑max approximation of the maximal convolution
    entry (temperature β).  β is increased exponentially during the run.
    """
    n = a0.size
    eps = 1e-12
    x = np.log(a0 + eps)                         # log‑variables
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    step = 0
    best_a = None
    best_score = np.inf
    beta = 0.5
    max_beta = 5000.0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # a = target_sum * softmax(x)
        mx = np.max(x)
        exp_x = np.exp(x - mx)
        a = exp_x / exp_x.sum() * target_sum

        # gradient of the smoothed max‑convolution
        grad_a, conv = _gradient_softmax(a, beta)

        # gradient w.r.t. x (mirror‑descent on the simplex)
        avg = np.dot(grad_a, a) / target_sum
        grad_x = a * (grad_a - avg)

        step += 1
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)

        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)

        # step size decays with β
        alpha = 0.2 / (beta + 1.0)
        x -= alpha * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional jitter to escape plateaus
        if step % 500 == 0:
            jitter = rng.normal(scale=0.01 * target_sum / n, size=n)
            x += jitter

        # evaluate current score (exact max‑conv, no extra FFT)
        max_b = conv.max()
        score = 2.0 * n * max_b / (target_sum * target_sum)

        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={step:5d} β={beta:6.1f} score={best_score:.6f}")

        # increase temperature gradually
        if step % 2000 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

    if best_a is None:
        best_a = a0.copy()
        best_score = 2.0 * n * _max_convolution_fft(best_a) / (target_sum * target_sum)
    return best_a, best_score

def _mass_transfer_refine(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Move a modest fraction of mass from high‑contributing pairs to low‑contributing ones."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.size
    best_a = a.copy()
    best_val = 2.0 * n * _max_convolution_fft(a) / (target_sum * target_sum)
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    while time.time() < deadline:
        k_max = int(np.argmax(conv))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = a[k_max - idx]                       # a_{k_max‑i}
        weight = a[idx] * partner                       # contribution a_i·a_{k_max‑i}

        donor = idx[np.argmax(weight)]

        low_thr = np.percentile(weight, 10)
        low_idx = idx[weight <= low_thr]
        if low_idx.size:
            receiver = low_idx[np.argmin(weight[weight <= low_thr])]
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            donor = rng.choice(idx)
            receiver = rng.integers(n)
            if donor == receiver:
                continue

        max_frac = 0.04
        delta = a[donor] * rng.random() * max_frac
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta

        conv = _conv_fft(a)
        val = 2.0 * n * conv.max() / (target_sum * target_sum)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Transfer improve score={best_val:.6f}")
        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta
            conv = _conv_fft(a)

        # renormalise occasional numerical drift
        cur_sum = a.sum()
        if abs(cur_sum - target_sum) > 1e-9:
            a = a * (target_sum / cur_sum)

    return best_a, best_val

def _subgradient_max_descent(a0: np.ndarray,
                             target_sum: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true maximal convolution entry."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = 2.0 * a.size * _max_convolution_fft(a) / (target_sum * target_sum)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())

        # sub‑gradient of that entry: 2·a[k_max‑i] on feasible indices
        grad = np.zeros_like(a)
        lo = max(0, k_max - (a.shape[0] - 1))
        hi = min(k_max, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        grad[idx] = 2.0 * a[partner]

        norm = np.linalg.norm(grad)
        if norm == 0:
            break

        # maximal safe step so we stay non‑negative before projection
        safe_eta = np.min(np.where(grad > 0, a / grad, np.inf))
        eta = min(0.5 * safe_eta, 0.5 / np.sqrt(it + 1))

        a = a - eta * grad
        a = _project_to_simplex(a, target_sum)

        val = 2.0 * a.size * _max_convolution_fft(a) / (target_sum * target_sum)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} score={best_val:.6f}")

        # occasional jitter
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        it += 1

    return best_a, best_val

    # Phase 1 – Adam mirror‑descent on soft‑max surrogate
    # ------------------------------------------------------------------
    if time.time() < deadline:
        adam_deadline = time.time() + (deadline - time.time()) * 0.45
        best_seq, best_score = _adam_softmax_descent(init, target_sum, adam_deadline)
        print(f"[{time.time():.2f}] After Adam best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – deterministic mass‑transfer refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        mt_deadline = time.time() + (deadline - time.time()) * 0.25
        seq_mt, val_mt = _mass_transfer_refine(best_seq, target_sum, mt_deadline)
        if val_mt < best_score:
            best_seq = seq_mt
            best_score = val_mt
        print(f"[{time.time():.2f}] After mass‑transfer best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – sub‑gradient descent on the exact max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_sg, val_sg = _subgradient_max_descent(best_seq, target_sum, deadline)
        if val_sg < best_score:
            best_seq = seq_sg
            best_score = val_sg
        print(f"[{time.time():.2f}] After sub‑gradient best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Clean up and final evaluation
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # Store for possible warm‑starts in subsequent calls
    globals()["height_sequence_1"] = final_seq
    return final_seq
