def _double_power_sequence(n, p, target_sum):
    i = np.arange(n, dtype=np.float64)
    a = ((i+1)*(n-i))**(-p)
    sum_ = a.sum()
    a = a * target_sum / sum_
    return a

    return a

# Convolution utilities (FFT based)
# --------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (∑a)²."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# --------------------------------------------------------------
# Simplex projection (Euclidean)
# --------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Families for the fast parametric sweep
# --------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _double_power_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    a *= target_sum / a.sum()
    return a


def _exp_double_power_sequence(
    n: int, p: float, q: float, target_sum: float
) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _edge_sequence(n: int, frac: float, target_sum: float) -> np.ndarray:
    """Mass concentrated on the two borders; length of each border = frac·n."""
    L = max(1, int(frac * n))
    # safety: avoid overlapping borders
    if 2 * L > n:
        L = n // 2
    a = np.zeros(n, dtype=np.float64)
    h = target_sum / (2.0 * L)
    a[:L] = h
    a[-L:] = h
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# --------------------------------------------------------------
# Soft‑max gradient of the max‑convolution (used in GD)
# --------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where conv is the full convolution a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# --------------------------------------------------------------
# Projected gradient descent with annealed soft‑max temperature
# --------------------------------------------------------------


def _annealed_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 2e5,
    step_factor: float = 0.6,
    gamma: float = 0.99,
) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max approximation.
    Returns (best_vector, best_score).
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    step_base = target_sum * step_factor
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 300 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(
    init: np.ndarray, target_sum: float, deadline: float
) -> tuple[np.ndarray, float]:
    """
    Multiple short GD runs with random jitter; keep the best result.
    """
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent(
            start,
            time.time() + run_budget,
            target_sum,
            max_beta=2e5,
            step_factor=0.8,
            gamma=0.99,
        )
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val


# --------------------------------------------------------------
# Deterministic mass‑transfer refinement (phase B)
# --------------------------------------------------------------


def _deterministic_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    start_temp: float = 0.5,
    end_temp: float = 1e-5,
    low_frac: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Greedy moves that directly reduce the current maximal convolution entry.
    Uses a Metropolis rule with an exponentially decaying temperature.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to the max convolution entry
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0.0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0.0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# --------------------------------------------------------------
# Global random mass‑transfer (extra exploration)
# --------------------------------------------------------------


def _global_random_mass_transfer(
    a0: np.ndarray, deadline: float, target_sum: float
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing moves that shift a small amount of mass
    from the largest entries to the smallest ones.
    """
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random new best={best_score:.6f}")

    return best_a, best_score


# --------------------------------------------------------------
# Main entry point required by the competition harness
# --------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with a low benchmark score.
    The routine obeys the time budget, returns the best sequence found,
    and prints progress information.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ----------------------------------------------------------
    # Choose a set of problem sizes (different n values)
    # ----------------------------------------------------------
    possible_n = [8000, 12000, 16000, 20000]
    best_seq = None
    best_val = np.inf
    best_n = None

    rng = np.random.default_rng()

    # ----------------------------------------------------------
    # Phase 0 – wide parametric pre‑search over several n
    # ----------------------------------------------------------
    for n in possible_n:
        target_sum = np.sqrt(2.0 * n)

        candidates = []

        # pure power‑law
        for p in np.linspace(0.30, 0.90, 12):
            candidates.append(_power_law_sequence(n, p, target_sum))

        # power‑law + exponential tail
        for p in np.linspace(0.30, 0.90, 6):
            for q in np.linspace(0.0, 5.0, 6):
                candidates.append(_exp_power_law_sequence(n, p, q, target_sum))

        # double power‑law (symmetric)
        for p in np.linspace(0.30, 0.90, 10):
            candidates.append(_double_power_sequence(n, p, target_sum))

        # double power‑law + exponential
        for p in np.linspace(0.30, 0.90, 4):
            for q in np.linspace(0.0, 5.0, 4):
                candidates.append(_exp_double_power_sequence(n, p, q, target_sum))

        # edge‑concentrated families
        for frac in np.linspace(0.01, 0.10, 8):
            candidates.append(_edge_sequence(n, frac, target_sum))

        # few Dirichlet‑sampled random vectors
        for _ in range(30):
            vec = rng.gamma(shape=1.0, size=n)
            vec = _project_to_simplex(vec, target_sum)
            candidates.append(vec)

        # jittered variants of everything above (helps break symmetry)
        for seq in list(candidates):
            jitter = rng.normal(scale=target_sum * 0.005, size=n)
            seq_j = seq + jitter
            seq_j = np.maximum(seq_j, 0.0)
            seq_j = _project_to_simplex(seq_j, target_sum)
            candidates.append(seq_j)

        # evaluate a manageable subset
        max_initial = min(250, len(candidates))
        sample_idx = rng.choice(len(candidates), size=max_initial, replace=False)

        for i in sample_idx:
            seq = candidates[i]
            val = _score(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                best_n = n

        print(
            f"[{time.time():.2f}] After parametric sweep (n={n}) best={best_val:.6f}"
        )
        # early exit if time is running out
        if time.time() > deadline - 15.0:
            break

    if best_seq is None:
        # fallback – simple uniform sequence
        best_n = possible_n[0]
        target_sum = np.sqrt(2.0 * best_n)
        best_seq = np.full(best_n, target_sum / best_n)
        best_val = _score(best_seq)

    # ----------------------------------------------------------
    # Phase A – short projected GD restarts
    # ----------------------------------------------------------
    target_sum = np.sqrt(2.0 * best_n)

    while time.time() < deadline - 20.0:
        remaining = deadline - time.time()
        run_budget = min(12.0, remaining * 0.30)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.01, size=best_seq.shape[0])
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _gradient_descent_with_restarts(
            start_vec, target_sum, time.time() + run_budget
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    # ----------------------------------------------------------
    # Phase B – deterministic mass‑transfer refinement
    # ----------------------------------------------------------
    if time.time() < deadline - 10.0:
        a_ref, val_ref = _deterministic_mass_transfer(
            best_seq,
            deadline,
            target_sum,
            start_temp=0.5,
            end_temp=1e-6,
            low_frac=0.04,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] DMT improved best={best_val:.6f}")

    # ----------------------------------------------------------
    # Phase C – global random mass‑transfer
    # ----------------------------------------------------------
    if time.time() < deadline - 5.0:
        a_ref2, val_ref2 = _global_random_mass_transfer(best_seq, deadline, target_sum)
        if val_ref2 < best_val:
            best_val = val_ref2
            best_seq = a_ref2.copy()
            print(f"[{time.time():.2f}] Global random improved best={best_val:.6f}")

    # ----------------------------------------------------------
    # Phase D – final polishing gradient descent
    # ----------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _annealed_gradient_descent(
            best_seq,
            deadline,
            target_sum,
            max_beta=1e6,
            step_factor=0.6,
            gamma=0.99,
        )
        if val_fin < best_val:
            best_val = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best={best_val:.6f}")

    # ----------------------------------------------------------
    # Post‑processing: trim trailing (near‑zero) entries
    # ----------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final evaluation (harmless if evaluate_sequence is not defined)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
