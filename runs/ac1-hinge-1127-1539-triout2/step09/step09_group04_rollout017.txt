def propose_candidate(budget_s=1000, **kwargs):
    start = time.time()
    deadline = start + budget_s - 0.5

    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    # initialize
    if 'height_sequence_1' in globals():
        init = _resize_sequence(np.array(globals()['height_sequence_1'], dtype=np.float64), n)
        init = init * (target_sum / init.sum())
    else:
        init = _power_law_sequence(n, 0.55, target_sum)

    # Stage 1: param search on small scale
    small_n = max(2000, n // 4)
    small_target = np.sqrt(2.0 * small_n)
    param_deadline = time.time() + min(10.0, (deadline - time.time()) * 0.07)
    seq_small, _ = _grid_param_search(small_n, small_target, param_deadline)
    if seq_small is not None:
        init = _resize_sequence(seq_small, n)
        init = init * (target_sum / init.sum())
        print(f"[{time.time():.2f}] Warmup from param search done")

    # Stage 2: exponential descent
    exp_deadline = time.time() + (deadline - time.time()) * 0.15
    a_exp, val_exp = _exponential_descent(init, target_sum, exp_deadline)
    best_a = a_exp
    best_val = val_exp
    print(f"[{time.time():.2f}] After ExpDesc best={best_val:.6f}")

    # Stage 3: gradient descent with restarts (softmax)
    if time.time() < deadline:
        gd_deadline = time.time() + (deadline - time.time()) * 0.25
        a_gd, val_gd = _gradient_descent_with_restarts(best_a, target_sum, gd_deadline)
        if val_gd < best_val:
            best_a, best_val = a_gd, val_gd
        print(f"[{time.time():.2f}] After GD best={best_val:.6f}")

    # Stage 4: aggressive mass transfer
    if time.time() < deadline:
        mt_deadline = time.time() + (deadline - time.time()) * 0.20
        a_mt, val_mt = _mass_transfer_aggressive(best_a, target_sum, mt_deadline)
        if val_mt < best_val:
            best_a, best_val = a_mt, val_mt
        print(f"[{time.time():.2f}] After aggressive MT best={best_val:.6f}")

    # Stage 5: random hill climbing
    if time.time() < deadline:
        hc_deadline = time.time() + (deadline - time.time()) * 0.15
        a_hc, val_hc = _random_hill_climb(best_a, target_sum, hc_deadline)
        if val_hc < best_val:
            best_a, best_val = a_hc, val_hc
        print(f"[{time.time():.2f}] After hillclimb best={best_val:.6f}")

    # Stage 6: subgradient descent with jitter
    if time.time() < deadline:
        sub_deadline = deadline
        a_sub, val_sub = _subgradient_descent_with_jitter(best_a, target_sum, sub_deadline)
        if val_sub < best_val:
            best_a, best_val = a_sub, val_sub
        print(f"[{time.time():.2f}] After subgrad final best={best_val:.6f}")

    # Trim zeros at both ends to reduce length
    arr = np.array(best_a, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[nz[0]: nz[-1] + 1]
    final_seq = arr.tolist()

    # Evaluate for logging
    try:
        eval_val = evaluate_sequence(final_seq)
        print(f"[{time.time():.2f}] Final evaluated score: {eval_val:.6f}")
    except Exception as e:
        print(f"[{time.time():.2f}] Evaluation failed: {e}")

    return final_seq

"""
Improved step‑function search for the convolution benchmark.

The algorithm follows a multi‑stage pipeline:

1. Warm‑start from the previously discovered best sequence (global variable
   ``height_sequence_1``) or from a simple power‑law shape.
2. A fast parametric sweep over a 2‑parameter family (power‑law + exponential
   taper) on a reduced scale, interpolated to the target length.
3. Exponential (mirror) descent on a soft‑max surrogate of the maximal
   convolution entry (β is increased exponentially).
4. Projected gradient descent with restarts on the same surrogate (momentum,
   occasional jitter).
5. Aggressive deterministic mass‑transfer: move mass from the index that
   contributes most to the current maximal convolution entry to a location
   that does not participate in that entry.
6. Random hill‑climbing with a simple simulated‑annealing acceptance rule.
7. Sub‑gradient descent on the exact max‑convolution entry, with a
   safe‑step computation that ignores zero components and periodic jitter.

All stages respect the non‑negativity and ℓ¹‑norm constraints, and each
intermediate improvement is kept.  The routine observes the supplied time
budget (default 1000 s) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution / scoring utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) shape, normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where conv is the full convolution a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Stage 1 – cheap parametric search on a reduced scale
# ----------------------------------------------------------------------


def _grid_param_search(small_n: int,
                       target_sum: float,
                       deadline: float) -> tuple[np.ndarray | None, float]:
    """Search a two‑parameter family a_i = (i+1)^(-p)·exp(-λ·i/small_n)."""
    p_vals = np.linspace(0.1, 2.0, 25)
    lam_vals = np.linspace(0.0, 12.0, 25)

    best_seq = None
    best_score = np.inf
    i = np.arange(small_n, dtype=np.float64)

    for p in p_vals:
        for lam in lam_vals:
            if time.time() > deadline:
                break
            seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
            seq = seq * (target_sum / seq.sum())
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()
                print(f"[{time.time():.2f}] Grid best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")
        else:
            continue
        break

    rng = np.random.default_rng()
    # a few random probes
    while time.time() < deadline:
        p = rng.uniform(0.1, 2.0)
        lam = rng.exponential(1.0) * 6.0
        seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
        seq = seq * (target_sum / seq.sum())
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Random best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stage 2 – exponential (multiplicative) descent
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Mirror‑descent with exponential updates using a soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 1 << 16            # 65536
    base_eta = 0.03
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 3 – projected gradient descent with restarts (soft‑max)
# ----------------------------------------------------------------------


def _annealed_gradient_descent_v2(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the soft‑max surrogate."""
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 1 << 16
    step_base = target_sum * 0.45
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.93
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0015 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GDv2 it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several short GD restarts with random jitter; keep the best."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(28.0, remaining * 0.34)
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.018, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_v2(start,
                                               time.time() + run_budget,
                                               target_sum)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val

# Stage 4 – aggressive deterministic mass‑transfer
# ----------------------------------------------------------------------


def _mass_transfer_aggressive(a0: np.ndarray,
                             target_sum: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    """
    Move mass from the index that contributes most to the current maximal
    convolution entry to an index that does not participate in that entry.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    while time.time() < deadline:
        k_max = int(conv.argmax())
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        weight = a[idx] * a[partner]

        donor = idx[np.argmax(weight)]

        # pick a receiver outside the active interval – contribution zero
        outside = np.setdiff1d(np.arange(n), idx, assume_unique=True)
        if outside.size > 0:
            receiver = outside[np.argmin(a[outside])]
        else:
            # fallback: pick a low‑weight inside interval
            low_thr = np.percentile(weight, 5)
            cand = idx[weight <= low_thr]
            receiver = cand[np.argmin(weight[weight <= low_thr])] if cand.size else rng.integers(n)

        if donor == receiver:
            # ensure a genuine move
            receiver = rng.integers(n)
            if donor == receiver:
                continue

        max_frac = 0.025
        delta = a[donor] * rng.random() * max_frac
        if delta <= 0:
            continue

        # perform tentative move
        a[donor] -= delta
        a[receiver] += delta

        # renormalise (numerical drift)
        a = a * (target_sum / a.sum())
        conv = _conv_fft(a)
        val = _score(a)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Aggressive MT improve score={best_val:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = a * (target_sum / a.sum())
            conv = _conv_fft(a)

    return best_a, best_val

# Stage 5 – random hill‑climbing with simulated annealing
# ----------------------------------------------------------------------


def _random_hill_climb(a0: np.ndarray,
                       target_sum: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """Small stochastic mass moves accepted with a cooling schedule."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)
    n = a.size
    rng = np.random.default_rng()
    start_time = time.time()

    while time.time() < deadline:
        # linear cooling
        t_elapsed = time.time() - start_time
        t_total = deadline - start_time
        temperature = 0.02 * (1.0 - t_elapsed / t_total) + 1e-5

        i = rng.integers(n)
        j = rng.integers(n)
        if i == j or a[i] <= 1e-15:
            continue

        frac = rng.random() * 0.008  # ≤0.8 % of donor mass
        delta = a[i] * frac

        a[i] -= delta
        a[j] += delta
        a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Hillclimb improve score={best_val:.6f}")
        else:
            # Metropolis acceptance
            prob = np.exp((best_val - val) / temperature)
            if rng.random() < prob:
                best_val = val
                best_a = a.copy()
            else:
                # revert
                a[i] += delta
                a[j] -= delta
                a = a * (target_sum / a.sum())

    return best_a, best_val

def _subgradient_descent_with_jitter(a0: np.ndarray,
                                    target_sum: float,
                                    deadline: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true maximal convolution entry."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    it = 0
    n = a.size

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[partner]

        # safe step that respects non‑negativity (ignore zero components)
        pos_mask = (a > 1e-14) & (grad > 0)
        if np.any(pos_mask):
            safe_eta = np.min(a[pos_mask] / grad[pos_mask])
        else:
            safe_eta = np.inf

        eta = min(0.45 * safe_eta, 0.45 / np.sqrt(it + 1))

        a = a - eta * grad
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it} score={best_val:.6f}")

        # occasional jitter to escape flat plateaus
        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Choose problem size based on the allotted budget
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Warm‑start (fallback to power‑law if no previous result)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"],
                                        dtype=np.float64), n)
        init = init * (target_sum / init.sum())
    else:
        init = _power_law_sequence(n, 0.55, target_sum)

    # ------------------------------------------------------------------
    # Stage 1 – cheap parametric search on a reduced scale
    # ------------------------------------------------------------------
    small_n = max(2000, n // 4)
    small_target = np.sqrt(2.0 * small_n)
    param_deadline = time.time() + min(12.0,
                                      (deadline - time.time()) * 0.07)
    seq_small, _ = _grid_param_search(small_n, small_target, param_deadline)

    if seq_small is not None:
        init = _resize_sequence(seq_small, n)
        init = init * (target_sum / init.sum())
        print(f"[{time.time():.2f}] Parametric warm‑start ready (n={n})")

    # ------------------------------------------------------------------
    # Stage 2 – exponential (multiplicative) descent
    # ------------------------------------------------------------------
    exp_deadline = time.time() + (deadline - time.time()) * 0.15
    a_exp, val_exp = _exponential_descent(init, target_sum, exp_deadline)
    best_seq = a_exp
    best_score = val_exp
    print(f"[{time.time():.2f}] After ExpGD best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – projected gradient descent with restarts (soft‑max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        gd_deadline = time.time() + (deadline - time.time()) * 0.25
        a_gd, val_gd = _gradient_descent_with_restarts(best_seq,
                                                       target_sum,
                                                       gd_deadline)
        if val_gd < best_score:
            best_seq, best_score = a_gd, val_gd
        print(f"[{time.time():.2f}] After GD best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – aggressive deterministic mass‑transfer
    # ------------------------------------------------------------------
    if time.time() < deadline:
        mt_deadline = time.time() + (deadline - time.time()) * 0.20
        a_mt, val_mt = _mass_transfer_aggressive(best_seq,
                                                 target_sum,
                                                 mt_deadline)
        if val_mt < best_score:
            best_seq, best_score = a_mt, val_mt
        print(f"[{time.time():.2f}] After Aggressive MT best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – random hill‑climbing with annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        hc_deadline = time.time() + (deadline - time.time()) * 0.15
        a_hc, val_hc = _random_hill_climb(best_seq,
                                          target_sum,
                                          hc_deadline)
        if val_hc < best_score:
            best_seq, best_score = a_hc, val_hc
        print(f"[{time.time():.2f}] After Hill‑climb best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 6 – sub‑gradient descent on exact max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_sub, val_sub = _subgradient_descent_with_jitter(best_seq,
                                                          target_sum,
                                                          deadline)
        if val_sub < best_score:
            best_seq, best_score = a_sub, val_sub
        print(f"[{time.time():.2f}] After Sub‑gradient best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Cleanup – drop leading/trailing near‑zeros and final evaluation
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[nz[0]: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # keep the best sequence for possible warm‑starts in the next call
    globals()["height_sequence_1"] = final_seq

    return final_seq
