def solve_lp_max_sum(f, M, S):
    n = len(f)
    A = build_convolution_matrix(f)
    b_ub = np.full(A.shape[0], M)
    c = -np.ones(n)  # maximize sum => minimize negative sum
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        # fallback: return uniform maybe
        return np.full(n, S / n)
    b = res.x
    total = b.sum()
    if total <= 0:
        return np.full(n, S / n)
    b_scaled = b * (S / total)
    return b_scaled
```

def _solve_lp_max_sum(f, M, S):
    n = len(f)
    # Build convolution constraints matrix A (2n-1 x n)
    A = _build_convolution_matrix(f)
    b_ub = np.full(A.shape[0], M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)   # minimize -sum => maximize sum
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback uniform
        b = np.full(n, S / n)
    else:
        b = res.x
        total = b.sum()
        if total <= 0:
            b = np.full(n, S / n)
        else:
            b = b * (S / total)
    return b
```

"""
Improved step‑function optimizer with a linear‑programming “mass‑transfer”
stage.

The objective of ``evaluate_sequence`` is the maximal entry of the
self‑convolution a∗a (the sum constraint is normalised to √(2 n)).  The
algorithm consists of the following phases, each respecting the 1000 s
budget:

1. **Random warm‑starts** – draw feasible vectors from several parametric
   families (power‑law, exponential, linear, …) and normalise them to
   ∑a=√(2n).

2. **LP‑based mass transfer** – for a moderate length (≈2000) we repeatedly
   solve the linear programme  

        maximise   ∑b_j
        subject to  (a ∗ b)[k] ≤ M   for all k,
                    b_j ≥ 0,

   where M = max_k (a∗a)[k] of the current sequence a.
   The solution b is scaled to the correct ℓ₁‑norm and mixed with a
   convex combination a←(1‑t)a+t b (t≈0.5).  This provably never increases
   the peak value and usually reduces it.

3. **Length scaling & projected sub‑gradient descent** – the best
   sequence found in (2) is interpolated to larger lengths (up to ≈ 20 000)
   and refined with a classic projected sub‑gradient method on
   f(a)=max(a∗a).  The projection onto the simplex {a≥0,∑a=√(2n)} uses the
   O(n) algorithm of Duchi‑Shalev‑Shwartz‑Singer‑Chandra.

4. **Simulated‑annealing hill‑climb** – a cheap stochastic mass‑transfer
   loop that moves a random fraction of mass between two positions and
   accepts worsening moves with an exponentially decaying temperature.

The final sequence is stored in the global variable ``height_sequence_1``
and returned as a plain list.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags

# Low‑level utilities (FFT convolution, projection, scaling)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a∗a."""
    return _full_convolution_fft(a).max()


def _max_convolution_and_gradient(a: np.ndarray):
    """Sub‑gradient of max_k (a∗a)[k] and the current max value."""
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_val = b.max()
    mask = (b == max_val).astype(float)

    a_rev = a[::-1]
    conv_len = len(mask) + n - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad, max_val


def _project_onto_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Projection onto {w ≥ 0, ∑w = S} (Duchi‑Shalev‑Shwartz algorithm)."""
    if S < 0:
        raise ValueError("Simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Parametric families for random initialisation
# ----------------------------------------------------------------------
def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    raw = mix * ush + (1.0 - mix) * base
    return raw


def _init_sine(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    vals = np.sin(np.pi * (i + 1) / (n + 1))
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * vals + (1.0 - mix) * base
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    return rng.random(n)


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """Generate a feasible start vector (non‑negative, ℓ₁‑norm = √(2 n))."""
    S = np.sqrt(2.0 * n)

    # occasional warm‑start
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
        return a

    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'ushape', 'sine', 'uniform', 'dirichlet'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'sine':
        raw = _init_sine(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Sub‑gradient descent (same as in the previous state‑of‑the‑art)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         global_best: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_val = _max_convolution_and_gradient(a)

        norm_sq = np.dot(grad, grad)
        if norm_sq > 0:
            step = (cur_val - best_val) / (norm_sq + 1e-12)
            step = np.clip(step, 1e-7, 0.2)
        else:
            step = 0.0

        a = _project_onto_simplex(a - step * grad, S)

        cur_val = _max_convolution(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()

        if it % 300 == 0:
            jitter = rng.normal(scale=0.001 * np.mean(a), size=n)
            a = _project_onto_simplex(a + jitter, S)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (same as before)
# ----------------------------------------------------------------------
def _anneal_local_search(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    cur_val = _max_convolution(a)
    best_a = a.copy()
    best_val = cur_val

    T = 0.05 * cur_val
    decay = 0.9999
    it = 0
    while time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        frac = rng.uniform(0.02, 0.18)
        delta = a[i] * frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _max_convolution(a_new)
        delta_val = new_val - cur_val
        if delta_val < 0 or rng.random() < np.exp(-delta_val / (T + 1e-12)):
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()

        T *= decay
        if it % 500 == 0:
            jitter = rng.normal(scale=0.001 * S, size=n)
            a = _project_onto_simplex(a + jitter, S)
            cur_val = _max_convolution(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑programming mass‑transfer stage
# ----------------------------------------------------------------------
def _build_convolution_matrix(f: np.ndarray):
    """
    Return a sparse matrix A such that A @ b = f ∗ b (full linear
    convolution) where ``f`` is fixed and ``b`` is the variable.
    """
    n = f.shape[0]
    offsets = np.arange(n)                # 0 … n‑1
    # each diagonal consists of a constant f[i]
    data = [np.full(n, fi, dtype=np.float64) for fi in f]
    A = diags(data, offsets, shape=(2 * n - 1, n), format='csr')
    return A


def _solve_lp_max_sum(f: np.ndarray, M: float, S: float):
    """
    Solve  max ∑b_j   s.t. (f∗b)[k] ≤ M  (∀k), b ≥ 0.
    The returned vector is scaled to have ℓ₁‑norm S.
    """
    n = len(f)
    A = _build_convolution_matrix(f)
    b_ub = np.full(A.shape[0], M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)   # minimise –∑b  ⇒ maximise ∑b
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    del A  # free memory early

    if not res.success:
        # fallback: uniform distribution
        b = np.full(n, S / n)
    else:
        b = res.x
        total = b.sum()
        if total <= 0:
            b = np.full(n, S / n)
        else:
            b = b * (S / total)
    return b


def _lp_refine(a0: np.ndarray, deadline: float, rng: np.random.Generator):
    """
    Repeatedly solve the LP described above and mix the resulting vector
    with the current iterate.  Returns the best vector seen and its peak
    value.
    """
    a = a0.copy()
    n = len(a)
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1
        M = _max_convolution(a)

        # solve LP for the current peak bound M
        g = _solve_lp_max_sum(a, M, S)

        # convex combination; start with a fairly aggressive mixing
        t = 0.5
        a_new = (1 - t) * a + t * g
        val = _max_convolution(a_new)

        if val + 1e-12 < best_val:
            best_val = val
            best_a = a_new.copy()
            print(f"[lp] iter={it:3d}  improved={best_val:.6f}")

        # accept only if we made progress; otherwise try a smaller step
        if val < M:
            a = a_new
        else:
            t = 0.2
            a_new = (1 - t) * a + t * g
            val = _max_convolution(a_new)
            if val + 1e-12 < best_val:
                best_val = val
                best_a = a_new.copy()
                print(f"[lp] iter={it:3d} (small t) improved={best_val:.6f}")
            if val < M:
                a = a_new
            else:
                # jitter to escape a possible plateau
                jitter = rng.normal(scale=0.001 * S, size=n)
                a = _project_onto_simplex(a + jitter, S)

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises the provided
    ``evaluate_sequence``.  The routine first runs an LP‑based mass‑transfer
    stage on a moderate length, then scales the result to larger lengths
    and applies projected sub‑gradient descent followed by a short
    simulated‑annealing refinement.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # Warm‑start handling
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    best_seq = None
    best_score = np.inf
    best_n = None
    attempts = 0

    # ------------------------------------------------------------------
    # Phase 1 – LP‑based refinement on a moderate length
    # ------------------------------------------------------------------
    n0 = int(rng.integers(1800, 2600))
    phase1_deadline = time.time() + min(180.0, (end_time - time.time()) * 0.2)

    while time.time() < phase1_deadline:
        attempts += 1
        a0 = _initial_vector(n0, rng, warm_seq)
        # give each LP run a small slice of time (≈15 s)
        slice_deadline = time.time() + min(15.0, phase1_deadline - time.time())
        a_opt, val = _lp_refine(a0, slice_deadline, rng)
        if val < best_score:
            best_score = val
            best_seq = a_opt.copy()
            best_n = n0
            print(f"[phase1] attempt={attempts:4d}  new best={best_score:.6f}  n={n0}")

    # ------------------------------------------------------------------
    # Phase 2 – scaling & sub‑gradient descent on larger lengths
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback to a simple uniform vector
        n = 2500
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = _max_convolution(best_seq)
        best_n = n

    # pick a few larger target lengths (geometric progression)
    lengths = [best_n]
    for factor in [1.5, 2.0, 3.0]:
        cand = int(best_n * factor)
        if 3000 <= cand <= 20000:
            lengths.append(cand)
    lengths = sorted(set(lengths))

    for target_n in lengths:
        if time.time() >= end_time - 60:
            break
        # interpolate current best to the new length
        a_scaled = _scale_sequence(best_seq, target_n)
        S = np.sqrt(2.0 * target_n)
        a_scaled = np.clip(a_scaled, 0.0, None)
        a_scaled = a_scaled * (S / a_scaled.sum())

        # allocate a time slice for sub‑gradient refinement
        remaining = end_time - time.time()
        slice_time = min(remaining * 0.2, 80.0)
        a_opt, val = _subgradient_descent(a_scaled,
                                          time.time() + slice_time,
                                          rng,
                                          best_score)
        if val < best_score:
            best_score = val
            best_seq = a_opt.copy()
            best_n = target_n
            print(f"[subgrad] length={target_n:5d}  improved={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing hill‑climb (final polishing)
    # ------------------------------------------------------------------
    if time.time() < end_time - 5.0:
        remaining = end_time - time.time()
        a_opt, val = _anneal_local_search(best_seq,
                                          time.time() + remaining,
                                          rng)
        if val < best_score:
            best_score = val
            best_seq = a_opt.copy()
            print(f"[anneal] final improvement={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f}  internal={best_score:.6f}  length={len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
