def _smooth_gradient_descent(a0, deadline, rng, best_val_global):
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_val = _max_convolution(a)  # actual max

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    epsilon_adam = 1e-8
    lr = 0.2

    # smoothing schedule
    max_val = _max_convolution(a)
    eps = max_val * 0.5  # initial smoothing
    eps_decay = 0.9995

    while time.time() < deadline:
        # compute smooth grad and smooth objective
        grad, smooth_val = _smooth_max_and_gradient(a, eps)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1)
        v_hat = v / (1 - beta2)
        step = lr * m_hat / (np.sqrt(v_hat) + epsilon_adam)

        # update a
        a = _project_onto_simplex(a - step, S)

        # evaluate true max
        cur_val = _max_convolution(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()
            # reduce eps more aggressively if improvement
        # decay eps
        eps = max(eps * eps_decay, 0.001)  # lower bound

        # occasional jitter
        if rng.random() < 0.01:
            jitter = rng.normal(scale=0.001 * S, size=n)
            a = _project_onto_simplex(a + jitter, S)

    return best_a, best_val

"""
Improved step‑function optimizer.

Goal: minimise  max_k (a∗a)[k]  with a_i ≥ 0 and  Σ a_i = √(2 n).
The evaluator returns exactly this maximum, so we minimise the
peak of the self‑convolution.

Approach
--------
1. **Broad random search** – generate many candidates from several
   parametric families (power‑law, exponential, linear, U‑shape,
   sine, Dirichlet, etc.) for lengths 5 000–25 000.  Keep the best few
   according to the true max‑convolution value.

2. **Smoothed gradient descent** – minimise a smooth approximation of
   the max using the log‑sum‑exp trick:
        M_ε(a) = ε·log ∑_k exp((a∗a)[k]/ε) .
   The gradient is 2·conv(w, rev(a)) where the weights
   w_k = exp((a∗a)[k]/ε)/∑_j exp((a∗a)[j]/ε) .
   An Adam‑style optimiser with projection onto the ℓ₁‑simplex
   (radius √(2 n)) drives the search.  ε is decreased
   exponentially so the optimisation gradually focuses on the true peak.

3. **Length scaling** – the best sequence is interpolated to a larger
   length (up to 25 000) and refined again with the same smoothed
   optimiser.

4. **Simulated‑annealing hill‑climb** – a cheap mass‑transfer move
   improves any remaining local irregularities.

The whole routine respects the 1000 s budget and stores the final
construction in the global variable ``height_sequence_1``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT, simplex projection, scaling)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a∗a."""
    return float(_full_convolution_fft(a).max())


def _project_onto_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex
        { w | w_i ≥ 0 , ∑ w_i = S }.
    Implements the O(n) algorithm of Duchi et al.
    """
    if S < 0:
        raise ValueError("Simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Random initialisers (families)
# ----------------------------------------------------------------------
def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    raw = mix * ush + (1.0 - mix) * base
    return raw


def _init_sine(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    vals = np.sin(np.pi * (i + 1) / (n + 1))
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * vals + (1.0 - mix) * base
    return raw


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    # concentration <1 encourages sparsity – sample a random alpha
    alpha = rng.uniform(0.1, 1.0)
    sample = rng.gamma(alpha, 1.0, size=n)
    return sample


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Occasionally warm‑starts from the globally best sequence and otherwise
    draws from a random parametric family.
    """
    S = np.sqrt(2.0 * n)

    # Warm start with probability ≈0.2
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
        return a

    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'ushape', 'sine', 'dirichlet', 'uniform'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'sine':
        raw = _init_sine(n, rng)
    elif style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    else:
        raw = _init_uniform(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Smoothed max + gradient (log‑sum‑exp) utilities
# ----------------------------------------------------------------------
def _smooth_max_and_gradient(a: np.ndarray, eps: float):
    """
    Returns (grad, smooth_val) where grad is the gradient of the
    smoothed maximum ε·log∑exp((a∗a)[k]/ε) and smooth_val is that value.
    """
    n = a.shape[0]

    # full convolution a∗a
    b = _full_convolution_fft(a)          # length 2n‑1

    # log‑sum‑exp in a numerically stable way
    scaled = b / eps
    max_scaled = np.max(scaled)
    exp_diff = np.exp(scaled - max_scaled)
    sum_exp = np.sum(exp_diff)
    lse = max_scaled + np.log(sum_exp)    # = log∑exp(b/ε)
    smooth_val = eps * lse

    # normalized weights w_k = exp(b_k/ε) / ∑exp(b/ε)
    w = exp_diff / sum_exp                # size 2n‑1, sums to 1

    # gradient = 2·conv(w, rev(a)), central part of length n
    a_rev = a[::-1]
    conv_len = len(w) + n - 1             # = 3n‑2
    f_len = _next_pow2(conv_len)
    W_fft = np.fft.rfft(w, n=f_len)
    A_fft = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(W_fft * A_fft, n=f_len)[:conv_len]
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad, smooth_val


def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             rng: np.random.Generator,
                             global_best: float) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimiser on the smoothed objective.
    Returns the best sequence seen and its *true* max‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _max_convolution(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.2

    # smoothing parameter – start relatively large, then decay
    cur_max = best_val
    eps = max(cur_max * 0.5, 1e-3)
    eps_decay = 0.9995

    it = 0
    while time.time() < deadline:
        it += 1

        grad, smooth_val = _smooth_max_and_gradient(a, eps)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # (optional) clip step size to avoid gigantic jumps
        max_step = 5.0 * S
        step_norm = np.linalg.norm(step)
        if step_norm > max_step:
            step = step * (max_step / step_norm)

        a = _project_onto_simplex(a - step, S)

        cur_val = _max_convolution(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()
            eps = max(eps * 0.9, 1e-3)      # accelerate smoothing decay after improvement

        # decay epsilon even if no improvement
        eps = max(eps * eps_decay, 1e-3)

        # occasional random jitter to escape tiny plateaux
        if it % 250 == 0 and rng.random() < 0.5:
            jitter = rng.normal(scale=0.001 * S, size=n)
            a = _project_onto_simplex(a + jitter, S)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb (post‑processing)
# ----------------------------------------------------------------------
def _anneal_local_search(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Mass‑transfer annealing: move a random fraction of mass from i to j.
    Accept worsening moves with a temperature that decays exponentially.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    cur_val = _max_convolution(a)
    best_a = a.copy()
    best_val = cur_val

    T = 0.05 * cur_val           # start temperature (~5 % of the peak)
    decay = 0.9999

    it = 0
    while time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        # transfer a random fraction of the mass at i to j
        frac = rng.uniform(0.02, 0.18)
        delta = a[i] * frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _max_convolution(a_new)
        delta_val = new_val - cur_val
        if delta_val < 0 or rng.random() < np.exp(-delta_val / (T + 1e-12)):
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()

        T *= decay
        if it % 500 == 0:
            jitter = rng.normal(scale=0.001 * S, size=n)
            a = _project_onto_simplex(a + jitter, S)
            cur_val = _max_convolution(a)

    return best_a, best_val

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    if 'height_sequence_1' in globals():

        except Exception:
            warm_seq = None

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    K = 8                                         # keep up to 8 best
    attempts = 0
    while time.time() < explore_deadline:
        attempts += 1
