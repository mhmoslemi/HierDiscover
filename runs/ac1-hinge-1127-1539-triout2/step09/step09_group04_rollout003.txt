def _local_search(a, target_sum, deadline):
    a = a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)
    cur_val = best_val
    cur_a = a.copy()
    cur_conv = _conv_fft(cur_a)
    # temperature schedule initial based on best_val; e.g., T0 = best_val * 0.1
    start_temp = best_val * 0.1
    min_temp = 1e-6
    # total steps approximate: maybe 200k iterations fits time.
    while time.time() < deadline:
        # update temperature based on elapsed fraction
        fraction = (time.time() - start) / (deadline - start)
        temp = start_temp * (1 - fraction) + min_temp * fraction

        # find max shift
        k_max = int(np.argmax(cur_conv))
        # donor candidate: index that contributes max to this conv entry
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        contributions = cur_a[idx] * cur_a[partner]
        donor = idx[np.argmax(contributions)]

        # receiver: index with smallest a (could be zero) but not same as donor
        # optionally choose random low a
        # choose low_idx among indices with low a less than median
        # or random
        # Here choose random index weighted by inverse a?
        candidate_indices = np.arange(n)
        # To avoid selecting donor
        cand = candidate_indices[candidate_indices != donor]
        # pick receiver with probability proportional to 1/(a+eps)
        weights = 1.0/(cur_a[cand] + 1e-12)
        weights = weights / weights.sum()
        receiver = np.random.choice(cand, p=weights)

        # choose delta
        max_delta = min(cur_a[donor] * 0.5, target_sum * 0.02)  # 0.5 fraction or 2% of total
        delta = np.random.random() * max_delta

        # move delta
        cur_a[donor] -= delta
        cur_a[receiver] += delta

        # recompute conv
        new_conv = _conv_fft(cur_a)
        new_val = new_conv.max()

        # acceptance
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val-cur_val)/temp):
            cur_val = new_val
            cur_conv = new_conv
            # update best
            if new_val < best_val:
                best_val = new_val
                best_a = cur_a.copy()
                print(f"[{time.time():.2f}] Local search new best {best_val:.6f}")
        else:
            # revert
            cur_a[donor] += delta
            cur_a[receiver] -= delta

        # optionally small jitter
        if time.time() - start > (deadline - start) * 0.2 and np.random.rand() < 0.001:
            jitter = np.random.normal(scale=target_sum * 0.001, size=n)
            cur_a = np.maximum(cur_a + jitter, 0)

def _conv_fft(a):
    n = a.shape[0]
    size = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    return conv

def _simulated_annealing_search(a0, target_sum, deadline):
    a = a0.copy()
    n = a.shape[0]
    # compute current conv and score
    conv = _conv_fft(a)
    cur_val = conv.max()
    best_seq = a.copy()
    best_val = cur_val
    start = time.time()
    rng = np.random.default_rng()
    # annealing schedule: temperature start as cur_val * 0.05 (or 0.1) etc.
    start_temp = cur_val * 0.1
    min_temp = cur_val * 1e-4
    while time.time() < deadline:
        # compute temperature based on elapsed proportion
        t_factor = (time.time() - start) / (deadline - start)
        temp = start_temp * (1 - t_factor) + min_temp * t_factor

        # Choose donor (largest contribution to max shift) and receiver (low a)
        k_max = int(np.argmax(conv))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        contributions = a[idx] * a[partner]
        donor = idx[np.argmax(contributions)]

        # receiver selection
        # pick index from low tail of a values
        # compute 10th percentile value
        thr = np.percentile(a, 10)
        low_idxs = np.where(a <= thr)[0]
        if low_idxs.size == 0:
            receiver = rng.integers(n)
            while receiver == donor:
                receiver = rng.integers(n)
        else:
            # choose random among low_idxs not donor
            if low_idxs.size == 1 and low_idxs[0] == donor:
                receiver = rng.integers(n)
                while receiver == donor:
                    receiver = rng.integers(n)
            else:
                # exclude donor if present
                low_idxs = low_idxs[low_idxs != donor]
                if low_idxs.size == 0:
                    receiver = rng.integers(n)
                    while receiver == donor:
                        receiver = rng.integers(n)
                else:
                    receiver = rng.choice(low_idxs)

        # delta magnitude
        max_delta = min(a[donor] * 0.5, target_sum * 0.02)  # up to 50% donor or 2% total
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # move
        a[donor] -= delta
        a[receiver] += delta

        new_conv = _conv_fft(a)
        new_val = new_conv.max()

        # acceptance test
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(temp, 1e-12)):
            cur_val = new_val
            conv = new_conv
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[{time.time():.2f}] SA new best {best_val:.6f}")
        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta

"""
Propose a step‑function (non‑negative sequence) that minimizes
    2·n·max(a*a) / (sum(a)^2)

Because sum(a)=√(2n) is enforced, the objective equals the maximal
convolution entry itself.  The algorithm combines several complementary
heuristics:

1. Quick parametric warm‑start on a reduced size (grid over a
   power‑law × exponential family, then linear‑interpolate).

2. Multiplicative (mirror) descent on a soft‑max surrogate of the
   maximum convolution entry.

3. Projected gradient descent with momentum and random restarts.

4. Projected sub‑gradient descent on the exact maximal entry.

5. Simulated‑annealing style local search: repeatedly transfer a small
   amount of mass from a “donor’’ index that contributes most to the
   current maximal convolution entry to a “receiver’’ with low mass,
   accepting uphill moves with a temperature schedule.

All stages respect the ℓ¹‑simplex (non‑negativity, fixed sum) and the
time budget.  The best sequence found is returned as a plain Python
list and stored globally as ``height_sequence_1`` for warm‑starts in
subsequent runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution / scoring utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) shape, normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Surrogate gradient (soft‑max of convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where conv is the full convolution a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * (reverse(a) convolve p)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Stage 1 – cheap parametric search on a reduced scale
# ----------------------------------------------------------------------


def _grid_param_search(small_n: int,
                       target_sum: float,
                       deadline: float) -> tuple[np.ndarray | None, float]:
    """Search a two‑parameter family a_i = (i+1)^(-p)·exp(-λ·i/small_n)."""
    p_vals = np.linspace(0.2, 1.2, 17)
    lam_vals = np.linspace(0.0, 6.0, 13)

    best_seq = None
    best_score = np.inf
    i = np.arange(small_n, dtype=np.float64)

    for p in p_vals:
        for lam in lam_vals:
            if time.time() > deadline:
                break
            seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
            seq = seq * (target_sum / seq.sum())
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()
                print(f"[{time.time():.2f}] Grid best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")
        else:
            continue
        break

    rng = np.random.default_rng()
    while time.time() < deadline:
        p = rng.uniform(0.2, 1.2)
        lam = rng.exponential(1.0) * 3.0
        seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
        seq = seq * (target_sum / seq.sum())
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Random best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stage 2 – exponential (multiplicative) descent
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Mirror‑descent with exponential updates using a soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 16384.0
    base_eta = 0.04
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 3 – projected gradient descent with restarts
# ----------------------------------------------------------------------


def _annealed_gradient_descent_v2(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the soft‑max surrogate."""
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 16384.0
    step_base = target_sum * 0.5
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.95
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.002 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GDv2 it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several short GD restarts with random jitter; keep the best."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_v2(start,
                                               time.time() + run_budget,
                                               target_sum)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 4 – exact sub‑gradient descent on the max‑convolution
# ----------------------------------------------------------------------


def _subgradient_max_descent(a0: np.ndarray,
                             target_sum: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true maximal convolution entry."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())

        # sub‑gradient of conv_{k_max}: 2·a[k_max‑i] on the feasible indices
        grad = np.zeros_like(a)
        lo = max(0, k_max - (a.shape[0] - 1))
        hi = min(k_max, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        grad[idx] = 2.0 * a[partner]

        norm = np.linalg.norm(grad)
        if norm == 0:
            break

        # safe step size to stay non‑negative before projection
        safe_eta = np.min(np.where(grad > 0, a / grad, np.inf))
        eta = min(0.5 * safe_eta, 0.5 / np.sqrt(it + 1))

        a = a - eta * grad
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it} score={best_val:.6f}")

        # occasional jitter to escape flat regions
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 5 – simulated‑annealing local search
# ----------------------------------------------------------------------


def _simulated_annealing_search(a0: np.ndarray,
                                target_sum: float,
                                deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple SA / mass‑transfer search:
    repeatedly move a small amount of mass from a donor (large contribution
    to the current maximal entry) to a low‑mass receiver.  Moves that
    increase the max are accepted with a temperature that decays linearly.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_fft(a)
    cur_val = conv.max()
    best_seq = a.copy()
    best_val = cur_val
    start = time.time()
    rng = np.random.default_rng()

    # temperature schedule (linear decay)
    start_temp = cur_val * 0.10
    min_temp = cur_val * 1e-5

    while time.time() < deadline:
        # compute temperature
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        temp = start_temp * (1 - frac) + min_temp * frac

        # current maximal shift
        k_max = int(np.argmax(conv))

        # indices that actually contribute to the maximal entry
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        contrib = a[idx] * a[partner]

        # donor = index with largest contribution
        donor = idx[np.argmax(contrib)]

        # receiver = an index with low mass (10‑percentile) excluding donor
        low_thr = np.percentile(a, 10)
        low_candidates = np.where(a <= low_thr)[0]
        low_candidates = low_candidates[low_candidates != donor]
        if low_candidates.size == 0:
            receiver = rng.integers(n)
            while receiver == donor:
                receiver = rng.integers(n)
        else:
            receiver = rng.choice(low_candidates)

        # amount to move (up to 50 % of donor or 2 % of total mass)
        max_delta = min(a[donor] * 0.5, target_sum * 0.02)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # perform the transfer
        a[donor] -= delta
        a[receiver] += delta

        # evaluate new max convolution
        new_conv = _conv_fft(a)
        new_val = new_conv.max()

        # accept if better or with Metropolis probability
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(temp, 1e-12)):
            cur_val = new_val
            conv = new_conv
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[{time.time():.2f}] SA new best {best_val:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta

        # occasional global jitter to avoid stagnation
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
            conv = _conv_fft(a)
            cur_val = conv.max()

    return best_seq, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose problem size
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Initial sequence (warm‑start from previous run if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"],
                                        dtype=np.float64), n)
        init = init * (target_sum / init.sum())
    else:
        init = _power_law_sequence(n, 0.55, target_sum)

    best_seq = init.copy()
    best_val = _score(best_seq)

    # ------------------------------------------------------------------
    # Stage 1 – cheap parametric search on a reduced scale
    # ------------------------------------------------------------------
    small_n = max(2000, n // 4)
    small_target = np.sqrt(2.0 * small_n)
    param_deadline = time.time() + min(12.0,
                                      (deadline - time.time()) * 0.07)
    small_seq, _ = _grid_param_search(small_n, small_target, param_deadline)
    if small_seq is not None:
        init = _resize_sequence(small_seq, n)
        init = init * (target_sum / init.sum())
        val = _score(init)
        if val < best_val:
            best_val = val
            best_seq = init.copy()
            print(f"[{time.time():.2f}] Parametric warm‑start improved best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – exponential (multiplicative) descent (~10 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        exp_deadline = time.time() + (deadline - time.time()) * 0.10
        a_exp, val_exp = _exponential_descent(best_seq, target_sum, exp_deadline)
        if val_exp < best_val:
            best_val = val_exp
            best_seq = a_exp.copy()
            print(f"[{time.time():.2f}] Exp descent improved best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – projected gradient descent with restarts (~25 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        gd_deadline = time.time() + (deadline - time.time()) * 0.25
        a_gd, val_gd = _gradient_descent_with_restarts(best_seq,
                                                       target_sum,
                                                       gd_deadline)
        if val_gd < best_val:
            best_val = val_gd
            best_seq = a_gd.copy()
            print(f"[{time.time():.2f}] GD restarts improved best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – sub‑gradient descent on exact max (~15 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + (deadline - time.time()) * 0.15
        a_sub, val_sub = _subgradient_max_descent(best_seq,
                                                  target_sum,
                                                  sub_deadline)
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑gradient improved best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – simulated‑annealing local search (remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_sa, val_sa = _simulated_annealing_search(best_seq,
                                                   target_sum,
                                                   deadline)
        if val_sa < best_val:
            best_val = val_sa
            best_seq = a_sa.copy()
            print(f"[{time.time():.2f}] Simulated annealing improved best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Clean up and final evaluation
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # store for possible warm‑starts in a subsequent call
    globals()["height_sequence_1"] = final_seq

    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    return final_seq
