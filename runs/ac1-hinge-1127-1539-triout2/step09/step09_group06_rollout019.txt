"""
Hybrid optimizer for the step‑function minimisation problem.

Strategy
--------
1. Warm‑start from the previous solution (if any) and down‑scale it to a
   modest size (≈2500) – this keeps the LP step cheap.
2. Fast random exploration of a few parametric families to obtain a
   decent start.
3. Refine the best start with many independent runs of the Adam optimiser
   (log‑softmax parametrisation on the simplex).  The runs are short and
   diverse (different noise seeds) to escape local minima.
4. A short sub‑gradient descent on the *hard* maximum gives a quick
   polishing.
5. One (or two) LP‑based “exchange” steps (see the description above):
   maximise the ℓ₁‑norm of a vector `b` under the linear constraints
   `f * b ≤ M`.  The resulting `b` is rescaled to the required ℓ₁‑norm
   and mixed with the current iterate.  This step provably never
   increases the peak and often yields a noticeable drop.
6. A cheap stochastic hill‑climb finishes the search.
7. The final sequence is returned (and stored in ``height_sequence_1``).

All steps respect the 1000 s wall‑time budget and keep every entry
non‑negative while fixing the ℓ₁‑norm to `sqrt(2·n)`.  The score
returned by `evaluate_sequence` is exactly the maximum entry of the
self‑convolution after normalisation.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy import sparse

# ----------------------------------------------------------------------
# low‑level convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a∗a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a∗a."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# initial‑vector generators (various parametric families)
# ----------------------------------------------------------------------
def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    return mix * edge + (1.0 - mix) * base


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    return mix * exp_part + (1.0 - mix) * base


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    return mix * lin + (1.0 - mix) * base


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    return mix * ush + (1.0 - mix) * base


def _init_cosine(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    raw = 1.0 - np.cos(np.pi * i / (n - 1))
    raw = raw * rng.uniform(0.5, 2.0)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * raw + (1.0 - mix) * base


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    return rng.random(n)


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.05, 0.9)
    return rng.gamma(shape, 1.0, size=n)


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Approximately 20 % of the time we warm‑start from `warm_seq`,
    otherwise we draw from one of the parametric families.
    The returned vector satisfies sum = sqrt(2·n) and is non‑negative.
    """
    S = np.sqrt(2.0 * n)

    if warm_seq is not None and rng.random() < 0.23:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
        return a

    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'gamma', 'uniform', 'dirichlet', 'ushape', 'cosine'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'cosine':
        raw = _init_cosine(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # grad = 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex (log‑softmax parametrisation).
    Returns the best sequence found and its *hard* peak value.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                     # required ℓ₁‑norm

    # initialise logits from a feasible a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                         # logits (log‑softmax)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.45

    # temperature start – a few units of smoothing
    tau = max(_full_convolution_fft(a0).max(), 5.0)

    best_a = a0.copy()
    best_peak = _max_convolution(a0)
    it = 0

    while time.time() < deadline:
        it += 1

        # recover a from logits (log‑softmax)
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # gradient (τ‑soft‑max)
        g, peak = _softmax_gradient(a, tau)

        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # chain rule to logits
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 6.0 * S:
            grad_p = grad_p * (6.0 * S / clip_val)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # schedules – very gentle decay
        tau *= 0.99997
        lr *= 0.99997

        # occasional jitter
        if it % 900 == 0:
            p += rng.normal(scale=0.005, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Sub‑gradient descent on the *hard* max (no smoothing)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple sub‑gradient descent that follows the entry attaining the
    current maximum, with projection onto the simplex.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    a = a0.copy()
    best_a = a.copy()
    best_val = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1

        b = _full_convolution_fft(a)
        k = int(b.argmax())
        val = b[k]

        if val < best_val:
            best_val = val
            best_a = a.copy()

        # sub‑gradient of (a*a)_k  :  g_i = 2 * a_{k-i} when defined
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        jdx = k - idx
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[jdx]

        step = 0.9 * S / (np.sqrt(it) + 1.0)
        a = a - step * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, S / n)
        else:
            a = a * (S / s)

        # occasional random jitter
        if it % 600 == 0:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                frac = rng.uniform(0.015, 0.08)
                amount = a[i] * frac
                a[i] -= amount
                a[j] += amount
                a = a * (S / a.sum())

    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑programming polishing step (exchange algorithm)
# ----------------------------------------------------------------------
def _lp_polish(a0: np.ndarray,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 4) -> tuple[np.ndarray, float]:
    """
    Perform a few LP‑based exchange steps.
    For the current iterate f we solve:
        maximise   ∑ b_j
        s.t.       f * b ≤ M   (M = max_k (f*f)_k)
                    b_j ≥ 0
    The solution b is rescaled to the required ℓ₁‑norm and mixed with f.
    The mixing coefficient is chosen by a tiny line‑search.
    """
    a = a0.copy()
    n = len(a)
    S = np.sqrt(2.0 * n)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        M = _max_convolution(a)

        # build sparse matrix representing convolution with fixed a
        # rows correspond to k = i + j, cols to j
        i_idx = np.arange(n, dtype=np.int32)[:, None]          # shape (n,1)
        j_idx = np.arange(n, dtype=np.int32)[None, :]          # shape (1,n)
        rows = (i_idx + j_idx).ravel()                        # length n*n
        cols = np.tile(np.arange(n, dtype=np.int32), n)        # length n*n
        data = np.repeat(a, n)                                # length n*n
        A = sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

        # linear programme: maximise sum(b)  ⇔ minimise -∑b
        c = -np.ones(n)
        ub = np.full(2 * n - 1, M)

        res = linprog(c, A_ub=A, b_ub=ub, bounds=(0, None),
                      method='highs', options={'presolve': True})
        if not res.success:
            # if LP fails, break early
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0.0:
            break

        g = b * (S / sum_b)   # rescale to the correct ℓ₁‑norm

        # line‑search over a few mixing coefficients
        best_t = 0.0
        best_val = M
        best_a = a
        for t in (0.2, 0.4, 0.6, 0.8):
            cand = (1.0 - t) * a + t * g
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s == 0.0:
                continue
            cand = cand * (S / s)
            val = _max_convolution(cand)
            if val < best_val:
                best_val = val
                best_t = t
                best_a = cand

        if best_t == 0.0:
            # no improvement – stop early
            break
        a = best_a

        # small random jitter to avoid stagnation
        if rng.random() < 0.12:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                frac = rng.uniform(0.01, 0.06)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = a * (S / a.sum())

    return a, _max_convolution(a)


# ----------------------------------------------------------------------
# Cheap stochastic hill‑climb (final polishing)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_val = _max_convolution(a)
    n = len(a)

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        frac = rng.uniform(0.015, 0.10)
        amount = a[i] * frac
        a[i] -= amount
        a[j] += amount
        val = _max_convolution(a)
        if val + 1e-12 < best_val:
            best_val = val
            best_a = a.copy()
        else:
            # reject – revert
            a[i] += amount
            a[j] -= amount

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence as a plain Python list
    and stores it in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    end_time = start_time + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # Warm‑start handling
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # ------------------------------------------------------------------
    # Choose a modest target length (fast LP)
    # ------------------------------------------------------------------
    target_n = 2500
    if warm_seq is not None and warm_seq.shape[0] >= target_n:
        # downscale if we have a longer warm‑start
        warm_seq = _scale_sequence(warm_seq, target_n)

    # ------------------------------------------------------------------
    # Phase 0 – quick random exploration to get a decent start
    # ------------------------------------------------------------------
    explore_deadline = start_time + min(70.0, (end_time - start_time) * 0.07)
    best_seq = None
    best_score = np.inf
    attempts = 0

    while time.time() < explore_deadline:
        attempts += 1
        a0 = _initial_vector(target_n, rng, warm_seq)
        score = _max_convolution(a0)
        if score < best_score:
            best_score = score
            best_seq = a0.copy()
            print(f"[explore] attempt={attempts:4d} score={score:.6f}")

    if best_seq is None:                     # fallback (should not happen)
        best_seq = np.full(target_n, np.sqrt(2.0 * target_n) / target_n)
        best_score = _max_convolution(best_seq)

    # ------------------------------------------------------------------
    # Phase 1 – several independent Adam runs (diverse restarts)
    # ------------------------------------------------------------------
    remaining = end_time - time.time()
    if remaining > 120.0:
        # allocate ~30 s per short restart, remainder for a long final run
        short_run = 30.0
        n_restart = max(1, int((remaining - 80.0) // short_run))
        per_run = min(short_run, (remaining - 80.0) / n_restart)

        for r in range(n_restart):
            if time.time() > end_time - 80.0:
                break
            # jitter the start
            noise = rng.normal(scale=0.015 * np.sqrt(2.0 * target_n) / target_n,
                               size=target_n)
            a_start = np.maximum(best_seq + noise, 0.0)
            a_start = a_start * (np.sqrt(2.0 * target_n) / a_start.sum())
            a_opt, peak = _adam_optimize(a_start, time.time() + per_run, rng)
            if peak < best_score:
                best_score = peak
                best_seq = a_opt.copy()
                print(f"[adam] restart={r} new best={peak:.6f}")

        # final longer Adam run (uses the currently best sequence)
        if time.time() < end_time - 80.0:
            a_opt, peak = _adam_optimize(best_seq,
                                          time.time() + (remaining - 80.0) / 2,
                                          rng)
            if peak < best_score:
                best_score = peak
                best_seq = a_opt.copy()
                print(f"[adam] final run best={peak:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – sub‑gradient descent (hard max)
    # ------------------------------------------------------------------
    remaining = end_time - time.time() - 40.0
    if remaining > 20.0:
        a_opt, peak = _subgradient_descent(best_seq,
                                           time.time() + remaining,
                                           rng)
        if peak < best_score:
            best_score = peak
            best_seq = a_opt.copy()
            print(f"[subgrad] after sub‑grad best={peak:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – LP polishing (exchange algorithm)
    # ------------------------------------------------------------------
    remaining = end_time - time.time() - 20.0
    if remaining > 10.0:
        a_opt, peak = _lp_polish(best_seq,
                                 time.time() + remaining,
                                 rng,
                                 max_iter=5)
        if peak < best_score:
            best_score = peak
            best_seq = a_opt.copy()
            print(f"[lp] polished best={peak:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – cheap hill‑climb
    # ------------------------------------------------------------------
    remaining = end_time - time.time() - 5.0
    if remaining > 5.0:
        a_opt, peak = _hill_climb(best_seq,
                                  time.time() + remaining,
                                  rng)
        if peak < best_score:
            best_score = peak
            best_seq = a_opt.copy()
            print(f"[hill] final hill‑climb best={peak:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f}  internal max={best_score:.6f}  length={len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
