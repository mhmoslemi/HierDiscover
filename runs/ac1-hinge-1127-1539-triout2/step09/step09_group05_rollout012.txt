"""
Improved step‑function optimizer (v2).

Goal: minimise   2·n·max(a∗a) / (∑a)²   with a_i ≥ 0.
The evaluation function returns exactly this quantity, i.e. the peak of the
full linear self‑convolution when the vector is normalised so that
∑a = √(2 n).

The optimiser works in several stages:

1. **Random exploration** – many candidates of varying lengths (3 000–25 000)
   are drawn from a diverse set of parametric families (power‑law,
   exponential, linear, U‑shape, sinusoid, Dirichlet, random‑walk and the
   uniform vector).  The best few (by the hard convolution peak) are kept
   as seeds.

2. **Smoothed descent** – each seed is refined by projected gradient descent
   on a *soft‑max* approximation of the peak:
        f_ε(a)=ε·log(∑_k exp((a∗a)_k/ε)).
   The gradient of f_ε is a convolution of the soft‑max weights with the
   reversed vector.  A Polyak step size
        α = (current_hard_max – best_hard_max) / ‖g‖²
   is used, followed by Euclidean projection onto the simplex
   {a≥0, ∑a = √(2 n)}.  The smoothing parameter ε is gradually reduced,
   allowing the optimiser to first move in a smooth landscape and then
   concentrate on the true max.

3. **Fine‑grained refinement** – the current global best is further
   polished with a few rounds of the classic hard‑max sub‑gradient method
   (the same technique used in the previous version) and a short
   simulated‑annealing hill‑climb.

4. **Length scaling** – the best construction is linearly interpolated to
   a larger length (up to 25 000) and the whole optimisation pipeline is
   repeated, giving the optimiser more degrees of freedom.

All stages respect the 1000 s global budget.  The final sequence is stored
in the global variable ``height_sequence_1`` and returned as a plain list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT, convolution, simplex projection, scaling)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a∗a."""
    return _full_convolution_fft(a).max()


def _project_onto_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex
        { w | w_i ≥ 0 , ∑ w_i = S }.
    Implements the O(n) algorithm of Duchi et al.
    """
    if S < 0:
        raise ValueError("Simplex sum must be non‑negative")
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]

def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    raw = mix * ush + (1.0 - mix) * base
    return raw


def _init_sine(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    vals = np.sin(np.pi * (i + 1) / (n + 1))
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * vals + (1.0 - mix) * base
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    # draw from a flat Dirichlet (gamma(1,1) i.i.d.)
    return rng.random(n)


def _init_random_walk(n: int, rng: np.random.Generator) -> np.ndarray:
    """A positively shifted random walk – creates a mildly smooth shape."""
    inc = rng.normal(loc=0.0, scale=1.0, size=n)
    seq = np.cumsum(inc)
    seq = seq - seq.min() + 1e-3   # shift to be strictly positive
    return seq


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    20 % of the time it is a (scaled) warm‑start from the current global best,
    otherwise it is sampled from a random parametric family.
    """
    S = np.sqrt(2.0 * n)

    # Warm start with prob 0.2
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
        return a

    # Choose a family
    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'ushape', 'sine', 'uniform', 'dirichlet',
                        'randomwalk'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'sine':
        raw = _init_sine(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    else:
        raw = _init_random_walk(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a

def _hard_subgradient_descent(a0: np.ndarray,
                              deadline: float,
                              rng: np.random.Generator,
                              global_best: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=max(a∗a) (hard max).
    Polyak step size (cur−best)/‖g‖², periodic jitter.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1
        # ----- gradient of the hard max -----
        b = _full_convolution_fft(a)
        max_val = b.max()
        mask = (b == max_val).astype(float)

        a_rev = a[::-1]
        conv_len = len(mask) + n - 1
        f_len = _next_pow2(conv_len)
        Fmask = np.fft.rfft(mask, n=f_len)
        Fa_rev = np.fft.rfft(a_rev, n=f_len)
        conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
        start = n - 1
        grad = 2.0 * conv[start:start + n]

        # ----- Polyak step -----
        norm_sq = np.dot(grad, grad)
        if norm_sq > 0:
            step = (max_val - best_val) / (norm_sq + 1e-12)
            step = np.clip(step, 1e-7, 0.25)
        else:
            step = 0.0

        a = _project_onto_simplex(a - step * grad, S)

        cur_val = _max_convolution(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to escape plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=0.001 * np.mean(a), size=n)
            a = _project_onto_simplex(a + jitter, S)

    return best_a, best_val


def _smoothed_descent(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      global_best: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on a smoothed approximation of max(a∗a):
        f_ε(a) = ε·log∑_k exp((a∗a)_k / ε).

    The smoothing ε is gradually reduced (multiplicative decay) so the method
    first moves in a gentle landscape and then focuses on the true peak.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _max_convolution(a)

    # initialise eps as a fraction of the current peak
    cur_peak = best_val
    eps = max(0.005, cur_peak * 0.2)          # start fairly smooth
    eps_decay = 0.985                          # multiplicative decay per iteration

    it = 0
    while time.time() < deadline:
        it += 1

        # ---- convolution and smoothed weights ----
        b = _full_convolution_fft(a)
        cur_peak = b.max()

        # stable soft‑max weights
        w = np.exp((b - cur_peak) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(b, 1.0 / b.size)
        else:
            w = w / w_sum

        # ---- gradient of the smoothed objective ----
        a_rev = a[::-1]
        conv_len = len(w) + n - 1
        f_len = _next_pow2(conv_len)
        Fw = np.fft.rfft(w, n=f_len)
        Fa_rev = np.fft.rfft(a_rev, n=f_len)
        conv = np.fft.irfft(Fw * Fa_rev, n=f_len)[:conv_len]
        start = n - 1
        grad = 2.0 * conv[start:start + n]

        # ---- Polyak step (using the **hard** peak as lower bound) ----
        norm_sq = np.dot(grad, grad)
        if norm_sq > 0:
            step = (cur_peak - best_val) / (norm_sq + 1e-12)
            step = np.clip(step, 1e-8, 0.3)
        else:
            step = 0.0

        a = _project_onto_simplex(a - step * grad, S)

        # ---- record improvement on the true objective ----
        new_val = _max_convolution(a)
        if new_val + 1e-12 < best_val:
            best_val = new_val
            best_a = a.copy()
            # when we improve the hard objective we can afford a slightly larger ε
            eps = max(eps, new_val * 0.15)

        # ---- reduce smoothing ----
        eps = max(eps * eps_decay, 0.001)

        # ---- occasional random jitter ----
        if it % 250 == 0:
            jitter = rng.normal(scale=0.001 * S, size=n)
            a = _project_onto_simplex(a + jitter, S)

    return best_a, best_val


def _anneal_local_search(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing hill‑climb that moves a random fraction of mass
    between two positions.  The temperature decays exponentially.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    cur_val = _max_convolution(a)
    best_a = a.copy()
    best_val = cur_val

    T = 0.05 * cur_val            # start temperature ≈ 5 % of the current peak
    decay = 0.9999

    it = 0
    while time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        frac = rng.uniform(0.02, 0.18)
        delta = a[i] * frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _max_convolution(a_new)
        d = new_val - cur_val
        if d < 0 or rng.random() < np.exp(-d / (T + 1e-12)):
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()

        T *= decay
        if it % 500 == 0:
            jitter = rng.normal(scale=0.001 * S, size=n)
            a = _project_onto_simplex(a + jitter, S)
            cur_val = _max_convolution(a)

    return best_a, best_val

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine follows the multi‑stage plan described in the module doc‑string.
    The final construction is stored in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    end_time = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Warm‑start handling (previous best, if any)
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # ------------------------------------------------------------------
    # Phase 1 – broad random exploration (≈ 45 % of the budget)
    # ------------------------------------------------------------------
    explore_deadline = end_time - 350.0   # keep ~350 s for later phases
    K = 8                                 # keep the top‑K seeds
    top_seeds = []                        # list of (value, array)

    attempts = 0
    while time.time() < explore_deadline:
        attempts += 1
        # favour longer sequences but keep variety
        if rng.random() < 0.7:
            n = int(rng.integers(3000, 15000))
        else:
            n = int(rng.integers(15000, 25000))

        a0 = _initial_vector(n, rng, warm_seq)
        val = _max_convolution(a0)        # hard peak

        # keep the best K
        if len(top_seeds) < K:
            top_seeds.append((val, a0.copy()))
            top_seeds.sort(key=lambda x: x[0])
        elif val < top_seeds[-1][0]:
            top_seeds[-1] = (val, a0.copy())
            top_seeds.sort(key=lambda x: x[0])

        if attempts % 250 == 0:
            print(f"[explore] attempts={attempts:4d} best_K={top_seeds[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – smoothed descent on the K seeds
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    for idx, (seed_val, seed_seq) in enumerate(top_seeds):
        remaining = end_time - time.time()
        if remaining < 120.0:    # not enough time left for a decent run
            break
        slice_deadline = time.time() + min(80.0, remaining * 0.3)

        print(f"[seed {idx+1}/{len(top_seeds)}] length={len(seed_seq)} start_val={seed_val:.6f}")

        a_opt, val = _smoothed_descent(seed_seq, slice_deadline, rng, best_score)
        print(f"   [smoothed]   improved to {val:.6f}")

        # a short hard‑max refinement (helps to get out of the smooth approximation)
        a_opt2, val2 = _hard_subgradient_descent(a_opt,
                                                 slice_deadline,
                                                 rng,
                                                 best_score)
        print(f"   [hard subgrad] improved to {val2:.6f}")

        if val2 < best_score:
            best_score = val2
            best_seq = a_opt2.copy()
            print(f"   *** new global best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – long refinement on the current global best
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < end_time - 150.0:
        remaining = end_time - time.time() - 50.0
        slice_deadline = time.time() + remaining

        print("[global refinement] starting long smoothed descent")
        a_opt, val = _smoothed_descent(best_seq, slice_deadline, rng, best_score)
        if val < best_score:
            best_score = val
            best_seq = a_opt.copy()
            print(f"   [global] smoothed improved to {best_score:.6f}")

        # finish with hard subgradient
        a_opt2, val2 = _hard_subgradient_descent(best_seq,
                                                 slice_deadline,
                                                 rng,
                                                 best_score)
        if val2 < best_score:
            best_score = val2
            best_seq = a_opt2.copy()
            print(f"   [global] hard subgrad improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – length scaling & repeat refinement (optional)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < end_time - 80.0:
        n0 = len(best_seq)
        for factor in [1.2, 1.5, 2.0]:
            new_n = int(n0 * factor)
            if new_n <= 25000:
                print(f"[scaling] length {n0} → {new_n}")
                seq_scaled = _scale_sequence(best_seq, new_n)
                S = np.sqrt(2.0 * new_n)
                seq_scaled = np.clip(seq_scaled, 0.0, None)
                seq_scaled = seq_scaled * (S / seq_scaled.sum())

                slice_deadline = time.time() + 40.0
                a_opt, val = _smoothed_descent(seq_scaled, slice_deadline, rng, best_score)
                if val < best_score:
                    best_score = val
                    best_seq = a_opt.copy()
                    n0 = new_n
                    print(f"   [scaled] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < end_time - 20.0:
        slice_deadline = time.time() + 15.0
        a_opt, val = _anneal_local_search(best_seq, slice_deadline, rng)
        if val < best_score:
            best_score = val
            best_seq = a_opt.copy()
            print(f"[anneal] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f}  internal={best_score:.6f}  length={len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
