    T0 = best_val * 0.2
    last_recalc = start
    while time.time() < deadline:
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1)
        prod = a[idx] * a[k_max - idx]
        # donor selection via prod weighted sample
        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)
        # receiver: pick from low contributions
        threshold = np.percentile(prod, 20)
        low_candidates = idx[prod <= threshold]
        if low_candidates.size > 0:
            receiver = rng.choice(low_candidates)
        else:
            receiver = rng.integers(n)
        if donor == receiver:
            continue

            continue

import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray, size: int | None = None) -> np.ndarray:
    """Full linear convolution a*a using FFT.  Returns length 2·n‑1 array."""
    n = a.shape[0]
    if size is None:
        size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray, size: int | None = None) -> float:
    """Maximum entry of the convolution a*a."""
    return float(_conv_fft(a, size).max())


# ----------------------------------------------------------------------
# Objective -------------------------------------------------------------
# ----------------------------------------------------------------------


def _score(a: np.ndarray, size: int | None = None) -> float:
    """
    2·n·max(a*a) / (sum(a)²).  For ∑a = √(2n) this equals max(a*a).
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a, size)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max surrogate and its gradient (via FFT)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float, size: int) -> tuple[np.ndarray, np.ndarray]:
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    Returns (grad, full_convolution).
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    # safe soft‑max (subtract max to avoid overflow)
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    rev_fft = np.fft.rfft(a_rev, n=size)
    p_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(rev_fft * p_fft, n=size)[: 2 * n - 1]

    # extract the n entries that correspond to the gradient w.r.t. a
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase 1 – Multiplicative (mirror) descent on the soft‑max surrogate
# ----------------------------------------------------------------------


def _mirror_descent(a0: np.ndarray,
                    target_sum: float,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Exponential (mirror) descent on the soft‑max surrogate.
    Returns the best sequence observed and its score.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    size = 1 << ((2 * a.shape[0] - 1).bit_length())
    best_val = _score(a, size)

    beta = 1.0
    max_beta = 5e6
    base_eta = 0.018
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta, size)

        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * grad)

        # exact renormalisation (project onto the ℓ¹‑simplex)
        a = a * (target_sum / a.sum())

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0006, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a, size)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] MirrorGD it={it:5d} β={beta:8.1f} score={best_val:.6f}")

        if it % 300 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – Adam‑style projected gradient descent on the surrogate
# ----------------------------------------------------------------------


def _adam_descent(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max surrogate, with simplex projection.
    Returns the best sequence observed and its score.
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    size = 1 << ((2 * a.shape[0] - 1).bit_length())
    best_a = a.copy()
    best_val = _score(a, size)

    beta = 1.0
    max_beta = 5e6
    lr0 = 0.032
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta, size)

        t += 1
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad ** 2)
        m_hat = m / (1.0 - 0.9 ** t)
        v_hat = v / (1.0 - 0.999 ** t)

        lr = lr0 / (beta + 1.0)
        a -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        a = _project_to_simplex(a, target_sum)

        if t % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.0012 / (beta + 1.0), size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a, size)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={t:5d} β={beta:8.1f} score={best_val:.6f}")

        if t % 300 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 3 – aggressive local mass‑transfer refinement (simulated annealing)
# ----------------------------------------------------------------------


def _local_refine_enhanced(a0: np.ndarray,
                           target_sum: float,
                           deadline: float) -> tuple[np.ndarray, float]:
    """
    Exact hill‑climbing on the true max‑convolution.
    Small mass is moved from “bad’’ to “good’’ indices.
    The move is accepted according to an annealing schedule.
    """
    a = a0.copy()
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    b = _conv_fft(a, size)
    best_a = a.copy()
    best_val = _score(a, size)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.20                     # initial temperature
    last_recalc = start

    while time.time() < deadline:
        # current worst (maximal) convolution entry
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to the current max entry
        prod = a[idx] * a[k_max - idx]

        # donor – pick proportionally to its harmful contribution
        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – sample from the low‑product tail (30‑th percentile)
        thresh = np.percentile(prod, 20)                # a bit more aggressive
        low_candidates = idx[prod <= thresh]
        if low_candidates.size:
            receiver = rng.choice(low_candidates)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # propose a transfer (larger than before)
        max_delta = min(a[donor] * 0.15, target_sum * 0.004)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # update the full convolution in O(n) using slice tricks
        b_new = b.copy()
        b_new[donor : donor + n] -= 2.0 * delta * a
        b_new[receiver : receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # simulated‑annealing acceptance
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] LocalRefine improved val={best_val:.6f}")

        # periodic full recomputation to avoid drift
        if time.time() - last_recalc > 12.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a, size)
            last_recalc = time.time()
            cur_val = _score(a, size)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑calc improved val={best_val:.6f}")

        # occasional jitter to break plateaus
        if int((elapsed * 150)) % 1100 == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.02, target_sum * 0.0007)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a, size)
                cur_val = _score(a, size)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved val={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point -------------------------------------------------------
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin for the evaluator

    # --------------------------------------------------------------
    # Problem size (larger n gives a finer search but costs more time)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 15000
    elif budget_s >= 500:
        n = 10000
    else:
        n = 6000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Warm‑start (reuse best known construction if available)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
    else:
        # try two simple analytical shapes and keep the better one
        pos = np.arange(1, n + 1, dtype=np.float64)
        shape1 = pos ** (-0.55)                     # decreasing power‑law
        shape2 = np.sin(np.pi * (pos - 0.5) / n) ** 2   # smooth bell‑shape
        shape1 = shape1 * (target_sum / shape1.sum())
        shape2 = shape2 * (target_sum / shape2.sum())
        s1 = _score(shape1)
        s2 = _score(shape2)
        init = shape1 if s1 < s2 else shape2

    # --------------------------------------------------------------
    # Phase 1 – mirror (exponential) descent
    # --------------------------------------------------------------
    phase1_deadline = time.time() + 0.18 * (deadline - time.time())
    a1, val1 = _mirror_descent(init, target_sum, phase1_deadline)
    best_seq = a1
    best_val = val1
    print(f"[{time.time():.2f}] After Mirror descent best={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam projected gradient descent on the surrogate
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase2_deadline = time.time() + 0.32 * (deadline - time.time())
        a2, val2 = _adam_descent(best_seq, target_sum, phase2_deadline)
        if val2 < best_val:
            best_seq = a2
            best_val = val2
        print(f"[{time.time():.2f}] After Adam GD best={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – exact local refinement (simulated annealing)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a3, val3 = _local_refine_enhanced(best_seq, target_sum, deadline)
        if val3 < best_val:
            best_seq = a3
            best_val = val3
        print(f"[{time.time():.2f}] After local refinement best={best_val:.6f}")

    # --------------------------------------------------------------
    # Final clean‑up (trim trailing near‑zeros)
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # --------------------------------------------------------------
    # Final evaluation (the external checker)
    # --------------------------------------------------------------
    try:
        final_val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for warm‑starts in future calls
    globals()["height_sequence_1"] = final_seq

    return final_seq
