"""
Proposes a non‑negative step function that minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)^2)

The algorithm is an enhanced version of the previously used method:

1.  A schedule of increasing lengths `n` is processed.  The best known
    sequence (if any) is up‑sampled to the new length and then refined.

2.  For each length we spend a fraction of the remaining time on three
    phases:
        * A – many restarts of projected Adam on a smooth‑max surrogate
          of the convolution (soft‑max).  The number of restarts adapts
          to the time budget.
        * B – a deterministic greedy hill‑climb that repeatedly moves
          mass from the index that contributes most to the current
          worst convolution entry to a more favourable index.
        * C – a short extra Adam run (if time permits).

3.  After all lengths have been processed, a final polishing step
    (long Adam + short hill‑climb) uses whatever time is left.

All vectors are projected onto the ℓ¹‑simplex Σa = √(2 n) after each
update, which keeps the sum (and therefore the scaling of the objective)
constant while protecting against overflow/underflow.

The code is fully vectorised, uses FFTs for fast convolutions and respects
the 1000 s wall‑clock budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float,
                    deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max surrogate.
    Runs until the wall‑clock `deadline`.
    """
    a = a0.copy()
    n = a.size

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # learning‑rate base (scaled with the simplex radius)
    base_lr = 0.15 * target_sum

    # smoothed‑max schedule
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1

        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        # occasional tiny jitter to escape flat regions
        if t % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 1e-4 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # increase smoothing
        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a

# Random initialisation / up‑sampling
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

# Deterministic hill‑climb (exact max)
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that repeatedly attacks the current worst
    convolution entry.  Returns the best vector seen and its maximal
    convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()

    rng = np.random.default_rng()

    while time.time() < deadline:
        # Index of the current maximal convolution entry
        M = int(np.argmax(b))

        # Source indices that affect b[M]
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]

        # Pick source with largest contribution
        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break   # nothing left to move

        # Counterpart values for all possible destinations
        counterpart = np.zeros(n, dtype=float)
        j_low = i_low
        j_high = i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        # Choose destination with minimal counterpart (excluding i)
        order = np.argsort(counterpart[j_range])
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        else:
            break

        # Amount of mass to move (up to 90 % of source)
        delta = a[i] * 0.9
        if delta <= 0.0:
            break

        # ----- evaluate the move (exact O(n) update) -----
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        # ----- accept if we improve -----
        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
            continue

        # ----- otherwise try a smaller step (binary search) -----
        improved = False
        delta2 = delta * 0.5
        while delta2 > 1e-14:
            b_test = b.copy()
            b_test[i:i + n] -= 2.0 * delta2 * a
            b_test[j:j + n] += 2.0 * delta2 * a
            if k2i < b_test.size:
                b_test[k2i] += delta2 * delta2
            if k2j < b_test.size:
                b_test[k2j] += delta2 * delta2
            if kij < b_test.size:
                b_test[kij] -= 2.0 * delta2 * delta2
            if b_test.max() < b.max() - 1e-12:
                a[i] -= delta2
                a[j] += delta2
                b = b_test
                if b.max() < best_score - 1e-12:
                    best_score = b.max()
                    best_a = a.copy()
                improved = True
                break
            delta2 *= 0.5

        if not improved:
            # small random jitter to try to escape a plateau
            jitter = rng.normal(scale=1e-6, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            target = math.sqrt(2.0 * n)
            a = _project_to_simplex(a, target)
            b = _full_convolution(a)

    return best_a, best_score

        total_future_n = sum(schedule[idx:])
        time_for_n = remaining * (n / total_future_n) * 0.90
        # enforce a modest floor so that every stage gets a chance
        time_for_n = max(time_for_n, 30.0)

        gd_budget = time_for_n * 0.55
        hc_budget = time_for_n * 0.35
        extra_budget = time_for_n * 0.10

        # --------------------------------------------------------------
        # Phase A – many restarts of projected Adam
        # --------------------------------------------------------------
        min_restart_time = 5.0
        n_restarts = max(1, int(gd_budget // min_restart_time))
        per_restart = gd_budget / n_restarts

        local_best_a = None
        local_best_score = np.inf

        for r in range(n_restarts):
            if time.time() >= deadline - 2.0:
                break
            this_deadline = min(deadline - 0.3, time.time() + per_restart)
            start_a = a_init.copy() if r == 0 else _random_initial_sequence(n, rng)
            a_opt = _adam_optimize(start_a, target_sum, this_deadline)
            score = _max_convolution_fft(a_opt)
            if score < local_best_score:
                local_best_score = score
                local_best_a = a_opt.copy()

        if local_best_a is not None and local_best_score < best_score - 1e-12:
            best_seq = local_best_a.copy()
            best_score = local_best_score
            print(f"[{time.time():.1f}] Adam improvement for n={n} "
                  f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # Phase B – deterministic hill‑climb on the exact convolution
        # --------------------------------------------------------------
        if time.time() < deadline - 2.0:
            a_cur = best_seq.copy()
            a_cur = _project_to_simplex(a_cur, target_sum)
            b_cur = _full_convolution(a_cur)
            hc_deadline = min(deadline - 0.3, time.time() + hc_budget)
            a_hc, score_hc = _hill_climb_deterministic(a_cur, b_cur, hc_deadline)
            if score_hc < best_score - 1e-12:
                best_seq = a_hc.copy()
                best_score = score_hc
                print(f"[{time.time():.1f}] Hill‑climb improvement for n={n} "
                      f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # Phase C – short extra Adam run if time permits
        # --------------------------------------------------------------
        if extra_budget > 2.0 and time.time() < deadline - 1.0:
            extra_deadline = deadline - 0.5
            a_ref = _adam_optimize(best_seq, target_sum, extra_deadline)
            score_ref = _max_convolution_fft(a_ref)
            if score_ref < best_score - 1e-12:
                best_seq = a_ref.copy()
                best_score = score_ref
                print(f"[{time.time():.1f}] Post‑HC Adam improvement for n={n} "
                      f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Final polishing with whatever time is left
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        n = best_seq.size
        target = math.sqrt(2.0 * n)

        # final Adam (long)
        final_deadline = deadline - 0.8
        best_seq = _adam_optimize(best_seq, target, final_deadline)

        # final short hill‑climb
        b_final = _full_convolution(best_seq)
        final_hc_deadline = deadline - 0.5
        best_seq, final_score = _hill_climb_deterministic(
            best_seq, b_final, final_hc_deadline)

        if final_score < best_score - 1e-12:
            best_score = final_score
            print(f"[{time.time():.1f}] Final polishing improvement: "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Clean up – drop trailing near‑zeros
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: simple uniform distribution (should never be hit)
        n = 2000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # 4) Report the official evaluation value
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:               # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
