def _adam_subgradient_max(a0, target_sum, deadline,
                          lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    a = _project_to_simplex(a0.copy(), target_sum)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_a = a.copy()
    best_val = _score(a)
    while time.time() < deadline:
        t += 1
        # compute convolution, find kmax
        b = _conv_fft(a)
        kmax = int(np.argmax(b))
        # compute gradient g
        g = np.zeros_like(a)
        lo = max(0, kmax - (len(a)-1))
        hi = min(kmax, len(a)-1)
        idx = np.arange(lo, hi+1)
        # g[idx] = 2 * a[kmax - idx]
        g[idx] = 2 * a[kmax - idx]
        # update m,v
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        # projection
        a = _project_to_simplex(a, target_sum)
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam iter={t} best={best_val:.6f}")
    return best_a, best_val
```

"""
Optimiser for the step-function benchmark.

The optimiser proceeds in three stages:

1. Generate a variety of parametric candidate sequences (power‑law,
   power‑law with exponential tail, uniform and the previous best
   sequence) and keep the best one w.r.t. the objective.

2. Refine the candidate with a fast ADAM based sub‑gradient descent on
   the exact maximal convolution entry.  The gradient of max(b) w.r.t.
   the coefficients is easily obtained as a shifted copy of the vector
   itself, therefore each iteration costs only one FFT (to locate the
   index of the current maximum) and O(n) arithmetic.

3. Run a short deterministic mass‑transfer phase and a final soft‑max
   gradient descent (annealed temperature) to polish the solution.

All steps respect the global time budget; the routine returns the best
sequence found (trimmed of trailing zeros) as a plain Python list of
floats.
"""
import time
import numpy as np
import math

def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())

def _score(a: np.ndarray) -> float:
    """Benchmark score 2·n·max_conv/(sum(a)²).  For the canonical scaling
    sum(a)=sqrt(2·n) this equals max_conv.
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))

def _scale_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale a non‑negative vector to have Σv = target_sum (simple scaling)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        v = v * (target_sum / s)
    else:
        v.fill(target_sum / v.size)
    return v

def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = target_sum}."""
    # implementation from Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a

def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a

def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)

def _adam_subgradient_max(a0: np.ndarray,
                          target_sum: float,
                          deadline: float,
                          lr: float = None,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam‑style sub‑gradient descent on the exact maximum convolution entry.
    The gradient of max(b) at the current maximal index k is
        g_i = 2 * a_{k-i}  (if 0 ≤ k-i < n, else 0).
    """
    a = _scale_to_simplex(a0.copy(), target_sum)
    n = a.shape[0]
    if lr is None:
        lr = 0.2 * target_sum   # default step scale
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow = beta1
    beta2_pow = beta2
    best_a = a.copy()
    best_val = _score(a)
    it = 0
    while time.time() < deadline:
        it += 1
        # full convolution and index of the current maximum
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # gradient of the maximal entry (sub‑gradient)
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[kmax - idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _scale_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} score={best_val:.6f}")

    return best_a, best_val

def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 2e5,
                               step_factor: float = 0.6,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max approximation
    of the maximal convolution entry.  This is the same routine used in
    the previous baseline but with a few tuned defaults.
    """
    a = _scale_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    step_base = target_sum * step_factor
    it = 0

    while time.time() < deadline:
        # compute soft‑max gradient
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_b = b.max()
        exps = np.exp(beta * (b - max_b))
        p = exps / exps.sum()

        # gradient = 2 * (a reversed) ⋆ p
        a_rev = a[::-1]
        Ar_fft = np.fft.rfft(a_rev, n=size)
        Pr_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
        g = 2.0 * conv_rev[n - 1: n - 1 + n]

        # Nesterov‑style momentum
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _scale_to_simplex(a, target_sum)

        # occasional jitter
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _scale_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 300 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-5,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy mass‑transfer moves that directly lower the current maximal
    convolution entry (the same idea as before but with a simpler schedule)."""
    a = a0.copy()
    n = a.shape[0]
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0.0:

"""
Optimiser for the step‑function benchmark.

The algorithm works in three stages:

1. **Parametric pre‑search** – a handful of power‑law / exponential‑tail
   families (plus the previous best sequence, if available) are generated
   and the best one is kept.

2. **ADAM sub‑gradient descent on the exact maximum convolution entry**.
   The gradient of max(b) at the current maximal index k is
   `g_i = 2 * a_{k-i}` for the indices where it is defined.
   Each iteration costs one FFT to locate the maximal index and
   O(n) arithmetic to form the gradient.  After the step the vector is
   scaled back onto the simplex (non‑negative, fixed sum).

3. **Deterministic mass‑transfer + random moves + a final soft‑max
   gradient descent** – these phases are identical to the previous
   baseline but use the improved sequence from stage 2 as a start.
   All phases respect the global time budget.

The routine returns the best sequence found (trailing zeros trimmed)
as a plain Python list of floats.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Objective
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Benchmark score 2·n·max(b)/(sum(a)²).  For Σa=√(2n) this equals max(b)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex handling
# ----------------------------------------------------------------------
def _scale_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale a non‑negative vector to have Σv = target_sum (cheap scaling)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        v = v * (target_sum / s)
    else:
        v.fill(target_sum / v.size)
    return v


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = target_sum}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# ADAM sub‑gradient descent on max(b)
# ----------------------------------------------------------------------
def _adam_subgradient_max(a0: np.ndarray,
                          target_sum: float,
                          deadline: float,
                          lr: float = None,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam‑style sub‑gradient descent on the exact maximal convolution entry.
    Gradient at the current maximal index k is `g_i = 2*a_{k-i}` (if defined).
    """
    a = _scale_to_simplex(a0.copy(), target_sum)
    n = a.shape[0]
    if lr is None:
        lr = 0.15 * target_sum
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow = beta1
    beta2_pow = beta2
    best_a = a.copy()
    best_val = _score(a)
    it = 0

    while time.time() < deadline:
        it += 1
        # locate current maximal convolution entry
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # sub‑gradient of that entry
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[kmax - idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = _scale_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Soft‑max gradient descent (used for final polishing)
# ----------------------------------------------------------------------
def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 2e5,
                               step_factor: float = 0.6,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on a soft‑max approximation of
    max(convolution).  Very similar to the baseline implementation but
    with a few tuned defaults.
    """
    a = _scale_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    step_base = target_sum * step_factor
    it = 0
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    while time.time() < deadline:
        # FFT of the current vector
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_b = b.max()
        exps = np.exp(beta * (b - max_b))
        p = exps / exps.sum()

        # gradient = 2 * (a reversed) ⋆ p
        a_rev = a[::-1]
        Ar_fft = np.fft.rfft(a_rev, n=size)
        Pr_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
        g = 2.0 * conv_rev[n - 1: n - 1 + n]

        # Nesterov‑style momentum
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _scale_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = _scale_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 300 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

        candidates.append(_power_law_sequence(n, p, target_sum))

    # (b) power‑law + exponential tail
    for p in np.linspace(0.35, 0.70, 6):
        for q in np.linspace(0.0, 5.0, 6):
            candidates.append(_exp_power_law_sequence(n, p, q, target_sum))

    # (c) uniform (baseline)
    candidates.append(np.full(n, target_sum / n, dtype=np.float64))

    # (d) previous best, if available
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=np.float64)
        candidates.append(_resize_sequence(prev, n))

    # jittered variants to increase diversity
    for seq in list(candidates):
        jitter = rng.normal(scale=target_sum * 0.003, size=n)
        seq_j = seq + jitter
        seq_j = _scale_to_simplex(seq_j, target_sum)
        candidates.append(seq_j)

    # quick evaluation of a manageable subset
    max_initial = min(200, len(candidates))
    idxs = rng.choice(len(candidates), size=max_initial, replace=False)

    best_seq = None
    best_score = np.inf
    for i in idxs:
        val = _score(candidates[i])
        if val < best_score:
            best_score = val
            best_seq = candidates[i].copy()
    print(f"[{time.time():.2f}] Initial best score = {best_score:.6f}")

    # -----------------------------------------------------------
    # Phase 1 – ADAM sub‑gradient descent on the exact max entry
    # -----------------------------------------------------------
    phase1_deadline = time.time() + (deadline - time.time()) * 0.35
    while time.time() < phase1_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        start_vec = best_seq + jitter
        start_vec = _scale_to_simplex(start_vec, target_sum)

        sub_deadline = time.time() + min(12.0, phase1_deadline - time.time())
        a_opt, val_opt = _adam_subgradient_max(start_vec,
                                               target_sum,
                                               sub_deadline,
                                               lr=target_sum * 0.15)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Adam improved best = {best_score:.6f}")

    # -----------------------------------------------------------
    # Phase 2 – Deterministic + random mass transfer
    # -----------------------------------------------------------
    while time.time() < deadline - 5.0:
        a_ref, val_ref = _deterministic_mass_transfer(best_seq,
                                                      deadline,
                                                      target_sum,
                                                      start_temp=0.5,
                                                      end_temp=1e-6,
                                                      low_frac=0.04)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] DMT improved best = {best_score:.6f}")

        a_ref2, val_ref2 = _global_random_mass_transfer(best_seq,
                                                        deadline,
                                                        target_sum)
        if val_ref2 < best_score:
            best_score = val_ref2
            best_seq = a_ref2.copy()
            print(f"[{time.time():.2f}] Global random improved best = {best_score:.6f}")

        if time.time() > deadline - 8.0:
            break

    # -----------------------------------------------------------
    # Phase 3 – Final soft‑max based polishing
    # -----------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _annealed_gradient_descent(best_seq,
                                                    deadline,
                                                    target_sum,
                                                    max_beta=8e5,
                                                    step_factor=0.5,
                                                    gamma=0.98)
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polished best = {best_score:.6f}")

    # -----------------------------------------------------------
    # Trim trailing zeros and return the result
    # -----------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional evaluation for logging (harmless if evaluate_sequence is absent)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
