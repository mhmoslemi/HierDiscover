def _adam_descent(a0, deadline, target_sum, max_beta=1e6, init_lr=0.2,
                  beta1=0.9, beta2=0.999, eps=1e-8):
    a = _project_to_simplex(a0.copy(), target_sum)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _score(a)
    beta = 0.5
    t = 0
    while time.time() < deadline:
        t += 1
        grad, conv = _gradient_softmax(a, beta)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = init_lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = _project_to_simplex(a, target_sum)

        # anneal beta
        if t % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.7, max_beta)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={t} β={beta:.1f} best={best_score:.6f}")
    return best_a, best_score
```

def _simulated_annealing(a0, deadline, target_sum, max_iter=200000):
    a = a0.copy()
    n = a.shape[0]
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    # schedule: temperature start T0 = 0.1 * cur_score perhaps, decays to near 0.
    while time.time() < deadline:
        # compute current convolution only if needed? We'll compute each iteration but maybe slow.
        # Instead we can compute conv only when evaluating after move (which we need for max).
        # We'll compute conv using _conv_fft each iteration.
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))
        # choose donor and receiver with contributions for that kmax
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi+1)
        w = a[idx] + a[kmax - idx]
        donor = idx[np.argmax(w)]
        # choose receiver among low w (smallest) but also global low
        low_cnt = max(1, int(len(idx) * 0.1))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        max_delta = a[donor] * 0.2
        delta = rng.random() * max_delta

        # propose move
        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        new_score = _score(a)

        # determine temperature
        elapsed = time.time() - start_time
        total = deadline - start_time
        # linear decay temperature from start_T to end_T
        start_T = cur_score * 0.1  # 10% of current maybe
        end_T = 1e-6
        T = start_T * ((total - elapsed) / total) + end_T * (elapsed / total)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_score - cur_score) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA new best {best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # optional periodic full jitter to escape stagnation
        if time.time() - start_time > 0 and int((time.time() - start_time) * 10) % 5000 == 0:
            # Add small random noise globally
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
    return best_a, best_score

"""
Improved optimizer for the step‑function benchmark.

Main ideas
----------
* Use a larger problem size (up to 25000 coefficients) – more
  degrees of freedom allow a lower maximal convolution value.
* Start from a richer parametric family (power‑law with exponential
  tail) and keep a few jittered variants.
* Adam‑style projected gradient descent on a soft‑max approximation of
  the max‑convolution, with an exponential annealing of the soft‑max
  temperature.  Adam’s adaptive step‑size often reaches a better local
  minimum than plain momentum GD.
* Deterministic mass‑transfer that directly attacks the current maximal
  convolution entry (as in the reference code) – now run for a larger
  share of the budget.
* A simple simulated‑annealing random‑mass‑transfer phase that can
  escape shallow local minima.
* Final polishing with a very high soft‑max temperature.

All steps respect the global time budget (default 1000 s).  The best
sequence found is trimmed, stored in the global variable
`height_sequence_1` and returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Benchmark objective 2·n·max(a*a)/(sum(a)^2)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple families for the quick parametric search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Generate (i+1)^(-p) of length n and scale to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float,
                            target_sum: float) -> np.ndarray:
    """(i+1)^(-p) * exp(-q * i / n) scaled to target_sum."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max approximation of the max‑convolution and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where conv is the full convolution a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    # avoid overflow – shift by max_b
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * (a reversed) ⋆ p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on the soft‑max objective
# ----------------------------------------------------------------------


def _adam_descent(a0: np.ndarray,
                  deadline: float,
                  target_sum: float,
                  max_beta: float = 2e6,
                  init_lr: float = 0.2,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam projected gradient descent on the soft‑max approximation.
    Returns (best_vector, best_score).
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta1_pow = beta1
    beta2_pow = beta2

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    it = 0
    while time.time() < deadline:
        it += 1
        # gradient of the current soft‑max
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = init_lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = _project_to_simplex(a, target_sum)

        # update bias‑correction factors
        beta1_pow *= beta1
        beta2_pow *= beta2

        # anneal the soft‑max temperature
        if it % 250 == 0 and beta < max_beta:
            beta = min(beta * 1.8, max_beta)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (phase B)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-6,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy mass transfer that directly reduces the maximal convolution entry,
    with a simple Metropolis acceptance rule and exponential temperature decay.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w.max() - w.min()
        if weight_diff > 0.0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0.0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            # revert the move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing random mass‑transfer (global exploration)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         max_iter: int = 200000) -> tuple[np.ndarray, float]:
    """
    Simple SA that repeatedly moves a fraction of mass from a large entry
    to a small entry, accepting with a temperature that decays linearly.
    """
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = deadline - start_time

    it = 0
    while time.time() < deadline and it < max_iter:
        # pick donor from top fraction and receiver from bottom fraction
        top_frac = 0.02
        low_frac = 0.02
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        new_score = _score(a)

        # linear temperature schedule
        elapsed = time.time() - start_time
        T_start = cur_score * 0.05  # 5 % of current score
        T_end = 1e-6
        T = max(T_end, T_start * (1.0 - elapsed / total_time))

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with a low benchmark score.
    The routine respects the time budget, prints progress, and returns the
    best sequence found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5     # safety margin

    # ---------------------------------------------------------------
    # Choose problem size (larger n gives more freedom)
    # ---------------------------------------------------------------
    if budget_s >= 900:
        n = 25000
    elif budget_s >= 600:
        n = 20000
    elif budget_s >= 300:
        n = 15000
    else:
        n = 10000

    target_sum = np.sqrt(2.0 * n)   # Σa is fixed to this value (scaling invariant)

    rng = np.random.default_rng()
    candidates = []

    # ----------------------------------------------------------------
    # Phase 0 – diverse parametric pre‑search
    # ----------------------------------------------------------------
    # (a) previous best, if any
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            prev = _resize_sequence(prev, n)
        candidates.append(prev)

    # (b) pure power‑law
    for p in np.linspace(0.30, 0.85, 12):
        candidates.append(_power_law_sequence(n, p, target_sum))

    # (c) power‑law + exponential tail
    for p in np.linspace(0.30, 0.85, 9):
        for q in np.linspace(0.0, 8.0, 9):
            candidates.append(_exp_power_law_sequence(n, p, q, target_sum))

    # (d) jittered variants of everything above
    jittered = []
    for seq in candidates:
        jitter = rng.normal(scale=target_sum * 0.008, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        jittered.append(seq_j)
    candidates.extend(jittered)

    # (e) a few pure random vectors (useful for large n)
    for _ in range(5):
        rand = rng.random(n)
        rand = _project_to_simplex(rand, target_sum)
        candidates.append(rand)

    # ---------------------------------------------------------------
    # Cheap evaluation of the pool, keep the best few
    # ---------------------------------------------------------------
    max_initial_evals = min(250, len(candidates))
    sample_idx = rng.choice(len(candidates), size=max_initial_evals, replace=False)
    best_seq = None
    best_score = np.inf
    for i in sample_idx:
        seq = candidates[i]
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
    print(f"[{time.time():.2f}] Initial best score = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase A – Adam‑style projected gradient descent with restarts
    # ---------------------------------------------------------------
    phase_a_deadline = time.time() + (deadline - time.time()) * 0.30
    while time.time() < phase_a_deadline - 5.0:
        # jitter a fresh start around the current best
        start_vec = best_seq + rng.normal(scale=target_sum * 0.015, size=n)
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _adam_descent(start_vec,
                                       deadline=phase_a_deadline,
                                       target_sum=target_sum,
                                       max_beta=5e5,
                                       init_lr=0.25)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Adam improved best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase B – Deterministic mass‑transfer (greedy)
    # ---------------------------------------------------------------
    phase_b_deadline = time.time() + (deadline - time.time()) * 0.30
    a_ref, val_ref = _deterministic_mass_transfer(best_seq,
                                                  deadline=phase_b_deadline,
                                                  target_sum=target_sum,
                                                  start_temp=0.4,
                                                  end_temp=1e-6,
                                                  low_frac=0.04)
    if val_ref < best_score:
        best_score = val_ref
        best_seq = a_ref.copy()
        print(f"[{time.time():.2f}] DMT improved best = {best_score:.6f}")

    # ---------------------------------------------------------------

    # ---------------------------------------------------------------
    if time.time() < deadline - 5.0:
        a_sa, val_sa = _simulated_annealing(best_seq,

                                            target_sum=target_sum,
                                            max_iter=200000)

    arr = np.array(best_seq, dtype=float)

        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for possible future runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
