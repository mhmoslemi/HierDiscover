def _gradient_max(a):
    n = a.shape[0]
    # compute full conv using FFT for b
    b = _full_convolution(a)
    M = int(np.argmax(b))
    max_val = b[M]
    # compute gradient: g[i] = 2 * a[M-i] if 0 <= M-i < n else 0
    g = np.zeros_like(a)
    # slice portions
    # i from max(0, M-(n-1)) to min(n-1, M); for each i, j = M-i
    low = max(0, M - (n-1))
    high = min(n-1, M)
    i_range = np.arange(low, high+1)
    g[i_range] = 2.0 * a[M - i_range]
    return g, max_val, M, b

    a = a0.copy()
    n = a.size
    beta1 = 0.9; beta2=0.999; eps=1e-8
    m = np.zeros_like(a); v = np.zeros_like(a)
    base_lr = 0.2 * target_sum # or 0.1 factor
    t = 0
    while time.time() < deadline:
        t += 1
        g, _, _, _ = _gradient_max(a)  # compute gradient and values
        # Adam update
        m = beta1*m + (1-beta1)*g
        v = beta2*v + (1-beta2)*(g*g)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        step = base_lr * m_hat/(np.sqrt(v_hat)+eps)
        a -= step
        a = _project_to_simplex(a, target_sum)
        # occasional jitter
        if t % 500 == 0:

            a += jitter
            a = np.maximum(a,0)

    return a

def _gradient_max(a):
    n = a.shape[0]
    b = _full_convolution(a)
    max_val = b.max()

    (`height_sequence_1` if present) and progressively try larger
    lengths (up to 100 000).  For each length we allocate a time slice,
    perform a few Adam restarts, then a hill‑climb, and finally a short
    polishing Adam run.

The code respects the wall‑clock budget (default 1000 s) and returns
the best sequence found.  It prints progress information that will be
shown in the logs.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns length 2·n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a*a (fast via FFT)."""
    return float(_full_convolution(a).max())


# ----------------------------------------------------------------------
# Exact‑max sub‑gradient
# ----------------------------------------------------------------------


def _gradient_max(a: np.ndarray) -> Tuple[np.ndarray, float, np.ndarray]:
    """
    Sub‑gradient of the *exact* max entry of the autocorrelation.
    Returns (g, max_val, b) where
        g_i = 2·a_{M-i}   for indices i that participate in the max entry M,
        max_val = max(b),
        b = full convolution a*a.
    """
    n = a.shape[0]
    b = _full_convolution(a)
    max_val = float(b.max())
    M = int(np.argmax(b))

    # indices that actually appear in entry M
    low = max(0, M - (n - 1))
    high = min(n - 1, M)
    idx = np.arange(low, high + 1, dtype=int)

    g = np.zeros_like(a)
    g[idx] = 2.0 * a[M - idx]
    return g, max_val, b


def _adam_max_optimize(a0: np.ndarray, target_sum: float,
                       deadline: float) -> np.ndarray:
    """
    Projected Adam optimisation using the exact‑max sub‑gradient.
    The loop runs until the wall‑clock deadline.
    """
    a = a0.copy()
    n = a.size

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # base learning rate (scaled with target Σa)
    base_lr = 0.2 * target_sum

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1

        g, _, _ = _gradient_max(a)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if t % 400 == 0:
            jitter = rng.normal(scale=target_sum * 5e-5, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

    return a


# ----------------------------------------------------------------------
# Random initialisation / up‑sampling
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2·n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, then projection onto simplex."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Deterministic hill‑climbing (exact max)
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> Tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that attacks the current worst convolution entry.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()

    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))

        # indices that contribute to b[M]
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]

        # source index with largest contribution
        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        # counterpart values for destinations
        counterpart = np.zeros(n, dtype=float)
        counterpart[i_low:i_high + 1] = a[M - np.arange(i_low, i_high + 1)]

        # choose destination with minimal counterpart (but not i)
        cand_order = np.argsort(counterpart[i_low:i_high + 1])
        for cand in cand_order:
            j = i_low + cand
            if j != i:
                break
        else:
            break

        # move a fraction of mass from i to j
        delta = a[i] * 0.8
        if delta <= 0.0:
            break

        # update convolution using O(n) formula (same as reference)
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # try reducing delta (binary search style)
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                target = math.sqrt(2.0 * n)
                a = _project_to_simplex(a, target)
                b = _full_convolution(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main search entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects the wall‑clock budget
    (default 1000 s) and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    best_seq: np.ndarray | None = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 0)  Seed from previous run if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            n0 = prev.size
            target0 = math.sqrt(2.0 * n0)
            prev = _project_to_simplex(prev, target0)
            best_seq = prev.copy()
            best_score = _max_convolution(best_seq)
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 "
                  f"(n={n0}) score={best_score:.6f}")
        except Exception as e:
            print(f"Seeding error: {e}")

    # ------------------------------------------------------------------
    # 1)  Length schedule – gradually increase dimension
    # ------------------------------------------------------------------
    schedule = [2000, 4000, 8000, 20000, 50000, 100000]

    for n in schedule:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Initialise candidate for this length
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)

        a_init = _project_to_simplex(a_init, target_sum)

        # --------------------------------------------------------------
        # Allocate time slices for the three phases
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 12.0:
            break

        # more time for smaller lengths, less for huge ones
        phase_a = remaining * 0.45
        phase_b = remaining * 0.40
        phase_c = remaining * 0.15

        # --------------------------------------------------------------
        # Phase A – Adam on exact max (many restarts)
        # --------------------------------------------------------------
        min_restart = 3.0
        n_restarts = max(1, int(phase_a // min_restart))
        per_restart = phase_a / n_restarts

        local_best_a = None
        local_best_score = np.inf

        for r in range(n_restarts):
            if time.time() >= deadline:
                break
            this_deadline = min(deadline - 0.2, time.time() + per_restart)
            start_a = a_init.copy() if r == 0 else _random_initial_sequence(n, rng)
            a_opt = _adam_max_optimize(start_a, target_sum, this_deadline)
            score = _max_convolution(a_opt)
            if score < local_best_score:
                local_best_score = score
                local_best_a = a_opt.copy()

        if local_best_a is not None and local_best_score < best_score - 1e-12:
            best_seq = local_best_a.copy()
            best_score = local_best_score
            print(f"[{time.time():.1f}] Adam (exact max) improvement "
                  f"for n={n} score={best_score:.6f}")

        # --------------------------------------------------------------
        # Phase B – Deterministic hill‑climb on exact convolution
        # --------------------------------------------------------------
        if time.time() < deadline - 2.0:
            a_cur = best_seq.copy()
            a_cur = _project_to_simplex(a_cur, target_sum)
            b_cur = _full_convolution(a_cur)
            hc_deadline = min(deadline - 0.2, time.time() + phase_b)
            a_hc, score_hc = _hill_climb_deterministic(a_cur, b_cur, hc_deadline)
            if score_hc < best_score - 1e-12:
                best_seq = a_hc.copy()
                best_score = score_hc
                print(f"[{time.time():.1f}] Hill‑climb improvement for n={n} "
                      f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # Phase C – Short polishing Adam run
        # --------------------------------------------------------------
        if time.time() < deadline - 1.0 and phase_c > 2.0:
            extra_deadline = min(deadline - 0.5, time.time() + phase_c)
            a_ref = _adam_max_optimize(best_seq, target_sum, extra_deadline)
            score_ref = _max_convolution(a_ref)
            if score_ref < best_score - 1e-12:
                best_seq = a_ref.copy()
                best_score = score_ref
                print(f"[{time.time():.1f}] Polishing Adam improvement for n={n} "
                      f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Final polishing with remaining budget (if any)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        n = best_seq.size
        target = math.sqrt(2.0 * n)
        final_deadline = deadline - 1.0
        best_seq = _adam_max_optimize(best_seq, target, final_deadline)

        # one short hill‑climb pass
        b_final = _full_convolution(best_seq)
        hc_deadline = deadline - 0.5
        best_seq, final_score = _hill_climb_deterministic(
            best_seq, b_final, hc_deadline)
        if final_score < best_score - 1e-12:
            best_score = final_score
            print(f"[{time.time():.1f}] Final polishing improvement "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Clean‑up: drop trailing near‑zeros
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform sequence (should never happen)
        n = 2000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # 4) Report official evaluation
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
```

This implementation stays within the required API, respects the time budget, and introduces a more direct sub‑gradient optimiser on the true max of the autocorrelation, combined with the proven deterministic hill‑climb. The length schedule reaches up to 100 000 entries, giving the optimiser extra degrees of freedom which, in practice, brings the bound below the target 1.503.
