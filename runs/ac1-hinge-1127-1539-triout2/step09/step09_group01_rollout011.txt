def _mass_transfer_annealing(a, target_sum, deadline):
    a = a.copy()
    n = a.size
    best_a = a.copy()
    best_max = _max_convolution_fft(a)
    cur_max = best_max
    start = time.time()
    while time.time() < deadline:
        # compute conv and mask
        conv = _conv_fft(a)
        M = conv.max()
        # mask for near-peak entries
        eps = max(M * 0.001, 1e-12) # or maybe 0.005
        mask = (conv >= M - eps).astype(np.float64)
        # compute weight contributions via convolution
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        mask_fft = np.fft.rfft(mask, n=size)
        w_full = np.fft.irfft(a_fft * mask_fft, n=size)[:2*n-1]
        w = w_full[n-1:n-1+n]  # length n
        donor = int(np.argmax(w))
        # receiver sample from low weight indices: choose among indices where w is small
        low_idxs = np.where(w <= np.percentile(w, 10))[0]  # bottom 10%
        if low_idxs.size == 0:
            receiver = donor
        else:
            receiver = int(np.random.choice(low_idxs))

        if donor == receiver:
            continue

        # choose delta: fraction of donor, scaled by temperature
        elapsed = time.time() - start
        ratio = elapsed / (deadline - start)
        temp = 0.5 * ((1e-4) / 0.5) ** ratio  # start_temp=0.5, end_temp=1e-4
        # adapt delta
        max_delta = a[donor] * 0.5
        delta = max_delta * temp * 5.0  # temperature scaled, adjust factor
        # ensure delta not too small
        if delta < 1e-9:
            delta = max_delta * 0.01
        # apply
        a[donor] -= delta
        a[receiver] += delta
        # ensure non-neg
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        new_max = _max_convolution_fft(a)

        if new_max < cur_max or np.random.random() < np.exp(-(new_max - cur_max) / max(temp, 1e-12)):
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT anneal new best={best_max:.6f}")

        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # optional jitter
        if np.random.rand() < 0.001:
            jitter = np.random.normal(scale=target_sum * 0.0005, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            cur_max = _max_convolution_fft(a)
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.2f}] jitter improved best={best_max:.6f}")

    return best_a, best_max

    cur_max = best_max
    start = time.time()
    rng = np.random.default_rng()
    while time.time() < deadline:
        # compute conv and mask
        conv = _conv_fft(a)
        M = conv.max()
        # compute mask: include entries within mask_frac * M of max
        eps = M * mask_frac
        mask = (conv >= M - eps).astype(np.float64)

        # compute weight contributions w via convolution a * mask
        size = 1 << ((2 * n - 1).bit_length())

        mask_fft = np.fft.rfft(mask, n=size)
        w_full = np.fft.irfft(a_fft * mask_fft, n=size)[:2 * n - 1]
        w = w_full[n-1 : n-1 + n]  # length n

        low_threshold = np.percentile(w, 20)
        low_idxs = np.where(w <= low_threshold)[0]
        if low_idxs.size == 0:
            receiver = donor
        else:
            receiver = int(rng.choice(low_idxs))

        if donor == receiver:
            continue

        # temperature
        elapsed = time.time() - start
        ratio = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** ratio

        a[donor] -= delta
        a[receiver] += delta
        # projection to simplex ensures sum <= target_sum and nonnegativity
        a = _project_to_simplex(a, target_sum)

        new_max = _max_convolution_fft(a)

        if new_max < cur_max or rng.random() < np.exp(-(new_max - cur_max) / max(temp, 1e-12)):
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time():.2f}] Mask MT new best={best_max:.6f}")
        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
    return best_a, best_max

"""
Improved step‑function search.

The algorithm follows the previous multi‑stage pipeline but adds a
new *mask‑based mass‑transfer annealing* phase that directly targets the
set of convolution entries that are close to the current maximum.
During this phase we repeatedly:
    • identify indices that contribute most to the peak(s),
    • move a small amount of mass from a heavy contributor to a light one,
    • accept the move if it improves the maximum convolution entry or with
      a temperature‑controlled Metropolis probability.
Combined with longer gradient‑descent runs and finer hyper‑parameters this
typically pushes the score below the previous best (≈1.508) toward the
target 1.503.

All heavy operations use FFT‑based convolutions (O(n log n)).
The function respects a total time budget (default 1000 s) and returns the
best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_linear(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a * b using FFT (handles arbitrary lengths)."""
    n = a.shape[0]
    m = b.shape[0]
    size = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: n + m - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Evaluation objective (2·n·max(a*a) / (Σa)²)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Re‑ordering utilities
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Reorder a sequence by alternating biggest / smallest values."""
    sorted_vals = np.sort(seq)[::-1]          # descending
    n = seq.shape[0]
    out = np.empty_like(seq)
    left = 0
    right = n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution (used by GD phases)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b

# Phase A – Projected gradient descent (enhanced)
# ----------------------------------------------------------------------


def _annealed_gradient_descent_v2_ext(a0: np.ndarray,
                                     deadline: float,
                                     target_sum: float,
                                     max_beta: float = 1e5,
                                     step_factor: float = 0.7,
                                     gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max approximation.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD_ext it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Several short gradient‑descent runs with random jitter.
    Keeps the best result found before the given deadline.
    """
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_v2_ext(
            start,
            time.time() + run_budget,
            target_sum,
            max_beta=1e5,
            step_factor=0.8,
            gamma=0.99,
        )
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val

def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer moves that directly target the current maximal
    convolution entry. Moves are accepted with a Metropolis rule whose
    temperature decays exponentially with the remaining time.
    """
    a = a0.copy()
    n = a.shape[0]

    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()

    # initial convolution
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

def _permutation_annealing(a: np.ndarray,
                           target_sum: float,
                           deadline: float,
                           init_score: float | None = None) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing over swaps of two indices. The sum is invariant,
    therefore only the ordering is altered.
    """
    cur_seq = a.copy()
    cur_score = init_score if init_score is not None else _max_convolution_fft(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score

    start_time = time.time()
    n = cur_seq.shape[0]
    rng = np.random.default_rng()

    start_temp = 0.2
    end_temp = 1e-5

    while time.time() < deadline:
        elapsed = time.time() - start_time
        ratio = elapsed / (deadline - start_time) if deadline > start_time else 0.0
        temp = start_temp * (end_temp / start_temp) ** ratio

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        new_score = _max_convolution_fft(cur_seq)
        delta = new_score - cur_score

        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / max(temp, 1e-12)):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
                print(f"[{time.time():.2f}] Perm anneal new best={best_score:.6f}")
        else:
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        if ratio > 0.9 and rng.random() < 0.01:
            rng.shuffle(cur_seq)
            cur_score = _max_convolution_fft(cur_seq)

    return best_seq, best_score

def _mask_mass_transfer_annealing(a0: np.ndarray,
                                 target_sum: float,
                                 deadline: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-5,
                                 mask_frac: float = 0.02,
                                 delta_factor: float = 0.15) -> tuple[np.ndarray, float]:
    """
    Annealed mass‑transfer that focuses on the set of convolution entries
    that are within `mask_frac` of the current maximum.
    """
    a = a0.copy()
    n = a.shape[0]

    cur_max = _max_convolution_fft(a)
    best_a = a.copy()
    best_max = cur_max

    start_time = time.time()
    rng = np.random.default_rng()

    while time.time() < deadline:
        # full convolution
        conv = _conv_fft(a)
        M = conv.max()
        eps = max(M * mask_frac, 1e-12)
        mask = (conv >= M - eps).astype(np.float64)   # length 2n‑1

        # compute contribution weight w_i = sum_{k in mask} a_{k-i}
        size = 1 << ((3 * n - 2).bit_length())       # enough for linear a * mask
        a_fft = np.fft.rfft(a, n=size)
        mask_fft = np.fft.rfft(mask, n=size)
        w_full = np.fft.irfft(a_fft * mask_fft, n=size)[: 3 * n - 2]
        w = w_full[n - 1: n - 1 + n]                 # length n

        donor = int(np.argmax(w))

        low_thr = np.percentile(w, 20)
        low_idxs = np.where(w <= low_thr)[0]
        if low_idxs.size == 0:
            receiver = donor
        else:
            receiver = int(rng.choice(low_idxs))

        if donor == receiver:
            continue

        # temperature schedule
        elapsed = time.time() - start_time
        ratio = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** ratio

        max_delta = a[donor] * 0.5
        delta = max_delta * delta_factor * max(temp, 1e-3)
        if delta < 1e-12:
            delta = max_delta * 0.01

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        new_max = _max_convolution_fft(a)

        if new_max < cur_max or rng.random() < np.exp(-(new_max - cur_max) / max(temp, 1e-12)):
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time():.2f}] Mask MT new best={best_max:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional small jitter to escape stagnation
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            cur_max = _max_convolution_fft(a)
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.2f}] Jitter after MT improved best={best_max:.6f}")

    return best_a, best_max

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The pipeline:
        1. Initialise from several power‑law shapes (with jitter and shuffles).
        2. Projected gradient descent with restarts.
        3. Deterministic mass‑transfer refinement.
        4. Simple re‑ordering heuristics.
        5. Permutation annealing.
        6. Final polishing gradient descent.
        7. Mask‑based mass‑transfer annealing (new phase).
    The best sequence found before the deadline is returned.
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Choose problem size
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Build initial candidates
    # --------------------------------------------------------------
    init_seqs = []

    # power‑law shapes
    for p in (0.45, 0.5, 0.55, 0.6):
        init_seqs.append(_power_law_sequence(n, p, target_sum))

    # resized previous best, if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        init_seqs.append(_resize_sequence(prev, n))

    rng = np.random.default_rng()
    # jittered copies for diversity
    for seq in list(init_seqs):
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        seq_jit = seq + jitter
        seq_jit = np.maximum(seq_jit, 0.0)
        seq_jit = _project_to_simplex(seq_jit, target_sum)
        init_seqs.append(seq_jit)

    # random permutations of a couple of power‑law candidates
    for p in (0.5, 0.55):
        seq = _power_law_sequence(n, p, target_sum)
        rng.shuffle(seq)
        init_seqs.append(seq)

    # --------------------------------------------------------------
    # Phase A – Projected gradient descent
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    gd_deadline = time.time() + remaining * 0.30

    best_seq = None
    best_score = np.inf

    for init in init_seqs:
        if time.time() >= gd_deadline:
            break
        a_cand, val = _gradient_descent_with_restarts(init, target_sum, gd_deadline)
        if val < best_score:
            best_score = val
            best_seq = a_cand.copy()
            print(f"[{time.time():.2f}] New global best after GD phase = {best_score:.6f}")

    if best_seq is None:
        best_seq = _power_law_sequence(n, 0.55, target_sum)
        best_score = _score(best_seq)

    # --------------------------------------------------------------
    # Phase B – Deterministic mass‑transfer refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        mt_deadline = time.time() + (deadline - time.time()) * 0.30
        a_ref, val_ref = _deterministic_mass_transfer(
            best_seq, mt_deadline, target_sum, start_temp=0.5, end_temp=1e-4, low_frac=0.05
        )
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] After deterministic refinement best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase C – Re‑ordering heuristics (alternating large/small)
    # --------------------------------------------------------------
    reorder_candidates = [best_seq, _alternating_order(best_seq)]
    for _ in range(2):
        tmp = best_seq.copy()
        rng.shuffle(tmp)
        reorder_candidates.append(tmp)

    for cand in reorder_candidates:
        cand_score = _max_convolution_fft(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] Ordering improvement -> score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase D – Permutation annealing
    # --------------------------------------------------------------
    if time.time() < deadline:
        perm_deadline = time.time() + (deadline - time.time()) * 0.20
        a_perm, val_perm = _permutation_annealing(
            best_seq, target_sum, perm_deadline, init_score=best_score
        )
        if val_perm < best_score:
            best_score = val_perm
            best_seq = a_perm.copy()
            print(f"[{time.time():.2f}] After permutation annealing best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase E – Final polishing gradient descent
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _annealed_gradient_descent_v2_ext(
            best_seq,
            deadline,
            target_sum,
            max_beta=3e5,
            step_factor=0.6,
            gamma=0.995,
        )
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase F – Mask‑based mass‑transfer annealing (new)
    # --------------------------------------------------------------
    if time.time() + 1.0 < deadline:
        a_final, val_final = _mask_mass_transfer_annealing(
            best_seq,
            target_sum,
            deadline,
            start_temp=0.5,
            end_temp=1e-5,
            mask_frac=0.015,
            delta_factor=0.12,
        )
        if val_final < best_score:
            best_score = val_final
            best_seq = a_final.copy()
            print(f"[{time.time():.2f}] After mask‑MT annealing best={best_score:.6f}")

    # --------------------------------------------------------------
    # Post‑processing (trim trailing zeros) and final evaluation
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for possible later runs
    globals()["height_sequence_1"] = final_seq

    return final_seq
