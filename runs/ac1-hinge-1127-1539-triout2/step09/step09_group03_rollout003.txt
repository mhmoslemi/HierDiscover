def _subgradient_descent(a0, target_sum, deadline):
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)
    step = 0.1 * target_sum   # base step for additive updates maybe.
    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        # compute convolution via FFT (maybe compute full conv)
        b = _conv_fft(a)  # returns full conv length 2n-1
        k_max = int(np.argmax(b))
        # gradient of max(b)
        lo = max(0, k_max - (len(a) - 1))
        hi = min(k_max, len(a) - 1)
        idx = np.arange(lo, hi+1)
        grad = np.zeros_like(a)
        grad[idx] = a[idx] + a[k_max - idx]
        # optionally weight gradient to be normalized
        # choose step size proportional to target_sum / np.linalg.norm(grad)
        norm = np.linalg.norm(grad)
        if norm == 0:
            break
        eta = step / norm
        # Perform exponentiated update or additive update
        a = a * np.exp(-eta * grad)  # multiplicative
        # renorm
        a = a * (target_sum / a.sum())
        # Evaluate
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it} score={best_val:.6f}")
        # optionally adjust step after progress
        it += 1
        # if time passes, reduce step size
        if it % 500 == 0:
            step *= 0.7
    return best_a, best_val

import time

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective  2·n·max(a*a) / (sum(a)²).  For Σa=√(2n) this equals max(a*a)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)

def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where conv is the full convolution a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # grad = 2 * (reverse(a) convolve p)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b

    Sub‑gradient of max(b) at the current worst index is computed in O(n).

    a = a * (target_sum / a.sum())
    best_a = a.copy()

            best_a = a.copy()

                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5            # safety margin

    # ------------------------------------------------------------------
    # Choose problem size based on the time budget
    # ------------------------------------------------------------------
    if budget_s >= 950:
        n = 18000
    elif budget_s >= 800:
        n = 12000
    elif budget_s >= 500:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Warm‑start (reuse the best known construction if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
    else:
        # simple decreasing power‑law as a fallback
        init = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))

    init = init * (target_sum / init.sum())

    # ------------------------------------------------------------------
    # Stage 2b – direct sub‑gradient descent (~15 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + 0.15 * (deadline - time.time())
        a_sub, val_sub = _subgradient_descent(init, target_sum, sub_deadline)
        best_seq = a_sub
        best_score = val_sub
        print(f"[{time.time():.2f}] After sub‑gradient descent best score={best_score:.6f}")
    else:
        best_seq = init
        best_score = _score(init)

    # ------------------------------------------------------------------
    # Stage 2 – exponential (multiplicative) descent (~15 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        exp_deadline = time.time() + 0.15 * (deadline - time.time())
        a_exp, val_exp = _exponential_descent(best_seq, target_sum, exp_deadline)
        if val_exp < best_score:
            best_seq = a_exp
            best_score = val_exp
        print(f"[{time.time():.2f}] After exponential descent best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – projected gradient descent with restarts (~35 % budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        gd_deadline = time.time() + 0.35 * (deadline - time.time())
        a_gd, val_gd = _gradient_descent_with_restarts(best_seq,
                                                       target_sum,
                                                       gd_deadline)
        if val_gd < best_score:
            best_seq = a_gd
            best_score = val_gd
        print(f"[{time.time():.2f}] After GD restarts best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – exact local mass‑transfer refinement (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, val_ref = _local_refine_improved(best_seq, target_sum, deadline)
        if val_ref < best_score:
            best_seq = a_ref
            best_score = val_ref
        print(f"[{time.time():.2f}] After local refinement best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Clean up and final evaluation
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in a subsequent call
    globals()["height_sequence_1"] = final_seq

    return final_seq
