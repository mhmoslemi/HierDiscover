    globals()['height_sequence_1'] = final_seq

"""
Improved step‑function optimizer

The evaluator is scale‑invariant; after normalising the ℓ₁‑norm to
sqrt(2·n) the score equals the maximum entry of the self‑convolution.
The algorithm therefore directly minimises that maximum.

Strategy
--------
1. Load any previously found sequence as a warm‑start.
2. Fast random exploration using a richer collection of parametric
   families (including asymmetric Gaussian and Beta shapes) over a
   wide range of lengths up to 25000.
3. Upscale the current best shape to several larger lengths and keep the
   length giving the smallest max‑convolution.
4. Refine the best candidate with
   * Adam (entropic mirror descent) on a τ‑soft‑max approximation of the
     maximum (a smooth log‑sum‑exp surrogate);
   * a short L‑BFGS‑B fine‑tune on the same surrogate (if SciPy is
     available);
   * a sub‑gradient descent on the hard max for final polishing.
5. End with a cheap stochastic hill‑climb that moves a small fraction of
   mass between two positions.
6. Return the final sequence and store it in the global variable
   ``height_sequence_1`` for future runs.

The code respects the 1000 s budget, prints progress messages and
contains all helper functions at top level as required.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, scaling, etc.)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a∗a."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# τ‑soft‑max gradient (used by Adam and L‑BFGS)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient: 2 * conv(mask, a_rev) restricted to the n‑length centre
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser (mirror descent on the simplex)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex (log‑softmax parametrisation).
    Minimises the hard maximum of the convolution by a slowly decaying τ‑soft‑max.
    Returns the best sequence found and its hard peak value.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                     # target ℓ₁‑norm

    # Initialise logits from the feasible a0 (entropic parametrisation)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                         # logits (log‑softmax)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.45

    # temperature schedule – start relatively large for smoothness
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 5.0)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ---- recover a from logits (log‑softmax) ----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ---- τ‑soft‑max objective & gradient ----
        g, peak = _softmax_gradient(a, tau)

        # keep the best *hard* peak seen so far
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ---- gradient w.r.t. logits (chain rule) ----
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 6.0 * S:
            grad_p = grad_p * (6.0 * S / clip_val)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- schedules (very slow decay) ----
        tau *= 0.99995
        lr *= 0.99995

        # occasional jitter to escape plateaus
        if it % 800 == 0:
            p += rng.normal(scale=0.006, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Sub‑gradient descent on the *hard* max (no smoothing)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple sub‑gradient descent that always follows the entry of the
    convolution attaining the current maximum.  After each step the
    vector is projected back onto the simplex (positivity + fixed ℓ₁‑norm).
    Returns the best sequence seen and its hard peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    a = a0.copy()
    best_a = a.copy()
    best_val = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1

        b = _full_convolution_fft(a)
        k = int(b.argmax())
        val = b[k]

        if val < best_val:
            best_val = val
            best_a = a.copy()

        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        jdx = k - idx
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[jdx]

        step = 0.8 * S / (np.sqrt(it) + 1.0)

        a = a - step * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, S / n)
        else:
            a = a * (S / s)

        if it % 500 == 0:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                frac = rng.uniform(0.02, 0.12)
                amount = a[i] * frac
                a[i] -= amount
                a[j] += amount
                a = a * (S / a.sum())

    return best_a, best_val

def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    raw = mix * edge + (1.0 - mix) * np.ones(n)
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    raw = mix * np.exp(-lam * d) + (1.0 - mix) * np.ones(n)
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    raw = mix * lin + (1.0 - mix) * rng.uniform(0.0, 0.25, size=n)
    return raw


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    raw = mix * prod.astype(float) ** (-p) + (1.0 - mix) * np.ones(n)
    return raw


def _init_cosine(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    raw = 1.0 - np.cos(np.pi * i / (n - 1))
    raw = raw * rng.uniform(0.5, 2.0)
    mix = rng.uniform(0.0, 1.0)
    raw = mix * raw + (1.0 - mix) * np.ones(n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    return rng.random(n)


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.05, 0.9)
    return rng.gamma(shape, 1.0, size=n)


def _init_asym_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    """Gaussian bump placed at a random offset (asymmetric)."""
    idx = np.arange(n)
    offset = rng.uniform(0.2, 0.8) * (n - 1)
    sigma = rng.uniform(0.1, 0.35) * n
    raw = np.exp(-((idx - offset) ** 2) / (2.0 * sigma ** 2))
    return raw


def _init_beta_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Beta‑distribution‑shaped profile on [0,1]."""
    x = np.linspace(0.0, 1.0, n)
    alpha = rng.uniform(0.5, 5.0)
    beta = rng.uniform(0.5, 5.0)
    # add a tiny epsilon to avoid infinities when alpha<1 or beta<1
    eps = 1e-12
    raw = (np.maximum(x, eps) ** (alpha - 1.0)) * (np.maximum(1.0 - x, eps) ** (beta - 1.0))
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    With probability ≈0.2 we warm‑start from ``warm_seq`` (scaled).
    Otherwise we draw from one of the random families (now richer).
    The returned vector satisfies sum = sqrt(2·n) and all entries ≥0.
    """
    S = np.sqrt(2.0 * n)

    if warm_seq is not None and rng.random() < 0.23:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
        return a

    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'gamma', 'uniform', 'dirichlet',
                        'ushape', 'cosine', 'asym_gaussian', 'beta'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'cosine':
        raw = _init_cosine(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'asym_gaussian':
        raw = _init_asym_gaussian(n, rng)
    elif style == 'beta':
        raw = _init_beta_shape(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a

def _lbfgs_optimize(a0: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    L‑BFGS‑B optimisation of the τ‑soft‑max surrogate.
    Starts from logits derived from ``a0`` and stops at ``deadline``.
    Returns the refined sequence and its hard peak value.
    """
    if not _has_scipy:
        return a0, _max_convolution(a0)

    n = len(a0)
    S = np.sqrt(2.0 * n)

    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p0 = np.log(soft)

    # A small but non‑zero temperature gives a smooth objective.
    tau = 0.05

    def obj_and_grad(p):
        # a from logits
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # smooth max (log‑sum‑exp)
        b = _full_convolution_fft(a)
        val = tau * np.log(np.exp(b / tau).sum())

        # gradient w.r.t. a
        g_a, _ = _softmax_gradient(a, tau)

        # chain rule to logits
        mu = np.dot(a, g_a) / S
        grad_p = a * (g_a - mu)
        return val, grad_p.astype(np.float64)

    # SciPy does not allow a hard time limit inside `minimize`,
    # so we manually enforce it by a simple callback.
    start_time = time.time()
    maxiter = 500

    def callback(xk):
        if time.time() > deadline:
            raise RuntimeError("time limit reached")

    try:
        res = minimize(lambda p: obj_and_grad(p)[0],
                       p0,
                       jac=lambda p: obj_and_grad(p)[1],
                       method='L-BFGS-B',
                       callback=callback,
                       options={'maxiter': maxiter,
                                'ftol': 1e-12,
                                'disp': False})
    except RuntimeError:
        # time limit hit – fall back to current best
        pass

    p_opt = res.x if 'res' in locals() else p0
    maxp = np.max(p_opt)
    exp_p = np.exp(p_opt - maxp)
    soft = exp_p / exp_p.sum()
    a_opt = soft * S
    peak = _max_convolution(a_opt)
    return a_opt, peak

            n = int(rng.integers(2000, 8000))
        else:
            n = int(rng.integers(8000, 25000))

        a0 = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a0)

        if peak < best_score:
            best_score = peak
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempts:4d} n={n:5d}  new best={best_score:.6f}")

    # fallback in the extremely unlikely case that no candidate was generated
    if best_seq is None:
        best_n = 4000
        best_seq = np.full(best_n, np.sqrt(2.0 * best_n) / best_n)
        best_score = _max_convolution(best_seq)

    # ------------------------------------------------------------------
    # Phase 2 – upscale / downscale to a few larger lengths
    # ------------------------------------------------------------------
    cand_lengths = [best_n]
    for factor in [1.25, 1.5, 2.0, 3.0, 4.0, 5.0]:
        cand = int(best_n * factor)
        if 2000 <= cand <= 25000:
            cand_lengths.append(cand)
    cand_lengths = sorted(set(cand_lengths))

    cur_seq = best_seq
    cur_score = best_score
    cur_n = best_n

    for L in cand_lengths:
        if time.time() > end_time - 300.0:
            break
        a_scaled = _scale_sequence(cur_seq, L)
        a_scaled = np.clip(a_scaled, 0.0, None)
        S = np.sqrt(2.0 * L)
        a_scaled = a_scaled * (S / a_scaled.sum())
        peak = _max_convolution(a_scaled)
        if peak < cur_score:
            cur_score = peak
            cur_seq = a_scaled.copy()
            cur_n = L
            print(f"[scale] length={L:5d}  improved max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam optimisation on the current best length
    # ------------------------------------------------------------------
    remaining = end_time - time.time() - 120.0   # reserve time for later phases
    if remaining > 15.0:
        n_restart = max(1, int(remaining // 60.0))
        per_run = remaining / n_restart
        for r in range(n_restart):
            if time.time() > end_time - 120.0:
                break
            # add a small random perturbation to encourage diversity
            noise = rng.normal(scale=0.015 * np.sqrt(2.0 * cur_n) / cur_n, size=cur_n)
            a_start = np.maximum(cur_seq + noise, 0.0)
            a_start = a_start * (np.sqrt(2.0 * cur_n) / a_start.sum())
            a_opt, peak = _adam_optimize(a_start,
                                          time.time() + per_run,
                                          rng)
            if peak < cur_score:
                cur_score = peak
                cur_seq = a_opt.copy()
                print(f"[adam] restart={r} new best max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – optional L‑BFGS fine‑tuning (if SciPy is available)
    # ------------------------------------------------------------------
    remaining = end_time - time.time() - 90.0
    if remaining > 30.0 and _has_scipy:
        try:
            a_opt, peak = _lbfgs_optimize(cur_seq, time.time() + remaining, rng)
            if peak < cur_score:
                cur_score = peak
                cur_seq = a_opt.copy()
                print(f"[lbfgs] improved max={cur_score:.6f}")
        except Exception as e:
            print("[lbfgs] failed:", e)

    # ------------------------------------------------------------------
    # Phase 5 – sub‑gradient descent (final polishing)
    # ------------------------------------------------------------------
    remaining = end_time - time.time() - 10.0
    if remaining > 8.0:
        a_opt, peak = _subgradient_descent(cur_seq,
                                           time.time() + remaining,
                                           rng)
        if peak < cur_score:
            cur_score = peak
            cur_seq = a_opt.copy()
            print(f"[subgrad] final max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – cheap stochastic hill‑climb (few seconds)
    # ------------------------------------------------------------------
    while time.time() < end_time - 1.0:
        a = cur_seq.copy()
        n = a.shape[0]
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        frac = rng.uniform(0.01, 0.14)
        amount = a[i] * frac
        a[i] -= amount
        a[j] += amount
        a = a * (np.sqrt(2.0 * n) / a.sum())
        peak = _max_convolution(a)
        if peak + 1e-9 < cur_score:   # strict improvement
            cur_score = peak
            cur_seq = a.copy()
            print(f"[hill] improved max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    final_seq = cur_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f}  internal max={cur_score:.6f}  length={len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
