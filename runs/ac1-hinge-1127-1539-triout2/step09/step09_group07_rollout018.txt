"""
Propose a step‑function sequence that minimises ``evaluate_sequence``.

The algorithm works in three stages.
1. **Exploration** – many random starts of moderate length (≈2‑8 k) are
   generated from a rich set of parametric families.  The best ~30
   candidates are retained.
2. **Multi‑scale refinement** – each retained candidate is repeatedly
   *up‑sampled* to larger lengths (up to 80 k) and optimised with a
   Adam optimiser that uses the exact sub‑gradient of the hard max of the
   self‑convolution.  Hyper‑parameters (learning rate, temperature decay)
   are varied for diversity.
3. **Fine‑tuning** – the current global best is taken at the maximum
   length, optimised for a longer time, then improved by a cheap
   simulated‑annealing walk, a strict greedy hill‑climber and a
   simple smoothing step.  The final sequence is returned and stored in
   ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions for convolution (FFT based, linear‑time for large n)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Peak value of a*a (full convolution)."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Exact (hard‑max) sub‑gradient of the peak convolution
# ----------------------------------------------------------------------
def _max_gradient(a: np.ndarray, tau: float):
    """
    Returns (grad, peak) where grad = ∂ max(b) / ∂a
    and peak = max(b) for b = a*a (full convolution).
    If ``tau`` is tiny we use the exact active set of indices
    attaining the maximum; otherwise a τ‑soft‑max is used.
    """
    b = _full_convolution_fft(a)
    peak = b.max()

    if tau < 1e-12:
        eps = peak * 1e-9 + 1e-12
        mask = (b >= peak - eps).astype(float)
        s = mask.sum()
        if s == 0:
            mask = np.ones_like(b) / b.size
        else:
            mask /= s
    else:
        diff = b - peak
        exps = np.exp(diff / tau)
        mask = exps / exps.sum()

    # gradient = 2 * conv(mask, a_rev) restricted to the original length
    a_rev = a[::-1]
    conv_len = mask.size + a.size - 1
    f_len = _next_pow2(conv_len)

    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)

    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    grad = 2.0 * conv[a.size - 1:a.size - 1 + a.size]
    return grad, peak


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   lr0: float = 0.55,
                   tau_decay: float = 0.9995,
                   lr_decay: float = 0.99985,
                   jitter_every: int = 400) -> tuple[np.ndarray, float]:
    """
    Minimise the peak of the self‑convolution under the ℓ₁‑norm constraint.
    Returns the best vector seen (still ℓ₁‑normalised to sqrt(2·n))
    and its hard peak value.
    """
    n = a0.shape[0]
    S = math.sqrt(2.0 * n)                # required ℓ₁‑norm

    # initialise logits from the feasible a0 (log‑softmax)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                       # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr0

    # initialise temperature for τ‑soft‑max
    tau = max(_full_convolution_fft(a0).max(), 0.5)

    best_a = a0.copy()
    best_peak = _max_convolution(a0)
    it = 0

    while time.time() < deadline:
        it += 1

        # ------- recover a from logits (log‑softmax) -------
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)               # shift for stability
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ------- gradient of (soft‑)max convolution -------
        g, peak = _max_gradient(a, tau)

        # keep best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ------- gradient w.r.t. logits (chain rule) -------
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ------- Adam update -------
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ------- annealing schedules -------
        tau *= tau_decay
        lr *= lr_decay

        # occasional jitter to escape plateaus
        if jitter_every and (it % jitter_every) == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak

def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Gamma‑draw → Dirichlet‑like vector."""
    alpha = rng.uniform(0.2, 5.0)
    raw = rng.gamma(alpha, 1.0, size=n)
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sparse Gamma entries (shape < 1)."""
    shape = rng.uniform(0.1, 0.8)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    """All ones."""
    return np.full(n, 1.0)


def _init_left_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotone decreasing power‑law, optionally mixed with a uniform background."""
    idx = np.arange(n)
    p = rng.uniform(0.3, 3.0)                # exponent
    offset = rng.uniform(0.0, 10.0)           # horizontal shift
    mix = rng.uniform(0.0, 1.0)               # background weight
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * (idx + offset) ** (-p)
    return raw


def _init_right_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotone increasing power‑law."""
    idx = np.arange(n)[::-1]                 # reverse indices
    p = rng.uniform(0.3, 3.0)
    offset = rng.uniform(0.0, 10.0)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * (idx + offset) ** (-p)
    return raw


def _init_left_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotone decreasing exponential, optionally mixed with a uniform background."""
    idx = np.arange(n)
    lam = rng.uniform(0.001, 0.02)           # decay rate
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * np.exp(-lam * idx)
    return raw


def _init_sparse_bumps(n: int, rng: np.random.Generator) -> np.ndarray:
    """Few random rectangular bumps on a tiny flat background."""
    raw = np.full(n, rng.uniform(0.0, 0.15))
    nbumps = rng.integers(1, max(2, n // 250))
    for _ in range(nbumps):
        center = rng.integers(0, n)
        width = rng.integers(3, max(5, n // 100))
        height = rng.uniform(0.8, 4.0)
        lo = max(0, center - width // 2)
        hi = min(n, lo + width)
        raw[lo:hi] = np.maximum(raw[lo:hi], height)
    return raw


def _init_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    """Centered Gaussian bump (continuous) with random width."""
    mu = (n - 1) / 2.0
    sigma = rng.uniform(0.05, 0.30) * n   # 5‑30 % of the length
    x = np.arange(n)
    raw = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    return raw


def _init_half_sine(n: int, rng: np.random.Generator) -> np.ndarray:
    """Half‑sine shape (zero at the ends, peak in the middle)."""
    x = np.arange(n)
    raw = np.sin(np.pi * x / (n - 1))
    raw = raw * rng.uniform(0.5, 2.0) + rng.uniform(0.0, 0.2)
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Occasionally warm‑start from the global best, otherwise draw from a random family.
    """
    S = math.sqrt(2.0 * n)

    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    style = rng.choice(['dirichlet', 'gamma', 'uniform',
                        'left_powerlaw', 'right_powerlaw',
                        'left_exponential', 'sparse', 'gaussian', 'half_sine'])
    if style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'left_powerlaw':
        raw = _init_left_powerlaw(n, rng)
    elif style == 'right_powerlaw':
        raw = _init_right_powerlaw(n, rng)
    elif style == 'left_exponential':
        raw = _init_left_exponential(n, rng)
    elif style == 'sparse':
        raw = _init_sparse_bumps(n, rng)
    elif style == 'gaussian':
        raw = _init_gaussian(n, rng)
    else:  # half_sine
        raw = _init_half_sine(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Simple stochastic fine‑tuner (mass‑transfer walk)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Mass‑transfer walk with exponential acceptance."""
    n = len(seq)
    S = math.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score
    temperature = 0.15 * cur_score

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] == 0.0:
            continue
        delta = rng.uniform(0.0, 0.3) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())
        new_score = _max_convolution(new)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temperature, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

        temperature *= 0.99995
        if temperature < 1e-6:
            temperature = 1e-6

    return best, best_score


def _hill_climb(seq: np.ndarray,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Strict downhill moves only (very cheap)."""
    n = len(seq)
    S = math.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] < 1e-12:
            continue
        delta = rng.uniform(0.0, 0.15) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())
        new_score = _max_convolution(new)
        if new_score < cur_score:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score
    return best, best_score


def _smooth_and_improve(seq: np.ndarray,
                        deadline: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Apply a tiny three‑point smoothing repeatedly (accepting only
    improvements).  This often irons out tiny spikes left from the
    gradient phase.
    """
    n = len(seq)
    S = math.sqrt(2.0 * n)
    kernel = np.array([0.25, 0.5, 0.25], dtype=float)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score

    while time.time() < deadline:
        # smooth (same size, reflect at the boundaries)
        padded = np.pad(cur, (1, 1), mode='edge')
        smoothed = np.convolve(padded, kernel, mode='valid')
        smoothed = smoothed * (S / smoothed.sum())
        smoothed_score = _max_convolution(smoothed)

        if smoothed_score < best_score:
            best = smoothed.copy()
            best_score = smoothed_score
            cur = smoothed
            cur_score = smoothed_score
        else:
            # occasional random jitter to escape tiny plateaus
            if rng.random() < 0.07:
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i != j and cur[i] > 0:
                    delta = rng.uniform(0.0, 0.2) * cur[i]
                    cur[i] -= delta
                    cur[j] += delta
                    cur = cur * (S / cur.sum())
                    cur_score = _max_convolution(cur)

    return best, best_score

    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the given time budget, prints progress, stores the
    best construction in the global variable ``height_sequence_1`` and returns
    a plain Python list.
    """
    rng = np.random.default_rng(seed)
    safety_margin = 5.0
    end_time = time.time() + max(budget_s - safety_margin, 0.0)
    if end_time <= time.time():
        raise RuntimeError("Budget too small after safety margin.")

    # ------------------------------------------------------------------
    # Warm‑start from a previously stored construction (if any)
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # ------------------------------------------------------------------
    # Phase 0 – fast random exploration (≈15 % of the budget)
    # ------------------------------------------------------------------
    explore_deadline = time.time() + 0.15 * budget_s
    top_k = 30
    top_candidates = []          # list of (score, seq)
    best_seq = None
    best_score = np.inf
    attempt = 0

    while time.time() < explore_deadline:
        attempt += 1
        # length distribution – favour the 3‑8 k band but occasionally larger
        if rng.random() < 0.75:
            n = int(rng.integers(2500, 7000))
        else:
            n = int(rng.integers(7000, 15000))

        a0 = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a0)

        if peak < best_score:
            best_score = peak
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempt:5d} n={n:5d}  new best={best_score:.6f}")

        # maintain top‑k list (sorted increasingly)
        if len(top_candidates) < top_k:
            top_candidates.append((peak, a0.copy()))
            top_candidates.sort(key=lambda x: x[0])
        elif peak < top_candidates[-1][0]:
            top_candidates[-1] = (peak, a0.copy())
            top_candidates.sort(key=lambda x: x[0])

    # safety net
    if best_seq is None:
        n = 4000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n)
        best_score = _max_convolution(best_seq)
        best_n = n
        top_candidates = [(best_score, best_seq.copy())]

    # ------------------------------------------------------------------
    # Phase 1 – multi‑scale Adam refinement (≈70 % of the budget)
    # ------------------------------------------------------------------
    # geometric length schedule (up to the algorithmic ceiling)
    max_len = min(80000, best_n * 12)
    lengths = [best_n]
    cur_len = best_n
    while cur_len * 1.5 < max_len:
        cur_len = int(cur_len * 1.5)
        lengths.append(cur_len)
    lengths.append(max_len)
    lengths = sorted(set(lengths))

    for init_idx, (init_score, init_seq) in enumerate(top_candidates):
        cur_seq = init_seq.copy()
        for target_n in lengths:
            if time.time() >= end_time - 30.0:
                break

            # interpolate to target length and renormalise
            a_scaled = _scale_sequence(cur_seq, target_n)
            a_scaled = np.clip(a_scaled, 0.0, None)
            a_scaled = a_scaled * (math.sqrt(2.0 * target_n) / a_scaled.sum())

            # allocate a slice of remaining time (but never more than 120 s)
            remaining = end_time - time.time()
            slice_budget = min(remaining * 0.45, 120.0)

            # randomise a few hyper‑parameters for diversity
            lr0 = rng.choice([0.48, 0.55, 0.62, 0.70])
            tau_decay = rng.choice([0.9993, 0.9995, 0.9997])
            refined, peak = _adam_optimize(a_scaled,
                                           time.time() + slice_budget,
                                           rng,
                                           lr0=lr0,
                                           tau_decay=tau_decay,
                                           lr_decay=0.99985,
                                           jitter_every=350)

            if peak < best_score:
                best_score = peak
                best_seq = refined.copy()
                best_n = target_n
                print(f"[refine] init#{init_idx+1} len={target_n:5d}  improved score={best_score:.6f}")

            cur_seq = refined

        if time.time() >= end_time - 30.0:
            break

    # ------------------------------------------------------------------
    # Phase 2 – long‑run high‑resolution optimisation (≈10 % of the budget)
    # ------------------------------------------------------------------
    remaining = end_time - time.time()
    if remaining > 30.0:
        # take the current best, push it to the maximal length
        big_n = max_len
        a_big = _scale_sequence(best_seq, big_n)
        a_big = np.clip(a_big, 0.0, None)
        a_big = a_big * (math.sqrt(2.0 * big_n) / a_big.sum())

        # a more generous time slice
        slice_deadline = time.time() + remaining * 0.65
        a_big, big_peak = _adam_optimize(a_big,
                                          slice_deadline,
                                          rng,
                                          lr0=0.65,
                                          tau_decay=0.9996,
                                          lr_decay=0.9999,
                                          jitter_every=300)
        if big_peak < best_score:
            best_score = big_peak
            best_seq = a_big.copy()
            best_n = big_n
            print(f"[big‑refine] length={big_n}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – cheap stochastic fine‑tuning
    # ------------------------------------------------------------------
    remaining = end_time - time.time()
    if remaining > 5.0:
        # simulated annealing first
        best_seq, best_score = _anneal(best_seq,
                                       time.time() + 0.6 * remaining,
                                       rng)
        remaining = end_time - time.time()
        # greedy hill‑climber
        if remaining > 3.0:
            best_seq, best_score = _hill_climb(best_seq,
                                               time.time() + remaining,
                                               rng)

    # final smoothing pass (very cheap)
    remaining = end_time - time.time()
    if remaining > 1.0:
        best_seq, best_score = _smooth_and_improve(best_seq,
                                                   time.time() + remaining,
                                                   rng)

    # ------------------------------------------------------------------
    # Final sanity check via the supplied black‑box evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  internal max = {best_score:.6f}"
          f"  length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
