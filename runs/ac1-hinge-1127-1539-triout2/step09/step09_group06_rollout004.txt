"""
Step‑function optimizer – hierarchical length expansion + Adam + sub‑gradient.

The objective is  M = max_k (a*a)_k  after normalising
    sum_i a_i = sqrt(2·n) .
M is exactly the value returned by ``evaluate_sequence`` (no other scaling).

Strategy
--------
1. Load a previously found sequence (if any) for warm‑starting.
2. Choose a modest base length (≈2000) and optimise it with Adam on a
   τ‑soft‑max approximation of the maximum.  Several random restarts are
   performed.
3. Iteratively enlarge the sequence by linear interpolation (factors
   1.2, 1.5, 2.0, 3.0 … up to a hard cap of 25000).  After each upscale we
   continue Adam optimisation – the shape can adapt to the larger support.
4. A final sub‑gradient descent on the *hard* maximum refines the result.
5. A cheap stochastic hill‑climb explores tiny mass transfers.
6. The best sequence is returned (and stored in the global variable
   ``height_sequence_1``).

All optimisation respects the 1000 s budget.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = rfft(a, n=f_len)
    conv = irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a∗a."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(a*a).
    Returns (grad, max_b) where grad = ∂/∂a (τ‑soft‑max).
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        # stable soft‑max
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # grad = 2 * (mask ⋆ a_rev)
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Mmask = rfft(mask, n=f_len)
    Ma_rev = rfft(a_rev, n=f_len)
    conv = irfft(Mmask * Ma_rev, n=f_len)[:conv_len]
    # extract the part that aligns with a (indices n‑1 … 2n‑2)
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, deadline: float, rng: np.random.Generator):
    """
    Adam optimiser on the simplex (log‑softmax).
    Returns (best_a, best_peak) where best_peak = max_k (a*a)_k.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)          # required ℓ1‑norm

    # initialise logits from a feasible a0 (entropic parametrisation)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)               # logits (log‑softmax)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.45

    # initial temperature – start relatively large for a smooth landscape
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 5.0)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # --- recover a from logits (log‑softmax) ---
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # --- gradient of τ‑soft‑max ---
        g, peak = _softmax_gradient(a, tau)

        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # --- chain rule: gradient wrt logits ---
        mu = np.dot(a, g) / S          # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)

        # optional clipping for numerical stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 6.0 * S:
            grad_p = grad_p * (6.0 * S / clip_val)

        # --- Adam update ---
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # --- schedules (very slow decay) ---
        tau *= 0.99990          # keep smoothing a bit longer
        lr *= 0.99990           # gentle learning‑rate decay

        # jitter to escape shallow plateaus
        if it % 1000 == 0:
            p += rng.normal(scale=0.004, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Sub‑gradient descent on the *hard* max
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, deadline: float, rng: np.random.Generator):
    """
    Simple sub‑gradient descent that follows the exact maximum entry.
    Returns (best_a, best_peak).
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    a = a0.copy()
    best_a = a.copy()
    best_val = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1

        b = _full_convolution_fft(a)
        k = int(b.argmax())
        val = b[k]

        if val < best_val:
            best_val = val
            best_a = a.copy()

        # sub‑gradient of (a*a)_k  (only indices that actually appear)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        jdx = k - idx
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[jdx]

        # diminishing step size
        step = 0.85 * S / (np.sqrt(it) + 1.0)

        a = a - step * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, S / n)
        else:
            a = a * (S / s)

        # occasional random jitter to avoid tiny plateaus
        if it % 800 == 0:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                frac = rng.uniform(0.01, 0.08)
                amount = a[i] * frac
                a[i] -= amount
                a[j] += amount
                a = a * (S / a.sum())

    return best_a, best_val

def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    raw = mix * ush + (1.0 - mix) * base
    return raw


def _init_cosine(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    raw = 1.0 - np.cos(np.pi * i / (n - 1))
    raw = raw * rng.uniform(0.5, 2.0)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * raw + (1.0 - mix) * base


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    # draw from a Dirichlet with concentration 0.5…1.5
    alpha = rng.uniform(0.3, 1.7, size=n)
    raw = rng.gamma(alpha, 1.0)
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.05, 0.9)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_beta(n: int, rng: np.random.Generator) -> np.ndarray:
    # Beta‑like (x^(a-1)*(1-x)^(b-1)) on a regular grid
    a = rng.uniform(0.2, 3.0)
    b = rng.uniform(0.2, 3.0)
    x = np.linspace(0.0, 1.0, n)
    raw = (x ** (a - 1)) * ((1 - x) ** (b - 1) + 1e-12)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * raw + (1.0 - mix) * base


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Either warm‑starts from ``warm_seq`` (scaled) or draws from a random family.
    The vector satisfies sum = sqrt(2·n) and all entries ≥0.
    """
    S = np.sqrt(2.0 * n)

    # occasional warm‑start – useful when a good shape is already known
    if warm_seq is not None and rng.random() < 0.23:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
        return a

    # pick a family
    families = ['powerlaw', 'exponential', 'linear', 'gamma',
                'uniform', 'dirichlet', 'ushape', 'cosine', 'beta']
    style = rng.choice(families)
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'cosine':
        raw = _init_cosine(n, rng)
    elif style == 'beta':
        raw = _init_beta(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):

    Returns the best sequence as a plain Python list and stores it in the
    global variable ``height_sequence_1`` for future runs.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Warm‑start handling
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # ------------------------------------------------------------------
    # Phase 0 – choose a base length and obtain a decent start
    # ------------------------------------------------------------------
    base_n = int(rng.integers(1800, 3000))
    a_cur = _initial_vector(base_n, rng, warm_seq)
    cur_peak = _max_convolution(a_cur)
    print(f"[init] base length={base_n}  peak={cur_peak:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam on the base length (several restarts)
    # ------------------------------------------------------------------
    remaining = end_time - time.time() - 350.0   # reserve later phases
    n_restart = max(1, int(remaining // 70))     # ~70 s per run at base size
    per_run = remaining / n_restart if n_restart > 0 else 0.0

    for r in range(n_restart):
        if time.time() > end_time - 350.0:
            break
        # perturb a little before each restart
        noise = rng.normal(scale=0.025 * np.sqrt(2.0 * base_n) / base_n, size=base_n)
        a_start = np.maximum(a_cur + noise, 0.0)
        a_start = a_start * (np.sqrt(2.0 * base_n) / a_start.sum())
        a_opt, peak = _adam_optimize(a_start,
                                      time.time() + per_run,
                                      rng)
        if peak < cur_peak:
            cur_peak = peak
            a_cur = a_opt.copy()
            print(f"[adam‑base] restart={r}  new best peak={cur_peak:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – hierarchical up‑scaling
    # ------------------------------------------------------------------
    # list of scaling factors (progressively larger)
    scale_factors = [1.25, 1.5, 2.0, 3.0, 4.0]
    for factor in scale_factors:
        if time.time() > end_time - 250.0:
            break
        new_n = int(min(25000, a_cur.shape[0] * factor))
        if new_n == a_cur.shape[0]:
            continue

        a_up = _scale_sequence(a_cur, new_n)
        a_up = np.clip(a_up, 0.0, None)
        S = np.sqrt(2.0 * new_n)
        a_up = a_up * (S / a_up.sum())
        up_peak = _max_convolution(a_up)
        print(f"[scale‑up]  old={a_cur.shape[0]} → new={new_n}  peak={up_peak:.6f}")

        # allocate a modest amount of time for Adam at the new size
        time_allow = min(120.0, (end_time - time.time()) / 2.0)
        a_opt, peak = _adam_optimize(a_up,
                                      time.time() + time_allow,
                                      rng)
        if peak < cur_peak:
            cur_peak = peak
            a_cur = a_opt.copy()
            print(f"[adam‑up]  factor={factor:.2f}  improved peak={cur_peak:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Sub‑gradient descent for final polishing
    # ------------------------------------------------------------------
    if time.time() < end_time - 120.0:
        a_opt, peak = _subgradient_descent(a_cur,
                                           time.time() + (end_time - time.time() - 120.0),
                                           rng)
        if peak < cur_peak:
            cur_peak = peak
            a_cur = a_opt.copy()
            print(f"[subgrad] final polishing peak={cur_peak:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – cheap stochastic hill‑climb (the last few seconds)
    # ------------------------------------------------------------------
    while time.time() < end_time - 1.0:
        a = a_cur.copy()
        n = a.shape[0]
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        frac = rng.uniform(0.02, 0.11)
        amount = a[i] * frac
        a[i] -= amount
        a[j] += amount
        # renormalise (numerical drift)
        a = a * (np.sqrt(2.0 * n) / a.sum())
        new_peak = _max_convolution(a)
        if new_peak + 1e-9 < cur_peak:
            cur_peak = new_peak
            a_cur = a.copy()
            print(f"[hill] improved peak={cur_peak:.6f}")

    # ------------------------------------------------------------------
    # Final verification via the supplied evaluator
    # ------------------------------------------------------------------
    final_seq = a_cur.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f}  internal max={cur_peak:.6f}  length={len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
