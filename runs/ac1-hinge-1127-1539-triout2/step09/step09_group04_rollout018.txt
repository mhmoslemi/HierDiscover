"""
Propose a non‑negative step sequence with a low benchmark score.

Algorithm
---------
1. Choose a problem size (≈20 000 for the full budget – larger n gives a
   tighter theoretical lower bound 2 n/(2 n‑1) → 1).
2. Warm‑start from the previously found sequence (global variable
   ``height_sequence_1``) or from a simple power‑law shape.
3. Perform a cheap random search over a two‑parameter family
   a_i = (i+1)^{-p}·exp(-λ·i/n) on the full size.
4. Refine the shape with several optimisation stages, all using the fast
   FFT‑based convolution:
   * mirror‑descent on a soft‑max surrogate of max (a*a);
   * projected gradient descent on the same surrogate;
   * exact sub‑gradient descent on the true maximal convolution entry;
   * aggressive greedy mass‑transfer that moves a fraction of the mass
     from the index contributing most to the maximal entry to a low‑mass
     index.
5. A final smoothing step (3‑point moving average) followed by projection
   onto the ℓ¹‑simplex.
6. Trim trailing zeros, evaluate with the provided ``evaluate_sequence`` and
   store the result for warm‑starts.

All steps keep the ℓ¹‑norm fixed (Σa = √(2 n)) and preserve non‑negativity.
The routine respects the 1000 s budget and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based – O(n log n))
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) shape, normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Gradient of the soft‑max surrogate of max‐convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(conv).
    Returns (grad, conv) where conv is the full convolution a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()          # soft‑max weights over all shifts

    # grad = 2 * (reverse(a) convolve p)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b

# Optimisation stages
# ----------------------------------------------------------------------


def _softmax_mirror_descent(a0: np.ndarray,
                            target_sum: float,
                            deadline: float) -> tuple[np.ndarray, float]:
    """Multiplicative mirror‑descent on the soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 16384.0
    base_eta = 0.02
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)                # multiplicative update
        a = a * (target_sum / a.sum())          # renormalise (keeps sum)

        # occasional jitter to avoid stagnation
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] MirrGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best, best_val


def _projected_gradient_descent(a0: np.ndarray,
                               target_sum: float,
                               deadline: float) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the soft‑max surrogate."""
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 16384.0
    step_coeff = target_sum * 0.4
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.9
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_coeff / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # jitter occasional
        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] ProjGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 600 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best, best_val


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on max (a*a)."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(conv.argmax())

        # Build sub‑gradient for this shift
        n = a.shape[0]
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = kmax - idx
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[partner]

        # maximal safe step that keeps a non‑negative
        pos = grad > 0
        if not pos.any():
            break
        safe_eta = np.min(a[pos] / grad[pos])
        eta = min(0.9 * safe_eta, 0.5 / np.sqrt(it + 1))

        a = a - eta * grad
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        # small jitter every few iterations
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        it += 1

    return best, best_val


def _greedy_mass_transfer(a0: np.ndarray,
                          target_sum: float,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Aggressive greedy transfer of mass from the index that contributes most
    to the current maximal convolution entry to a low‑mass index.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(conv.argmax())

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = kmax - idx
        weight = a[idx] * a[partner]

        donor = idx[np.argmax(weight)]

        # choose a receiver among the smallest values (bottom 0.5 %)
        low_cnt = max(1, n // 200)
        low_idx = np.argpartition(a, low_cnt)[:low_cnt]
        receiver = rng.choice(low_idx)

        if donor == receiver:
            continue

        # move a modest fraction (5‑20 %) of the donor's mass
        frac = rng.uniform(0.05, 0.20)
        delta = a[donor] * frac
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta

        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] MassTransfer improved to {best_val:.6f}")

        # renormalise numerical drift
        cur = a.sum()
        if abs(cur - target_sum) > 1e-9:
            a = a * (target_sum / cur)

    return best, best_val


def _smoothening_stage(a0: np.ndarray,
                       target_sum: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """Apply a few iterations of 3‑point moving‑average smoothing."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best = a.copy()
    best_val = _score(a)

    kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)

    while time.time() < deadline:
        a_s = np.convolve(a, kernel, mode='same')
        a_s = _project_to_simplex(a_s, target_sum)

        val = _score(a_s)
        if val < best_val:
            best_val = val
            best = a_s.copy()
            print(f"[{time.time():.2f}] Smoothing improved to {best_val:.6f}")

        a = a_s

    return best, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # small safety margin

    # ------------------------------------------------------------------
    # Choose problem size – larger n gives a tighter lower bound.
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 400:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Warm‑start (use previous best if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = np.array(globals()["height_sequence_1"], dtype=np.float64)
        init = _resize_sequence(init, n)
        init = init * (target_sum / init.sum())
    else:
        init = _power_law_sequence(n, p=0.55, target_sum=target_sum)

    best_seq = init.copy()
    best_val = _score(best_seq)
    print(f"[{time.time():.2f}] Initial score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 0 – cheap random exploration of a 2‑parameter family
    # ------------------------------------------------------------------
    param_deadline = time.time() + min(30.0,
                                       (deadline - time.time()) * 0.10)
    rng = np.random.default_rng()
    i = np.arange(n, dtype=np.float64)

    while time.time() < param_deadline:
        p = rng.uniform(0.2, 1.2)
        lam = rng.uniform(0.0, 8.0)
        seq = (i + 1) ** (-p) * np.exp(-lam * i / n)
        seq = seq * (target_sum / seq.sum())
        val = _score(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Random param better: p={p:.3f} λ={lam:.3f} score={best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – mirror‑descent on soft‑max surrogate
    # ------------------------------------------------------------------
    if time.time() < deadline:
        md_dead = time.time() + (deadline - time.time()) * 0.20
        a_md, v_md = _softmax_mirror_descent(best_seq, target_sum, md_dead)
        if v_md < best_val:
            best_val, best_seq = v_md, a_md.copy()
        print(f"[{time.time():.2f}] After mirror‑descent best score={best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – projected gradient descent on soft‑max surrogate
    # ------------------------------------------------------------------
    if time.time() < deadline:
        pg_dead = time.time() + (deadline - time.time()) * 0.20
        a_pg, v_pg = _projected_gradient_descent(best_seq, target_sum, pg_dead)
        if v_pg < best_val:
            best_val, best_seq = v_pg, a_pg.copy()
        print(f"[{time.time():.2f}] After proj‑GD best score={best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – exact sub‑gradient descent on true max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sg_dead = time.time() + (deadline - time.time()) * 0.25
        a_sg, v_sg = _subgradient_descent(best_seq, target_sum, sg_dead)
        if v_sg < best_val:
            best_val, best_seq = v_sg, a_sg.copy()
        print(f"[{time.time():.2f}] After sub‑grad best score={best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – greedy mass‑transfer refinement (large steps)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        mt_dead = time.time() + (deadline - time.time()) * 0.20
        a_mt, v_mt = _greedy_mass_transfer(best_seq, target_sum, mt_dead)
        if v_mt < best_val:
            best_val, best_seq = v_mt, a_mt.copy()
        print(f"[{time.time():.2f}] After mass‑transfer best score={best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – final smoothing pass
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sm_dead = time.time() + (deadline - time.time()) * 0.05
        a_sm, v_sm = _smoothening_stage(best_seq, target_sum, sm_dead)
        if v_sm < best_val:
            best_val, best_seq = v_sm, a_sm.copy()
        print(f"[{time.time():.2f}] After smoothing best score={best_val:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up and final evaluation
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_score:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
