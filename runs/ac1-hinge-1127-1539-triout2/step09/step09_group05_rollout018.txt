def _smooth_descent(a0, deadline, rng, best_val):
    a = a0.copy()
    n = len(a)
    S = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_f = _max_convolution(a)  # actual objective
    alphas = [1.0, 2.0, 5.0, 10.0]
    while time.time() < deadline:
        for alpha in alphas:
            for _ in range(10):
                if time.time() >= deadline:
                    break
                # compute convolution
                b = _full_convolution_fft(a)
                max_b = b.max()
                # compute softmax weights
                w = np.exp(alpha * (b - max_b))
                w_sum = w.sum()
                if w_sum == 0:
                    w = np.ones_like(b) / b.size
                else:
                    w = w / w_sum
                # compute gradient
                a_rev = a[::-1]
                conv = _fft_convolve(w, a_rev)
                grad = 2.0 * conv[n-1:n-1+ n]   # start index n-1
                # step size
                norm_sq = np.dot(grad, grad)
                if norm_sq > 0:
                    step = (max_b - best_f) / (norm_sq + 1e-12)
                    step = np.clip(step, 1e-7, 0.2)
                else:
                    step = 0.0
                a = _project_onto_simplex(a - step * grad, S)
                cur_f = _max_convolution(a)
                if cur_f < best_f:
                    best_f = cur_f
                    best_a = a.copy()
    return best_a, best_f

"""
Improved step‑function optimiser.

The optimiser consists of three phases:

1. **Heavy‑tailed random search** – generate many symmetric “edge‑heavy”
   sequences of the form (d+δ)^{-p} (where d=min(i,n−1−i)+1) with random
   exponents p and offsets δ, together with a few other simple families.
   The sequences are projected onto the simplex
   {a_i≥0, Σa_i=√(2n)} and the best few candidates (lowest max‑convolution)
   are retained.

2. **Smoothed gradient descent** – minimise a smooth proxy of the
   max‑convolution:
        φ_α(a)= (1/α)·log Σ_k exp(α·(a∗a)[k])
   whose gradient is 2·(softmax_α(b)∗a_rev) (b=a∗a).  We run this descent
   for a schedule of temperatures α∈{1,2,5,10} with a Polyak step size and
   project back onto the simplex after each step.

3. **Fine‑grained sub‑gradient + annealing** – finish the optimisation with
   the classic sub‑gradient method (binary mask of the maximal entries) and
   a short simulated‑annealing mass‑transfer loop.

The whole routine respects the 1000 s budget, prints intermediate bounds
and stores the final sequence in the global variable ``height_sequence_1``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT, convolutions, projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using an FFT."""
    n = x.shape[0] + y.shape[0] - 1
    f_len = _next_pow2(n)
    X = np.fft.rfft(x, n=f_len)
    Y = np.fft.rfft(y, n=f_len)
    conv = np.fft.irfft(X * Y, n=f_len)[:n]
    return conv


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a∗a."""
    return _fft_convolve(a, a)


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a∗a."""
    return _full_convolution_fft(a).max()


def _project_onto_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex
        { w | w_i ≥ 0 , Σ w_i = S }.
    """
    if S < 0:
        raise ValueError("Simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Families of initial vectors (all are symmetric and non‑negative)
# ----------------------------------------------------------------------
def _init_edge_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """a_i ∝ (d_i + δ)^{-p} ,  d_i = min(i,n‑1‑i)+1."""
    i = np.arange(n)
    d = np.minimum(i, n - 1 - i) + 1.0
    p = rng.uniform(0.2, 3.5)          # exponent
    delta = rng.uniform(0.0, 5.0)      # small offset to avoid singularities
    raw = (d + delta) ** (-p)
    return raw


def _init_edge_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """a_i ∝ exp(‑λ·d_i) , d_i as above."""
    i = np.arange(n)
    d = np.minimum(i, n - 1 - i) + 1.0
    lam = rng.uniform(0.0005, 0.03)
    raw = np.exp(-lam * d)
    return raw


def _init_linear_decrease(n: int, rng: np.random.Generator) -> np.ndarray:
    """Pure linear decrease from the left edge to the right edge."""
    i = np.arange(n)
    raw = 1.0 - i / (n - 1)          # values in [1,0]
    return raw


def _init_two_gaussians(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of two narrow Gaussians centred at the two ends."""
    i = np.arange(n)
    sigma = rng.uniform(0.02, 0.12) * n
    left = np.exp(-((i) ** 2) / (2 * sigma ** 2))
    right = np.exp(-((i - (n - 1)) ** 2) / (2 * sigma ** 2))
    raw = left + right
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    With probability 0.2 we warm‑start from the best known sequence,
    otherwise we draw from one of the families above.
    """
    S = np.sqrt(2.0 * n)

    # ---- warm start ----------------------------------------------------
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total > 0.0:
            a = a * (S / total)
            return a

    # ---- random families ------------------------------------------------
    choice = rng.choice(['powerlaw', 'exponential', 'linear', 'gauss'])
    if choice == 'powerlaw':
        raw = _init_edge_powerlaw(n, rng)
    elif choice == 'exponential':
        raw = _init_edge_exponential(n, rng)
    elif choice == 'linear':
        raw = _init_linear_decrease(n, rng)
    else:
        raw = _init_two_gaussians(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) gradient descent
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             rng: np.random.Generator,
                             best_val: float) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate φ_α(a) = (1/α)·log Σ exp(α·(a∗a)[k])
    with a decreasing schedule α∈{1,2,5,10}.  After each update the vector
    is projected onto the simplex of radius √(2n).
    Returns the best vector seen (measured by the true max‑convolution).
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_obj = _max_convolution(a)

    alphas = [1.0, 2.0, 5.0, 10.0]

    while time.time() < deadline:
        # pick a temperature at random (gives some stochasticity)
        alpha = rng.choice(alphas)

        # convolution of a with itself
        b = _full_convolution_fft(a)
        max_b = b.max()

        # soft‑max weights (log‑sum‑exp gradient)
        # subtract max for numerical stability
        w = np.exp(alpha * (b - max_b))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient = 2·(w ∗ reverse(a)), extracted on the central window
        a_rev = a[::-1]
        conv = _fft_convolve(w, a_rev)                 # length = 3n‑2
        start = n - 1
        grad = 2.0 * conv[start:start + n]

        # Polyak step (use the best true objective seen so far)
        norm_sq = np.dot(grad, grad)
        if norm_sq > 0.0:
            step = (max_b - best_obj) / (norm_sq + 1e-12)
            # keep step size reasonable
            step = np.clip(step, 1e-7, 0.15)
        else:
            step = 0.0

        a = _project_onto_simplex(a - step * grad, S)

        cur_obj = _max_convolution(a)
        if cur_obj + 1e-12 < best_obj:
            best_obj = cur_obj
            best_a = a.copy()

    return best_a, best_obj


# ----------------------------------------------------------------------
# Classic sub‑gradient descent (binary mask)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         best_val: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on f(a)=max(a∗a)."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_obj = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1
        # binary mask of maximal positions
        b = _full_convolution_fft(a)
        cur_val = b.max()
        mask = (b == cur_val).astype(float)

        # gradient = 2 * (mask ∗ reverse(a))
        a_rev = a[::-1]
        conv = _fft_convolve(mask, a_rev)
        start = n - 1
        grad = 2.0 * conv[start:start + n]

        norm_sq = np.dot(grad, grad)
        if norm_sq > 0.0:
            step = (cur_val - best_obj) / (norm_sq + 1e-12)
            step = np.clip(step, 1e-7, 0.2)
        else:
            step = 0.0

        a = _project_onto_simplex(a - step * grad, S)

        cur_val = _max_convolution(a)
        if cur_val + 1e-12 < best_obj:
            best_obj = cur_val
            best_a = a.copy()

        # occasional jitter to avoid stagnation
        if it % 250 == 0:
            jitter = rng.normal(scale=0.001 * np.mean(a), size=n)
            a = _project_onto_simplex(a + jitter, S)

    return best_a, best_obj


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb
# ----------------------------------------------------------------------
def _anneal_local_search(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealing that moves a random fraction of mass between two cells."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    cur_val = _max_convolution(a)
    best_a = a.copy()
    best_val = cur_val

    T = 0.05 * cur_val   # initial temperature (~5 % of the peak)
    decay = 0.9999

    it = 0
    while time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        frac = rng.uniform(0.02, 0.18)
        delta = a[i] * frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _max_convolution(a_new)
        delta_val = new_val - cur_val
        if delta_val < 0 or rng.random() < np.exp(-delta_val / (T + 1e-12)):
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()

        T *= decay
        if it % 500 == 0:
            jitter = rng.normal(scale=0.001 * S, size=n)
            a = _project_onto_simplex(a + jitter, S)
            cur_val = _max_convolution(a)

    return best_a, best_val
