def _apply_move(a, b, i, j, delta):
    n = a.size
    # subtract contributions of source i
    b[i : i + n] -= 2.0 * delta * a
    # add contributions of destination j
    b[j : j + n] += 2.0 * delta * a
    # adjust self-convolution terms
    k2i = 2*i
    k2j = 2*j
    kij = i + j
    if k2i < b.size:
        b[k2i] += delta * delta
    if k2j < b.size:
        b[k2j] += delta * delta
    if kij < b.size:
        b[kij] -= 2.0 * delta * delta
    # update a
    a[i] -= delta
    a[j] += delta

def _random_local_search(a0, b0, deadline):
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()
    rng = np.random.default_rng()
    while time.time() < deadline:
        # Choose source with probability proportional to its contribution to max conv?
        # Simpler: random i where a[i] > 0
        i = rng.integers(0, n)
        if a[i] < 1e-14:
            continue
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        # Choose delta
        frac = rng.uniform(0.05, 0.5)
        delta = a[i] * frac
        if delta < 1e-14:
            continue
        # copy b temporarily?
        # We'll compute b_new via temporary copy.
        b_new = b.copy()
        # Apply move on b_new using original a (still before move)
        b_new[i : i + n] -= 2.0 * delta * a
        b_new[j : j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta
        new_score = b_new.max()
        if new_score < best_score - 1e-12:
            # accept move
            a[i] -= delta
            a[j] += delta
            b = b_new
            best_score = new_score
            best_a = a.copy()
        else:
            # maybe accept with small prob? Not necessary.
            continue
    return best_a, best_score

"""
Search for a non‑negative step function that minimises
    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)**2)

The objective is scale‑invariant; after projecting the vector onto the
ℓ¹‑simplex Σa = √(2 n) the value of evaluate_sequence equals the
maximum entry of the self‑convolution (max b).  The algorithm therefore
optimises max b directly.

Strategy
--------
1.  Seed from a previously found sequence (if available) and project it
    onto the simplex.
2.  Process a schedule of increasing lengths – larger n give a better
    theoretical bound, but are more expensive.
3.  For each length:
    * Run several *projected Adam* optimisations on the smooth
      log‑sum‑exp surrogate of the max‑convolution.
    * Perform a deterministic hill‑climb that moves a large fraction of
      the mass responsible for the current worst convolution entry.
    * Run a cheap random local‑search that randomly shifts a part of the
      mass from one index to another and accepts the move if it improves
      the current maximal convolution value.
    * Optionally run a short extra Adam pass.
4.  After the schedule finish, polish the best candidate with a long
    Adam run followed by a final hill‑climb and random search.
5.  Truncate trailing near‑zero entries and return the sequence.

All heavy work (convolutions, gradients) uses FFTs, the code respects a
hard wall‑clock deadline (default 1000 s) and prints progress messages.
"""

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using an FFT (fast, O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float,
                    deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max surrogate.
    Runs until the wall‑clock deadline and returns the final vector.
    """
    a = a0.copy()
    n = a.size

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # learning‑rate base (scaled with the simplex radius)
    base_lr = 0.15 * target_sum

    # smoothed‑max schedule
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1

        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        # occasional tiny jitter to escape flat regions
        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4 / (beta + 1.0),
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # increase smoothing
        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 7)          # added a Gaussian family (6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    elif family == 5:                   # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    else:                               # Gaussian (centered)
        mu = (n - 1) / 2.0
        sigma = rng.uniform(0.05, 0.25) * n
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """Greedy hill‑climb that always attacks the current worst convolution entry."""
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()

    rng = np.random.default_rng()

    while time.time() < deadline:
        # index of the current maximal convolution entry
        M = int(np.argmax(b))

        # source indices influencing b[M]
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]

        # pick source with largest contribution
        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        # counterpart values for each possible destination
        counterpart = np.zeros(n, dtype=float)
        j_low = i_low
        j_high = i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        # destination with minimal counterpart (but not i)
        order = np.argsort(counterpart[j_range])
        j = None
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        # move a fraction of the source mass
        delta = a[i] * 0.8
        if delta <= 0.0:
            break

        # evaluate move (linear convolution update, O(n))
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # try a smaller move (binary‑search style)
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                # small jitter to escape plateau
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                target = math.sqrt(2.0 * n)
                a = _project_to_simplex(a, target)
                b = _full_convolution(a)

    return best_a, best_score

def _random_local_search(a0: np.ndarray, b0: np.ndarray,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple stochastic search.  Repeatedly move a random fraction of the
    mass from a random source index to a random destination index and
    keep the move if it reduces the maximal convolution entry.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()
    rng = np.random.default_rng()

    while time.time() < deadline:
        # pick a source with non‑zero mass
        i = rng.integers(0, n)
        if a[i] < 1e-14:
            continue
        # pick a distinct destination
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # pick a fraction of the source mass to move
        delta = a[i] * rng.uniform(0.05, 0.5)
        if delta < 1e-14:
            continue

        # apply the move to a copy of the convolution (O(n))
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()
        if new_score < best_score - 1e-12:
            # accept move
            a[i] -= delta
            a[j] += delta
            b = b_new
            best_score = new_score
            best_a = a.copy()

    return best_a, best_score

        min_restart_time = 5.0
        n_restarts = max(1, int(gd_budget // min_restart_time))
        per_restart = gd_budget / n_restarts

        local_best_a = None
        local_best_score = np.inf

        for r in range(n_restarts):
            if time.time() >= deadline:
                break
            this_deadline = min(deadline - 0.2, time.time() + per_restart)
            start_a = a_init.copy() if r == 0 else _random_initial_sequence(n, rng)
            a_opt = _adam_optimize(start_a, target_sum, this_deadline)
            score = _max_convolution_fft(a_opt)
            if score < local_best_score:
                local_best_score = score
                local_best_a = a_opt.copy()

        if local_best_a is not None and local_best_score < best_score - 1e-12:
            best_seq = local_best_a.copy()
            best_score = local_best_score
            print(f"[{time.time() - start_time:.1f}s] Adam improvement for n={n} "
                  f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # Phase B – deterministic hill‑climbing on the exact convolution
        # --------------------------------------------------------------
        if time.time() < deadline - 2.0:
            a_cur = best_seq.copy()
            a_cur = _project_to_simplex(a_cur, target_sum)
            b_cur = _full_convolution(a_cur)
            hc_deadline = min(deadline - 0.2, time.time() + hc_budget)
            a_hc, score_hc = _hill_climb_deterministic(a_cur, b_cur, hc_deadline)
            if score_hc < best_score - 1e-12:
                best_seq = a_hc.copy()
                best_score = score_hc
                print(f"[{time.time() - start_time:.1f}s] Hill‑climb improvement for n={n} "
                      f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # Phase C – cheap random local search
        # --------------------------------------------------------------
        if time.time() < deadline - 1.0:
            a_cur = best_seq.copy()
            a_cur = _project_to_simplex(a_cur, target_sum)
            b_cur = _full_convolution(a_cur)
            rnd_deadline = min(deadline - 0.2, time.time() + rnd_budget)
            a_rand, score_rand = _random_local_search(a_cur, b_cur, rnd_deadline)
            if score_rand < best_score - 1e-12:
                best_seq = a_rand.copy()
                best_score = score_rand
                print(f"[{time.time() - start_time:.1f}s] Random‑search improvement for n={n} "
                      f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # Phase D – extra short Adam run
        # --------------------------------------------------------------
        if extra_budget > 1.0 and time.time() < deadline - 1.0:
            extra_deadline = min(deadline - 0.5, time.time() + extra_budget)
            a_extra = _adam_optimize(best_seq, target_sum, extra_deadline)
            score_extra = _max_convolution_fft(a_extra)
            if score_extra < best_score - 1e-12:
                best_seq = a_extra.copy()
                best_score = score_extra
                print(f"[{time.time() - start_time:.1f}s] Extra Adam improvement for n={n} "
                      f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Final polishing (remaining budget)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        n = best_seq.size
        target = math.sqrt(2.0 * n)

        # longer Adam run on the remaining time
        final_deadline = deadline - 1.0
        best_seq = _adam_optimize(best_seq, target, final_deadline)

        # final deterministic hill‑climb
        b_final = _full_convolution(best_seq)
        best_seq, final_score = _hill_climb_deterministic(
            best_seq, b_final, deadline - 0.5)
        if final_score < best_score - 1e-12:
            best_score = final_score
            print(f"[{time.time() - start_time:.1f}s] Final hill‑climb improvement: "
                  f"score={best_score:.6f}")

        # final random local search (if time permits)
        if time.time() < deadline - 0.5:
            b_final = _full_convolution(best_seq)
            best_seq, final_score = _random_local_search(
                best_seq, b_final, deadline - 0.5)
            if final_score < best_score - 1e-12:
                best_score = final_score
                print(f"[{time.time() - start_time:.1f}s] Final random search improvement: "
                      f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Clean‑up: drop trailing near‑zeros
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 2000
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # 5) Report official evaluation
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.1f}s] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:   # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
