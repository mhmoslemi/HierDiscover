def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT."""
    conv = _conv_fft(a)
    return float(conv.max())

# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _score(a: np.ndarray) -> float:
    """Exact benchmark score (2·n·max(conv) / (sum(a)²))."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _gradient_softmax(a: np.ndarray, beta: float):

    Gradient of the soft‑max surrogate of max(convolution).
    Returns (grad, conv) where conv is a*a (full length).

    a_rev_fft = np.fft.rfft(a_rev, n=size)
    p_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase 1 – random parametric warm‑start
# ----------------------------------------------------------------------


def _random_parametric_start(n: int, S: float, deadline: float):
    """Search a two‑parameter family a_i = (i+1)^{-p}·exp(-λ·i/n)."""
    best = None
    best_val = np.inf
    rng = np.random.default_rng()
    i = np.arange(n, dtype=np.float64)

    # a few deterministic grid points first
    p_grid = np.linspace(0.2, 1.4, 13)
    lam_grid = np.linspace(0.0, 8.0, 13)

    for p in p_grid:
        for lam in lam_grid:
            if time.time() > deadline:
                return best, best_val
            seq = (i + 1) ** (-p) * np.exp(-lam * i / n)
            seq *= S / seq.sum()
            val = _score(seq)
            if val < best_val:
                best_val = val
                best = seq.copy()
                print(f"[{time.time():.2f}] grid p={p:.3f} λ={lam:.3f} score={best_val:.6f}")

    # random search
    while time.time() < deadline:
        p = rng.uniform(0.2, 1.4)
        lam = rng.exponential(1.0) * 4.0          # favour small λ but allow larger
        seq = (i + 1) ** (-p) * np.exp(-lam * i / n)
        seq *= S / seq.sum()
        val = _score(seq)
        if val < best_val:
            best_val = val
            best = seq.copy()
            print(f"[{time.time():.2f}] rnd p={p:.3f} λ={lam:.3f} score={best_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Phase 2 – soft‑max mirror descent
# ----------------------------------------------------------------------


def _softmax_mirror_descent(a0: np.ndarray, S: float, deadline: float):
    """Mirror‑descent w.r.t. the soft‑max approximation of the max‑conv."""
    a = a0.copy()
    a = a * (S / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 1e6
    base_eta = 0.3
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        # exponential (multiplicative) update – preserves positivity
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * grad)
        a = a * (S / a.sum())   # renormalise

        # occasional jitter to avoid stagnation
        if it % 500 == 0:
            jitter = rng.normal(scale=S * 0.0005, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (S / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpMD it={it:5d} β={beta:.1e} score={best_val:.6f}")

        # increase soft‑max sharpness
        if it % 300 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 3 – stochastic mass‑transfer local search
# ----------------------------------------------------------------------


def _mass_transfer_local_search(a0: np.ndarray, S: float, deadline: float):
    """Hill‑climbing: move a modest fraction of mass from high‑ to low‑contrib. indices."""
    a = a0.copy()
    a *= S / a.sum()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()

    # pre‑allocate convolution buffer
    conv = _conv_fft(a)

    it = 0
    # fraction schedule – start relatively aggressive and decay
    frac_start = 0.06
    frac_end = 0.001

    while time.time() < deadline:
        # linear decay of the maximal transfer fraction
        progress = (time.time() - (deadline - (deadline - time.time())))  # dummy, will be overwritten
        # we simply compute a fraction based on iteration count
        frac = max(frac_end, frac_start * (0.5 ** (it // 2000)))

        k_max = int(conv.argmax())
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each participating index to the maximal entry
        partner = k_max - idx
        weight = a[idx] * a[partner]

        donor = idx[np.argmax(weight)]
        # pick a receiver among the lowest 10 % contributors (deterministic)
        low_thr = np.percentile(weight, 10)
        low_candidates = idx[weight <= low_thr]
        if low_candidates.size:
            receiver = low_candidates[np.argmin(weight[weight <= low_thr])]
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            donor = rng.choice(idx)
            receiver = rng.integers(n)
            if donor == receiver:
                it += 1
                continue

        delta = a[donor] * rng.random() * frac
        if delta <= 0.0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta

        conv = _conv_fft(a)
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Transfer it={it:5d} score={best_val:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            conv = _conv_fft(a)

        # occasional jitter to shake out of plateaus
        if it % 1000 == 0:
            jitter = rng.normal(scale=S * 0.0003, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (S / a.sum())
            conv = _conv_fft(a)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 4 – exact sub‑gradient descent with back‑tracking line search
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, S: float, deadline: float):
    """Projected sub‑gradient descent on the true max‑convolution."""
    a = a0.copy()
    a = a * (S / a.sum())
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())

        # sub‑gradient of conv_{k_max}
        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[partner]

        # safe step: largest α that keeps a - α·grad ≥ 0
        safe_alpha = np.min(np.where(grad > 0, a / grad, np.inf))
        # line‑search parameters
        alpha = 0.9 * safe_alpha
        improved = False

        for _ in range(5):
            a_candidate = a - alpha * grad
            a_candidate = _project_to_simplex(a_candidate, S)
            val = _score(a_candidate)
            if val < best_val:
                best_val = val
                best_a = a_candidate.copy()
                a = a_candidate
                improved = True
                print(f"[{time.time():.2f}] Subgrad it={it:5d} α={alpha:.3e} score={best_val:.6f}")
                break
            else:
                alpha *= 0.5  # shrink step and retry

        if not improved:
            # small random perturbation to escape a potential flat region
            if it % 200 == 0:
                jitter = rng.normal(scale=S * 0.0002, size=n)
                a = np.maximum(a + jitter, 0.0)
                a = _project_to_simplex(a, S)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – a little larger than the previous 12000 gives a
    # finer discretisation and a slightly lower theoretical bound.
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 15000
    elif budget_s >= 400:
        n = 12000
    else:
        n = 8000

    S = np.sqrt(2.0 * n)          # ℓ¹‑norm used for the internal scaling

    # --------------------------------------------------------------
    # Warm‑start (either from a previous run or from a random parametric family)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if init.shape[0] != n:
            x_old = np.linspace(0.0, 1.0, init.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            init = np.interp(x_new, x_old, init)
        init = init * (S / init.sum())
    else:
        # simple power‑law start
        init = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))
        init = init * (S / init.sum())

    best_seq = init.copy()
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] Initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – random parametric warm‑start (≈10 % of budget)
    # --------------------------------------------------------------
    phase1_deadline = time.time() + (deadline - time.time()) * 0.10
    cand, cand_score = _random_parametric_start(n, S, phase1_deadline)
    if cand is not None and cand_score < best_score:
        best_seq = cand
        best_score = cand_score
        print(f"[{time.time():.2f}] Phase‑1 best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – soft‑max mirror descent (≈30 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase2_deadline = time.time() + (deadline - time.time()) * 0.30
        seq2, score2 = _softmax_mirror_descent(best_seq, S, phase2_deadline)
        if score2 < best_score:
            best_seq = seq2
            best_score = score2
        print(f"[{time.time():.2f}] After Phase‑2 score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – stochastic mass‑transfer (≈30 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase3_deadline = time.time() + (deadline - time.time()) * 0.30
        seq3, score3 = _mass_transfer_local_search(best_seq, S, phase3_deadline)
        if score3 < best_score:
            best_seq = seq3
            best_score = score3
        print(f"[{time.time():.2f}] After Phase‑3 score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – exact sub‑gradient descent (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        seq4, score4 = _subgradient_descent(best_seq, S, deadline)
        if score4 < best_score:
            best_seq = seq4
            best_score = score4
        print(f"[{time.time():.2f}] After Phase‑4 (final) score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Clean‑up: trim trailing zeros (tiny values) and final evaluation
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts on the next call
    globals()["height_sequence_1"] = final_seq
    return final_seq
