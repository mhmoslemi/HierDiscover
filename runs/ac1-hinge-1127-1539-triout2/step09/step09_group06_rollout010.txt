"""Hybrid random search + Frank‑Wolfe (LP) + hill‑climb for minimizing ‖a⋆a‖∞.

The algorithm:
1. Generate a feasible random start (ℓ₁‑norm = √(2 n)).
2. Improve it by a Frank‑Wolfe–style step:
   – given the current shape a compute M = max(a⋆a)
   – solve a linear program max ∑bₖ s.t. a⋆b ≤ M, b≥0
   – scale b to the same ℓ₁‑norm and take a convex combination (1‑t)a + t b
     with a line‑search over a few t‑values.
   The step is repeated while time permits.
3. Finish with a cheap stochastic hill‑climb that moves a small
   fraction of mass between two coordinates.
4. Return the best sequence and store it in the global variable
   ``height_sequence_1``.
"""
import time
import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog

# -------------------------------------------------------------
# Low‑level convolution utilities
# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv

def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a∗a."""
    return _full_convolution_fft(a).max()

def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]

# -------------------------------------------------------------
# Random families for initial vectors
# -------------------------------------------------------------
def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw

def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw

def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    raw = mix * lin + (1.0 - mix) * base
    return raw

def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    raw = mix * ush + (1.0 - mix) * base
    return raw

def _init_cosine(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    raw = 1.0 - np.cos(np.pi * i / (n - 1))
    raw = raw * rng.uniform(0.5, 2.0)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * raw + (1.0 - mix) * base

def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)

def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    return rng.random(n)

def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.05, 0.9)
    return rng.gamma(shape, 1.0, size=n)

def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """Return a feasible vector of length n (ℓ₁‑norm = √(2 n))."""
    S = np.sqrt(2.0 * n)

    # 20 % chance to warm‑start from the previous best
    if warm_seq is not None and rng.random() < 0.23:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
        return a

    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'gamma', 'uniform', 'dirichlet', 'ushape', 'cosine'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'cosine':
        raw = _init_cosine(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a

# -------------------------------------------------------------
# Linear‑programming (Frank‑Wolfe) step
# -------------------------------------------------------------
def _build_conv_matrix(a: np.ndarray) -> csr_matrix:
    """Construct the (2n‑1)×n matrix C with C·b = a∗b (linear in b)."""
    n = a.shape[0]
    nnz = n * n
    rows = np.empty(nnz, dtype=np.int32)
    cols = np.empty(nnz, dtype=np.int32)
    data = np.empty(nnz, dtype=np.float64)

    idx = 0
    for j in range(n):
        rows[idx:idx + n] = np.arange(j, j + n, dtype=np.int32)
        cols[idx:idx + n] = j
        data[idx:idx + n] = a
        idx += n

    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _lp_direction(a: np.ndarray, M: float) -> np.ndarray | None:
    """Solve max ∑b s.t. a∗b ≤ M, b≥0.  Returns the optimal b or None."""
    n = a.shape[0]
    A = _build_conv_matrix(a)                     # shape (2n‑1, n)
    c = -np.ones(n)                               # maximise sum → minimise –sum
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)   # right‑hand side
    bounds = [(0, None)] * n

    # Use HiGHS – most robust and fast for large sparse LPs
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs',
                  options={'presolve': True, 'time_limit': 30})

    if not res.success:
        return None
    return res.x

# -------------------------------------------------------------
# Simple stochastic hill‑climb (final polishing)
# -------------------------------------------------------------
def _hill_climb(a: np.ndarray,
                time_deadline: float,
                rng: np.random.Generator) -> np.ndarray:
    """Move a small random fraction of mass between two positions."""
    n = a.shape[0]
    cur = a.copy()
    cur_score = _max_convolution(cur)

    while time.time() < time_deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur[i] <= 0.0:
            continue
        frac = rng.uniform(0.02, 0.12)
        amount = cur[i] * frac
        cur[i] -= amount
        cur[j] += amount
        # re‑normalize (ℓ₁‑norm is invariant under this move, but
        # rounding errors may accumulate)
        cur = cur * (np.sqrt(2.0 * n) / cur.sum())
        new_score = _max_convolution(cur)
        if new_score + 1e-9 < cur_score:
            cur_score = new_score
        else:
            # undo if not improving (optional – we keep the move anyway
            # because later moves may compensate)
            pass
    return cur

def propose_candidate(seed: int = 42,

    # -----------------------------------------------------------------

    if 'height_sequence_1' in globals():

        except Exception:
            warm_seq = None

    # Phase 1 – fast random exploration to obtain a decent start
    # -----------------------------------------------------------------
    best_seq = None
    best_M = np.inf
    attempts = 0

        M = _max_convolution(a0)
        if M < best_M:
            best_M = M
            best_seq = a0.copy()
            print(f"[explore] attempt={attempts:4d} n={n:5d}  M={best_M:.6f}")

    if best_seq is None:
        # fall back to a simple uniform vector (should never happen)
        n = 3000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_M = _max_convolution(best_seq)

    # -----------------------------------------------------------------
    # Phase 2 – optional up‑sampling (try a few lengths)
    # -----------------------------------------------------------------
    cur_seq = best_seq
    cur_M = best_M
    cur_n = len(cur_seq)

    # try a few scalings up to factor 2.0 (paid for only if time allows)
    for factor in [1.25, 1.5, 2.0]:
        if time.time() > end_time - 250.0:
            break
        L = int(cur_n * factor)
        if L < 2000 or L > 15000:
            continue
        a_scaled = _scale_sequence(cur_seq, L)
        a_scaled = np.clip(a_scaled, 0.0, None)
        a_scaled = a_scaled * (np.sqrt(2.0 * L) / a_scaled.sum())
        M = _max_convolution(a_scaled)
        if M < cur_M:
            cur_M = M
            cur_seq = a_scaled.copy()
            cur_n = L
            print(f"[scale] length={L:5d}  M={cur_M:.6f}")

    # -----------------------------------------------------------------
    # Phase 3 – Frank‑Wolfe style LP steps (the main driver of improvement)
    # -----------------------------------------------------------------
    lp_deadline = end_time - 150.0   # keep ~150 s for final polishing
    iteration = 0
    while time.time() < lp_deadline:
        iteration += 1
        n = cur_n
        S = np.sqrt(2.0 * n)

        # compute current peak and build LP
        cur_M = _max_convolution(cur_seq)
        b = _lp_direction(cur_seq, cur_M)
        if b is None:
            print("[lp] LP failed – leaving loop")
            break
        # scale the direction to the same ℓ₁‑norm
        sum_b = b.sum()
        if sum_b == 0.0:
            print("[lp] LP returned zero vector – leaving loop")
            break
        b_scaled = b * (S / sum_b)

        # line‑search over a few convex‑combination weights
        best_local_seq = cur_seq
        best_local_M = cur_M
        for t in (0.2, 0.4, 0.6, 0.8):
            a_new = (1.0 - t) * cur_seq + t * b_scaled
            M_new = _max_convolution(a_new)
            if M_new < best_local_M:

"""Hybrid random search + Frank‑Wolfe (LP) + hill‑climb for minimizing ‖a⋆a‖∞.

The algorithm:
1. Generate a feasible random start (ℓ₁‑norm = √(2 n)).
2. Improve it by a Frank‑Wolfe–style step:
   – given the current shape a compute M = max(a⋆a)
   – solve a linear program max ∑bₖ s.t. a⋆b ≤ M, b≥0
   – scale b to the same ℓ₁‑norm and take a convex combination
     (1‑t)a + t b with a short line‑search over a few t‑values.
   The step is repeated while time permits.
3. Finish with a cheap stochastic hill‑climb that moves a small
   fraction of mass between two coordinates.
4. Return the best sequence and store it in the global variable
   ``height_sequence_1`` for warm‑starts in later runs.
"""
import time
import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog

# -----------------------------------------------------------------
# Convolution utilities
# -----------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a⋆a."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# -----------------------------------------------------------------
# Random families for initial vectors
# -----------------------------------------------------------------
def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    raw = mix * ush + (1.0 - mix) * base
    return raw


def _init_cosine(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    raw = 1.0 - np.cos(np.pi * i / (n - 1))
    raw = raw * rng.uniform(0.5, 2.0)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * raw + (1.0 - mix) * base


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    return rng.random(n)


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.05, 0.9)
    return rng.gamma(shape, 1.0, size=n)


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """Return a feasible vector of length n (ℓ₁‑norm = √(2 n))."""
    S = np.sqrt(2.0 * n)

    # Occasionally warm‑start from a previously found sequence
    if warm_seq is not None and rng.random() < 0.23:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
        return a

    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'gamma', 'uniform', 'dirichlet', 'ushape', 'cosine'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'cosine':
        raw = _init_cosine(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# -----------------------------------------------------------------
# Frank‑Wolfe (LP) utilities
# -----------------------------------------------------------------
def _build_conv_matrix(a: np.ndarray) -> csr_matrix:
    """Return the (2n‑1)×n matrix C with C·b = a⋆b (linear in b)."""
    n = a.shape[0]
    nnz = n * n
    rows = np.empty(nnz, dtype=np.int32)
    cols = np.empty(nnz, dtype=np.int32)
    data = np.empty(nnz, dtype=np.float64)

    idx = 0
    for j in range(n):
        rows[idx:idx + n] = np.arange(j, j + n, dtype=np.int32)
        cols[idx:idx + n] = j
        data[idx:idx + n] = a
        idx += n

    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _lp_direction(a: np.ndarray, M: float) -> np.ndarray | None:
    """Solve   max Σ bₖ   s.t. a⋆b ≤ M, b ≥ 0.  Returns the optimal b or None."""
    n = a.shape[0]
    A = _build_conv_matrix(a)
    c = -np.ones(n)                                 # maximise sum → minimise –sum
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)  # right‑hand side
    bounds = [(0, None)] * n

    # HiGHS is fast and respects sparse matrices
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': 30})
    except Exception:  # pragma: no cover
        return None

    if not res.success:
        return None
    return res.x


# -----------------------------------------------------------------
# Simple stochastic hill‑climb (final polishing)
# -----------------------------------------------------------------
def _hill_climb(a: np.ndarray,
                deadline: float,
                rng: np.random.Generator) -> np.ndarray:
    """Repeatedly move a random fraction of mass between two positions."""
    n = a.shape[0]
    cur = a.copy()
    cur_score = _max_convolution(cur)

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur[i] <= 0.0:
            continue
        frac = rng.uniform(0.02, 0.12)
        amount = cur[i] * frac
        cur[i] -= amount
        cur[j] += amount
        # renormalise to keep exact ℓ₁‑norm (numerical drift)
        cur = cur * (np.sqrt(2.0 * n) / cur.sum())
        new_score = _max_convolution(cur)
        if new_score + 1e-9 < cur_score:
            cur_score = new_score
    return cur
