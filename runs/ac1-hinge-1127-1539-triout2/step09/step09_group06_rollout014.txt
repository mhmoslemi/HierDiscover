    best_val = _max_convolution(a)
    n = len(a)

        # compute b and max index
        b = _full_convolution_fft(a)
        k = int(b.argmax())
        val = b[k]

        lo = max(0, k - (n - 1))

        idx = np.arange(lo, hi + 1)

def _project_simplex(v, S):
    """Project vector v onto the simplex {x >= 0, sum x = S}."""
    # standard algorithm
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        # all entries zero? then return uniform?
        return np.full(n, S / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

√(2·n).  The algorithm proceeds in several stages:

 1. Random exploration of many parametric families (including a new
    Gaussian shape) to obtain a good warm‑start.
 2. Length scaling: the best candidate is linearly interpolated to a few
    larger lengths and the best length is kept.
 3. Continuous optimisation:
        • Adam on a τ‑soft‑max approximation of the convolution peak.
        • Simple sub‑gradient descent on the hard maximum.
 4. Targeted hill‑climbing: using the sub‑gradient of the current
    maximum we repeatedly move a tiny fraction of mass from indices
    that contribute most to the peak to indices that contribute least.
    This fine‑tunes the solution and usually yields the last few
    thousandths of improvement.
 5. Final sanity check via `evaluate_sequence`.

The routine respects a 1000 s budget, prints progress messages and
stores the best sequence in the global variable ``height_sequence_1``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, scaling, etc.)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a∗a."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Soft‑max (τ‑smooth) gradient – used by Adam
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient: 2*(mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser (mirror descent on the simplex)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex (log‑softmax parametrisation).
    Minimises the *hard* maximum of the convolution by a slowly
    decaying τ‑soft‑max.
    Returns the best sequence found and its (hard) peak value.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                     # ℓ1‑norm we must keep

    # initialise logits from the feasible a0 (entropic parametrisation)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                         # logits (log‑softmax)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.55                                 # a little more aggressive

    # temperature schedule – start relatively large for smoothness
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 5.0)                 # a few units of smoothing

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ---- recover a from logits (log‑softmax) ----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ---- convolution & gradient (τ‑soft‑max) ----
        g, peak = _softmax_gradient(a, tau)

        # keep the best hard peak seen so far
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ---- gradient w.r.t. logits (chain rule) ----
        mu = np.dot(a, g) / S                 # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)                 # shape (n,)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 6.0 * S:
            grad_p = grad_p * (6.0 * S / clip_val)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- schedules (very slow decay) ----
        tau *= 0.99996                         # keep smoothing longer
        lr *= 0.99996                          # gentle learning‑rate decay

        # occasional jitter to escape plateaus
        if it % 800 == 0:
            p += rng.normal(scale=0.006, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Sub‑gradient descent on the *hard* max (no smoothing)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple sub‑gradient descent that always follows the entry of the
    convolution attaining the current maximum.  After each step the
    vector is projected back onto the simplex (positivity + fixed ℓ1‑norm).
    Returns the best sequence seen and its hard peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    a = a0.copy()
    best_a = a.copy()
    best_val = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # current convolution and its maximiser
        b = _full_convolution_fft(a)
        k = int(b.argmax())
        val = b[k]

        if val < best_val:
            best_val = val
            best_a = a.copy()

        # sub‑gradient of (a*a)_k :  g_i = 2 * a_{k-i} (if inside range)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        jdx = k - idx
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[jdx]

        # step size decreasing like 1/t^{0.6}
        step = 0.85 * S / (np.sqrt(it) ** 0.6 + 1.0)

        a = a - step * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, S / n)
        else:
            a = a * (S / s)

        # occasional random jitter to avoid tiny plateaus
        if it % 500 == 0:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                frac = rng.uniform(0.02, 0.12)
                amount = a[i] * frac
                a[i] -= amount
                a[j] += amount
                a = a * (S / a.sum())

    return best_a, best_val


# ----------------------------------------------------------------------
# Targeted hill‑climbing (tiny, gradient‑guided mass transfers)
# ----------------------------------------------------------------------
def _targeted_hill_climb(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Fine‑grained hill‑climbing that uses the hard‑max sub‑gradient to
    move a very small fraction of mass from high‑impact indices to
    low‑impact indices.  This often breaks the last plateau.
    """
    a = a0.copy()
    cur_max = _max_convolution(a)
    best_a = a.copy()
    best_val = cur_max
    n = len(a)
    it = 0

    while time.time() < deadline:
        it += 1

        # current max index and sub‑gradient
        b = _full_convolution_fft(a)
        k = int(b.argmax())
        val = b[k]

        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        jdx = k - idx
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[jdx]

        # choose source (large impact, positive mass)
        positive = np.where(a > 1e-12)[0]
        if positive.size == 0:
            break
        src_metric = g[positive] * a[positive]          # heuristic
        src = positive[np.argmax(src_metric)]

        # choose destination (minimal impact)
        dest = np.argmin(g)
        if dest == src:
            # find the next‑best destination that is not src
            order = np.argsort(g)
            for d in order:
                if d != src:
                    dest = d
                    break

        # tiny transfer
        frac = rng.uniform(0.001, 0.008)                # 0.1 %–0.8 %
        delta = a[src] * frac
        a2 = a.copy()
        a2[src] -= delta
        a2[dest] += delta

        new_val = _max_convolution(a2)

        if new_val + 1e-12 < best_val:
            best_val = new_val
            best_a = a2.copy()
            a = a2
            if it % 200 == 0:
                print(f"[target] iter={it:6d}  new best max={best_val:.6f}")

        else:
            # occasional random jitter to keep exploring
            if rng.random() < 0.01:
                i, j = rng.integers(0, n, size=2)
                if i != j and a[i] > 0.0:
                    f2 = rng.uniform(0.0005, 0.003)
                    d2 = a[i] * f2
                    a[i] -= d2
                    a[j] += d2

    return best_a, best_val


# ----------------------------------------------------------------------
# Parametric families for random initialisation (including Gaussian)
# ----------------------------------------------------------------------
def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    raw = mix * ush + (1.0 - mix) * base
    return raw


def _init_cosine(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    raw = 1.0 - np.cos(np.pi * i / (n - 1))
    raw = raw * rng.uniform(0.5, 2.0)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * raw + (1.0 - mix) * base


def _init_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    """Gaussian bump centred in the middle of the interval."""
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    sigma = rng.uniform(0.08, 0.22) * n        # relative width
    raw = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * raw + (1.0 - mix) * base


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    return rng.random(n)


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.05, 0.9)
    return rng.gamma(shape, 1.0, size=n)


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    With probability ≈0.2 we warm‑start from ``warm_seq`` (scaled).
    Otherwise we draw from one of the random families.
    The returned vector satisfies sum = sqrt(2·n) and all entries ≥0.
    """
    S = np.sqrt(2.0 * n)

    if warm_seq is not None and rng.random() < 0.23:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
        return a

    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'gamma', 'uniform', 'dirichlet', 'ushape',
                        'cosine', 'gaussian'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'cosine':
        raw = _init_cosine(n, rng)
    elif style == 'gaussian':
        raw = _init_gaussian(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence found as a plain
    Python list and stores it in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # tiny safety margin

    # ------------------------------------------------------------------
    # Warm‑start handling
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # ------------------------------------------------------------------
    # Phase 1 – fast random exploration (≈ 20 % of the budget)
    # ------------------------------------------------------------------
    explore_deadline = end_time - 350.0   # keep at least ~350 s for refinement
    best_seq = None
    best_score = np.inf
    best_n = None
    attempts = 0

    while time.time() < explore_deadline:
        attempts += 1
        # favour a few‑thousand entries but allow some variety
        if rng.random() < 0.70:
            n = int(rng.integers(1800, 6000))
        else:
            n = int(rng.integers(6000, 15000))

        a0 = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a0)

        if peak < best_score:
            best_score = peak
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempts:4d} n={n:5d}  new best={best_score:.6f}")

    if best_seq is None:
        best_n = 3000
        best_seq = np.full(best_n, np.sqrt(2.0 * best_n) / best_n)
        best_score = _max_convolution(best_seq)

    # ------------------------------------------------------------------
    # Phase 2 – upscale and evaluate a few larger lengths (no optimisation)
    # ------------------------------------------------------------------
    cand_lengths = [best_n]
    for factor in [1.25, 1.5, 2.0, 3.0, 4.0, 5.0]:
        cand = int(best_n * factor)
        if 2000 <= cand <= 25000:
            cand_lengths.append(cand)
    cand_lengths = sorted(set(cand_lengths))

    cur_seq = best_seq
    cur_score = best_score
    cur_n = best_n

    for L in cand_lengths:
        if time.time() > end_time - 250.0:
            break
        a_scaled = _scale_sequence(cur_seq, L)
        a_scaled = np.clip(a_scaled, 0.0, None)
        S = np.sqrt(2.0 * L)
        a_scaled = a_scaled * (S / a_scaled.sum())
        peak = _max_convolution(a_scaled)

        if peak < cur_score:
            cur_score = peak
            cur_seq = a_scaled.copy()
            cur_n = L
            print(f"[scale] length={L:5d}  improved max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam optimisation (multiple short restarts)
    # ------------------------------------------------------------------
    remaining = end_time - time.time() - 130.0   # reserve time for later phases
    if remaining > 15.0:
        n_restart = max(1, int(remaining // 60))   # roughly one minute per restart
        per_run = remaining / n_restart
        for r in range(n_restart):
            if time.time() > end_time - 130.0:
                break
            # jitter start point
            noise = rng.normal(scale=0.02 * np.sqrt(2.0 * cur_n) / cur_n, size=cur_n)
            a_start = np.maximum(cur_seq + noise, 0.0)
            a_start = a_start * (np.sqrt(2.0 * cur_n) / a_start.sum())
            a_opt, peak = _adam_optimize(a_start,
                                          time.time() + per_run,
                                          rng)
            if peak < cur_score:
                cur_score = peak
                cur_seq = a_opt.copy()
                print(f"[adam] restart={r}  new best max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – sub‑gradient descent (continuous polishing)
    # ------------------------------------------------------------------
    remaining = end_time - time.time() - 130.0
    if remaining > 20.0:
        a_opt, peak = _subgradient_descent(cur_seq,
                                           time.time() + remaining,
                                           rng)
        if peak < cur_score:
            cur_score = peak
            cur_seq = a_opt.copy()
            print(f"[subgrad] after polishing max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – targeted hill‑climbing (final fine‑tuning)
    # ------------------------------------------------------------------
    remaining = end_time - time.time() - 5.0
    if remaining > 5.0:
        a_opt, peak = _targeted_hill_climb(cur_seq,
                                           time.time() + remaining,
                                           rng)
        if peak < cur_score:
            cur_score = peak
            cur_seq = a_opt.copy()
            print(f"[target] final max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – cheap random hill‑climb (very small moves, many tries)
    # ------------------------------------------------------------------
    while time.time() < end_time - 1.0:
        a = cur_seq.copy()
        n = a.shape[0]
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        frac = rng.uniform(0.0005, 0.004)   # ultra‑tiny shift
        amount = a[i] * frac
        a[i] -= amount
        a[j] += amount
        peak = _max_convolution(a)
        if peak + 1e-12 < cur_score:   # strict improvement
            cur_score = peak
            cur_seq = a.copy()
            print(f"[hill] refined max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    final_seq = cur_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f}  internal max={cur_score:.6f}  length={len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
