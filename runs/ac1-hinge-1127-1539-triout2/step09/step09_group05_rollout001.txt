"""
Propose a step‑function sequence that minimises the evaluation function.

The algorithm builds on the previous approach but introduces two key
innovations:

1. **Smooth max approximation** – instead of the hard‑max sub‑gradient
   we minimise a differentiable surrogate
   J(a)= (1/β)·log(∑exp(β·(a∗a)[k])).
   Its gradient is 2·(w∗rev(a)) where w are the soft‑max weights.
   By gradually increasing the smoothing parameter β the optimiser first
   flattens the whole autocorrelation and then concentrates on the true
   maximum.

2. **Symmetry enforcement** – the optimal sequences observed experimentally
   are nearly symmetric.  After each gradient step we symmetrise the vector
   (a←(a+a[::-1])/2) and project it onto the ℓ₁‑simplex
   {a≥0,∑a=√(2n)}.  This reduces the dimensionality of the search space
   and yields a lower peak.

The search consists of many restarts with random parametric families,
followed by the smooth projected‑gradient descent.  Lengths are
progressively increased (up to ≈20 000) and the best result is kept.
Finally the sequence is evaluated with the provided ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a∗a."""
    return _full_convolution_fft(a).max()


def _project_onto_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {w | w_i ≥ 0 , ∑ w_i = S }.
    Implements the O(n) algorithm of Duchi et al.
    """
    if S < 0:
        raise ValueError("Simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


def _make_symmetric(a: np.ndarray) -> np.ndarray:
    """Force symmetry a[i]=a[n-1-i] (preserves the sum)."""
    return (a + a[::-1]) / 2.0

def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    raw = mix * ush + (1.0 - mix) * base
    return raw


def _init_sine(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    vals = np.sin(np.pi * (i + 1) / (n + 1))
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * vals + (1.0 - mix) * base
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    return rng.random(n)


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Occasionally warm‑starts from the globally best sequence and otherwise
    draws from a random parametric family.
    """
    S = np.sqrt(2.0 * n)

    # Warm start with probability ≈0.22
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
        return a

    style = rng.choice(
        ['powerlaw', 'exponential', 'linear', 'ushape',
         'sine', 'uniform', 'dirichlet']
    )
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'sine':
        raw = _init_sine(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


def _initial_vector_symmetric(n: int,
                              rng: np.random.Generator,
                              warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Same as _initial_vector but enforce a≈a[::-1] and re‑project onto the
    simplex to keep feasibility.
    """
    a = _initial_vector(n, rng, warm_seq)
    a = _make_symmetric(a)
    S = np.sqrt(2.0 * n)
    a = _project_onto_simplex(a, S)
    return a

    Starts with a small smoothing β and doubles it periodically.

    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _max_convolution(a)

    it = 0
    beta = 0.02          # initial smoothing
    step0 = 1.0          # initial step‑size factor

        # --------------------------------------------------------------
        # compute convolution and smooth weights

        b = _full_convolution_fft(a)
        M = b.max()
        # soft‑max weights (stable)

        sum_exp = exp_term.sum()
        if sum_exp == 0.0:

            w = exp_term / sum_exp

        # --------------------------------------------------------------
        # gradient of the smooth objective
        # --------------------------------------------------------------
        a_rev = a[::-1]
        conv_len = len(w) + n - 1

        Fw = np.fft.rfft(w, n=f_len)

        start = n - 1

        # --------------------------------------------------------------

        # --------------------------------------------------------------
        it += 1

        a_new = _make_symmetric(a_new)
        a_new = _project_onto_simplex(a_new, S)

        # --------------------------------------------------------------
