def project_onto_simplex(v, S): # v can be any real vector.

   u = np.sort(v)[::-1]

   cssv = np.cumsum(u) - S

   rho = np.nonzero(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]

   theta = cssv[rho] / (rho+1)

   w = np.maximum(v - theta, 0)

   return w

"""
Step‑function optimizer (multi‑phase, heavy random exploration,
parameter sweep, Adam + soft‑max, tiny LP polishing, final hill‑climb).

The evaluator is scale‑invariant; after normalising the ℓ₁‑norm to
√(2·n) the score equals the peak of the self‑convolution.
We therefore minimise  maxₖ (a∗a)ₖ.

Phases
------
1. **Broad random search** – many lengths (2 000 … 20 000) and many
   parametric families (power‑law, exponential, linear, cosine,
   U‑shape, Gaussian, triangle).  The best candidate is kept.
2. **Length up‑scaling** – the best shape is interpolated to a few
   larger lengths (up to 5×) and the one with the smallest raw peak
   is retained.
3. **Adam optimisation** – a log‑softmax (mirror) parametrisation
   with a τ‑soft‑max surrogate of the max is run several independent
   restarts (≈30 s each).  The best sequence is stored.
4. **LP polishing** – for up to 2 000 variables the linear program
   “maximise Σ gⱼ  s.t. a∗g ≤ M” is solved (SciPy highs).  The solution
   is mixed with the current vector (t=0.5) and the result is projected
   back onto the simplex.  For longer vectors a compressed version
   (2 000 entries) is used.
5. **Stochastic hill‑climb** – tiny random transfers of mass between
   two positions are accepted only if they strictly improve the peak.
6. The final sequence is returned (and stored in ``height_sequence_1``).

The whole routine respects a 1000 s wall‑clock budget.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a∗a."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


def _project_onto_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx=S}."""
    if S < 0:
        raise ValueError("S must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    return mix * edge + (1.0 - mix) * base


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    return mix * exp_part + (1.0 - mix) * base


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    return mix * lin + (1.0 - mix) * base


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    return mix * ush + (1.0 - mix) * base


def _init_cosine(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    raw = 1.0 - np.cos(np.pi * i / (n - 1))
    raw = raw * rng.uniform(0.5, 2.0)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * raw + (1.0 - mix) * base


def _init_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    """Symmetric discrete Gaussian centred at (n‑1)/2."""
    sigma = rng.uniform(0.1, 0.45) * n
    i = np.arange(n)
    centre = (n - 1) / 2.0
    raw = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * raw + (1.0 - mix) * base


def _init_triangle(n: int, rng: np.random.Generator) -> np.ndarray:
    """Symmetric triangle (peak in the centre)."""
    i = np.arange(n)
    centre = (n - 1) / 2.0
    raw = 1.0 - np.abs(i - centre) / centre
    raw = np.maximum(raw, 0.0)
    raw = raw * rng.uniform(0.5, 2.0)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * raw + (1.0 - mix) * base

    'gaussian': _init_gaussian,
    'triangle': _init_triangle,
    'uniform': lambda n, rng: np.full(n, 1.0),
    'dirichlet': lambda n, rng: rng.random(n),
    'gamma': lambda n, rng: rng.gamma(rng.uniform(0.05, 0.9), 1.0, size=n)
}


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    With probability ≈0.2 we warm‑start from ``warm_seq`` (scaled).
    Otherwise a random parametric family is drawn.
    Returned vector satisfies Σa = √(2·n) and aⱼ ≥ 0.
    """
    S = np.sqrt(2.0 * n)

    # occasional warm‑start
    if warm_seq is not None and rng.random() < 0.23:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
        return a

    # pick a family uniformly at random
    style = rng.choice(list(_FAMILY_FUNCS.keys()))
    raw = _FAMILY_FUNCS[style](n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient used by Adam (mirror descent on the simplex)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """Gradient of the τ‑soft‑max approximation of max(a*a)."""
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # grad = 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam on the simplex (log‑softmax parametrisation) minimising the hard
    maximum of the convolution, with a τ‑soft‑max surrogate.
    Returns the best vector encountered and its *hard* peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)          # target ℓ₁ norm

    # initialise logits from feasible a0
    soft = a0 / S
    soft = np.maximum(soft, 1e-12)
    soft /= soft.sum()
    p = np.log(soft)                # logits for log‑softmax

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.45

    # temperature starts a few units above the current peak
    tau = max(_full_convolution_fft(a0).max(), 5.0)

    best_a = a0.copy()
    best_peak = _max_convolution(a0)
    it = 0

    while time.time() < deadline:
        it += 1

        # recover a from logits
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # gradient of τ‑soft‑max
        g, peak = _softmax_gradient(a, tau)

        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # chain rule to get gradient w.r.t. logits
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 6.0 * S:
            grad_p = grad_p * (6.0 * S / clip_val)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # schedules – slower decay keeps exploration longer
        tau *= 0.9995
        lr *= 0.9995

        # jitter every few hundred iterations
        if it % 800 == 0:
            p += rng.normal(scale=0.006, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Sub‑gradient descent on the *hard* max (Euclidean projection)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple sub‑gradient descent that follows the *hard* maximum.
    After each step the vector is projected onto the simplex.
    Returns the best vector seen and its peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    a = a0.copy()
    best_a = a.copy()
    best_val = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1

        b = _full_convolution_fft(a)
        k = int(b.argmax())
        val = b[k]

        if val < best_val:
            best_val = val
            best_a = a.copy()

        # sub‑gradient of (a*a)_k
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        jdx = k - idx
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[jdx]

        # step size ~ 1/√t
        step = 0.8 * S / (np.sqrt(it) + 1.0)

        a = _project_onto_simplex(a - step * g, S)

        # occasional random jitter to avoid tiny plateaus
        if it % 500 == 0:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                frac = rng.uniform(0.02, 0.12)
                amount = a[i] * frac
                a[i] -= amount
                a[j] += amount
                a = _project_onto_simplex(a, S)

    return best_a, best_val


# ----------------------------------------------------------------------
# Small linear‑program polishing (maximise Σg under a∗g ≤ M)
# ----------------------------------------------------------------------
def _lp_polish(a: np.ndarray,
               rng: np.random.Generator,
               max_iter: int = 1) -> np.ndarray:
    """
    One‑step LP polish: maximise Σg with a∗g ≤ M.
    For n > 2000 a compressed version (n=2000) is used.
    Returns the mixed vector (t=0.5) projected onto the simplex.
    """
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    M = _max_convolution(a)

    # work on a compressed version if n is large
    if n > 2000:
        a_small = _scale_sequence(a, 2000)
        a_small = _project_onto_simplex(a_small, np.sqrt(2.0 * 2000))
        a_work = a_small
        work_n = 2000
    else:
        a_work = a.copy()
        work_n = n

    # build dense constraint matrix A of shape (2*work_n-1, work_n)
    # row k contains a_work[k-j] for j in the valid range
    rows = 2 * work_n - 1
    A = np.zeros((rows, work_n), dtype=float)
    for k in range(rows):
        lo = max(0, k - (work_n - 1))
        hi = min(k, work_n - 1)
        j = np.arange(lo, hi + 1, dtype=int)
        A[k, j] = a_work[k - j]

    b_ub = np.full(rows, M, dtype=float)

    # linear program: maximise Σg  <=> minimise -Σg
    c = -np.ones(work_n, dtype=float)
    bounds = [(0, None)] * work_n

    for _ in range(max_iter):
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            break
        g0 = res.x
        if g0.sum() == 0.0:
            break
        # rescale to proper ℓ₁ norm
        g = g0 * (S / g0.sum())
        # mix and project back
        a_mix = 0.5 * a + 0.5 * _scale_sequence(g, n)
        a = _project_onto_simplex(a_mix, S)

    return a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises the provided
    ``evaluate_sequence``.  Returns the best sequence as a plain Python
    list and stores it globally as ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Warm‑start handling
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # ------------------------------------------------------------------
    # Phase 0 – broad random exploration (≈ 30 % of budget)
    # ------------------------------------------------------------------
    explore_deadline = end_time - 400.0   # keep at least 400 s for refinement
    best_seq = None
    best_score = np.inf
    attempts = 0

    # candidate length distribution – more weight to medium sizes
    while time.time() < explore_deadline:
        attempts += 1
        if rng.random() < 0.65:
            n = int(rng.integers(1800, 8000))
        else:
            n = int(rng.integers(8000, 18000))

        a0 = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a0)    # this equals the evaluation score
        if peak < best_score:
            best_score = peak
            best_seq = a0.copy()
            print(f"[explore] attempt={attempts:4d} n={n:5d}  new best={best_score:.6f}")

    # fallback
    if best_seq is None:
        n = 3000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = _max_convolution(best_seq)

    # ------------------------------------------------------------------
    # Phase 1 – scaling to larger lengths (no optimisation)
    # ------------------------------------------------------------------
    cur_seq = best_seq
    cur_score = best_score
    cur_n = len(best_seq)

    # list of candidate up‑scales
    cand_lengths = [cur_n]
    for factor in [1.2, 1.5, 1.9, 2.5, 3.5]:
        L = int(cur_n * factor)
        if 2000 <= L <= 25000:
            cand_lengths.append(L)
    cand_lengths = sorted(set(cand_lengths))

    for L in cand_lengths:
        if time.time() > end_time - 300.0:
            break
        a_scaled = _scale_sequence(cur_seq, L)
        a_scaled = np.clip(a_scaled, 0.0, None)
        S = np.sqrt(2.0 * L)
        a_scaled = a_scaled * (S / a_scaled.sum())
        peak = _max_convolution(a_scaled)
        if peak < cur_score:
            cur_score = peak
            cur_seq = a_scaled.copy()
            cur_n = L
            print(f"[scale] length={L:5d}  improved max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation (multiple restarts)
    # ------------------------------------------------------------------
    remaining = end_time - time.time() - 120.0   # reserve 2 min for later phases
    if remaining > 30.0:
        # decide number of restarts (≈ 30 s each)
        n_restart = max(1, int(remaining // 30))
        per_run = remaining / n_restart
        for r in range(n_restart):
            if time.time() > end_time - 120.0:
                break
            # jitter start a bit
            noise = rng.normal(scale=0.02 * np.sqrt(2.0 * cur_n) / cur_n,
                               size=cur_n)
            a_start = np.maximum(cur_seq + noise, 0.0)
            a_start = _project_onto_simplex(a_start, np.sqrt(2.0 * cur_n))
            a_opt, peak = _adam_optimize(a_start,
                                          time.time() + per_run,
                                          rng)
            if peak < cur_score:
                cur_score = peak
                cur_seq = a_opt.copy()
                print(f"[adam] restart={r} new best max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Sub‑gradient descent (hard max)
    # ------------------------------------------------------------------
    remaining = end_time - time.time() - 70.0
    if remaining > 20.0:
        a_opt, peak = _subgradient_descent(cur_seq,
                                           time.time() + remaining,
                                           rng)
        if peak < cur_score:
            cur_score = peak
            cur_seq = a_opt.copy()
            print(f"[subgrad] after descent max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Tiny linear‑program polish (if feasible)
    # ------------------------------------------------------------------
    remaining = end_time - time.time() - 50.0
    if remaining > 10.0:
        a_polished = _lp_polish(cur_seq, rng, max_iter=2)
        peak = _max_convolution(a_polished)
        if peak < cur_score:
            cur_score = peak
            cur_seq = a_polished.copy()
            print(f"[lp] polished max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – cheap stochastic hill‑climb (few seconds)
    # ------------------------------------------------------------------
    while time.time() < end_time - 1.0:
        a = cur_seq.copy()
        n = a.shape[0]
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        frac = rng.uniform(0.02, 0.12)
        amount = a[i] * frac
        a[i] -= amount
        a[j] += amount
        a = _project_onto_simplex(a, np.sqrt(2.0 * n))
        peak = _max_convolution(a)
        if peak + 1e-9 < cur_score:
            cur_score = peak
            cur_seq = a.copy()
            print(f"[hill] improved max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    final_seq = cur_seq.tolist()
    external = evaluate_sequence(final_seq)  # type: ignore[name-defined]
    print(f"[final] external={external:.6f}  internal max={cur_score:.6f}  length={len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
