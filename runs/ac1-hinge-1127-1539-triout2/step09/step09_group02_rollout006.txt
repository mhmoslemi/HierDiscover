def improve_by_lp(a, target_sum, deadline):
    a = a.copy()
    n = a.size
    M = _max_convolution_fft(a)
    iteration = 0
    while time.time() < deadline and iteration < 50:
        iteration += 1
        # Build matrix A
        c = np.concatenate([a, np.zeros(n-1, dtype=float)])
        r = np.concatenate([a[:1], np.zeros(n-1, dtype=float)])  # first row
        A = toeplitz(c, r)  # shape (2n-1, n)
        # Setup LP
        # maximize sum(g) -> minimize -sum(g)
        c_obj = -np.ones(n, dtype=float)
        b_ub = np.full(2*n-1, M, dtype=float)
        bounds = [(0, None)]*n
        # Solve
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            print(f"LP failed at iteration {iteration}: {res.message}")
            break
        g0 = res.x
        sum_g0 = np.sum(g0)
        # if sum_g0 close to 0 skip
        if sum_g0 < 1e-12:
            break
        # scale g to target sum
        scale = target_sum / sum_g0
        g = g0 * scale
        # try mixing
        improved = False
        t = 0.2
        while t > 1e-4 and time.time() < deadline:
            a_candidate = (1 - t) * a + t * g
            M_candidate = _max_convolution_fft(a_candidate)
            if M_candidate < M - 1e-12:
                a = a_candidate
                M = M_candidate
                improved = True
                print(f"[{time.time():.1f}] LP iter {iteration} mixing t={t:.4f}, M={M:.6f}")
                break
            t *= 0.5
        if not improved:
            print(f"[{time.time():.1f}] LP iter {iteration} no mixing improvement; stop")
            break
    return a, M

Propose a step function minimizing the max self‑convolution.

Algorithm:
1. Initialise from the previous best sequence (height_sequence_1) or a
   random simplex point of length n0=2000.
2. Apply a few projected‑Adam runs (as in the baseline) to obtain a decent
   starting point.
3. Perform a series of linear‑programming updates.  For the current
   sequence a we compute M = max(a*a).  We then solve
        maximise   Σ g_j
        subject to  a * g ≤ M   (elementwise)
                    g ≥ 0.
   The solution g pushes mass to positions that are “inactive’’ in the
   convolution.  After scaling g to the simplex radius we mix it with a
   using a small step size t.  If the mixing reduces M we keep the change;
   otherwise we decrease t.  This step is repeated until no improvement
   is possible or the time budget expires.
4. Optionally up‑sample the improved sequence to a larger length and
   repeat step 3 (if time permits).
5. Final polishing with a short Adam run and a deterministic hill‑climb,
   then return the trimmed sequence.

import time

from scipy.optimize import linprog
from scipy.linalg import toeplitz

# Simplex projection (Euclidean)
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x >= 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1, dtype=float)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w

def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (fast O(n log n))."""
    n = a.size
    size = 1 << ((2 * n - 2).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())

def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a."""
    n = a.size
    size = 1 << ((2 * n - 2).bit_length())

    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    a = rng.random(n)
    a = _project_to_simplex(a, target)
    return a

def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

# ----------------------------------------------------------------------
def _lp_improvement(a: np.ndarray, target_sum: float, deadline: float,
                    max_iter: int = 30) -> tuple[np.ndarray, float]:
    """
    Perform LP‑based improvements on sequence a.
    Returns (a_new, M_new) where M_new = max(conv(a_new,a_new)).
    Stops early if time runs out or no further improvement.
    """
    n = a.size
    a_cur = a.copy()
    M_cur = _max_convolution_fft(a_cur)
    iteration = 0

    # Pre‑allocate constraint matrix (dense) – will be rebuilt each iteration
    while time.time() < deadline and iteration < max_iter:
        iteration += 1
        # Build the convolution matrix A where (A g)_k = (a_cur * g)_k
        c = np.concatenate([a_cur, np.zeros(n - 1, dtype=float)])
        r = np.concatenate([a_cur[:1], np.zeros(n - 1, dtype=float)])
        A = toeplitz(c, r)   # shape (2n‑1, n)

        # Linear program: maximise Σ g_j  ⇔ minimise −Σ g_j
        c_obj = -np.ones(n, dtype=float)
        b_ub = np.full(2 * n - 1, M_cur, dtype=float)
        bounds = [(0.0, None)] * n

        # solve
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})

        if not res.success:
            print(f"[{time.time():.1f}] LP iteration {iteration} failed: {res.message}")
            break

        g0 = res.x
        sum_g0 = np.sum(g0)
        if sum_g0 < 1e-12:
            break

        # Scale g0 to satisfy the simplex radius
        scale = target_sum / sum_g0
        g = g0 * scale

        # Try to mix a_cur and g.
        improved = False
        t = 0.25  # starting step size
        while t > 1e-5 and time.time() < deadline:
            a_candidate = (1.0 - t) * a_cur + t * g
            M_candidate = _max_convolution_fft(a_candidate)
            if M_candidate < M_cur - 1e-12:
                a_cur = a_candidate
                M_cur = M_candidate
                improved = True
                print(f"[{time.time():.1f}] LP iter {iteration}, t={t:.5f}, M={M_cur:.6f}")
                break
            t *= 0.5
        if not improved:
            # No beneficial mixing – stop the LP loop
            print(f"[{time.time():.1f}] LP iter {iteration}: no mixing improvement.")
            break

    return a_cur, M_cur

        g0 = res.x
        sum_g0 = np.sum(g0)
        if sum_g0 < 1e-12:
            break

        # Scale g0 to satisfy the simplex radius.
        scale = target_sum / sum_g0
        g = g0 * scale

        # Try to mix a_cur and g.  Reduce the step size until improvement.
        improved = False
        t = 0.25
        while t > 1e-5 and time.time() < deadline:
            a_candidate = (1.0 - t) * a_cur + t * g
            M_candidate = _max_convolution_fft(a_candidate)
            if M_candidate < M_cur - 1e-12:
                a_cur = a_candidate
                M_cur = M_candidate
                improved = True
                print(f"[{time.time():.1f}] LP iter {it}, t={t:.5f}, M={M_cur:.6f}")
                break
            t *= 0.5

        if not improved:
            # No beneficial mixing – stop early.
            print(f"[{time.time():.1f}] LP iter {it}: no mixing improvement.")
            break

    return a_cur, M_cur

# Light deterministic hill‑climb (used only for final polishing)
def _quick_hill_climb(a: np.ndarray, b: np.ndarray,
                      deadline: float) -> tuple[np.ndarray, float]:
    """Very fast greedy hill‑climb that moves a fraction of the worst mass."""
    a = a.copy()
    best_a = a.copy()
    best_score = b.max()
    n = a.size
    rng = np.random.default_rng()

    while time.time() < deadline:
        M_idx = int(np.argmax(b))
        i_low = max(0, M_idx - (n - 1))
        i_high = min(n - 1, M_idx)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M_idx - idx]
        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        # Choose destination j with smallest counterpart (avoid i)
        counterpart = a[M_idx - idx]
        order = np.argsort(counterpart)
        for cand in idx[order]:
            if cand != i:
                j = cand
                break
        else:
            break

        delta = a[i] * 0.5
        if delta <= 0:
            break

        # Update convolution exactly (O(n))
        b[i:i + n] -= 2.0 * delta * a
        b[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b.size:
            b[k2i] += delta * delta
        if k2j < b.size:
            b[k2j] += delta * delta
        if kij < b.size:
            b[kij] -= 2.0 * delta * delta

        a[i] -= delta
        a[j] += delta

        cur_max = b.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
        else:
            # Small random jitter to escape plateaus
            jitter = rng.normal(scale=1e-7, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, math.sqrt(2.0 * n))
            b = _full_convolution(a)

    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects the wall‑clock budget and returns the best sequence
    found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1) Initialise from a previous run if possible
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            init = np.asarray(globals()["height_sequence_1"], dtype=float)
            n0 = init.size
            target0 = math.sqrt(2.0 * n0)
            init = _project_to_simplex(init, target0)
            best_seq = init.copy()
            best_M = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) M={best_M:.6f}")
        except Exception as e:
            print(f"Seeding error: {e}")
            best_seq = None
            best_M = np.inf
    else:
        best_seq = None
        best_M = np.inf

    # ------------------------------------------------------------------
    # 2) If no seed, start from a random point of length 2000
    # ------------------------------------------------------------------
    if best_seq is None:
        n0 = 2000
        best_seq = _random_initial_sequence(n0, rng)
        best_M = _max_convolution_fft(best_seq)
        print(f"[{time.time():.1f}] Random start (n={n0}) M={best_M:.6f}")

    # ------------------------------------------------------------------
    # 3) Very short Adam run (helps escaping very poor local minima)
    # ------------------------------------------------------------------
    def _adam_opt(a0: np.ndarray, target: float, t_dead: float) -> np.ndarray:
        """Projected Adam on the smooth‑max surrogate (few seconds)."""
        a = a0.copy()
        n = a.size
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        base_lr = 0.15 * target
        beta = 2.0
        beta_max = 1e6
        beta_factor = 2.0
        step_no = 0

        while time.time() < t_dead:
            step_no += 1

            # smooth max via log‑sum‑exp
            size = 1 << ((2 * n - 2).bit_length())
            a_fft = np.fft.rfft(a, n=size)
            b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]
            max_b = np.max(b)
            exps = np.exp(beta * (b - max_b))
            p = exps / np.sum(exps)

            # gradient g = 2·(a ⋆ p)
            a_rev = a[::-1]
            conv_len = n + (2 * n - 1) - 1
            size2 = 1 << ((conv_len - 1).bit_length())
            A_rev_fft = np.fft.rfft(a_rev, n=size2)
            P_fft = np.fft.rfft(p, n=size2)
            conv_rev = np.fft.irfft(A_rev_fft * P_fft, n=size2)[:conv_len]
            g = 2.0 * conv_rev[n - 1: n - 1 + n]

            # Adam update
            m = beta1 * m + (1 - beta1) * g
            v = beta2 * v + (1 - beta2) * (g * g)
            m_hat = m / (1 - beta1 ** step_no)
            v_hat = v / (1 - beta2 ** step_no)
            step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

            a -= step
            a = _project_to_simplex(a, target)

            if step_no % 200 == 0 and beta < beta_max:
                beta = min(beta * beta_factor, beta_max)

        return a

    if time.time() < deadline - 5.0:
        adam_deadline = min(deadline - 0.5, time.time() + 5.0)
        a_adam = _adam_opt(best_seq, math.sqrt(2.0 * best_seq.size), adam_deadline)
        M_adam = _max_convolution_fft(a_adam)
        if M_adam < best_M - 1e-12:
            best_seq, best_M = a_adam, M_adam
            print(f"[{time.time():.1f}] Adam refinement improved M={best_M:.6f}")

    # ------------------------------------------------------------------
    # 4) Core LP refinement on the current length
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        lp_deadline = deadline - 1.0
        a_lp, M_lp = _lp_improvement(best_seq,
                                      math.sqrt(2.0 * best_seq.size),
                                      lp_deadline)
        if M_lp < best_M - 1e-12:
            best_seq, best_M = a_lp, M_lp
            print(f"[{time.time():.1f}] LP refinement final M={best_M:.6f}")

    # ------------------------------------------------------------------
    # 5) Optional up‑sampling to larger lengths (if time permits)
    # ------------------------------------------------------------------
    for new_n in (4000, 8000, 20000):
        if time.time() >= deadline - 2.0:
            break
        a_big = _upsample_sequence(best_seq, new_n)
        target_big = math.sqrt(2.0 * new_n)
        lp_dead = time.time() + min(30.0, deadline - time.time() - 1.0)
        a_big_ref, M_big = _lp_improvement(a_big, target_big, lp_dead,
                                            max_iter=12)
        if M_big < best_M - 1e-12:
            best_seq, best_M = a_big_ref, M_big
            print(f"[{time.time():.1f}] Upscaled to n={new_n}, M={best_M:.6f}")

    # ------------------------------------------------------------------
    # 6) Final polishing: short Adam + quick hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        # short Adam
        adam_dead = min(deadline - 0.5, time.time() + 5.0)
        best_seq = _adam_opt(best_seq, math.sqrt(2.0 * best_seq.size), adam_dead)

        # quick hill‑climb
        b_cur = _full_convolution(best_seq)
        best_seq, final_M = _quick_hill_climb(best_seq, b_cur,
                                              deadline - 0.5)
        if final_M < best_M - 1e-12:
            best_M = final_M
            print(f"[{time.time():.1f}] Final hill‑climb improvement M={best_M:.6f}")

    # ------------------------------------------------------------------
    # 7) Trim trailing near‑zeros and report
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:   # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
