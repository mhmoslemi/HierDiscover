"""
Hybrid optimizer for the max‑convolution problem.

The objective (with the ℓ¹‑normalisation Σa_i = √(2·n)) reduces to
    max_{k} (a * a)[k] .
We minimise this quantity by a sequence of stages:

1.   Warm‑starts from the best known construction (if any) and from
     several random families (uniform, power‑law, exponential, …).
2.   Projected Adam on a smooth (log‑sum‑exp) surrogate of the max.
     The learning‑rate, β‑schedule and jitter are tuned for fast
     convergence.
3.   Deterministic hill‑climbing that moves mass from the source that
     contributes most to the current worst convolution entry to the
     most favourable destination.
4.   A cheap stochastic local search that performs random mass moves
     with an O(n) convolution update; it explores directions that the
     greedy hill‑climb may miss.
5.   Optional mixing of the current best sequence with the uniform
     distribution (and a pure exponential) – a cheap way of flattening
     peaks.

The routine respects a wall‑clock budget (default 1000 s) and always
returns the best sequence found.  All operations are vectorised and
use FFTs where appropriate.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Simple linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using an FFT (fast, O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float) -> Tuple[np.ndarray, np.ndarray]:
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float,
                   deadline: float,
                   beta0: float = 5.0,
                   beta_max: float = 1e7,
                   beta_mul: float = 1.5) -> np.ndarray:
    """
    Projected Adam on the smoothed‑max surrogate.
    Stops when wall‑clock time exceeds *deadline*.
    """
    a = a0.copy()
    n = a.size

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # learning‑rate scaled with the simplex radius
    base_lr = 0.22 * target_sum

    beta = beta0
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat plateaus
        if t % 200 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4 / (beta + 1.0),
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # increase smoothing slowly
        if t % 250 == 0 and beta < beta_max:
            beta = min(beta * beta_mul, beta_max)

    return a

def _random_initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)
    from a handful of simple families.
    """
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float,
                              move_frac: float = 0.8) -> Tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that always attacks the current worst convolution entry.
    *move_frac* is the fraction of source mass moved in each successful step.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()

    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))

        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]

        i = idx[np.argmax(contrib)]
        if a[i] < 1e-14:
            break

        # counterpart values for possible destinations
        counterpart = np.zeros(n, dtype=float)
        j_low = i_low
        j_high = i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        # destination with minimal counterpart (but not i)
        order = np.argsort(counterpart[j_range])
        j = -1
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j == -1:
            break

        delta = a[i] * move_frac
        if delta <= 0.0:
            break

        # Incremental convolution update (O(n))
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < b.max() - 1e-14:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-14:
                best_score = new_max
                best_a = a.copy()
        else:
            # try a smaller move (binary‑search style)
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-16:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-14:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-14:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                # tiny random jitter to escape a flat region
                jitter = rng.normal(scale=1e-7, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                target = math.sqrt(2.0 * n)
                a = _project_to_simplex(a, target)
                b = _full_convolution(a)

    return best_a, best_score

def _random_local_search(a0: np.ndarray, b0: np.ndarray,
                         deadline: float,
                         move_frac_range: Tuple[float, float] = (0.02, 0.08)) -> Tuple[np.ndarray, float]:
    """
    Very cheap random mass‑transfer local search.
    Each move transfers a random fraction of a source entry to a random destination.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_max = b.max()
    rng = np.random.default_rng()

    while time.time() < deadline:
        # pick a source with non‑trivial mass
        i = rng.integers(0, n)
        if a[i] < 1e-12:
            continue
        # random destination different from i
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # random fraction of source mass
        frac = rng.random() * (move_frac_range[1] - move_frac_range[0]) + move_frac_range[0]
        delta = a[i] * frac
        if delta < 1e-16:
            continue

        # incremental update of the full convolution
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        cur_max = b_new.max()
        if cur_max < best_max - 1e-14:
            a[i] -= delta
            a[j] += delta
            b = b_new
            best_max = cur_max
            best_a = a.copy()

    return best_a, best_max

def _mix_with_family(a: np.ndarray,
                     family_seq: np.ndarray,
                     target_sum: float,
                     mix_steps: int = 11) -> Tuple[np.ndarray, float]:
    """
    Form convex combinations λ*a + (1‑λ)*family_seq and keep the best.
    """
    best_seq = a.copy()
    best_score = _max_convolution_fft(best_seq)

    for t in range(mix_steps):
        lam = t / (mix_steps - 1)            # λ from 0 to 1
        cand = lam * a + (1.0 - lam) * family_seq
        cand = _project_to_simplex(cand, target_sum)
        score = _max_convolution_fft(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects the wall‑clock budget
    (default 1000 s) and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.6   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 0)  Warm‑start from a previous solution if present
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            n0 = prev.size
            target0 = math.sqrt(2.0 * n0)
            prev = _project_to_simplex(prev, target0)
            best_seq = prev.copy()
            best_score = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 "
                  f"(n={n0}) score={best_score:.6f}")
        except Exception as e:
            print(f"Seeding error: {e}")

    # ------------------------------------------------------------------
    # 1)  Length schedule – we start small and gradually increase n
    # ------------------------------------------------------------------
    schedule = [2000, 4000, 8000, 16000, 32000, 64000]

    for n in schedule:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Initialise a candidate for this length
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)
        a_init = _project_to_simplex(a_init, target_sum)

        # --------------------------------------------------------------
        # Time slicing for this dimension
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 12.0:
            break

        # We give a larger share to the larger dimensions
        # (roughly proportional to log n)
        factor = math.log(n) / sum(math.log(m) for m in schedule)
        gd_budget = remaining * (0.5 * factor + 0.25)   # ≥ 25 %
        hc_budget = remaining * (0.3 * factor + 0.15)   # ≥ 15 %
        extra_budget = remaining - gd_budget - hc_budget

        # --------------------------------------------------------------
        # Phase A – Projected Adam (multiple restarts)
        # --------------------------------------------------------------
        min_restart_time = 3.0
        n_restarts = max(1, int(gd_budget // min_restart_time))
        per_restart = gd_budget / n_restarts

        local_best_a = None
        local_best_score = np.inf

        for r in range(n_restarts):
            if time.time() >= deadline:
                break
            this_deadline = min(deadline - 0.3, time.time() + per_restart)
            start_a = a_init.copy() if r == 0 else _random_initial_sequence(n, rng)
            a_opt = _adam_optimize(start_a, target_sum, this_deadline,
                                   beta0=5.0, beta_max=1e7, beta_mul=1.5)
            score = _max_convolution_fft(a_opt)
            if score < local_best_score:
                local_best_score = score
                local_best_a = a_opt.copy()

        if local_best_a is not None and local_best_score < best_score - 1e-12:
            best_seq = local_best_a.copy()
            best_score = local_best_score
            print(f"[{time.time():.1f}] Adam improvement (n={n}) "
                  f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # Phase B – Deterministic hill‑climbing
        # --------------------------------------------------------------
        if time.time() < deadline - 2.0:
            a_cur = best_seq.copy()
            a_cur = _project_to_simplex(a_cur, target_sum)
            b_cur = _full_convolution(a_cur)
            hc_deadline = min(deadline - 0.3, time.time() + hc_budget)
            a_hc, score_hc = _hill_climb_deterministic(a_cur, b_cur, hc_deadline,
                                                       move_frac=0.6)
            if score_hc < best_score - 1e-12:
                best_seq = a_hc.copy()
                best_score = score_hc
                print(f"[{time.time():.1f}] Hill‑climb improvement (n={n}) "
                      f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # Phase C – Random local search (cheap stochastic moves)
        # --------------------------------------------------------------
        if time.time() < deadline - 2.0:
            a_cur = best_seq.copy()
            b_cur = _full_convolution(a_cur)
            rl_deadline = min(deadline - 0.3, time.time() + extra_budget * 0.5)
            a_rl, score_rl = _random_local_search(a_cur, b_cur, rl_deadline)
            if score_rl < best_score - 1e-12:
                best_seq = a_rl.copy()
                best_score = score_rl
                print(f"[{time.time():.1f}] Random‑search improvement (n={n}) "
                      f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # Phase D – Mixing with simple families (uniform / exponential)
        # --------------------------------------------------------------
        if time.time() < deadline - 1.0:
            # uniform family
            uniform_seq = np.full(n, target_sum / n)
            seq_mix, score_mix = _mix_with_family(best_seq, uniform_seq,
                                                  target_sum, mix_steps=9)
            # simple exponential family (rate chosen from a small set)
            lam = 0.005 * (1 + (n // 2000) % 5)   # a modest deterministic rate
            exp_seq = np.exp(-lam * np.arange(n))
            exp_seq = _project_to_simplex(exp_seq, target_sum)
            seq_mix2, score_mix2 = _mix_with_family(seq_mix, exp_seq,
                                                    target_sum, mix_steps=7)
            if score_mix2 < best_score - 1e-12:
                best_seq = seq_mix2.copy()
                best_score = score_mix2
                print(f"[{time.time():.1f}] Mixing improvement (n={n}) "
                      f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Final polishing on the largest dimension obtained so far
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        n = best_seq.size
        target_sum = math.sqrt(2.0 * n)

        # a longer Adam run using all remaining budget
        final_deadline = deadline - 0.5
        best_seq = _adam_optimize(best_seq, target_sum, final_deadline,
                                  beta0=5.0, beta_max=1e7, beta_mul=1.5)

        # deterministic hill‑climb again (short)
        b_final = _full_convolution(best_seq)
        hc_deadline = deadline - 0.3
        a_final, final_score = _hill_climb_deterministic(best_seq, b_final,
                                                          hc_deadline,
                                                          move_frac=0.5)
        if final_score < best_score - 1e-12:
            best_seq = a_final.copy()
            best_score = final_score
            print(f"[{time.time():.1f}] Final hill‑climb improvement: "
                  f"score={best_score:.6f}")

        # final tiny random search (last few seconds)
        if time.time() < deadline - 0.5:
            b_cur = _full_convolution(best_seq)
            rl_deadline = deadline - 0.1
            a_rl, score_rl = _random_local_search(best_seq, b_cur, rl_deadline)
            if score_rl < best_score - 1e-12:
                best_seq = a_rl.copy()
                best_score = score_rl
                print(f"[{time.time():.1f}] Final random‑search improvement: "
                      f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Clean‑up: drop trailing near‑zeros
    # ------------------------------------------------------------------
    if best_seq is None:
        # ultra‑fallback: uniform with n=2000
        n = 2000
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # 4) Final official evaluation (using the provided evaluator)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
