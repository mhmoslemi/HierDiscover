def _smooth_max_and_gradient(a: np.ndarray, alpha: float):
    n = a.shape[0]
    # c = a*a convolution
    c = _full_convolution_fft(a)
    max_c = c.max()
    # compute weight distribution (softmax)
    offset = c - max_c
    exp_arg = alpha * offset
    # To avoid overflow for large alpha if offset large negative => exp small; fine.
    w_raw = np.exp(exp_arg)
    w_sum = w_raw.sum()
    if w_sum == 0.0:
        w = np.full_like(w_raw, 1.0 / w_raw.size)
    else:
        w = w_raw / w_sum
    # compute gradient: conv(w, rev(a))
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    f_len = _next_pow2(conv_len)
    Fw = np.fft.rfft(w, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fw * Fa_rev, n=f_len)[:conv_len]
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    # smooth objective value
    smooth_val = max_c + np.log(w_sum) / alpha
    return grad, smooth_val, max_c

"""
Step‑function optimizer (≈ 1000 s budget).

Goal: minimise   2·n·max(a∗a)/(∑a)²   – with the sum forced to √(2 n) this
is exactly the maximum entry of the aperiodic autocorrelation.
The optimizer combines three ideas:

1. **Broad random search** – many feasible starts drawn from several
   parametric families (power‑law, exponential, linear, U‑shape,
   sinusoid, Gaussian, Dirichlet, uniform).  A warm‑start from the
   best sequence found so far is used occasionally.

2. **Smooth‑max gradient descent** – the non‑differentiable max is
   replaced by the log‑sum‑exp (smooth‑max) with a growing temperature
   parameter α.  The gradient of the smooth‑max is a simple weighted
   convolution, so each iteration costs only a few FFTs.  After each
   step the vector is projected onto the simplex
   {a≥0, ∑a=√(2 n)}.

3. **Final polishing** – a short projected sub‑gradient descent
   (the classic method used before) followed by a cheap simulated‑
   annealing mass‑transfer hill‑climb.

The whole routine respects the time budget, keeps the best sequence
seen and stores it in the global variable ``height_sequence_1``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT helpers, projection, scaling)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a∗a."""
    return _full_convolution_fft(a).max()


def _project_onto_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex
        { w | w_i ≥ 0 , ∑ w_i = S }.
    Implements the O(n) algorithm of Duchi et al.
    """
    if S < 0:
        raise ValueError("Simplex sum must be non‑negative")
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]

def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    raw = mix * ush + (1.0 - mix) * base
    return raw


def _init_sine(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    vals = np.sin(np.pi * (i + 1) / (n + 1))
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * vals + (1.0 - mix) * base
    return raw


def _init_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    mu = (n - 1) / 2.0
    sigma = rng.uniform(n / 10.0, n / 3.0)
    gauss = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * gauss + (1.0 - mix) * base
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    return rng.random(n)


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Occasionally warm‑starts from the globally best sequence and otherwise
    draws from a random parametric family.
    """
    S = np.sqrt(2.0 * n)

    # Warm start with probability ≈0.22
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
        return a

    # Choose a family at random
    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'ushape', 'sine', 'gaussian',
                        'uniform', 'dirichlet'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'sine':
        raw = _init_sine(n, rng)
    elif style == 'gaussian':
        raw = _init_gaussian(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Smooth‑max helpers
# ----------------------------------------------------------------------
def _smooth_max_and_gradient(a: np.ndarray, alpha: float):
    """
    Returns (grad, smooth_val, hard_max) where
      grad       – sub‑gradient of the *smooth* max with respect to a,
      smooth_val – (1/α)·log∑exp(α·c)  where c = a∗a,
      hard_max   – max(c) (the true objective we finally care about).
    """
    n = a.shape[0]
    # c = a∗a
    c = _full_convolution_fft(a)
    max_c = c.max()

    # soft‑max weights
    offset = c - max_c                     # ≤ 0
    exp_arg = alpha * offset
    w_raw = np.exp(exp_arg)                # safe, never overflows
    w_sum = w_raw.sum()
    if w_sum == 0.0:
        w = np.full_like(w_raw, 1.0 / w_raw.size)
    else:
        w = w_raw / w_sum                    # probability vector

    # gradient = 2 * (w ∗ rev(a))  (central n entries)
    a_rev = a[::-1]
    conv_len = len(w) + n - 1               # = 3n‑2
    f_len = _next_pow2(conv_len)
    Fw = np.fft.rfft(w, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fw * Fa_rev, n=f_len)[:conv_len]
    start = n - 1
    grad = 2.0 * conv[start:start + n]

    # smooth objective value
    smooth_val = max_c + np.log(w_sum) / alpha
    return grad, smooth_val, max_c


def _smooth_gradient_descent(a0: np.ndarray,
                             alphas,
                             max_iters_per_alpha: int,
                             step0: float,
                             rng: np.random.Generator,
                             deadline: float):
    """
    Projected gradient descent on the smooth‑max objective.
    The schedule of alphas grows (low temperature → high temperature).
    Returns the best vector (w.r.t. the *hard* max) found and its hard
    max value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _max_convolution(a)

    for alpha in alphas:
        # a modest diminishing step size inside each α‑phase
        step = step0
        for it in range(max_iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_val

            grad, smooth_val, hard_val = _smooth_max_and_gradient(a, alpha)

            # Polyak‑type step (only if we are above the best seen value)
            norm_sq = np.dot(grad, grad)
            if norm_sq > 0:
                step = max(0.0, (smooth_val - best_val) / (norm_sq + 1e-12))
                # keep step in a safe interval
                step = np.clip(step, 1e-7, 0.2)

            # projected gradient step
            a = _project_onto_simplex(a - step * grad, S)

            # evaluate hard max
            cur_hard = _max_convolution(a)
            if cur_hard < best_val:
                best_val = cur_hard
                best_a = a.copy()

            # occasional tiny jitter to escape plateaus
            if it % 250 == 0:
                jitter = rng.normal(scale=0.001 * np.mean(a), size=n)
                a = _project_onto_simplex(a + jitter, S)

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (as used previously)
# ----------------------------------------------------------------------
def _max_convolution_and_gradient(a: np.ndarray):
    """
    Returns (grad, max_val) where grad is a sub‑gradient of
    maxₖ(a∗a)[k] with respect to a and max_val is that maximum.
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_val = b.max()
    mask = (b == max_val).astype(float)

    a_rev = a[::-1]
    conv_len = len(mask) + n - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad, max_val


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         global_best: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=max(a∗a).
    Uses a Polyak step size and occasional Gaussian jitter.
    Returns the best vector seen and its hard‑max value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_val = _max_convolution_and_gradient(a)

        norm_sq = np.dot(grad, grad)
        if norm_sq > 0:
            step = (cur_val - best_val) / (norm_sq + 1e-12)
            step = np.clip(step, 1e-7, 0.2)
        else:
            step = 0.0

        a = _project_onto_simplex(a - step * grad, S)

        cur_val = _max_convolution(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()

        if it % 300 == 0:
            jitter = rng.normal(scale=0.001 * np.mean(a), size=n)
            a = _project_onto_simplex(a + jitter, S)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb (same as before)
# ----------------------------------------------------------------------
def _anneal_local_search(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Moves a random fraction of mass between two positions; accepts
    worsening moves with an exponentially decaying temperature.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    cur_val = _max_convolution(a)
    best_a = a.copy()
    best_val = cur_val

    T = 0.05 * cur_val
    decay = 0.9999

    it = 0
    while time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        frac = rng.uniform(0.02, 0.18)
        delta = a[i] * frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _max_convolution(a_new)
        delta_val = new_val - cur_val
        if delta_val < 0 or rng.random() < np.exp(-delta_val / (T + 1e-12)):
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()

        T *= decay
        if it % 500 == 0:
            jitter = rng.normal(scale=0.001 * S, size=n)
            a = _project_onto_simplex(a + jitter, S)
            cur_val = _max_convolution(a)

    return best_a, best_val

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises the provided
    ``evaluate_sequence``.  The routine follows the four‑stage plan
    described in the module docstring.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    end_time = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # warm‑start handling
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    best_seq = None
    best_val = np.inf
    attempts = 0

    # ------------------------------------------------------------------
    # Phase 1 – broad random exploration + cheap smooth‑max refinement
    # ------------------------------------------------------------------
    explore_deadline = end_time - 250.0   # reserve ~250 s for deeper refinement
    while time.time() < explore_deadline:
        attempts += 1
        # bias towards longer sequences (3 k–12 k) but also sometimes longer
        if rng.random() < 0.75:
            n = int(rng.integers(3000, 12000))
        else:
            n = int(rng.integers(12000, 20000))

        a0 = _initial_vector(n, rng, warm_seq)

        # a short smooth‑max run (few alphas, few iterations) to get a
        # decent candidate quickly
        cand_deadline = time.time() + min(6.0,
                                          explore_deadline - time.time())
        a_opt, val = _smooth_gradient_descent(
            a0,
            alphas=[0.5, 1.0, 2.0, 5.0],
            max_iters_per_alpha=50,
            step0=0.5,
            rng=rng,
            deadline=cand_deadline)

        if val < best_val:
            best_val = val
            best_seq = a_opt.copy()
            best_n = n
            print(f"[explore] attempt={attempts:4d} n={n:5d}  new best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – length scaling & aggressive sub‑gradient refinement
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback to a modest uniform candidate
        n = 4000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_val = _max_convolution(best_seq)
        best_n = n

    # a handful of larger target lengths (geometric progression)
    lengths = [best_n]
    for factor in [1.3, 1.8, 2.5, 3.5]:
        cand = int(best_n * factor)
        if 3000 <= cand <= 25000:
            lengths.append(cand)
    lengths = sorted(set(lengths))

    for target_n in lengths:
        if time.time() >= end_time - 150.0:
            break
        a_scaled = _scale_sequence(best_seq, target_n)
        S = np.sqrt(2.0 * target_n)
        a_scaled = np.clip(a_scaled, 0.0, None)
        a_scaled = a_scaled * (S / a_scaled.sum())

        # allocate a slice of the remaining budget (≈15 % of what is left)
        remaining = end_time - time.time()
        slice_time = min(remaining * 0.15, 80.0)

        a_opt, val = _subgradient_descent(a_scaled,
                                          time.time() + slice_time,
                                          rng,
                                          best_val)
        if val < best_val:
            best_val = val
            best_seq = a_opt.copy()
            best_n = target_n
            print(f"[subgrad] target_n={target_n:5d}  improved={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – long sub‑gradient run on the current best length
    # ------------------------------------------------------------------
    if time.time() < end_time - 50.0:
        remaining = end_time - time.time() - 5.0
        a_opt, val = _subgradient_descent(best_seq,
                                          time.time() + remaining,
                                          rng,
                                          best_val)
        if val < best_val:
            best_val = val
            best_seq = a_opt.copy()
            print(f"[final subgrad] length={len(best_seq)}  best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if time.time() < end_time - 5.0:
        remaining = end_time - time.time() - 1.0
        a_opt, val = _anneal_local_search(best_seq,
                                          time.time() + remaining,
                                          rng)
        if val < best_val:
            best_val = val
            best_seq = a_opt.copy()
            print(f"[anneal] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f}  internal={best_val:.6f}  length={len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
