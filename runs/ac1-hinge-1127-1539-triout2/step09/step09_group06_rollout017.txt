    end_time = time.time() + budget_s - 5

    # warm start

        except: warm_seq = None

    # Phase 1: random exploration
    best_seq, best_score, best_n = None, np.inf, None
    while time.time() < end_time - 600:
        # generate length
        if rng.random() < 0.7:
            n = int(rng.integers(1800, 6000))
        else:
            n = int(rng.integers(6000, 12000))
        a0 = _initial_vector(n, rng, warm_seq)
        # internal max conv value after scaling automatically computed by _initial_vector's scaling to S?
        # Actually _initial_vector returns vector with sum = sqrt(2*n).
        score = _max_convolution(a0)  # corresponds to evaluation since sum normalized
        if score < best_score:
            best_score = score
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] n={n} new best={best_score:.6f}")

    if best_seq is None:
        # fallback
        n = 2500
        best_seq = np.full(n, np.sqrt(2*n)/n)
        best_score = _max_convolution(best_seq)
        best_n = n

    # Phase 2: upscale lengths enumerated
    # List candidate lengths as before
    cand_lengths = [best_n] + [int(best_n * factor) for factor in [1.25,1.5,2.0,3.0]]
    cand_lengths = [L for L in cand_lengths if 2000 <= L <= 25000]
    cand_lengths = sorted(set(cand_lengths))
    cur_seq = best_seq; cur_score = best_score; cur_n = best_n

    for L in cand_lengths:

            cur_score = peak
            cur_n = L
            print(f"[scale] L={L} improved max={cur_score:.6f}")

    # Phase 3: Adam optimization
    # use some restarts: allocate time e.g., 120 sec for Adam
    adam_time_budget = min(120.0, (end_time - time.time()) / 2) # allocate half of remaining for Adam if enough
    if adam_time_budget > 10:

        a_opt, peak = _subgradient_descent(cur_seq, time.time() + sub_time, rng)
        if peak < cur_score:
            cur_score = peak
            cur_seq = a_opt.copy()
            print(f"[subgrad] new max={cur_score:.6f}")

            cur_seq = a_opt.copy()

    print(f"[final] external={external:.6f} internal max={cur_score:.6f} length={len(final_seq)}")

    return final_seq

def _lp_refine(a0, deadline, rng):
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)
    a = a0.copy()
    # ensure sum = S
    total = a.sum()
    if total != 0:
        a = a * (S / total)
    else:
        a = np.full(n, S / n)

    best_a = a.copy()
    best_M = _max_convolution(a)

    # compute initial active set indices where convolution close to max
    conv = _full_convolution_fft(a)
    active = set(np.where(conv >= best_M - 1e-12)[0].tolist())

    t = 0.5  # mixing coefficient
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        # Build constraint matrix
        active_list = sorted(active)
        m = len(active_list)
        A = np.zeros((m, n), dtype=np.float64)
        for row_idx, k in enumerate(active_list):
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            A[row_idx, i_start:i_end+1] = a[i_start:i_end+1]
        b = np.full(m, best_M)

        # solve LP
        c = -np.ones(n)  # maximize sum(g)
        bounds = [(0, None)] * n
        # Use scipy.optimize.linprog
        try:
            res = scipy.optimize.linprog(c, A_ub=A, b_ub=b,
                                        bounds=bounds,
                                        method='highs')
        except Exception as e:
            print("[lp] linprog failed:", e)
            break
        if not res.success:
            print("[lp] linprog not successful")
            break
        g = res.x

        # Compute f*g convolution
        conv_fg = _full_convolution_fft_two(a, g)
        # Identify any violations > M + eps
        eps = 1e-12
        viol = np.where(conv_fg > best_M + eps)[0]
        if viol.size > 0:
            # add them
            updated = False
            for idx in viol:
                if idx not in active:
                    active.add(int(idx))
                    updated = True
            if updated:
                continue  # re-solve with new constraints
        # No violations; proceed to mixing
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g_scaled = g * (S / sum_g)
        # mixing weight t
        a_new = (1 - t) * a + t * g_scaled
        # ensure sum = S
        a_new = a_new * (S / a_new.sum())
        M_new = _max_convolution(a_new)

        if M_new < best_M - 1e-12:
            best_M = M_new
            best_a = a_new.copy()
            a = a_new
            # recompute active set to new maxima
            conv_new = _full_convolution_fft(a)
            active = set(np.where(conv_new >= best_M - 1e-12)[0].tolist())
            # reduce t slowly
            t = max(0.05, t * 0.9)
            # break condition maybe
        else:
            # no improvement; decrease t
            t = max(0.05, t * 0.7)
            if t <= 0.05:
                # can't improve further
                break
        # safety break
        if iteration >= 200:
            break
    return best_a, best_M

def _build_convolution_constraint_matrix(f):
    n = len(f)
    rows = []
    cols = []
    data = []
    for i in range(n):
        # rows i to i + n - 1
        rows_i = np.arange(i, i + n, dtype=np.int32)
        cols_i = np.full(n, i, dtype=np.int32)
        rows.append(rows_i)
        cols.append(cols_i)
        data.append(np.full(n, f[i], dtype=np.float64))
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    data = np.concatenate(data)
    # shape (2n-1, n)
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return A

"""
Step‑function optimizer.

The algorithm works in several stages inside the 1000 s budget:

1. **Random exploration** – sample many parametric families (power‑law,
   exponential, linear, U‑shape, cosine, Dirichlet, gamma, uniform) and
   keep the best candidate (normalised to ℓ₁‑norm √(2·n)).

2. **Length up‑scaling** – linearly interpolate the current best shape to
   a few larger lengths and keep any improvement.

3. **Continuous optimisation** – a few restarts of Adam on a τ‑soft‑max
   surrogate of the max‑convolution, followed by a sub‑gradient descent
   on the hard maximum.

4. **Linear‑programming refinement** – the core new idea.
   For the current sequence *f* we compute M = max(f∗f).
   We then solve the LP

       maximise   Σ g_i
       subject to g_i ≥ 0
                  (f∗g)_k ≤ M   for all k

   (the convolution constraints are linear in *g*).  The LP is solved
   with SciPy’s HiGHS interior‑point method using a sparse constraint
   matrix.  The solution *g* is rescaled to the same ℓ₁‑norm as *f* and
   mixed with *f* (f ← (1‑t) f + t g).  Repeating this step reduces M
   and therefore the evaluation score.

5. **Cheap hill‑climbing** – stochastic moves of a small mass between
   two positions, accepting only strict improvements.

The final sequence is returned as a plain Python list and stored in the
global variable ``height_sequence_1`` for warm‑starting the next run.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse

# ----------------------------------------------------------------------
# Basic convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a∗a (full length) using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _full_convolution_fft_two(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a∗b using a real FFT."""
    n1 = a.shape[0]
    n2 = b.shape[0]
    total_len = n1 + n2 - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    B = np.fft.rfft(b, n=f_len)
    conv = np.fft.irfft(A * B, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution."""
    return _full_convolution_fft(a).max()


# ----------------------------------------------------------------------
# Sequence scaling / interpolation helpers
# ----------------------------------------------------------------------
def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]

def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    raw = mix * ush + (1.0 - mix) * base
    return raw


def _init_cosine(n: int, rng: np.random.Generator) -> np.ndarray:

    raw = 1.0 - np.cos(np.pi * i / (n - 1))
    raw = raw * rng.uniform(0.5, 2.0)

    base = np.ones(n)
    return mix * raw + (1.0 - mix) * base

    if warm_seq is not None and rng.random() < 0.23:

        a = np.clip(a, 0.0, None)

        return a

    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# τ‑soft‑max gradient (for Adam)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex (log‑softmax) aiming at the hard max.
    Returns best sequence and its hard peak value.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                     # ℓ₁‑norm to keep fixed

    # initialise logits from a feasible a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                         # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.45

    # temperature for the soft‑max approx.
    tau = max(_full_convolution_fft(a0).max(), 5.0)

    best_a = a0.copy()
    best_peak = _max_convolution(a0)
    it = 0

    while time.time() < deadline:
        it += 1

        # recover a from logits
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # gradient of τ‑soft‑max
        g, peak = _softmax_gradient(a, tau)

        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # gradient w.r.t. logits
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # clip for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 6.0 * S:
            grad_p = grad_p * (6.0 * S / clip_val)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # schedules
        tau *= 0.99995
        lr *= 0.99995

        if it % 800 == 0:
            p += rng.normal(scale=0.006, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Sub‑gradient descent on the hard max
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple sub‑gradient descent that follows the current maximising
    convolution entry.  Projects back onto the simplex after each step.
    Returns best sequence and its hard peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    a = a0.copy()
    best_a = a.copy()
    best_val = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1

        b = _full_convolution_fft(a)
        k = int(b.argmax())
        val = b[k]

        if val < best_val:
            best_val = val
            best_a = a.copy()

        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        jdx = k - idx
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[jdx]

        step = 0.8 * S / (np.sqrt(it) + 1.0)
        a = a - step * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, S / n)
        else:
            a = a * (S / s)

        if it % 500 == 0:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                frac = rng.uniform(0.02, 0.12)
                amount = a[i] * frac
                a[i] -= amount
                a[j] += amount
                a = a * (S / a.sum())

    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement (core new contribution)
# ----------------------------------------------------------------------
def _lp_refine(a0: np.ndarray,
               deadline: float,
               rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Iteratively improve a sequence by solving the LP
        max Σ g_i   s.t.   g_i ≥ 0,   (a∗g)_k ≤ M   ∀k,
    where M = max(a∗a).  The solution g is scaled to the same ℓ₁‑norm
    as a and mixed back (a ← (1‑t)a + t g).  Returns the improved
    sequence and its max‑convolution value.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    # start from a feasible point with the required norm
    a = np.maximum(a0, 0.0)
    total = a.sum()
    if total > 0:
        a = a * (S / total)
    else:
        a = np.full(n, S / n)

    best_a = a.copy()
    best_M = _max_convolution(a)

    # mixing coefficient (starts fairly aggressive)
    t = 0.4
    max_iters = 30

    for itr in range(max_iters):
        if time.time() > deadline - 10:
            break

        # ------------------------------------------------------------------
        # Build the full convolution constraint matrix A (sparse)
        # Row k corresponds to (a∗g)_k ≤ M
        # For each column i the coefficient a_i appears in rows i … i+n‑1
        # ------------------------------------------------------------------
        rows = np.empty(n * n, dtype=np.int32)
        cols = np.empty(n * n, dtype=np.int32)
        data = np.empty(n * n, dtype=np.float64)
        pos = 0
        for i in range(n):
            rows[pos:pos + n] = np.arange(i, i + n, dtype=np.int32)
            cols[pos:pos + n] = i
            data[pos:pos + n] = a[i]
            pos += n
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

        b_vec = np.full(2 * n - 1, best_M, dtype=np.float64)

        # ------------------------------------------------------------------
        # Solve the LP: maximise Σ g_i  ⇔ minimise -Σ g_i
        # ------------------------------------------------------------------
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None)] * n
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_vec,
                                    bounds=bounds, method='highs')
        if not res.success:
            # if the LP fails (unlikely), abort refinement
            print("[lp] linprog failed:", res.message)
            break

        g = res.x
        sum_g = g.sum()
        if sum_g == 0.0:
            break
        # Scale g to the same ℓ₁‑norm as a
        g_scaled = g * (S / sum_g)

        # Mix and renormalise
        a_new = (1.0 - t) * a + t * g_scaled
        a_new = a_new * (S / a_new.sum())
        M_new = _max_convolution(a_new)

        # Accept improvement
        if M_new < best_M - 1e-12:
            best_M = M_new
            best_a = a_new.copy()
            a = a_new
            # tighten mixing
            t = max(0.05, t * 0.9)
        else:
            # No progress – shrink the step size
            t = max(0.05, t * 0.6)
            if t <= 0.05:
                # cannot make further progress
                break

    return best_a, best_M


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best sequence as a plain list and stores it in the global
    variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Warm‑start handling
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # ------------------------------------------------------------------
    # Phase 1 – fast random exploration (≈ 20 % of the budget)
    # ------------------------------------------------------------------
    explore_deadline = end_time - 600.0   # keep at least ~600 s for refinement
    best_seq = None
    best_score = np.inf
    best_n = None
    attempts = 0

    while time.time() < explore_deadline:
        attempts += 1
        if rng.random() < 0.70:
            n = int(rng.integers(1800, 6000))
        else:
            n = int(rng.integers(6000, 12000))
        a0 = _initial_vector(n, rng, warm_seq)
        score = _max_convolution(a0)      # equal to external evaluation after normalisation
        if score < best_score:
            best_score = score
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempts:4d} n={n:5d}  new best={best_score:.6f}")

    if best_seq is None:
        # fallback to a modest uniform vector
        best_n = 2500
        best_seq = np.full(best_n, np.sqrt(2 * best_n) / best_n)
        best_score = _max_convolution(best_seq)

    # ------------------------------------------------------------------
    # Phase 2 – upscale to a few larger lengths (no optimiser)
    # ------------------------------------------------------------------
    cand_lengths = [best_n]
    for factor in [1.25, 1.5, 2.0, 3.0, 4.0, 5.0]:
        cand = int(best_n * factor)
        if 2000 <= cand <= 25000:
            cand_lengths.append(cand)
    cand_lengths = sorted(set(cand_lengths))

    cur_seq = best_seq
    cur_score = best_score
    cur_n = best_n

    for L in cand_lengths:
        if time.time() > end_time - 350.0:
            break
        a_scaled = _scale_sequence(cur_seq, L)
        a_scaled = np.clip(a_scaled, 0.0, None)
        S = np.sqrt(2.0 * L)
        a_scaled = a_scaled * (S / a_scaled.sum())
        peak = _max_convolution(a_scaled)
        if peak < cur_score:
            cur_score = peak
            cur_seq = a_scaled.copy()
            cur_n = L
            print(f"[scale] length={L:5d}  improved max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam optimisation (several restarts)
    # ------------------------------------------------------------------
    remaining = end_time - time.time() - 250.0   # reserve time for later phases
    if remaining > 30.0:
        per_run = 30.0
        n_restart = max(1, int(remaining // per_run))
        for r in range(n_restart):
            if time.time() > end_time - 250.0:
                break
            noise = rng.normal(scale=0.01 * np.sqrt(2.0 * cur_n) / cur_n, size=cur_n)
            a_start = np.maximum(cur_seq + noise, 0.0)
            a_start = a_start * (np.sqrt(2.0 * cur_n) / a_start.sum())
            a_opt, peak = _adam_optimize(a_start,
                                         time.time() + per_run,
                                         rng)
            if peak < cur_score:
                cur_score = peak
                cur_seq = a_opt.copy()
                print(f"[adam] restart={r} new best max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – sub‑gradient descent (polish)
    # ------------------------------------------------------------------
    if time.time() < end_time - 120.0:
        sub_time = min(120.0, end_time - time.time() - 80.0)
        a_opt, peak = _subgradient_descent(cur_seq,
                                           time.time() + sub_time,
                                           rng)
        if peak < cur_score:
            cur_score = peak
            cur_seq = a_opt.copy()
            print(f"[subgrad] new best max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – linear‑programming refinement (core new method)
    # ------------------------------------------------------------------
    if time.time() < end_time - 80.0:
        lp_time = end_time - time.time() - 20.0
        a_opt, peak = _lp_refine(cur_seq, time.time() + lp_time, rng)
        if peak < cur_score:
            cur_score = peak
            cur_seq = a_opt.copy()
            print(f"[lp] refined max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – cheap stochastic hill‑climb (final pass)
    # ------------------------------------------------------------------
    while time.time() < end_time - 1.0:
        a = cur_seq.copy()
        n = a.shape[0]
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        frac = rng.uniform(0.02, 0.12)
        amount = a[i] * frac
        a[i] -= amount
        a[j] += amount
        peak = _max_convolution(a)
        if peak + 1e-12 < cur_score:   # strict improvement
            cur_score = peak
            cur_seq = a.copy()
            print(f"[hill] improved max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    final_seq = cur_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f}  internal max={cur_score:.6f}  length={len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
