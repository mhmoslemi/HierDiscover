"""
Propose a step‑function minimizing   2·n·max(a*a) / (∑a)² .

Algorithm
---------
1. **Problem size** – choose n≈12000 (large enough degrees of freedom, still
   fast to FFT).

2. **Parametric pre‑search** – explore a three‑parameter family
        a_i = (i+1)^(-p) · exp( -q·i/n – r·(i/n)² )
   on a coarse grid (p,q,r).  Each candidate is projected onto the simplex
   Σa = sqrt(2·n).  The best candidate (few‑hundred evaluations) seeds the
   subsequent local search.

3. **Projected Adam gradient descent** – minimise a smooth soft‑max
   approximation of max(a*a).  The gradient is obtained via FFT, an Adam‑style
   adaptive step size is used, and after each update the vector is projected
   back onto the simplex.  The soft‑max temperature β is doubled every few
   hundred iterations up to ≈2·10⁵.

4. **Deterministic mass‑transfer refinement** – repeatedly move a small
   amount of mass from the index contributing most to the current maximal
   convolution entry to an index contributing least, accepting moves that
   lower the score (or with a Metropolis probability that cools).

5. **Global random mass transfer** – occasional simulated‑annealing moves
   that shift a little mass from the top fraction of entries to the bottom
   fraction, keeping improvements.

6. **Final polishing** – another Adam run with a very high β (≈10⁶) to
   sharpen the max‑approximation.

All phases respect the overall 1000 s budget.  The final sequence is trimmed
of trailing zeros before returning.

The code uses only NumPy and the supplied ``evaluate_sequence`` for optional
final verification.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Benchmark objective 2·n·max(a*a) / (∑a)²."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    rho = np.nonzero(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _quad_exp_power_law_sequence(n: int, p: float, q: float, r: float,
                                 target_sum: float) -> np.ndarray:
    """(i+1)^(-p) * exp(-q*i/n - r*(i/n)^2)  scaled to target_sum."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n - r * (i / n) ** 2)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Gradient of soft‑max approximation of max‑convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (grad, conv) where grad is the gradient of the soft‑max of
    a*a w.r.t. a, and conv is the full convolution.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    # soft‑max weights (stable)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * (a reversed) ⋆ p   (only the central n entries are needed)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on soft‑max objective
# ----------------------------------------------------------------------


def _adam_descent(a0: np.ndarray,
                  deadline: float,
                  target_sum: float,
                  max_beta: float = 2e5,
                  step0: float = 0.5,
                  beta_inc_iter: int = 250) -> tuple[np.ndarray, float]:
    """
    Adam optimizer with projection onto the simplex.
    - a0: starting point (will be projected)
    - deadline: absolute time to stop
    - target_sum: required Σa after projection
    - max_beta: final temperature for soft‑max
    - step0: base learning rate (relative to target_sum)
    - beta_inc_iter: how many iterations between β doublings
    Returns (best_vector, best_score).
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    t = 0
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step_base = step0 * target_sum

    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Adam update (with projection)
        a -= step_base * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if t % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        # evaluate
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={t:6d} β={beta:6.1f} best={best_val:.6f}")

        # increase temperature
        if t % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-6,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start = time.time()

    # pre‑compute convolution once; later updates recompute fully (still cheap)
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        # exponential temperature schedule
        tau = (time.time() - start) / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution for each idx to the maximal entry
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        # bound transfer
        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0.0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0.0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Global random mass‑transfer (exploratory)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        donor = np.argsort(-a)[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        receiver = np.argsort(a)[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random new best={best_score:.6f}")

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine respects the time budget, prints progress and returns the
    best sequence found (as Python list of floats).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ---------------------------------------------------------------
    # Choose problem size
    # ---------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)   # Σa is fixed (scaling invariant)

    # ---------------------------------------------------------------
    # Phase 0 – fast parametric pre‑search (incl. quadratic tail)
    # ---------------------------------------------------------------
    rng = np.random.default_rng()
    candidates = []

    # (a) pure power‑law
    for p in np.linspace(0.35, 0.70, 12):
        candidates.append(_power_law_sequence(n, p, target_sum))

    # (b) power‑law + exponential tail
    for p in np.linspace(0.35, 0.70, 6):
        for q in np.linspace(0.0, 5.0, 6):
            candidates.append(_exp_power_law_sequence(n, p, q, target_sum))

    # (c) power‑law + quadratic‑exponential tail (new family)
    for p in np.linspace(0.30, 0.80, 5):
        for q in np.linspace(0.0, 5.0, 5):
            for r in np.linspace(0.0, 6.0, 5):
                candidates.append(_quad_exp_power_law_sequence(n, p, q, r, target_sum))

    # (d) previous best (if any)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        candidates.append(_resize_sequence(prev, n))

    # (e) jittered variants
    for seq in list(candidates):
        jitter = rng.normal(scale=target_sum * 0.003, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        candidates.append(seq_j)

    # limit the number of preliminary evaluations to keep time bounded
    max_initial_evals = min(250, len(candidates))
    sample_idx = rng.choice(len(candidates), size=max_initial_evals, replace=False)

    best_seq = None
    best_score = np.inf

    for i in sample_idx:
        seq = candidates[i]
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    print(f"[{time.time():.2f}] Initial best after parametric search = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase A – Adam projected gradient descent with restarts
    # ---------------------------------------------------------------
    # allocate roughly 35% of remaining budget to this phase
    phase_a_deadline = time.time() + (deadline - time.time()) * 0.35

    while time.time() < phase_a_deadline - 5.0:
        # jittered start from current best
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _adam_descent(start_vec,
                                       phase_a_deadline,
                                       target_sum,
                                       max_beta=2e5,
                                       step0=0.6,
                                       beta_inc_iter=200)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Adam improved best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase B – Deterministic + Global random refinements
    # ---------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # deterministic mass‑transfer
        a_ref, val_ref = _deterministic_mass_transfer(best_seq,
                                                      deadline,
                                                      target_sum,
                                                      start_temp=0.5,
                                                      end_temp=1e-6,
                                                      low_frac=0.06)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] DMT improved best = {best_score:.6f}")

        # global random moves
        a_ref2, val_ref2 = _global_random_mass_transfer(best_seq,
                                                        deadline,
                                                        target_sum)
        if val_ref2 < best_score:
            best_score = val_ref2
            best_seq = a_ref2.copy()
            print(f"[{time.time():.2f}] Global random improved best = {best_score:.6f}")

        if time.time() > deadline - 2.0:
            break

    # ---------------------------------------------------------------
    # Phase C – Final polishing with higher temperature
    # ---------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_descent(best_seq,
                                       deadline,
                                       target_sum,
                                       max_beta=1e6,
                                       step0=0.5,
                                       beta_inc_iter=150)
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Post‑processing – trim trailing zeros
    # ---------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final verification (harmless if evaluate_sequence is absent)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
