def stochastic_hill_climb(a, target_sum, deadline):
    best = a.copy()
    best_maxb = get_maxb(a)
    cur = a.copy()
    cur_maxb = best_maxb
    n = len(a)
    rng = np.random.default_rng()
    T0 = 0.01 # maybe set as relative change.
    while time.time() < deadline:
        # compute time fraction for annealing schedule
        progress = (deadline - time.time()) / (deadline - start)
        # we can compute temperature decreasing linearly
        T = T0 * progress
        # random move: choose donor idx where cur[idx] > epsilon
        donor = rng.choice(np.where(cur > 1e-12)[0])
        # choose receiver different
        receiver = rng.integers(n)
        while receiver == donor:
            receiver = rng.integers(n)
        # choose delta as uniform(0, cur[donor] * max_frac)
        max_frac = min(0.05, T/ T0 * 0.05 + 0.01)  # gradually decreasing max move
        delta = cur[donor] * rng.random() * max_frac
        if delta < 1e-12:
            continue
        new_cur = cur.copy()
        new_cur[donor] -= delta
        new_cur[receiver] += delta
        # re-normalization not needed because sum unchanged.
        # compute maxb via convolution
        new_maxb = _max_convolution_fft(new_cur)

                best_maxb = new_maxb
                best = cur.copy()
                print(f"[{time.time():.2f}] HillClimb improved best={best_maxb:.6f}")

def _random_hill_climb(a, target_sum, deadline):
    cur = a.copy()
    cur_maxb = _max_convolution_fft(cur)
    best = cur.copy()
    best_maxb = cur_maxb
    rng = np.random.default_rng()
    n = cur.size

    # parameters
    max_frac_initial = 0.05
    max_frac_final = 0.005
    total_time = deadline - time.time()
    # We'll reduce max_frac linearly across time.

    it = 0
    while time.time() < deadline:
        progress = (deadline - time.time()) / total_time
        max_frac = max_frac_final + (max_frac_initial - max_frac_final) * progress
        donor = rng.integers(n)
        while cur[donor] < 1e-12:
            donor = rng.integers(n)
        receiver = rng.integers(n)
        while receiver == donor:
            receiver = rng.integers(n)
        delta = cur[donor] * rng.random() * max_frac
        if delta <= 0:
            continue
        cur[donor] -= delta
        cur[receiver] += delta
        new_maxb = _max_convolution_fft(cur)
        # Accept if improvement
        if new_maxb < cur_maxb:
            cur_maxb = new_maxb
            if new_maxb < best_maxb:
                best_maxb = new_maxb
                best = cur.copy()
                print(f"[{time.time():.2f}] Hill climb it={it} best={best_maxb:.6f}")
        else:
            # revert
            cur[donor] += delta
            cur[receiver] -= delta

        it += 1

    return best, best_maxb

"""
Propose a step‑function (non‑negative sequence) with a low benchmark score.

The algorithm
--------------
1. Choose a problem size n depending on the time budget (larger n gives a
   finer search space).  The ℓ¹‑norm Σaᵢ is fixed to √(2n) so that the
   objective coincides with the maximal convolution entry.

2. Warm‑start from a previously found solution (global ``height_sequence_1``)
   or from a two‑parameter power‑law family.  A cheap grid / random search
   on a reduced scale supplies a good initial shape.

3. Refine the shape with several optimisation stages, each limited by a time
   slice:
   * Exponential (multiplicative) mirror‑descent on a soft‑max surrogate of
     the max‑convolution.
   * Projected gradient descent with momentum (still on the soft‑max).
   * Deterministic mass‑transfer that moves a small fraction of the mass from
     the most harmful index to a low‑contributing one.
   * Greedy random hill‑climbing (simulated‑annealing style) that directly
     evaluates the exact max‑convolution after each tiny mass move.
   * Projected sub‑gradient descent on the true max‑convolution entry.

4. The best sequence seen during the whole run is returned (trailing zeros
   are stripped).  The resulting list can be reused as ``height_sequence_1``
   in subsequent calls.

All stages respect the overall time budget (≈1000 s) and keep the vector on
the ℓ¹‑simplex.  The code uses only NumPy and the supplied ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution / scoring utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())        # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  For Σa = √(2n) this equals max(a*a)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex (Σx = s, x ≥ 0)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helpers for resizing / generating families
# ----------------------------------------------------------------------


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Shape a_i = (i+1)^(-p) scaled to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max surrogate of the max‑convolution and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where conv is the full convolution a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # grad = 2 * (reverse(a) *convolve* p)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Stage 2 – exponential (multiplicative) descent on the soft‑max
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Mirror‑descent with exponential updates using a soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 16384.0
    base_eta = 0.04
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)                     # multiplicative step
        a = a * (target_sum / a.sum())               # renormalise

        # occasional jitter to avoid stagnation
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 3 – projected gradient descent with momentum (soft‑max)
# ----------------------------------------------------------------------


def _projected_gradient_descent(a0: np.ndarray,
                               target_sum: float,
                               deadline: float) -> tuple[np.ndarray, float]:
    """Projected gradient descent (additive) on the soft‑max surrogate."""
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 16384.0
    step_base = target_sum * 0.5
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.95
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.002 / (beta + 1.0), size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] PGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several short PGD restarts (random jitter) and keep the best."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _projected_gradient_descent(start,
                                             target_sum,
                                             time.time() + run_budget)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 4 – deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _mass_transfer_refine(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Move a modest fraction of mass from a high‑contributing index to a low‑one."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    while time.time() < deadline:
        k_max = int(np.argmax(conv))

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = a[k_max - idx]                     # a_{k_max-i}
        weight = a[idx] * partner                     # contribution a_i·a_{k_max-i}

        donor = idx[np.argmax(weight)]

        low_thr = np.percentile(weight, 10)
        low_idx = idx[weight <= low_thr]
        if low_idx.size:
            receiver = low_idx[np.argmin(weight[weight <= low_thr])]
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            donor = rng.choice(idx)
            receiver = rng.integers(n)
            if donor == receiver:
                continue

        max_frac = 0.04
        delta = a[donor] * rng.random() * max_frac
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta

        conv = _conv_fft(a)
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Transfer improve score={best_val:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            conv = _conv_fft(a)

        # re‑normalise occasional drift
        cur_sum = a.sum()
        if abs(cur_sum - target_sum) > 1e-9:
            a = a * (target_sum / cur_sum)

    return best_a, best_val

# Stage 5 – greedy random hill‑climbing on the exact max‑convolution
# ----------------------------------------------------------------------


def _hill_climb_max(a0: np.ndarray,
                    target_sum: float,
                    deadline: float) -> tuple[np.ndarray, float]:
    """Greedy hill‑climbing that evaluates the exact max‑convolution after each tiny move."""
    cur = a0.copy()
    cur_maxb = _max_convolution_fft(cur)
    best = cur.copy()
    best_maxb = cur_maxb
    n = cur.size
    rng = np.random.default_rng()

    # schedule for the maximal fraction of donor mass we are allowed to move
    max_frac_start = 0.05
    max_frac_end = 0.003
    total_time = deadline - time.time()
    it = 0

    while time.time() < deadline:
        # linear schedule for max_frac
        remaining = deadline - time.time()
        progress = remaining / total_time if total_time > 0 else 0.0
        max_frac = max_frac_end + (max_frac_start - max_frac_end) * progress

        donor = rng.integers(n)
        # ensure donor has some mass
        while cur[donor] < 1e-12:
            donor = rng.integers(n)

        receiver = rng.integers(n)
        while receiver == donor:
            receiver = rng.integers(n)

        delta = cur[donor] * rng.random() * max_frac
        if delta <= 1e-16:
            continue

        # perform the tentative move
        cur[donor] -= delta
        cur[receiver] += delta

        new_maxb = _max_convolution_fft(cur)

        # accept only if we improve (greedy)
        if new_maxb < cur_maxb:
            cur_maxb = new_maxb
            if new_maxb < best_maxb:
                best_maxb = new_maxb
                best = cur.copy()
                print(f"[{time.time():.2f}] Hill‑climb it={it} best={best_maxb:.6f}")
        else:
            # revert
            cur[donor] += delta
            cur[receiver] -= delta

        it += 1

    return best, best_maxb

# ----------------------------------------------------------------------


def _subgradient_max_descent(a0: np.ndarray,
                             target_sum: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true maximal convolution entry."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())

        grad = np.zeros_like(a)
        lo = max(0, k_max - (a.shape[0] - 1))
        hi = min(k_max, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        grad[idx] = 2.0 * a[partner]

        norm = np.linalg.norm(grad)
        if norm == 0:
            break

        # safe step respecting non‑negativity before projection
        safe_eta = np.min(np.where(grad > 0, a / grad, np.inf))
        eta = min(0.5 * safe_eta, 0.5 / np.sqrt(it + 1))

        a = a - eta * grad
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it} score={best_val:.6f}")

        # occasional jitter to escape flat regions
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        it += 1

    return best_a, best_val

# Cheap parametric search on a reduced scale
# ----------------------------------------------------------------------


def _parametric_search(small_n: int,
                       target_sum: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """Explore a (p, λ) power‑law/exponential family on a small scale."""
    best_seq = None
    best_score = np.inf
    i = np.arange(small_n, dtype=np.float64)

    # coarse grid
    p_vals = np.linspace(0.2, 1.2, 13)
    lam_vals = np.linspace(0.0, 6.0, 13)

    for p in p_vals:
        for lam in lam_vals:
            if time.time() > deadline:
                break
            seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
            seq = seq * (target_sum / seq.sum())
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()
                print(f"[{time.time():.2f}] Grid best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")

    # random refinement around the best point (if we found one)
    rng = np.random.default_rng()
    while time.time() < deadline:
        p = rng.uniform(0.2, 1.2)
        lam = rng.exponential(1.0) * 3.0
        seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
        seq = seq * (target_sum / seq.sum())
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Random best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")

    return best_seq, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5               # safety margin

    # ------------------------------------------------------------------
    # Problem size (larger n gives finer granularity but costs more time)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 600:
        n = 15000
    elif budget_s >= 400:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Warm‑start from previous run or from a simple power‑law family
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"],
                                        dtype=np.float64), n)
        init = init * (target_sum / init.sum())
    else:
        init = _power_law_sequence(n, 0.55, target_sum)

    # ------------------------------------------------------------------
    # Stage 1 – cheap parametric search on a reduced scale
    # ------------------------------------------------------------------
    small_n = max(2000, n // 5)
    small_target = np.sqrt(2.0 * small_n)
    param_deadline = time.time() + min(25.0,
                                       (deadline - time.time()) * 0.08)
    best_small_seq, _ = _parametric_search(small_n, small_target,
                                           param_deadline)

    if best_small_seq is not None:
        init = _resize_sequence(best_small_seq, n)
        init = init * (target_sum / init.sum())
        print(f"[{time.time():.2f}] Parametric warm‑start ready (n={n})")

    # ------------------------------------------------------------------
    # Stage 2 – exponential (multiplicative) descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        exp_deadline = time.time() + (deadline - time.time()) * 0.14
        a_exp, val_exp = _exponential_descent(init, target_sum, exp_deadline)
        best_seq = a_exp
        best_score = val_exp
        print(f"[{time.time():.2f}] After ExpGD best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – projected gradient descent with restarts
    # ------------------------------------------------------------------
    if time.time() < deadline:
        gd_deadline = time.time() + (deadline - time.time()) * 0.33
        a_gd, val_gd = _gradient_descent_with_restarts(best_seq,
                                                       target_sum,
                                                       gd_deadline)
        if val_gd < best_score:
            best_seq = a_gd
            best_score = val_gd
        print(f"[{time.time():.2f}] After GD best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – deterministic mass‑transfer refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        mt_deadline = time.time() + (deadline - time.time()) * 0.20
        a_mt, val_mt = _mass_transfer_refine(best_seq, target_sum, mt_deadline)
        if val_mt < best_score:
            best_seq = a_mt
            best_score = val_mt
        print(f"[{time.time():.2f}] After mass‑transfer best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – greedy random hill‑climbing on the exact max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline:
        hc_deadline = time.time() + (deadline - time.time()) * 0.15
        a_hc, val_hc = _hill_climb_max(best_seq, target_sum, hc_deadline)
        if val_hc < best_score:
            best_seq = a_hc
            best_score = val_hc
        print(f"[{time.time():.2f}] After hill‑climb best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 6 – sub‑gradient descent on exact max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_sub, val_sub = _subgradient_max_descent(best_seq,
                                                  target_sum,
                                                  deadline)
        if val_sub < best_score:
            best_seq = a_sub
            best_score = val_sub
        print(f"[{time.time():.2f}] After sub‑gradient best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final clean‑up and evaluation
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for warm‑starts in a subsequent call
    globals()["height_sequence_1"] = final_seq

    return final_seq
