"""
Propose a non‑negative step function that minimises ::

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)**2)

The search works in three stages:

1. **Projected Adam on a soft‑max surrogate** – many short restarts
   (the smooth max is gradually sharpened).  This quickly finds good
   “global” shapes.

2. **Heavy deterministic hill‑climbing** – the current worst convolution
   entry is identified; a large fraction of the responsible mass is
   moved towards indices that do not contribute to that entry.  The
   update of the full convolution is O(n) per move.

3. **Final polishing** – a long Adam run followed by a short hill‑climb.

The algorithm runs under the supplied wall‑clock budget (default 1000 s)
and prints intermediate scores.  It re‑uses any previously found
`height_sequence_1` as a seed and progressively upsamples the best
candidate to larger lengths.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using an FFT (fast, O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Soft‑max surrogate of the max (and its gradient)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float,
                    deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max surrogate.
    Runs until the supplied wall‑clock deadline.
    """
    a = a0.copy()
    n = a.size

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # base learning rate – scaled with the simplex radius
    base_lr = 0.13 * target_sum

    # smoothing schedule
    beta = 2.0          # start of the log‑sum‑exp temperature
    beta_max = 1e8
    beta_factor = 2.0

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1

        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat areas
        if t % 300 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4 / (beta + 1.0),
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # increase smoothing (make the surrogate sharper)
        if t % 400 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)

    # Choose a random “family” of shapes
    fam = rng.integers(0, 7)
    idx = np.arange(n, dtype=float)

    if fam == 0:                                   # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif fam == 1:                                 # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif fam == 2:                                 # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif fam == 3:                                 # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif fam == 4:                                 # piecewise exponential
        cut = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(cut)),
                            np.exp(-lam2 * np.arange(cut, n))])

    elif fam == 5:                                 # bimodal spikes at the ends
        k = max(2, n // 20)
        left = np.exp(-rng.exponential(scale=0.02) * np.arange(k))
        right = np.exp(-rng.exponential(scale=0.02) * np.arange(k)[::-1])
        middle = np.zeros(n - 2 * k)
        a = np.concatenate([left, middle, right])

    else:                                          # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

# Deterministic hill‑climbing (exact max) – improved version
# ----------------------------------------------------------------------


def _hill_climb_best(a0: np.ndarray, b0: np.ndarray,
                     deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that attacks the current worst convolution entry.
    Moves a large fraction of the responsible mass towards indices that
    do *not* contribute to that entry.  The full convolution is updated
    in O(n) per move.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()

    rng = np.random.default_rng()

    while time.time() < deadline:
        # worst (largest) entry of the convolution
        M = int(np.argmax(b))

        # source indices that influence entry M
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]

        # pick the source with the largest contribution
        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        # counterpart for each possible destination (a[M-j] when defined,
        # otherwise 0 because the pair does not appear in entry M)
        counterpart = np.zeros(n, dtype=float)
        j_low = max(0, M - (n - 1))
        j_high = min(n - 1, M)
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        # Prefer a destination *outside* the influencing window where
        # the counterpart is zero – this never increases the current max.
        zero_candidates = np.where(counterpart == 0)[0]
        if zero_candidates.size > 0:
            # pick the zero‑candidate farthest from i (to decorrelate)
            j = zero_candidates[np.argmax(np.abs(zero_candidates - i))]
        else:
            # otherwise pick the index with smallest counterpart (avoid i)
            candidates = np.where(np.arange(n) != i)[0]
            j = candidates[np.argmin(counterpart[candidates])]

        # move a large fraction of the source mass
        delta = a[i] * 0.7  # 70 % of the source
        if delta <= 0.0:
            break

        # --- evaluate the effect on the full convolution (O(n)) ----------
        b_new = b.copy()
        # contribution of the moved mass to all affected entries
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        # --- accept if we improve ----------------------------------------
        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
            continue

        # If the full move does not improve, try a smaller delta (binary search)
        improved = False
        delta2 = delta * 0.5
        while delta2 > 1e-14:
            b_test = b.copy()
            b_test[i:i + n] -= 2.0 * delta2 * a
            b_test[j:j + n] += 2.0 * delta2 * a
            if k2i < b_test.size:
                b_test[k2i] += delta2 * delta2
            if k2j < b_test.size:
                b_test[k2j] += delta2 * delta2
            if kij < b_test.size:
                b_test[kij] -= 2.0 * delta2 * delta2
            if b_test.max() < b.max() - 1e-12:
                a[i] -= delta2
                a[j] += delta2
                b = b_test
                if b.max() < best_score - 1e-12:
                    best_score = b.max()
                    best_a = a.copy()
                improved = True
                break
            delta2 *= 0.5

        if not improved:
            # small random jitter – helps escape plateaus
            jitter = rng.normal(scale=1e-6, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            target = math.sqrt(2.0 * n)
            a = _project_to_simplex(a, target)
            b = _full_convolution(a)

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:

    best_score = np.inf

    # ------------------------------------------------------------------

        try:

            n0 = prev.size

            prev = _project_to_simplex(prev, target0)
            best_seq = prev.copy()

    # ------------------------------------------------------------------

        # --------------------------------------------------------------

        # --------------------------------------------------------------

        # --------------------------------------------------------------

        # --------------------------------------------------------------

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # 4) Report the official evaluation
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
