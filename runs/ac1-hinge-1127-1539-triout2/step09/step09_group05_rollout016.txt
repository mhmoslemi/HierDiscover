def _smooth_max_and_grad(a, beta):
    n = len(a)
    conv = _full_convolution_fft(a)  # length 2n-1
    # compute softmax
    maxc = conv.max()
    # subtract max for stability
    expw = np.exp(beta * (conv - maxc))
    sumw = expw.sum()
    w = expw / sumw
    # gradient: 2 * conv(w, a_rev) central slice
    a_rev = a[::-1]
    # convolution via FFT (use same length)
    total_len = len(w) + n - 1
    f_len = _next_pow2(total_len)
    W = np.fft.rfft(w, n=f_len)

    conv_grad_full = np.fft.irfft(W * A_rev, n=f_len)[:total_len]
    grad = 2.0 * conv_grad_full[n-1:n-1+n]
    smooth_val = (maxc + np.log(sumw) / beta)  # as F = (1/β) log sum exp(...)
    return grad, smooth_val, conv.max()

   finally focuses on the true maximum.

3. **Final simulated‑annealing fine‑tuning** – a cheap mass‑transfer
   hill‑climb that accepts occasional worsening moves.  This helps to
   escape tiny plateaus left by the gradient method.

All stages respect a total wall‑clock budget of ≈ 1000 s.  The best
sequence found is stored in the global variable ``height_sequence_1`` and
returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT, projection, interpolation)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a∗a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a∗a."""
    return _full_convolution_fft(a).max()


def _project_onto_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {w | w_i ≥ 0 , Σw_i = S }.
    Implements the O(n) algorithm of Duchi et al.
    """
    if S < 0:
        raise ValueError("Simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Internal (fast) objective – identical to evaluate_sequence
# ----------------------------------------------------------------------
def _objective_val(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a∗a)/(Σa)² (no clipping)."""
    n = a.shape[0]
    maxb = _max_convolution(a)
    # Σa after projection equals √(2·n) → we can use that directly,
    # but we keep the explicit formula for safety.
    suma = a.sum()
    if suma < 1e-12:
        return np.inf
    return float(2.0 * n * maxb / (suma * suma))


# ----------------------------------------------------------------------
# Smoothed‑max helper (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def _smooth_max_and_gradient(a: np.ndarray, beta: float):
    """
    Returns (grad, smooth_val, hard_max) where

        smooth_val = (1/β)·log Σ exp(β·(a∗a)[k])
        grad = ∇ smooth_val  (shape of a)

    The gradient is computed via FFT‑based convolutions.
    """
    n = a.shape[0]
    conv = _full_convolution_fft(a)           # length 2n‑1
    maxc = conv.max()
    # softmax weights, stable version
    expw = np.exp(beta * (conv - maxc))
    sumw = expw.sum()
    w = expw / sumw                           # Σ w_k = 1
    # gradient = 2 * (w ∗ reverse(a)) on the central window
    a_rev = a[::-1]
    total_len = len(w) + n - 1
    f_len = _next_pow2(total_len)
    W = np.fft.rfft(w, n=f_len)
    A_rev = np.fft.rfft(a_rev, n=f_len)
    conv_full = np.fft.irfft(W * A_rev, n=f_len)[:total_len]
    grad = 2.0 * conv_full[n - 1:n - 1 + n]
    smooth_val = (maxc + np.log(sumw) / beta)  # = (1/β)·log Σ exp(β·conv)
    return grad, smooth_val, maxc


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective with simplex projection
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   beta_start: float = 1.0,
                   beta_max: float = 30.0) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smoothed max‑convolution objective."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)       # simplex radius (exact sum after projection)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta = beta_start
    beta_inc = (beta_max / beta_start) ** (1.0 / 2000)   # gradual increase
    lr = 0.8                   # base learning rate
    eps = 1e-8
    t = 0
    best_a = a.copy()
    best_val = _objective_val(a)

    while time.time() < deadline:
        t += 1
        if t % 200 == 0 and beta < beta_max:
            beta = min(beta * beta_inc, beta_max)

        grad, _, _ = _smooth_max_and_gradient(a, beta)

        # Adam moments
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad ** 2)

        m_hat = m / (1 - 0.9 ** t)
        v_hat = v / (1 - 0.999 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_onto_simplex(a - step, S)

        # hard objective (the one evaluate_sequence uses)
        cur_val = _objective_val(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to shake out plateaus
        if t % 500 == 0:
            jitter = rng.normal(scale=0.001 * S, size=n)
            a = _project_onto_simplex(a + jitter, S)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb (mass transfer)
# ----------------------------------------------------------------------
def _anneal_local_search(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    cur_val = _objective_val(a)
    best_a = a.copy()
    best_val = cur_val

    T = 0.05 * cur_val   # initial temperature
    decay = 0.9999

    it = 0
    while time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        frac = rng.uniform(0.02, 0.15)
        delta = a[i] * frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_val = _objective_val(a_new)
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
        T *= decay
        if it % 400 == 0:
            jitter = rng.normal(scale=0.001 * S, size=n)
            a = _project_onto_simplex(a + jitter, S)
            cur_val = _objective_val(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Parametric families for random initialisation
# ----------------------------------------------------------------------
def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.1, 4.0)
    off = rng.uniform(0.0, 15.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + off) ** (-p)
    base = np.ones(n)
    return mix * edge + (1 - mix) * base


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0002, 0.03)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    return mix * exp_part + (1 - mix) * base


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    peak = rng.uniform(0.5, 5.0)
    slope = rng.uniform(0.0, peak / (n / 2 + 1e-9))
    lin = np.maximum(0.0, peak - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.3))
    return mix * lin + (1 - mix) * base


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.3, 2.8)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    return mix * ush + (1 - mix) * base


def _init_sine(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    vals = np.sin(np.pi * (i + 1) / (n + 1))
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * vals + (1 - mix) * base


def _init_logistic(n: int, rng: np.random.Generator) -> np.ndarray:
    """A sigmoid‑shaped curve – cheap and often useful."""
    i = np.arange(n)
    mid = (n - 1) / 2.0
    steep = rng.uniform(0.01, 2.5)
    shift = rng.uniform(-3.0, 3.0)
    vals = 1.0 / (1.0 + np.exp(steep * (i - mid) + shift))
    # make it symmetric so that both ends are high
    vals = np.maximum(vals, vals[::-1])
    return vals


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    return rng.random(n)


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    With probability ≈0.2 we warm‑start from the globally best sequence,
    otherwise we draw from a random parametric family.
    """
    S = np.sqrt(2.0 * n)

    # warm start
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total > 0.0:
            a = a * (S / total)
            return a

    # pick a family
    family = rng.choice(['powerlaw', 'exponential', 'linear',
                         'ushape', 'sine', 'logistic',
                         'uniform', 'dirichlet'])
    if family == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif family == 'exponential':
        raw = _init_exponential(n, rng)
    elif family == 'linear':
        raw = _init_linear(n, rng)
    elif family == 'ushape':
        raw = _init_ushape(n, rng)
    elif family == 'sine':
        raw = _init_sine(n, rng)
    elif family == 'logistic':
        raw = _init_logistic(n, rng)
    elif family == 'uniform':
        raw = _init_uniform(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises the provided
    ``evaluate_sequence``.  The routine follows the three‑stage plan
    described in the module docstring and stores the final construction
    in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    # safety margin for housekeeping
    end_time = time.time() + budget_s - 5.0

    # ------------------------------------------------------------------
    # Retrieve warm‑start if available
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    best_seq = None
    best_score = np.inf
    attempts = 0

    # ------------------------------------------------------------------
    # Phase 1 – broad random exploration (≈ 35 % of the budget)
    # ------------------------------------------------------------------
    explore_deadline = end_time - 400.0           # reserve time for refinement
    while time.time() < explore_deadline:
        attempts += 1
        # favour longer sequences, but keep a slice of medium size
        if rng.random() < 0.7:
            n = int(rng.integers(5000, 25000))
        else:
            n = int(rng.integers(25000, 40000))
        a0 = _initial_vector(n, rng, warm_seq)
        score = _objective_val(a0)
        if score < best_score:
            best_score = score
            best_seq = a0.copy()
            print(f"[explore] attempt={attempts:4d} n={n:5d}  new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – smoothed‑max Adam descent on a few promising lengths
    # ------------------------------------------------------------------
    if best_seq is None:                     # extremely unlikely
        n = 8000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = _objective_val(best_seq)

    # choose a ladder of lengths (starting from the current best)
    target_lengths = [len(best_seq)]
    for factor in [1.3, 1.7, 2.2, 3.0]:
        cand = int(len(best_seq) * factor)
        if 5000 <= cand <= 40000:
            target_lengths.append(cand)
    target_lengths = sorted(set(target_lengths))

    for target_n in target_lengths:
        if time.time() >= end_time - 250.0:
            break
        # interpolate (or downsample) to the new length
        a_scaled = _scale_sequence(best_seq, target_n)
        # normalise to the simplex exactly
        S = np.sqrt(2.0 * target_n)
        a_scaled = np.clip(a_scaled, 0.0, None)
        a_scaled = a_scaled * (S / a_scaled.sum())

        # allocate a time slice for this length (≈ 15 % remaining)
        remaining = end_time - time.time()
        slice_time = min(remaining * 0.15, 120.0)

        a_opt, score = _adam_optimize(a_scaled,
                                      time.time() + slice_time,
                                      rng,
                                      beta_start=1.0,
                                      beta_max=30.0)
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[adam] target_n={target_n:5d}  improved={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if time.time() < end_time - 5.0:
        remaining = end_time - time.time() - 1.0
        a_opt, score = _anneal_local_search(best_seq,
                                            time.time() + remaining,
                                            rng)
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[anneal] final improve to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – sanity check with the external evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f}  internal={best_score:.6f}  length={len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
