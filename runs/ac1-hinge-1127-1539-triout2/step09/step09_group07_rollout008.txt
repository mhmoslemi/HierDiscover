"""
Improved step‑function search for minimise ``evaluate_sequence``.

The algorithm consists of three stages:
1️⃣ Fast random exploration of many families (including exponential,
   power‑law, sparse bumps, Gaussian and half‑sine) – we keep the best
   few candidates.
2️⃣ For each shortlisted candidate we run a longer Adam optimisation on
   the simplex (log‑softmax parametrisation).  The Adam schedule is
   tuned for a slower temperature decay and a larger learning‑rate so
   that the sub‑gradient of the hard max is used more often.
3️⃣ The best construction is polished by a deterministic projected‑gradient
   descent (PGD) that directly follows the exact sub‑gradient of the
   max‑convolution, followed by a cheap simulated‑annealing walk and a
   strict greedy hill‑climber.

The whole pipeline respects the 1000 s budget (a 5 s safety margin)
and stores the final sequence in the global ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Peak value of a*a."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Exact (hard‑max) sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------
def _hard_max_gradient(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Returns (grad, max_b) where grad = ∂ max(b) / ∂a,
    b = a*a (full convolution).  The gradient is obtained by averaging
    the true active set (indices whose value is within a tiny tolerance
    of the maximum) – this is an exact sub‑gradient.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()
    eps = max_b * 1e-9 + 1e-12
    mask = (b >= max_b - eps).astype(float)
    mask_sum = mask.sum()
    if mask_sum == 0:
        mask = np.ones_like(b) / b.size
    else:
        mask /= mask_sum
    # gradient = 2 * (mask * a_rev) convolved with a_rev
    a_rev = a[::-1]
    conv_len = mask.size + a.size - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    grad = 2.0 * conv[a.size - 1:a.size - 1 + a.size]
    return grad, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the probability simplex (log‑softmax)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   lr0: float = 0.6,
                   tau_decay: float = 0.9998,
                   lr_decay: float = 0.9999,
                   jitter_every: int = 300) -> tuple[np.ndarray, float]:
    """
    Minimise the peak of the self‑convolution under a fixed ℓ₁‑norm.
    Returns the best vector seen and its hard peak.
    """
    n = a0.shape[0]
    S = math.sqrt(2.0 * n)          # required ℓ₁‑norm (scale does not matter)

    # initialise logits from the feasible a0 (log‑softmax)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr0

    # initialise temperature with a reasonable max‑convolution value
    tau = max(_full_convolution_fft(a0).max(), 0.5)

    best_a = a0.copy()
    best_peak = _max_convolution(a0)
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- recover a from logits (log‑softmax) -----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)          # stabilise
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ----- gradient of (soft‑)max convolution -----
        # use hard‑max for the last part of the run (tau will be tiny)
        g, peak = _hard_max_gradient(a) if tau < 1e-12 else _softmax_gradient(a, tau)

        # keep best hard peak seen
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ----- gradient w.r.t. logits (chain rule) -----
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # clip extreme values for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- schedules -----
        tau *= tau_decay
        lr *= lr_decay

        # occasional jitter to escape plateaus
        if jitter_every and (it % jitter_every) == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak


def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Soft‑max approximation of the max‑gradient (used only while tau > 0).
    Returns (grad, max_b).
    """
    b = _full_convolution_fft(a)
    max_b = b.max()
    diff = b - max_b
    exps = np.exp(diff / tau)
    mask = exps / exps.sum()
    a_rev = a[::-1]
    conv_len = mask.size + a.size - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    grad = 2.0 * conv[a.size - 1:a.size - 1 + a.size]
    return grad, max_b


# ----------------------------------------------------------------------
# Random families for the exploration phase
# ----------------------------------------------------------------------
def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Gamma‑draw → Dirichlet‑like vector."""
    alpha = rng.uniform(0.2, 5.0)
    return rng.gamma(alpha, 1.0, size=n)


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sparse Gamma entries (shape < 1)."""
    shape = rng.uniform(0.1, 0.8)
    return rng.gamma(shape, 1.0, size=n)


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    """All ones."""
    return np.full(n, 1.0)


def _init_left_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotone decreasing power‑law."""
    idx = np.arange(n)
    p = rng.uniform(0.3, 3.0)
    offset = rng.uniform(0.0, 10.0)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * base + (1.0 - mix) * (idx + offset) ** (-p)


def _init_left_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotone decreasing exponential."""
    idx = np.arange(n)
    lam = rng.uniform(0.001, 0.02)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * base + (1.0 - mix) * np.exp(-lam * idx)


def _init_sparse_bumps(n: int, rng: np.random.Generator) -> np.ndarray:
    """Few random rectangular bumps on a tiny flat background."""
    raw = np.full(n, rng.uniform(0.0, 0.15))
    nbumps = rng.integers(1, max(2, n // 250))
    for _ in range(nbumps):
        center = rng.integers(0, n)
        width = rng.integers(3, max(5, n // 100))
        height = rng.uniform(0.8, 4.0)
        lo = max(0, center - width // 2)
        hi = min(n, lo + width)
        raw[lo:hi] = np.maximum(raw[lo:hi], height)
    return raw


def _init_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    """Centered Gaussian bump."""
    mu = (n - 1) / 2.0
    sigma = rng.uniform(0.05, 0.30) * n
    x = np.arange(n)
    return np.exp(-0.5 * ((x - mu) / sigma) ** 2)


def _init_half_sine(n: int, rng: np.random.Generator) -> np.ndarray:
    """Half‑sine shape (zero at the ends)."""
    x = np.arange(n)
    raw = np.sin(np.pi * x / (n - 1))
    raw = raw * rng.uniform(0.5, 2.0) + rng.uniform(0.0, 0.2)
    return raw


def _init_ramp(n: int, rng: np.random.Generator) -> np.ndarray:
    """Linear ramp decreasing from left to right."""
    start = rng.uniform(0.5, 2.0)
    end = rng.uniform(0.0, 0.5)
    return np.linspace(start, end, n)


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    With probability ~0.22 we warm‑start from the global best,
    otherwise we draw from one of the parametric families.
    """
    S = math.sqrt(2.0 * n)

    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    style = rng.choice(['dirichlet', 'gamma', 'uniform',
                        'left_powerlaw', 'left_exponential',
                        'sparse', 'gaussian', 'half_sine', 'ramp'])
    if style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'left_powerlaw':
        raw = _init_left_powerlaw(n, rng)
    elif style == 'left_exponential':
        raw = _init_left_exponential(n, rng)
    elif style == 'sparse':
        raw = _init_sparse_bumps(n, rng)
    elif style == 'gaussian':
        raw = _init_gaussian(n, rng)
    elif style == 'half_sine':
        raw = _init_half_sine(n, rng)
    else:    # ramp
        raw = _init_ramp(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Projected‑gradient descent on the simplex (exact hard‑max gradient)
# ----------------------------------------------------------------------
def _pgd_optimize(a0: np.ndarray,
                  deadline: float,
                  rng: np.random.Generator,
                  init_step: float = 0.15) -> tuple[np.ndarray, float]:
    """
    Deterministic projected‑gradient descent that follows the exact
    sub‑gradient of max(a*a).  The iterate is kept on the ℓ₁‑simplex
    (non‑negative, fixed sum).  Returns best vector seen and its peak.
    """
    n = a0.shape[0]
    S = math.sqrt(2.0 * n)

    a = a0.copy()
    a = a * (S / a.sum())          # enforce exact norm
    best = a.copy()
    best_peak = _max_convolution(a)
    step = init_step

    while time.time() < deadline:
        # exact sub‑gradient
        grad, peak = _hard_max_gradient(a)

        # make gradient sum‑zero (project onto hyperplane sum=0)
        grad_proj = grad - grad.mean()

        # trial step
        new_a = a - step * grad_proj
        new_a = np.maximum(new_a, 0.0)
        if new_a.sum() == 0.0:
            # avoid collapse – jitter a little and continue
            new_a = a + rng.normal(scale=1e-3, size=n)
            new_a = np.maximum(new_a, 0.0)

        new_a = new_a * (S / new_a.sum())
        new_peak = _max_convolution(new_a)

        if new_peak < best_peak:
            best = new_a.copy()
            best_peak = new_peak
            a = new_a
            # if we are improving we can be a bit bolder
            step = min(step * 1.02, 2.0)
        else:
            # reject but keep move if it does not increase the peak
            if new_peak < peak:
                a = new_a
                step = min(step * 1.05, 1.0)
            else:
                # too aggressive – shrink step
                step *= 0.65
                if step < 1e-6:
                    step = 1e-6
    return best, best_peak


# ----------------------------------------------------------------------
# Simulated‑annealing (mass‑transfer walk)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Mass‑transfer walk on the simplex with exponential acceptance."""
    n = len(seq)
    S = math.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score
    temperature = 0.15 * cur_score

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] == 0.0:
            continue
        delta = rng.uniform(0.0, 0.3) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())
        new_score = _max_convolution(new)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

        temperature *= 0.99995
        if temperature < 1e-6:
            temperature = 1e-6

    return best, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climber (strictly downhill)
# ----------------------------------------------------------------------
def _hill_climb(seq: np.ndarray,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Very cheap greedy hill‑climber (downhill moves only)."""
    n = len(seq)
    S = math.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] < 1e-12:
            continue
        delta = rng.uniform(0.0, 0.15) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())
        new_score = _max_convolution(new)
        if new_score < cur_score:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Three phases are executed: (1) fast random exploration, (2) multi‑scale
    Adam refinement, (3) a dedicated projected‑gradient descent followed by
    simulated annealing and a greedy hill‑climber.
    The resulting sequence is stored in the global ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    safety_margin = 5.0
    end_time = time.time() + budget_s - safety_margin
    if end_time <= time.time():
        raise RuntimeError("Budget too small after safety margin.")

    # --------------------------------------------------------------
    # Warm‑start from a previously stored construction, if any
    # --------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # --------------------------------------------------------------
    # Phase 1 – fast random exploration (≈20 % of budget)
    # --------------------------------------------------------------
    explore_deadline = time.time() + 0.20 * budget_s
    top_k = 12
    top_candidates = []                # (score, vector)
    best_seq = None
    best_score = np.inf
    attempt = 0

    while time.time() < explore_deadline:
        attempt += 1
        # length distribution – mix of moderate and larger sizes
        if rng.random() < 0.70:
            n = int(rng.integers(2500, 7000))
        else:
            n = int(rng.integers(7000, 17000))

        a0 = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a0)

        if peak < best_score:
            best_score = peak
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempt:5d} n={n:5d}  new best={best_score:.6f}")

        # maintain top‑k list (sorted ascending)
        if len(top_candidates) < top_k:
            top_candidates.append((peak, a0.copy()))
            top_candidates.sort(key=lambda x: x[0])
        elif peak < top_candidates[-1][0]:
            top_candidates[-1] = (peak, a0.copy())
            top_candidates.sort(key=lambda x: x[0])

    # safety net
    if best_seq is None:
        n = 4000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n)
        best_score = _max_convolution(best_seq)
        best_n = n
        top_candidates = [(best_score, best_seq.copy())]

    # --------------------------------------------------------------
    # Phase 2 – multi‑scale Adam refinement (≈60 % of budget)
    # --------------------------------------------------------------
    max_len = min(80000, best_n * 12)
    # geometric schedule of lengths
    lengths = [best_n]
    cur_len = best_n
    while cur_len * 1.5 < max_len:
        cur_len = int(cur_len * 1.5)
        lengths.append(cur_len)
    lengths.append(max_len)
    lengths = sorted(set(lengths))

    # iterate over shortlisted candidates (most promising first)
    for init_idx, (init_score, init_seq) in enumerate(top_candidates):
        cur_seq = init_seq.copy()
        for target_n in lengths:
            if time.time() >= end_time - 30.0:
                break

            a_scaled = _scale_sequence(cur_seq, target_n)
            a_scaled = np.clip(a_scaled, 0.0, None)
            a_scaled = a_scaled * (math.sqrt(2.0 * target_n) / a_scaled.sum())

            # give each scale a slice of the remaining budget
            remaining = end_time - time.time()
            slice_budget = min(remaining * 0.45, 180.0)

            # randomise a couple of hyper‑parameters for diversity
            lr0 = rng.choice([0.55, 0.60, 0.65])
            tau_decay = rng.choice([0.9996, 0.9998, 0.9999])
            refined, peak = _adam_optimize(a_scaled,
                                           time.time() + slice_budget,
                                           rng,
                                           lr0=lr0,
                                           tau_decay=tau_decay,
                                           lr_decay=0.9999,
                                           jitter_every=300)

            if peak < best_score:
                best_score = peak
                best_seq = refined.copy()
                best_n = target_n
                print(f"[refine] init#{init_idx+1} len={target_n:5d}  improved score={best_score:.6f}")

            cur_seq = refined   # seed the next scale

        if time.time() >= end_time - 30.0:
            break

    # --------------------------------------------------------------
    # Phase 3 – deterministic PGD polishing (≈15 % of budget)
    # --------------------------------------------------------------
    remaining = end_time - time.time()
    if remaining > 20.0:
        pgd_deadline = time.time() + 0.65 * remaining
        best_seq, best_score = _pgd_optimize(best_seq, pgd_deadline, rng, init_step=0.12)
        remaining = end_time - time.time()

    # --------------------------------------------------------------
    # Phase 4 – cheap stochastic fine‑tuning (≈5 % of budget)
    # --------------------------------------------------------------
    if remaining > 15.0:
        # simulated‑annealing walk
        best_seq, best_score = _anneal(best_seq,
                                       time.time() + 0.55 * remaining,
                                       rng)
        remaining = end_time - time.time()
        if remaining > 5.0:
            best_seq, best_score = _hill_climb(best_seq,
                                               time.time() + remaining,
                                               rng)

    # --------------------------------------------------------------
    # Final sanity check via the supplied black‑box evaluator
    # --------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  internal max = {best_score:.6f}"
          f"  length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
