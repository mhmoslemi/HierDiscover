Improved step‑function search.

Algorithm
---------
1) Random warm‑starts from a richer family of shapes (U‑shape,
   symmetric power‑law, Gaussian, sparse bumps …).  The sum of the
   vector is normalised to √(2 n) because the external objective
   then equals the peak of the self‑convolution.

2) A multi‑scale Adam optimiser works on the *log‑softmax*
   parametrisation of the simplex.  The optimiser uses an exact
   sub‑gradient of the hard max (through FFT) and periodically
   symmetrises the vector (a ↔ a[::-1]) – the convolution is
   invariant under this operation and the symmetry stabilises the
   search.

3) The best candidates are interpolated to larger supports (up to
   1.2·10⁵ entries).  At each scale a short Adam run is performed
   with randomised learning‑rate and temperature decay.  The
   longest scale receives several independent restarts to escape
   local minima.

4) Finally a stochastic “mass‑transfer’’ annealer followed by a
   greedy hill‑climber refines the incumbent.

The whole pipeline respects the 1000 s budget; progress and the best

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Peak value of a*a (full convolution)."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]

# Exact sub‑gradient of the hard max‑convolution
# ----------------------------------------------------------------------
def _max_gradient(a: np.ndarray, tau: float):

    Returns (grad, max_b) where grad = ∂ max(b) / ∂a

    tau is a temperature for the soft‑max.  When tau is tiny the
    exact active set (all indices within a tolerance of the maximum)
    is used.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:

        s = mask.sum()
        if s == 0:
            mask = np.ones_like(b) / b.size
        else:
            mask /= s
    else:

        exps = np.exp(diff / tau)
        mask = exps / exps.sum()

    # gradient = 2 * (mask ⊛ reverse(a))
    a_rev = a[::-1]
    conv_len = mask.size + a.size - 1
    f_len = _next_pow2(conv_len)

    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)

    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    grad = 2.0 * conv[a.size - 1:a.size - 1 + a.size]
    return grad, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax representation)
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    lr0: float = 0.5,
    tau_decay: float = 0.9995,
    lr_decay: float = 0.99985,
    jitter_every: int = 400,
    symmetrize: bool = True,
) -> tuple[np.ndarray, float]:
    """
    Minimise the peak of the self‑convolution under the ℓ₁‑norm
    constraint (∑a = √(2 n)).  Returns the best vector seen and its
    peak value.
    """
    n = a0.shape[0]
    S = math.sqrt(2.0 * n)                     # required ℓ₁‑norm

    # Initialise logits from feasible a0 (log‑softmax)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr0

    # temperature for τ‑soft‑max
    tau = max(_full_convolution_fft(a0).max(), 0.5)

    best_a = a0.copy()
    best_peak = _max_convolution(a0)
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- recover a from logits (log‑softmax) -----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        if symmetrize:
            a = (a + a[::-1]) * 0.5
            a = a * (S / a.sum())

        # ----- gradient of (soft‑)max convolution -----
        g, peak = _max_gradient(a, tau)

        # keep best
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ----- gradient wrt logits (chain rule) -----
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- schedules -----
        tau *= tau_decay
        lr *= lr_decay

        if jitter_every and (it % jitter_every) == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak

def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Gamma‑draw → Dirichlet‑like vector."""
    alpha = rng.uniform(0.2, 5.0)
    return rng.gamma(alpha, 1.0, size=n)


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sparse Gamma entries (shape < 1)."""
    shape = rng.uniform(0.1, 0.8)
    return rng.gamma(shape, 1.0, size=n)


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    """All ones."""
    return np.full(n, 1.0)


def _init_left_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotone decreasing power‑law, optionally mixed with a uniform background."""
    idx = np.arange(n)
    p = rng.uniform(0.3, 3.0)
    offset = rng.uniform(0.0, 10.0)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * (idx + offset) ** (-p)
    return raw


def _init_left_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotone decreasing exponential, optionally mixed with a uniform background."""
    idx = np.arange(n)
    lam = rng.uniform(0.001, 0.02)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * np.exp(-lam * idx)
    return raw


def _init_sparse_bumps(n: int, rng: np.random.Generator) -> np.ndarray:
    """Few random rectangular bumps on a tiny flat background."""
    raw = np.full(n, rng.uniform(0.0, 0.15))
    nbumps = rng.integers(1, max(2, n // 250))
    for _ in range(nbumps):
        center = rng.integers(0, n)
        width = rng.integers(3, max(5, n // 100))
        height = rng.uniform(0.8, 4.0)
        lo = max(0, center - width // 2)
        hi = min(n, lo + width)
        raw[lo:hi] = np.maximum(raw[lo:hi], height)
    return raw


def _init_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    mu = (n - 1) / 2.0
    sigma = rng.uniform(0.05, 0.30) * n
    x = np.arange(n)
    return np.exp(-0.5 * ((x - mu) / sigma) ** 2)


def _init_half_sine(n: int, rng: np.random.Generator) -> np.ndarray:
    x = np.arange(n)
    raw = np.sin(np.pi * x / (n - 1))
    raw = raw * rng.uniform(0.5, 2.0) + rng.uniform(0.0, 0.2)
    return raw


def _init_Ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    """U‑shaped: high at both ends, low in the middle."""
    left = _init_left_powerlaw(n // 2 + 1, rng)
    right = left[::-1][: n // 2]
    if n % 2 == 1:
        middle = np.array([rng.uniform(0.0, left[-1])])
        raw = np.concatenate([left, middle, right])
    else:
        raw = np.concatenate([left, right])
    return raw


def _initial_vector(
    n: int,
    rng: np.random.Generator,
    warm_seq: np.ndarray | None,
) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Occasionally warm‑starts from the global best; otherwise draws from
    one of the parametric families.
    """
    S = math.sqrt(2.0 * n)

    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a * (S / a.sum())
        return a

    # choose style
    style = rng.choice(
        [
            "dirichlet",
            "gamma",
            "uniform",
            "left_powerlaw",
            "left_exponential",
            "sparse",
            "gaussian",
            "half_sine",
            "Ushape",
        ]
    )
    if style == "dirichlet":
        raw = _init_dirichlet(n, rng)
    elif style == "gamma":
        raw = _init_gamma(n, rng)
    elif style == "uniform":
        raw = _init_uniform(n, rng)
    elif style == "left_powerlaw":
        raw = _init_left_powerlaw(n, rng)
    elif style == "left_exponential":
        raw = _init_left_exponential(n, rng)
    elif style == "sparse":
        raw = _init_sparse_bumps(n, rng)
    elif style == "gaussian":
        raw = _init_gaussian(n, rng)
    elif style == "half_sine":
        raw = _init_half_sine(n, rng)
    else:  # Ushape
        raw = _init_Ushape(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Mass‑transfer annealer / Hill‑climber
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray, deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Stochastic annealing on the simplex (exponential acceptance)."""
    n = len(seq)
    S = math.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score
    temperature = 0.15 * cur_score

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] == 0.0:
            continue
        delta = rng.uniform(0.0, 0.3) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())
        new_score = _max_convolution(new)

        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / max(temperature, 1e-12)):
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

        temperature *= 0.99995
        if temperature < 1e-6:
            temperature = 1e-6

    return best, best_score


def _hill_climb(seq: np.ndarray, deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Greedy downhill moves only."""
    n = len(seq)
    S = math.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] < 1e-12:
            continue
        delta = rng.uniform(0.0, 0.15) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())
        new_score = _max_convolution(new)
        if new_score < cur_score:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score
    return best, best_score

"""
Improved step‑function search.

Algorithm
---------
1) Random warm‑starts from a richer family of shapes (U‑shape,
   symmetric power‑law, Gaussian, sparse bumps …).  The sum of the
   vector is normalised to √(2 n) because the external objective then
   equals the peak of the self‑convolution.

2) A multi‑scale Adam optimiser works on the *log‑softmax* parametrisation
   of the simplex.  The optimiser uses an exact sub‑gradient of the hard
   max (through FFT) and periodically symmetrises the vector
   (a ↔ a[::-1]) – the convolution is invariant under this operation and
   the symmetry stabilises the search.

3) The best candidates are interpolated to larger supports (up to
   1.2·10⁵ entries).  At each scale a short Adam run is performed with
   randomised learning‑rate and temperature decay.  The longest scale
   receives several independent restarts to escape local minima.

4) Finally a stochastic “mass‑transfer’’ annealer followed by a greedy
   hill‑climber refines the incumbent.

The whole pipeline respects the 1000 s budget; progress and the best
external score are printed.  The resulting list is stored in the global
variable ``height_sequence_1`` and returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Peak value of a*a (full convolution)."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Exact sub‑gradient of the hard max‑convolution
# ----------------------------------------------------------------------
def _max_gradient(a: np.ndarray, tau: float):
    """
    Returns (grad, max_b) where grad = ∂ max(b) / ∂a
    and max_b = max(b) with b = a*a (full convolution).

    tau is a temperature for the soft‑max.  When tau is tiny the exact
    active set (all indices within a tolerance of the maximum) is used.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        eps = max_b * 1e-9 + 1e-12
        mask = (b >= max_b - eps).astype(float)
        s = mask.sum()
        if s == 0:
            mask = np.ones_like(b) / b.size
        else:
            mask /= s
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / exps.sum()

    a_rev = a[::-1]
    conv_len = mask.size + a.size - 1
    f_len = _next_pow2(conv_len)

    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)

    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    grad = 2.0 * conv[a.size - 1:a.size - 1 + a.size]
    return grad, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax representation)
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    lr0: float = 0.5,
    tau_decay: float = 0.9995,
    lr_decay: float = 0.99985,
    jitter_every: int = 400,
    symmetrize: bool = True,
) -> tuple[np.ndarray, float]:
    """
    Minimise the peak of the self‑convolution under the ℓ₁‑norm
    constraint (∑a = √(2 n)).  Returns the best vector seen and its peak.
    """
    n = a0.shape[0]
    S = math.sqrt(2.0 * n)                     # required ℓ₁‑norm

    # initialise logits from feasible a0 (log‑softmax)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr0
    tau = max(_full_convolution_fft(a0).max(), 0.5)

    best_a = a0.copy()
    best_peak = _max_convolution(a0)
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- recover a from logits (log‑softmax) -----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        if symmetrize:
            a = (a + a[::-1]) * 0.5
            a = a * (S / a.sum())

        # ----- gradient of (soft‑)max convolution -----
        g, peak = _max_gradient(a, tau)

        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ----- gradient w.r.t. logits (chain rule) -----
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # clip for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- schedules -----
        tau *= tau_decay
        lr *= lr_decay

        if jitter_every and (it % jitter_every) == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Random families for the exploration phase
# ----------------------------------------------------------------------
def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.2, 5.0)
    return rng.gamma(alpha, 1.0, size=n)


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.1, 0.8)
    return rng.gamma(shape, 1.0, size=n)


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_left_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    p = rng.uniform(0.3, 3.0)
    offset = rng.uniform(0.0, 10.0)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * (idx + offset) ** (-p)
    return raw


def _init_left_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    lam = rng.uniform(0.001, 0.02)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * np.exp(-lam * idx)
    return raw


def _init_sparse_bumps(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = np.full(n, rng.uniform(0.0, 0.15))
    nbumps = rng.integers(1, max(2, n // 250))
    for _ in range(nbumps):
        center = rng.integers(0, n)
        width = rng.integers(3, max(5, n // 100))
        height = rng.uniform(0.8, 4.0)
        lo = max(0, center - width // 2)
        hi = min(n, lo + width)
        raw[lo:hi] = np.maximum(raw[lo:hi], height)
    return raw


def _init_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    mu = (n - 1) / 2.0
    sigma = rng.uniform(0.05, 0.30) * n
    x = np.arange(n)
    return np.exp(-0.5 * ((x - mu) / sigma) ** 2)


def _init_half_sine(n: int, rng: np.random.Generator) -> np.ndarray:
    x = np.arange(n)
    raw = np.sin(np.pi * x / (n - 1))
    raw = raw * rng.uniform(0.5, 2.0) + rng.uniform(0.0, 0.2)
    return raw


def _init_Ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    U‑shaped: large values at both ends, small in the middle.
    Constructed by mirroring a left‑hand decreasing profile.
    """
    left = _init_left_powerlaw(n // 2 + 1, rng)
    right = left[::-1][: n // 2]
    if n % 2 == 1:
        middle = np.array([rng.uniform(0.0, left[-1])])
        raw = np.concatenate([left, middle, right])
    else:
        raw = np.concatenate([left, right])
    return raw


def _initial_vector(
    n: int,
    rng: np.random.Generator,
    warm_seq: np.ndarray | None,
) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    With prob. ≈0.22 warm‑starts from the global best, otherwise draws from
    one of the parametric families.
    """
    S = math.sqrt(2.0 * n)

    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a * (S / a.sum())
        return a

    style = rng.choice(
        [
            "dirichlet",
            "gamma",
            "uniform",
            "left_powerlaw",
            "left_exponential",
            "sparse",
            "gaussian",
            "half_sine",
            "Ushape",
        ]
    )
    if style == "dirichlet":
        raw = _init_dirichlet(n, rng)
    elif style == "gamma":
        raw = _init_gamma(n, rng)
    elif style == "uniform":
        raw = _init_uniform(n, rng)
    elif style == "left_powerlaw":
        raw = _init_left_powerlaw(n, rng)
    elif style == "left_exponential":
        raw = _init_left_exponential(n, rng)
    elif style == "sparse":
        raw = _init_sparse_bumps(n, rng)
    elif style == "gaussian":
        raw = _init_gaussian(n, rng)
    elif style == "half_sine":
        raw = _init_half_sine(n, rng)
    else:  # Ushape
        raw = _init_Ushape(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Mass‑transfer annealer / Hill‑climber
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray, deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Stochastic annealing on the simplex (exponential acceptance)."""
    n = len(seq)
    S = math.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score
    temperature = 0.15 * cur_score

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] == 0.0:
            continue
        delta = rng.uniform(0.0, 0.3) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())
        new_score = _max_convolution(new)

        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / max(temperature, 1e-12)):
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

        temperature *= 0.99995
        if temperature < 1e-6:
            temperature = 1e-6

    return best, best_score


def _hill_climb(seq: np.ndarray, deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Greedy hill‑climber (strictly downhill moves only)."""
    n = len(seq)
    S = math.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] < 1e-12:
            continue
        delta = rng.uniform(0.0, 0.15) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())
        new_score = _max_convolution(new)
        if new_score < cur_score:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The algorithm consists of random exploration → multi‑scale Adam refinement
    → stochastic annealing → greedy hill‑climbing.  The best sequence is stored
    in the global variable ``height_sequence_1`` and returned as a plain list.
    """
    rng = np.random.default_rng(seed)
    safety_margin = 6.0
    end_time = time.time() + budget_s - safety_margin
    if end_time <= time.time():
        raise RuntimeError("Budget too small after safety margin.")

    # --------------------------------------------------------------
    # Warm‑start from a previous construction (if any)
    # --------------------------------------------------------------
    warm_seq = None
    if "height_sequence_1" in globals():
        try:
            warm_seq = np.asarray(globals()["height_sequence_1"], dtype=float)
        except Exception:
            warm_seq = None

    # --------------------------------------------------------------
    # Phase 1 – fast random exploration (≈15 % of budget)
    # --------------------------------------------------------------
    explore_deadline = time.time() + 0.15 * budget_s
    top_k = 20
    top_candidates = []                     # (score, seq)
    best_seq = None
    best_score = np.inf
    attempt = 0

    while time.time() < explore_deadline:
        attempt += 1
        # length choice – mostly medium, sometimes large
        if rng.random() < 0.75:
            n = int(rng.integers(2500, 8000))
        else:
            n = int(rng.integers(8000, 25000))

        a0 = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a0)

        if peak < best_score:
            best_score = peak
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] #{attempt:5d} n={n:5d}  new best={best_score:.6f}")

        if len(top_candidates) < top_k:
            top_candidates.append((peak, a0.copy()))
            top_candidates.sort(key=lambda x: x[0])
        elif peak < top_candidates[-1][0]:
            top_candidates[-1] = (peak, a0.copy())
            top_candidates.sort(key=lambda x: x[0])

    # safety net
    if best_seq is None:
        n = 4000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n)
        best_score = _max_convolution(best_seq)
        best_n = n
        top_candidates = [(best_score, best_seq.copy())]

    # --------------------------------------------------------------
    # Phase 2 – multi‑scale Adam refinement (≈70 % of budget)
    # --------------------------------------------------------------
    max_len = min(150_000, best_n * 30)
    lengths = [best_n]
    cur_len = best_n
    while cur_len * 1.5 < max_len:
        cur_len = int(cur_len * 1.5)
        lengths.append(cur_len)
    lengths.append(max_len)
    lengths = sorted(set(lengths))

    for init_idx, (init_score, init_seq) in enumerate(top_candidates):
        cur_seq = init_seq.copy()
        cur_len = len(cur_seq)

        for target_n in lengths:
            if time.time() >= end_time - 30.0:
                break

            a_scaled = _scale_sequence(cur_seq, target_n)
            a_scaled = np.clip(a_scaled, 0.0, None)
            a_scaled = a_scaled * (math.sqrt(2.0 * target_n) / a_scaled.sum())

            remaining = end_time - time.time()
            slice_budget = min(remaining * 0.45, 180.0)

            lr0 = rng.choice([0.45, 0.55, 0.70, 0.85])
            tau_decay = rng.choice([0.9992, 0.9994, 0.9996, 0.9998])

            refined, peak = _adam_optimize(
                a_scaled,
                time.time() + slice_budget,
                rng,
                lr0=lr0,
                tau_decay=tau_decay,
                lr_decay=0.99985,
                jitter_every=500,
                symmetrize=True,
            )

            if peak < best_score:
                best_score = peak
                best_seq = refined.copy()
                best_n = target_n
                print(f"[refine] init#{init_idx+1} len={target_n:5d}  improved={best_score:.6f}")

            cur_seq = refined
            cur_len = target_n

        if time.time() >= end_time - 30.0:
            break

    # --------------------------------------------------------------
    # Phase 3 – extra Adam restarts on the final (largest) length
    # --------------------------------------------------------------
    remaining = end_time - time.time()
    if remaining > 25.0:
        n_final = best_n
        for restart in range(4):
            if time.time() >= end_time - 5.0:
                break
            start = _scale_sequence(best_seq, n_final)
            start = np.clip(start, 0.0, None)
            start = start * (math.sqrt(2.0 * n_final) / start.sum())
            slice_budget = min(remaining * 0.2, 30.0)
            refined, peak = _adam_optimize(
                start,
                time.time() + slice_budget,
                rng,
                lr0=0.6,
                tau_decay=0.9995,
                lr_decay=0.9999,
                jitter_every=300,
                symmetrize=True,
            )
            if peak < best_score:
                best_score = peak
                best_seq = refined.copy()
                print(f"[restart] #{restart+1} new best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – cheap stochastic fine‑tuning (≈10 % of budget)
    # --------------------------------------------------------------
    remaining = end_time - time.time()
    if remaining > 20.0:
        best_seq, best_score = _anneal(best_seq, time.time() + 0.6 * remaining, rng)
        remaining = end_time - time.time()
        if remaining > 5.0:
            best_seq, best_score = _hill_climb(best_seq, time.time() + remaining, rng)

    # --------------------------------------------------------------
    # Final evaluation via the supplied black‑box
    # --------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(
        f"[final] external score = {external:.6f}  internal max = {best_score:.6f}"
        f"  length = {len(final_seq)}"
    )
    globals()["height_sequence_1"] = final_seq
    return final_seq
