def refine_by_simulated_annealing(a, deadline):
    n = len(a)
    cur = a.copy()
    cur_score = _max_convolution_fft(cur)
    best = cur.copy()
    best_score = cur_score
    start = time.time()
    rng = np.random.default_rng()
    while time.time() < deadline:
        # choose donor and receiver random
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)
        # choose delta up to min(cur[i], max_change)
        max_delta = cur[i] * 0.5
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        # no need to project because sum unchanged and nonnegativity holds (ensured)
        new_score = _max_convolution_fft(new)
        # temperature schedule
        elapsed = time.time() - start
        t = 1.0 - elapsed/(deadline - start)
        temp = 0.2 * (0.001 / 0.2) ** (1 - t) # or other
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score)/max(temp, 1e-12)):
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score
                print(f"[{time.time():.2f}] SA improved best={best_score:.6f}")
        # also maybe alternate donor/receiver based on high contributions
    return best, best_score

    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

        # Adam update

"""
Propose a step function minimising the peak of its self‑convolution.

Algorithm
---------
1. Build a diverse pool of candidates (power‑law, exponential, sinusoidal,
   previous best, and jittered copies) and project each onto the ℓ¹‑simplex
   {x≥0, Σx = √(2 n)}.

2. Phase A – Adam‑based projected gradient descent on a soft‑max
   approximation of the max‑convolution.
   The “temperature’’ β is increased exponentially so that the optimisation
   progressively focuses on the true maximum.  The optimiser uses Adam’s
   adaptive learning rates and a Euclidean simplex projection after every
   update.

3. Phase B – Deterministic mass‑transfer (greedy moves that shift a fraction
   of mass from indices contributing heavily to the current peak towards
   low‑contributing indices).  An annealed Metropolis rule permits occasional
   uphill moves.

4. Phase C – Simple re‑ordering heuristics (alternating large/small
   layout) – a cheap way of exploiting the ordering freedom.

5. Phase D – Permutation annealing (swap‑based simulated annealing) that
   refines the ordering of the multiset of values.

6. Phase E – Final polishing with a high‑β Adam run.

All phases respect the overall 1000 s time budget.  The best sequence found
is returned, stored in the global variable ``height_sequence_1`` for reuse,
and its score is printed using the provided ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Benchmark objective: 2·n·max(a*a) / (Σa)² (for Σa=√(2n) this equals max)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """Create an exponential decay sequence e^{-lam·i} and normalise."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _sinusoid_sequence(n: int, target_sum: float) -> np.ndarray:
    """Half‑sine shape (positive on [0,π]) normalised to target_sum."""
    a = np.sin(np.pi * np.arange(n, dtype=np.float64) / (n - 1))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient: 2 * (a reversed) ⋆ p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b

# Phase A – Adam projected gradient descent on soft‑max max
# ----------------------------------------------------------------------


def _adam_descent(a0: np.ndarray,

                  step_factor: float = 0.6,
                  beta_start: float = 0.5,
                  beta_mul: float = 2.0,
                  beta_step: int = 200) -> tuple[np.ndarray, float]:
    """
    Adam optimiser with simplex projection.
    The soft‑max temperature β is increased every `beta_step` iterations.
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta = beta_start
    t = 1
    rng = np.random.default_rng()

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        # Adam moments
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad * grad)

        # bias‑corrected
        m_hat = m / (1.0 - 0.9 ** t)
        v_hat = v / (1.0 - 0.999 ** t)

        # adaptive step size
        norm = np.linalg.norm(m_hat)
        if norm == 0:
            lr = 0.0
        else:
            lr = step_factor * target_sum / ((beta + 1.0) * norm)

        a -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={t:5d} β={beta:6.1f} score={best_val:.6f}")

        # beta schedule
        if t % beta_step == 0 and beta < max_beta:
            beta = min(beta * beta_mul, max_beta)

        # occasional jitter to escape tiny plateaus
        if t % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        t += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer moves that directly attack the current maximal
    convolution entry.  Moves are accepted with a Metropolis rule whose
    temperature decays exponentially with the remaining time.
    """
    a = a0.copy()
    n = a.shape[0]

    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()

    # initial convolution
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to the current peak
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Re‑ordering utilities
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """
    Reorder a sequence by placing the largest value leftmost,
    the second largest rightmost, the third next to the left, etc.
    """
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left = 0
    right = n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


# ----------------------------------------------------------------------
# Phase D – Permutation annealing (swap‑based simulated annealing)
# ----------------------------------------------------------------------


def _permutation_annealing(a: np.ndarray,
                           target_sum: float,
                           deadline: float,
                           init_score: float | None = None) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing over swaps of two indices.
    The ordering is altered while the multiset of values stays unchanged.
    """
    cur_seq = a.copy()
    cur_score = init_score if init_score is not None else _max_convolution_fft(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score

    start_time = time.time()
    n = cur_seq.shape[0]
    rng = np.random.default_rng()

    start_temp = 0.2
    end_temp = 1e-5

    while time.time() < deadline:
        elapsed = time.time() - start_time
        ratio = elapsed / (deadline - start_time) if deadline > start_time else 0.0
        temp = start_temp * (end_temp / start_temp) ** ratio

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # swap
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        new_score = _max_convolution_fft(cur_seq)
        delta = new_score - cur_score

        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / max(temp, 1e-12)):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
                print(f"[{time.time():.2f}] Perm anneal new best = {best_score:.6f}")
        else:
            # revert
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        if ratio > 0.9 and rng.random() < 0.01:
            rng.shuffle(cur_seq)
            cur_score = _max_convolution_fft(cur_seq)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The algorithm respects the 1000 s time budget and returns the best
    sequence discovered.
    """
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Choose problem size (larger n gives more degrees of freedom)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 15000
    elif budget_s >= 400:
        n = 10000
    else:
        n = 6000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Build initial candidate pool
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    init_seqs = []

    # power‑law family
    for p in np.linspace(0.3, 0.9, 7):
        init_seqs.append(_power_law_sequence(n, float(p), target_sum))

    # exponential decays
    for lam in (0.0005, 0.001, 0.002, 0.005):
        init_seqs.append(_exponential_sequence(n, lam, target_sum))

    # sinusoid
    init_seqs.append(_sinusoid_sequence(n, target_sum))

    # previous best if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        init_seqs.append(_resize_sequence(prev, n))

    # jittered copies (add diversity)
    for base in list(init_seqs):
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        seq = np.maximum(base + jitter, 0.0)
        seq = _project_to_simplex(seq, target_sum)
        init_seqs.append(seq)

    # a few random permutations of early candidates
    for base in init_seqs[:4]:
        seq = base.copy()
        rng.shuffle(seq)
        init_seqs.append(seq)

    # ------------------------------------------------------------------
    # Phase A – Adam projected gradient descent
    # ------------------------------------------------------------------
    phase_a_end = start + (deadline - start) * 0.45
    best_seq = None
    best_score = np.inf

    for init in init_seqs:
        if time.time() >= phase_a_end:
            break
        # allocate a modest budget per run (30 s or remaining share)
        remaining = phase_a_end - time.time()
        run_budget = min(30.0, remaining * 0.7 / max(1, len(init_seqs)))
        a_opt, val_opt = _adam_descent(
            init,
            target_sum,
            time.time() + run_budget,
            max_beta=1e5,
            step_factor=0.6,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] New global best after Adam = {best_score:.6f}")

    if best_seq is None:  # fallback
        best_seq = _power_law_sequence(n, 0.5, target_sum)
        best_score = _score(best_seq)

    # ------------------------------------------------------------------
    # Phase B – Deterministic mass‑transfer refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        mt_deadline = time.time() + (deadline - time.time()) * 0.30
        a_mt, val_mt = _deterministic_mass_transfer(
            best_seq,
            mt_deadline,
            target_sum,
            start_temp=0.8,
            end_temp=1e-5,
            low_frac=0.04,
        )
        if val_mt < best_score:
            best_score = val_mt
            best_seq = a_mt.copy()
            print(f"[{time.time():.2f}] After mass‑transfer best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase C – Simple re‑ordering heuristics
    # ------------------------------------------------------------------
    reorder_cands = [best_seq, _alternating_order(best_seq)]
    # a couple of random shuffles
    for _ in range(2):
        tmp = best_seq.copy()
        rng.shuffle(tmp)
        reorder_cands.append(tmp)

    for cand in reorder_cands:
        cand_score = _max_convolution_fft(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] Ordering improvement -> score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase D – Permutation annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        perm_deadline = time.time() + (deadline - time.time()) * 0.20
        a_perm, val_perm = _permutation_annealing(
            best_seq,
            target_sum,
            perm_deadline,
            init_score=best_score,
        )
        if val_perm < best_score:
            best_score = val_perm
            best_seq = a_perm.copy()
            print(f"[{time.time():.2f}] After permutation annealing best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase E – Final high‑β Adam polishing
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_descent(
            best_seq,
            target_sum,
            deadline,
            max_beta=3e5,
            step_factor=0.5,
        )
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing (trim trailing zeros) and final evaluation
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose the sequence for possible later runs
    globals()["height_sequence_1"] = final_seq

    return final_seq
