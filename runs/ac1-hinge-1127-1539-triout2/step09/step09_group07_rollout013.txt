def _beta_shape(n, p):
    idx = np.arange(n)
    raw = ( (idx + 1) * (n - idx) ) ** p
    return raw

def _sine_shape(n,p):
    idx = np.arange(n)
    raw = np.sin(np.pi * idx / (n - 1))
    # raw = 0 at ends; to avoid zeros maybe add epsilon: raw = raw + eps
    raw = raw ** p
    return raw

def _gaussian_shape(n, sigma_factor):
    idx = np.arange(n)
    mu = (n - 1)/2.0
    sigma = sigma_factor * n
    raw = np.exp(-0.5 * ((idx - mu)/sigma)**2)
    return raw

def _exponential_shape(n, lam):
    idx = np.arange(n)
    raw = np.exp(-lam * idx) + np.exp(-lam * (n-1-idx))
    return raw

    return final_seq

"""
Propose a step‑function sequence that minimises ``evaluate_sequence``.

New ideas compared to the reference implementation
--------------------------------------------------
* **Deterministic shape sweep** – systematic search over a few
  parametric families (exponential, power‑law, sinusoidal, Gaussian,
  β‑type) for many lengths.  This often discovers a better “global”
  region than random draws alone.
* **Symmetry projection** – after every Adam update the candidate is
  mirrored onto the symmetric sub‑space (a ← (a+a[::-1])/2) which
  strongly reduces the peak of the self‑convolution.
* **Two‑stage Adam** – for each candidate we run a short Adam run with
  a high learning‑rate, followed by a longer low‑learning‑rate run.
* **Extended refinement schedule** – lengths are grown geometrically up
  to ≈ 8·10⁴ and the remaining budget is split among the most promising
  candidates.
* **Final annealing / hill‑climb** – as before, to capture any remaining
  tiny improvements.

The function respects the 1000 s budget, prints progress, stores the
best construction in the global variable ``height_sequence_1`` and
returns it as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based – O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum value of a*a (full convolution)."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Gradient of (soft‑)max convolution – used by Adam optimiser
# ----------------------------------------------------------------------
def _max_gradient(a: np.ndarray, tau: float):
    """
    Returns (grad, max_b) where grad = ∂ max(b) / ∂a
    and max_b = max(b) with b = a*a (full convolution).

    If tau is tiny we fall back to the exact active set (all indices
    within a tiny tolerance of the maximum) and average their
    gradients; otherwise a soft‑max (τ‑soft‑max) is used.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        eps = max_b * 1e-9 + 1e-12
        mask = (b >= max_b - eps).astype(float)
        s = mask.sum()
        if s == 0:
            mask = np.ones_like(b) / b.size
        else:
            mask /= s
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / exps.sum()

    a_rev = a[::-1]
    conv_len = mask.size + a.size - 1
    f_len = _next_pow2(conv_len)

    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)

    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    grad = 2.0 * conv[a.size - 1:a.size - 1 + a.size]
    return grad, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   lr0: float = 0.5,
                   tau_decay: float = 0.9995,
                   lr_decay: float = 0.99985,
                   jitter_every: int = 400,
                   enforce_symmetry: bool = False) -> tuple[np.ndarray, float]:
    """
    Minimise the peak of the self‑convolution under the ℓ₁‑norm
    constraint (sum = sqrt(2 n)).  Returns the best vector seen and its
    hard peak (max convolution value).
    """
    n = a0.shape[0]
    S = math.sqrt(2.0 * n)                     # required ℓ₁‑norm

    # initialise logits from a feasible a0 (log‑softmax)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr0

    # initialise temperature for τ‑soft‑max
    tau = max(_full_convolution_fft(a0).max(), 0.5)

    best_a = a0.copy()
    best_peak = _max_convolution(a0)
    it = 0

    while time.time() < deadline:
        it += 1

        # ------- recover a from logits (log‑softmax) -------
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # optional symmetry projection
        if enforce_symmetry:
            a = (a + a[::-1]) * 0.5
            a = a * (S / a.sum())

        # ------- gradient of (soft‑)max convolution -------
        g, peak = _max_gradient(a, tau)

        # keep best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ------- gradient w.r.t. logits (chain rule) -------
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ------- Adam update -------
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ------- annealing schedules -------
        tau *= tau_decay
        lr *= lr_decay

        # occasional jitter to escape plateaus
        if jitter_every and (it % jitter_every) == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Random families used for occasional diversification
# ----------------------------------------------------------------------
def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.2, 5.0)
    raw = rng.gamma(alpha, 1.0, size=n)
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.1, 0.8)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_left_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    p = rng.uniform(0.3, 3.0)
    offset = rng.uniform(0.0, 10.0)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * (idx + offset) ** (-p)
    return raw


def _init_left_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    lam = rng.uniform(0.001, 0.02)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * np.exp(-lam * idx)
    return raw


def _init_sparse_bumps(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = np.full(n, rng.uniform(0.0, 0.15))
    nbumps = rng.integers(1, max(2, n // 250))
    for _ in range(nbumps):
        center = rng.integers(0, n)
        width = rng.integers(3, max(5, n // 100))
        height = rng.uniform(0.8, 4.0)
        lo = max(0, center - width // 2)
        hi = min(n, lo + width)
        raw[lo:hi] = np.maximum(raw[lo:hi], height)
    return raw


def _init_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    mu = (n - 1) / 2.0
    sigma = rng.uniform(0.05, 0.30) * n
    x = np.arange(n)
    raw = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    return raw


def _init_half_sine(n: int, rng: np.random.Generator) -> np.ndarray:
    x = np.arange(n)
    raw = np.sin(np.pi * x / (n - 1))
    raw = raw * rng.uniform(0.5, 2.0) + rng.uniform(0.0, 0.2)
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Occasionally warm‑starts from the global best, otherwise draws
    from one of the parametric families.
    """
    S = math.sqrt(2.0 * n)

    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    style = rng.choice(['dirichlet', 'gamma', 'uniform',
                        'left_powerlaw', 'left_exponential',
                        'sparse', 'gaussian', 'half_sine'])
    if style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'left_powerlaw':
        raw = _init_left_powerlaw(n, rng)
    elif style == 'left_exponential':
        raw = _init_left_exponential(n, rng)
    elif style == 'sparse':
        raw = _init_sparse_bumps(n, rng)
    elif style == 'gaussian':
        raw = _init_gaussian(n, rng)
    else:  # half_sine
        raw = _init_half_sine(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Simulated‑annealing / greedy fine‑tuner (unchanged)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    n = len(seq)
    S = math.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score
    temperature = 0.15 * cur_score

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] == 0.0:
            continue
        delta = rng.uniform(0.0, 0.3) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())
        new_score = _max_convolution(new)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temperature, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

        temperature *= 0.99995
        if temperature < 1e-6:
            temperature = 1e-6

    return best, best_score


def _hill_climb(seq: np.ndarray,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    n = len(seq)
    S = math.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] < 1e-12:
            continue
        delta = rng.uniform(0.0, 0.15) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())
        new_score = _max_convolution(new)
        if new_score < cur_score:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score
    return best, best_score


# ----------------------------------------------------------------------
# Deterministic shape families for systematic sweep
# ----------------------------------------------------------------------
def _shape_beta(n: int, p: float) -> np.ndarray:
    """
    Symmetric beta‑type shape: ((i+1)*(n-i))**p.
    """
    idx = np.arange(n)
    raw = ((idx + 1) * (n - idx)) ** p
    return raw


def _shape_sine(n: int, p: float) -> np.ndarray:
    """Half‑sine raised to a power."""
    x = np.arange(n)
    raw = np.sin(np.pi * x / (n - 1)) ** p
    return raw


def _shape_gaussian(n: int, sigma_factor: float) -> np.ndarray:
    """Centered Gaussian with width sigma = sigma_factor * n."""
    mu = (n - 1) / 2.0
    x = np.arange(n)
    sigma = sigma_factor * n
    raw = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    return raw


def _shape_exponential(n: int, lam: float) -> np.ndarray:
    """Monotone decreasing exponential (mirrored)."""
    idx = np.arange(n)
    raw = np.exp(-lam * idx) + np.exp(-lam * (n - 1 - idx))
    return raw

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The algorithm consists of:

    1. Systematic sweep over several parametric families for a range
       of lengths.
    2. Multi‑scale Adam refinement (with an optional symmetry
       projection) on the most promising candidates.
    3. A cheap stochastic annealer followed by a strict hill‑climber.
    """
    rng = np.random.default_rng(seed)
    safety_margin = 5.0
    end_time = time.time() + budget_s - safety_margin
    if end_time <= time.time():
        raise RuntimeError("Budget too small after safety margin.")

    # --------------------------------------------------------------
    # Warm‑start from a previously stored construction, if any
    # --------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # --------------------------------------------------------------
    # Phase 1 – systematic shape sweep (≈20 % of budget)
    # --------------------------------------------------------------
    sweep_deadline = time.time() + 0.20 * budget_s
    best_seq = None
    best_score = np.inf
    best_n = None

    # Lengths to investigate (chosen to give good coverage)
    length_pool = [2000, 3000, 4000, 5000, 6000, 8000,
                   12000, 20000, 32000, 48000]

    # Parameter grids for the families
    beta_ps = np.linspace(0.0, 4.0, 21)          # 0,0.2,…,4
    sine_ps = np.linspace(0.2, 5.0, 25)          # 0.2,0.4,…,5
    sigma_factors = np.linspace(0.05, 0.30, 6)   # 0.05,…,0.30
    lam_grid = np.linspace(0.001, 0.02, 20)      # 0.001‑0.02

    attempt = 0
    for n in length_pool:
        if time.time() >= sweep_deadline:
            break
        S = math.sqrt(2.0 * n)

        # ----- beta family -----
        for p in beta_ps:
            raw = _shape_beta(n, p)
            raw = np.clip(raw, 0.0, None)
            a = raw * (S / raw.sum())
            maxb = _max_convolution(a)
            attempt += 1
            if maxb < best_score:
                best_score = maxb
                best_seq = a.copy()
                best_n = n
                print(f"[sweep] #{attempt:5d} n={n:5d} β p={p:4.2f} score={maxb:.6f}")

        # ----- sine family -----
        for p in sine_ps:
            raw = _shape_sine(n, p)
            raw = np.clip(raw, 0.0, None)
            a = raw * (S / raw.sum())
            maxb = _max_convolution(a)
            attempt += 1
            if maxb < best_score:
                best_score = maxb
                best_seq = a.copy()
                best_n = n
                print(f"[sweep] #{attempt:5d} n={n:5d} sine p={p:4.2f} score={maxb:.6f}")

        # ----- Gaussian family -----
        for sf in sigma_factors:
            raw = _shape_gaussian(n, sf)
            raw = np.clip(raw, 0.0, None)
            a = raw * (S / raw.sum())
            maxb = _max_convolution(a)
            attempt += 1
            if maxb < best_score:
                best_score = maxb
                best_seq = a.copy()
                best_n = n
                print(f"[sweep] #{attempt:5d} n={n:5d} gauss σ={sf:.2f} score={maxb:.6f}")

        # ----- exponential family -----
        for lam in lam_grid:
            raw = _shape_exponential(n, lam)
            raw = np.clip(raw, 0.0, None)
            a = raw * (S / raw.sum())
            maxb = _max_convolution(a)
            attempt += 1
            if maxb < best_score:
                best_score = maxb
                best_seq = a.copy()
                best_n = n
                print(f"[sweep] #{attempt:5d} n={n:5d} exp λ={lam:.4f} score={maxb:.6f}")

    # --------------------------------------------------------------
    # Safety net – guarantee at least one candidate
    # --------------------------------------------------------------
    if best_seq is None:
        n = 4000
        S = math.sqrt(2.0 * n)
        best_seq = np.full(n, S / n)
        best_score = _max_convolution(best_seq)
        best_n = n
        print("[fallback] created uniform baseline")

    # --------------------------------------------------------------
    # Phase 2 – multi‑scale Adam refinement (≈70 % of budget)
    # --------------------------------------------------------------
    # Assemble a short list of promising candidates
    top_candidates = [(best_score, best_seq.copy())]

    # add a few random families for diversity
    for _ in range(5):
        n = int(rng.integers(3000, 15000))
        a0 = _initial_vector(n, rng, warm_seq)
        sc = _max_convolution(a0)
        top_candidates.append((sc, a0.copy()))
        if sc < best_score:
            best_score, best_seq, best_n = sc, a0.copy(), n

    # possibly add warm‑started versions at other lengths
    if warm_seq is not None:
        for target_n in length_pool:
            if target_n == best_n:
                continue
            scaled = _scale_sequence(warm_seq, target_n)
            scaled = np.clip(scaled, 0.0, None)
            scaled = scaled * (math.sqrt(2.0 * target_n) / scaled.sum())
            sc = _max_convolution(scaled)
            top_candidates.append((sc, scaled.copy()))
            if sc < best_score:
                best_score, best_seq, best_n = sc, scaled.copy(), target_n

    top_candidates.sort(key=lambda x: x[0])
    top_candidates = top_candidates[:8]   # keep the 8 best

    # Build a geometric length schedule for refinement
    max_refine_len = min(80000, best_n * 12)
    schedule = []
    cur = best_n
    while cur < max_refine_len:
        cur = int(cur * 1.5)
        schedule.append(cur)
    schedule.append(max_refine_len)
    schedule = sorted(set(schedule))

    # Run Adam (symmetrised & plain) on each candidate
    for cid, (init_score, init_seq) in enumerate(top_candidates, start=1):
        if time.time() >= end_time - 30.0:
            break
        cur_seq = init_seq.copy()
        for target_n in schedule:
            if time.time() >= end_time - 30.0:
                break

            # interpolate to the new length and re‑normalise
            a_scaled = _scale_sequence(cur_seq, target_n)
            a_scaled = np.clip(a_scaled, 0.0, None)
            a_scaled = a_scaled * (math.sqrt(2.0 * target_n) / a_scaled.sum())

            remaining = end_time - time.time()
            slice_budget = min(remaining * 0.20, 150.0)   # at most 150 s per slice

            # Choose optimiser variant randomly
            sym = rng.random() < 0.6
            lr0 = rng.choice([0.42, 0.48, 0.55, 0.62])
            tau_decay = rng.choice([0.9992, 0.9994, 0.9996, 0.9998])
            refined, peak = _adam_optimize(a_scaled,
                                           time.time() + slice_budget,
                                           rng,
                                           lr0=lr0,
                                           tau_decay=tau_decay,
                                           lr_decay=0.99985,
                                           jitter_every=350,
                                           enforce_symmetry=sym)

            # Enforce symmetry again just in case the optimiser was plain
            if sym:
                refined = (refined + refined[::-1]) * 0.5
                refined = refined * (math.sqrt(2.0 * target_n) / refined.sum())
                peak = _max_convolution(refined)

            if peak < best_score:
                best_score = peak
                best_seq = refined.copy()
                best_n = target_n
                print(f"[refine] cand={cid} len={target_n:5d} sym={sym} new_best={peak:.6f}")

            cur_seq = refined.copy()

    # --------------------------------------------------------------
    # Phase 3 – cheap stochastic fine‑tuning (≈10 % of budget)
    # --------------------------------------------------------------
    remaining = end_time - time.time()
    if remaining > 20.0:
        best_seq, best_score = _anneal(best_seq,
                                       time.time() + 0.6 * remaining,
                                       rng)
        remaining = end_time - time.time()
        if remaining > 5.0:
            best_seq, best_score = _hill_climb(best_seq,
                                               time.time() + remaining,
                                               rng)

    # --------------------------------------------------------------
    # Final sanity check via the supplied black‑box evaluator
    # --------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  internal max = {best_score:.6f}"
          f"  length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
