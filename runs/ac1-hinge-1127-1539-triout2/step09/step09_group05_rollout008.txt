"""
Optimiser for the non‑negative step‑function problem.

Main ideas
----------

* **Broad random exploration** – generate many candidates from a rich
  collection of parametric families (power‑law, exponential, linear,
  U‑shape, sine, random‑FFT, Dirichlet, uniform).  Keep the best few
  based on the *hard* peak of the autocorrelation.

* **Annealed smooth‑max gradient descent** – minimise a smooth
  approximation of the peak
        τ·log∑ₖexp((a∗a)[k]/τ)
  using an Adam‑style update and projection onto the simplex
  {a≥0,∑a=√(2n)}.  τ is reduced exponentially during the run
  (simulated‑annealing on the temperature) so the algorithm starts
  from a wide‑band gradient that sees many entries of the convolution
  and ends with a true sub‑gradient of the peak.

* **Length scaling** – after optimisation at one length, linearly
  interpolate the best vector to a larger length (up to 25000) and
  continue the annealed descent.  Larger n gives the optimiser more
  degrees of freedom to flatten the autocorrelation.

* **Final sanity check** – the best vector is evaluated with the
  supplied ``evaluate_sequence`` function and stored in the global
  variable ``height_sequence_1``.

The whole routine respects a 1000 s budget; if the time limit is
approached the algorithm stops early and returns the best sequence
found so far.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT, scaling, projection, gradient)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a∗a."""
    return _full_convolution_fft(a).max()


def _smooth_max_grad(a: np.ndarray, tau: float):
    """
    Returns (grad, max_val) where ``grad`` is the gradient of the
    smooth max  fₜ(a)=τ·log∑exp((a∗a)[k]/τ) and ``max_val`` is the
    (hard) peak maxₖ (a∗a)[k].

    The gradient is
        ∇fₜ(a) = 2 * (w ∗ rev(a)) / ∑w,
    where wₖ = exp((bₖ‑max_b)/τ) and b = a∗a.
    """
    n = a.shape[0]
    total_len = 2 * n - 1               # length of b
    conv_len = total_len + n - 1         # length of w∗rev(a)
    f_len = _next_pow2(conv_len)

    # --- convolution a∗a -------------------------------------------------
    A = np.fft.rfft(a, n=f_len)
    b = np.fft.irfft(A * A, n=f_len)[:total_len]
    max_b = b.max()

    # --- soft‑max weights ------------------------------------------------
    # shift by max_b to avoid overflow
    w_hat = np.exp((b - max_b) / tau)
    w_sum = w_hat.sum()    # ≥ 1 because the maximal entry contributes 1

    # --- gradient (w ∗ rev(a)) -----------------------------------------
    a_rev = a[::-1]
    Fw = np.fft.rfft(w_hat, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fw * Fa_rev, n=f_len)[:conv_len]

    start = n - 1
    grad = 2.0 * conv[start:start + n] / w_sum
    return grad, max_b


def _project_onto_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the simplex
        { w | w_i ≥ 0 , ∑ w_i = S }.
    Implements the O(n) algorithm of Duchi et al.
    """
    if S <= 0:
        return np.zeros_like(v)

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho_arr = u - cssv / np.arange(1, len(v) + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]

# Parametric families for random initialisation
# ----------------------------------------------------------------------
def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    raw = mix * ush + (1.0 - mix) * base
    return raw


def _init_sine(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    vals = np.sin(np.pi * (i + 1) / (n + 1))
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * vals + (1.0 - mix) * base
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    return rng.random(n)


def _init_random_fft(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Random phase spectrum with constant magnitude.
    The inverse FFT is made non‑negative and scaled to the simplex.
    """
    # real‑valued sequence of length n, we build its half‑spectrum
    half = n // 2 + 1
    phases = rng.uniform(0.0, 2 * np.pi, half)
    # magnitude = 1 for all frequencies
    real = np.cos(phases)
    imag = np.sin(phases)
    # enforce Hermitian symmetry
    spec = np.empty(half, dtype=np.complex128)
    spec.real = real
    spec.imag = imag
    a = np.fft.irfft(spec, n=n)
    # shift to non‑negative
    a -= a.min()
    a += 1e-6
    return a


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Occasionally warm‑starts from the globally best sequence and otherwise
    draws from a random parametric family.
    """
    S = np.sqrt(2.0 * n)

    # warm start with probability ≈0.2
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
        return a

    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'ushape', 'sine', 'uniform', 'dirichlet',
                        'randfft'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'sine':
        raw = _init_sine(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    else:                     # randfft
        raw = _init_random_fft(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Annealed smooth‑max gradient descent (Adam + simplex projection)
# ----------------------------------------------------------------------
def _annealed_adam_descent(a0: np.ndarray,
                           deadline: float,
                           rng: np.random.Generator,
                           initial_hard_max: float) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation of the smooth max objective.
    ``deadline`` is an absolute UNIX time; the routine stops before it.
    Returns the best (hard‑max) vector seen and its peak value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.15                # base learning‑rate before projection

    best_a = a.copy()
    best_val = initial_hard_max

    start = time.time()
    it = 0

    while time.time() < deadline:
        it += 1

        #--- temperature schedule (exponential decay) --------------------
        elapsed = time.time() - start
        total = deadline - start
        # ratio ∈ [0,1]; we decay τ from 0.5·max_b to 0.001·max_b
        # we obtain the current max via a quick convolution inside the
        # smooth‑max routine.
        grad, cur_max = _smooth_max_grad(a, tau=1.0)    # placeholder τ=1.0
        # now replace the placeholder τ with a proper schedule:
        #   τ = cur_max * (0.5 ** (ratio*7))  → after ~7 halvings τ ≈ 0.008·cur_max
        tau = cur_max * (0.5 ** (ratio * 7.0))
        # recompute grad with the correct τ
        grad, cur_max = _smooth_max_grad(a, tau)

        #--- Adam update -------------------------------------------------
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        #--- projection onto the simplex --------------------------------
        a = _project_onto_simplex(a - step, S)

        #--- evaluate hard peak -----------------------------------------
        hard_peak = _max_convolution(a)
        if hard_peak + 1e-12 < best_val:
            best_val = hard_peak
            best_a = a.copy()

        # occasional jitter to escape tiny plateaus
        if it % 500 == 0:
            jitter = rng.normal(scale=0.001 * S, size=n)
            a = _project_onto_simplex(a + jitter, S)

    return best_a, best_val

# Main entry point required by the harness

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises the provided
    ``evaluate_sequence``.  The routine follows the plan outlined in the
    module docstring and stores the final construction in the global
    variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    # safety margin so we never overrun the allotted time
    end_time = time.time() + budget_s - 5.0

    # ------------------------------------------------------------------
    # Warm‑start handling
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # ------------------------------------------------------------------
    # Phase 1 – broad random exploration (≈30 % of the budget)
    # ------------------------------------------------------------------
    top_k = 6                         # keep the best six candidates
    candidates = []                  # list of (hard_peak, seq, n)
    explore_deadline = end_time - 300.0   # reserve ~5 min for refinement

    attempts = 0
    while time.time() < explore_deadline:
        attempts += 1
        # prefer longer sequences but keep some variety
        if rng.random() < 0.78:
            n = int(rng.integers(3000, 15000))
        else:
            n = int(rng.integers(15000, 25000))
        seq = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(seq)

        # keep only the best ``top_k`` candidates
        if len(candidates) < top_k:
            candidates.append((peak, seq, n))
            candidates.sort(key=lambda x: x[0])
        else:
            if peak < candidates[-1][0]:
                candidates[-1] = (peak, seq, n)
                candidates.sort(key=lambda x: x[0])

        if attempts % 250 == 0:
            print(f"[explore] attempts={attempts:4d} best={candidates[0][0]:.6f}")

    if not candidates:
        # fallback – uniform of moderate size
        n = 4000
        uniform = np.full(n, np.sqrt(2.0 * n) / n)
        candidates = [( _max_convolution(uniform), uniform, n )]

    # ------------------------------------------------------------------
    # Phase 2 – annealed Adam refinement on the top candidates
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    best_n = None

    for idx, (init_peak, seq, n) in enumerate(candidates):
        if time.time() >= end_time - 120.0:
            break
        # allocate a fair slice of remaining time
        remaining = end_time - time.time()
        slice_time = min(remaining * 0.18, 80.0)

        print(f"[refine] candidate {idx} n={n:5d} init_peak={init_peak:.6f}")

        opt_seq, opt_val = _annealed_adam_descent(seq,
                                                  time.time() + slice_time,
                                                  rng,
                                                  init_peak)

        print(f"          -> improved peak={opt_val:.6f}")

        if opt_val < best_val:
            best_val = opt_val
            best_seq = opt_seq.copy()
            best_n = n

    # ------------------------------------------------------------------
    # Phase 3 – length scaling & extra refinement (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < end_time - 30.0:
        # try a modest upscale (up to 25000)
        target_n = int(best_n * 1.7)
        if target_n > 25000:
            target_n = 25000
        if target_n != best_n:
            scaled = _scale_sequence(best_seq, target_n)
            S = np.sqrt(2.0 * target_n)
            scaled = np.clip(scaled, 0.0, None)
            scaled = scaled * (S / scaled.sum())

            init_peak = _max_convolution(scaled)
            print(f"[scale] upscaling {best_n}->{target_n}, init_peak={init_peak:.6f}")

            opt_seq, opt_val = _annealed_adam_descent(scaled,
                                                      end_time - 8.0,
                                                      rng,
                                                      init_peak)
            if opt_val < best_val:
                best_val = opt_val
                best_seq = opt_seq.copy()
                best_n = target_n
                print(f"[scale] improved peak={best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    if best_seq is None:
        # extreme fallback – a tiny uniform vector
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)

    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f} internal={best_val:.6f} length={len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
