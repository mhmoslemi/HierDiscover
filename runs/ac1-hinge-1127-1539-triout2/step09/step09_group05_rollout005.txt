def _optimize_sequence(a0, deadline, rng, eps_schedule, steps_per_eps, etc):

   a = a0.copy()
   n = len(a); S = sqrt(2*n)
   m = np.zeros_like(a)
   v = np.zeros_like(a)
   beta1=0.9; beta2=0.999; epsilon_adam=1e-8; lr=0.1

   time_now = time.time()
   for eps in eps_schedule:
       # loop until deadline per eps or given iteration count.
       while time.time() < deadline:
           # compute gradient using smoothed max at eps.
           conv = _full_convolution_fft(a)
           max_c = conv.max()
           # compute weight
           w = np.exp(eps * (conv - max_c))  # careful around large epsilon, shift
           w_sum = w.sum()
           if w_sum == 0:
               w = (conv == max_c).astype(float)
               w_sum = w.sum()
           w /= w_sum
           # gradient = 2 * conv(w, rev(a))
           # compute using FFT
           a_rev = a[::-1]
           # compute conv_w_rev = convolve(w, a_rev)
           # use fft approach with appropriate length
           conv_len = len(w) + n - 1
           f_len = _next_pow2(conv_len)
           Fw = np.fft.rfft(w, n=f_len)
           Fa_rev = np.fft.rfft(a_rev, n=f_len)
           conv_w_rev = np.fft.irfft(Fw * Fa_rev, n=f_len)[:conv_len]
           start = n - 1
           grad = 2.0 * conv_w_rev[start:start+n]

           # Adam update
           m = beta1 * m + (1 - beta1) * grad
           v = beta2 * v + (1 - beta2) * (grad ** 2)
           m_hat = m / (1 - beta1**(iteration+1))
           v_hat = v / (1 - beta2**(iteration+1))
           step = lr * m_hat / (np.sqrt(v_hat) + epsilon_adam)

           a = _project_onto_simplex(a - step, S)

           # optionally evaluate actual max aims for early stop
           cur_val = _max_convolution(a)
           # keep best
           if cur_val < best_val:
               best_val = cur_val; best_a = a.copy()
           # Maybe break early if no improvement.

   return best_a, best_val

"""
Improved step‑function optimizer.

The search consists of three layers:
1. **Broad random sampling** – many candidates from several parametric
   families (power‑law, exponential, linear, U‑shape, sine, Dirichlet…)
   of lengths 3 000–25 000 are created.  Each candidate is refined for a
   few seconds by an Adam optimiser that minimises a *smoothed* version
   of the max‑convolution (log‑sum‑exp surrogate).  The best few
   candidates are kept.

2. **Length scaling & deep Adam optimisation** – the current best vector
   is linearly interpolated to larger lengths (up to ≈ 25 000) and
   optimised for longer periods.  The Adam optimiser starts with a very
   smooth surrogate (ε≈0.2) and gradually sharpens it (ε→≈8), thereby
   providing a smooth gradient early on and converging to the true
   max‑convolution later.  After each Adam run the vector is projected
   back onto the simplex  {a_i≥0, Σa_i=√(2 n)}.

3. **Final stochastic mass‑transfer hill‑climb** – a cheap simulated‑annealing
   loop moves a random fraction of mass between two positions and accepts
   worsening moves with a slowly decaying temperature.  This stage
   escapes tiny plateaus that gradient descent may miss.

All stages respect the 1000 s budget; the best sequence is stored in the
global variable ``height_sequence_1`` and returned as a plain list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a∗a (hard max)."""
    return _full_convolution_fft(a).max()


def _project_onto_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex
        { w | w_i ≥ 0 , Σ w_i = S }.
    Implements the O(n) algorithm of Duchi et al.
    """
    if S < 0:
        raise ValueError("Simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]

def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    raw = mix * edge + (1.0 - mix) * np.ones(n)
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    raw = mix * exp_part + (1.0 - mix) * np.ones(n)
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    raw = mix * ush + (1.0 - mix) * np.ones(n)
    return raw


def _init_sine(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    vals = np.sin(np.pi * (i + 1) / (n + 1))
    mix = rng.uniform(0.0, 1.0)
    raw = mix * vals + (1.0 - mix) * np.ones(n)
    return raw


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet with random concentration; small concentration yields spiky vectors."""
    conc = rng.uniform(0.05, 3.0)
    return rng.gamma(conc, 1.0, size=n)


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Occasionally warm‑starts from the globally best sequence,
    otherwise draws from a random parametric family.
    """
    S = np.sqrt(2.0 * n)

    # Warm start with modest probability
    if warm_seq is not None and rng.random() < 0.18:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        tot = a.sum()
        if tot == 0.0:
            a = np.full(n, 1.0)
            tot = n
        a = a * (S / tot)
        return a

    style = rng.choice(
        ['powerlaw', 'exponential', 'linear', 'ushape', 'sine', 'dirichlet'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'sine':
        raw = _init_sine(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    tot = raw.sum()
    if tot == 0.0:
        raw = np.full(n, 1.0)
        tot = n
    a = raw * (S / tot)
    return a

def _smooth_gradient(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Gradient of the smoothed max:
        f_ε(a) = (1/ε) log Σ_k exp(ε·c_k),
    where c = a∗a.
    Returns ∇f_ε(a) as a dense array of length n.
    """
    n = a.shape[0]
    c = _full_convolution_fft(a)
    max_c = c.max()
    # soft‑max weights (shifted by max_c for stability)
    w = np.exp(eps * (c - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = (c == max_c).astype(float)
        w_sum = w.sum()
    w /= w_sum

    # convolution w * reverse(a)
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    f_len = _next_pow2(conv_len)
    Fw = np.fft.rfft(w, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fw * Fa_rev, n=f_len)[:conv_len]
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   start_eps: float = 0.2,
                   end_eps: float = 8.0,
                   lr: float = 0.25) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed max objective.
    eps is increased exponentially from start_eps to end_eps,
    providing a smooth-to‑sharp transition.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _max_convolution(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    it = 0
    start_time = time.time()
    total_time = deadline - start_time
    # schedule based on elapsed proportion
    while time.time() < deadline:
        it += 1
        # linear interpolation of ε in log‑space
        prop = (time.time() - start_time) / total_time
        eps = np.exp(np.log(start_eps) + prop * (np.log(end_eps) - np.log(start_eps)))

        grad = _smooth_gradient(a, eps)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        a = _project_onto_simplex(a - step, S)

        # occasional hard evaluation (every 20 iters)
        if it % 20 == 0:
            cur_val = _max_convolution(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                # when a good improvement is found we reset Adam moments a bit
                m *= 0.5
                v *= 0.5

        # jitter to avoid stagnation (very cheap)
        if it % 300 == 0:
            jitter = rng.normal(scale=0.0005 * S, size=n)
            a = _project_onto_simplex(a + jitter, S)

    # final hard evaluation
    final_val = _max_convolution(best_a)
    return best_a, final_val


def _mass_transfer_hill_climb(a0: np.ndarray,
                              deadline: float,
                              rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing move: transfer a random fraction of mass
    from one index to another, accepting worsening moves with a
    temperature that decays exponentially.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_val = _max_convolution(a)
    best_a = a.copy()
    best_val = cur_val

    T = 0.03 * cur_val  # start temperature as a few % of current peak
    decay = 0.9997

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        frac = rng.uniform(0.01, 0.20)
        delta = a[i] * frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _max_convolution(a_new)
        delta_val = new_val - cur_val
        if delta_val < 0.0 or rng.random() < np.exp(-delta_val / (T + 1e-12)):
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()

        T *= decay
        # occasional small jitter to keep exploring
        if rng.random() < 0.005:
            jitter = rng.normal(scale=0.0005 * np.sqrt(2.0 * n), size=n)
            a = _project_onto_simplex(a + jitter, np.sqrt(2.0 * n))
            cur_val = _max_convolution(a)

    return best_a, best_val

            target_lengths.append(cand)
    target_lengths = sorted(set(target_lengths))

    for target_n in target_lengths:
        if time.time() >= end_time - 200.0:
            break
        # scale current best to the new length
        a_scaled = _scale_sequence(best_seq, target_n)
        a_scaled = np.clip(a_scaled, 0.0, None)
        a_scaled = a_scaled * (np.sqrt(2.0 * target_n) / a_scaled.sum())

        # allocate a generous time slice (≈15 % of remaining budget)
        remaining = end_time - time.time()
        slice_time = min(remaining * 0.15, 85.0)

        a_opt, val = _adam_optimize(a_scaled,
                                    time.time() + slice_time,
                                    rng,
                                    start_eps=0.2,
                                    end_eps=8.0,
                                    lr=0.25)
        if val < best_score:
            best_score = val
            best_seq = a_opt.copy()
            best_n = target_n
            print(f"[scale‑adam] target_n={target_n:5d}  improved={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final long Adam run on the current best length
    # ------------------------------------------------------------------
    if time.time() < end_time - 60.0:
        remaining = end_time - time.time() - 5.0
        a_opt, val = _adam_optimize(best_seq,
                                    time.time() + remaining,
                                    rng,
                                    start_eps=0.2,
                                    end_eps=8.0,
                                    lr=0.25)
        if val < best_score:
            best_score = val
            best_seq = a_opt.copy()
            print(f"[final‑adam] length={len(best_seq)}  best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – stochastic mass‑transfer hill‑climb
    # ------------------------------------------------------------------
    if time.time() < end_time - 5.0:
        remaining = end_time - time.time() - 1.0
        a_opt, val = _mass_transfer_hill_climb(best_seq,
                                               time.time() + remaining,
                                               rng)
        if val < best_score:
            best_score = val
            best_seq = a_opt.copy()
            print(f"[anneal] final improve to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f}  internal={best_score:.6f}  length={len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
