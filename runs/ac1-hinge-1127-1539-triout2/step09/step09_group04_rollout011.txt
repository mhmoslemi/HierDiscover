The goal is to minimise  

\[
\text{score}(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2}
\]

with the constraints `a_i ≥ 0`.  
If `∑a = √(2n)` the denominator equals `2n` and the score collapses to the
maximum entry of the linear convolution `a*a`.  
Consequently the problem is:

*find a non‑negative sequence of a given length whose *autocorrelation* has the
smallest possible peak*.

The optimiser below follows a three‑phase strategy

1. **Quick sub‑gradient polish** – a few hundred iterations of the exact
   sub‑gradient method give a good starting shape.
2. **Simulated‑annealing local search** – repeatedly move a small amount of
   mass from a “donor” (large contribution to the current maximal
   convolution entry) to a “receiver” (small contribution).  
   Moves that improve the score are always kept; worse moves are accepted
   with a temperature‑dependent probability, allowing the algorithm to escape
   local minima.
3. **Final sub‑gradient refinement** – after the annealing phase a short
   sub‑gradient descent cleans up the remaining irregularities.

All steps respect the ℓ¹‑norm (`∑a = √(2n)`) by a projection onto the simplex.
The code is fully self‑contained, uses only `numpy` and `time`,
and respects the 1000 s budget.

```python
"""
Hybrid optimiser for the max‑convolution benchmark.

1️⃣  Warm‑start from the previous best (or a simple power‑law) and project
    to the ℓ¹‑simplex (∑a = √(2 n)).

2️⃣  A short exact sub‑gradient descent quickly reduces the peak.

3️⃣  Simulated‑annealing moves a small fraction of mass from a “donor”
    (large contribution to the current maximal convolution entry) to a
    “receiver” (small contribution).  Moves are accepted
    greedily or with Metropolis probability (temperature slowly cools).

4️⃣  A final sub‑gradient descent polishes the solution.

All operations keep the sequence non‑negative; the final list is trimmed
of trailing zeros and returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²)."""
    n = a.size
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Sequence a_i = (i+1)^(-p), normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the max‑convolution entry
# ----------------------------------------------------------------------


def _subgradient_max_descent(a0: np.ndarray,
                             target_sum: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true maximal convolution entry.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())

        # sub‑gradient of conv_{k_max}: 2·a[k_max‑i] on feasible indices
        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[partner]

        # step size – safe (no sign change) and diminishing
        norm = np.linalg.norm(grad)
        if norm == 0:
            break
        safe_eta = np.min(np.where(grad > 0, a / grad, np.inf))
        eta = min(0.5 * safe_eta, 0.5 / np.sqrt(it + 1))

        a = a - eta * grad
        a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it} best={best_val:.6f}")

        # occasional jitter to escape flat regions
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing mass‑transfer optimiser
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         init_best: float) -> tuple[np.ndarray, float]:
    """
    Annealing search that repeatedly transfers a small amount of mass from a
    high‑contributing index (donor) to a low‑contributing one (receiver).
    """
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()

    # current state
    conv = _conv_fft(a)
    cur_score = float(conv.max())

    best_a = a.copy()
    best_score = init_best if init_best is not None else cur_score
    if cur_score < best_score:
        best_score = cur_score
        best_a = a.copy()

    # temperature schedule
    temperature = max(0.05 * cur_score, 1e-4)
    it = 0

    while time.time() < deadline:
        # ---- choose donor / receiver based on the current maximal entry ----
        k_max = int(conv.argmax())
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)          # indices that appear in b_{k_max}
        partner = k_max - idx                                 # the matching partner for each idx
        weight = a[idx] * a[partner]                          # contribution a_i·a_{k_max‑i}

        donor = idx[np.argmax(weight)]

        # receiver: one of the 10‑percentile contributors (or random fallback)
        low_thr = np.percentile(weight, 10)
        low_idx = idx[weight <= low_thr]
        if low_idx.size:
            receiver = low_idx[rng.integers(low_idx.size)]
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            # cheap fallback to a completely random pair
            donor = rng.integers(n)
            receiver = rng.integers(n)
            if donor == receiver:
                continue

        # ---- move a small fraction of the donor's mass ----
        max_frac = 0.04                     # at most 4 % of the donor mass per move
        delta = a[donor] * max_frac * rng.random()
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta

        # evaluate new state
        conv_new = _conv_fft(a_new)
        new_score = float(conv_new.max())

        # acceptance test (Metropolis)
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_new
            conv = conv_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it} T={temperature:.5f} best={best_score:.6f}")

        # temperature cooling
        it += 1
        if it % 1000 == 0:
            temperature = max(temperature * 0.95, 1e-8)

            # occasional jitter to avoid dead‑ends
            if rng.random() < 0.2:
                jitter = rng.normal(scale=target_sum * 0.0005, size=n)
                a = np.maximum(a + jitter, 0.0)
                a = a * (target_sum / a.sum())
                conv = _conv_fft(a)
                cur_score = float(conv.max())

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The algorithm respects the given time budget and returns the best
    sequence found.
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # a tiny safety margin

    # --------------------------------------------------------------
    # problem size
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # warm‑start
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = np.array(globals()["height_sequence_1"], dtype=np.float64)
        init = _resize_sequence(init, n)
    else:
        init = _power_law_sequence(n, 0.55, target_sum)

    init = init * (target_sum / init.sum())

    # --------------------------------------------------------------
    # Phase 1 – short sub‑gradient polish (≈10 % of budget)
    # --------------------------------------------------------------
    phase1_deadline = time.time() + (deadline - time.time()) * 0.10
    a, best_val = _subgradient_max_descent(init, target_sum, phase1_deadline)

    # --------------------------------------------------------------
    # Phase 2 – simulated annealing (≈75 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase2_deadline = time.time() + (deadline - time.time()) * 0.75
        a, best_val = _simulated_annealing(a, target_sum, phase2_deadline, best_val)

    # --------------------------------------------------------------
    # Phase 3 – final sub‑gradient refinement (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a, best_val = _subgradient_max_descent(a, target_sum, deadline)

    # --------------------------------------------------------------
    # Trim trailing zeros and return
    # --------------------------------------------------------------
    nz = np.where(a > 1e-12)[0]
    if nz.size:
        a = a[: nz[-1] + 1]

    final_seq = a.tolist()

    # sanity check (prints the final benchmark value)
    try:
        val = evaluate_sequence(final_seq)          # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for warm‑starts in a subsequent call
    globals()["height_sequence_1"] = final_seq

    return final_seq
```

The optimiser respects all constraints, stays within the 1000 s limit,
