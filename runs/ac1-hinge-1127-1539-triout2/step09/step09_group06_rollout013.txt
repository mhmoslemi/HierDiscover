"""
Search for a low‑peak non‑negative step function.

The score to minimise is   2·n·max(a*a) / (sum(a)**2)
which is invariant under scaling of the whole vector.
We therefore fix the ℓ₁‑norm to 1 during the search.
The algorithm consists of four stages:

1. **Random warm‑starts** (including the previous best if available) of modest
   size (≈1500 entries) and an LP‑based refinement.  For a given vector f,
   the LP maximises  Σ g_i  subject to   f * g ≤ M   (convolution) and
   g ≥ 0, where M = max(f*f).  After solving we normalise g and blend
   it into f (f ← (1‑t)f + t g) – this provably never increases the peak.
   A few LP iterations dramatically flatten the shape.

2. **Upscaling** – the refined short vector is linearly interpolated to a
   handful of longer lengths (3000‑5000).  The best length w.r.t. the exact
   score is kept.

3. **Continuous optimisation** – the chosen vector is refined with the
   Adam‑based optimiser (mirror descent on the simplex) and a simple
   sub‑gradient method, both of which work on the τ‑soft‑max approximation
   of the ∞‑norm of the self‑convolution.

4. **Cheap stochastic hill‑climb** – small random mass transfers between two
   positions are accepted only if the exact score improves.

The whole pipeline respects a 1000 s budget and stores the final sequence in
the global name ``height_sequence_1`` for warm‑starts in later runs.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level utilities ---------------------------------------------------
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a∗a."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


def _build_convolution_matrix(f: np.ndarray) -> sparse.csr_matrix:
    """
    Build the matrix A such that (A @ g) = f * g  (full convolution).
    A has shape (2·n‑1, n) and contains only non‑negative entries,
    therefore it can be used directly in a linear program.
    """
    n = f.shape[0]
    # i repeats each row index n times, j tiles the column index n times
    i = np.repeat(np.arange(n), n)
    j = np.tile(np.arange(n), n)
    rows = i + j                      # convolution index
    cols = j
    data = np.repeat(f, n)            # coefficient f_i for each column j
    return sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


# ----------------------------------------------------------------------
# Adam optimiser (mirror descent on the simplex) -----------------------
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex (log‑softmax parametrisation).
    Returns the best vector (still having ℓ₁‑norm = sqrt(2·n)) and its
    *hard* peak value max(a*a).
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                     # ℓ₁‑norm we must keep

    # initialise logits from the feasible a0 (entropic parametrisation)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                         # logits (log‑softmax)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.45                                # aggressive start

    # temperature schedule – start relatively large for smoothness
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 5.0)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ---- recover a from logits (log‑softmax) ----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ---- convolution & gradient (τ‑soft‑max) ----
        g, peak = _softmax_gradient(a, tau)

        # keep the best hard peak seen so far
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ---- gradient w.r.t. logits (chain rule) ----
        mu = np.dot(a, g) / S                 # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)                 # shape (n,)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 6.0 * S:
            grad_p = grad_p * (6.0 * S / clip_val)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- schedules (very slow decay) ----
        tau *= 0.99995                         # keep smoothing longer
        lr *= 0.99995                          # gentle learning‑rate decay

        # occasional jitter to escape plateaus
        if it % 800 == 0:
            p += rng.normal(scale=0.006, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Sub‑gradient descent on the *hard* max (no smoothing)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple sub‑gradient descent that follows the entry of the convolution
    attaining the current maximum.  After each step the vector is projected
    back onto the simplex (non‑negative entries with fixed ℓ₁‑norm).
    Returns the best vector (ℓ₁‑norm = sqrt(2·n)) and its hard peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    a = a0.copy()
    best_a = a.copy()
    best_val = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1

        b = _full_convolution_fft(a)
        k = int(b.argmax())
        val = b[k]

        if val < best_val:
            best_val = val
            best_a = a.copy()

        # sub‑gradient of (a*a)_k
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        jdx = k - idx
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[jdx]

        step = 0.8 * S / (np.sqrt(it) + 1.0)

        a = a - step * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, S / n)
        else:
            a = a * (S / s)

        # occasional random jitter to avoid tiny plateaus
        if it % 500 == 0:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                frac = rng.uniform(0.02, 0.12)
                amount = a[i] * frac
                a[i] -= amount
                a[j] += amount
                a = a * (S / a.sum())

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple random initialisation (Dirichlet) ----------------------------
# ----------------------------------------------------------------------


def _random_initial_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random non‑negative vector of length n with ℓ₁‑norm = 1."""
    raw = rng.random(n)
    total = raw.sum()
    if total == 0.0:
        raw = np.ones(n)
        total = n
    return raw / total


# ----------------------------------------------------------------------
# LP refinement ---------------------------------------------------------
# ----------------------------------------------------------------------


def _lp_refine(f: np.ndarray,
               t: float,
               max_iters: int,
               deadline: float,
               rng: np.random.Generator) -> np.ndarray:
    """
    Perform a few LP‑based refinement steps on vector f (ℓ₁‑norm = 1).
    Each step solves:
        maximise   Σ g_i
        subject to (f * g)_k ≤ M   for all k
                   g_i ≥ 0
    where M = max(f * f).  The resulting g is normalised to sum 1 and
    blended back: f ← (1‑t)·f + t·g .
    Returns the refined vector.
    """
    n = f.shape[0]
    for it in range(max_iters):
        if time.time() > deadline - 20.0:   # keep a buffer for later phases
            break
        M = _max_convolution(f)

        # Build the convolution matrix A (sparse)
        A = _build_convolution_matrix(f)

        # Linear programme: maximise Σ g_i  ⇔ minimise −Σ g_i
        c = -np.ones(n)
        bounds = [(0, None)] * n
        b_ub = np.full(2 * n - 1, M)

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            print(f"[LP] iteration {it} failed: {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            break
        g_norm = g / sum_g

        # Blend and renormalise (sum stays = 1)
        f = (1.0 - t) * f + t * g_norm
        f = f / f.sum()

        M_new = _max_convolution(f)
        print(f"[LP] iter={it}  M={M:.6f} → {M_new:.6f}")

        # If progress stalls, shrink the blending factor
        if M_new >= M * 0.9999:
            t *= 0.7
        # stop early if t gets tiny
        if t < 1e-3:
            break

    return f

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found (as a plain Python list) and stores it in
    the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0          # small safety margin

    # ------------------------------------------------------------------
    # Warm‑start handling
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    n_lp = 1500                       # size used for the LP phase
    candidates = []

    if warm_seq is not None:
        candidates.append(_scale_sequence(warm_seq, n_lp))

    # a few random Dirichlet candidates for diversity
    for _ in range(4):
        candidates.append(_random_initial_vector(n_lp, rng))

    best_seq = None
    best_score = np.inf   # internal score = 2·n·max_conv (sum=1)

    # ------------------------------------------------------------------
    # Stage 1 – LP refinement + up‑scaling of each candidate
    # ------------------------------------------------------------------
    target_lengths = [3000, 3500, 4000, 4500, 5000]

    for cid, f0 in enumerate(candidates):
        if time.time() > end_time - 600:    # leave ample time for later phases
            break
        # Normalise to ℓ₁ = 1
        f = f0.copy()
        f = np.maximum(f, 0.0)
        f = f / f.sum()

        # LP refinement (few iterations, moderate blending)
        f = _lp_refine(f, t=0.5, max_iters=5,

            score = 2.0 * L * max_b               # exact internal score
            print(f"[candidate] id={cid} L={L:4d} score={score:.6f}")

            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_len = L

    # ------------------------------------------------------------------
    # Stage 2 – Adam optimisation on the best length found so far
    # ------------------------------------------------------------------
    if best_seq is None:                     # very unlikely, but be safe

    remaining = end_time - time.time() - 120.0
    if remaining > 30.0:

        a_start = best_seq * (S_target / best_seq.sum())

                                     time.time() + remaining,
                                     rng)
        # Bring back to ℓ₁ = 1 and compute exact score
        best_seq = a_opt / a_opt.sum()
        max_b = _max_convolution(best_seq)
        best_score = 2.0 * n * max_b
        print(f"[adam] post‑opt score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Sub‑gradient descent (final polishing)
    # ------------------------------------------------------------------
    remaining = end_time - time.time() - 5.0

        S_target = np.sqrt(2.0 * n)
        a_start = best_seq * (S_target / best_seq.sum())
        a_opt, peak = _subgradient_descent(a_start,
                                           time.time() + remaining,
                                           rng)
        best_seq = a_opt / a_opt.sum()
        max_b = _max_convolution(best_seq)
        best_score = 2.0 * n * max_b
        print(f"[subgrad] post‑opt score={best_score:.6f}")

    # ------------------------------------------------------------------
    while time.time() < end_time - 1.0:

        seq[i] -= amount

        if score + 1e-9 < best_score:

    # ------------------------------------------------------------------

    final_seq = best_seq.tolist()
