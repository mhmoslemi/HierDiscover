def build_conv_matrix(a):

def lp_refine(a, M, t=0.2, num_iter=10, deadline=...):
    n = len(a)
    S = np.sqrt(2.0 * n)
    a_curr = a.copy()
    for it in range(num_iter):
        if time.time() > deadline:
            break
        M = _max_convolution(a_curr) # compute current max
        # Build constraint matrix for a_curr
        A = _conv_matrix(a_curr) # returns csr
        b_ub = np.full(2*n-1, M)
        # liner prog
        res = scipy.optimize.linprog(c=-np.ones(n), A_ub=A, b_ub=b_ub,
                                   bounds=[(0, None)]*n,
                                   method='highs')
        if not res.success:
            print("[LP] iteration", it, "failed:", res.message)
            break
        g = res.x
        # Normalize sum to S
        sum_g = g.sum()
        if sum_g > 0:
            g = g * (S / sum_g)
        # Mix
        a_curr = (1 - t) * a_curr + t * g
    return a_curr

"""
Hybrid LP + Adam optimizer for minimizing the peak of the self‑convolution.

The algorithm works in several stages:

1. **Random exploration** – generate many feasible sequences of moderate
   length (≈2k–4k) using a collection of parametric families and keep the
   best (lowest max‑convolution) one.

2. **LP refinement** – for the current best sequence f, solve a linear
   programme
        maximize   Σ g_i
        subject to f * g ≤ M   (element‑wise) ,   g_i ≥ 0
   where M = max_k (f*f)_k.  The matrix of the convolution is built as a
   sparse Toeplitz matrix, so the LP is tractable for n≈2500.  The optimum
   g is then rescaled to the required ℓ₁‑norm (√(2n)) and mixed with f:
        f ← (1‑t)·f + t·g .
   A few mixing iterations quickly lower the peak.

3. **Adam optimisation** – treat the sequence as a soft‑max parametrisation
   on the simplex and minimise a τ‑soft‑max approximation of the peak.
   The temperature τ is slowly annealed and the optimiser runs with
   Adam‑style momentum.

4. **Sub‑gradient polishing** – pure hard‑max sub‑gradient descent with a
   decreasing step size gives a final fine‑tuning.

5. **Cheap hill‑climb** – random mass transfers between two positions for
   the last seconds.

All steps respect the 1000 s budget and the final sequence is stored in the
global variable ``height_sequence_1`` for warm‑starts in future runs.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, very fast)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a∗a."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Soft‑max gradient of the peak (used by Adam)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()
    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / exps.sum()
    # gradient: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                     # required ℓ₁‑norm

    # initialise logits from the feasible a0 (entropic parametrisation)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                         # logits (log‑softmax)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.45

    # temperature start
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 5.0)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1
        # ---- recover a from logits (log‑softmax) ----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ---- gradient of τ‑soft‑max ----
        g, peak = _softmax_gradient(a, tau)

        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ---- gradient w.r.t. logits (mirror‑descent) ----
        mu = np.dot(a, g) / S                 # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 6.0 * S:
            grad_p = grad_p * (6.0 * S / clip_val)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- schedules ----
        tau *= 0.99995
        lr *= 0.99995

        # occasional jitter
        if it % 800 == 0:
            p += rng.normal(scale=0.006, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Sub‑gradient descent on the hard peak
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    a = a0.copy()
    best_a = a.copy()
    best_val = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1
        b = _full_convolution_fft(a)
        k = int(b.argmax())
        val = b[k]

        if val < best_val:
            best_val = val
            best_a = a.copy()

        # sub‑gradient of (a*a)_k
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        jdx = k - idx
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[jdx]

        step = 0.8 * S / (np.sqrt(it) + 1.0)
        a = a - step * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, S / n)
        else:
            a = a * (S / s)

        if it % 500 == 0:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                frac = rng.uniform(0.02, 0.12)
                amount = a[i] * frac
                a[i] -= amount
                a[j] += amount
                a = a * (S / a.sum())

    return best_a, best_val

def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    raw = mix * ush + (1.0 - mix) * base
    return raw


def _init_cosine(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    raw = 1.0 - np.cos(np.pi * i / (n - 1))
    raw = raw * rng.uniform(0.5, 2.0)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * raw + (1.0 - mix) * base


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    return rng.random(n)


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.05, 0.9)
    return rng.gamma(shape, 1.0, size=n)


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """Generate a feasible vector of length n (ℓ₁‑norm = √(2n))."""
    S = np.sqrt(2.0 * n)

    if warm_seq is not None and rng.random() < 0.23:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
        return a

    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'gamma', 'uniform', 'dirichlet', 'ushape', 'cosine'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'cosine':
        raw = _init_cosine(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Convolution matrix (used by the LP step)
# ----------------------------------------------------------------------
def _conv_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Build the sparse matrix A such that A @ g = a * g (full linear convolution).
    Shape of A is (2n‑1) × n.
    """
    n = a.shape[0]
    i = np.arange(n)[:, None]          # (n,1)
    j = np.arange(n)[None, :]          # (1,n)
    rows = (i + j).ravel()
    cols = np.broadcast_to(j, (n, n)).ravel()
    data = np.repeat(a, n)
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------
def _lp_refine(a0: np.ndarray,
               deadline: float,
               rng: np.random.Generator,
               mix_factor: float = 0.15,
               max_iter: int = 7) -> np.ndarray:
    """
    Repeatedly solve a linear programme that keeps the current peak M as a hard bound.
    The optimiser returns a new sequence that is mixed into the current one.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)
    a = a0.copy()
    iter_done = 0

    while time.time() < deadline and iter_done < max_iter:
        iter_done += 1
        M = _max_convolution(a)

        # Build the convolution matrix for the *current* a
        A = _conv_matrix(a)
        b_ub = np.full(2 * n - 1, M)

        # Linear programme: maximise Σ g_i  ⇔ minimise −Σ g_i
        c = -np.ones(n)
        res = scipy.optimize.linprog(c,
                                     A_ub=A,
                                     b_ub=b_ub,
                                     bounds=[(0, None)] * n,
                                     method='highs',
                                     options={'presolve': True, 'time_limit': max(10, int(deadline - time.time()))})
        if not res.success:
            # If the LP fails (numerical issues) fall back to a tiny random jitter.
            print(f"[LP] iteration {iter_done} failed: {res.message}")
            jitter = rng.normal(scale=0.01 * S / n, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (S / a.sum())
            continue

        g = res.x
        sum_g = g.sum()
        if sum_g > 0.0:
            g = g * (S / sum_g)

        # Mix a little bit of the LP solution into the current sequence
        a = (1.0 - mix_factor) * a + mix_factor * g

    return a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (list of floats) and stores it in the global variable
    ``height_sequence_1`` for future runs.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    end_time = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Warm‑start handling
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # ------------------------------------------------------------------
    # Phase 1 – fast random exploration (≈15 % of the budget)
    # ------------------------------------------------------------------
    explore_deadline = start_time + 0.15 * budget_s
    best_seq = None
    best_score = np.inf
    attempts = 0
    while time.time() < explore_deadline:
        attempts += 1
        if rng.random() < 0.70:
            n = int(rng.integers(1800, 6000))
        else:
            n = int(rng.integers(6000, 12000))
        a0 = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a0)
        if peak < best_score:
            best_score = peak
            best_seq = a0.copy()
            print(f"[explore] attempt={attempts:4d} n={n:5d}  new best={best_score:.6f}")

    if best_seq is None:                               # safeguard
        n = 3000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = _max_convolution(best_seq)

    # ------------------------------------------------------------------
    # Phase 2 – LP refinement (≈20 % of the budget)
    # ------------------------------------------------------------------
    lp_deadline = time.time() + 0.20 * budget_s
    a_lp = _lp_refine(best_seq, lp_deadline, rng,
                      mix_factor=0.12, max_iter=8)
    lp_score = _max_convolution(a_lp)
    if lp_score < best_score:
        best_score = lp_score
        best_seq = a_lp.copy()
        print(f"[LP] refined max={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – up‑scale to a slightly larger length (no optimisation)
    # ------------------------------------------------------------------
    # Larger length often reduces the peak after re‑normalisation.
    cur_n = len(best_seq)
    target_n = int(min(5000, cur_n * 1.6))           # stay within a tractable range
    if target_n != cur_n:
        a_scaled = _scale_sequence(best_seq, target_n)
        a_scaled = np.clip(a_scaled, 0.0, None)
        a_scaled = a_scaled * (np.sqrt(2.0 * target_n) / a_scaled.sum())
        cur_seq = a_scaled
        cur_score = _max_convolution(cur_seq)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur_seq.copy()
            cur_n = target_n
            print(f"[scale] up‑scaled to n={cur_n}  max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Adam optimisation (≈30 % of the budget)
    # ------------------------------------------------------------------
    remaining = end_time - time.time() - 120.0   # reserve time for sub‑grad
    if remaining > 15.0:
        adam_deadline = time.time() + remaining
        # Run a few independent Adam restarts to diversify
        restarts = max(1, int(remaining // 45))
        per_run = remaining / restarts
        cur_a = best_seq.copy()
        cur_best = best_score
        for r in range(restarts):
            if time.time() > end_time - 130.0:
                break
            # small random perturbation before each restart
            noise = rng.normal(scale=0.02 * np.sqrt(2.0 * cur_n) / cur_n, size=cur_n)
            a_start = np.maximum(cur_a + noise, 0.0)
            a_start = a_start * (np.sqrt(2.0 * cur_n) / a_start.sum())
            a_opt, peak = _adam_optimize(a_start,
                                          time.time() + per_run,
                                          rng)
            if peak < cur_best:
                cur_best = peak
                cur_a = a_opt.copy()
                print(f"[adam] restart={r} new best max={cur_best:.6f}")
        best_seq = cur_a
        best_score = cur_best

    # ------------------------------------------------------------------
    # Phase 5 – Sub‑gradient polishing (≈15 % of the budget)
    # ------------------------------------------------------------------
    remaining = end_time - time.time() - 20.0
    if remaining > 10.0:
        a_opt, peak = _subgradient_descent(best_seq,
                                           time.time() + remaining,
                                           rng)
        if peak < best_score:
            best_score = peak
            best_seq = a_opt.copy()
            print(f"[subgrad] final max={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – cheap random hill‑climb (last few seconds)
    # ------------------------------------------------------------------
    while time.time() < end_time - 1.0:
        a = best_seq.copy()
        n = a.shape[0]
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        frac = rng.uniform(0.02, 0.12)
        amount = a[i] * frac
        a[i] -= amount
        a[j] += amount
        peak = _max_convolution(a)
        if peak + 1e-9 < best_score:
            best_score = peak
            best_seq = a.copy()
            print(f"[hill] improved max={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f}  internal max={best_score:.6f}  length={len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
