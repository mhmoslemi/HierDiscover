"""
Propose a step‑function minimizing ``evaluate_sequence``.

Algorithm (≈1000 s total)
-------------------------
1. **Exploration** – generate many random candidates of varying length.
   Families include Dirichlet‑like, Gamma, power‑law, exponential,
   sparse rectangular bumps, Gaussian, half‑sine and a new *geometric*
   (r^i) family.  Keep the best ~10 candidates.

2. **Multi‑scale refinement** – for each shortlisted candidate
   upscale to a large target length (≈80 000) and run a full‑gradient
   Adam optimiser on the simplex (log‑softmax parametrisation) with
   an exact sub‑gradient of the hard max‑convolution.  Hyper‑parameters
   (learning‑rate, τ‑decay, jitter) are jittered per run.

3. **Polishing** – a few steps of projected gradient descent (PGD)
   with the exact hard‑max sub‑gradient, followed by a short simulated
   annealing walk and a greedy hill‑climber.

4. **Verification** – the best sequence is evaluated with the supplied
   ``evaluate_sequence`` and stored in the global ``height_sequence_1``.

The code respects the 1000 s budget (5 s safety margin) and prints
progress information.
"""

import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Peak value of a*a (full convolution)."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Exact sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------
def _max_gradient(a: np.ndarray, tau: float):
    """
    Return (grad, max_b) where ``grad`` is a sub‑gradient of
    max(a*a) w.r.t. ``a`` and ``max_b`` the current peak.
    With very small ``tau`` a hard‑max (active‑set) gradient is used.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()
    if tau < 1e-12:
        eps = max_b * 1e-9 + 1e-12
        mask = (b >= max_b - eps).astype(float)
        s = mask.sum()
        if s == 0:
            mask = np.ones_like(b) / b.size
        else:
            mask /= s
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / exps.sum()
    a_rev = a[::-1]
    conv_len = mask.size + a.size - 1
    f_len = _next_pow2(conv_len)

    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)

    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    grad = 2.0 * conv[a.size - 1:a.size - 1 + a.size]
    return grad, max_b


# ----------------------------------------------------------------------
# Projection onto the ℓ₁‑simplex (sum = S, non‑negative)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of ``v`` onto {x ≥ 0, sum x = S}."""
    if S < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   lr0: float = 0.5,
                   tau_decay: float = 0.9995,
                   lr_decay: float = 0.99985,
                   jitter_every: int = 400):
    """
    Minimise the peak of the self‑convolution under the ℓ₁‑norm
    constraint (sum = √(2 n)).  Returns the best vector seen and its
    peak value.
    """
    n = a0.shape[0]
    S = math.sqrt(2.0 * n)                     # required ℓ₁‑norm

    # initialise logits from the feasible a0 (log‑softmax)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                           # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr0

    # initialise temperature for τ‑soft‑max
    init_max = _full_convolution_fft(a0).max()
    tau = max(init_max, 0.5)

    best_a = a0.copy()
    best_peak = _max_convolution(a0)
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- recover a from logits (log‑softmax) -----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)               # shift for stability
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ----- gradient of (soft‑)max convolution -----
        g, peak = _max_gradient(a, tau)

        # keep best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ----- gradient w.r.t. logits (chain rule) -----
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- annealing -----
        tau *= tau_decay
        lr *= lr_decay

        # occasional jitter to escape plateaus
        if jitter_every and (it % jitter_every) == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Simple projected‑gradient descent (hard‑max sub‑gradient)
# ----------------------------------------------------------------------
def _pgd_optimize(a0: np.ndarray,
                  deadline: float,
                  rng: np.random.Generator,
                  init_lr: float = 0.08):
    """
    A few hundred cheap PGD steps using the exact hard‑max gradient.
    Returns the best vector seen and its peak.
    """
    n = a0.shape[0]
    S = math.sqrt(2.0 * n)
    a = a0.copy()
    best_a = a.copy()
    best_peak = _max_convolution(a)
    lr = init_lr
    it = 0
    while time.time() < deadline:
        it += 1
        grad, peak = _max_gradient(a, tau=0.0)   # hard max
        a = a - lr * grad
        a = _project_simplex(a, S)
        new_peak = _max_convolution(a)
        if new_peak < best_peak:
            best_peak = new_peak
            best_a = a.copy()
        # decay learning‑rate slowly
        lr = max(lr * 0.9995, 1e-5)
    return best_a, best_peak

# Simulated annealing + hill climbing (as in the reference)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Mass‑transfer walk on the simplex with exponential acceptance."""
    n = len(seq)
    S = math.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score
    temperature = 0.15 * cur_score

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] == 0.0:
            continue
        delta = rng.uniform(0.0, 0.3) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())
        new_score = _max_convolution(new)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

        temperature *= 0.99995
        if temperature < 1e-6:
            temperature = 1e-6

    return best, best_score


def _hill_climb(seq: np.ndarray,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Strict downhill hill‑climber."""
    n = len(seq)
    S = math.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] < 1e-12:
            continue
        delta = rng.uniform(0.0, 0.15) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())
        new_score = _max_convolution(new)
        if new_score < cur_score:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score
    return best, best_score

def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.2, 5.0)
    return rng.gamma(alpha, 1.0, size=n)


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.1, 0.8)
    return rng.gamma(shape, 1.0, size=n)


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_left_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    p = rng.uniform(0.3, 3.0)
    offset = rng.uniform(0.0, 10.0)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * base + (1.0 - mix) * (idx + offset) ** (-p)


def _init_left_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    lam = rng.uniform(0.001, 0.02)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * base + (1.0 - mix) * np.exp(-lam * idx)


def _init_sparse_bumps(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = np.full(n, rng.uniform(0.0, 0.15))
    nbumps = rng.integers(1, max(2, n // 250))
    for _ in range(nbumps):
        center = rng.integers(0, n)
        width = rng.integers(3, max(5, n // 100))
        height = rng.uniform(0.8, 4.0)
        lo = max(0, center - width // 2)
        hi = min(n, lo + width)
        raw[lo:hi] = np.maximum(raw[lo:hi], height)
    return raw


def _init_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    mu = (n - 1) / 2.0
    sigma = rng.uniform(0.05, 0.30) * n
    x = np.arange(n)
    return np.exp(-0.5 * ((x - mu) / sigma) ** 2)


def _init_half_sine(n: int, rng: np.random.Generator) -> np.ndarray:
    x = np.arange(n)
    raw = np.sin(np.pi * x / (n - 1))
    raw = raw * rng.uniform(0.5, 2.0) + rng.uniform(0.0, 0.2)
    return raw


def _init_geometric(n: int, rng: np.random.Generator) -> np.ndarray:
    """Geometric decay r**i (0<r<1) possibly plus a small uniform offset."""
    r = rng.uniform(0.90, 0.9995)
    raw = r ** np.arange(n, dtype=float)
    # optional uniform background
    mix = rng.uniform(0.0, 0.4)
    raw = mix * np.mean(raw) + (1 - mix) * raw
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Occasionally warm‑start from the global best; otherwise draw from
    a random family.
    """
    S = math.sqrt(2.0 * n)

    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    style = rng.choice(['dirichlet', 'gamma', 'uniform',
                        'left_powerlaw', 'left_exponential',
                        'sparse', 'gaussian', 'half_sine', 'geometric'])
    if style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'left_powerlaw':
        raw = _init_left_powerlaw(n, rng)
    elif style == 'left_exponential':
        raw = _init_left_exponential(n, rng)
    elif style == 'sparse':
        raw = _init_sparse_bumps(n, rng)
    elif style == 'gaussian':
        raw = _init_gaussian(n, rng)
    elif style == 'half_sine':
        raw = _init_half_sine(n, rng)
    else:  # geometric
        raw = _init_geometric(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a

def propose_candidate(seed: int = 42,

    global variable ``height_sequence_1``.
    """

    if end_time <= time.time():
        raise RuntimeError("Budget too small after safety margin.")

    # Warm‑start from a previously stored best sequence, if any
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # Phase 1 – fast random exploration (≈20 % of budget)
    # ------------------------------------------------------------------

        attempt += 1

        else:
            n = int(rng.integers(8000, 40000))

        peak = _max_convolution(a0)

            best_n = n
            print(f"[explore] attempt={attempt:5d} n={n:5d}  new best={best_score:.6f}")

        # keep top‑k
        if len(top_candidates) < top_k:
            top_candidates.append((peak, a0.copy()))
            top_candidates.sort(key=lambda x: x[0])
        elif peak < top_candidates[-1][0]:
            top_candidates[-1] = (peak, a0.copy())
            top_candidates.sort(key=lambda x: x[0])

    # Safety net
    if best_seq is None:
        n = 4000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n)
        best_score = _max_convolution(best_seq)
        best_n = n
        top_candidates = [(best_score, best_seq.copy())]

    # ------------------------------------------------------------------
    # Phase 2 – multi‑scale Adam refinement (≈70 % of budget)
    # ------------------------------------------------------------------
    max_len = 80000                     # final target length
    remaining_for_refine = end_time - time.time() - 30.0   # keep a little headroom
    num_candidates = len(top_candidates)
    # allocate roughly equal time slices for each candidate
    slice_per_candidate = remaining_for_refine / max(num_candidates, 1)

    for idx, (init_peak, init_seq) in enumerate(top_candidates):
        if time.time() + slice_per_candidate > end_time - 30.0:
            break

        # Scale to target length
        a_scaled = _scale_sequence(init_seq, max_len)
        a_scaled = np.clip(a_scaled, 0.0, None)
        a_scaled = a_scaled * (math.sqrt(2.0 * max_len) / a_scaled.sum())

        # Randomise a few hyper‑parameters for diversity
        lr0 = rng.choice([0.48, 0.55, 0.62, 0.70])
        tau_decay = rng.choice([0.9993, 0.9995, 0.9997])
        jitter_every = rng.integers(300, 600)

        # Adam optimisation
        refined, peak = _adam_optimize(
            a_scaled,
            deadline=time.time() + slice_per_candidate * 0.55,
            rng=rng,
            lr0=lr0,
            tau_decay=tau_decay,
            lr_decay=0.99985,
            jitter_every=jitter_every,
        )

        # Quick PGD polishing
        pgd_deadline = time.time() + slice_per_candidate * 0.25
        refined, pgd_peak = _pgd_optimize(
            refined,
            deadline=pgd_deadline,
            rng=rng,
            init_lr=0.04,
        )
        if pgd_peak < peak:
            peak = pgd_peak

        # Keep the best overall
        if peak < best_score:
            best_score = peak
            best_seq = refined.copy()
            best_n = max_len
            print(f"[refine] cand#{idx+1} len={max_len:5d}  improved score={best_score:.6f}")

        # Give a tiny chance to continue refinement at an intermediate length
        # (helps escaping local minima for very large n)
        if rng.random() < 0.12 and time.time() < end_time - 30.0:
            mid_len = int(max_len * 0.6)
            a_mid = _scale_sequence(refined, mid_len)
            a_mid = a_mid * (math.sqrt(2.0 * mid_len) / a_mid.sum())
            refined_mid, _ = _adam_optimize(
                a_mid,
                deadline=time.time() + slice_per_candidate * 0.20,
                rng=rng,
                lr0=lr0 * 0.9,
                tau_decay=tau_decay,
                lr_decay=0.99985,
                jitter_every=jitter_every,
            )
            # Scale back up to max_len to keep the improvement if any
            refined_mid = _scale_sequence(refined_mid, max_len)
            refined_mid = refined_mid * (math.sqrt(2.0 * max_len) / refined_mid.sum())
            peak_mid = _max_convolution(refined_mid)
            if peak_mid < best_score:
                best_score = peak_mid
                best_seq = refined_mid.copy()
                print(f"[mid‑refine] cand#{idx+1} -> score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – stochastic fine‑tuning (≈10 % of budget)
    # ------------------------------------------------------------------
    remaining = end_time - time.time()
    if remaining > 20.0:
        best_seq, best_score = _anneal(best_seq,
                                       deadline=time.time() + 0.6 * remaining,
                                       rng=rng)
        remaining = end_time - time.time()
        if remaining > 5.0:
            best_seq, best_score = _hill_climb(best_seq,
                                               deadline=time.time() + remaining,
                                               rng=rng)

    # ------------------------------------------------------------------
    # Final verification and return
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  internal max = {best_score:.6f}"
          f"  length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
