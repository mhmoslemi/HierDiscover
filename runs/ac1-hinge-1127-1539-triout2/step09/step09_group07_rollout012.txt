   def _lp_improve(a0, S, deadline, rng, t=0.05, epsilon_rel=1e-8, max_iter=2000):
       best_a = a0.copy()
       best_score = _max_convolution(a0)
       a = a0.copy()
       while time.time() < deadline and max_iter > 0:
           max_iter -= 1
           conv = _full_convolution_fft(a)
           max_val = conv.max()
           # Update best
           if max_val < best_score:
               best_score = max_val
               best_a = a.copy()
           # active indices
           eps = max_val * 1e-8 + 1e-12
           active = np.where(conv >= max_val - eps)[0]
           # maybe limit active to at most 200 randomly
           if len(active) > 200:
               active = rng.choice(active, size=200, replace=False)
           # Build constraints matrix A_ub
           n = len(a)
           m = len(active)
           rows = []
           cols = []
           data = []
           for row_idx, k in enumerate(active):
               lo = max(0, k - (n - 1))
               hi = min(k, n - 1)
               # j from lo..hi
               # for each j: coeff = a[k - j]

               j_vals = np.arange(lo, hi + 1)

               rows.extend([row_idx] * len(j_vals))
               cols.extend(j_vals.tolist())
               data.extend(a[i_vals].tolist())
           A = csr_matrix((data, (rows, cols)), shape=(m, n))
           # Solve LP
           c = -np.ones(n)
           b_ub = np.full(m, max_val)
           bounds = [(0, None)] * n
           # Use tiny time limit like 5 seconds; but we can not specify; SciPy will use default.
           res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
           if not res.success:

           sum_g0 = g0.sum()
           if sum_g0 <= 0:
               break
           g = g0 * (S / sum_g0)
           # Mix
           a = (1 - t) * a + t * g
           # Optionally jitter occasionally

"""
Propose candidate step‑function sequence via a two‑stage optimisation:

1. **Linear‑programming refinement** – starting from a random feasible
   vector `f` (ℓ₁‑norm = √(2·n)) we iteratively solve a tiny LP that
   maximises the total mass under the current peak‐constraint
   `f*f ≤ M`.  Only the constraints that are (almost) active
   (indices where the convolution attains its maximum) are kept,
   which keeps the LP tiny (O(|active|·n) non‑zeros).  The optimal
   `g` is rescaled to the required norm and mixed with the current
   iterate: `f ← (1−t)·f + t·g`.  The peak never increases and
   usually drops.

2. **Adam polishing + hill‑climbing** – after a few seconds of LP steps
   the current best iterate is handed to the soft‑max Adam optimiser
   from the reference implementation, followed by a cheap stochastic
   hill‑climber.  Multiple random restarts are performed until the
   global time budget (≈ 1000 s) expires.

The final sequence is stored in the global variable ``height_sequence_1``
and returned as a plain list of non‑negative numbers.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Convolution utilities (FFT based – fast for all n we consider)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Peak value of a*a (full convolution)."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Random families for the exploration phase (including new symmetric ones)
# ----------------------------------------------------------------------
def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.2, 5.0)
    return rng.gamma(alpha, 1.0, size=n)


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.1, 0.8)
    return rng.gamma(shape, 1.0, size=n)


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_left_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    p = rng.uniform(0.3, 3.0)
    offset = rng.uniform(0.0, 10.0)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * base + (1.0 - mix) * (idx + offset) ** (-p)


def _init_left_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    lam = rng.uniform(0.001, 0.02)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * base + (1.0 - mix) * np.exp(-lam * idx)


def _init_sparse_bumps(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = np.full(n, rng.uniform(0.0, 0.15))
    nbumps = rng.integers(1, max(2, n // 250))
    for _ in range(nbumps):
        center = rng.integers(0, n)
        width = rng.integers(3, max(5, n // 100))
        height = rng.uniform(0.8, 4.0)
        lo = max(0, center - width // 2)
        hi = min(n, lo + width)
        raw[lo:hi] = np.maximum(raw[lo:hi], height)
    return raw


def _init_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    mu = (n - 1) / 2.0
    sigma = rng.uniform(0.05, 0.30) * n
    x = np.arange(n)
    return np.exp(-0.5 * ((x - mu) / sigma) ** 2)


def _init_half_sine(n: int, rng: np.random.Generator) -> np.ndarray:
    x = np.arange(n)
    raw = np.sin(np.pi * x / (n - 1))
    raw = raw * rng.uniform(0.5, 2.0) + rng.uniform(0.0, 0.2)
    return raw


def _init_symmetric_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """Decreasing power‑law from both ends towards the centre."""
    idx = np.arange(n)
    dist = np.minimum(idx, n - 1 - idx) + 1
    p = rng.uniform(0.3, 3.0)
    offset = rng.uniform(0.0, 10.0)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * base + (1.0 - mix) * (dist + offset) ** (-p)


def _init_two_end_bumps(n: int, rng: np.random.Generator) -> np.ndarray:
    """Two rectangular bumps at the two ends."""
    raw = np.full(n, rng.uniform(0.0, 0.05))
    w = rng.integers(max(3, n // 200), max(10, n // 50))
    h1 = rng.uniform(0.8, 3.0)
    h2 = rng.uniform(0.8, 3.0)
    raw[:w] = np.maximum(raw[:w], h1)
    raw[-w:] = np.maximum(raw[-w:], h2)
    return raw


def _initial_vector(n: int, rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """Produce a feasible start vector of length n (ℓ₁ = √(2n))."""
    S = math.sqrt(2.0 * n)

    # occasional warm start
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    style = rng.choice(['dirichlet', 'gamma', 'uniform',
                        'left_powerlaw', 'left_exponential',
                        'sparse', 'gaussian', 'half_sine',
                        'sym_powerlaw', 'two_end'])
    if style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'left_powerlaw':
        raw = _init_left_powerlaw(n, rng)
    elif style == 'left_exponential':
        raw = _init_left_exponential(n, rng)
    elif style == 'sparse':
        raw = _init_sparse_bumps(n, rng)
    elif style == 'gaussian':
        raw = _init_gaussian(n, rng)
    elif style == 'half_sine':
        raw = _init_half_sine(n, rng)
    elif style == 'sym_powerlaw':
        raw = _init_symmetric_powerlaw(n, rng)
    else:  # two_end
        raw = _init_two_end_bumps(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    return raw * (S / total)


# ----------------------------------------------------------------------
# Adam optimiser (same as in the reference, kept for polishing)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())


def _max_gradient(a: np.ndarray, tau: float):
    """Sub‑gradient of max convolution (hard‑max for small τ)."""
    b = _full_convolution_fft(a)
    max_b = b.max()
    if tau < 1e-12:
        eps = max_b * 1e-9 + 1e-12
        mask = (b >= max_b - eps).astype(float)
        s = mask.sum()
        mask = mask / s if s > 0 else np.ones_like(b) / b.size
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / exps.sum()
    a_rev = a[::-1]
    conv_len = mask.size + a.size - 1
    f_len = _next_pow2(conv_len)
    grad = np.fft.irfft(np.fft.rfft(mask, n=f_len) *
                        np.fft.rfft(a_rev, n=f_len), n=f_len)[:conv_len]
    grad = 2.0 * grad[a.size - 1:a.size - 1 + a.size]
    return grad, max_b


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   lr0: float = 0.5,
                   tau_decay: float = 0.9995,
                   lr_decay: float = 0.99985,
                   jitter_every: int = 400) -> tuple[np.ndarray, float]:
    """Adam on the simplex (log‑softmax) for a fixed length."""
    n = a0.shape[0]
    S = math.sqrt(2.0 * n)                     # ℓ₁‑norm target

    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                           # logits

    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr0
    tau = max(_full_convolution_fft(a0).max(), 0.5)

    best_a = a0.copy()
    best_peak = _max_convolution(a0)
    it = 0

    while time.time() < deadline:
        it += 1
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        g, peak = _max_gradient(a, tau)

        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        tau *= tau_decay
        lr *= lr_decay

        if jitter_every and (it % jitter_every) == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Cheap hill‑climber (same as reference)
# ----------------------------------------------------------------------
def _hill_climb(seq: np.ndarray,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Strict downhill mass‑transfer walk on the simplex."""
    n = len(seq)
    S = math.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] < 1e-12:
            continue
        delta = rng.uniform(0.0, 0.15) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())
        new_score = _max_convolution(new)
        if new_score < cur_score:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score
    return best, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement based on the active‑set idea
# ----------------------------------------------------------------------
def _lp_refine(a0: np.ndarray,
               deadline: float,
               rng: np.random.Generator,
               mix_factor: float = 0.12,
               max_active: int = 200) -> tuple[np.ndarray, float]:
    """
    Alternate LP improvement (active constraints only) and mixing.
    Returns the best vector seen and its peak value.
    """
    n = a0.shape[0]
    S = math.sqrt(2.0 * n)

    a = a0.copy()
    best_a = a.copy()
    best_peak = _max_convolution(a)

    # Pre‑compute for every convolution index k the range of g‑indices that
    # actually appear (this does not depend on the values of a).
    lo_for_k = np.maximum(0, np.arange(2 * n - 1) - (n - 1))
    hi_for_k = np.minimum(np.arange(2 * n - 1), n - 1)

    while time.time() < deadline:
        conv = _full_convolution_fft(a)
        M = conv.max()

        if M < best_peak:
            best_peak = M
            best_a = a.copy()

        # ---------- active set ----------
        eps = M * 1e-8 + 1e-12
        active_idx = np.where(conv >= M - eps)[0]
        if active_idx.size > max_active:
            active_idx = rng.choice(active_idx, size=max_active, replace=False)

        m = active_idx.size
        # Build sparse matrix A (m × n) where A[k, j] = a[k - j] if valid
        rows = []
        cols = []
        data = []
        for ridx, k in enumerate(active_idx):
            lo = lo_for_k[k]
            hi = hi_for_k[k]
            js = np.arange(lo, hi + 1, dtype=int)
            is_ = k - js
            rows.extend([ridx] * js.size)
            cols.extend(js.tolist())
            data.extend(a[is_].tolist())
        A = csr_matrix((data, (rows, cols)), shape=(m, n))

        # ---------- solve tiny LP ----------
        c = -np.ones(n)                                 # maximise sum(g)  ⇔ minimise -∑g
        bounds = [(0, None)] * n
        b_ub = np.full(m, M)

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})

        if not res.success:
            # LP failed – abort refinement for this length
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            break
        g = g0 * (S / sum_g0)

        # ---------- mixing ----------
        a = (1.0 - mix_factor) * a + mix_factor * g

        # occasional jitter to escape plateaus
        if rng.random() < 0.03:
            a = np.maximum(a + rng.normal(scale=0.005, size=n), 0.0)
            a = a * (S / a.sum())

    return best_a, best_peak

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The algorithm mixes an active‑set LP improvement with Adam polishing
    and a final hill‑climber.  The best construction found within the
    time budget is stored globally as ``height_sequence_1`` and returned.
    """
    rng = np.random.default_rng(seed)
    # keep a small safety margin for the harness
    safety_margin = 5.0
    end_time = time.time() + budget_s - safety_margin
    if end_time <= time.time():
        raise RuntimeError("Budget too small after safety margin.")

    # ------------------------------------------------------------------
    # Warm‑start from a previously stored construction, if any
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    best_seq = None
    best_score = np.inf
    iteration = 0

    # ------------------------------------------------------------------
    # Main loop – keep trying random initialisations until time runs out
    # ------------------------------------------------------------------
    while time.time() < end_time - 5.0:      # keep a few seconds for final reports
        iteration += 1

        # choose a length (biased towards a few‑thousand entries)
        if rng.random() < 0.6:
            n = int(rng.integers(3000, 7000))
        else:
            n = int(rng.integers(7000, 15000))

        a0 = _initial_vector(n, rng, warm_seq)

        # ---- Phase A : active‑set LP refinement ----
        lp_deadline = time.time() + min(20.0, (end_time - time.time()) * 0.25)
        a_lp, peak_lp = _lp_refine(a0, lp_deadline, rng,
                                   mix_factor=0.12,
                                   max_active=180)

        # ---- Phase B : Adam polishing (short) ----
        adam_deadline = time.time() + min(15.0, (end_time - time.time()) * 0.20)
        a_adam, peak_adam = _adam_optimize(a_lp, adam_deadline, rng,
                                           lr0=0.48,
                                           tau_decay=0.9994,
                                           lr_decay=0.99985,
                                           jitter_every=300)

        # ---- Phase C : cheap hill‑climber ----
        hill_deadline = time.time() + min(8.0, (end_time - time.time()) * 0.15)
        a_final, peak_final = _hill_climb(a_adam, hill_deadline, rng)

        # ---- Record best ----
        if peak_final < best_score:
            best_score = peak_final
            best_seq = a_final.copy()
            print(f"[iter {iteration:4d}] n={n:5d}  new best = {best_score:.6f}")

        # Optional early exit if we already beat the target
        if best_score < 1.503:
            print("[target reached] breaking early.")
            break

    # ------------------------------------------------------------------
    # Final sanity check via the supplied black‑box evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  internal max = {best_score:.6f}"
          f"  length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
