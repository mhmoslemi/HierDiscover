def _build_conv_matrix(f):
    n = len(f)
    m = 2*n - 1
    data = []
    rows = []
    cols = []
    for i in range(n):
        # for column i: rows i..i+n-1
        rows.extend(range(i, i+n))
        cols.extend([i]*n)
        data.extend([f[i]]*n)
    A = csr_matrix((data, (rows, cols)), shape=(m, n))

"""
Improved step‑function optimizer for the “max‑convolution” objective.

Main ideas
----------
1. **Broad random exploration** – generate many candidates from several
   parametric families (power‑law, exponential, linear, U‑shape,
   sinusoid, Dirichlet, uniform) at lengths between 3 000 and 20 000.
   Keep the best one seen according to the hard max‑convolution.

2. **Projected sub‑gradient descent** – on the current best candidate we
   run a classic sub‑gradient method for the objective
   `f(a)=max_k (a∗a)[k]`.  The step size is the Polyak rule
   ` (f(a)−best)/(‖g‖²) ` and after each step the vector is projected
   onto the simplex `{a_i≥0, Σ a_i = √(2n)}` using the O(n) Duchi‑Shalev‑Shwartz
   algorithm.

3. **Linear‑programming refinement** – given a feasible `a`, the LP

       maximise   Σ g_i
       s.t.      (a∗g)[k] ≤ M    (k = 0..2n‑2)
                 g_i ≥ 0

   with `M = max_k (a∗a)[k]` finds a direction `g` that can lower the
   worst convolution value.  After solving we renormalise `g` to the
   simplex and replace `a ← (1‑t) a + t g` (t≈0.5).  Repeating a few
   iterations rapidly improves the bound; the matrix of the
   convolution constraints is built as a sparse Toeplitz matrix.

4. **Further sub‑gradient polishing** – after the LP phase we run a
   longer projected sub‑gradient descent to fine‑tune the solution.

5. **Simulated‑annealing hill climbing** – a cheap stochastic
   mass‑transfer loop that occasionally accepts worsening moves, helping
   escape tiny plateaus.

All phases respect the 1000 s budget.  The final sequence is stored in
the global variable ``height_sequence_1`` and returned as a plain list.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, projection, scaling)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a∗a."""
    return _full_convolution_fft(a).max()


def _project_onto_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex
        { w | w_i ≥ 0 , Σ w_i = S }.
    Implements the O(n) algorithm of Duchi et al.
    """
    if S < 0:
        raise ValueError("Simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]

# Parametric families for random initialization

def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    raw = mix * edge + (1.0 - mix) * np.ones(n)
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    raw = mix * exp_part + (1.0 - mix) * np.ones(n)
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    raw = mix * ush + (1.0 - mix) * np.ones(n)
    return raw


def _init_sine(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    vals = np.sin(np.pi * (i + 1) / (n + 1))
    mix = rng.uniform(0.0, 1.0)
    raw = mix * vals + (1.0 - mix) * np.ones(n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    return rng.random(n)


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Occasionally warm‑starts from the globally best sequence and otherwise
    draws from a random parametric family.
    """
    S = np.sqrt(2.0 * n)

    # Warm start with probability ≈0.2
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        tot = a.sum()
        if tot == 0.0:
            a = np.full(n, 1.0)
            tot = n
        a = a * (S / tot)
        return a

    style = rng.choice(
        ['powerlaw', 'exponential', 'linear', 'ushape',
         'sine', 'uniform', 'dirichlet'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'sine':
        raw = _init_sine(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    tot = raw.sum()
    if tot == 0.0:
        raw = np.full(n, 1.0)
        tot = n
    a = raw * (S / tot)
    return a

def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         global_best: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=max(a∗a).
    Uses Polyak step size and occasional jitter.
    Returns best vector and its max‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1
        # compute sub‑gradient analytically (same as in the reference code)
        b = _full_convolution_fft(a)
        max_val = b.max()
        mask = (b == max_val).astype(float)

        # gradient = 2 * (mask ∗ reverse(a))
        a_rev = a[::-1]
        conv_len = len(mask) + n - 1
        f_len = _next_pow2(conv_len)
        Fmask = np.fft.rfft(mask, n=f_len)
        Fa_rev = np.fft.rfft(a_rev, n=f_len)
        conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
        start = n - 1
        grad = 2.0 * conv[start:start + n]

        # Polyak step size (target = best_val)
        norm_sq = np.dot(grad, grad)
        if norm_sq > 0:
            step = (max_val - best_val) / (norm_sq + 1e-12)
            step = np.clip(step, 1e-7, 0.25)   # keep steps reasonable
        else:
            step = 0.0

        a = _project_onto_simplex(a - step * grad, S)

        cur_val = _max_convolution(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()

        # jitter every few hundred iterations
        if it % 350 == 0:
            jitter = rng.normal(scale=0.001 * np.mean(a), size=n)
            a = _project_onto_simplex(a + jitter, S)

    return best_a, best_val


def _anneal_local_search(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing hill‑climb that transfers a random
    fraction of mass between two positions.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    cur_val = _max_convolution(a)
    best_a = a.copy()
    best_val = cur_val

    T = 0.05 * cur_val         # initial temperature
    decay = 0.9999

    it = 0
    while time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        frac = rng.uniform(0.02, 0.18)
        delta = a[i] * frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _max_convolution(a_new)
        delta_val = new_val - cur_val
        if delta_val < 0 or rng.random() < np.exp(-delta_val / (T + 1e-12)):
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()

        T *= decay
        if it % 600 == 0:
            jitter = rng.normal(scale=0.001 * S, size=n)
            a = _project_onto_simplex(a + jitter, S)
            cur_val = _max_convolution(a)

    return best_a, best_val


def _build_conv_matrix(a: np.ndarray) -> csr_matrix:
    """
    Build the (2n‑1) × n matrix A with A[k,i] = a[i] if 0≤k‑i<n else 0.
    This encodes the linear constraints (a∗g)[k] ≤ M.
    Returned as a CSR sparse matrix.
    """
    n = a.shape[0]
    # rows = i + j  for i,j in [0,n)
    row_ind = (np.arange(n, dtype=np.int32)[:, None] + np.arange(n, dtype=np.int32)).ravel()
    col_ind = np.repeat(np.arange(n, dtype=np.int32), n)
    data = np.repeat(a, n)
    return csr_matrix((data, (row_ind, col_ind)), shape=(2 * n - 1, n))


def _lp_refine(a0: np.ndarray,
               deadline: float,
               rng: np.random.Generator,
               mix: float = 0.5,
               max_iters: int = 8) -> tuple[np.ndarray, float]:
    """
    Linear‑programming refinement.
    Starting from `a0`, repeatedly:
        – solve max Σ g_i  s.t. (a∗g) ≤ M, g≥0,
        – renormalise g to the simplex,
        – replace a ← (1‑mix) a + mix g.
    Returns the refined vector and its current max‑convolution.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    M = _max_convolution(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        A = _build_conv_matrix(a)
        b_ub = np.full(2 * n - 1, M)
        c = -np.ones(n)                     # maximise Σ g_i  ↔ minimise −Σ g_i
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=(0, None), method='highs')
        if not res.success:
            print("[lp] linprog failed, status:", res.status)
            break
        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 == 0.0:
            print("[lp] trivial LP solution")
            break
        g = g0 * (S / sum_g0)

        a = (1.0 - mix) * a + mix * g
        M_new = _max_convolution(a)
        it += 1
        print(f"[lp] iter={it:2d}  M={M_new:.6f}")
        # stop if improvement stalls
        if M_new >= M - 1e-7:
            break
        M = M_new
    return a, M
