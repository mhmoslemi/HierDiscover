"""
Propose a step‑function with a very low max‑convolution score.

Algorithm
---------

1. Warm‑start from the best construction found so far (global
   ``height_sequence_1``).  If none is available a simple power‑law is used
   and resized to the target length.

2. **Exponential descent** – a mirror‑descent on a soft‑max approximation of
   the maximal auto‑convolution.  The temperature parameter β is doubled
   regularly, which quickly sharpens the surrogate.

3. **Projected GD with restarts** – several short gradient‑descent runs on
   the same surrogate, each started from a jittered copy of the current
   best solution.  Momentum and a simple adaptive step‑size keep the
   optimisation stable.

4. **Aggressive local refinement** – a simulated‑annealing hill‑climber that
   works on the *exact* max‑convolution.  At each iteration a donor index
   (large contribution to the current maximal entry) and a receiver index
   (small contribution) are selected and a small amount of mass Δ is
   transferred.  The full convolution is updated in O(n) using slice
   arithmetic, a quadratic term Δ² is added, and a Metropolis acceptance
   rule with a linearly decreasing temperature is applied.  The allowed Δ
   starts relatively large (up to 20 % of the donor mass) and is cooled
   together with the acceptance temperature, allowing fast early progress
   and fine‑grained final tuning.

5. Final clean‑up – trim trailing zeros, recompute the official score with
   ``evaluate_sequence`` and store the result for future warm‑starts.

All steps respect non‑negativity and the ℓ¹‑norm constraint
(Σa = √(2 n)).  The code never uses closures, only top‑level functions,
and runs within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of power‑of‑two size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the auto‑convolution of a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  For Σa=√(2n) this equals max(a*a)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to ``new_len`` entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max surrogate utilities
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, full_convolution) where ``grad`` has the same length as ``a``.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()                       # soft‑selection mask

    # grad = 2 * reverse(a) convolve p
    a_rev = a[::-1]
    grad_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(grad_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Stage 2 – exponential (multiplicative) descent
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exp‑GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

# Stage 3 – projected gradient descent with restarts
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               target_sum: float,
                               deadline: float) -> tuple[np.ndarray, float]:
    """Projected GD on the soft‑max surrogate with momentum and jitter."""
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    step_base = target_sum * 0.45
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.93
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.0015 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several short GD restarts, keep the best result."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent(start,
                                            target_sum,
                                            time.time() + run_budget)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val

def _aggressive_local_refine(a0: np.ndarray,
                             target_sum: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climber on the *true* max‑convolution.
    Moves a (temperature‑controlled) amount of mass from a high‑contributing
    index to a low‑contributing one.  The full convolution is updated in O(n)
    using slice operations.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)               # exact auto‑convolution
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start_time = time.time()
    total_time = deadline - start_time

    # temperature schedule – linear cooling
    T0 = best_val * 0.04 + 1e-12   # a small positive constant

    while time.time() < deadline:
        # --------------------------------------------------------------
        # Identify the worst convolution index and donors / receivers
        # --------------------------------------------------------------
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each possible donor to the offending entry
        prod = a[idx] * a[k_max - idx]

        # deterministic donor: the index giving the *largest* product
        donor = idx[np.argmax(prod)]

        # receivers are taken from the low‑product tail (≤ 20‑th percentile)
        thresh = np.percentile(prod, 20)
        low_candidates = idx[prod <= thresh]
        if low_candidates.size:
            receiver = rng.choice(low_candidates)
        else:
            # fallback: any index except donor
            receiver = rng.integers(n)
            while receiver == donor:
                receiver = rng.integers(n)

        if donor == receiver:
            continue

        # --------------------------------------------------------------
        # Propose a transfer amount Δ
        # --------------------------------------------------------------
        elapsed = time.time() - start_time
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))

        # early in the run we allow larger moves; they shrink with temperature
        max_delta = min(
            a[donor] * 0.20 * (0.5 + 0.5 * temperature / T0),
            target_sum * 0.005 * (0.5 + 0.5 * temperature / T0),
        )
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # --------------------------------------------------------------
        # Update the convolution exactly (O(n))
        # --------------------------------------------------------------
        b_new = b.copy()
        # linear parts (the two ordered contributions)
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic corrections
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # --------------------------------------------------------------
        # Acceptance rule (Metropolis)
        # --------------------------------------------------------------
        accept = False
        if new_val < best_val:
            accept = True
        else:
            # probability decays with temperature
            prob = np.exp(-(new_val - best_val) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Aggressive refine improved score={best_val:.6f}")

        # --------------------------------------------------------------
        # Periodic re‑normalisation (to fight numerical drift)
        # --------------------------------------------------------------
        if time.time() - start_time > total_time * 0.25:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑norm improved score={best_val:.6f}")
            start_time = time.time()      # restart the clock for the next quarter

        # occasional tiny jitter to escape tiny plateaus
        if int(elapsed * 100) % 2500 == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 0:
                eps = min(a[i] * 0.005, target_sum * 0.0004)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.

    The routine respects the supplied time budget (seconds) and returns the
    best sequence it could find as a plain ``list`` of ``float``.
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # tiny safety margin

    # --------------------------------------------------------------
    # Choose a problem size – larger n gives more expressive power
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 500:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Warm‑start (reuse the best known construction if available)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"],
                                        dtype=np.float64), n)
    else:
        # simple decreasing power‑law as a fallback
        init = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))

    init = init * (target_sum / init.sum())

    # --------------------------------------------------------------
    # Stage 2 – exponential (multiplicative) descent (≈12 % of budget)
    # --------------------------------------------------------------
    exp_deadline = time.time() + 0.12 * (deadline - time.time())
    a_exp, val_exp = _exponential_descent(init, target_sum, exp_deadline)
    best_seq = a_exp
    best_score = val_exp
    print(f"[{time.time():.2f}] After exponential descent best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – projected GD with restarts (≈25 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        gd_deadline = time.time() + 0.25 * (deadline - time.time())
        a_gd, val_gd = _gradient_descent_with_restarts(best_seq,
                                                       target_sum,
                                                       gd_deadline)
        if val_gd < best_score:
            best_seq = a_gd
            best_score = val_gd
        print(f"[{time.time():.2f}] After GD restarts best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – aggressive exact local refinement (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_ref, val_ref = _aggressive_local_refine(best_seq,
                                                   target_sum,
                                                   deadline)
        if val_ref < best_score:
            best_seq = a_ref
            best_score = val_ref
        print(f"[{time.time():.2f}] After aggressive refinement best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final clean‑up
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # --------------------------------------------------------------
    # Official evaluation (may be a no‑op if the caller already has it)
    # --------------------------------------------------------------
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in a later call
    globals()["height_sequence_1"] = final_seq

    return final_seq
