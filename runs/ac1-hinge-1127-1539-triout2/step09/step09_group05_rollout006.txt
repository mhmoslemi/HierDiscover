def _smooth_grad_descent(a0, tau, deadline, rng, best_val_init=None):
    a = a0.copy()
    n = len(a)
    S = np.sqrt(2.0 * n)
    best_a = a.copy()
    # compute initial objective fval
    fval, _ = _smooth_max_and_gradient(a, tau)
    best_val = fval
    it = 0
    while time.time() < deadline:
        it += 1
        fval, grad = _smooth_max_and_gradient(a, tau)
        # step size: use Polyak: (fval - best_val) / (norm_sq + eps)
        norm_sq = np.dot(grad, grad)
        if norm_sq > 0:
            step = (fval - best_val) / (norm_sq + 1e-12)
            step = np.clip(step, 1e-6, 0.1)
        else:
            step = 0.0
        # gradient descent
        a = _project_onto_simplex(a - step * grad, S)
        # evaluate actual hard max
        cur_max = _max_convolution(a)
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
        # occasional random reparam (jitter)
        if it % 200 == 0:
            jitter = rng.normal(scale=0.001 * S, size=n)
            a = _project_onto_simplex(a + jitter, S)
    return best_a, best_val

    # compute convolution c = a*a (full)
    c = _full_convolution_fft(a)
    # compute softmax weights
    max_c = c.max()

    sum_exp = exp_term.sum()

    # gradient: 2 * conv(w, rev(a)), extract central region
    a_rev = a[::-1]

    start = n - 1

    smooth_max = max_c + np.log(sum_exp) / tau

"""
Improved step‑function optimiser.

The optimiser combines three ideas that complement the earlier
sub‑gradient / annealing pipeline:

1. **Random families + scaling** – a quick broad search over many
   parametric shapes (power‑law, exponential, linear, U‑shape,
   sinusoid, Gaussian, Dirichlet) for lengths between 3 000 and 20 000.

2. **Smoothed‑max gradient descent** – instead of the non‑smooth
   ``max(a*a)`` we minimise the log‑sum‑exp approximation
   ``smooth_max = (log Σ exp(τ·c))/τ`` with ``c = a*a``.
   Its gradient is  ``2·conv(w, rev(a))`` where ``w`` are the softmax
   weights of the convolution entries.  A continuation on τ
   (τ=5,10,20,40,80) allows large steps early and fine tuning later.
   After each gradient step we project onto the ℓ₁‑simplex
   {a≥0, Σa = √(2 n)}.

3. **Final refinement** – the classic projected sub‑gradient descent
   and a cheap simulated‑annealing mass‑transfer hill‑climb are applied
   as a finishing touch.

The whole search respects the 1000 s budget, keeps the best construction
found and stores it in the global variable ``height_sequence_1``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a∗a."""
    return _full_convolution_fft(a).max()


def _project_onto_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex { w | w_i ≥ 0 , Σ w_i = S }.
    Implements the O(n) algorithm of Duchi et al.
    """
    if S < 0:
        raise ValueError("Simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]

def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    raw = mix * ush + (1.0 - mix) * base
    return raw


def _init_sine(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    vals = np.sin(np.pi * (i + 1) / (n + 1))
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * vals + (1.0 - mix) * base
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    return rng.random(n)


def _init_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    centre = (n - 1) / 2.0
    sigma = rng.uniform(0.1 * n, 0.4 * n)
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * vals + (1.0 - mix) * base
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Occasionally warm‑starts from the globally best sequence and otherwise
    draws from a random parametric family.
    """
    S = np.sqrt(2.0 * n)

    # Warm start with probability ≈0.2
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
        return a

    # Sample a family
    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'ushape', 'sine', 'gaussian',
                        'uniform', 'dirichlet'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'sine':
        raw = _init_sine(n, rng)
    elif style == 'gaussian':
        raw = _init_gaussian(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a

# Smoothed‑max utilities

def _smooth_max_and_gradient(a: np.ndarray, tau: float):
    """
    Returns (smooth_max, grad) where smooth_max = (log Σ exp(τ·c))/τ  with
    c = a∗a, and grad = ∂smooth_max/∂a.
    """
    n = a.shape[0]
    # 1) full self‑convolution
    c = _full_convolution_fft(a)          # length 2n‑1
    max_c = c.max()

    # 2) softmax weights
    exp_term = np.exp(tau * (c - max_c))   # stability trick
    sum_exp = exp_term.sum()
    w = exp_term / sum_exp                # Σ w = 1, length 2n‑1

    # 3) gradient = 2 * conv(w, rev(a))   (central window)
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    f_len = _next_pow2(conv_len)
    Fw = np.fft.rfft(w, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fw * Fa_rev, n=f_len)[:conv_len]
    start = n - 1
    grad = 2.0 * conv[start:start + n]

    # 4) smooth max value
    smooth_max = max_c + np.log(sum_exp) / tau
    return float(smooth_max), grad


def _smooth_grad_descent(a0: np.ndarray,
                         tau: float,
                         deadline: float,
                         rng: np.random.Generator,
                         best_hard: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the τ‑smoothed max.
    Returns the best sequence seen (w.r.t. hard max) and its hard max value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1

        smooth_val, grad = _smooth_max_and_gradient(a, tau)

        # Polyak step size using the current best hard max as target
        norm_sq = np.dot(grad, grad)
        if norm_sq > 0:
            step = (smooth_val - best_val) / (norm_sq + 1e-12)
            step = np.clip(step, 1e-7, 0.2)
        else:
            step = 0.0

        a = _project_onto_simplex(a - step * grad, S)

        cur_val = _max_convolution(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=0.001 * S, size=n)
            a = _project_onto_simplex(a + jitter, S)

    return best_a, best_val

def _anneal_local_search(a0: np.ndarray,

    best_a = a.copy()
    best_val = cur_val

    decay = 0.9999

            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()

        T *= decay
        if it % 500 == 0:
            jitter = rng.normal(scale=0.001 * S, size=n)
            a = _project_onto_simplex(a + jitter, S)
            cur_val = _max_convolution(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The best sequence is stored in the global ``height_sequence_1`` and
    also returned as a plain list.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # Warm start handling
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    best_seq = None
    best_score = np.inf
    best_n = None
    attempts = 0

    # ------------------------------------------------------------------
    # Phase 1 – broad random exploration (≈ 55 % of budget)
    # ------------------------------------------------------------------
    explore_deadline = end_time - 300.0   # reserve ≈300 s for deeper work
    while time.time() < explore_deadline:
        attempts += 1
        # favour longer sequences but keep some variety
        if rng.random() < 0.7:
            n = int(rng.integers(3000, 12000))
        else:
            n = int(rng.integers(12000, 20000))

        a0 = _initial_vector(n, rng, warm_seq)
        val = _max_convolution(a0)
        if val < best_score:
            best_score = val
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempts:4d} n={n:5d}  new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – smoothed‑max gradient descent on selected lengths
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform medium sized
        n = 4000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = _max_convolution(best_seq)
        best_n = n

    # decide on a short list of target lengths (geometric progression)
    target_lengths = [best_n]
    for factor in [1.3, 1.7, 2.2, 3.0]:
        cand = int(best_n * factor)
        if 3000 <= cand <= 25000:
            target_lengths.append(cand)
    # guarantee uniqueness & sort
    target_lengths = sorted(set(target_lengths))

    for target_n in target_lengths:
        if time.time() >= end_time - 120.0:
            break
        # scale current best to new length
        a_scaled = _scale_sequence(best_seq, target_n)
        a_scaled = np.clip(a_scaled, 0.0, None)
        a_scaled = a_scaled * (np.sqrt(2.0 * target_n) / max(a_scaled.sum(), 1e-12))

        # schedule of τ values (continuation)
        tau_schedule = [5, 10, 20, 40, 80]
        a_current = a_scaled
        for tau in tau_schedule:
            if time.time() >= end_time - 80.0:
                break
            # allocate a slice of the remaining budget for this (τ, n) pair
            remaining = end_time - time.time()
            slice_deadline = time.time() + min(remaining * 0.25, 45.0)

            a_opt, cur_hard = _smooth_grad_descent(a_current,
                                                   tau,
                                                   slice_deadline,
                                                   rng,
                                                   best_score)
            # update trackers
            if cur_hard < best_score:
                best_score = cur_hard
                best_seq = a_opt.copy()
                best_n = target_n
                print(f"[smooth] n={target_n:5d} τ={tau:2d}  new best={best_score:.6f}")

            a_current = a_opt  # continue from the refined vector

    # ------------------------------------------------------------------
    # Phase 3 – long projected sub‑gradient run on the current best length
    # ------------------------------------------------------------------
    if time.time() < end_time - 30.0:
        remaining = end_time - time.time() - 5.0
        a_opt, val = _subgradient_descent(best_seq,
                                          time.time() + remaining,
                                          rng,
                                          best_score)
        if val < best_score:
            best_score = val
            best_seq = a_opt.copy()
            print(f"[final subgrad] length={len(best_seq)}  best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated‑annealing hill‑climb as final touch
    # ------------------------------------------------------------------
    if time.time() < end_time - 5.0:
        remaining = end_time - time.time() - 1.0
        a_opt, val = _anneal_local_search(best_seq,
                                          time.time() + remaining,
                                          rng)
        if val < best_score:
            best_score = val
            best_seq = a_opt.copy()
            print(f"[anneal] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – sanity check with the official evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f}  internal={best_score:.6f}  length={len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
