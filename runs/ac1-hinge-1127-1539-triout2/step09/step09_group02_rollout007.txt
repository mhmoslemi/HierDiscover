"""
Propose a step‑function that minimizes ``evaluate_sequence``.

Algorithm outline
-----------------
1. **Parametric family search** – cheap scans over exponential,
   power‑law and mixed families produce strong starting points.
2. **Ensemble averaging** – the best few candidates are averaged
   (projected to the ℓ¹‑simplex Σa = √(2 n)) which often flattens the
   autocorrelation.
3. **Projected Adam** – a few restarts on the smooth‑max (log‑sum‑exp)
   surrogate quickly improve the shape.
4. **Deterministic hill‑climb** – exact moves shift mass from the
   current worst convolution entry to a more favorable location.
5. **Final polishing** – remaining time is spent on a longer Adam run
   and a short hill‑climb.

The routine respects the global time budget (default 1000 s) and
returns the best sequence found.
"""

import time
import math
import numpy as np
from operator import itemgetter

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using an FFT (fast, O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float,
                    deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max surrogate.
    The algorithm runs until the wall‑clock deadline.
    """
    a = a0.copy()
    n = a.size

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # learning‑rate base (scaled with the simplex radius)
    base_lr = 0.15 * target_sum

    # smoothed‑max schedule
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1

        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        # occasional tiny jitter to escape flat regions
        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4 / (beta + 1.0),
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # increase smoothing
        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Deterministic hill‑climbing (exact max)
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that always attacks the current worst convolution
    entry.  Moves a (large) fraction of the responsible mass to the
    index that contributes the least to that entry.  The full convolution
    is updated in O(n) per move.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()

    rng = np.random.default_rng()

    while time.time() < deadline:
        # index of the current maximal convolution entry
        M = int(np.argmax(b))

        # range of source indices that influence b[M]
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]

        # source with the largest contribution
        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        # counterpart values for possible destinations
        counterpart = np.zeros(n, dtype=float)
        j_low = i_low
        j_high = i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        # choose destination with minimal counterpart (but not i)
        order = np.argsort(counterpart[j_range])
        j = None
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        # amount of mass to move (up to 80 % of source)
        delta = a[i] * 0.8
        if delta <= 0.0:
            break

        # --- evaluate the move ------------------------------------------------
        b_new = b.copy()
        # linear convolution update (O(n))
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        # --- accept if we improve ------------------------------------------------
        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # try a smaller move (binary search style)
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                # small random jitter to escape a possible plateau
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                # re‑project onto the simplex and recompute convolution
                target = math.sqrt(2.0 * n)
                a = _project_to_simplex(a, target)
                b = _full_convolution(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Small parametric family search
# ----------------------------------------------------------------------


def _family_candidates(n: int, rng, top_k: int = 8):
    """
    Scan several simple families (exponential, power‑law, mixed,
    piecewise, Dirichlet) and keep the best ``top_k`` candidates.
    Returns a list of (score, vector) pairs.
    """
    target = math.sqrt(2.0 * n)
    candidates = []

    # exponential decay
    for _ in range(30):
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * np.arange(n, dtype=float))
        a = _project_to_simplex(a, target)
        score = _max_convolution_fft(a)
        candidates.append((score, a))

    # power‑law
    for _ in range(30):
        p = rng.uniform(0.1, 1.5)
        a = (np.arange(1, n + 1, dtype=float)) ** (-p)
        a = _project_to_simplex(a, target)
        score = _max_convolution_fft(a)
        candidates.append((score, a))

    # mixed power‑law × exponential
    for _ in range(30):
        p = rng.uniform(0.1, 1.5)
        lam = rng.exponential(scale=0.015)
        a = (np.arange(1, n + 1, dtype=float)) ** (-p) * np.exp(-lam * np.arange(n))
        a = _project_to_simplex(a, target)
        score = _max_convolution_fft(a)
        candidates.append((score, a))

    # piecewise exponential
    for _ in range(10):
        b = rng.integers(int(0.2 * n), int(0.8 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])
        a = _project_to_simplex(a, target)
        score = _max_convolution_fft(a)
        candidates.append((score, a))

    # Dirichlet (uniform on simplex)
    for _ in range(20):
        a = rng.dirichlet(alpha=np.ones(n)) * target
        a = _project_to_simplex(a, target)
        score = _max_convolution_fft(a)
        candidates.append((score, a))

    # sort and keep the best ``top_k``
    candidates.sort(key=itemgetter(0))
    return candidates[:top_k]


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence`` under a wall‑clock budget.

    The strategy blends a fast parametric family scan, ensemble averaging,
    a few Adam runs, deterministic hill‑climbing and a final polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 1)  Seed from a previously found construction, if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            n0 = prev.size
            target0 = math.sqrt(2.0 * n0)
            prev = _project_to_simplex(prev, target0)
            best_seq = prev.copy()
            best_score = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 "
                  f"(n={n0}) score={best_score:.6f}")
        except Exception as e:
            print(f"Seeding error: {e}")

    # ------------------------------------------------------------------
    # 2)  Length schedule – increase dimension gradually
    # ------------------------------------------------------------------
    schedule = [2000, 4000, 8000, 20000, 50000]

    for n in schedule:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Initialise candidate (upsample previous best if possible)
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)

        a_init = _project_to_simplex(a_init, target_sum)

        remaining = deadline - time.time()
        if remaining < 10.0:
            break

        # --------------------------------------------------------------
        # 2a)  Quick parametric family scan (≈10 % of remaining time)
        # --------------------------------------------------------------
        fam_budget = min(remaining * 0.10, 20.0)
        fam_deadline = time.time() + fam_budget
        top_candidates = _family_candidates(n, rng, top_k=8)
        # ensure the up‑sampled initial vector participates
        init_score = _max_convolution_fft(a_init)
        top_candidates.append((init_score, a_init))
        top_candidates.sort(key=itemgetter(0))
        top_candidates = top_candidates[:8]

        # --------------------------------------------------------------
        # 2b)  Ensemble averaging of the best few candidates
        # --------------------------------------------------------------
        ensemble = np.mean([vec for _s, vec in top_candidates], axis=0)
        ensemble = _project_to_simplex(ensemble, target_sum)
        ensemble_score = _max_convolution_fft(ensemble)
        if ensemble_score < best_score - 1e-12:
            best_seq = ensemble.copy()
            best_score = ensemble_score
            print(f"[{time.time():.1f}] Ensemble avg improvement (n={n}) "
                  f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # 2c)  Adam optimisation – a few restarts
        # --------------------------------------------------------------
        # allocate most of the remaining budget to Adam / hill‑climb
        remaining = deadline - time.time()
        if remaining < 2.0:
            break
        gd_budget = remaining * 0.55
        hc_budget = remaining * 0.35
        extra_budget = remaining * 0.10

        # number of Adam restarts (≈5 s each)
        min_restart_time = 5.0
        n_restarts = max(1, int(gd_budget // min_restart_time))
        per_restart = gd_budget / n_restarts

        local_best_a = None
        local_best_score = np.inf

        for r in range(n_restarts):
            if time.time() >= deadline:
                break
            this_deadline = min(deadline - 0.2, time.time() + per_restart)
            start_a = a_init.copy() if r == 0 else _random_initial_sequence(n, rng)
            a_opt = _adam_optimize(start_a, target_sum, this_deadline)
            score = _max_convolution_fft(a_opt)
            if score < local_best_score:
                local_best_score = score
                local_best_a = a_opt.copy()

        if local_best_a is not None and local_best_score < best_score - 1e-12:
            best_seq = local_best_a.copy()
            best_score = local_best_score
            print(f"[{time.time():.1f}] Adam improvement (n={n}) "
                  f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # 2d)  Deterministic hill‑climb on the exact convolution
        # --------------------------------------------------------------
        if time.time() < deadline - 2.0:
            a_cur = best_seq.copy()
            a_cur = _project_to_simplex(a_cur, target_sum)
            b_cur = _full_convolution(a_cur)
            hc_deadline = min(deadline - 0.2, time.time() + hc_budget)
            a_hc, score_hc = _hill_climb_deterministic(a_cur, b_cur, hc_deadline)
            if score_hc < best_score - 1e-12:
                best_seq = a_hc.copy()
                best_score = score_hc
                print(f"[{time.time():.1f}] Hill‑climb improvement (n={n}) "
                      f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # 2e)  Tiny random perturbations (simulated‑annealing style)
        # --------------------------------------------------------------
        while time.time() + 1.5 < deadline:
            cand = best_seq.copy()
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            delta = cand[i] * 0.05
            cand[i] -= delta
            cand[j] += delta
            cand = _project_to_simplex(cand, target_sum)
            score = _max_convolution_fft(cand)
            if score < best_score - 1e-12:
                best_seq = cand
                best_score = score
                print(f"[{time.time():.1f}] Random tweak improvement (n={n}) "
                      f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # 2f)  Optional short extra Adam run (if time permits)
        # --------------------------------------------------------------
        if time.time() < deadline - 1.0 and extra_budget > 2.0:
            extra_deadline = deadline - 0.5
            a_ref = _adam_optimize(best_seq, target_sum, extra_deadline)
            score_ref = _max_convolution_fft(a_ref)
            if score_ref < best_score - 1e-12:
                best_seq = a_ref.copy()
                best_score = score_ref
                print(f"[{time.time():.1f}] Post‑HC Adam improvement (n={n}) "
                      f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Final polishing on the remaining budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        n = best_seq.size
        target = math.sqrt(2.0 * n)

        final_deadline = deadline - 1.0
        best_seq = _adam_optimize(best_seq, target, final_deadline)

        b_final = _full_convolution(best_seq)
        best_seq, final_score = _hill_climb_deterministic(
            best_seq, b_final, deadline - 0.5)

        if final_score < best_score - 1e-12:
            best_score = final_score
            print(f"[{time.time():.1f}] Final polishing improvement: "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Clean‑up: drop trailing near‑zeros
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # 5) Report official evaluation
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
