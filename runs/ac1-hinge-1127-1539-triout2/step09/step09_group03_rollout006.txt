"""
Propose a step‑function with a low max‑convolution value.

Algorithm
---------

1. **Problem size** – Choose a length *n* based on the allotted
   time‑budget (larger *n* gives a finer search space).

2. **Warm‑starts** – Build a few sensible initial sequences
   * (i) the best sequence discovered in a previous run
   * (ii) a simple decreasing power‑law
   * (iii) a symmetric power‑law.
   All are rescaled to satisfy Σa = √(2 n).

3. **Two‑phase smooth optimisation**
   * exponential (mirror) descent on a soft‑max surrogate of the
     max‑convolution (the same code that was used before);
   * projected gradient descent with momentum, many random restarts
     and occasional jitter.

   The best of the three warm‑starts after this phase becomes the
   current candidate.

4. **Direct sub‑gradient descent on the true max‑convolution**.
   At every iteration we compute the full convolution, pick the
   *k* indices with the largest values (default *k* = 3), form the
   sub‑gradient of the max‑convolution (averaged over those indices),
   take a diminishing‑step update and project back onto the simplex.

5. **Exact local mass‑transfer refinement** – the same O(n) hill‑climbing
   routine that was used previously, now run on the candidate obtained
   after step 4.

6. **Final clean‑up** – drop trailing zeros, evaluate the score with
   the provided ``evaluate_sequence`` and store the sequence globally
   for warm‑starts in future calls.

All steps respect non‑negativity and the ℓ¹‑norm constraint
(Σa = √(2 n)), therefore the returned list can be fed directly to
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a power‑of‑2 FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  For Σa=√(2n) this equals max(a*a)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple linear interpolation for resizing
# ----------------------------------------------------------------------


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max surrogate gradient (used by the original code)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # grad = 2 * (reverse(a) convolve p)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase 2 – exponential (multiplicative) descent
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 3 – projected gradient descent with restarts
# ----------------------------------------------------------------------


def _annealed_gradient_descent_v2(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    step_base = target_sum * 0.45
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.93
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0015 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GDv2 it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_v2(start,
                                               time.time() + run_budget,
                                               target_sum)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 4 – direct sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         top_k: int = 3) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on   max_k (a*a)[k]   with projection onto the simplex.
    ``top_k`` allows us to average gradients over the biggest peaks,
    which helps to smooth the descent when several entries are close to the maximum.
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    mom = np.zeros_like(a)
    mu = 0.9                     # momentum coefficient
    it = 0
    # initial step size – a fraction of the total ℓ¹‑mass
    step0 = target_sum * 0.2

    while time.time() < deadline:
        # full convolution and its peak(s)
        b = _conv_fft(a)
        max_val = b.max()
        # indices of the largest values (at most top_k)
        if top_k == 1:
            ks = np.array([int(b.argmax())], dtype=int)
        else:
            # np.argpartition is O(n); we grab the top_k entries
            ks = np.argpartition(b, -top_k)[-top_k:]

        # accumulate sub‑gradient over the selected peaks
        g = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=int)
            j = k - idx
            # contribution 2*a[j] except when i=j (i.e. 2*idx==k)
            grad_contrib = 2.0 * a[j]
            eq = (idx == j)
            grad_contrib[eq] = a[idx[eq]]          # only a_i when i=j
            g[idx] += grad_contrib
        # average
        g /= max(1, len(ks))

        # momentum update
        mom = mu * mom + g

        # diminishing step size
        lr = step0 / np.sqrt(it + 1.0)

        a -= lr * mom
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        # occasional jitter to escape plateaus
        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 5 – exact local refinement (same as the previous version)
# ----------------------------------------------------------------------


def _local_refine(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    # temperature for simulated‑annealing acceptance
    T0 = best_val * 0.04
    last_recalc = start

    while time.time() < deadline:
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        prod = a[idx] * a[k_max - idx]

        # donor weighted by contribution to the current maximum
        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – bias toward indices that contribute little
        weight_all = np.zeros(n, dtype=a.dtype)
        weight_all[lo:hi + 1] = prod
        thresh = np.percentile(weight_all, 30)
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # propose a modest transfer
        max_delta = min(a[donor] * 0.04, target_sum * 0.0012)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # update convolution exactly (O(n))
        b_new = b.copy()
        b_new[donor:donor + n] += -2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Local refine improved score={best_val:.6f}")

        # periodic full recomputation to avoid drift
        if time.time() - last_recalc > 12.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_recalc = time.time()
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑norm improved score={best_val:.6f}")

        # occasional tiny jitter
        if int((time.time() - start) * 100) % 2500 == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.005, target_sum * 0.0004)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point expected by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the supplied time‑budget (seconds) and returns
    the best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.4   # tiny safety margin

    # --------------------------------------------------------------
    # Choose problem size based on the remaining budget
    # --------------------------------------------------------------
    if budget_s >= 900:
        n = 16000
    elif budget_s >= 600:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Build a few warm‑start candidates
    # --------------------------------------------------------------
    warm_starts = []

    if "height_sequence_1" in globals():
        warm = _resize_sequence(np.array(globals()["height_sequence_1"],
                                         dtype=np.float64), n)
        warm_starts.append(warm)

    # power‑law decreasing
    pow_seq = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))
    warm_starts.append(pow_seq)

    # symmetric power‑law
    half = (n + 1) // 2
    half_seq = np.arange(1, half + 1, dtype=np.float64) ** (-0.55)
    sym_seq = np.concatenate([half_seq,
                              half_seq[::-1][: n - half]])
    warm_starts.append(sym_seq)

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Quickly run the original two‑phase smooth optimiser on each start
    # --------------------------------------------------------------
    per_start_budget = (deadline - time.time()) * 0.2 / len(warm_starts)

    for idx, seq in enumerate(warm_starts):
        seq = seq * (target_sum / seq.sum())
        exp_dead = time.time() + per_start_budget * 0.3
        a_exp, val_exp = _exponential_descent(seq, target_sum, exp_dead)
        gd_dead = time.time() + per_start_budget * 0.7
        a_gd, val_gd = _gradient_descent_with_restarts(a_exp,
                                                       target_sum,
                                                       gd_dead)
        if val_gd < best_score:
            best_score = val_gd
            best_seq = a_gd.copy()
        print(f"[{time.time():.2f}] Warm‑start #{idx} score={val_gd:.6f}")

    # --------------------------------------------------------------
    # Direct sub‑gradient descent on the true max‑convolution
    # --------------------------------------------------------------
    if time.time() < deadline:
        sub_dead = time.time() + (deadline - time.time()) * 0.35
        a_sub, val_sub = _subgradient_descent(best_seq,
                                              target_sum,
                                              sub_dead,
                                              top_k=3)
        if val_sub < best_score:
            best_score = val_sub
            best_seq = a_sub.copy()
        print(f"[{time.time():.2f}] After sub‑gradient best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Exact local refinement (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_ref, val_ref = _local_refine(best_seq, target_sum, deadline)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
        print(f"[{time.time():.2f}] After local refinement best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final clean‑up and evaluation
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # store for future warm‑starts
    globals()["height_sequence_1"] = final_seq

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    return final_seq
