def _subgradient_line_search(a, target_sum, deadline):
    a = a.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())
        # gradient
        grad = np.zeros_like(a)
        lo = max(0, k_max - (a.shape[0] - 1))
        hi = min(k_max, a.shape[0] - 1)
        idx = np.arange(lo, hi+1)
        partner = k_max - idx
        grad[idx] = 2.0 * a[partner]
        # safe eta: min_{i: grad_i>0} a_i / grad_i
        pos = grad > 0
        if not np.any(pos):
            break
        safe_eta = np.min(a[pos] / grad[pos])
        # initial step size maybe 0.5 * safe_eta
        eta = 0.5 * safe_eta
        # line search: try decreasing step until improvement
        improved = False
        for _ in range(10):
            a_candidate = a - eta * grad
            a_candidate = _project_to_simplex(a_candidate, target_sum)
            val = _score(a_candidate)
            if val < best_val:
                best_val = val
                best_a = a_candidate.copy()
                improved = True
                a = a_candidate
                break
            eta *= 0.5
        if not improved:
            # random jitter small and continue
            jitter = rng.normal(scale=target_sum*0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
        it += 1
    return best_a, best_val

"""
propose_candidate
~~~~~~~~~~~~~~~~~

A multi‑stage optimiser for the max‑convolution benchmark.

The method combines three ideas:

1. **Coarse parametric warm‑start** – random draws from a
   two‑parameter power‑law / exponential family are tested and the
   best (scaled to the required ℓ¹‑norm) is used as the starting point.

2. **Projected sub‑gradient descent with line‑search** – the exact
   sub‑gradient of the active (maximal) convolution entry is computed,
   a safe step size is taken and the iterate is projected onto the
   ℓ¹‑simplex (Σa = √(2n), a≥0).  A simple back‑tracking line‑search
   allows larger steps than a plain diminishing schedule.

3. **Light‑weight simulated‑annealing fine tuning** – after the
   deterministic descent a cheap two‑index “mass‑transfer” move is
   repeatedly tried; moves that improve the objective are kept,
   otherwise they are accepted with a temperature‑controlled probability.
   Because the move preserves the total mass, no projection is needed.

All stages respect the time budget (default 1000 s) and keep the best
sequence found.  The final list is trimmed of trailing zeros and stored
in ``globals()["height_sequence_1"]`` for possible warm‑starts in a
subsequent run.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution / scoring utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  For a correctly normalised a
    this equals the maximal convolution entry."""
    n = a.size
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) shape, normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Stage 2 – projected sub‑gradient descent with line‑search
# ----------------------------------------------------------------------


def _subgradient_line_search(a0: np.ndarray,
                             target_sum: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution entry.
    Uses a simple back‑tracking line‑search to obtain larger, productive steps.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it_without_improve = 0
    max_without_improve = 200  # after this we inject a small jitter

    while time.time() < deadline:
        # current convolution and active index
        conv = _conv_fft(a)
        k_max = int(conv.argmax())

        # sub‑gradient of the active entry: 2·a[k_max‑i] on the feasible i‑range
        n = a.size
        grad = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        grad[idx] = 2.0 * a[partner]

        # safe step size that keeps a - η·grad non‑negative (before projection)
        pos = grad > 0
        if not np.any(pos):
            break
        safe_eta = np.min(a[pos] / grad[pos])
        eta = 0.5 * safe_eta  # start with a fairly aggressive step

        improved = False
        # back‑tracking line‑search (max 8 reductions)
        for _ in range(8):
            a_candidate = a - eta * grad
            a_candidate = _project_to_simplex(a_candidate, target_sum)
            val = _score(a_candidate)
            if val < best_val:
                best_val = val
                best_a = a_candidate.copy()
                a = a_candidate
                improved = True
                it_without_improve = 0
                print(f"[{time.time():.2f}] SubgradLS it={_:.0f} η={eta:.4e} score={best_val:.6f}")
                break
            eta *= 0.5  # shrink step size

        if not improved:
            it_without_improve += 1
            # occasional random jitter to escape flat regions
            if it_without_improve >= max_without_improve:
                jitter = rng.normal(scale=target_sum * 0.001, size=n)
                a = np.maximum(a + jitter, 0.0)
                a = a * (target_sum / a.sum())
                it_without_improve = 0
        # tiny safety: keep the loop from overrunning the deadline
        if time.time() >= deadline:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 3 – cheap simulated‑annealing fine tuning
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap SA that moves a small amount of mass between two indices.
    The total ℓ¹‑mass is preserved, therefore no projection is required.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    n = a.size
    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start

    while time.time() < deadline:
        # linear cooling schedule
        elapsed = time.time() - start
        frac = max(0.0, 1.0 - elapsed / total_time)
        temperature = frac * cur_score * 0.02   # temperature scale
        step_factor = max(0.001, frac * 0.04)   # up to ~4 % of donor mass

        # pick a donor proportional to its mass (more mass ⇒ more likely to move)
        donor = rng.choice(n, p=a / a.sum())
        # pick a receiver uniformly (different from donor)
        receiver = rng.integers(n - 1)
        if receiver >= donor:
            receiver += 1

        delta = a[donor] * step_factor * rng.random()
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        # mass is preserved, no need for projection
        new_score = _score(a_new)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temperature, 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] SA improve score={best_score:.6f}")

    return best_a, best_score

# ----------------------------------------------------------------------

    best_seq = init.copy()

    # --------------------------------------------------------------

        val = _score(seq)

    # make the sequence available for the next run
