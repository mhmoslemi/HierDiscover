"""

Improvements over the previous baseline

* **Long final refinement** – after a cheap random exploration we

  after the long Adam optimisation to capture any remaining tiny

All helper functions are defined at top level; the entry point required
by the harness is ``propose_candidate``.

import time
import math
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Peak value of a*a (full convolution)."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Exact sub‑gradient of the hard max (or a softmax when tau>0)
# ----------------------------------------------------------------------
def _max_gradient(a: np.ndarray, tau: float):
    """
    Returns (grad, max_b) where ``grad`` is ∂max(b)/∂a for ``b = a*a``
    (full convolution) and ``max_b`` is the value of the maximum.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        eps = max_b * 1e-9 + 1e-12
        mask = (b >= max_b - eps).astype(float)
        s = mask.sum()
        if s == 0:
            mask = np.ones_like(b) / b.size
        else:
            mask /= s
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / exps.sum()

    # convolution of mask with reversed a
    a_rev = a[::-1]
    conv_len = mask.size + a.size - 1
    f_len = _next_pow2(conv_len)

    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)

    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    grad = 2.0 * conv[a.size - 1:a.size - 1 + a.size]
    return grad, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   lr0: float = 0.4,
                   tau_decay: float = 0.9996,
                   lr_decay: float = 0.99985,
                   jitter_every: int = 200) -> tuple[np.ndarray, float]:
    """
    minimise the peak of the self‑convolution under the ℓ₁‑norm
    constraint (sum = √(2 n)).  Returns the best vector seen and its
    peak value.
    """
    n = a0.shape[0]
    S = math.sqrt(2.0 * n)                     # required ℓ₁‑norm

    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                           # logits (log‑softmax)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr0

    tau = max(_full_convolution_fft(a0).max(), 0.5)

    best_a = a0.copy()
    best_peak = _max_convolution(a0)
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- recover a from logits (log‑softmax) -----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ----- gradient of (soft‑)max convolution -----
        g, peak = _max_gradient(a, tau)

        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ----- gradient w.r.t. logits (chain rule) -----
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- schedules -----
        tau *= tau_decay
        lr *= lr_decay

        # occasional jitter to escape plateaus
        if jitter_every and (it % jitter_every) == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak

def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.2, 5.0)
    raw = rng.gamma(alpha, 1.0, size=n)
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.1, 0.8)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_left_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    p = rng.uniform(0.3, 3.0)
    offset = rng.uniform(0.0, 10.0)
    mix = rng.uniform(0.0, 1.0)
    raw = mix + (1.0 - mix) * (idx + offset) ** (-p)
    return raw


def _init_left_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    lam = rng.uniform(0.001, 0.02)
    mix = rng.uniform(0.0, 1.0)
    raw = mix + (1.0 - mix) * np.exp(-lam * idx)
    return raw


def _init_sparse_bumps(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = np.full(n, rng.uniform(0.0, 0.15))
    nbumps = rng.integers(1, max(2, n // 250))
    for _ in range(nbumps):
        center = rng.integers(0, n)
        width = rng.integers(3, max(5, n // 100))
        height = rng.uniform(0.8, 4.0)
        lo = max(0, center - width // 2)
        hi = min(n, lo + width)
        raw[lo:hi] = np.maximum(raw[lo:hi], height)
    return raw


def _init_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    mu = (n - 1) / 2.0
    sigma = rng.uniform(0.05, 0.30) * n
    x = np.arange(n)
    raw = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    return raw


def _init_half_sine(n: int, rng: np.random.Generator) -> np.ndarray:
    x = np.arange(n)
    raw = np.sin(np.pi * x / (n - 1))
    raw = raw * rng.uniform(0.5, 2.0) + rng.uniform(0.0, 0.2)
    return raw


def _init_binary_spike(n: int, rng: np.random.Generator) -> np.ndarray:
    """Very sparse binary vector: a few entries equal to 1, rest 0."""
    raw = np.zeros(n)
    k = rng.integers(1, max(2, n // 50))  # ~2 % non‑zeros
    idx = rng.choice(n, size=k, replace=False)
    raw[idx] = 1.0
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    With probability ≈0.2 we warm‑start from the global best,
    otherwise we draw from one of several parametric families.
    """
    S = math.sqrt(2.0 * n)

    if warm_seq is not None and rng.random() < 0.2:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    style = rng.choice(['dirichlet', 'gamma', 'uniform',
                        'left_powerlaw', 'left_exponential',
                        'sparse', 'gaussian', 'half_sine',
                        'binary_spike'])
    if style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'left_powerlaw':
        raw = _init_left_powerlaw(n, rng)
    elif style == 'left_exponential':
        raw = _init_left_exponential(n, rng)
    elif style == 'sparse':
        raw = _init_sparse_bumps(n, rng)
    elif style == 'gaussian':
        raw = _init_gaussian(n, rng)
    elif style == 'half_sine':
        raw = _init_half_sine(n, rng)
    else:               # binary_spike
        raw = _init_binary_spike(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Simple stochastic local search – mass transfer walk
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Mass‑transfer walk with exponential acceptance."""
    n = len(seq)
    S = math.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score
    temperature = 0.15 * cur_score

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] == 0.0:
            continue
        delta = rng.uniform(0.0, 0.3) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())
        new_score = _max_convolution(new)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temperature, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

        temperature *= 0.99995
        if temperature < 1e-6:
            temperature = 1e-6

    return best, best_score


def _hill_climb(seq: np.ndarray,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Pure downhill moves only – cheap and fast."""
    n = len(seq)
    S = math.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] < 1e-12:
            continue
        delta = rng.uniform(0.0, 0.15) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())
        new_score = _max_convolution(new)
        if new_score < cur_score:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The algorithm consists of three phases:
        1. cheap random exploration,
        2. a long Adam refinement on a large up‑sampled length,
        3. stochastic fine‑tuning (anneal + greedy hill‑climb).

    The final sequence is stored in the global variable ``height_sequence_1``
    and also returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    safety_margin = 5.0
    end_time = time.time() + budget_s - safety_margin
    if end_time <= time.time():
        raise RuntimeError("Budget too small after safety margin.")

    # ------------------------------------------------------------------
    # Warm‑start from a previously stored construction, if any
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # ------------------------------------------------------------------
    # Phase 1 – fast random exploration (≈10 % of budget)
    # ------------------------------------------------------------------
    explore_deadline = time.time() + 0.10 * budget_s
    best_seq = None
    best_score = np.inf
    attempt = 0

    while time.time() < explore_deadline:
        attempt += 1
        # length distribution – focus on the medium‑large range
        n = int(rng.integers(4000, 15000))

        a0 = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a0)

        if peak < best_score:
            best_score = peak
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempt:5d} n={n:5d}  new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety net – ensure at least one candidate exists
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 5000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n)
        best_score = _max_convolution(best_seq)
        best_n = n
        print("[explore] fallback to uniform default")

    # ------------------------------------------------------------------
    # Phase 2 – long Adam refinement on a large scale (≈70 % of budget)
    # ------------------------------------------------------------------
    max_len = min(80000, best_n * 15)          # generous upper bound
    a_scaled = _scale_sequence(best_seq, max_len)
    a_scaled = np.clip(a_scaled, 0.0, None)
    a_scaled = a_scaled * (math.sqrt(2.0 * max_len) / a_scaled.sum())

    remaining = end_time - time.time()
    adam_deadline = time.time() + remaining * 0.65   # ~65 % of what is left
    print(f"[refine] scaling from {best_n} → {max_len}, "
          f"Adam run until {adam_deadline - time.time():.1f}s left")

    refined, refined_peak = _adam_optimize(a_scaled,
                                           adam_deadline,
                                           rng,
                                           lr0=0.4,
                                           tau_decay=0.9996,
                                           lr_decay=0.99985,
                                           jitter_every=200)

    if refined_peak < best_score:
        best_score = refined_peak
        best_seq = refined.copy()
        best_n = max_len
        print(f"[refine] improved after long Adam: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – cheap stochastic fine‑tuning (≈20 % of budget)
    # ------------------------------------------------------------------
    remaining = end_time - time.time()
    if remaining > 15.0:
        # simulated annealing first
        best_seq, best_score = _anneal(best_seq,
                                       time.time() + 0.6 * remaining,
                                       rng)
        remaining = end_time - time.time()
        # then a strict greedy hill‑climber
        if remaining > 5.0:
            best_seq, best_score = _hill_climb(best_seq,
                                               time.time() + remaining,
                                               rng)

    # ------------------------------------------------------------------
    # Final sanity check via the supplied black‑box evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f}  internal={best_score:.6f}  "
          f"length={len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
