"""
Hybrid optimisation for the self‑convolution max‑norm problem.

The pipeline (all phases respect a shared 1000 s deadline):

1. **Candidate pool** – power‑law, sinusoidal and Gaussian shapes,
   the previous best (if any), plus jittered copies and random
   permutations.  All candidates are projected onto the ℓ¹‑simplex
   {x ≥ 0, Σx = √(2 n)}.

2. **Projected gradient descent (soft‑max)** – momentum optimiser on a
   smooth log‑sum‑exp approximation of the max‑convolution.  The
   temperature β is annealed up to 1e5 and the step size is
   automatically scaled.  Several restarts with random jitter are
   performed; the best result is retained.

3. **Deterministic mass‑transfer** – a greedy Metropolis move that
   transfers a small amount of mass from an index heavily contributing
   to the current peak to a low‑contributing index.  The temperature
   decays exponentially with the remaining time.

4. **Re‑ordering heuristics** – alternating large/small ordering and
   a simple “even‑spacing’’ ordering that distributes large values as
   uniformly as possible.

5. **Permutation annealing** – simulated‑annealing over swaps of two
   positions (ordering only).  Swaps are accepted according to a
   decreasing temperature schedule.

6. **Exact sub‑gradient descent** – after the ordering is near‑optimal,
   the true sub‑gradient of the max‑convolution (indicator of the
   maximal entries) is used with a diminishing step size.  This refines
   the values without the smoothing bias of the soft‑max.

7. **Final high‑β polishing** – a last projected gradient descent with
   β up to 5·10⁶ to squeeze any remaining slack.

All heavy convolutions use FFT (O(n log n)).  The algorithm stops a few
seconds before the global deadline and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators (different shapes)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _sinusoid_sequence(n: int, target_sum: float) -> np.ndarray:
    # Positive half‑wave of sin², shifted to avoid exact zeros at ends
    x = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (x + 0.5) / n) ** 2
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma_frac: float, target_sum: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, decay: float, target_sum: float) -> np.ndarray:
    a = np.exp(-decay * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max)
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """Return (grad, conv) for the log‑sum‑exp approximation of max."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # grad = 2 * (a_rev ⋆ p)
    a_rev = a[::-1]
    ar_fft = np.fft.rfft(a_rev, n=size)
    pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(ar_fft * pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Projected gradient descent with soft‑max annealing
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 1e5,
                               step_factor: float = 0.7,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max approximation.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional tiny jitter to escape shallow plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} best={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform several short GD runs with random jitter; keep the best.
    """
    best_a = init.copy()
    best_val = _max_convolution_fft(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent(
            start,
            time.time() + run_budget,
            target_sum,
            max_beta=1e5,
            step_factor=0.8,
            gamma=0.99,
        )
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val

def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer attacks the current maximal convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]

    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]  # contribution of each index to that peak
        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Re‑ordering heuristics
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place largest, then smallest, then second‑largest, … (alternating)."""
    sorted_vals = np.sort(seq)[::-1]  # descending
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spacing_order(seq: np.ndarray) -> np.ndarray:
    """
    Distribute large values as evenly as possible.
    Largest value goes to position 0, next to position ≈ n/2, then
    n/4, 3n/4, …
    """
    n = seq.shape[0]
    order = np.empty_like(seq)
    sorted_idx = np.argsort(seq)[::-1]   # indices of values descending
    step = max(1, n // len(sorted_idx))
    pos = 0
    for idx in sorted_idx:
        order[pos] = seq[idx]
        pos = (pos + step) % n
    # fill any leftovers (should not happen)
    return order


# ----------------------------------------------------------------------
# Permutation annealing (swap based)
# ----------------------------------------------------------------------


def _permutation_annealing(a: np.ndarray,
                           target_sum: float,
                           deadline: float,
                           init_score: float | None = None) -> tuple[np.ndarray, float]:
    cur_seq = a.copy()
    cur_score = init_score if init_score is not None else _max_convolution_fft(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score

    start_time = time.time()
    n = cur_seq.shape[0]
    rng = np.random.default_rng()

    start_temp = 0.2
    end_temp = 1e-5

    while time.time() < deadline:
        elapsed = time.time() - start_time
        ratio = elapsed / (deadline - start_time) if deadline > start_time else 0.0
        temp = start_temp * (end_temp / start_temp) ** ratio

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # swap
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        new_score = _max_convolution_fft(cur_seq)
        delta = new_score - cur_score

        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / max(temp, 1e-12)):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
                print(f"[{time.time():.2f}] Perm anneal new best={best_score:.6f}")
        else:
            # revert
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        # occasional full reshuffle near the end
        if ratio > 0.9 and rng.random() < 0.01:
            rng.shuffle(cur_seq)
            cur_score = _max_convolution_fft(cur_seq)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         step_factor: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Uses the true sub‑gradient (indicator of the maximal convolution
    entries) with a diminishing step size.  No smoothing bias.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_b = conv.max()
        mask = conv == max_b
        p = mask.astype(np.float64) / mask.sum()  # uniform over maximal positions

        # gradient = 2 * (a_rev ⋆ p)
        a_rev = a[::-1]
        size = 1 << ((2 * n - 1).bit_length())
        ar_fft = np.fft.rfft(a_rev, n=size)
        pr_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(ar_fft * pr_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # momentum with a bit of damping
        momentum = 0.9 * momentum + grad

        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_factor / (norm * (it + 1) ** 0.5)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} best={best_val:.6f}")

        # occasional tiny jitter to avoid stagnation
        if it % 800 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid search for a non‑negative step function with a low
    evaluate_sequence score.  Returns the best sequence found before the
    deadline.
    """
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Problem size (larger n gives more flexibility but costs more time)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Build an initial pool of diverse candidates
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    init_seqs = []

    # power‑law variants
    for p in (0.45, 0.5, 0.55, 0.6):
        init_seqs.append(_power_law_sequence(n, p, target_sum))

    # sinusoidal and Gaussian shapes
    init_seqs.append(_sinusoid_sequence(n, target_sum))
    init_seqs.append(_gaussian_sequence(n, sigma_frac=0.15, target_sum=target_sum))
    init_seqs.append(_gaussian_sequence(n, sigma_frac=0.10, target_sum=target_sum))
    init_seqs.append(_exponential_sequence(n, decay=1.0 / (0.25 * n), target_sum=target_sum))

    # resized previous best if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        init_seqs.append(_resize_sequence(prev, n))

    # jittered copies for diversity
    for seq in list(init_seqs):
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        seq_jit = seq + jitter
        seq_jit = np.maximum(seq_jit, 0.0)
        seq_jit = _project_to_simplex(seq_jit, target_sum)
        init_seqs.append(seq_jit)

    # random permutations of a couple of promising candidates
    for p in (0.5, 0.55):
        seq = _power_law_sequence(n, p, target_sum)
        rng.shuffle(seq)
        init_seqs.append(seq)

    # ------------------------------------------------------------------
    # Phase A – Soft‑max projected gradient descent (majority of time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    gd_deadline = time.time() + remaining * 0.35

    best_seq = None
    best_score = np.inf

    for init in init_seqs:
        if time.time() >= gd_deadline:
            break
        a_cand, val = _gradient_descent_with_restarts(init, target_sum, gd_deadline)
        if val < best_score:
            best_score = val
            best_seq = a_cand.copy()
            print(f"[{time.time():.2f}] New global best after GD = {best_score:.6f}")

    if best_seq is None:
        best_seq = _power_law_sequence(n, 0.55, target_sum)
        best_score = _max_convolution_fft(best_seq)

    # ------------------------------------------------------------------
    # Phase B – Deterministic mass‑transfer refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        mt_deadline = time.time() + (deadline - time.time()) * 0.30
        a_ref, val_ref = _deterministic_mass_transfer(
            best_seq, mt_deadline, target_sum, start_temp=0.5, end_temp=1e-4, low_frac=0.05
        )
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] After DMT best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase C – Re‑ordering heuristics
    # ------------------------------------------------------------------
    reorder_candidates = [best_seq, _alternating_order(best_seq), _even_spacing_order(best_seq)]
    for _ in range(2):
        tmp = best_seq.copy()
        rng.shuffle(tmp)
        reorder_candidates.append(tmp)

    for cand in reorder_candidates:
        cand_score = _max_convolution_fft(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] Ordering improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase D – Permutation annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        perm_deadline = time.time() + (deadline - time.time()) * 0.15
        a_perm, val_perm = _permutation_annealing(
            best_seq, target_sum, perm_deadline, init_score=best_score
        )
        if val_perm < best_score:
            best_score = val_perm
            best_seq = a_perm.copy()
            print(f"[{time.time():.2f}] After perm anneal best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase E – Exact sub‑gradient descent (fine‑tuning)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        subgrad_deadline = time.time() + (deadline - time.time()) * 0.15
        a_sub, val_sub = _subgradient_descent(
            best_seq, target_sum, subgrad_deadline, step_factor=0.6
        )
        if val_sub < best_score:
            best_score = val_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] After sub‑grad best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase F – Final high‑β polishing (very high temperature soft‑max)
    # ------------------------------------------------------------------
    if time.time() + 1.0 < deadline:
        a_final, val_final = _annealed_gradient_descent(
            best_seq,
            deadline,
            target_sum,
            max_beta=5e6,
            step_factor=0.5,
            gamma=0.995,
        )
        if val_final < best_score:
            best_score = val_final
            best_seq = a_final.copy()
            print(f"[{time.time():.2f}] Final polish best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zeros and return
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # optional evaluation output (won't affect the competition harness)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    globals()["height_sequence_1"] = final_seq  # remember for future runs
    return final_seq
