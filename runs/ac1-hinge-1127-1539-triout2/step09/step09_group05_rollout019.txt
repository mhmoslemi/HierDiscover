"""Improved step‑function optimizer.

The new version adds a few carefully chosen parametric families (beta
distributions, half‑cosine, etc.) and replaces the simple Polyak
sub‑gradient step by an Adam‑style adaptive optimiser.  The search
stages are

1. **Broad random exploration** – generate many candidates from
   several families (including a beta‑pdf family) for lengths up to
   25000.  The cheap FFT‑based max‑convolution is used for scoring.
2. **Deterministic beta‑grid sweep** – a shallow grid over symmetric
   beta‑pdfs (α=β) often uncovers shapes with a very flat autocorrelation.
3. **Adam refinement** – an adaptive projected Adam optimiser runs
   on the current best sequence for the bulk of the remaining budget.
4. **Final annealing** – a light simulated‑annealing mass‑transfer
   loop as a finishing touch.
5. **External check** – the provided ``evaluate_sequence`` is called
   once at the end and the result is stored in ``height_sequence_1``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a via a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a∗a using the FFT routine."""
    return _full_convolution_fft(a).max()


def _max_convolution_and_gradient(a: np.ndarray):
    """
    Returns (grad, max_val) where grad is a sub‑gradient of
    maxₖ (a∗a)[k] and max_val is that maximum.
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_val = b.max()
    mask = (b == max_val).astype(float)

    # convolution of mask with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + n - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad, max_val


# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. O(n))
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto {w ≥ 0, Σ w = S}."""
    if S < 0:
        raise ValueError("Simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: scaling a vector to a new length (linear interpolation)
# ----------------------------------------------------------------------
def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]

# Parametric families for initial vectors

def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    off = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + off) ** (-p)
    raw = mix * edge + (1.0 - mix) * np.ones(n)
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    raw = mix * exp_part + (1.0 - mix) * np.ones(n)
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    raw = mix * ush + (1.0 - mix) * np.ones(n)
    return raw


def _init_sine(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    vals = np.sin(np.pi * (i + 1) / (n + 1))
    mix = rng.uniform(0.0, 1.0)
    raw = mix * vals + (1.0 - mix) * np.ones(n)
    return raw


def _init_cosine_half(n: int, rng: np.random.Generator) -> np.ndarray:
    """Half‑cosine shape: f(i)=1‑cos(2π*i/(n‑1)) (zero at ends, peak=2)."""
    i = np.arange(n)
    vals = 1.0 - np.cos(2.0 * np.pi * i / (n - 1))
    mix = rng.uniform(0.0, 1.0)
    raw = mix * vals + (1.0 - mix) * np.ones(n)
    return raw


def _init_beta(n: int, rng: np.random.Generator) -> np.ndarray:
    """Symmetric beta‑pdf (t^{α‑1}(1‑t)^{α‑1}) sampled at mid‑points."""
    alpha = rng.uniform(0.3, 3.0)
    i = np.arange(n)
    # use mid‑point rule to avoid exact zeros at the boundary
    t = (i + 0.5) / n
    pdf = (t ** (alpha - 1.0)) * ((1.0 - t) ** (alpha - 1.0))
    raw = pdf
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Occasionally warm‑starts from the globally best sequence; otherwise
    picks a random parametric family (now including a beta‑pdf family).
    """
    S = np.sqrt(2.0 * n)

    # Warm start with probability ≈0.22
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
        return a

    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'ushape', 'sine', 'cosine', 'beta', 'uniform'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'sine':
        raw = _init_sine(n, rng)
    elif style == 'cosine':
        raw = _init_cosine_half(n, rng)
    elif style == 'beta':
        raw = _init_beta(n, rng)
    else:
        raw = _init_uniform(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a

def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   best_val: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on f(a)=max(a∗a).  The method runs until the
    time limit and returns the best vector seen.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_a = a.copy()
    cur_val = _max_convolution(a)
    best_score = cur_val

    while time.time() < deadline:
        t += 1
        grad, cur_val = _max_convolution_and_gradient(a)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size – tuned for this problem
        step = 0.1 / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_onto_simplex(a, S)

        cur_val = _max_convolution(a)
        if cur_val < best_score:
            best_score = cur_val
            best_a = a.copy()

        # occasional tiny jitter to escape plateaus
        if t % 300 == 0:
            jitter = rng.normal(scale=0.001 * np.mean(a), size=n)
            a = _project_onto_simplex(a + jitter, S)

    return best_a, best_score


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         global_best: float) -> tuple[np.ndarray, float]:
    """
    Simple Polyak‑step sub‑gradient descent (kept for occasional use).
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_val = _max_convolution_and_gradient(a)

        norm_sq = np.dot(grad, grad)
        if norm_sq > 0:
            step = (cur_val - best_val) / (norm_sq + 1e-12)
            step = np.clip(step, 1e-7, 0.2)
        else:
            step = 0.0

        a = _project_onto_simplex(a - step * grad, S)

        cur_val = _max_convolution(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()

        if it % 300 == 0:
            jitter = rng.normal(scale=0.001 * np.mean(a), size=n)
            a = _project_onto_simplex(a + jitter, S)

    return best_a, best_val


def _anneal_local_search(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Mass‑transfer simulated‑annealing hill‑climb."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    cur_val = _max_convolution(a)
    best_a = a.copy()
    best_val = cur_val

    T = 0.05 * cur_val
    decay = 0.9999

    it = 0
    while time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        frac = rng.uniform(0.02, 0.18)
        delta = a[i] * frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _max_convolution(a_new)
        dval = new_val - cur_val
        if dval < 0.0 or rng.random() < np.exp(-dval / (T + 1e-12)):
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()

        T *= decay
        if it % 500 == 0:
            jitter = rng.normal(scale=0.001 * S, size=n)
            a = _project_onto_simplex(a + jitter, S)
            cur_val = _max_convolution(a)

    return best_a, best_val

                                    best_score)
        if val < best_score:
            best_score = val
            best_seq = a_opt.copy()
            print(f"[adam] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – fallback sub‑gradient refinement (short burst)
    # ------------------------------------------------------------------
    if time.time() < end_time - 80.0:
        remaining = end_time - time.time() - 5.0
        a_opt, val = _subgradient_descent(best_seq,
                                          time.time() + remaining,
                                          rng,
                                          best_score)
        if val < best_score:
            best_score = val
            best_seq = a_opt.copy()
            print(f"[subgrad] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – simulated‑annealing hill‑climb (final polish)
    # ------------------------------------------------------------------
    if time.time() < end_time - 20.0:
        remaining = end_time - time.time() - 5.0
        a_opt, val = _anneal_local_search(best_seq,
                                          time.time() + remaining,
                                          rng)
        if val < best_score:
            best_score = val
            best_seq = a_opt.copy()
            print(f"[anneal] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f}  internal={best_score:.6f}  length={len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
