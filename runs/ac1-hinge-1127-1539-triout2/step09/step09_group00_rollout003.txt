"""
Improved step‑function optimizer.

The goal is to minimise

    2·n·max(a*a) / (sum(a)^2)

subject to a_i ≥ 0 and sum(a)=√(2n).

The algorithm consists of three stages:

1️⃣  **Fast parametric sweep** – generate a diverse set of candidates:
   * pure power‑law,
   * power‑law with an exponential tail,
   * a simple plateau‑+‑exponential‑tail family (new),
   * jittered variants of all of the above.
   The best among these seeds the next stage.

2️⃣  **Adam‑based projected gradient descent** on a soft‑max
   approximation of the max‑convolution (β is annealed up to 1e7).
   Several restarts with random jitter are performed while time permits.

3️⃣  **Deterministic mass‑transfer refinement** (greedy moves that
   directly reduce the current maximal convolution entry) followed
   by a short final Adam polishing.

All vectors are kept on the simplex {x ≥ 0, Σx = √(2n)} after each update.
The routine respects the global time budget and returns the best sequence
found, trimmed of trailing zeros.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Benchmark objective 2·n·max(a*a) / (sum(a)^2)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) scaled to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) * exp(-q * i / n) scaled to target_sum."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _plateau_tail_sequence(n: int,
                           plateau_frac: float,
                           decay: float,
                           target_sum: float) -> np.ndarray:
    """
    Plateau of constant height for the first `plateau_frac*n` entries,
    followed by a geometric decay with factor `decay` (0<decay<1).
    Scaled so that the sum equals target_sum.
    """
    k = int(np.clip(plateau_frac, 0.0, 1.0) * n)
    tail_len = n - k
    if tail_len == 0 or decay >= 1.0:
        # pure plateau (uniform)
        a = np.full(n, target_sum / n, dtype=np.float64)
        return a

    # denominator = k + (1 - decay**tail_len) / (1 - decay)
    denom = k + (1.0 - decay ** tail_len) / (1.0 - decay)
    const = target_sum / denom
    a = np.empty(n, dtype=np.float64)
    a[:k] = const
    if tail_len > 0:
        a[k:] = const * (decay ** np.arange(tail_len, dtype=np.float64))
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution (used by GD / Adam)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where conv is the full convolution a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    # stable soft‑max
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    p_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   target_sum: float,
                   max_beta: float = 1e7,
                   init_beta: float = 0.5,
                   step_factor: float = 0.5,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected gradient descent on the soft‑max of the max‑convolution.
    Returns (best_vector, best_score).
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = init_beta
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    step_base = target_sum * step_factor

    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # direction normalised to avoid huge jumps
        direction = m_hat / (np.sqrt(v_hat) + eps)
        norm = np.linalg.norm(direction)
        if norm > 0.0:
            lr = step_base / (beta + 1.0) / norm
            a -= lr * direction

        # projection onto simplex
        a = _project_to_simplex(a, target_sum)

        # occasional small jitter to escape plateaus
        if t % 1000 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        # anneal β (exponential schedule)
        if t % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.6, max_beta)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={t:5d} β={beta:6.1f} score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (greedy reduction of max entry)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-6,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer that directly attacks the current maximal convolution entry.
    Uses a Metropolis rule with an exponentially decaying temperature.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]
        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0.0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0.0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        # Metropolis acceptance
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter
        if it % 3000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Global random mass‑transfer (exploration)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing moves that shift a small amount of mass."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()

    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 0.5:
        # donor from top fraction
        k_top = max(1, int(top_frac * n))
        donor = np.argsort(-a)[rng.integers(k_top)]

        # receiver from bottom fraction
        k_low = max(1, int(low_frac * n))
        receiver = np.argsort(a)[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine respects the time budget, prints progress, and returns
    the best sequence found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose problem size (larger n gives more degrees of freedom)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 14000
    elif budget_s >= 400:
        n = 10000
    else:
        n = 6000

    target_sum = np.sqrt(2.0 * n)   # fixed Σa (scale‑invariant)

    rng = np.random.default_rng()
    candidates = []

    # ------------------------------------------------------------------
    # Phase 0 – fast parametric pre‑search (diverse families)
    # ------------------------------------------------------------------
    # (a) pure power‑law
    for p in np.linspace(0.30, 0.80, 15):
        candidates.append(_power_law_sequence(n, p, target_sum))

    # (b) power‑law + exponential tail
    for p in np.linspace(0.30, 0.80, 8):
        for q in np.linspace(0.0, 5.0, 8):
            candidates.append(_exp_power_law_sequence(n, p, q, target_sum))

    # (c) plateau + exponential tail (new family)
    for pf in np.linspace(0.0, 0.40, 9):          # plateau fraction
        for decay in np.linspace(0.85, 0.997, 12):  # decay factor
            candidates.append(_plateau_tail_sequence(n, pf, decay, target_sum))

    # (d) previous best if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        candidates.append(_resize_sequence(prev, n))

    # (e) jittered variants of everything above
    for seq in list(candidates):
        jitter = rng.normal(scale=target_sum * 0.006, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        candidates.append(seq_j)

    # limited cheap scoring pass
    max_initial = min(250, len(candidates))
    sample_idx = rng.choice(len(candidates), size=max_initial, replace=False)

    best_seq = None
    best_score = np.inf
    for idx in sample_idx:
        seq = candidates[idx]
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
    print(f"[{time.time():.2f}] Initial best after parametric search = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam gradient descent with restarts
    # ------------------------------------------------------------------
    # Allocate roughly 55 % of remaining time to this phase
    phase_a_deadline = time.time() + (deadline - time.time()) * 0.55
    while time.time() < phase_a_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.02, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        # each run gets a slice of the remaining time
        run_deadline = min(phase_a_deadline,
                           time.time() + min(35.0,
                                            (deadline - time.time()) * 0.30))
        a_opt, val_opt = _adam_optimize(start_vec,
                                         run_deadline,
                                         target_sum,
                                         max_beta=1e7,
                                         step_factor=0.5)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Adam restart improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Deterministic mass‑transfer + Global random moves
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # deterministic refinement (takes a chunk of the remaining budget)
        dmt_deadline = time.time() + min(30.0, (deadline - time.time()) * 0.30)
        a_ref, val_ref = _deterministic_mass_transfer(best_seq,
                                                      dmt_deadline,
                                                      target_sum,
                                                      start_temp=0.5,
                                                      end_temp=1e-6,
                                                      low_frac=0.07)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] DMT improved best = {best_score:.6f}")

        # a short burst of global random moves
        gr_deadline = time.time() + min(12.0, (deadline - time.time()) * 0.15)
        a_gr, val_gr = _global_random_mass_transfer(best_seq,
                                                    gr_deadline,
                                                    target_sum)
        if val_gr < best_score:
            best_score = val_gr
            best_seq = a_gr.copy()
            print(f"[{time.time():.2f}] Global random improved best = {best_score:.6f}")

        # early exit guard
        if time.time() > deadline - 2.0:
            break

    # ------------------------------------------------------------------
    # Phase C – Final polishing (short Adam run with very high β)
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_optimize(best_seq,
                                         deadline,
                                         target_sum,
                                         max_beta=5e7,
                                         step_factor=0.4)
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing: trim trailing zeros and return as Python list
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final evaluation (harmless if evaluate_sequence is not defined)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
