"""
Improved step‑function search.

The algorithm consists of three phases:

1️⃣  **Random exploration** – many short runs from several parametric families
    (power‑law, exponential, linear, U‑shape, sine, uniform, Dirichlet) are
    generated with lengths between 3 000 and 25 000.  The best candidate
    (lowest convolution peak) is kept.

2️⃣  **Smooth max‑gradient descent** – the non‑smooth objective
    `max_k (a∗a)[k]` is approximated by a log‑sum‑exp:
        f_ε(a) = ε·log Σ_k exp( (a∗a)[k] / ε )
    Its gradient is cheap to compute via FFT:
        ∇f_ε(a) = 2·conv(p , a_rev)[n‑1 : n‑1+n] ,
    where `p` are the soft‑max weights of the convolution entries.
    An Adam optimiser with projection onto the simplex
    `{a_i≥0, Σ a_i = √(2 n)}` is run for a decreasing schedule of ε.
    The method provides a dense, informative descent direction (unlike a
    pure sub‑gradient that only looks at the max entries) and converges
    quickly to a much lower peak.

3️⃣  **Length scaling & final refinement** – the current best shape is
    interpolated to larger lengths (up to 25 000) and the smooth descent
    is repeated.  A short projected sub‑gradient pass (the classic
    max‑only method) finishes the optimisation.

The whole routine respects the 1000 s budget, prints intermediate
bounds and stores the final sequence in the global variable
`height_sequence_1`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Duchi et al.)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Project v onto {w ≥ 0, Σ w = S}."""
    if S < 0:
        raise ValueError("Simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho_arr = u - cssv / np.arange(1, len(v) + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Parametric families for random initialisation
# ----------------------------------------------------------------------
def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    return mix * edge + (1.0 - mix) * base


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    return mix * exp_part + (1.0 - mix) * base


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    return mix * lin + (1.0 - mix) * base


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    return mix * ush + (1.0 - mix) * base


def _init_sine(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    vals = np.sin(np.pi * (i + 1) / (n + 1))
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * vals + (1.0 - mix) * base


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    return rng.random(n)


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    20 % of the time we warm‑start from an existing good sequence,
    otherwise we draw from a random parametric family.
    """
    S = np.sqrt(2.0 * n)

    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
        return a

                        'ushape', 'sine', 'uniform', 'dirichlet'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'sine':
        raw = _init_sine(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) gradient
# ----------------------------------------------------------------------
def _smooth_max_and_gradient(a: np.ndarray, eps: float):
    """
    Returns (gradient, max_val, full_conv) where gradient is the
    gradient of the smooth approximation of max_k (a∗a)[k] with
    smoothing parameter eps.
    """
    n = a.shape[0]

    # full convolution a∗a
    B = _full_convolution_fft(a)                     # length 2n‑1
    B_max = B.max()

    # soft‑max weights (stable)
    z = (B - B_max) / eps
    exp_z = np.exp(z)
    sum_exp = exp_z.sum()
    p = exp_z / sum_exp                               # length 2n‑1, sums to 1

    # gradient = 2 * conv(p , a_rev) restricted to centre window
    a_rev = a[::-1]
    conv_len = len(p) + n - 1                         # = 3n‑2
    f_len = _next_pow2(conv_len)

    Fp = np.fft.rfft(p, n=f_len)
    Fa = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fp * Fa, n=f_len)[:conv_len]

    start = n - 1
    grad = 2.0 * conv[start:start + n]

    return grad, B_max, B


# ----------------------------------------------------------------------
# Adam optimiser with simplex projection
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   eps_schedule,
                   lr0: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed max objective.
    Returns the best vector seen and its hard max value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # initialise Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    t = 0

    # best tracking (hard max)
    B = _full_convolution_fft(a)
    best_val = B.max()
    best_a = a.copy()

    for eps in eps_schedule:
        # shrink learning rate a little when we tighten epsilon
        lr = lr0 * (eps / eps_schedule[0])

        while time.time() < deadline:
            t += 1
            # gradient of smoothed max
            grad, cur_max, _ = _smooth_max_and_gradient(a, eps)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            a = a - step
            a = _project_onto_simplex(a, S)

            # evaluate hard max for this new point
            B_new = _full_convolution_fft(a)
            max_new = B_new.max()
            if max_new < best_val - 1e-12:
                best_val = max_new
                best_a = a.copy()

            # occasional jitter to escape tiny plateaus
            if t % 400 == 0:
                jitter = np.random.default_rng().normal(scale=0.001 * S,
                                                        size=n)
                a = _project_onto_simplex(a + jitter, S)

            # stop inner loop early if we are close to deadline
            if time.time() + 0.5 > deadline:
                break
        # after finishing an epsilon, inject a tiny random shake
        if time.time() + 1.0 < deadline:
            shake = np.random.default_rng().normal(scale=0.0005 * S,
                                                   size=n)
            a = _project_onto_simplex(a + shake, S)

    return best_a, best_val

# Classic max‑only sub‑gradient descent (used for final polishing)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         global_best: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # initialise best values
    best_a = a.copy()
    best_val = _full_convolution_fft(a).max()

    it = 0
    while time.time() < deadline:
        it += 1

        # ---- max‑only sub‑gradient ----
        B = _full_convolution_fft(a)
        cur_val = B.max()
        mask = (B == cur_val).astype(float)          # hard mask
        a_rev = a[::-1]
        conv_len = len(mask) + n - 1
        f_len = _next_pow2(conv_len)
        Fmask = np.fft.rfft(mask, n=f_len)
        Fa_rev = np.fft.rfft(a_rev, n=f_len)
        conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
        start = n - 1
        grad = 2.0 * conv[start:start + n]

        # Polyak step size
        norm_sq = np.dot(grad, grad)
        if norm_sq > 0:
            step = (cur_val - best_val) / (norm_sq + 1e-12)
            step = np.clip(step, 1e-7, 0.15)
        else:
            step = 0.0

        a = _project_onto_simplex(a - step * grad, S)

        # evaluate new point
        new_val = _full_convolution_fft(a).max()
        if new_val + 1e-12 < best_val:
            best_val = new_val
            best_a = a.copy()

        # occasional random jitter
        if it % 300 == 0:
            jitter = np.random.default_rng().normal(scale=0.001 * S,
                                                    size=n)
            a = _project_onto_simplex(a + jitter, S)

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine follows the three‑phase plan described in the module docstring.
    The final sequence is stored in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    end_time = start_time + budget_s - 3.0   # leave a small safety margin

    # ------------------------------------------------------------------
    # Warm‑start handling
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # ------------------------------------------------------------------
    # Phase 1 – broad random exploration
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    best_n = None
    attempts = 0

    explore_deadline = end_time - 250.0   # reserve time for optimisation phases
    while time.time() < explore_deadline:
        attempts += 1
        # favour longer sequences, but keep variety
        if rng.random() < 0.7:
            n = int(rng.integers(5000, 18000))
        else:
            n = int(rng.integers(2000, 5000))

        a0 = _initial_vector(n, rng, warm_seq)
        cur_val = _full_convolution_fft(a0).max()
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempts:4d} n={n:5d}  best={best_val:.6f}")

    # fallback if nothing was generated (should not happen)
    if best_seq is None:
        n = 4000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_val = _full_convolution_fft(best_seq).max()
        best_n = n

    # ------------------------------------------------------------------
    # Phase 2 – smooth‑max Adam optimisation (including length scaling)
    # ------------------------------------------------------------------
    eps_schedule = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.001]
    length_factors = [1.0, 1.3, 1.6, 2.0]   # progressive up‑scaling

    for factor in length_factors:
        if time.time() >= end_time - 150.0:
            break
        target_n = int(best_n * factor)
        if target_n < 3000:
            target_n = 3000
        if target_n > 25000:
            target_n = 25000

        # resize and renormalise
        a_scaled = _scale_sequence(best_seq, target_n)
        S = np.sqrt(2.0 * target_n)
        a_scaled = np.clip(a_scaled, 0.0, None)
        a_scaled = a_scaled * (S / a_scaled.sum())

        # allocate a time slice for this length
        remaining = end_time - time.time()
        slice_time = min(remaining * 0.45, 80.0)   # generous slice

        a_opt, val = _adam_optimize(a_scaled,
                                    time.time() + slice_time,
                                    eps_schedule,
                                    lr0=0.3)
        if val < best_val:
            best_val = val
            best_seq = a_opt.copy()
            best_n = target_n
            print(f"[adam]  length={target_n:5d}  improved={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final max‑only sub‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < end_time - 30.0:
        remaining = end_time - time.time() - 5.0
        a_opt, val = _subgradient_descent(best_seq,
                                          time.time() + remaining,
                                          best_val)
        if val < best_val:
            best_val = val
            best_seq = a_opt.copy()
            print(f"[final subgrad] length={len(best_seq)}  best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f}  internal={best_val:.6f}  length={len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
