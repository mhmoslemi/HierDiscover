def solve_lp_given_f(f, M):
    n = len(f)
    b = cp.Variable(n)
    constraints = [b >= 0]
    conv_fb = cp.conv(f, b)  # returns length 2n-1
    constraints += [conv_fb <= M]  # elementwise inequality
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    if b.value is None:
        return None
    else:
        return np.array(b.value)

"""
propose_candidate
-----------------

Hybrid search for the step‑function that minimizes the evaluator
``evaluate_sequence`` (i.e. the peak of the self‑convolution under a fixed
ℓ₁‑norm).  The algorithm proceeds in several stages:

1. **Random families** – generate many feasible candidates of moderate
   length (1500‑2500) using a variety of parametric families and keep the
   best raw max‑convolution value.

2. **Linear‑program refinement** – for the current best vector `f`,
   solve a linear program
        maximise   ∑ g_j
        s.t.      g ≥ 0,  f ∗ g ≤ M,
   where `M = max(f∗f)`.  The solution `g` is renormalised to the required
   ℓ₁‑norm and mixed with `f` (`f ← (1‑t)f + t g`).  Repeating a few
   iterations monotonically reduces the peak.

3. **Upscaling** – linearly interpolate the improved shape to a larger
   length (up to ≈ 6000) and renormalise.

4. **Continuous optimisation** – run several independent Adam
   (soft‑max mirror descent) and hard‑max sub‑gradient runs on the
   upscaled vector.

5. **Cheap hill‑climbing** – stochastic mass swaps for a final polish.

All phases respect the overall 1000 s budget and the best sequence is
stored in the global variable ``height_sequence_1``.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Basic convolution utilities (FFT based)
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a∗a."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation (preserves shape)."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ------------------------------------------------------------
# Parametric families – used for random initial vectors
# ------------------------------------------------------------
def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    return mix * edge + (1.0 - mix) * base


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    return mix * exp_part + (1.0 - mix) * base


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    return mix * lin + (1.0 - mix) * base


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    return mix * ush + (1.0 - mix) * base


def _init_cosine(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    raw = 1.0 - np.cos(np.pi * i / (n - 1))
    raw = raw * rng.uniform(0.5, 2.0)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * raw + (1.0 - mix) * base


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    return rng.random(n)


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.05, 0.9)
    return rng.gamma(shape, 1.0, size=n)


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """Return a feasible vector of length n (non‑negative, ℓ₁ = √(2n))."""
    S = np.sqrt(2.0 * n)

    # occasional warm‑start
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
        return a

    style = rng.choice([
        'powerlaw', 'exponential', 'linear',
        'gamma', 'uniform', 'dirichlet',
        'ushape', 'cosine'
    ])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'cosine':
        raw = _init_cosine(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ------------------------------------------------------------
# Adam optimiser on the τ‑soft‑max (mirrors the previous code)
# ------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """Gradient of the τ‑soft‑max of the self‑convolution."""
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient = 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Adam on the simplex (log‑softmax parametrisation)."""
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                     # required ℓ₁‑norm

    # initialise logits from feasible a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                         # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.45

    # start temperature
    tau = max(_full_convolution_fft(a0).max(), 5.0)

    best_a = a0.copy()
    best_peak = _max_convolution(a0)
    it = 0

    while time.time() < deadline:
        it += 1

        # recover a from logits
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # gradient of τ‑soft‑max
        g, peak = _softmax_gradient(a, tau)

        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # chain rule to logits
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # optional gradient clipping
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 6.0 * S:
            grad_p = grad_p * (6.0 * S / clip_val)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # schedules (slow decay)
        tau *= 0.99995
        lr *= 0.99995

        # occasional jitter
        if it % 800 == 0:
            p += rng.normal(scale=0.006, size=n)

    return best_a, best_peak


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Hard‑max sub‑gradient descent (projected onto the simplex)."""
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    a = a0.copy()
    best_a = a.copy()
    best_val = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1

        b = _full_convolution_fft(a)
        k = int(b.argmax())
        val = b[k]

        if val < best_val:
            best_val = val
            best_a = a.copy()

        # sub‑gradient of entry k: g_i = 2 * a_{k-i}
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        jdx = k - idx
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[jdx]

        step = 0.8 * S / (np.sqrt(it) + 1.0)
        a = a - step * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, S / n)
        else:
            a = a * (S / s)

        # jitter occasionally
        if it % 500 == 0:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                frac = rng.uniform(0.02, 0.12)
                amount = a[i] * frac
                a[i] -= amount
                a[j] += amount
                a = a * (S / a.sum())

    return best_a, best_val


# ------------------------------------------------------------
# Linear‑program step (LP refinement)
# ------------------------------------------------------------
def _lp_refine_step(f: np.ndarray,
                    M: float,
                    t_mix: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float] | tuple[None, None]:
    """One LP‑based refinement: maximise ∑ g_j subject to f∗g ≤ M."""
    try:
        import cvxpy as cp
    except Exception:
        return None, None

    n = len(f)
    S = np.sqrt(2.0 * n)

    # variable
    g = cp.Variable(n, nonneg=True)

    # constant parameter for f
    f_param = cp.Parameter(n, value=f)

    # convolution (linear in g)
    conv_fg = cp.conv(f_param, g)          # length 2n‑1
    constraints = [conv_fg <= M]

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:
        return None, None

    if g.value is None:
        return None, None

    g_vec = np.array(g.value, dtype=float)
    # numerical cleanup
    g_vec = np.maximum(g_vec, 0.0)
    total = g_vec.sum()
    if total == 0.0:
        return None, None
    g_scaled = g_vec * (S / total)

    # convex combination (maintains ℓ₁ = S)
    f_new = (1.0 - t_mix) * f + t_mix * g_scaled
    return f_new, g_scaled

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`. Returns the best sequence as a Python list and
    stores it in the global variable `height_sequence_1`.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # Warm‑start (if any)
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # ------------------------------------------------------------------
    # Phase 1 – random exploration on moderate lengths (≈ 20 % of budget)
    # ------------------------------------------------------------------
    explore_deadline = end_time - 250.0   # reserve ~250 s for later phases
    best_seq = None
    best_score = np.inf
    best_n = None
    attempts = 0

    while time.time() < explore_deadline:
        attempts += 1
        # moderate lengths give a tractable LP later
        n = int(rng.integers(1500, 2500))
        a0 = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a0)
        if peak < best_score:
            best_score = peak
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempts:4d} n={n:4d}  max={best_score:.6f}")

    # fallback
    if best_seq is None:
        best_n = 2000
        best_seq = np.full(best_n, np.sqrt(2.0 * best_n) / best_n)
        best_score = _max_convolution(best_seq)

    # ------------------------------------------------------------------
    # Phase 2 – LP refinement (few iterations)
    # ------------------------------------------------------------------
    lp_deadline = end_time - 150.0   # keep ~150 s for gradient phases
    t_mix = 0.45                     # mixing coefficient (moderate)
    lp_iter = 0
    while time.time() < lp_deadline:
        M = _max_convolution(best_seq)
        f_new, _ = _lp_refine_step(best_seq, M, t_mix, rng)
        if f_new is None:
            break
        new_M = _max_convolution(f_new)
        if new_M < best_score - 1e-7:
            best_seq = f_new
            best_score = new_M
            lp_iter += 1
            print(f"[LP] iter={lp_iter:2d}  max={best_score:.6f}")
            # slowly reduce mixing to fine‑tune
            t_mix = max(0.15, t_mix * 0.92)
        else:
            # no improvement – stop early
            break

    # ------------------------------------------------------------------
    # Phase 3 – upscale to larger length (if time permits)
    # ------------------------------------------------------------------
    upscale_deadline = end_time - 120.0
    cur_n = best_n
    cur_seq = best_seq
    cur_score = best_score

    if time.time() < upscale_deadline:
        # try a few scaling factors (1.5×, 2×, 3×) and keep the best
        for factor in [1.5, 2.0, 2.8]:
            if time.time() > upscale_deadline:
                break
            L = int(cur_n * factor)
            if L < 2000 or L > 8000:
                continue
            a_scaled = _scale_sequence(cur_seq, L)
            a_scaled = np.clip(a_scaled, 0.0, None)
            S_new = np.sqrt(2.0 * L)
            a_scaled = a_scaled * (S_new / a_scaled.sum())
            peak = _max_convolution(a_scaled)
            if peak < cur_score:
                cur_seq = a_scaled
                cur_score = peak
                cur_n = L
                print(f"[scale] length={L:4d}  improved max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Adam optimisation (multiple restarts)
    # ------------------------------------------------------------------
    remaining = end_time - time.time() - 50.0   # keep ~50 s for sub‑grad & hill
    if remaining > 20.0:
        n_restart = max(1, int(remaining // 30))
        per_run = remaining / n_restart
        for r in range(n_restart):
            if time.time() > end_time - 55.0:
                break
            # perturb start a little to diversify
            noise = rng.normal(scale=0.018 * np.sqrt(2.0 * cur_n) / cur_n, size=cur_n)
            a_start = np.maximum(cur_seq + noise, 0.0)
            a_start = a_start * (np.sqrt(2.0 * cur_n) / a_start.sum())
            a_opt, peak = _adam_optimize(a_start,
                                          time.time() + per_run,
                                          rng)
            if peak < cur_score:
                cur_seq = a_opt
                cur_score = peak
                print(f"[adam] restart={r}  new max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – Sub‑gradient descent (final polishing)
    # ------------------------------------------------------------------
    remaining = end_time - time.time() - 5.0
    if remaining > 5.0:
        a_opt, peak = _subgradient_descent(cur_seq,
                                           time.time() + remaining,
                                           rng)
        if peak < cur_score:
            cur_seq = a_opt
            cur_score = peak
            print(f"[subgrad] final max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – cheap stochastic hill‑climb (few seconds)
    # ------------------------------------------------------------------
    while time.time() < end_time - 1.0:
        a = cur_seq.copy()
        n = a.shape[0]
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        frac = rng.uniform(0.02, 0.13)
        amount = a[i] * frac
        a[i] -= amount
        a[j] += amount
        peak = _max_convolution(a)
        if peak + 1e-9 < cur_score:
            cur_seq = a
            cur_score = peak
            print(f"[hill] improved max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Final check via the external evaluator
    # ------------------------------------------------------------------
    final_seq = cur_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f}  internal max={cur_score:.6f}  length={len(final_seq)}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
