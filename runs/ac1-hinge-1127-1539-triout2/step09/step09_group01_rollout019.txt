"""
Improved step‑function search.

The algorithm proceeds as follows:

1. **Problem size** – Choose a length `n` proportional to the time budget.
2. **Candidate pool** – Generate many simple shapes:
   * power‑law families a_i = (i+1)^(-p) for a fine grid of exponents,
   * a geometric family a_i ∝ exp(-x·i) where `x` is optimised by a fast
     golden‑section search on the exact convolution maximum,
   * jittered copies of the above and the previous best (if any).
   All candidates are projected onto the ℓ¹‑simplex
   {x ≥ 0, Σx = √(2 n)}.
3. **Fast greedy pre‑selection** – Evaluate every candidate with the
   exact convolution (FFT) and keep the one with the smallest score.
4. **Projected gradient descent (soft‑max)** – A momentum optimiser
   minimises a smooth approximation of `max(a*a)`.  The temperature β
   is annealed up to 5·10⁵ and many short restarts with jitter are
   performed.
5. **Deterministic mass‑transfer** – Repeatedly move a small amount of
   mass from the index that contributes most to the current peak to a
   low‑contributing index.  The step size follows a cooling schedule.
6. **Permutation annealing** – Simulated annealing over swaps of two
   positions improves the ordering (the multiset of heights stays the
   same).  Acceptance follows the usual Metropolis rule.
7. **Final polishing** – A last high‑β gradient descent cleans up any
   remaining irregularities.
8. **Post‑processing** – Trailing zeros are removed; the final sequence
   is stored globally (`height_sequence_1`) for reuse in the next run.

The whole pipeline respects the 1000 s time limit.  The combination of
a richer initial pool, a tighter geometric search and more time spent

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Equation used by the competition harness (with Σa = √(2 n)
    the score equals the maximal convolution entry)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – Ω = {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the ℓ¹‑simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple families of candidate sequences
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i = (i+1)^(-p)  (i starts from 0) and normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _geometric_sequence(x: float, n: int, target_sum: float) -> np.ndarray:
    """
    a_i = exp(-x·i)  (i = 0 … n‑1)  and normalised to target_sum.
    Parameter x = -log(r)  (>0).  The case x→0 corresponds to the uniform
    distribution.
    """
    # unnormalised values
    i = np.arange(n, dtype=np.float64)
    a_raw = np.exp(-x * i)

    # normalisation factor (stable using expm1)
    # sum_{i=0}^{n-1} exp(-x i) = (1 - exp(-n x)) / (1 - exp(-x))
    # handle x≈0 with series expansion (expm1)
    if x < 1e-12:
        sum_raw = n
    else:
        sum_raw = np.expm1(-n * x) / np.expm1(-x)   # both negative → positive
    a = a_raw * (target_sum / sum_raw)
    return a


def _geom_score(x: float, n: int, target_sum: float) -> float:
    """
    Exact maximal convolution entry for a geometric sequence with decay
    parameter x ( = -log r).  Returned value is the *score* (max_b) because
    the sum is fixed to target_sum.
    """
    if x <= 0.0:
        # uniform case – max_b = 2
        return 2.0

    # D = (1‑e^{-x})/(1‑e^{-n x})   –   stable via expm1
    D = np.expm1(-x) / np.expm1(-n * x)    # both numerators are negative
    C2 = 2.0 * n * D * D                   # C² = 2 n D²

    k = np.arange(2 * n - 1, dtype=np.float64)
    r_pow = np.exp(-x * k)

    # triangular weighting: 1,2,…,n,…,2,1
    s = np.concatenate([np.arange(1, n + 1), np.arange(n - 1, 0, -1)])
    vals = C2 * r_pow * s
    return float(vals.max())


def _best_geometric_sequence(n: int, target_sum: float, deadline: float):
    """
    Golden‑section search for the decay parameter x that minimises the
    geometric‑shape score.  Returns (sequence, score).
    """
    # reasonable search interval – the optimum is always < 0.001 for our n
    lo, hi = 1e-12, 0.001
    phi = (1 + 5 ** 0.5) / 2.0
    best_x = None
    best_val = np.inf

    # pre‑compute once in case the deadline is very short
    while time.time() < deadline and hi - lo > 1e-12:
        m1 = hi - (hi - lo) / phi
        m2 = lo + (hi - lo) / phi

        v1 = _geom_score(m1, n, target_sum)
        v2 = _geom_score(m2, n, target_sum)

        if v1 < v2:
            hi = m2
            if v1 < best_val:
                best_val, best_x = v1, m1
        else:
            lo = m1
            if v2 < best_val:
                best_val, best_x = v2, m2

    # fallback – uniform sequence if the optimisation never ran
    if best_x is None:
        seq = np.full(n, target_sum / n, dtype=np.float64)
        best_val = _score(seq)
        return seq, best_val

    seq = _geometric_sequence(best_x, n, target_sum)
    # a final projection guards against tiny rounding errors
    seq = _project_to_simplex(seq, target_sum)
    return seq, best_val


# ----------------------------------------------------------------------
# Soft‑max gradient of max‑convolution (used by the gradient descent)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * (a reversed) ⋆ p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – Momentum projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.6,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max approximation.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Multiple short gradient‑descent runs with random jitter.
    Keeps the best result before the deadline.
    """
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent(start,
                                            time.time() + run_budget,
                                            target_sum,
                                            max_beta=5e5,
                                            step_factor=0.8,
                                            gamma=0.99)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer moves that directly attack the current
    maximal convolution entry.  Moves are accepted with an annealed
    Metropolis rule.
    """
    a = a0.copy()
    n = a.shape[0]

    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        # temperature schedule (exponential decay)
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # index where convolution is maximal
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # w_i = a_i + a_{kmax-i}
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        # transfer amount (bounded)
        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        # recompute convolution and score
        conv = _conv_fft(a)
        new_score = _score(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter to shake the system
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Re‑ordering utilities
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """
    Place the largest value on the leftmost position,
    the second largest on the rightmost, the third on the next left,
    etc.  This is a simple “big–small’’ pattern known to help the
    convolution peak.
    """
    sorted_vals = np.sort(seq)[::-1]          # descending
    n = seq.shape[0]
    out = np.empty_like(seq)
    left = 0
    right = n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


# ----------------------------------------------------------------------
# Phase D – Permutation annealing (swap‑based simulated annealing)
# ----------------------------------------------------------------------


def _permutation_annealing(a: np.ndarray,
                           target_sum: float,
                           deadline: float,
                           init_score: float | None = None) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing over swaps of two indices.  The ordering only is
    altered; the multiset of heights stays unchanged.
    """
    cur_seq = a.copy()
    cur_score = init_score if init_score is not None else _max_convolution_fft(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score

    start_time = time.time()
    n = cur_seq.shape[0]
    rng = np.random.default_rng()

    start_temp = 0.3
    end_temp = 1e-5

    while time.time() < deadline:
        elapsed = time.time() - start_time
        ratio = elapsed / (deadline - start_time) if deadline > start_time else 0.0
        temp = start_temp * (end_temp / start_temp) ** ratio

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # swap
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        new_score = _max_convolution_fft(cur_seq)
        delta = new_score - cur_score

        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / max(temp, 1e-12)):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
                print(f"[{time.time():.2f}] Perm anneal new best = {best_score:.6f}")
        else:
            # revert
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        # occasional global reshuffle near the end
        if ratio > 0.9 and rng.random() < 0.02:
            rng.shuffle(cur_seq)
            cur_score = _max_convolution_fft(cur_seq)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point – the competition harness will call this
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The procedure follows the outline described at the top of the file.
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n gives a richer search space)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Build a diverse pool of simple candidates
    # --------------------------------------------------------------
    init_seqs = []

    # fine grid of power‑law exponents
    for p in np.linspace(0.45, 0.65, 21):
        init_seqs.append(_power_law_sequence(n, p, target_sum))

    # geometric family – optimise the decay parameter x
    if time.time() < deadline - 30.0:
        geom_seq, geom_val = _best_geometric_sequence(n, target_sum,
                                                      time.time() + 25.0)
        init_seqs.append(geom_seq)
        print(f"[{time.time():.2f}] Geometric init score = {geom_val:.6f}")

    # jittered copies of the above (adds stochastic diversity)
    rng = np.random.default_rng()
    for seq in list(init_seqs):
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        seq_jit = seq + jitter
        seq_jit = np.maximum(seq_jit, 0.0)
        seq_jit = _project_to_simplex(seq_jit, target_sum)
        init_seqs.append(seq_jit)

    # reuse previous best if it exists
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        init_seqs.append(_project_to_simplex(_resize_sequence(prev, n), target_sum))

    # --------------------------------------------------------------
    # Quick evaluation of all seeds – keep the best one
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    for seq in init_seqs:
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] New seed best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase A – Projected gradient descent (soft‑max)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    gd_deadline = time.time() + remaining * 0.35
    a_gd, val_gd = _gradient_descent_with_restarts(best_seq, target_sum, gd_deadline)
    if val_gd < best_score:
        best_score = val_gd
        best_seq = a_gd.copy()
        print(f"[{time.time():.2f}] After GD best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – Deterministic mass‑transfer refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        mt_deadline = time.time() + (deadline - time.time()) * 0.30
        a_mt, val_mt = _deterministic_mass_transfer(best_seq, mt_deadline,
                                                    target_sum,
                                                    start_temp=0.7,
                                                    end_temp=1e-4,
                                                    low_frac=0.05)
        if val_mt < best_score:
            best_score = val_mt
            best_seq = a_mt.copy()
            print(f"[{time.time():.2f}] After DMT best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase C – Re‑ordering heuristics (alternating big/small)
    # --------------------------------------------------------------
    reorder_candidates = [best_seq, _alternating_order(best_seq)]
    # add a couple of random shuffles
    for _ in range(2):
        tmp = best_seq.copy()
        rng.shuffle(tmp)
        reorder_candidates.append(tmp)

    for cand in reorder_candidates:
        cand_score = _max_convolution_fft(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] Ordering improvement -> score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase D – Permutation annealing (swap‑based simulated annealing)
    # --------------------------------------------------------------
    if time.time() < deadline:
        perm_deadline = time.time() + (deadline - time.time()) * 0.20
        a_perm, val_perm = _permutation_annealing(best_seq,
                                                  target_sum,
                                                  perm_deadline,
                                                  init_score=best_score)
        if val_perm < best_score:
            best_score = val_perm
            best_seq = a_perm.copy()
            print(f"[{time.time():.2f}] After permutation annealing best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase E – Final polishing (high‑β gradient descent)
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _annealed_gradient_descent(best_seq,
                                                    deadline,
                                                    target_sum,
                                                    max_beta=8e5,
                                                    step_factor=0.5,
                                                    gamma=0.995)
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best={best_score:.6f}")

    # --------------------------------------------------------------
    # Post‑processing – trim trailing zeros and final evaluation
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # final check (the harness will also do this, but we want to see the number)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for the next run
    globals()["height_sequence_1"] = final_seq
    return final_seq


# ----------------------------------------------------------------------
# Helper for resizing a previous best (used only if a previous run exists)
# ----------------------------------------------------------------------


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)
