def _simulated_annealing(a, target_sum, deadline):
    a_cur = a.copy()
    val_cur = _score(a_cur)
    best_a = a_cur.copy()
    best_val = val_cur
    rng = np.random.default_rng()
    n = len(a_cur)

    # Temperature schedule: start_temp = 0.01 maybe times objective? we set start_temp proportional to (max_b - avg)?
    start_temp = 0.05 * val_cur
    # end temp near 0
    while time.time() < deadline:
        # compute progress fraction
        t = (deadline - time.time()) / (deadline - start_time)
        temp = start_temp * t

        # choose random donor, receiver, delta
        donor = rng.integers(n)
        if a_cur[donor] <= 0: continue
        receiver = rng.integers(n)
        if receiver == donor: continue
        max_delta = a_cur[donor] * 0.2  # up to 20%
        if max_delta <= 0: continue
        delta = rng.random() * max_delta
        a_new = a_cur.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        # ensure nonneg: donor remains nonneg (true)
        val_new = _score(a_new)
        if val_new < val_cur or rng.random() < math.exp(-(val_new - val_cur) / max(temp, 1e-12)):
            a_cur = a_new
            val_cur = val_new
            if val_cur < best_val:
                best_val = val_cur
                best_a = a_cur.copy()
                print(f"[{time.time():.2f}] SA improved best={best_val:.6f}")
        # maybe occasional jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a_cur = np.maximum(a_cur + jitter, 0)
            a_cur = a_cur * (target_sum / a_cur.sum())
            val_cur = _score(a_cur)
            if val_cur < best_val:
                best_val = val_cur
                best_a = a_cur.copy()
    return best_a, best_val

"""
Propose a low‑score step function.

The algorithm follows the successful multi‑stage pipeline from the previous
solution and adds a final stochastic local‑search phase that repeatedly moves
a small amount of mass between two indices and keeps any improvement.
All stages respect a global time budget (default 1000 s) and work on the
ℓ¹‑simplex (Σa = √(2 n)).  The extra random refinement often squeezes a
few thousandths out of the best score obtained by the deterministic phases.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution / scoring utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) shape, normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where conv is the full convolution a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * (reverse(a) convolve p)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b

# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Mirror‑descent with exponential updates using a soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 16384.0
    base_eta = 0.04
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

# ----------------------------------------------------------------------


def _annealed_gradient_descent_v2(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the soft‑max surrogate."""
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 16384.0
    step_base = target_sum * 0.5
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.95
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.002 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GDv2 it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several short GD restarts with random jitter; keep the best."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_v2(start,
                                               time.time() + run_budget,
                                               target_sum)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val

def _grid_param_search(small_n: int,
                       target_sum: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """Search a two‑parameter family a_i = (i+1)^(-p)·exp(-λ·i/small_n)."""
    p_vals = np.linspace(0.2, 1.2, 17)
    lam_vals = np.linspace(0.0, 6.0, 13)

    best_seq = None
    best_score = np.inf
    i = np.arange(small_n, dtype=np.float64)

    for p in p_vals:
        for lam in lam_vals:
            if time.time() > deadline:
                break
            seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
            seq = seq * (target_sum / seq.sum())
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()
                print(f"[{time.time():.2f}] Grid best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")
        else:
            continue
        break

    rng = np.random.default_rng()
    while time.time() < deadline:
        p = rng.uniform(0.2, 1.2)
        lam = rng.exponential(1.0) * 3.0
        seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
        seq = seq * (target_sum / seq.sum())
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Random best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")

    return best_seq, best_score

def _mass_transfer_refine(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Move a modest fraction of mass from high‑contributing indices to low‑contributing ones."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    while time.time() < deadline:
        k_max = int(np.argmax(conv))

        # indices that actually participate in the maximal entry
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = a[k_max - idx]                       # a_{k_max-i}
        weight = a[idx] * partner                       # contribution a_i·a_{k_max-i}

        donor = idx[np.argmax(weight)]

        low_thr = np.percentile(weight, 10)
        low_idx = idx[weight <= low_thr]
        if low_idx.size:
            receiver = low_idx[np.argmin(weight[weight <= low_thr])]
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            donor = rng.choice(idx)
            receiver = rng.integers(n)
            if donor == receiver:
                continue

        max_frac = 0.04
        delta = a[donor] * rng.random() * max_frac
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta

        conv = _conv_fft(a)
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Transfer improve score={best_val:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            conv = _conv_fft(a)

        # renormalise numerically if needed
        cur_sum = a.sum()
        if abs(cur_sum - target_sum) > 1e-9:
            a = a * (target_sum / cur_sum)

    return best_a, best_val

# ----------------------------------------------------------------------


def _subgradient_max_descent(a0: np.ndarray,
                             target_sum: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true maximal convolution entry."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())

        # sub‑gradient of conv_{k_max}: 2·a[k_max‑i] on the feasible indices
        grad = np.zeros_like(a)
        lo = max(0, k_max - (a.shape[0] - 1))
        hi = min(k_max, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        grad[idx] = 2.0 * a[partner]

        norm = np.linalg.norm(grad)
        if norm == 0:
            break

        # safe step bound: ensure we stay non‑negative after the update
        safe_eta = np.min(np.where(grad > 0, a / grad, np.inf))
        eta = min(0.5 * safe_eta, 0.5 / np.sqrt(it + 1))

        a = a - eta * grad
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it} score={best_val:.6f}")

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 7 – final stochastic local‑search refinement
# ----------------------------------------------------------------------


def _random_local_search(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a small random fraction of mass from one index to another.
    Accept only moves that improve the objective.  The loop runs until the
    deadline is reached.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        donor = rng.integers(n)
        if a[donor] <= 0.0:
            continue
        receiver = rng.integers(n)
        if receiver == donor:
            continue

        # move at most 5 % of the donor's mass
        max_frac = 0.05
        delta = a[donor] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # apply move in‑place
        a[donor] -= delta
        a[receiver] += delta

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] RandomLocal it={it} improve score={best_val:.6f}")
        else:
            # revert the move
            a[donor] += delta
            a[receiver] -= delta

        it += 1

        # occasional tiny jitter to shake off plateaus
        if it % 5000 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The routine respects the allotted time budget and returns the best
    sequence found (as a plain list of Python floats).
    """
    start = time.time()
    deadline = start + budget_s - 0.5  # small safety margin

    # --------------------------------------------------------------
    # Choose problem size according to budget
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 15000          # a bit larger than before – still cheap for FFT
    elif budget_s >= 400:
        n = 10000
    else:
        n = 6000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Warm‑start (reuse a previous construction if available)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"],
                                        dtype=np.float64), n)
        init = init * (target_sum / init.sum())
    else:
        init = _power_law_sequence(n, 0.55, target_sum)

    # --------------------------------------------------------------
    # Stage 1 – cheap parametric search on a reduced scale
    # --------------------------------------------------------------
    small_n = max(2000, n // 4)
    small_target = np.sqrt(2.0 * small_n)
    param_deadline = time.time() + min(15.0,
                                      (deadline - time.time()) * 0.07)
    best_small_seq, _ = _grid_param_search(small_n, small_target,
                                           param_deadline)

    if best_small_seq is not None:
        init = _resize_sequence(best_small_seq, n)
        init = init * (target_sum / init.sum())
        print(f"[{time.time():.2f}] Parametric warm‑start ready (n={n})")

    # --------------------------------------------------------------
    # Stage 2 – exponential (multiplicative) descent
    # --------------------------------------------------------------
    exp_deadline = time.time() + (deadline - time.time()) * 0.12
    a_exp, val_exp = _exponential_descent(init, target_sum, exp_deadline)
    best_seq = a_exp
    best_score = val_exp
    print(f"[{time.time():.2f}] After ExpGD best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – projected gradient descent with restarts
    # --------------------------------------------------------------
    if time.time() < deadline:
        gd_deadline = time.time() + (deadline - time.time()) * 0.30
        a_gd, val_gd = _gradient_descent_with_restarts(best_seq,
                                                       target_sum,
                                                       gd_deadline)
        if val_gd < best_score:
            best_seq = a_gd
            best_score = val_gd
        print(f"[{time.time():.2f}] After GD best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – deterministic mass‑transfer refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        mt_deadline = time.time() + (deadline - time.time()) * 0.18
        a_mt, val_mt = _mass_transfer_refine(best_seq, target_sum, mt_deadline)
        if val_mt < best_score:
            best_seq = a_mt
            best_score = val_mt
        print(f"[{time.time():.2f}] After mass‑transfer best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 5 – sub‑gradient descent on the exact max convolution
    # --------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + (deadline - time.time()) * 0.20
        a_sub, val_sub = _subgradient_max_descent(best_seq,
                                                  target_sum,
                                                  sub_deadline)
        if val_sub < best_score:
            best_seq = a_sub
            best_score = val_sub
        print(f"[{time.time():.2f}] After sub‑gradient best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 6 – final stochastic local‑search refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        rs_deadline = deadline
        a_rs, val_rs = _random_local_search(best_seq,
                                             target_sum,
                                             rs_deadline)
        if val_rs < best_score:
            best_seq = a_rs
            best_score = val_rs
        print(f"[{time.time():.2f}] After random local search best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Clean‑up and final evaluation
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # Store for warm‑starts in a future call
    globals()["height_sequence_1"] = final_seq

    return final_seq
