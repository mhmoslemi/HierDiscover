def _targeted_mass_transfer(a, deadline, rng, S):

    best_a = a.copy(); best_val = _max_convolution(a); cur = a.copy(); cur_val = best_val

    while time.time() < deadline:

        # compute gradient to target high gradient indices

        grad, _ = _max_gradient(cur, tau=0)

        # compute probabilities proportional to gradient

        probs_i = grad / (grad.sum() + 1e-12)

        i = rng.choice(len(cur), p=probs_i)

        # choose j with low gradient (inverse)

        inv = 1.0 / (grad + 1e-12)

        probs_j = inv / inv.sum()

        j = rng.choice(len(cur), p=probs_j)

        if i == j or cur[i] < 1e-12:

            continue

        delta = rng.uniform(0.0, 0.2) * cur[i]  # move up to 20%

        new = cur.copy()

        new[i] -= delta

        new[j] += delta

        new = new * (S / new.sum())

        if new_val < cur_val:

            cur = new

            cur_val = new_val

            if new_val < best_val:

                best_val = new_val

                best_a = new.copy()

"""
Improved search for a step‑function that minimises
``evaluate_sequence``.

The algorithm works in three stages:

1. **Fast random exploration** – generate a handful of short candidate
   vectors (length 2‑8 k) with many random families, run a cheap
   sub‑gradient descent (exact “hard‑max” gradient) for a few seconds
   and keep the most promising ones.

2. **Multi‑scale refinement** – start from the best short vector,
   geometrically increase the support (up to ~8·10⁴ entries) and run a
   longer projected sub‑gradient descent on each scale.  The sub‑gradient
   of the max‑convolution is computed analytically, the simplex constraint
   (non‑negative + fixed ℓ₁‑norm) is enforced by a simple rescaling after
   each step.

3. **Gradient‑guided mass transfer** – a final stochastic walk that moves
   mass from positions with a large gradient (i.e. those that increase the
   current peak) to positions with a small gradient.  This often squeezes
   a few more digits out of the objective.

The whole search respects the 1000 s budget (a 5 s safety margin) and
stores the best construction in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution using a real FFT.
    Returns an array of length 2·len(a)−1.
    """
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Peak value of the full self‑convolution a*a."""
    return _full_convolution_fft(a).max()


def _max_gradient(a: np.ndarray, tau: float):
    """
    Sub‑gradient of max_k (a*a)_k.
    Returns (grad, max_b) where ``grad = ∂ max(b)/∂a`` (shape = a.shape)
    and ``max_b = max_k (a*a)_k``.
    When ``tau`` is very small a hard max is used (active set of binding
    indices).  For larger ``tau`` a smooth soft‑max is employed.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        eps = max_b * 1e-9 + 1e-12
        mask = (b >= max_b - eps).astype(float)
        s = mask.sum()
        if s == 0:
            mask = np.ones_like(b) / b.size
        else:
            mask /= s
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / exps.sum()

    a_rev = a[::-1]
    conv_len = mask.size + a.size - 1
    f_len = _next_pow2(conv_len)

    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)

    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    grad = 2.0 * conv[a.size - 1:a.size - 1 + a.size]
    return grad, max_b


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]

# Random families for the exploration phase

def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.2, 5.0)
    return rng.gamma(alpha, 1.0, size=n)


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.1, 0.8)
    return rng.gamma(shape, 1.0, size=n)


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_left_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    p = rng.uniform(0.3, 3.0)
    offset = rng.uniform(0.0, 10.0)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * base + (1.0 - mix) * (idx + offset) ** (-p)


def _init_left_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    lam = rng.uniform(0.001, 0.02)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * base + (1.0 - mix) * np.exp(-lam * idx)


def _init_sparse_bumps(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = np.full(n, rng.uniform(0.0, 0.15))
    nbumps = rng.integers(1, max(2, n // 250))
    for _ in range(nbumps):
        center = rng.integers(0, n)
        width = rng.integers(3, max(5, n // 100))
        height = rng.uniform(0.8, 4.0)
        lo = max(0, center - width // 2)
        hi = min(n, lo + width)
        raw[lo:hi] = np.maximum(raw[lo:hi], height)
    return raw


def _init_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    mu = (n - 1) / 2.0
    sigma = rng.uniform(0.05, 0.30) * n
    x = np.arange(n)
    return np.exp(-0.5 * ((x - mu) / sigma) ** 2)


def _init_half_sine(n: int, rng: np.random.Generator) -> np.ndarray:
    x = np.arange(n)
    raw = np.sin(np.pi * x / (n - 1))
    raw = raw * rng.uniform(0.5, 2.0) + rng.uniform(0.0, 0.2)
    return raw


def _initial_vector(n: int, rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    About 22 % of the time we interpolate from the global best.
    Otherwise we draw from one of the parametric families.
    """
    S = math.sqrt(2.0 * n)

    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    style = rng.choice(['dirichlet', 'gamma', 'uniform',
                        'left_powerlaw', 'left_exponential',
                        'sparse', 'gaussian', 'half_sine'])
    if style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'left_powerlaw':
        raw = _init_left_powerlaw(n, rng)
    elif style == 'left_exponential':
        raw = _init_left_exponential(n, rng)
    elif style == 'sparse':
        raw = _init_sparse_bumps(n, rng)
    elif style == 'gaussian':
        raw = _init_gaussian(n, rng)
    else:   # half_sine
        raw = _init_half_sine(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (hard max)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         S: float) -> tuple[np.ndarray, float]:
    """
    Minimise max(a*a) under the ℓ₁‑norm constraint (sum = S).
    Uses the exact sub‑gradient of the hard maximum and a simple
    projection (clip + renormalise) after each step.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _max_convolution(a)
    it = 0
    # initial step size (a few % of the total mass)
    step = 0.05 * S

    while time.time() < deadline:
        it += 1

        grad, cur_val = _max_gradient(a, tau=0.0)   # exact active set
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # direction that respects sum‑=0 : subtract mean(grad)
        d = - (grad - np.mean(grad))

        # diminishing step – simple 1/√t rule
        eta = step / math.sqrt(it)

        a_candidate = a + eta * d
        a_candidate = np.maximum(a_candidate, 0.0)

        s = a_candidate.sum()
        if s == 0.0:            # should not happen, but guard against it
            break
        a_candidate = a_candidate * (S / s)

        new_val = _max_convolution(a_candidate)

        # accept if we made progress, otherwise shrink step
        if new_val < cur_val:
            a = a_candidate
            # slightly enlarge step after a successful move
            step *= 1.02
        else:
            step *= 0.7

        # occasional jitter to escape flat regions
        if (it & 0x3FF) == 0:   # every 1024 iterations
            jitter = rng.normal(scale=0.008 * S, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (S / a.sum())

    return best_a, best_val


# ----------------------------------------------------------------------
# Gradient‑guided stochastic mass transfer (fine‑tuning)
# ----------------------------------------------------------------------
def _targeted_mass_transfer(a0: np.ndarray,
                            deadline: float,
                            rng: np.random.Generator,
                            S: float) -> tuple[np.ndarray, float]:
    """
    A cheap stochastic walk that preferentially moves mass from indices
    with a large max‑convolution gradient to indices with a small one.
    """
    cur = a0.copy()
    cur_val = _max_convolution(cur)
    best = cur.copy()
    best_val = cur_val
    n = len(cur)

    while time.time() < deadline:
        grad, _ = _max_gradient(cur, tau=0.0)

        # sample i from high‑gradient region, j from low‑gradient region
        prob_i = grad / (grad.sum() + 1e-12)
        i = rng.choice(n, p=prob_i)

        inv = 1.0 / (grad + 1e-12)
        prob_j = inv / inv.sum()
        j = rng.choice(n, p=prob_j)

        if i == j or cur[i] < 1e-12:
            continue

        delta = rng.uniform(0.0, 0.25) * cur[i]    # move up to 25 % of mass at i
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())

        new_val = _max_convolution(new)
        if new_val < cur_val:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new.copy()

    return best, best_val
