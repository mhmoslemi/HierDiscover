def _lp_refinement(a0, deadline, target_sum):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    while time.time() < deadline:
        conv = _conv_fft(a)
        M = conv.max()
        # active set indices: top K largest entries
        K = min(200, len(conv))
        top_idxs = np.argpartition(conv, -K)[-K:]
        # Build constraint matrix A of shape (K, n)
        A = np.zeros((K, n), dtype=np.float64)
        for row_idx, k in enumerate(top_idxs):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            # indices j from lo to hi inclusive
            # for each j, coefficient is a[k - j]; assign
            js = np.arange(lo, hi + 1)
            A[row_idx, js] = a[k - js]   # because a_{k - j}
        # linprog
        c = -np.ones(n)   # maximize sum
        b_ub = np.full(K, M)
        bounds = [(0.0, 1000.0) for _ in range(n)]
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            print("LP failed", res.message)
            break
        g = res.x
        # rescale g to target_sum
        sum_g = g.sum()
        if sum_g == 0:
            break
        g = g * (target_sum / sum_g)
        # update a as convex combination
        t = 0.5
        a_new = (1 - t) * a + t * g
        a_new = _project_to_simplex(a_new, target_sum)
        # compute new score
        new_score = _score(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            print(f"[{time.time():.2f}] LP step improved score={best_score:.6f}")

            # maybe accept anyway? Keep a as old? We'll keep a unchanged to avoid deterioration.
            pass
        # set a <- a_new regardless? maybe follow improvement path?
        a = a_new

Algorithm:
- Initialize with a power‑law step function (normalized to Σ=√(2n)).
- Optional linear‑programming (LP) refinement: solve a LP that maximises the total
  mass under the current convolution peak constraints (only the K largest
  convolution entries are kept as constraints).  The solution is rescaled and
  blended with the current sequence.
- Projected gradient descent on a soft‑max surrogate of the max‑convolution.
- Deterministic mass‑transfer annealing that moves mass from indices that
  contribute heavily to the current peak to indices that contribute little.
- Final high‑β gradient polishing.
All steps respect the 1000 s budget and the sequence is projected onto the
ℓ¹‑simplex after each update.
"""

import time
import numpy as np

# try to import scipy for the LP step
try:
    from scipy.optimize import linprog
    _SCIPY_FOUND = True

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())

def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

def _score(a: np.ndarray) -> float:
    """Benchmark objective: 2·n·max(a*a) / (Σa)² (with Σa≈√(2n) this reduces to max)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution (used by gradient descent)
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of soft‑max approximation of max(convolution). Returns (grad, conv)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # grad = 2 * (a reversed) ⋆ p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b

# ----------------------------------------------------------------------
# Phase A – Projected gradient descent (momentum, annealed soft‑max)
# ----------------------------------------------------------------------
def _annealed_gradient_descent_v2_ext(a0: np.ndarray,
                                     deadline: float,
                                     target_sum: float,
                                     max_beta: float = 1e5,
                                     step_factor: float = 0.7,
                                     gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Momentum projected gradient descent on soft‑max approximation."""
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD_ext it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several short gradient‑descent jobs with jitter, keep the best."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_v2_ext(
            start,
            time.time() + run_budget,
            target_sum,
            max_beta=1e5,
            step_factor=0.8,
            gamma=0.99,
        )
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val

# ----------------------------------------------------------------------
# Phase B – Linear‑programming refinement (uses only the K largest
# convolution coefficients as constraints)
# ----------------------------------------------------------------------
def _lp_refinement(a0: np.ndarray,
                   deadline: float,
                   target_sum: float,
                   K: int = 150) -> tuple[np.ndarray, float]:
    """
    Solve a sequence of small linear programmes that maximise total mass
    under the current convolution‑peak constraints.
    Only the K largest convolution entries are kept as constraints,
    which makes the LP tractable even for n≈10⁴.
    """
    if not _SCIPY_FOUND:
        # fallback – no LP capability
        return a0.copy(), _score(a0)

    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)

    while time.time() < deadline:
        conv = _conv_fft(a)
        M = conv.max()
        # indices of K largest entries (including the maximum)
        if K >= conv.size:
            idxs = np.arange(conv.size)
        else:
            idxs = np.argpartition(conv, -K)[-K:]

        # Build constraint matrix A (|idxs| × n)
        rows = len(idxs)
        A = np.zeros((rows, n), dtype=np.float64)
        for r, k in enumerate(idxs):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1, dtype=np.int64)
            A[r, js] = a[k - js]   # coefficient = a_{k-j}

        # Linear programme: maximise Σ g_j   (i.e. minimise –Σ g_j)
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(rows, M, dtype=np.float64)
        bounds = [(0.0, 1000.0) for _ in range(n)]

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
        if not res.success:
            # if LP fails we abort the refinement
            print(f"[{time.time():.2f}] LP failed: {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g == 0.0:
            break
        # rescale to the required ℓ¹ norm
        g = g * (target_sum / sum_g)

        # convex combination with a modest step size
        t = 0.5
        a_new = (1 - t) * a + t * g
        a_new = _project_to_simplex(a_new, target_sum)

        new_val = _score(a_new)
        if new_val < best_val:
            best_val = new_val
            best_a = a_new.copy()
            print(f"[{time.time():.2f}] LP refinement improved score={best_val:.6f}")

        a = a_new

    return best_a, best_val

# ----------------------------------------------------------------------
# Phase C – Deterministic mass‑transfer refinement (Metropolis)
# ----------------------------------------------------------------------
def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy mass‑transfer moves that directly attack the current peak."""
    a = a0.copy()
    n = a.shape[0]

    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()

    # initial convolution
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

# ----------------------------------------------------------------------
# Re‑ordering utilities
# ----------------------------------------------------------------------
def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Reorder by alternating largest‑to‑left, second‑largest‑to‑right, …"""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left = 0
    right = n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out

# ----------------------------------------------------------------------
# Phase D – Permutation annealing (swap‑based simulated annealing)
# ----------------------------------------------------------------------
def _permutation_annealing(a: np.ndarray,
                           target_sum: float,
                           deadline: float,
                           init_score: float | None = None) -> tuple[np.ndarray, float]:
    """Swap‑based simulated annealing on the ordering of the multiset."""
    cur_seq = a.copy()
    cur_score = init_score if init_score is not None else _max_convolution_fft(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score

    start_time = time.time()
    n = cur_seq.shape[0]
    rng = np.random.default_rng()

    start_temp = 0.2
    end_temp = 1e-5

    while time.time() < deadline:
        elapsed = time.time() - start_time
        ratio = elapsed / (deadline - start_time) if deadline > start_time else 0.0
        temp = start_temp * (end_temp / start_temp) ** ratio

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # swap
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        new_score = _max_convolution_fft(cur_seq)
        delta = new_score - cur_score

        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / max(temp, 1e-12)):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
                print(f"[{time.time():.2f}] Perm anneal new best = {best_score:.6f}")
        else:
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        if ratio > 0.9 and rng.random() < 0.01:
            rng.shuffle(cur_seq)
            cur_score = _max_convolution_fft(cur_seq)

    return best_seq, best_score

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_v2_ext(
            start,
            time.time() + run_budget,
            target_sum,
            max_beta=1e5,
            step_factor=0.8,
            gamma=0.99,
        )
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Linear‑programming refinement (uses only the K largest
# convolution coefficients as constraints)
# ----------------------------------------------------------------------
def _lp_refinement(a0: np.ndarray,
                   deadline: float,
                   target_sum: float,
                   K: int = 150) -> tuple[np.ndarray, float]:
    """
    Small LPs that maximise total mass under the current peak constraints.
    Only the K largest convolution entries are kept as constraints,
    making the problem tractable for n≈10⁴.
    """
    if not _SCIPY_FOUND:               # pragma: no cover
        return a0.copy(), _score(a0)

    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)

    while time.time() < deadline:
        conv = _conv_fft(a)
        M = conv.max()
        if K >= conv.size:
            idxs = np.arange(conv.size)
        else:
            idxs = np.argpartition(conv, -K)[-K:]

        rows = len(idxs)
        A = np.zeros((rows, n), dtype=np.float64)
        for r, k in enumerate(idxs):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1, dtype=np.int64)
            A[r, js] = a[k - js]               # coefficient = a_{k-j}

        c = -np.ones(n, dtype=np.float64)       # maximise Σ g_j
        b_ub = np.full(rows, M, dtype=np.float64)
        bounds = [(0.0, 1000.0) for _ in range(n)]

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            print(f"[{time.time():.2f}] LP failed: {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g == 0.0:
            break
        g = g * (target_sum / sum_g)

        t = 0.5
        a_new = (1 - t) * a + t * g
        a_new = _project_to_simplex(a_new, target_sum)

        new_val = _score(a_new)
        if new_val < best_val:
            best_val = new_val
            best_a = a_new.copy()
            print(f"[{time.time():.2f}] LP refinement improved score={best_val:.6f}")

        a = a_new

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – Deterministic mass‑transfer refinement (Metropolis style)
# ----------------------------------------------------------------------
def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy moves that directly attack the current maximal convolution entry."""
    a = a0.copy()
    n = a.shape[0]

    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]               # contribution of each index
        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Re‑ordering utilities
# ----------------------------------------------------------------------
def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place the largest value leftmost, second largest rightmost, …."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    left_side = True
    for v in sorted_vals:
        if left_side:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        left_side = not left_side
    return out


# ----------------------------------------------------------------------
# Phase D – Permutation annealing (swap‑based simulated annealing)
# ----------------------------------------------------------------------
def _permutation_annealing(a: np.ndarray,
                           target_sum: float,
                           deadline: float,
                           init_score: float | None = None) -> tuple[np.ndarray, float]:
    """Swap‑based annealing that only changes the ordering of the multiset."""
    cur_seq = a.copy()
    cur_score = init_score if init_score is not None else _max_convolution_fft(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score

    start_time = time.time()
    n = cur_seq.shape[0]
    rng = np.random.default_rng()

    start_temp = 0.2
    end_temp = 1e-5

    while time.time() < deadline:
        elapsed = time.time() - start_time
        ratio = elapsed / (deadline - start_time) if deadline > start_time else 0.0
        temp = start_temp * (end_temp / start_temp) ** ratio

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        new_score = _max_convolution_fft(cur_seq)
        delta = new_score - cur_score

        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / max(temp, 1e-12)):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
                print(f"[{time.time():.2f}] Perm anneal new best = {best_score:.6f}")
        else:
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        if ratio > 0.9 and rng.random() < 0.01:
            rng.shuffle(cur_seq)
            cur_score = _max_convolution_fft(cur_seq)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The pipeline combines LP refinement, gradient descent, mass‑transfer,
    re‑ordering, permutation annealing and a final polishing stage.
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    # ---- choose problem size -------------------------------------------------
    if budget_s >= 800:
        n = 16000
    elif budget_s >= 400:
        n = 10000
    else:
        n = 6000

    target_sum = np.sqrt(2.0 * n)
    rng = np.random.default_rng()

    # ---- 0. initial candidate ------------------------------------------------
    a = _power_law_sequence(n, 0.55, target_sum)
    a += rng.normal(scale=target_sum * 0.01, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target_sum)
    print(f"[{time.time():.2f}] init n={n} score={_score(a):.6f}")

    # ---- 1. optional LP refinement -----------------------------------------
    if _SCIPY_FOUND and time.time() < deadline - 30.0:
        lp_deadline = time.time() + (deadline - time.time()) * 0.20
        a, val = _lp_refinement(a, lp_deadline, target_sum, K=180)
        print(f"[{time.time():.2f}] after LP refinement score={val:.6f}")

    # ---- 2. gradient descent with restarts ----------------------------------
    if time.time() < deadline - 5.0:
        gd_deadline = time.time() + (deadline - time.time()) * 0.30
        a, val = _gradient_descent_with_restarts(a, target_sum, gd_deadline)
        print(f"[{time.time():.2f}] after GD phase score={val:.6f}")

    # ---- 3. deterministic mass‑transfer ------------------------------------
    if time.time() < deadline - 5.0:
        mt_deadline = time.time() + (deadline - time.time()) * 0.30
        a, val = _deterministic_mass_transfer(a, mt_deadline, target_sum,
                                             start_temp=0.8,
                                             end_temp=5e-5,
                                             low_frac=0.07)
        print(f"[{time.time():.2f}] after mass‑transfer score={val:.6f}")

    # ---- 4. simple re‑ordering heuristics -----------------------------------
    if time.time() < deadline:
        reorder_candidates = [a, _alternating_order(a)]
        for _ in range(2):
            tmp = a.copy()
            rng.shuffle(tmp)
            reorder_candidates.append(tmp)

        for cand in reorder_candidates:
            cand_score = _max_convolution_fft(cand)
            if cand_score < _score(a):
                a = cand.copy()
                print(f"[{time.time():.2f}] ordering improvement -> score={cand_score:.6f}")

    # ---- 5. permutation annealing -------------------------------------------
    if time.time() < deadline:
        perm_deadline = time.time() + (deadline - time.time()) * 0.15
        a, val = _permutation_annealing(a, target_sum, perm_deadline,
                                        init_score=_max_convolution_fft(a))
        print(f"[{time.time():.2f}] after permutation annealing score={val:.6f}")

    # ---- 6. final polishing (high‑β gradient descent) -----------------------
    if time.time() + 2.0 < deadline:
        a, val = _annealed_gradient_descent_v2_ext(
            a,
            deadline,
            target_sum,
            max_beta=3e5,
            step_factor=0.6,
            gamma=0.995,
        )
        print(f"[{time.time():.2f}] final polish score={val:.6f}")

    # ---- post‑processing ----------------------------------------------------
    arr = np.array(a, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] final evaluated score = {val:.6f}")
    except Exception:               # pragma: no cover
        pass

    globals()["height_sequence_1"] = final_seq
    return final_seq
