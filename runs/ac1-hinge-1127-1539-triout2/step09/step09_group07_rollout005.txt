"""
LP‑driven flattening of the self‑convolution.

We start from the best known construction (global ``height_sequence_1`` if
available) and repeatedly perform the following steps until the time budget
runs out:

1.  If the current support is far from the allowed maximum we enlarge it by
    linear interpolation and renormalise to the required ℓ₁‑norm
    ``S = sqrt(2·n)``.  A longer support gives the optimiser more freedom to
    spread the mass.

2.  For the current sequence ``f`` we compute the full convolution ``b = f*f``.
    The peak value ``M = max(b)`` is the current objective.  The indices where
    ``b`` attains (or almost attains) this peak form the *active set*.

3.  Keeping ``f`` fixed we solve a linear programme

        maximise   sum(g_j)
        subject to g_j ≥ 0
                   (f * g)[s] ≤ M    for every s in the active set

    The constraints are linear in the unknown vector ``g`` because ``f`` is
    known.  The LP is solved with ``scipy.optimize.linprog`` (HiGHS) using a
    sparse constraint matrix.

    The solution ``g₀`` is scaled to the correct ℓ₁‑norm ``S`` (the LP ignores
    the norm, we only maximise the total mass before scaling).

4.  We replace a small fraction of ``f`` by the new vector ``g``:

        f_new = (1‑t)·f + t·g ,   t ∈ {0.2,0.1,0.05,0.02,0.01}

    After each trial we renormalise to the exact norm and keep the best
    improvement (the peak must strictly decrease).

5.  The process (2‑4) is repeated many times.  When the time budget permits we
    finish with a short simulated‑annealing / greedy hill‑climbing phase
    (identical to the reference implementation) to capture any remaining tiny
    improvements.

The algorithm respects the 1000 s budget, prints progress, stores the final
construction in the global ``height_sequence_1`` variable and returns it as a
plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, stable for large n)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Peak value of a*a (full convolution)."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation (preserves total mass)."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]

def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.2, 5.0)
    raw = rng.gamma(alpha, 1.0, size=n)
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.1, 0.8)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_left_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    p = rng.uniform(0.3, 3.0)
    offset = rng.uniform(0.0, 10.0)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * (idx + offset) ** (-p)
    return raw


def _init_left_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    lam = rng.uniform(0.001, 0.02)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * np.exp(-lam * idx)
    return raw


def _init_sparse_bumps(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = np.full(n, rng.uniform(0.0, 0.15))
    nbumps = rng.integers(1, max(2, n // 250))
    for _ in range(nbumps):
        center = rng.integers(0, n)
        width = rng.integers(3, max(5, n // 100))
        height = rng.uniform(0.8, 4.0)
        lo = max(0, center - width // 2)
        hi = min(n, lo + width)
        raw[lo:hi] = np.maximum(raw[lo:hi], height)
    return raw


def _init_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    mu = (n - 1) / 2.0
    sigma = rng.uniform(0.05, 0.30) * n
    x = np.arange(n)
    raw = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    return raw


def _init_half_sine(n: int, rng: np.random.Generator) -> np.ndarray:
    x = np.arange(n)
    raw = np.sin(np.pi * x / (n - 1))
    raw = raw * rng.uniform(0.5, 2.0) + rng.uniform(0.0, 0.2)
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Occasionally warm‑starts from the global best, otherwise draws from a rich
    family of random shapes.
    """
    S = math.sqrt(2.0 * n)

    # Warm start with probability ≈0.22
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a * (S / a.sum())
        return a

    # Choose a random family
    style = rng.choice(['dirichlet', 'gamma', 'uniform',
                        'left_powerlaw', 'left_exponential',
                        'sparse', 'gaussian', 'half_sine'])
    if style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'left_powerlaw':
        raw = _init_left_powerlaw(n, rng)
    elif style == 'left_exponential':
        raw = _init_left_exponential(n, rng)
    elif style == 'sparse':
        raw = _init_sparse_bumps(n, rng)
    elif style == 'gaussian':
        raw = _init_gaussian(n, rng)
    else:   # half_sine
        raw = _init_half_sine(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Linear‑programming improvement step
# ----------------------------------------------------------------------
def _active_indices(b: np.ndarray, rel_tol: float = 1e-9) -> np.ndarray:
    """Return indices where the convolution is within tolerance of its maximum."""
    M = b.max()
    eps = M * rel_tol + 1e-12
    return np.where(b >= M - eps)[0]


def _build_constraint_matrix(f: np.ndarray, active_idx: np.ndarray) -> "csr_matrix":
    """
    Build a sparse matrix A (shape |active| × n) such that
        (A @ g)[r] = (f * g)[active_idx[r]].

    The rows correspond to the active convolution positions.
    """
    from scipy.sparse import csr_matrix
    n = f.shape[0]
    rows = []
    cols = []
    data = []

    for row, s in enumerate(active_idx):
        lo = max(0, s - (n - 1))
        hi = min(n - 1, s)
        # variable indices that appear in this constraint
        j_vals = np.arange(lo, hi + 1, dtype=int)
        coeffs = f[s - j_vals]          # f_{s-j}
        rows.extend([row] * len(j_vals))
        cols.extend(j_vals.tolist())
        data.extend(coeffs.tolist())

    A = csr_matrix((data, (rows, cols)), shape=(len(active_idx), n))
    return A


def _lp_step(f: np.ndarray, M: float, rng: np.random.Generator) -> tuple[np.ndarray, bool]:
    """
    Perform one LP‑based improvement step.
    Returns (new_f, improved_flag).  ``new_f`` is identical to ``f`` if the step
    does not yield a strict improvement.
    """
    n = f.shape[0]
    S = math.sqrt(2.0 * n)

    # 1) active set
    b = _full_convolution_fft(f)
    active = _active_indices(b, rel_tol=1e-9)
    # Limit the size of the active set for speed – keep the most extreme indices
    if len(active) > 50:
        # keep the first 25 and last 25 (they are usually the peaks)
        active = np.concatenate([active[:25], active[-25:]])

    # 2) build sparse constraints (f * g)[s] ≤ M  for all active s
    A = _build_constraint_matrix(f, active)

    # 3) solve LP: maximise sum(g)  subject to A g ≤ M , g ≥ 0
    from scipy.optimize import linprog

    c = -np.ones(n)                     # minimise -sum → maximise sum
    ub = np.full(A.shape[0], M)

    res = linprog(c, A_ub=A, b_ub=ub,
                  bounds=[(0, None)] * n,
                  method='highs',
                  options={'presolve': True, 'time_limit': 30})

    if not res.success:
        # LP failed – fall back to a no‑op
        return f, False

    g0 = res.x
    if g0.sum() <= 0:
        return f, False

    # 4) scale to the required ℓ₁‑norm
    g = g0 * (S / g0.sum())

    # 5) try a few convex‑combination steps (t = 0.2 … 0.01)
    best_f = f
    best_M = M
    for t in (0.2, 0.1, 0.05, 0.02, 0.01):
        cand = (1 - t) * f + t * g
        cand = cand * (S / cand.sum())          # renorm (numerical drift)
        cand_M = _max_convolution(cand)
        if cand_M < best_M - 1e-12:
            best_f = cand
            best_M = cand_M
            break

    improved = best_M < M - 1e-12
    return best_f, improved

# Simulated‑annealing / greedy hill‑climber (reuse from the reference code)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Mass‑transfer walk with exponential acceptance."""
    n = len(seq)
    S = math.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score
    temperature = 0.15 * cur_score

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] == 0.0:
            continue
        delta = rng.uniform(0.0, 0.3) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())
        new_score = _max_convolution(new)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temperature, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

        temperature *= 0.99995
        if temperature < 1e-6:
            temperature = 1e-6

    return best, best_score


def _hill_climb(seq: np.ndarray,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Strict downhill neighbour walk (very cheap)."""
    n = len(seq)
    S = math.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] < 1e-12:
            continue
        delta = rng.uniform(0.0, 0.15) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())
        new_score = _max_convolution(new)
        if new_score < cur_score:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

    return best, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The algorithm combines a linear‑programming flattening step with a final
    annealing/hill‑climbing fine‑tuner.  Returned list is stored globally as
    ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    safety_margin = 5.0               # keep a small buffer for the harness
    end_time = time.time() + budget_s - safety_margin
    if end_time <= time.time():
        raise RuntimeError("Insufficient budget after safety margin.")

    # ------------------------------------------------------------------
    # Get a warm start if possible
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # ------------------------------------------------------------------
    # Initialise the first candidate
    # ------------------------------------------------------------------
    if warm_seq is not None:
        cur_seq = warm_seq.copy()
        # ensure exact norm (might have drifted due to previous scaling)
        n0 = len(cur_seq)
        cur_seq = cur_seq * (math.sqrt(2.0 * n0) / cur_seq.sum())
    else:
        # choose a moderate length if nothing is known
        n0 = rng.integers(3000, 6000)
        cur_seq = _initial_vector(n0, rng, None)

    cur_M = _max_convolution(cur_seq)
    best_seq = cur_seq.copy()
    best_M = cur_M
    best_n = len(cur_seq)
    print(f"[init] length={best_n}  peak={best_M:.6f}")

    # ------------------------------------------------------------------
    # Upper bound on the final length (≈ 8·10⁴ as in the reference)
    # ------------------------------------------------------------------
    max_len = min(80000, best_n * 12)

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < end_time - 30.0:          # keep ~30 s for final polishing
        # ---- possible up‑scale -------------------------------------------------
        if best_n < max_len:
            new_n = int(min(best_n * 1.5, max_len))
            best_seq = _scale_sequence(best_seq, new_n)
            best_seq = best_seq * (math.sqrt(2.0 * new_n) / best_seq.sum())
            best_n = new_n
            best_M = _max_convolution(best_seq)
            print(f"[scale] length={best_n}  peak={best_M:.6f}")

        # ---- LP‑based flattening ------------------------------------------------
        # Give each scale a modest time slice (≈ 40 % of the remaining budget)
        slice_budget = min((end_time - time.time()) * 0.4, 180.0)
        slice_deadline = time.time() + slice_budget

        iter_count = 0
        while time.time() < slice_deadline:
            iter_count += 1
            new_seq, improved = _lp_step(best_seq, best_M, rng)
            if improved:
                best_seq = new_seq
                best_M = _max_convolution(best_seq)
                print(f"[lp] iter={iter_count:4d}  length={best_n}  new peak={best_M:.6f}")
            else:
                # If a single step makes no progress, break to give the
                # scheduler a chance to upscale or switch strategy.
                break

        # ---- occasional random restart (to escape very flat plateaus) ----------
        if rng.random() < 0.07:
            # create a fresh random vector of the current length and blend a bit
            fresh = _initial_vector(best_n, rng, None)
            blend = 0.15
            best_seq = (1 - blend) * best_seq + blend * fresh
            best_seq = best_seq * (math.sqrt(2.0 * best_n) / best_seq.sum())
            best_M = _max_convolution(best_seq)
            print(f"[restart] blended fresh start, peak={best_M:.6f}")

    # ------------------------------------------------------------------
    # Final stochastic polish (annealing + greedy hill‑climb)
    # ------------------------------------------------------------------
    remaining = end_time - time.time()
    if remaining > 20.0:
        best_seq, best_M = _anneal(best_seq,
                                   time.time() + 0.6 * remaining,
                                   rng)
        remaining = end_time - time.time()
        if remaining > 5.0:
            best_seq, best_M = _hill_climb(best_seq,
                                           time.time() + remaining,
                                           rng)

    # ------------------------------------------------------------------
    # Final sanity check with the supplied black‑box evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  internal max = {best_M:.6f}"
          f"  length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
