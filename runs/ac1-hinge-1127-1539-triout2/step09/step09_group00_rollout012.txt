def _hard_max_gradient(a):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    # compute convolution a*a to get b
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
    max_val = b.max()
    # mask
    mask = (b == max_val).astype(np.float64)
    mask_fft = np.fft.rfft(mask, n=size)
    conv_rev = np.fft.irfft(a_rev_fft * mask_fft, n=size)[:2*n-1]
    grad = 2.0 * conv_rev[n-1:n-1+n]
    return grad, max_val, b

"""
Propose a low‑score step‑function.

The algorithm combines several complementary optimisation phases:

1. **Parametric initialisation** – a cheap grid over power‑law /
   exponential shapes (plus a resized previous best) to obtain a good
   starting vector.

2. **Hard‑max sub‑gradient descent** – uses the exact gradient of the
   max‑convolution (the true objective) rather than a soft‑max
   approximation.  A Nesterov‑style momentum term and a diminishing step
   size drive rapid improvement while keeping the vector on the simplex
   Σa = √(2 n).

3. **Soft‑max annealed gradient descent** – refines the solution with the
   smoother surrogate used in the original baseline.

4. **Deterministic mass‑transfer** – a greedy “move mass from the
   dominant convolution contribution to a weak one’’ heuristic,
   equipped with a temperature schedule and Metropolis acceptance.

5. **Global random mass‑transfer** – occasional simulated‑annealing‑type
   moves that shift a small amount of mass from the largest to the
   smallest entries.

6. **Final polishing** – a long annealed gradient‑descent run with a
   very high soft‑max temperature.

All phases respect a single 1000 s time budget.  The final sequence is
trimmed of trailing zeros and returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Benchmark objective 2·n·max(a*a) / (∑a)² (equals max‑convolution when Σa=√(2n))."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple families for the quick parametric search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Generate (i+1)^(-p) of length n and scale to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) * exp(-q * i / n) scaled to target_sum."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution (used in GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where conv is the full convolution a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    # avoid overflow – subtract max first
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * (a reversed) ⋆ p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Hard‑max sub‑gradient (exact max) – new component
# ----------------------------------------------------------------------


def _hard_max_gradient(a: np.ndarray):
    """
    Exact sub‑gradient of the max‑convolution objective.
    Returns (grad, max_val, full_convolution).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    # full convolution a*a
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_val = conv.max()
    eps = 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt > 1:
        mask /= cnt                     # average over all active indices

    # gradient = 2 * (a_rev ⋆ mask)
    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    mask_fft = np.fft.rfft(mask, n=size)
    conv_rev = np.fft.irfft(a_rev_fft * mask_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, max_val, conv

# Projected gradient descent with annealed soft‑max temperature
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 2e5,
                               step_factor: float = 0.6,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max approximation.
    Returns (best_vector, best_score).
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    step_base = target_sum * step_factor
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # Nesterov‑style momentum update
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 300 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:

        start = np.maximum(start, 0.0)

                                            time.time() + run_budget,

            best_a = a.copy()

def _deterministic_mass_transfer(a0: np.ndarray,

                                end_temp: float = 1e-5,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy moves that directly reduce the current maximal convolution entry.
    Uses a Metropolis rule with an exponentially decaying temperature.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]
        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0.0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0.0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

    # pure power‑law
    for p in np.linspace(0.45, 0.65, 12):
        candidates.append(_power_law_sequence(n, p, target_sum))

    # power‑law + exponential tail
    for p in np.linspace(0.45, 0.65, 8):
        for q in np.linspace(0.0, 5.0, 8):
            candidates.append(_exp_power_law_sequence(n, p, q, target_sum))

    # previous best (if available)
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and prev:
        prev_arr = np.array(prev, dtype=np.float64)
        candidates.append(_resize_sequence(prev_arr, n))

    # jittered variants of a subset of the above
    for seq in list(candidates)[:30]:
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        candidates.append(seq_j)

    # evaluate a (random) sample of the candidates
    max_initial_evals = min(300, len(candidates))
    sample_idx = rng.choice(len(candidates), size=max_initial_evals, replace=False)

    best_seq = None
    best_score = np.inf

    for i in sample_idx:
        seq = candidates[i]
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
    print(f"[{time.time():.2f}] Initial best score = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase 1 – Hard‑max sub‑gradient descent (exact max)
    # ---------------------------------------------------------------
    sub_deadline = time.time() + (deadline - time.time()) * 0.30
    a = best_seq.copy()
    momentum = np.zeros_like(a)
    gamma = 0.9
    step0 = target_sum * 0.5
    it = 0
    while time.time() < sub_deadline:
        grad, _, _ = _hard_max_gradient(a)

        # Nesterov‑style momentum
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step0 / (norm * math.sqrt(it + 1))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_score - 1e-12:
            best_score = cur_val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it} improved score={best_score:.6f}")

        it += 1

    # ---------------------------------------------------------------
    # Phase 2 – Soft‑max gradient descent with restarts
    # ---------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        gd_deadline = time.time() + remaining * 0.50
        a_opt, val_opt = _gradient_descent_with_restarts(best_seq,
                                                         target_sum,
                                                         gd_deadline)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] GD improved best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase 3 – Deterministic mass‑transfer refinement
    # ---------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        a_ref, val_ref = _deterministic_mass_transfer(best_seq,
                                                      deadline,
                                                      target_sum,
                                                      start_temp=0.5,
                                                      end_temp=1e-6,
                                                      low_frac=0.05)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] DMT improved best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase 4 – Global random mass‑transfer (extra exploration)
    # ---------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        a_ref2, val_ref2 = _global_random_mass_transfer(best_seq,
                                                        deadline,
                                                        target_sum)
        if val_ref2 < best_score:
            best_score = val_ref2
            best_seq = a_ref2.copy()
            print(f"[{time.time():.2f}] Global random improved best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase 5 – Final polishing (high‑beta soft‑max GD)
    # ---------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        a_fin, val_fin = _annealed_gradient_descent(best_seq,
                                                    deadline,
                                                    target_sum,
                                                    max_beta=2e6,
                                                    step_factor=0.5,
                                                    gamma=0.99)
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Post‑processing: trim trailing zeros, return as Python list
    # ---------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final evaluation (harmless if evaluate_sequence is not defined)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
