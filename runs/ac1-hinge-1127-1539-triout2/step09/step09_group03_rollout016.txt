def _local_refine(a0, target_sum, deadline):
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    # initial temperature set to a fraction of best_val
    T0 = best_val * 0.08  # increased
    while time.time() < deadline:
        # current max index
        k_max = int(np.argmax(b))
        # range for indices that contribute to k_max
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        # contributions from each i in idx: a[i] * a[k_max-i]
        prod = a[idx] * a[k_max - idx]
        # pick donor weighted by prod
        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # choose receiver from low-weight region
        # compute weight per index based on a's magnitude
        # prefer indices with small a
        low_candidates = np.where(a <= np.percentile(a, 30))[0]
        if low_candidates.size:
            receiver = rng.choice(low_candidates)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # limit by donor amount and also by not overshooting temperature
        max_delta = min(a[donor] * 0.2, target_sum * 0.005)
        if max_delta <= 0:

        delta = rng.random() * max_delta

        # update convolution efficiently
        b_new = b.copy()
        # linear term updates (ordered pairs)
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic term updates
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))

        accept = False
        if new_val < best_val:

        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:

        if accept:
            a[donor] -= delta

            b = b_new

"""
Propose a step‑function (non‑negative sequence) with a very low benchmark score.

Algorithm
---------
1. Choose a problem size `n` based on the time budget (larger `n` → more
   degrees of freedom, but also more work per iteration).
2. Build a *spiky* initial sequence:
      • Pick `k` ≈ √(2n) / 0.85 positions (≈ 180–200 for n≈12000) roughly
        equally spaced and add moderate jitter.
      • Put the same mass `w = sqrt(2n)/k` on each selected index.
      • Add a tiny random noise and finally project onto the simplex
        {x ≥ 0, Σx = sqrt(2n)}.
   The idea is to keep each product `a_i a_j` small (≈ w²) while using many
   spikes so that the total ℓ₁‑norm is satisfied.  This gives a good
   starting point for the max‑convolution objective.
3. Refine the sequence with an aggressive simulated‑annealing / mass‑transfer
   local search (O(n) updates of the full convolution).  Each move transfers
   a small amount of mass from an index that contributes heavily to the
   current maximal convolution entry to an index with little mass.  The move
   is accepted if it lowers the objective or with a probability that slowly
   cools down.
4. Optional short gradient‑descent (soft‑max surrogate) restarts are run if
   time permits – they help to escape plateaus.
5. Keep the best sequence seen, trim trailing zeros, store it in
   ``height_sequence_1`` for future warm‑starts and return it.

All steps respect non‑negativity and the ℓ¹‑norm constraint, therefore the
result can be fed directly to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of power‑of‑2 length."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  For Σa=√(2n) this equals max(a*a)."""
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Gradient descent on soft‑max surrogate (optional)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    grad_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(grad_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


def _annealed_gradient_descent_v2(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the soft‑max surrogate."""
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    step_base = target_sum * 0.45
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.93
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0015 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GDv2 it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val - 1e-9:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Local refine improved score={best_val:.6f}")

        # --------------------------------------------------------------
        # Periodic re‑normalisation to avoid numerical drift
        # --------------------------------------------------------------
        if time.time() - start > 30.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑normalisation improved score={best_val:.6f}")

    return best_a, best_val

# Main entry point

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5            # safety margin

    # ------------------------------------------------------------------
    # Problem size (larger `n` gives more freedom, but each iteration costs O(n))
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 15000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = math.sqrt(2.0 * n)

    best_seq = None
    best_score = float('inf')
    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # 1️⃣ Warm‑start from the globally stored sequence (if any)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
        init = init * (target_sum / init.sum())
        # a quick local refinement to see whether the warm‑start can be improved
        warm_deadline = time.time() + min(80.0, (deadline - time.time()) * 0.2)
        a_warm, val_warm = _local_refine(init, target_sum, warm_deadline)
        if val_warm < best_score:
            best_seq = a_warm
            best_score = val_warm
            print(f"[{time.time():.2f}] Warm‑start refined score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Repeated attempts with a spiky (few‑spike) initialization
    # ------------------------------------------------------------------
    attempt = 0
    while time.time() < deadline - 10.0:
        remaining = deadline - time.time()
        attempt_time = min(30.0, remaining * 0.35)
        if attempt_time < 5.0:
            break
        attempt += 1

        # ---- build a spiky start -------------------------------------------------
        # aim for ~0.85·√(2n) per spike → k spikes ≈ √(2n) / 0.85
        k = max(180, int(target_sum / 0.85))
        if k > n:
            k = n
        base_pos = np.linspace(0, n - 1, k)
        jitter = rng.normal(scale=n / (k * 4), size=k)
        pos = np.clip(np.rint(base_pos + jitter), 0, n - 1).astype(int)
        # guarantee uniqueness
        pos = np.unique(pos)
        while pos.size < k:
            extra = rng.choice(n, size=k - pos.size, replace=False)
            pos = np.unique(np.concatenate([pos, extra]))
        w = target_sum / k
        a_init = np.zeros(n, dtype=np.float64)
        a_init[pos] = w
        # small random perturbation to break perfect symmetry
        a_init += rng.normal(scale=w * 0.05, size=n)
        a_init = np.maximum(a_init, 0.0)
        a_init = a_init * (target_sum / a_init.sum())

        # ---- aggressive local refinement -----------------------------------------
        a_ref, val_ref = _local_refine(a_init, target_sum,
                                       time.time() + attempt_time)
        if val_ref < best_score:
            best_seq = a_ref
            best_score = val_ref
            print(f"[{time.time():.2f}] Attempt {attempt} improved best to {best_score:.6f}")

        # ---- optional short gradient‑descent restarts (if time left) ------------
        if time.time() < deadline - 5.0:
            gd_deadline = time.time() + min(12.0, (deadline - time.time()) * 0.15)
            a_gd, val_gd = _gradient_descent_with_restarts(a_ref,
                                                           target_sum,
                                                           gd_deadline)
            if val_gd < best_score:
                best_seq = a_gd
                best_score = val_gd
                print(f"[{time.time():.2f}] GD after attempt {attempt} improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never happen) – uniform distribution
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _score(best_seq)

    # ------------------------------------------------------------------
    # Final clean‑up: truncate trailing zeros and evaluate once more
    # ------------------------------------------------------------------
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    final_seq = best_seq.tolist()
    globals()["height_sequence_1"] = final_seq

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    return final_seq
