```
def _subgradient_descent_max(a0, target_sum, deadline):
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a) # but _score uses max; fine.
    iteration = 0
    # Parameter schedule
    eta0 = target_sum * 0.2   # maybe step scaling.
    while time.time() < deadline:
        # compute convolution via FFT
        b = _conv_fft(a)
        max_val = b.max()
        k_max = int(np.argmax(b))
        # gradient vector
        g = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        # for i from lo..hi: gradient g[i] = 2*a_{k_max - i}
        idx = np.arange(lo, hi+1)
        g[idx] = 2.0 * a[k_max - idx]
        # adjust gradient to stay on simplex: subtract mean
        g_centered = g - np.mean(g)
        # step size schedule: decreasing
        eta = eta0 / (1.0 + iteration * 0.01)  # arbitrary
        a = a - eta * g_centered
        # ensure nonnegativity and projection onto simplex
        a = np.maximum(a, 0.0)
        a = a * (target_sum / a.sum())
        val = max_val  # maybe recompute after update
        # after update compute new max
        # actual new max will be recomputed in next loop; but evaluate after projection
        # compute new b for val
        b_new = _conv_fft(a)
        new_val = b_new.max()
        if new_val < best_val:
            best_val = new_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={iteration} score={best_val:.6f}")
        iteration += 1
        # optionally break if iteration too high
    return best_a, best_val
```

"""
Propose a step‑function with a low benchmark score.

Algorithm
---------
1. Warm‑start from the best known construction (global `height_sequence_1`);
   if unavailable start from a decreasing power‑law.
2. Exponential (mirror) descent on a soft‑max approximation of the max‑convolution
   (≈15 % of the time budget).
3. Projected gradient descent with random restarts (≈30 % of the budget).
4. Sub‑gradient descent on the *true* max‑convolution (≈15 % of the budget).
   The sub‑gradient of  maxₖ (a⋆a)ₖ is  g_i = 2·a_{k*‑i} for the current maximising
   index k*.  After each step we project onto the simplex Σa = √(2n).
5. Exact simulated‑annealing refinement that transfers a tiny amount of mass
   from a “bad’’ index (large contribution to the current maximal entry)
   to a “good’’ index, updating the convolution in O(n) per move.
   The remaining budget is spent here.
6. Truncate trailing zeros, evaluate the final score and store the sequence
   for future warm‑starts.

All helper routines are defined at top level; the only required external
function is `evaluate_sequence` which the platform provides.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a⋆a) / (sum(a)²).  For Σa=√(2n) this equals max(a⋆a)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, Σx = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(a⋆a).
    Returns (grad, full_convolution) where grad has the same length as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # grad = 2 * (reverse(a) convolve p)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b

# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (multiplicative) on the soft‑max surrogate.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

def _annealed_gradient_descent_v2(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the soft‑max surrogate.
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    step_base = target_sum * 0.45
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.93
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0015 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GDv2 it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several short GD restarts with random jitter; keep the best."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_v2(start,
                                               time.time() + run_budget,
                                               target_sum)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val

def _subgradient_descent_max(a0: np.ndarray,
                             target_sum: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the exact objective max(a⋆a) with a simplex projection.
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    n = a.shape[0]
    b = _conv_fft(a)
    best_val = b.max()
    best_a = a.copy()
    it = 0
    rng = np.random.default_rng()
    # initial step size (tuned experimentally)
    eta0 = target_sum * 0.3

    while time.time() < deadline:
        # index of the current maximal convolution entry
        k_max = int(np.argmax(b))
        # gradient of max entry w.r.t. a (sub‑gradient)
        g = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g[idx] = 2.0 * a[k_max - idx]

        # remove the component parallel to the all‑ones vector to keep Σa fixed
        g_centered = g - g.mean()

        # decreasing step size
        eta = eta0 / (1.0 + it * 0.01)

        # small random jitter every few hundred steps to avoid stagnation
        if it % 400 == 0:
            g_centered += rng.normal(scale=target_sum * 0.001, size=n)

        a = a - eta * g_centered
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        # recompute convolution and max
        b = _conv_fft(a)
        cur_val = b.max()
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val

def _local_refine(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climbing on the true max‑convolution.
    Moves a tiny amount of mass from a “bad’’ index to a “good’’ index,
    updating the convolution in O(n) per move.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    # Temperature schedule (proportional to the current best value)
    T0 = best_val * 0.05
    last_recalc = start

    while time.time() < deadline:
        # --- locate the current maximal entry and compute contributions ---
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        prod = a[idx] * a[k_max - idx]               # each pair’s contribution

        # donor – sample proportionally to its harmful contribution
        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – pick an index with small gradient (i.e. little effect on max)
        # gradient for index i is 2*a_{k_max‑i} (zero outside the window)
        grad = np.zeros(n)
        grad[idx] = 2.0 * a[k_max - idx]
        thresh = np.percentile(grad, 30)  # 30‑th percentile as “low’’ gradient
        low_candidates = np.where(grad <= thresh)[0]
        if low_candidates.size:
            receiver = rng.choice(low_candidates)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # --- propose a small transfer ---
        max_delta = min(a[donor] * 0.04, target_sum * 0.001)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # --- update convolution exactly (O(n)) ---
        b_new = b.copy()
        # linear terms from -2*delta*a (donor) and +2*delta*a (receiver)
        b_new[donor:donor + n] += -2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic terms
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # --- simulated‑annealing acceptance ---
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Local refine improved score={best_val:.6f}")

        # --- occasional full recomputation to avoid numerical drift ---
        if time.time() - last_recalc > 12.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_recalc = time.time()
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑norm improved score={best_val:.6f}")

        # tiny jitter now and then to shake out of plateaus
        if (int(elapsed * 100) % 2500) == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.005, target_sum * 0.0004)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5            # safety margin

    # --------------------------------------------------------------
    # Choose problem size based on the time budget
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 500:
        n = 15000
    else:
        n = 12000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Warm‑start (reuse the best known construction if available)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
    else:
        # fallback: decreasing power‑law
        init = (np.arange(1, n + 1, dtype=np.float64) ** (-0.58))

    init = init * (target_sum / init.sum())
    best_seq = init.copy()
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] Initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – exponential (mirror) descent
    # --------------------------------------------------------------
    if time.time() < deadline:
        exp_deadline = time.time() + 0.15 * (deadline - time.time())
        a_exp, val_exp = _exponential_descent(best_seq, target_sum, exp_deadline)
        if val_exp < best_score:
            best_seq, best_score = a_exp, val_exp
            print(f"[{time.time():.2f}] After exponential descent best={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – projected gradient descent with restarts
    # --------------------------------------------------------------
    if time.time() < deadline:
        gd_deadline = time.time() + 0.30 * (deadline - time.time())
        a_gd, val_gd = _gradient_descent_with_restarts(best_seq,
                                                       target_sum,
                                                       gd_deadline)
        if val_gd < best_score:
            best_seq, best_score = a_gd, val_gd
            print(f"[{time.time():.2f}] After GD restarts best={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – sub‑gradient descent on true max‑convolution
    # --------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + 0.15 * (deadline - time.time())
        a_sub, val_sub = _subgradient_descent_max(best_seq,
                                                   target_sum,
                                                   sub_deadline)
        if val_sub < best_score:
            best_seq, best_score = a_sub, val_sub
            print(f"[{time.time():.2f}] After sub‑gradient descent best={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 5 – exact local refinement (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_ref, val_ref = _local_refine(best_seq, target_sum, deadline)
        if val_ref < best_score:
            best_seq, best_score = a_ref, val_ref
        print(f"[{time.time():.2f}] After local refinement best={best_score:.6f}")

    # --------------------------------------------------------------
    # Clean up, final evaluation and store for warm‑starts
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
