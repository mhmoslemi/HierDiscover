"""
Propose a step‑function with a low benchmark score.

Algorithm
---------

1. **Warm‑start** – use the best known construction if it exists,
   otherwise initialise several simple candidates:
   * the previous global best (`height_sequence_1`);
   * a short geometric series a_i = (1‑c/n) (1‑c/n)^i for a few
     values of `c` (c∈{0.6,1.0,1.4,1.8});

   Keep the one with the smallest true score returned by
   ``evaluate_sequence``.

2. **Hard‑max sub‑gradient descent** – minimise the exact
   objective 2·n·max(a*a)/(∑a)².
   At each iteration we
   * compute the full linear convolution by FFT,
   * locate the index `k_max` of the maximal entry,
   * form the sub‑gradient of the max‑entry (the sum of the two
     participating coefficients for every pair that contributes to
     `k_max`),
   * take a small step opposite to the gradient,
   * project back onto the simplex {x≥0, Σx = target_sum}.
   The step size decays like 1/√t.  The loop runs for a configurable
   fraction of the time budget and keeps the best iterate.

3. **Exact mass‑transfer local refinement** – a fast O(n) hill‑climbing
   on the true max‑convolution value (identical to the original
   implementation).  Tiny chunks of mass are moved from a “bad’’ index
   that heavily contributes to the current maximum to a “good’’ index.
   Simulated‑annealing acceptance allows occasional uphill moves.

4. **Final clean‑up** – strip trailing zeros, evaluate the final
   score, store the sequence for warm‑starts and return it.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based – fast for the sizes we use)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> tuple[float, int, np.ndarray]:
    """
    Returns (max_value, index_of_max, full_convolution).
    """
    b = _conv_fft(a)
    idx = int(np.argmax(b))
    return float(b[idx]), idx, b


# ----------------------------------------------------------------------
# Simple optimisation helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_direct(a: np.ndarray, target_sum: float) -> float:
    """
    Compute the exact benchmark score using the same formula as
    ``evaluate_sequence`` (but without the safety clamping).
    """
    n = a.shape[0]
    max_b, _, _ = _max_convolution(a)
    return float(2.0 * n * max_b / (target_sum * target_sum))

# 1. Warm‑start candidates
# ----------------------------------------------------------------------


def _candidate_geometric(n: int, c: float) -> np.ndarray:
    """
    Geometric (exponential) prototype:
        a_i = (1 - c/n) * (1 - c/n)^i   (i = 0 … n‑1)
    The factor (1‑c/n) forces the sum of the infinite series to be 1.
    """
    eps = c / n
    r = 1.0 - eps
    a = (1.0 - r) * (r ** np.arange(n, dtype=np.float64))
    return a


def _initial_population(n: int, target_sum: float) -> np.ndarray:
    """Return the best of a few simple constructions."""
    candidates = []

    # warm‑start from previous global best if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            # linear interpolation to the new length
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            prev = np.interp(new_x, old_x, prev)
        candidates.append(prev)

    # geometric prototypes with several values of c
    for c in (0.6, 1.0, 1.4, 1.8):
        candidates.append(_candidate_geometric(n, c))

    # simple decreasing power‑law (fallback used by the previous code)
    candidates.append(np.arange(1, n + 1, dtype=np.float64) ** (-0.55))

    best_seq = None
    best_val = np.inf
    for seq in candidates:
        seq = seq * (target_sum / seq.sum())
        val = _score_direct(seq, target_sum)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
    print(f"[{time.time():.2f}] Warm‑start best score={best_val:.6f}")
    return best_seq, best_val

# 2. Hard‑max sub‑gradient descent
# ----------------------------------------------------------------------


def _hard_max_subgradient_descent(a0: np.ndarray,
                                 target_sum: float,
                                 deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution
    objective (no soft‑max smoothing).  The step size is η_t = η0 / sqrt(t+1).
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score_direct(a, target_sum)

    rng = np.random.default_rng()
    t = 0
    eta0 = 0.05 * target_sum                     # base step size

    while time.time() < deadline:
        max_b, k_max, _ = _max_convolution(a)

        # sub‑gradient of the maximal entry
        n = a.size
        grad = np.zeros_like(a)

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        jdx = k_max - idx

        # each pair (i,j) contributes a_i + a_j
        grad[idx] += a[jdx]
        grad[jdx] += a[idx]

        # step size schedule
        step = eta0 / np.sqrt(t + 1.0)
        a -= step * grad
        a = np.maximum(a, 0.0)                   # stay non‑negative
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if t % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score_direct(a, target_sum)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={t:5d} score={best_val:.6f}")

        t += 1

    return best_a, best_val

# 3. Exact local‑refinement (identical to the previous implementation)
# ----------------------------------------------------------------------


def _local_refine(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Exact hill‑climbing / simulated‑annealing on the true max‑convolution.
    Moves a tiny amount of mass from a “bad’’ index (large contribution
    to the current maximal entry) to a low‑contributing index.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    best_a = a.copy()
    best_val = _score_direct(a, target_sum)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.04                 # initial temperature for annealing
    last_recalc = start

    while time.time() < deadline:
        # --------------------------------------------------------------
        # Identify donor (high contribution) and receiver (low contribution)
        # --------------------------------------------------------------
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        prod = a[idx] * a[k_max - idx]               # contribution of each index

        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – pick from the low‑product tail
        weight_all = np.zeros(n, dtype=a.dtype)
        weight_all[lo:hi + 1] = prod
        thresh = np.percentile(weight_all, 30)       # 30‑th percentile
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # --------------------------------------------------------------
        # Propose a small transfer
        # --------------------------------------------------------------
        max_delta = min(a[donor] * 0.04, target_sum * 0.0008)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # --------------------------------------------------------------
        # Update the convolution exactly (O(n) slice operations)
        # --------------------------------------------------------------
        b_new = b.copy()
        # linear part: 2 * a * Δ
        b_new[donor:donor + n] += -2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ*Δ)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # --------------------------------------------------------------
        # Acceptance (simulated annealing)
        # --------------------------------------------------------------
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Local refine improved score={best_val:.6f}")

        # --------------------------------------------------------------
        # Periodic full re‑evaluation to avoid numerical drift
        # --------------------------------------------------------------
        if time.time() - last_recalc > 12.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_recalc = time.time()
            cur_val = _score_direct(a, target_sum)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑normalisation improved score={best_val:.6f}")

        # occasional tiny jitter
        if (int(elapsed * 100) % 2500) == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.005, target_sum * 0.0004)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score_direct(a, target_sum)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5            # safety margin

    # ------------------------------------------------------------------
    # Choose a problem size – a little smaller than the previous 12k
    # makes the LP‑style sub‑gradient cheap while still giving a good
    # bound.
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 10000
    elif budget_s >= 500:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # 1) Warm‑start
    # ------------------------------------------------------------------
    seq, best_score = _initial_population(n, target_sum)

    # ------------------------------------------------------------------
    # 2) Hard‑max sub‑gradient descent  (~30 % of budget)
    # ------------------------------------------------------------------
    subgrad_deadline = time.time() + 0.30 * (deadline - time.time())
    seq, score = _hard_max_subgradient_descent(seq, target_sum, subgrad_deadline)
    if score < best_score:
        best_score, seq = score, seq.copy()
    print(f"[{time.time():.2f}] After sub‑grad descent best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Exact local mass‑transfer refinement (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq, score = _local_refine(seq, target_sum, deadline)
        if score < best_score:
            best_score, seq = score, seq.copy()
        print(f"[{time.time():.2f}] After local refinement best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up and final evaluation
    # ------------------------------------------------------------------
    arr = np.array(seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in a subsequent call
    globals()["height_sequence_1"] = final_seq

    return final_seq
