"""
Subgradient + Gradient Optimiser for the step‑function benchmark.

The algorithm proceeds in several phases:

1. **Parametric pre‑search** – generates a modest set of candidate
   sequences from simple power‑law / exponential families (incl. jittered
   variants) and picks the best according to the exact score.

2. **Projected gradient descent (soft‑max)** – runs a few short
   momentum‑GD sessions on a smooth soft‑max approximation of the
   maximal convolution entry, with an annealed temperature β.

3. **Deterministic mass‑transfer** – a greedy Metropolis style refinement
   that moves mass from the “donor” index contributing most to the current
   maximal convolution to a low‑contributing “receiver”.

4. **Exact sub‑gradient descent** – uses the true sub‑gradient of the
   max‑convolution (average over all maximal indices).  The step size
   decays as O(1/√t) and a small momentum term is applied; the best
   vector seen is kept.

5. **Final polishing** – a short high‑β soft‑max GD run.

All updates respect the simplex constraint Σa = √(2 n) and a ≥ 0.
The routine respects the overall time budget and returns the best
sequence found (trailing zeros are trimmed).  The final score is printed
via `evaluate_sequence` if available.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Benchmark objective 2·n·max(a*a) / (∑a)²."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple families for the quick parametric search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Generate (i+1)^(-p) of length n and scale to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) * exp(-q * i / n) scaled to target_sum."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution (used in GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where conv is the full convolution a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))          # avoid overflow
    p = exps / np.sum(exps)                    # soft‑max weights

    # gradient = 2 * (a reversed) ⋆ p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Projected gradient descent with annealed soft‑max temperature
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 2e5,
                               step_factor: float = 0.6,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Momentum projected GD on the soft‑max approx. of max‑conv."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    step_base = target_sum * step_factor
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 300 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Multiple short GD runs with random jitter; keep the best result."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent(start,
                                            time.time() + run_budget,
                                            target_sum,
                                            max_beta=2e5,
                                            step_factor=0.8,
                                            gamma=0.99)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (phase B)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-5,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy moves that directly reduce the current maximal convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0.0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0.0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on max‑convolution (new phase)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         momentum_decay: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    The step size decays as O(1/√t) and a simple momentum term is used.
    Returns the best vector encountered and its score.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_score = _score(a)

    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        # Full convolution and mask of maximal entries
        conv = _conv_fft(a)
        max_b = conv.max()
        mask = (conv == max_b)                     # exact mask (typically 1–2 entries)
        w = mask.astype(np.float64)
        w_sum = w.sum()
        if w_sum == 0:
            # fallback – shouldn't happen
            w = np.ones_like(conv) / conv.size
        else:
            w /= w_sum                               # normalize to a probability vector

        # Sub‑gradient = 2 * (a_rev ⋆ w)_{n-1 : n-1+n}
        a_rev = a[::-1]
        Ar_fft = np.fft.rfft(a_rev, n=size)
        Wr_fft = np.fft.rfft(w, n=size)
        grad_rev = np.fft.irfft(Ar_fft * Wr_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * grad_rev[n - 1: n - 1 + n]

        # Momentum update
        momentum = momentum_decay * momentum + grad

        # Step size – decreasing with √iteration count
        if it == 0:
            lr = step_factor * target_sum
        else:
            lr = step_factor * target_sum / np.sqrt(it + 1)

        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            a -= lr * momentum / norm

        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it:5d} score={best_score:.6f}")

        # occasional jitter to escape tiny plateaus
        if it % 1500 == 0 and it > 0:
            rng = np.random.default_rng()
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine respects the time budget, returns the best sequence found,
    and prints progress information.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose problem size
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)   # Σa fixed to this value (scale‑invariant)

    # ------------------------------------------------------------------
    # Phase 0 – fast parametric pre‑search
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    candidates = []

    # pure power‑law
    for p in np.linspace(0.45, 0.65, 13):
        candidates.append(_power_law_sequence(n, p, target_sum))

    # power‑law + exponential tail
    for p in np.linspace(0.45, 0.65, 7):
        for q in np.linspace(0.0, 5.0, 9):
            candidates.append(_exp_power_law_sequence(n, p, q, target_sum))

    # previous best (if available)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        candidates.append(_resize_sequence(prev, n))

    # jittered variants of everything above
    for seq in list(candidates):
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        candidates.append(seq_j)

    # keep a manageable subset for the cheap scoring pass
    max_initial_evals = min(200, len(candidates))
    sample_idx = rng.choice(len(candidates), size=max_initial_evals, replace=False)

    best_seq = None
    best_score = np.inf

    for i in sample_idx:
        seq = candidates[i]
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
    print(f"[{time.time():.2f}] Initial best after parametric search = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Projected gradient descent with restarts (soft‑max)
    # ------------------------------------------------------------------
    gd_deadline = time.time() + (deadline - time.time()) * 0.30
    while time.time() < gd_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _gradient_descent_with_restarts(start_vec,
                                                         target_sum,
                                                         gd_deadline)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] GD improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Deterministic mass‑transfer refinement
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        a_ref, val_ref = _deterministic_mass_transfer(best_seq,
                                                      deadline,
                                                      target_sum,
                                                      start_temp=0.5,
                                                      end_temp=1e-6,
                                                      low_frac=0.05)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] DMT improved best = {best_score:.6f}")

        # break early if we are very close to the deadline
        if time.time() > deadline - 2.0:
            break

    # ------------------------------------------------------------------
    # Phase C – Exact sub‑gradient descent (new phase)
    # ------------------------------------------------------------------
    subgrad_deadline = time.time() + (deadline - time.time()) * 0.25
    if time.time() < subgrad_deadline - 2.0:
        a_sub, val_sub = _subgradient_descent(best_seq,
                                              subgrad_deadline,
                                              target_sum,
                                              step_factor=0.55,
                                              momentum_decay=0.9)
        if val_sub < best_score:
            best_score = val_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑grad improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase D – Final polishing (high‑β soft‑max GD)
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _annealed_gradient_descent(best_seq,
                                                    deadline,
                                                    target_sum,
                                                    max_beta=1e6,
                                                    step_factor=0.4,
                                                    gamma=0.99)
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing: trim trailing zeros, return as Python list
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final evaluation (harmless if evaluate_sequence is not defined)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
