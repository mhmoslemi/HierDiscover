"""
Propose a non‑negative step function with a low maximal self‑convolution.

The pipeline:
1. Build a diverse pool of shapes (power‑law, Gaussian, triangular, random,
   and the previous best).  All are projected onto the ℓ¹‑simplex
   {x≥0, Σx = √(2 n)} – with this normalisation the evaluation score equals the
   maximal convolution entry.

2. Keep the best few pool members and run a fast Adam‑based projected gradient
   descent on a soft‑max approximation of the max‑convolution.  The optimiser
   uses an annealing schedule for the soft‑max temperature β and a per‑coordinate
   Adam step size.

3. Refine the result with a greedy deterministic mass‑transfer that moves a
   small amount of mass from indices that contribute heavily to the current
   peak to indices that contribute little.  The temperature for Metropolis
   acceptance decays exponentially.

4. Apply cheap re‑ordering tricks (alternating large/small, reversal, sorting).

5. Perform a simulated‑annealing over swaps of two positions (permutation
   annealing) to explore the combinatorial space of orderings.

6. Polish the best sequence with a final Adam pass at a very high β, and a
   short sub‑gradient descent on the exact max‑convolution.

All phases respect the global time budget (default 1000 s).  The final list
is returned and also stored in the global variable ``height_sequence_1``
for use as a warm‑start in a later run.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalize to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma_frac: float, target_sum: float) -> np.ndarray:
    """Gaussian centred at the middle of the array, σ = sigma_frac·n."""
    sigma = max(1.0, sigma_frac * n)
    xs = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-((xs - centre) ** 2) / (2.0 * sigma * sigma))
    a *= target_sum / a.sum()
    return a


def _triangular_sequence(n: int, target_sum: float) -> np.ndarray:
    """Linear decreasing from 1 to 0."""
    a = np.linspace(1.0, 0.0, n, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient: 2 * (a reversed)⋆p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑based projected gradient descent on the soft‑max
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           target_sum: float,
                           deadline: float,
                           max_beta: float = 1e5,
                           step_factor: float = 0.1) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max approximation of the max‑convolution.
    The sequence is projected onto the ℓ¹‑simplex after each update.
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    base_lr = target_sum * step_factor
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        # gradient of the smooth max
        g, _ = _gradient_softmax(a, beta)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # element‑wise step size
        lr_vec = base_lr / (np.sqrt(v_hat) + eps)

        a = a - lr_vec * m_hat
        a = _project_to_simplex(a, target_sum)

        # beta schedule (exponential annealing)
        if t % 200 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        # occasional jitter to escape plateaus
        if t % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={t:5d} β={beta:6.1f} score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer that lowers the current maximal convolution entry.
    Moves are accepted with a Metropolis rule whose temperature decays
    exponentially with the remaining time.
    """
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    # initialise convolution once; updated after each accepted move
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        # temperature schedule
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # index of the current maximal convolution entry
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution for each participating index (a_i + a_{kmax-i})
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        # bound the amount transferred
        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        # perform move
        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)  # recompute after revert

        # occasional global jitter
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Re‑ordering utilities
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """
    Reorder a sequence by alternating the largest remaining element to the left
    and the right side (big–small pattern).
    """
    sorted_vals = np.sort(seq)[::-1]          # descending
    n = seq.shape[0]
    out = np.empty_like(seq)
    left = 0
    right = n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


# ----------------------------------------------------------------------
# Permutation annealing (swap‑based simulated annealing)
# ----------------------------------------------------------------------


def _permutation_annealing(a: np.ndarray,
                           target_sum: float,
                           deadline: float,
                           init_score: float | None = None) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing over swaps of two indices, keeping the mass fixed.
    The objective is the maximal convolution entry.
    """
    cur_seq = a.copy()
    cur_score = (init_score if init_score is not None
                 else _max_convolution_fft(cur_seq))
    best_seq = cur_seq.copy()
    best_score = cur_score

    start_time = time.time()
    n = cur_seq.shape[0]
    rng = np.random.default_rng()

    # temperature schedule parameters
    start_temp = 0.2
    end_temp = 1e-5

    while time.time() < deadline:
        elapsed = time.time() - start_time
        ratio = elapsed / (deadline - start_time) if deadline > start_time else 0.0
        temp = start_temp * (end_temp / start_temp) ** ratio

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # perform swap
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        new_score = _max_convolution_fft(cur_seq)
        delta = new_score - cur_score

        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / max(temp, 1e-12)):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
                print(f"[{time.time():.2f}] Perm anneal new best = {best_score:.6f}")
        else:
            # revert swap
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        # occasional full reshuffle near the end
        if ratio > 0.9 and rng.random() < 0.01:
            rng.shuffle(cur_seq)
            cur_score = _max_convolution_fft(cur_seq)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Sub‑gradient refinement on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_refinement(a0: np.ndarray,
                            target_sum: float,
                            deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent directly on the max‑convolution.
    The sub‑gradient of max_k (a*a)[k] is 2·a_{k*−i} on the active index k*.
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    it = 0
    # step size decays like 1/√t
    base_lr = target_sum * 0.5

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # build sub‑gradient
        idx = np.arange(a.shape[0])
        partner = kmax - idx
        mask = (partner >= 0) & (partner < a.shape[0])
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[partner[mask]]

        lr = base_lr / np.sqrt(it + 1.0)
        a = a - lr * grad
        a = _project_to_simplex(a, target_sum)

        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it:5d} best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with a low benchmark score.
    The routine orchestrates several optimisation stages described above.
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n gives a tighter theoretical lower bound)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 400:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Build an initial pool of diverse candidates
    # --------------------------------------------------------------
    candidates = []

    # power‑law family
    for p in np.linspace(0.4, 0.8, 9):
        candidates.append(_power_law_sequence(n, p, target_sum))
        candidates.append(_power_law_sequence(n, p, target_sum)[::-1])

    # Gaussian family (different widths)
    for frac in (0.12, 0.15, 0.20, 0.25, 0.30):
        candidates.append(_gaussian_sequence(n, frac, target_sum))
        candidates.append(_gaussian_sequence(n, frac, target_sum)[::-1])

    # triangular shape
    candidates.append(_triangular_sequence(n, target_sum))
    candidates.append(_triangular_sequence(n, target_sum)[::-1])

    # previous best (if any) – resized if necessary
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.shape[0] != n:
            prev = _resize_sequence(prev, n)
        candidates.append(prev)
        rng = np.random.default_rng()
        jitter = rng.normal(scale=target_sum * 0.02, size=n)
        prev_j = prev + jitter
        prev_j = np.maximum(prev_j, 0.0)
        prev_j = _project_to_simplex(prev_j, target_sum)
        candidates.append(prev_j)

    # a handful of pure random starts
    rng = np.random.default_rng()
    for _ in range(5):
        rnd = rng.random(n)
        rnd = _project_to_simplex(rnd, target_sum)
        candidates.append(rnd)

    # --------------------------------------------------------------
    # Evaluate pool and keep the few best
    # --------------------------------------------------------------
    pool_scores = [_max_convolution_fft(seq) for seq in candidates]
    top_k = 5
    top_idxs = np.argsort(pool_scores)[:top_k]
    pool = [candidates[i] for i in top_idxs]
    best_seq = pool[0].copy()
    best_val = pool_scores[top_idxs[0]]
    print(f"[{time.time():.2f}] Initial pool best score = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase A – Adam projected gradient descent (soft‑max)
    # --------------------------------------------------------------
    if time.time() < deadline:
        gd_deadline = time.time() + (deadline - time.time()) * 0.45
        best_adam_seq = None
        best_adam_val = np.inf
        for init in pool:
            if time.time() >= gd_deadline:
                break
            a_opt, val_opt = _adam_gradient_descent(
                init,
                target_sum,
                gd_deadline,
                max_beta=2e5,
                step_factor=0.08,
            )
            if val_opt < best_adam_val:
                best_adam_val = val_opt
                best_adam_seq = a_opt.copy()
                print(f"[{time.time():.2f}] Adam improved best to {best_adam_val:.6f}")

        if best_adam_seq is not None and best_adam_val < best_val:
            best_val = best_adam_val
            best_seq = best_adam_seq.copy()
            print(f"[{time.time():.2f}] Global best after GD = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase B – Deterministic mass‑transfer refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        mt_deadline = time.time() + (deadline - time.time()) * 0.20
        a_mt, val_mt = _deterministic_mass_transfer(
            best_seq, mt_deadline, target_sum,
            start_temp=0.6, end_temp=1e-5, low_frac=0.04
        )
        if val_mt < best_val:
            best_val = val_mt
            best_seq = a_mt.copy()
            print(f"[{time.time():.2f}] After mass‑transfer best={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase C – Simple re‑ordering heuristics
    # --------------------------------------------------------------
    if time.time() < deadline:
        reorderings = [
            best_seq,
            _alternating_order(best_seq),
            best_seq[::-1],
            np.sort(best_seq)[::-1],    # descending order
        ]
        for cand in reorderings:
            val = _max_convolution_fft(cand)
            if val < best_val:
                best_val = val
                best_seq = cand.copy()
                print(f"[{time.time():.2f}] Reorder improvement -> score={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase D – Permutation annealing (swap‑based SA)
    # --------------------------------------------------------------
    if time.time() < deadline:
        perm_deadline = time.time() + (deadline - time.time()) * 0.20
        a_perm, val_perm = _permutation_annealing(
            best_seq, target_sum, perm_deadline, init_score=best_val
        )
        if val_perm < best_val:
            best_val = val_perm
            best_seq = a_perm.copy()
            print(f"[{time.time():.2f}] After permutation annealing best={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase E – Final Adam polishing with a higher β
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(
            best_seq,
            target_sum,
            deadline,
            max_beta=5e5,
            step_factor=0.04,
        )
        if val_fin < best_val:
            best_val = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polishing best={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase F – Sub‑gradient refinement on the exact max
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_sub, val_sub = _subgradient_refinement(best_seq, target_sum, deadline)
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑gradient refinement best={best_val:.6f}")

    # --------------------------------------------------------------
    # Trim trailing zeros and final evaluation
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    globals()["height_sequence_1"] = final_seq   # make it visible for the next run

    # sanity check (optional, ignored if evaluate_sequence is not imported)
    try:
        chk = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {chk:.6f}")
    except Exception:
        pass

    return final_seq
