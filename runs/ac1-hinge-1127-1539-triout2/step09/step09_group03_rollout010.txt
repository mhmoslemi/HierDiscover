def lp_step(a, target_sum, M, t=0.2):
    n = a.shape[0]
    # build constraint matrix
    # Nice to precompute row & col arrays.
    i = np.arange(n)
    j = np.arange(n)
    row = (i[:, None] + j).ravel()
    col = np.tile(j, n)
    data = np.repeat(a, n)
    A = csr_matrix((data, (row, col)), shape=(2*n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)   # maximize sum(b)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})
    if not res.success:
        # fallback: return original a
        return a
    b0 = res.x
    if b0.sum() == 0:
        # unlikely
        return a
    b = b0 * (target_sum / b0.sum())
    # combine
    new_a = (1 - t) * a + t * b
    # ensure non-negative
    new_a = np.maximum(new_a, 0.0)
    # renorm
    new_a = new_a * (target_sum / new_a.sum())
    return new_a

"""
Propose a step‑function (non‑negative sequence) with a low benchmark score.

Algorithm
---------
1. **Coarse multi‑scale stage** (≈30 % of the time budget)
   * Build a modest‑size problem (≈2000 points).
   * Run a short exponential‑descent + projected‑gradient phase.
   * If SciPy is available, perform a few Frank‑Wolfe style LP steps:
     maximise Σ b_j subject to a * b ≤ M (the current maximal convolution)
     and b≥0.  The result is blended into the current vector.
   * Upsample the coarse solution to the final size.

2. **Warm‑start candidates**
   * The up‑sampled coarse solution.
   * The previous best global sequence (if any) resized to the final length.
   * Their simple average.
   * Keep the best of the three w.r.t. the surrogate score.

3. **Fine optimisation on the full size**
   * Exponential (mirror) descent on a soft‑max surrogate of the max‑convolution.
   * Several projected‑gradient‑descent restarts with momentum and jitter.
   * Exact local refinement: move a tiny amount of mass from a “bad’’
     index (large contribution to the current maximal convolution entry) to a
     “good’’ index, updating the convolution in O(n).  Simulated‑annealing
     occasionally accepts uphill moves.

All steps respect non‑negativity and the ℓ¹‑norm constraint
(Σa = √(2n)).  The final sequence (truncated after the last non‑zero entry)
is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (LP stage)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    _SCIPY_AVAILABLE = True

    _SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  For Σa=√(2n) this equals max(a*a)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))

# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, Σx = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where conv is the full convolution a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # grad = 2 * (reverse(a) convolve p)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase‑2: exponential (multiplicative) descent
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (multiplicative) on the soft‑max surrogate.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase‑3: projected gradient descent with restarts
# ----------------------------------------------------------------------


def _annealed_gradient_descent_v2(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the soft‑max surrogate.
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    step_base = target_sum * 0.45
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.93
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0015 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GDv2 it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several short GD restarts with random jitter; keep the best."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_v2(start,
                                               time.time() + run_budget,
                                               target_sum)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase‑5: exact mass‑transfer local refinement (simulated annealing)
# ----------------------------------------------------------------------


def _local_refine(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Exact hill‑climbing / simulated‑annealing on the true max‑convolution.
    Moves a tiny amount of mass from an index that heavily contributes to
    the current maximal entry to a low‑contributing index.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    # temperature for occasional uphill moves
    T0 = best_val * 0.04
    last_recalc = start

    while time.time() < deadline:
        # --------------------------------------------------------------
        # Identify donor (high contribution) and receiver (low contribution)
        # --------------------------------------------------------------
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        prod = a[idx] * a[k_max - idx]               # contribution of each index

        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – sample from the low‑product tail
        weight_all = np.zeros(n, dtype=a.dtype)
        weight_all[lo:hi + 1] = prod
        thresh = np.percentile(weight_all, 30)       # 30‑th percentile
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # --------------------------------------------------------------
        # Propose a small transfer
        # --------------------------------------------------------------
        max_delta = min(a[donor] * 0.04, target_sum * 0.0008)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # --------------------------------------------------------------
        # Update convolution exactly (O(n) slice ops)
        # --------------------------------------------------------------
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # --------------------------------------------------------------
        # Acceptance (simulated annealing)
        # --------------------------------------------------------------
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Local refine improved score={best_val:.6f}")

        # --------------------------------------------------------------
        # Periodic re‑normalisation & full recomputation
        # --------------------------------------------------------------
        if time.time() - last_recalc > 12.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_recalc = time.time()
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑normalisation improved score={best_val:.6f}")

        # occasional tiny jitter
        if (int(elapsed * 100) % 2500) == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.005, target_sum * 0.0004)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Optional LP (coarse) stage
# ----------------------------------------------------------------------


def _build_conv_matrix(a: np.ndarray) -> csr_matrix:
    """
    Build the (2n‑1)×n constraint matrix A such that A @ b = a * b.
    Each column j of A is the vector a shifted by j rows.
    """
    n = a.shape[0]
    # row indices i + j
    i = np.arange(n, dtype=np.int64)[:, None]
    j = np.arange(n, dtype=np.int64)[None, :]
    row = (i + j).ravel()
    col = np.tile(j.ravel(), n)  # each column index repeated n times
    data = np.repeat(a, n)
    return csr_matrix((data, (row, col)), shape=(2 * n - 1, n))


def _lp_optimize(a: np.ndarray,
                 target_sum: float,
                 max_iters: int,
                 blend: float,
                 deadline: float) -> np.ndarray:
    """
    Perform a few Frank‑Wolfe style LP steps on a small problem.
    In each step we solve:
        maximize   Σ b_j
        subject to a * b ≤ M   (M = max(a*a)),   b ≥ 0
    The solution is rescaled to the target ℓ¹‑norm and blended into a.
    """
    if not _SCIPY_AVAILABLE:
        return a  # fallback – nothing to do

    a_cur = a.copy()
    n = a_cur.shape[0]

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        M = _max_convolution_fft(a_cur)

        # Build the constraint matrix for the current a_cur
        A = _build_conv_matrix(a_cur)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n)               # maximise Σ b_j  → minimise -Σ b_j
        bounds = [(0, None)] * n

        res = linprog(c,
                      A_ub=A,
                      b_ub=b_ub,
                      bounds=bounds,
                      method='highs')
        if not res.success:
            print(f"[{time.time():.2f}] LP step {it} failed: {res.message}")
            break

        b0 = res.x
        sum_b0 = b0.sum()
        if sum_b0 <= 0:
            break

        # Rescale to the required ℓ¹‑norm
        b_scaled = b0 * (target_sum / sum_b0)

        # Blend into the current vector
        a_cur = (1.0 - blend) * a_cur + blend * b_scaled
        a_cur = np.maximum(a_cur, 0.0)
        a_cur = a_cur * (target_sum / a_cur.sum())

        cur_score = _score(a_cur)
        print(f"[{time.time():.2f}] LP iter {it} score={cur_score:.6f}")

    return a_cur

# Main entry point

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5            # safety margin

    # ------------------------------------------------------------------
    # Choose final problem size
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n_final = 15000
    elif budget_s >= 500:
        n_final = 12000
    else:
        n_final = 8000

    target_sum = np.sqrt(2.0 * n_final)

    # ------------------------------------------------------------------
    # Warm‑start from the global best if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev_seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
        prev_seq = _resize_sequence(prev_seq, n_final)
        prev_seq = prev_seq * (target_sum / prev_seq.sum())
    else:
        prev_seq = None

    # ------------------------------------------------------------------
    # Coarse multi‑scale stage (≈30 % of total budget)
    # ------------------------------------------------------------------
    n_coarse = min(2000, max(600, n_final // 6))
    target_sum_coarse = np.sqrt(2.0 * n_coarse)

    if "height_sequence_1" in globals():
        coarse = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n_coarse)
    else:
        # simple power‑law as a cheap initial guess
        coarse = (np.arange(1, n_coarse + 1, dtype=np.float64) ** (-0.55))
    coarse = coarse * (target_sum_coarse / coarse.sum())

    # exponential descent (small slice of time)
    if time.time() < deadline:
        exp_dead_c = time.time() + 0.07 * (deadline - time.time())
        coarse, _ = _exponential_descent(coarse, target_sum_coarse, exp_dead_c)

    # projected GD restarts (small slice)
    if time.time() < deadline:
        gd_dead_c = time.time() + 0.13 * (deadline - time.time())
        coarse, _ = _gradient_descent_with_restarts(coarse,
                                                    target_sum_coarse,
                                                    gd_dead_c)

    # LP refinement (only if SciPy is present)
    if time.time() < deadline and _SCIPY_AVAILABLE:
        lp_dead_c = time.time() + 0.10 * (deadline - time.time())
        coarse = _lp_optimize(coarse,
                              target_sum_coarse,
                              max_iters=5,
                              blend=0.2,
                              deadline=lp_dead_c)

    coarse_score = _score(coarse)
    print(f"[{time.time():.2f}] Coarse stage best score={coarse_score:.6f}")

    # ------------------------------------------------------------------
    # Upsample to final size and assemble warm‑start candidates
    # ------------------------------------------------------------------
    upsampled = _resize_sequence(coarse, n_final)
    upsampled = upsampled * (target_sum / upsampled.sum())

    candidates = [upsampled]
    if prev_seq is not None:
        candidates.append(prev_seq)
        avg = 0.5 * (upsampled + prev_seq)
        avg = avg * (target_sum / avg.sum())
        candidates.append(avg)

    best_seq = None
    best_val = np.inf
    for cand in candidates:
        val = _score(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
    print(f"[{time.time():.2f}] After warm‑up best score={best_val:.6f}")

    # ------------------------------------------------------------------
    # Fine optimisation on the full size
    # ------------------------------------------------------------------
    # Exponential descent
    if time.time() < deadline:
        exp_dead_f = time.time() + 0.15 * (deadline - time.time())
        a_exp, val_exp = _exponential_descent(best_seq, target_sum, exp_dead_f)
        if val_exp < best_val:
            best_val = val_exp
            best_seq = a_exp.copy()
        print(f"[{time.time():.2f}] After final exponential descent best={best_val:.6f}")

    # Projected GD restarts
    if time.time() < deadline:
        gd_dead_f = time.time() + 0.30 * (deadline - time.time())
        a_gd, val_gd = _gradient_descent_with_restarts(best_seq,
                                                       target_sum,
                                                       gd_dead_f)
        if val_gd < best_val:
            best_val = val_gd
            best_seq = a_gd.copy()
        print(f"[{time.time():.2f}] After final GD restarts best={best_val:.6f}")

    # Exact local refinement (uses up the remaining time)
    if time.time() < deadline:
        a_ref, val_ref = _local_refine(best_seq, target_sum, deadline)
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
        print(f"[{time.time():.2f}] After final local refinement best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up and final evaluation
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # Store for possible warm‑starts in a subsequent call
    globals()["height_sequence_1"] = final_seq

    return final_seq
