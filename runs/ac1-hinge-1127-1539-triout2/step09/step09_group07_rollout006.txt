"""
Propose a step‑function sequence that minimises ``evaluate_sequence``.

Algorithm outline
-----------------
1. **Exploration** – generate many random candidates from a rich
   family of parametric shapes (Dirichlet, Gamma, power‑law,
   exponential, sparse rectangular bumps, Gaussians, half‑sine,
   logistic, inverse‑power‑law, triangular).  Keep the best *K*
   candidates (by the peak of the self‑convolution).

2. **Multi‑scale Adam refinement** – for each shortlisted candidate
   successively upscale the vector (geometric schedule up to
   ≈100 k entries) and run a custom Adam optimiser that works on the
   log‑softmax parametrisation (so the ℓ₁‑norm constraint
   ``sum = √(2 n)`` is satisfied automatically).  The optimiser uses the
   exact sub‑gradient of the hard maximum (soft‑max with a temperature
   that is annealed to zero).

3. **Focused large‑scale optimisation** – take the best vector,
   stretch it to the largest allowed size and devote the remaining
   wall‑clock time to a longer Adam run (larger learning rate,
   slower decay).  A small amount of Gaussian jitter is injected
   periodically to escape plateaus.

4. **Fine‑tuning** – a brief simulated‑annealing walk followed by a
   cheap greedy hill‑climber on the final candidate.

All steps respect the 1000 s budget (a 5 s safety margin).  The best
sequence found is stored in the global ``height_sequence_1`` and also
returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Peak value of a*a (full convolution)."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Gradient of the exact max‑convolution (hard‑max sub‑gradient)
# ----------------------------------------------------------------------
def _max_gradient(a: np.ndarray, tau: float):
    """
    Returns (grad, max_b) where grad = ∂ max(b) / ∂a
    and max_b = max(b) with b = a*a (full convolution).

    If ``tau`` is very small we use the exact active set (all indices
    within a tiny tolerance of the maximum) and average their
    gradients; otherwise a τ‑soft‑max is employed.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        eps = max_b * 1e-9 + 1e-12
        mask = (b >= max_b - eps).astype(float)
        s = mask.sum()
        if s == 0:
            mask = np.ones_like(b) / b.size
        else:
            mask /= s
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / exps.sum()

    # gradient: 2 * conv(mask, rev(a)) restricted to length n
    a_rev = a[::-1]
    conv_len = mask.size + a.size - 1
    f_len = _next_pow2(conv_len)

    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)

    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    grad = 2.0 * conv[a.size - 1:a.size - 1 + a.size]
    return grad, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   lr0: float = 0.5,
                   tau_decay: float = 0.9995,
                   lr_decay: float = 0.99985,
                   jitter_every: int = 400) -> tuple[np.ndarray, float]:
    """
    Minimise the peak of the self‑convolution under the ℓ₁‑norm
    constraint (sum = √(2 n)).  Returns the best vector seen and its
    hard peak.
    """
    n = a0.shape[0]
    S = math.sqrt(2.0 * n)                     # required ℓ₁‑norm

    # initialise logits from the feasible a0 (log‑softmax)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                           # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr0

    # initialise temperature for τ‑soft‑max
    tau = max(_full_convolution_fft(a0).max(), 0.5)

    best_a = a0.copy()
    best_peak = _max_convolution(a0)
    it = 0

    while time.time() < deadline:
        it += 1

        # recover a from logits (log‑softmax)
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)               # shift for stability
        soft = exp_p / exp_p.sum()
        a = soft * S

        # gradient of (soft‑)max convolution
        g, peak = _max_gradient(a, tau)

        # keep best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # chain rule to gradients w.r.t. logits
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # annealing schedules
        tau *= tau_decay
        lr *= lr_decay

        # occasional jitter
        if jitter_every and (it % jitter_every) == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak

def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Gamma‑draw → Dirichlet‑like vector."""
    alpha = rng.uniform(0.2, 5.0)
    raw = rng.gamma(alpha, 1.0, size=n)
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sparse Gamma entries (shape < 1)."""
    shape = rng.uniform(0.1, 0.8)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    """All ones."""
    return np.full(n, 1.0)


def _init_left_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotone decreasing power‑law, optionally mixed with a uniform background."""
    idx = np.arange(n)
    p = rng.uniform(0.3, 3.0)                # exponent
    offset = rng.uniform(0.0, 10.0)           # horizontal shift
    mix = rng.uniform(0.0, 1.0)               # background weight
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * (idx + offset) ** (-p)
    return raw


def _init_left_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotone decreasing exponential, optionally mixed with a uniform background."""
    idx = np.arange(n)
    lam = rng.uniform(0.001, 0.02)           # decay rate
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * np.exp(-lam * idx)
    return raw


def _init_sparse_bumps(n: int, rng: np.random.Generator) -> np.ndarray:
    """Few random rectangular bumps on a tiny flat background."""
    raw = np.full(n, rng.uniform(0.0, 0.15))
    nbumps = rng.integers(1, max(2, n // 250))
    for _ in range(nbumps):
        center = rng.integers(0, n)
        width = rng.integers(3, max(5, n // 100))
        height = rng.uniform(0.8, 4.0)
        lo = max(0, center - width // 2)
        hi = min(n, lo + width)
        raw[lo:hi] = np.maximum(raw[lo:hi], height)
    return raw


def _init_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    """Centered Gaussian bump (continuous) with random width."""
    mu = (n - 1) / 2.0
    sigma = rng.uniform(0.05, 0.30) * n   # 5‑30 % of the length
    x = np.arange(n)
    raw = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    return raw


def _init_half_sine(n: int, rng: np.random.Generator) -> np.ndarray:
    """Half‑sine shape (zero at the ends, peak in the middle)."""
    x = np.arange(n)
    raw = np.sin(np.pi * x / (n - 1))
    raw = raw * rng.uniform(0.5, 2.0) + rng.uniform(0.0, 0.2)
    return raw


def _init_logistic(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sigmoidal shape with random steepness and offset."""
    x = np.arange(n)
    steep = rng.uniform(0.02, 0.15) * n
    mid = rng.uniform(0.3, 0.7) * (n - 1)
    raw = 1.0 / (1.0 + np.exp(-(x - mid) / steep))
    if rng.random() < 0.3:
        raw = 1.0 - raw          # occasional flip
    raw = raw * rng.uniform(0.5, 2.0) + rng.uniform(0.0, 0.3)
    return raw


def _init_inverse_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """Decreasing inverse‑power‑law, optionally mixed with a uniform background."""
    idx = np.arange(n)
    p = rng.uniform(0.5, 2.5)
    offset = rng.uniform(0.0, 5.0)
    raw = (idx + offset) ** (-p)
    mix = rng.uniform(0.0, 0.5)
    raw = mix * np.ones(n) + (1 - mix) * raw
    return raw


def _init_triangular(n: int, rng: np.random.Generator) -> np.ndarray:
    """Symmetric triangular bump (peak in the middle)."""
    raw = np.minimum(np.arange(1, n + 1), np.arange(n, 0, -1)).astype(float)
    raw = raw * rng.uniform(0.5, 1.5) + rng.uniform(0.0, 0.3)
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Occasionally warm‑start from the global best; otherwise draw from a
    randomly chosen parametric family.
    """
    S = math.sqrt(2.0 * n)

    # Warm start occasionally
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    styles = ['dirichlet', 'gamma', 'uniform',
              'left_powerlaw', 'left_exponential',
              'sparse', 'gaussian', 'half_sine',
              'logistic', 'inverse_powerlaw', 'triangular']
    style = rng.choice(styles)

    if style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'left_powerlaw':
        raw = _init_left_powerlaw(n, rng)
    elif style == 'left_exponential':
        raw = _init_left_exponential(n, rng)
    elif style == 'sparse':
        raw = _init_sparse_bumps(n, rng)
    elif style == 'gaussian':
        raw = _init_gaussian(n, rng)
    elif style == 'half_sine':
        raw = _init_half_sine(n, rng)
    elif style == 'logistic':
        raw = _init_logistic(n, rng)
    elif style == 'inverse_powerlaw':
        raw = _init_inverse_powerlaw(n, rng)
    else:   # triangular
        raw = _init_triangular(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Simulated annealing / greedy hill climbing fine‑tuner
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Mass‑transfer walk on the simplex with exponential acceptance."""
    n = len(seq)
    S = math.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score
    temperature = 0.15 * cur_score  # initial temperature

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] == 0.0:
            continue
        delta = rng.uniform(0.0, 0.3) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())
        new_score = _max_convolution(new)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temperature, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

        temperature *= 0.99995
        if temperature < 1e-6:
            temperature = 1e-6

    return best, best_score


def _hill_climb(seq: np.ndarray,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Very cheap greedy hill‑climber (strictly downhill moves only)."""
    n = len(seq)
    S = math.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] < 1e-12:
            continue
        delta = rng.uniform(0.0, 0.15) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())
        new_score = _max_convolution(new)
        if new_score < cur_score:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The algorithm follows the stages described in the module docstring.
    The best construction is stored globally as ``height_sequence_1`` and
    also returned as a plain list.
    """
    rng = np.random.default_rng(seed)
    safety_margin = 5.0
    end_time = time.time() + budget_s - safety_margin
    if end_time <= time.time():
        raise RuntimeError("Budget too small after safety margin.")

    # ------------------------------------------------------------------
    # Warm‑start from a previously stored construction, if any
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # ------------------------------------------------------------------
    # Phase 1 – fast random exploration
    # ------------------------------------------------------------------
    explore_deadline = time.time() + 0.15 * budget_s
    top_k = 20
    top_candidates = []            # list of (score, seq)
    best_seq = None
    best_score = np.inf
    attempts = 0

    while time.time() < explore_deadline:
        attempts += 1
        # length distribution – mix of moderate and larger sizes
        if rng.random() < 0.6:
            n = int(rng.integers(1500, 6000))
        else:
            n = int(rng.integers(6000, 25000))

        a0 = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a0)

        if peak < best_score:
            best_score = peak
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempts:5d} n={n:5d}  new best={best_score:.6f}")

        # maintain top‑k list
        if len(top_candidates) < top_k:
            top_candidates.append((peak, a0.copy()))
            top_candidates.sort(key=lambda x: x[0])
        elif peak < top_candidates[-1][0]:
            top_candidates[-1] = (peak, a0.copy())
            top_candidates.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # Safety net – ensure at least one candidate exists
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 3000
        S = math.sqrt(2.0 * n)
        best_seq = np.full(n, S / n)
        best_score = _max_convolution(best_seq)
        best_n = n
        top_candidates = [(best_score, best_seq.copy())]

    # ------------------------------------------------------------------
    # Phase 2 – multi‑scale Adam refinement on the top candidates
    # ------------------------------------------------------------------
    max_len = min(120000, best_n * 30)          # generous upper bound
    lengths = [best_n]
    cur_len = best_n
    while cur_len * 1.4 < max_len:
        cur_len = int(cur_len * 1.4)
        lengths.append(cur_len)
    lengths.append(max_len)
    lengths = sorted(set(lengths))

    for init_idx, (init_score, init_seq) in enumerate(top_candidates):
        cur_seq = init_seq.copy()
        for target_n in lengths:
            if time.time() >= end_time - 30.0:
                break

            # interpolate to the new length and renormalise
            a_scaled = _scale_sequence(cur_seq, target_n)
            a_scaled = np.clip(a_scaled, 0.0, None)
            a_scaled = a_scaled * (math.sqrt(2.0 * target_n) / a_scaled.sum())

            # allocate a slice of the remaining time (capped)
            remaining = end_time - time.time()
            slice_budget = min(remaining * 0.4, 180.0)

            # randomise a few hyper‑parameters for diversity
            lr0 = rng.choice([0.55, 0.60, 0.65, 0.70])
            tau_decay = rng.choice([0.9993, 0.9995, 0.9997])
            refined, peak = _adam_optimize(a_scaled,
                                           time.time() + slice_budget,
                                           rng,
                                           lr0=lr0,
                                           tau_decay=tau_decay,
                                           lr_decay=0.99988,
                                           jitter_every=300)

            if peak < best_score:
                best_score = peak
                best_seq = refined.copy()
                best_n = target_n
                print(f"[refine] init#{init_idx+1} len={target_n:5d}  improved={best_score:.6f}")

            cur_seq = refined

        if time.time() >= end_time - 30.0:
            break

    # ------------------------------------------------------------------
    # Phase 3 – large‑scale fine‑tuning on the current best candidate
    # ------------------------------------------------------------------
    if time.time() < end_time - 5.0:
        # upscale to the maximum length if not already there
        if len(best_seq) != max_len:
            best_seq = _scale_sequence(best_seq, max_len)
            best_seq = np.clip(best_seq, 0.0, None)
            best_seq = best_seq * (math.sqrt(2.0 * max_len) / best_seq.sum())

        remaining = end_time - time.time()
        # a longer Adam run (most of the remaining budget)
        adam_budget = remaining * 0.7
        best_seq, best_score = _adam_optimize(best_seq,
                                              time.time() + adam_budget,
                                              rng,
                                              lr0=0.62,
                                              tau_decay=0.9994,
                                              lr_decay=0.99992,
                                              jitter_every=200)

        remaining = end_time - time.time()
        # simulated annealing
        if remaining > 10.0:
            best_seq, best_score = _anneal(best_seq,
                                           time.time() + 0.6 * remaining,
                                           rng)
            remaining = end_time - time.time()

        # final greedy hill‑climb
        if remaining > 5.0:
            best_seq, best_score = _hill_climb(best_seq,
                                               time.time() + remaining,
                                               rng)

    # ------------------------------------------------------------------
    # Final sanity check via the supplied black‑box evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  internal max = {best_score:.6f}"
          f"  length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
