def _init_arcsine(n, rng):
    i = np.arange(n)
    # product distance to edges
    d = i * (n-1-i) + 0.5
    exponent = rng.uniform(0.3, 0.7)
    offset = rng.uniform(0.0, 10.0)
    raw = 1.0 / np.sqrt(d * exponent + offset)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * raw + (1 - mix) * base

* Use a richer family of random initial shapes, including an
  “arcsine‑like’’ distribution that concentrates mass near the ends.
* Enforce symmetry (a_i = a_{n‑1‑i}) after each update – the optimal
  solution is believed to be symmetric, which reduces the effective
  dimension and smoothes the search surface.
* Adam optimiser on a τ‑soft‑max approximation of the max, with a very
  slow temperature decay and a modest learning‑rate schedule.
* A hard‑max sub‑gradient descent for final polishing.
* A short simulated‑annealing / hill‑climb phase that moves a small
  amount of mass between two positions, accepting occasional
  uphill moves.
* All phases respect the overall 1000 s budget; timing is adapted
  dynamically.

The final sequence is stored in the global variable ``height_sequence_1``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, scaling, etc.)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a∗a."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce a_i = a_{n‑1‑i}."""
    return (a + a[::-1]) * 0.5


# ----------------------------------------------------------------------
# τ‑soft‑max gradient (used by Adam)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser (mirror descent on the simplex, symmetric version)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex (log‑softmax parametrisation) with
    enforced symmetry. Returns the best feasible vector and its hard‑max
    convolution value.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)               # target ℓ1‑norm

    # initialise logits from a0 (entropic parametrisation)
    eps = 1e-12
    prob = np.maximum(a0 / S, eps)
    prob = prob / prob.sum()
    p = np.log(prob)                   # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.3                           # a safe start

    # temperature initialisation
    tau = max(_full_convolution_fft(a0).max(), 5.0)

    best_a = a0.copy()
    best_peak = _max_convolution(a0)
    it = 0
    while time.time() < deadline:
        it += 1

        # recover a from logits (log‑softmax) and enforce symmetry
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S
        a = _symmetrize(a)                     # symmetry
        a = a * (S / a.sum())                 # renormalise (tiny drift correction)

        # convolution & τ‑soft‑max gradient
        g, peak = _softmax_gradient(a, tau)

        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # gradient w.r.t. logits (mirror‑descent formula)
        mu = np.dot(a, g) / S                 # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)

        # clip for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 6.0 * S:
            grad_p = grad_p * (6.0 * S / clip_val)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # schedules (very slow decay)
        tau *= 0.99995
        lr *= 0.99997

        # occasional jitter to escape plateaus
        if it % 700 == 0:
            p += rng.normal(scale=0.004, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Sub‑gradient descent on the *hard* max (symmetric version)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple sub‑gradient descent that follows the entry attaining the
    current maximum, with projection onto the symmetric simplex.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    a = a0.copy()
    best_a = a.copy()
    best_val = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1

        b = _full_convolution_fft(a)
        k = int(b.argmax())
        val = b[k]

        if val < best_val:
            best_val = val
            best_a = a.copy()

        # sub‑gradient of (a*a)_k
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        jdx = k - idx
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[jdx]

        step = 0.6 * S / (np.sqrt(it) + 1.0)

        a = a - step * g
        a = np.maximum(a, 0.0)

        # enforce symmetry and renormalise
        a = _symmetrize(a)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, S / n)
        else:
            a = a * (S / s)

        # occasional jitter
        if it % 400 == 0:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                frac = rng.uniform(0.02, 0.10)
                amount = a[i] * frac
                a[i] -= amount
                a[j] += amount
                a = _symmetrize(a)
                a = a * (S / a.sum())

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing / hill‑climb (final polishing)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Very cheap simulated‑annealing that moves a random fraction of mass
    between two positions, accepting improvements or occasional
    uphill moves with a linearly decreasing temperature.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    cur = a0.copy()
    cur_val = _max_convolution(cur)
    best = cur.copy()
    best_val = cur_val

    start = time.time()
    # initial temperature: 5 % of current value
    T0 = max(cur_val * 0.05, 1e-6)

    while time.time() < deadline:
        # linearly decreasing temperature
        elapsed = time.time() - start
        T = max(T0 * (1.0 - elapsed / (deadline - start)), 1e-12)

        i, j = rng.integers(0, n, size=2)
        if i == j or cur[i] <= 0.0:
            continue
        frac = rng.uniform(0.01, 0.18)
        amount = cur[i] * frac
        new = cur.copy()
        new[i] -= amount
        new[j] += amount

        # re‑symmetrize and renorm
        new = _symmetrize(new)
        new = new * (S / new.sum())
        val = _max_convolution(new)

        if val < cur_val or rng.random() < np.exp(-(val - cur_val) / T):
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best = new.copy()

    return best, best_val


# ----------------------------------------------------------------------
# Parametric families for random initialisation
# ----------------------------------------------------------------------
def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    raw = mix * ush + (1.0 - mix) * base
    return raw


def _init_cosine(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    raw = 1.0 - np.cos(np.pi * i / (n - 1))
    raw = raw * rng.uniform(0.5, 2.0)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * raw + (1.0 - mix) * base


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    return rng.random(n)


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.05, 0.9)
    return rng.gamma(shape, 1.0, size=n)


def _init_arcsine(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Arcsine‑like shape: density ∝ 1/√(i·(n‑1‑i) + offset).  Concentrates
    mass near the ends, which tends to flatten the self‑convolution.
    """
    i = np.arange(n, dtype=float)
    prod = i * (n - 1 - i) + rng.uniform(0.0, 8.0)  # offset to avoid singularity
    alpha = rng.uniform(0.3, 0.7)                  # exponent ≈0.5
    raw = prod ** (-alpha)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * raw + (1.0 - mix) * base


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    With probability ≈0.2 we warm‑start from ``warm_seq`` (scaled).
    Otherwise we draw from one of the random families, now including an
    arcsine‑like distribution.
    The returned vector satisfies sum = √(2·n) and all entries ≥0.
    """
    S = np.sqrt(2.0 * n)

    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
        a = _symmetrize(a)
        a = a * (S / a.sum())
        return a

    # pick a random family
    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'gamma', 'uniform', 'dirichlet', 'ushape',
                        'cosine', 'arcsine'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'cosine':
        raw = _init_cosine(n, rng)
    elif style == 'arcsine':
        raw = _init_arcsine(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    a = _symmetrize(a)
    a = a * (S / a.sum())
    return a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence found as a plain
    Python list and stores it in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Warm‑start handling
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # ------------------------------------------------------------------
    # Phase 1 – fast random exploration (≈ 150 s)
    # ------------------------------------------------------------------
    explore_deadline = end_time - 600.0   # reserve ~600 s for refined phases
    best_seq = None
    best_score = np.inf
    attempts = 0

    while time.time() < explore_deadline:
        attempts += 1
        # Sample lengths favouring a few thousand entries
        if rng.random() < 0.75:
            n = int(rng.integers(1800, 6200))
        else:
            n = int(rng.integers(6200, 15000))

        a0 = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a0)

        if peak < best_score:
            best_score = peak
            best_seq = a0.copy()
            print(f"[explore] attempt={attempts:4d} n={n:5d}  new best={best_score:.6f}")

    if best_seq is None:
        # fallback to a simple uniform vector
        n = 3000
        S = np.sqrt(2.0 * n)
        best_seq = np.full(n, S / n)
        best_score = _max_convolution(best_seq)

    # ------------------------------------------------------------------
    # Phase 2 – upscale to several larger lengths (no optimisation)
    # ------------------------------------------------------------------
    cand_lengths = [len(best_seq)]
    for factor in [1.2, 1.5, 1.8, 2.2, 3.0]:
        L = int(len(best_seq) * factor)
        if 2000 <= L <= 25000:
            cand_lengths.append(L)
    cand_lengths = sorted(set(cand_lengths))

    cur_seq = best_seq
    cur_score = best_score
    cur_n = len(best_seq)

    for L in cand_lengths:
        if time.time() > end_time - 530.0:   # ensure time for later phases
            break
        a_scaled = _scale_sequence(cur_seq, L)
        a_scaled = np.clip(a_scaled, 0.0, None)
        S = np.sqrt(2.0 * L)
        a_scaled = a_scaled * (S / a_scaled.sum())
        a_scaled = _symmetrize(a_scaled)
        a_scaled = a_scaled * (S / a_scaled.sum())
        peak = _max_convolution(a_scaled)

        if peak < cur_score:
            cur_score = peak
            cur_seq = a_scaled.copy()
            cur_n = L
            print(f"[scale] length={L:5d}  improved max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam optimisation (≈ 350 s total, multiple restarts)
    # ------------------------------------------------------------------
    remaining = end_time - time.time() - 200.0   # reserve 200 s for later phases
    if remaining > 20.0:
        n_restart = max(1, int(remaining // 70.0))   # ~70 s per restart
        per_run = remaining / n_restart
        for r in range(n_restart):
            if time.time() > end_time - 200.0:
                break
            # add small random jitter before each restart
            noise = rng.normal(scale=0.015 * np.sqrt(2.0 * cur_n) / cur_n, size=cur_n)
            a_start = np.maximum(cur_seq + noise, 0.0)
            a_start = a_start * (np.sqrt(2.0 * cur_n) / a_start.sum())
            a_start = _symmetrize(a_start)
            a_start = a_start * (np.sqrt(2.0 * cur_n) / a_start.sum())

            a_opt, peak = _adam_optimize(a_start,
                                          time.time() + per_run,
                                          rng)
            if peak < cur_score:
                cur_score = peak
                cur_seq = a_opt.copy()
                print(f"[adam] restart={r} new best max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – sub‑gradient descent (≈ 120 s)
    # ------------------------------------------------------------------
    remaining = end_time - time.time() - 70.0
    if remaining > 15.0:
        a_opt, peak = _subgradient_descent(cur_seq,
                                           time.time() + remaining,
                                           rng)
        if peak < cur_score:
            cur_score = peak
            cur_seq = a_opt.copy()
            print(f"[subgrad] after polishing max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – simulated annealing (≈ 70 s)
    # ------------------------------------------------------------------
    remaining = end_time - time.time() - 20.0
    if remaining > 15.0:
        a_opt, peak = _simulated_annealing(cur_seq,
                                           time.time() + remaining,
                                           rng)
        if peak < cur_score:
            cur_score = peak
            cur_seq = a_opt.copy()
            print(f"[anneal] final max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – cheap random hill‑climb for the last few seconds
    # ------------------------------------------------------------------
    while time.time() < end_time - 1.0:
        a = cur_seq.copy()
        n = a.shape[0]
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        frac = rng.uniform(0.02, 0.12)
        amount = a[i] * frac
        a[i] -= amount
        a[j] += amount
        a = _symmetrize(a)
        a = a * (np.sqrt(2.0 * n) / a.sum())
        peak = _max_convolution(a)
        if peak + 1e-9 < cur_score:
            cur_score = peak
            cur_seq = a.copy()
            print(f"[hill] improved max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    final_seq = cur_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f}  internal max={cur_score:.6f}  length={len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
