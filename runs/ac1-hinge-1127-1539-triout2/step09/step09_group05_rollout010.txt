def _smooth_objective_and_grad(a, alpha):
    # compute conv = a*a via FFT
    b = _full_convolution_fft(a)
    # compute w = softmax(alpha * b)
    # careful to avoid overflow: subtract max
    maxb = b.max()
    exp_term = np.exp(alpha * (b - maxb))
    Z = exp_term.sum()
    w = exp_term / Z
    # gradient: g = 2 * (w * a_rev)[n-1:n-1+len(a)]
    a_rev = a[::-1]
    # convolve w with a_rev
    conv_len = len(w) + len(a_rev) - 1
    f_len = _next_pow2(conv_len)
    Fw = np.fft.rfft(w, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fw * Fa_rev, n=f_len)[:conv_len]

    best_val = _max_convolution(a)
    m = np.zeros_like(a)

        while time.time() < deadline:

"""
Improved step‑function optimiser.

Main ideas
-----------
* Keep the original random‑exploration + sub‑gradient pipeline as a fallback.
* Add a *smooth* optimisation stage that minimises a log‑sum‑exp
  approximation of the max‑convolution.  The gradient of this
  smoothed objective is computed analytically via FFTs and an Adam
  optimiser with Euclidean simplex projection is used.
* Use a longer schedule of lengths (up to ~30 000) and a new
  “inverse‑sqrt’’ parametric family to diversify the search.
* The whole routine respects the 1000 s budget and stores the final
  sequence in the global variable ``height_sequence_1``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a∗a."""
    return _full_convolution_fft(a).max()


# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. O(n))
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection onto {w | w_i ≥ 0, ∑ w_i = S}."""
    if S < 0:
        raise ValueError("Simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Rescaling utilities
# ----------------------------------------------------------------------
def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]

# Parametric families for random initialisation
# ----------------------------------------------------------------------
def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    raw = mix * ush + (1.0 - mix) * base
    return raw


def _init_sine(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    vals = np.sin(np.pi * (i + 1) / (n + 1))
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * vals + (1.0 - mix) * base
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:

def _init_invsqrt(n: int, rng: np.random.Generator) -> np.ndarray:
    """Inverse‑sqrt shape (larger at the edges, smaller in the middle)."""
    i = np.arange(n, dtype=float)
    d = (i + 0.5) * (n - i - 0.5)
    raw = 1.0 / np.sqrt(d + rng.uniform(1e-3, 1e-1))
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)

def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Occasionally warm‑starts from the globally best sequence and otherwise
    draws from a random parametric family.
    """
    S = np.sqrt(2.0 * n)

    # Warm start with probability ≈0.2
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
        return a

    # Choose a family
    style = rng.choice(
        ['powerlaw', 'exponential', 'linear', 'ushape',
         'sine', 'uniform', 'dirichlet', 'invsqrt'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'sine':
        raw = _init_sine(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    else:   # invsqrt
        raw = _init_invsqrt(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a

def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         global_best: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on max(a∗a)."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # --- gradient of the hard max ---
        b = _full_convolution_fft(a)
        max_val = b.max()
        mask = (b == max_val).astype(float)

        a_rev = a[::-1]
        conv_len = len(mask) + n - 1
        f_len = _next_pow2(conv_len)
        Fmask = np.fft.rfft(mask, n=f_len)
        Fa_rev = np.fft.rfft(a_rev, n=f_len)
        conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
        start = n - 1
        grad = 2.0 * conv[start:start + n]

        # Polyak step size
        norm_sq = np.dot(grad, grad)
        if norm_sq > 0:
            step = (max_val - best_val) / (norm_sq + 1e-12)
            step = np.clip(step, 1e-7, 0.2)
        else:
            step = 0.0

        a = _project_onto_simplex(a - step * grad, S)

        cur_val = _max_convolution(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()

        if it % 300 == 0:
            jitter = rng.normal(scale=0.001 * np.mean(a), size=n)
            a = _project_onto_simplex(a + jitter, S)

    return best_a, best_val

    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    cur_val = _max_convolution(a)
    best_a = a.copy()
    best_val = cur_val

    T = 0.05 * cur_val
    decay = 0.9999
    it = 0
    while time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        frac = rng.uniform(0.02, 0.18)
        delta = a[i] * frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _max_convolution(a_new)
        delta_val = new_val - cur_val
        if delta_val < 0 or rng.random() < np.exp(-delta_val / (T + 1e-12)):
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()

        T *= decay
        if it % 500 == 0:
            jitter = rng.normal(scale=0.001 * S, size=n)
            a = _project_onto_simplex(a + jitter, S)
            cur_val = _max_convolution(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Smoothed optimisation (new component)
# ----------------------------------------------------------------------
def _smooth_objective_and_grad(a: np.ndarray, alpha: float):
    """
    Returns (smooth_val, grad, hard_max) where
    smooth_val = (1/α) log ∑_k exp(α·(a∗a)[k])
    grad is the gradient of the smooth objective.
    """
    n = a.shape[0]

    # self‑convolution
    b = _full_convolution_fft(a)                 # length 2n‑1
    maxb = b.max()

    # stable softmax on α·b
    shifted = alpha * (b - maxb)                  # ≤ 0
    exp_term = np.exp(shifted)
    Z = exp_term.sum()
    w = exp_term / Z                              # weights, sum‑to‑1

    # gradient = 2 * (w ∗ a_rev)  (central window)
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    f_len = _next_pow2(conv_len)
    Fw = np.fft.rfft(w, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fw * Fa_rev, n=f_len)[:conv_len]
    start = n - 1
    grad = 2.0 * conv[start:start + n]

    smooth = maxb + np.log(Z) / alpha           # (1/α)log‑sum‑exp
    return smooth, grad, maxb


def _smooth_descent(a0: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator,
                    alphas) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed objective.
    alphas – list of increasing temperature parameters (α values).
    Returns the best feasible vector and its hard max value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _max_convolution(a)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    for alpha in alphas:
        # a modest number of inner steps for each α (adjustable)
        max_iters = 3000
        for _ in range(max_iters):
            if time.time() >= deadline:
                return best_a, best_val

            t += 1
            smooth, grad, cur_max = _smooth_objective_and_grad(a, alpha)

            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)

            # learning rate slowly decreasing with α
            lr = 0.2 / (np.sqrt(alpha) + 1e-8)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_onto_simplex(a - step, S)

            # true hard max (expensive but cheap enough)
            cur_val = _max_convolution(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                # reset moments – gives fresh direction after improvement
                m.fill(0.0)
                v.fill(0.0)
                t = 0

            # occasional random jitter to shake out plateaus
            if t % 500 == 0:
                jitter = rng.normal(scale=0.001 * S, size=n)
                a = _project_onto_simplex(a + jitter, S)

    return best_a, best_val

# Main entry point required by the harness

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises the provided
    ``evaluate_sequence``.  The routine combines random exploration,
    smoothed Adam optimisation, sub‑gradient descent and a final
    annealing hill‑climb.  The best sequence is stored in the global
    variable ``height_sequence_1`` and also returned as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    end_time = start_time + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # Warm‑start handling
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    best_seq = None
    best_val = np.inf
    attempts = 0

    # ------------------------------------------------------------------
    # Phase 1 – broad random exploration (≈ 50 % of budget)
    # ------------------------------------------------------------------
    explore_deadline = end_time - 300.0   # reserve ~300 s for refinement
    while time.time() < explore_deadline:
        attempts += 1
        # favour longer sequences (5 000 – 20 000)
        n = int(rng.integers(5000, 20001))
        a0 = _initial_vector(n, rng, warm_seq)
        val = _max_convolution(a0)
        if val < best_val:
            best_val = val
            best_seq = a0.copy()
            print(f"[explore] attempt={attempts:4d} n={n:5d}  new best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – scaled smoothing optimisation on a few lengths
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback to a modest uniform start
        n = 6000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_val = _max_convolution(best_seq)

    # pick a ladder of lengths (up to ~30 000)
    lengths = [len(best_seq)]
    for factor in [1.4, 2.0, 2.8, 3.6]:
        cand = int(len(best_seq) * factor)
        if 3000 <= cand <= 30000:
            lengths.append(cand)
    lengths = sorted(set(lengths))

    for target_n in lengths:
        if time.time() >= end_time - 200.0:
            break
        # rescale current best to the new length
        a_scaled = _scale_sequence(best_seq, target_n)
        S = np.sqrt(2.0 * target_n)
        a_scaled = np.clip(a_scaled, 0.0, None)
        a_scaled = a_scaled * (S / a_scaled.sum())

        # allocate a slice of the remaining budget (≈15 % each)
        remaining = end_time - time.time()
        slice_time = min(remaining * 0.15, 120.0)

        # smoothing schedule – start gentle, finish sharp
        alphas = [2.0, 5.0, 10.0, 20.0, 40.0, 80.0]
        a_opt, val = _smooth_descent(a_scaled,
                                      time.time() + slice_time,
                                      rng,
                                      alphas)

        if val < best_val:
            best_val = val
            best_seq = a_opt.copy()
            print(f"[smooth] target_n={target_n:5d}  improved={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – long sub‑gradient run on the current best length
    # ------------------------------------------------------------------
    if time.time() < end_time - 30.0:
        remaining = end_time - time.time() - 5.0
        a_opt, val = _subgradient_descent(best_seq,
                                          time.time() + remaining,
                                          rng,
                                          best_val)
        if val < best_val:
            best_val = val
            best_seq = a_opt.copy()
            print(f"[final subgrad] length={len(best_seq)}  best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated‑annealing hill‑climb (final polish)
    # ------------------------------------------------------------------
    if time.time() < end_time - 5.0:
        remaining = end_time - time.time() - 1.0
        a_opt, val = _anneal_local_search(best_seq,
                                          time.time() + remaining,
                                          rng)
        if val < best_val:
            best_val = val
            best_seq = a_opt.copy()
            print(f"[anneal] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f}  internal={best_val:.6f}  length={len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
