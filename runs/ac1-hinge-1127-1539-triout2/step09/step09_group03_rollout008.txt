def _subgradient_descent(a0, target_sum, deadline):
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng()

        b = _conv_fft(a)
        max_b = b.max()

        # compute subgradient

        n = a.shape[0]
        for k in act:
            lo = max(0, k - (n-1))
            hi = min(k, n-1)
            for i in range(lo, hi+1):
                j = k - i
                if i == j:
                    g[i] += 2.0 * a[i]
                else:
                    g[i] += a[j]
                    g[j] += a[i]

        norm = np.linalg.norm(g)
        if norm == 0:
            break
        step = 0.5 / (np.sqrt(it+1) * (norm / target_sum))
        a -= step * g
        # projection
        a = _project_to_simplex(a, target_sum)
        val = _score(a)
        if val < best_score:

            best_a = a.copy()

            jitter = rng.normal(scale=target_sum*0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)
        it += 1
    return best_a, best_score

"""
Step–function optimizer for the “max‑convolution / ℓ¹‑norm’’ benchmark.

The algorithm consists of three phases:

1. **Global search** – a few iterations of mirror (exponential) descent on a
   soft‑max surrogate of the maximal autocorrelation.  This quickly moves the
   mass away from the most harmful indices.

2. **Exact sub‑gradient descent** – the true non‑smooth objective
   ``max_k (a*a)_k`` is tackled directly.  The sub‑gradient is obtained from the
   active (maximal) convolution indices and a simple projected sub‑gradient
   method with diminishing steps and occasional jitter is applied.

3. **Deterministic mass‑transfer refinement** – a fast O(n) update scheme
   repeatedly transfers a tiny amount of mass from a “donor’’ index that
   contributes most to the current maximal convolution entry to a “receiver’’
   that contributes least.  An annealed acceptance rule (simulated annealing)
   permits occasional uphill moves, helping to escape shallow plateaus.

All phases respect non‑negativity and the ℓ¹‑norm constraint
``∑ a_i = √(2 n)`` by projecting onto the simplex after each update.
The routine adapts the problem size (a power‑of‑two length) to the allotted
time budget (≤1000 s) and stores the best sequence in the global variable
``height_sequence_1`` for warm‑starting future runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (full linear convolution a*a)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using an FFT of size power‑of‑two."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the full convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (sum(a)²).
    For a normalized sum (∑a = √(2n)) the expression equals max(a*a).
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Phase 1 – exponential (mirror) descent on a smooth max surrogate
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Multiplicative (mirror) descent on the soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        # --- soft‑max gradient -------------------------------------------------
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_b = b.max()
        exps = np.exp(beta * (b - max_b))
        p = exps / exps.sum()

        a_rev = a[::-1]
        Ar_fft = np.fft.rfft(a_rev, n=size)
        Pr_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # --- update -----------------------------------------------------------
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * grad)
        a = a * (target_sum / a.sum())

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 8e-04, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – exact sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _subgradient_step(a: np.ndarray,
                      target_sum: float,
                      rng: np.random.Generator) -> np.ndarray:
    """One projected sub‑gradient step on the exact max‑convolution."""
    n = a.shape[0]
    b = _conv_fft(a)
    max_b = b.max()
    act = np.where(b >= max_b - 1e-12)[0]          # active (near‑max) shifts

    grad = np.zeros_like(a)

    for k in act:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=np.int64)
        j = k - i
        mask = i != j
        grad[i[mask]] += a[j[mask]]
        grad[j[mask]] += a[i[mask]]
        eq = i == j
        grad[i[eq]] += 2.0 * a[i[eq]]

    # diminishing step size proportional to 1/√t
    step = 0.4 / (np.sqrt(rng.random()) * (np.linalg.norm(grad) + 1e-12))
    a_new = a - step * grad
    a_new = _project_to_simplex(a_new, target_sum)
    return a_new


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution."""
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        a = _subgradient_step(a, target_sum, rng)
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} score={best_val:.6f}")

        # occasional jitter to leave plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


def _subgradient_descent_with_restarts(init: np.ndarray,
                                      target_sum: float,
                                      deadline: float) -> tuple[np.ndarray, float]:
    """Run several short sub‑gradient runs with random jitter and keep the best."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _subgradient_descent(start, target_sum,
                                      time.time() + run_budget)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 3 – deterministic mass‑transfer refinement (local search)
# ----------------------------------------------------------------------


def _mass_transfer_refine(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Fast hill‑climbing on the true max‑convolution.
    Moves a tiny amount of mass from a “donor’’ (large contribution to the
    current maximal entry) to a “receiver’’ (small contribution).  Acceptance
    follows a simple annealing schedule.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start_time = time.time()
    total_time = deadline - start_time
    T0 = best_val * 0.06                  # initial temperature
    it = 0
    last_full = start_time

    while time.time() < deadline:
        # --------------------------------------------------------------
        # identify a maximal shift and the best donor / receiver
        # --------------------------------------------------------------
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        prod = a[idx] * a[k_max - idx]               # contribution of each index
        donor = idx[np.argmax(prod)]
        receiver = idx[np.argmin(prod)]

        if donor == receiver or a[donor] <= 0.0:
            it += 1
            continue

        # --------------------------------------------------------------
        # propose a transfer amount (larger than before)
        # --------------------------------------------------------------
        max_delta = min(a[donor] * 0.18, target_sum * 0.0015)
        if max_delta <= 0.0:
            it += 1
            continue
        delta = rng.random() * max_delta

        # --------------------------------------------------------------
        # update convolution exactly (O(n))
        # --------------------------------------------------------------
        b_new = b.copy()

        # linear part: ±2*delta*a
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a

        # quadratic corrections
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max / (target_sum * target_sum)

        # --------------------------------------------------------------
        # annealed acceptance
        # --------------------------------------------------------------
        elapsed = time.time() - start_time
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Mass‑transfer improved score={best_val:.6f}")

        # --------------------------------------------------------------
        # periodic full recomputation (to avoid drift)
        # --------------------------------------------------------------
        if time.time() - last_full > 15.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑norm improved score={best_val:.6f}")
            last_full = time.time()

        # occasional tiny jitter (every few thousand iterations)
        if it % 4000 == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.006, target_sum * 0.0005)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)               # recompute after jitter
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # tiny safety margin

    # --------------------------------------------------------------
    # Choose a power‑of‑two problem size (FFT friendly)
    # --------------------------------------------------------------
    if budget_s >= 900:
        n = 16384
    elif budget_s >= 600:
        n = 12288
    else:
        n = 8192

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Warm‑start (reuse the best known construction if available)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"],
                                         dtype=np.float64), n)
    else:
        # fallback: decreasing power law (exponent tuned empirically)
        init = (np.arange(1, n + 1, dtype=np.float64) ** -0.58)

    init = init * (target_sum / init.sum())

    best_seq = init.copy()
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] Initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – exponential (mirror) descent (≈10 % of remaining budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        exp_deadline = time.time() + 0.10 * (deadline - time.time())
        a_exp, val_exp = _exponential_descent(best_seq, target_sum, exp_deadline)
        if val_exp < best_score:
            best_seq, best_score = a_exp, val_exp
        print(f"[{time.time():.2f}] After exponential descent best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – exact sub‑gradient descent with restarts (≈30 % budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + 0.30 * (deadline - time.time())
        a_sub, val_sub = _subgradient_descent_with_restarts(best_seq,
                                                            target_sum,
                                                            sub_deadline)
        if val_sub < best_score:
            best_seq, best_score = a_sub, val_sub
        print(f"[{time.time():.2f}] After sub‑gradient phase best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – deterministic mass‑transfer refinement (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_ref, val_ref = _mass_transfer_refine(best_seq,
                                               target_sum,
                                               deadline)
        if val_ref < best_score:
            best_seq, best_score = a_ref, val_ref
        print(f"[{time.time():.2f}] After local refinement best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final clean‑up and evaluation
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in a subsequent call
    globals()["height_sequence_1"] = final_seq

    return final_seq
