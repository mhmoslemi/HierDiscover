def smooth_gradient(a, alpha):
    b = full_convolution(a)
    max_b = b.max()
    # compute w = exp(alpha * (b - max_b))

    a_rev = a[::-1]
    # Use convolution (FFT) of w and a_rev; note w has length convolve_len = 2*n-1, not less; we need conv over w and rev(a) which yields length (len(w)+n-1). Then slice.

"""
step‑function optimizer – smoothed top‑K descent + refinement

The objective that `evaluate_sequence` finally returns (because the
normalisation is Σa = √(2 n)) is simply

    max_k (a * a)[k]                (full linear self‑convolution)

subject to a_i ≥ 0 and Σa_i = √(2 n).  

The algorithm consists of five stages:

1. **Broad random exploration** – generate many candidates from several
   parametric families (power‑law, exponential, linear, U‑shape,
   sinusoid, uniform, Dirichlet).  The best few are kept.

2. **Projected sub‑gradient descent** – a classic sub‑gradient method
   on the true maximum (mask of the current peak) with Polyak step
   sizes and projection onto the ℓ₁‑simplex.

3. **Length scaling** – the current best sequence is interpolated to a
   few larger lengths and refined again (still using sub‑gradient).

4. **Smoothed “top‑K” descent** – instead of using only the single
   maximal entry, we repeatedly minimise the average of the
   `K`‑largest convolution values.  The mask of the `K` largest entries
   provides a sub‑gradient; `K` is gradually reduced (from a few per‑
   cent of all entries down to one) which gives a smooth homotopy
   from a smoothed objective to the true objective.  This stage
   typically pushes the bound well below the previous best.

5. **Light simulated‑annealing hill‑climb** – occasional stochastic
   mass‑transfer moves help escape tiny plateaus.

All stages respect the 1000 s overall budget.  The final sequence is
stored in the global variable ``height_sequence_1`` and returned as a
plain list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT, projection, scaling)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a∗a."""
    return _full_convolution_fft(a).max()


def _project_onto_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex { w ≥ 0, Σw = S }."""
    if S < 0:
        raise ValueError("Simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Parametric families for random initialisation
# ----------------------------------------------------------------------
def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    raw = mix * ush + (1.0 - mix) * base
    return raw


def _init_sine(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    vals = np.sin(np.pi * (i + 1) / (n + 1))
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * vals + (1.0 - mix) * base
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    return rng.random(n)


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """Create a feasible start vector of length n."""
    S = np.sqrt(2.0 * n)

    # warm‑start with modest probability
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total > 0.0:
            a = a * (S / total)
            return a

    # draw a parametric family
    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'ushape', 'sine', 'uniform', 'dirichlet'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'sine':
        raw = _init_sine(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Sub‑gradient machinery (hard max)
# ----------------------------------------------------------------------
def _max_convolution_and_gradient(a: np.ndarray):
    """Sub‑gradient for the true max (mask of maximal entries)."""
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_val = b.max()
    mask = (b == max_val).astype(float)
    a_rev = a[::-1]
    conv_len = len(mask) + n - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad, max_val


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         global_best: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max objective."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_val = _max_convolution_and_gradient(a)

        norm_sq = np.dot(grad, grad)
        if norm_sq > 0:
            step = (cur_val - best_val) / (norm_sq + 1e-12)
            step = np.clip(step, 1e-7, 0.2)
        else:
            step = 0.0

        a = _project_onto_simplex(a - step * grad, S)
        cur_val = _max_convolution(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()

        if it % 300 == 0:
            jitter = rng.normal(scale=0.001 * np.mean(a), size=n)
            a = _project_onto_simplex(a + jitter, S)

    return best_a, best_val


# ----------------------------------------------------------------------
# Smoothed “top‑K” descent
# ----------------------------------------------------------------------
def _gradient_from_mask(a: np.ndarray, mask: np.ndarray) -> np.ndarray:
    """Gradient for an arbitrary (non‑negative) mask of length 2n‑1."""
    n = a.shape[0]
    a_rev = a[::-1]
    conv_len = mask.shape[0] + n - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


def _topk_descent(a0: np.ndarray,
                  deadline: float,
                  rng: np.random.Generator,
                  global_best: float,
                  init_frac: float = 0.08,
                  min_frac: float = 0.001,
                  decay: float = 0.999) -> tuple[np.ndarray, float]:
    """
    Smoothed descent that minimises the average of the K‑largest
    convolution entries.  K = frac * (2n‑1) is gradually shrunk.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _max_convolution(a)
    cur_val = best_val
    top_frac = init_frac
    it = 0

    while time.time() < deadline:
        it += 1

        # full convolution
        b = _full_convolution_fft(a)
        cur_val = b.max()

        # build mask of the K‑largest entries
        k = max(1, int(top_frac * b.size))
        idx = np.argpartition(-b, k - 1)[:k]
        mask = np.zeros_like(b)
        mask[idx] = 1.0

        # gradient for this mask
        grad = _gradient_from_mask(a, mask)

        # Polyak step size (use the global best as the target)
        norm_sq = np.dot(grad, grad)
        if norm_sq > 0:
            step = (cur_val - global_best) / (norm_sq + 1e-12)
            step = np.clip(step, 1e-7, 0.25)
        else:
            step = 0.0

        a = _project_onto_simplex(a - step * grad, S)

        # evaluate true objective
        cur_val = _max_convolution(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()

        # decay the top‑K fraction
        top_frac = max(min_frac, top_frac * decay)

        # occasional jitter to escape plateaus
        if it % 500 == 0:
            jitter = rng.normal(scale=0.0015 * S, size=n)
            a = _project_onto_simplex(a + jitter, S)
            cur_val = _max_convolution(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (as a final polishing step)
# ----------------------------------------------------------------------
def _anneal_local_search(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Simple SA that moves a random fraction of mass between two positions."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    cur_val = _max_convolution(a)
    best_a = a.copy()
    best_val = cur_val

    # temperature schedule – start at ~5 % of current peak
    T = 0.05 * cur_val
    decay = 0.9999

    it = 0
    while time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        frac = rng.uniform(0.02, 0.18)
        delta = a[i] * frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _max_convolution(a_new)
        delta_val = new_val - cur_val
        if delta_val < 0 or rng.random() < np.exp(-delta_val / (T + 1e-12)):
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()

        T *= decay
        if it % 500 == 0:
            jitter = rng.normal(scale=0.001 * S, size=n)
            a = _project_onto_simplex(a + jitter, S)
            cur_val = _max_convolution(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step function that minimises the provided
    ``evaluate_sequence``.  The routine follows the five‑stage plan
    described in the module docstring and stores the final construction
    in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Warm‑start handling
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    best_seq = None
    best_score = np.inf
    best_n = None
    attempts = 0

    # ------------------------------------------------------------------
    # Phase 1 – broad random exploration (≈ 55 % of budget)
    # ------------------------------------------------------------------
    explore_deadline = end_time - 350.0   # reserve the rest for refinement
    while time.time() < explore_deadline:
        attempts += 1
        # favour longer sequences (3 000 – 15 000)
        if rng.random() < 0.78:
            n = int(rng.integers(3000, 15000))
        else:
            n = int(rng.integers(15000, 25000))

        a0 = _initial_vector(n, rng, warm_seq)
        val = _max_convolution(a0)
        if val < best_score:
            best_score = val
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempts:4d} n={n:5d}  new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – sub‑gradient refinement on a few scaled lengths
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 4000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = _max_convolution(best_seq)
        best_n = n

    # Choose a modest set of target lengths (geometric growth)
    lengths = [best_n]
    for factor in [1.3, 1.8, 2.5, 3.5]:
        cand = int(best_n * factor)
        if 3000 <= cand <= 25000:
            lengths.append(cand)
    lengths = sorted(set(lengths))

    for target_n in lengths:
        if time.time() >= end_time - 120.0:
            break
        a_scaled = _scale_sequence(best_seq, target_n)
        S = np.sqrt(2.0 * target_n)
        a_scaled = np.clip(a_scaled, 0.0, None)
        a_scaled = a_scaled * (S / a_scaled.sum())

        remaining = end_time - time.time()
        slice_time = min(remaining * 0.15, 80.0)

        a_opt, val = _subgradient_descent(a_scaled,
                                          time.time() + slice_time,
                                          rng,
                                          best_score)
        if val < best_score:
            best_score = val
            best_seq = a_opt.copy()
            best_n = target_n
            print(f"[subgrad] target_n={target_n:5d}  improved={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – long sub‑gradient run on current best length
    # ------------------------------------------------------------------
    if time.time() < end_time - 70.0:
        remaining = end_time - time.time() - 5.0
        a_opt, val = _subgradient_descent(best_seq,
                                          time.time() + remaining,
                                          rng,
                                          best_score)
        if val < best_score:
            best_score = val
            best_seq = a_opt.copy()
            print(f"[final subgrad] length={len(best_seq)}  best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – smoothed top‑K descent (the main new ingredient)
    # ------------------------------------------------------------------
    if time.time() < end_time - 30.0:
        remaining = end_time - time.time() - 5.0
        # allocate ~60 % of the remaining time to top‑K, rest to anneal
        topk_time = remaining * 0.60
        anneal_time = remaining - topk_time

        a_opt, val = _topk_descent(best_seq,
                                   time.time() + topk_time,
                                   rng,
                                   best_score,
                                   init_frac=0.09,
                                   min_frac=0.001,
                                   decay=0.999)
        if val < best_score:
            best_score = val
            best_seq = a_opt.copy()
            print(f"[topk descent] length={len(best_seq)}  best={best_score:.6f}")

        # ------------------------------------------------------------------
        # Phase 5 – simulated‑annealing hill‑climb (final polishing)
        # ------------------------------------------------------------------
        if anneal_time > 5.0:
            a_opt, val = _anneal_local_search(best_seq,
                                              time.time() + anneal_time,
                                              rng)
            if val < best_score:
                best_score = val
                best_seq = a_opt.copy()
                print(f"[anneal] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the external evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f}  internal={best_score:.6f}  length={len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
