def solve_lp_convolution(f, M, target):
    n = f.size
    # Build A
    rows = 2*n - 1
    # Use scipy.sparse.lil_matrix for incremental assignments, then convert to csr
    from scipy.sparse import lil_matrix, csr_matrix
    A = lil_matrix((rows, n))
    for i in range(n):
        A[i:i+n, i] = f
    A = csr_matrix(A)
    c = -np.ones(n)  # maximize sum = minimize -sum

    # bounds
    bounds = [(0, None) for _ in range(n)]

    # RHS b for inequality
    b_rhs = np.full(rows, M)

    # Use linprog (SciPy 1.9 supports sparse)
    result = linprog(c, A_ub=A, b_ub=b_rhs, bounds=bounds, method='highs')
    if result.success:
        g = result.x
        # scale to target sum
        sum_g = np.sum(g)
        if sum_g > 0:
            g = g * (target / sum_g)
        else:
            # fallback uniform
            g = np.full(n, target / n)
        return g
    else:
        return None

"""
Propose a step‑function (non‑negative sequence) that minimises
``evaluate_sequence``.
This implementation augments the previous Adam + hill‑climb pipeline with
a linear‑programming (LP) refinement step based on the classic alternating
procedure described in the statement:

    * normalise the current candidate f so that Σf = √(2 n),
    * let M = max(conv(f,f)),
    * solve   max Σg   s.t.   g≥0  and  conv(f,g) ≤ M (component‑wise),
    * rescale g to the same ℓ¹‑norm and replace   f ← (1‑t)·f + t·g.

The LP is solved with ``scipy.optimize.linprog`` (HiGHS).  After a few
alternations we up‑sample the vector, perform a short Adam optimisation,
and finish with a deterministic hill‑climb.  The routine respects the
global wall‑clock budget (default 1000 s) and always returns the best
sequence found.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# optional but highly useful for the LP step
try:
    from scipy.optimize import linprog
    from scipy.sparse import lil_matrix, csr_matrix
    _SCIPY_AVAILABLE = True
except Exception:          # pragma: no cover
    _SCIPY_AVAILABLE = False


# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = rfft(a, n=size)
    conv = irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using an FFT (fast, O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = rfft(a, n=size)
    conv = irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())

# Adam optimiser (smooth‑max surrogate)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]
    size1 = _next_pow2(2 * n - 1)
    a_fft = rfft(a, n=size1)
    b = irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = rfft(a_rev, n=size2)
    Pr_fft = rfft(p, n=size2)
    conv_rev = irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float,
                   deadline: float) -> np.ndarray:
    """Projected Adam minimising the smooth‑max surrogate of max(conv)."""
    a = a0.copy()
    n = a.size

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.15 * target_sum
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        # tiny jitter to escape plateaus
        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4 / (beta + 1.0),
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Random initialiser
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)

    family = rng.integers(0, 5)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (near‑uniform)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    else:                               # very flat with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Deterministic hill‑climb (exact max)
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that always attacks the current worst convolution entry.
    Moves a fraction of the responsible mass to a position that reduces that entry.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()
    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))

        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]

        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        # best destination (minimal counterpart)
        j_low = i_low
        j_high = i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart = a[M - j_range]
        order = np.argsort(counterpart)

        j = None
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        delta = a[i] * 0.8
        if delta <= 0.0:
            break

        # linear update of convolution (O(n))
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()
        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # try a smaller move (binary search)
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    break
                delta2 *= 0.5
            else:
                # small jitter to leave flat regions
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                target = math.sqrt(2.0 * n)
                a = _project_to_simplex(a, target)
                b = _full_convolution(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# LP refinement step
# ----------------------------------------------------------------------


def _lp_refine(f: np.ndarray, M: float, target_sum: float) -> np.ndarray | None:
    """
    Solve the linear programme:
        maximise   Σ g_j
        subject to   g_j ≥ 0
                     (f * g)_k ≤ M    for all k
    Returns the rescaled optimiser g (ℓ¹‑norm = target_sum) or None on failure.
    """
    if not _SCIPY_AVAILABLE:               # pragma: no cover
        return None

    n = f.size
    rows = 2 * n - 1

    # Build the Toeplitz‑like convolution matrix A where A @ g = f * g
    A = lil_matrix((rows, n), dtype=float)
    for i in range(n):
        A[i:i + n, i] = f
    A = csr_matrix(A)

    # Objective: maximise Σ g → minimise -Σ g
    c = -np.ones(n, dtype=float)

    # Upper bounds for each convolution entry
    b_rhs = np.full(rows, M, dtype=float)

    # Variable bounds: g ≥ 0
    bounds = [(0, None) for _ in range(n)]

    # Solve – HiGHS works with sparse matrices
    try:
        res = linprog(c, A_ub=A, b_ub=b_rhs, bounds=bounds,
                      method="highs", options={"presolve": True, "time_limit": 5.0})
    except Exception:                     # pragma: no cover
        return None

    if not res.success:
        return None

    g = res.x
    s = np.sum(g)
    if s <= 0:
        return None
    g = g * (target_sum / s)
    return g

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence`` under the 1000 s wall‑clock budget.
    The procedure interleaves:
        • a few LP refinements,
        • Adam optimisation on a smooth‑max surrogate,
        • deterministic hill‑climbing,
        • progressive up‑sampling to larger dimensions.
    The best sequence encountered is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 1)  Seed from previously discovered construction, if present
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            n0 = prev.size
            target0 = math.sqrt(2.0 * n0)
            prev = _project_to_simplex(prev, target0)
            best_seq = prev.copy()
            best_score = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 "
                  f"(n={n0}) score={best_score:.6f}")
        except Exception as e:
            print(f"Seeding error: {e}")

    # ------------------------------------------------------------------
    # 2)  Length schedule – start modest, finish with the target size
    # ------------------------------------------------------------------
    schedule = [500, 1000, 2000]       # keep total runtime modest

    for n in schedule:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Initialise a candidate for this length
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a_cur = _upsample_sequence(best_seq, n)
        else:
            a_cur = _random_initial_sequence(n, rng)

        a_cur = _project_to_simplex(a_cur, target_sum)

        # --------------------------------------------------------------
        # Allocate a time slice for this dimension
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 10.0:
            break

        # time budgeting (roughly)
        lp_budget = remaining * 0.20
        adam_budget = remaining * 0.55
        hc_budget = remaining * 0.20
        extra_budget = remaining * 0.05

        # --------------------------------------------------------------
        # Phase LP – a few alternating refinement steps
        # --------------------------------------------------------------
        lp_deadline = min(deadline - 0.2, time.time() + lp_budget)
        while time.time() < lp_deadline and _SCIPY_AVAILABLE:
            # current bound
            conv_a = _full_convolution(a_cur)
            M = conv_a.max()
            g = _lp_refine(a_cur, M, target_sum)
            if g is None:
                break
            # convex combination (t = 0.1)
            t_mix = 0.1
            a_cur = (1.0 - t_mix) * a_cur + t_mix * g
            a_cur = _project_to_simplex(a_cur, target_sum)

        # --------------------------------------------------------------
        # Phase A – projected Adam on the smooth‑max surrogate
        # --------------------------------------------------------------
        adam_deadline = min(deadline - 0.2, time.time() + adam_budget)
        a_cur = _adam_optimize(a_cur, target_sum, adam_deadline)

        # --------------------------------------------------------------
        # Phase B – deterministic hill‑climbing (exact max)
        # --------------------------------------------------------------
        if time.time() < deadline - 2.0:
            b_cur = _full_convolution(a_cur)
            hc_deadline = min(deadline - 0.2, time.time() + hc_budget)
            a_cur, score_cur = _hill_climb_deterministic(a_cur, b_cur, hc_deadline)
        else:
            score_cur = _max_convolution_fft(a_cur)

        # --------------------------------------------------------------
        # Record improvement
        # --------------------------------------------------------------
        if score_cur < best_score - 1e-12:
            best_score = score_cur
            best_seq = a_cur.copy()
            print(f"[{time.time():.1f}] Improvement at n={n} "
                  f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # Phase C – tiny extra Adam run if time permits
        # --------------------------------------------------------------
        if time.time() < deadline - 1.0 and extra_budget > 2.0:
            extra_deadline = deadline - 0.5
            a_tmp = _adam_optimize(best_seq, target_sum, extra_deadline)
            tmp_score = _max_convolution_fft(a_tmp)
            if tmp_score < best_score - 1e-12:
                best_seq = a_tmp.copy()
                best_score = tmp_score
                print(f"[{time.time():.1f}] Post‑LP Adam improvement for n={n} "
                      f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Final polishing (remaining budget)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        n = best_seq.size
        target = math.sqrt(2.0 * n)

        final_deadline = deadline - 1.0
        best_seq = _adam_optimize(best_seq, target, final_deadline)

        b_final = _full_convolution(best_seq)
        best_seq, final_score = _hill_climb_deterministic(
            best_seq, b_final, deadline - 0.5)

        if final_score < best_score - 1e-12:
            best_score = final_score
            print(f"[{time.time():.1f}] Final polishing improvement: "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Trim trailing near‑zeros (purely cosmetic)
    # ------------------------------------------------------------------
    if best_seq is None:                      # very unlikely fallback
        n = 500
        best_seq = np.full(n, math.sqrt(2.0 * n) / n)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # 5) Report official evaluation
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
