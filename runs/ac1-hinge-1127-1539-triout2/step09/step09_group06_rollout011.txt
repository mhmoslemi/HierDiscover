def _lp_refine(a, deadline, rng):
    n = len(a)
    S = np.sqrt(2.0 * n)
    a_curr = a.copy()
    best = a_curr.copy()
    best_M = _max_convolution(a_curr)
    it = 0
    while time.time() < deadline:
        it += 1
        M = _max_convolution(a_curr)
        # build Toeplitz matrix A
        rows = np.empty(n * n, dtype=int)
        cols = np.empty(n * n, dtype=int)
        data = np.empty(n * n, dtype=float)
        idx = 0
        for i in range(n):
            rows[idx:idx+n] = i + np.arange(n)
            cols[idx:idx+n] = np.arange(n)
            data[idx:idx+n] = a_curr[i]
            idx += n
        A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
        # solve LP
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            print("[lp] linprog failed", res.message)
            break
        g0 = res.x
        sum_g = g0.sum()
        if sum_g <= 0:
            break
        g = g0 * (S / sum_g)
        # mixture
        t = 0.2
        cand = (1 - t) * a_curr + t * g
        cand = np.maximum(cand, 0.0)
        cand = cand * (S / cand.sum())
        cand_M = _max_convolution(cand)
        if cand_M < best_M:
            best_M = cand_M
            best = cand.copy()
            a_curr = cand
            print(f"[lp] iter={it} improved max={best_M:.6f}")

                if it >= 10:
                    break
    return best, best_M

"""
Improved step‑function optimizer.

The evaluator is scale‑invariant after normalising the ℓ1‑norm to
√(2·n); the score equals the peak of the self‑convolution.
The algorithm proceeds in the following stages:

1.  Warm‑start from a previously stored sequence (if any).
2.  Fast random exploration of many parametric families for a wide range
    of lengths, keeping the best raw peak.
3.  Interpolation to several larger lengths (no optimisation) – the best
    length is retained.
4.  Continuous refinement with two complementary methods
        – Adam on a τ‑soft‑max surrogate of the peak,
        – a linear‑programming based “mix‑with‑g” step that is known to
          strictly decrease the peak (the LP step solves
          max ∑g s.t. f∗g ≤ M, g≥0).
5.  A final cheap hill‑climb that swaps a small amount of mass between
    two positions.
The whole routine respects the 1 000 s budget and prints progress.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Basic convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a∗a."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Parametric families for random initialisation
# ----------------------------------------------------------------------
def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    raw = mix * ush + (1.0 - mix) * base
    return raw


def _init_cosine(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    raw = 1.0 - np.cos(np.pi * i / (n - 1))
    raw = raw * rng.uniform(0.5, 2.0)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * raw + (1.0 - mix) * base


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    return rng.random(n)


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.05, 0.9)
    return rng.gamma(shape, 1.0, size=n)


def _initial_vector(n: int, rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    With probability ≈0.2 we warm‑start from ``warm_seq`` (scaled).
    Otherwise we draw from one of the random families.
    The returned vector satisfies sum = √(2·n) and all entries ≥0.
    """
    S = np.sqrt(2.0 * n)

    if warm_seq is not None and rng.random() < 0.23:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
        return a

    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'gamma', 'uniform', 'dirichlet', 'ushape', 'cosine'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'cosine':
        raw = _init_cosine(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# τ‑soft‑max gradient (used by Adam)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser (mirror descent on the simplex)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex (log‑softmax parametrisation).
    Returns the best sequence found and its hard peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                     # ℓ1‑norm we must keep

    # initialise logits from the feasible a0 (entropic parametrisation)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                         # logits (log‑softmax)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.45                                # aggressive start

    # temperature schedule – start relatively large for smoothness
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 5.0)                 # a few units of smoothing

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ---- recover a from logits (log‑softmax) ----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ---- convolution & gradient (τ‑soft‑max) ----
        g, peak = _softmax_gradient(a, tau)

        # keep the best hard peak seen so far
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ---- gradient w.r.t. logits (chain rule) ----
        mu = np.dot(a, g) / S                 # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)                 # shape (n,)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 6.0 * S:
            grad_p = grad_p * (6.0 * S / clip_val)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- schedules (very slow decay) ----
        tau *= 0.99995                         # keep smoothing longer
        lr *= 0.99995                          # gentle learning‑rate decay

        # occasional jitter to escape plateaus
        if it % 800 == 0:
            p += rng.normal(scale=0.006, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Linear‑programming refinement (LP‑mix step)
# ----------------------------------------------------------------------
def _lp_refine(a: np.ndarray,
               deadline: float,
               rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve the LP "max ∑g  s.t.  f∗g ≤ M, g≥0" and blend the
    solution g back into f.  Each successful blend reduces the hard peak.
    Returns the refined sequence and its peak.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)

    a_curr = a.copy()
    best_seq = a_curr.copy()
    best_M = _max_convolution(a_curr)
    it = 0

    # pre‑allocate the index vectors that define the Toeplitz matrix
    rows = np.repeat(np.arange(n), n) + np.tile(np.arange(n), n)
    cols = np.tile(np.arange(n), n)

    while time.time() < deadline:
        it += 1

        M = _max_convolution(a_curr)

        # Build the (2n‑1) × n constraint matrix A where (A·g)_k = (f∗g)_k
        data = np.repeat(a_curr, n)
        A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

        # Linear programme: minimise -∑g  (i.e. maximise ∑g)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})

        if not res.success:
            # LP failed – abort refinement
            print(f"[lp] iteration {it}: linprog failed – {res.message}")
            break

        g0 = res.x
        sum_g = g0.sum()
        if sum_g <= 0.0:
            break

        # Rescale to the required ℓ1‑norm
        g = g0 * (S / sum_g)

        # Try several mixing coefficients and keep the best result
        t_candidates = [0.05, 0.1, 0.2, 0.3, 0.5]
        best_local_seq = a_curr
        best_local_M = M
        for t in t_candidates:
            cand = (1.0 - t) * a_curr + t * g
            cand = np.maximum(cand, 0.0)
            cand = cand * (S / cand.sum())
            cand_M = _max_convolution(cand)
            if cand_M < best_local_M - 1e-12:
                best_local_M = cand_M
                best_local_seq = cand.copy()

        # Accept if we actually improved the peak
        if best_local_M < best_M - 1e-12:
            best_M = best_local_M
            best_seq = best_local_seq.copy()
            a_curr = best_local_seq.copy()
            print(f"[lp] iter={it} improved max={best_M:.6f}")
        else:
            # no improvement – stop early after a few futile attempts
            if it >= 15:
                break

    return best_seq, best_M


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence found as a plain
    Python list and stores it in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # tiny safety margin

    # --------------------------------------------------------------
    # Warm‑start handling
    # --------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # --------------------------------------------------------------
    # Phase 1 – fast random exploration
    # --------------------------------------------------------------
    explore_deadline = end_time - 350.0   # reserve ~350 s for refinements
    best_seq = None
    best_score = np.inf
    attempts = 0

    while time.time() < explore_deadline:
        attempts += 1
        # favour a few‑thousand entries but allow some variety
        if rng.random() < 0.70:
            n = int(rng.integers(1800, 6000))
        else:
            n = int(rng.integers(6000, 12000))

        a0 = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a0)

        if peak < best_score:
            best_score = peak
            best_seq = a0.copy()
            print(f"[explore] attempt={attempts:4d} n={n:5d}  new best={best_score:.6f}")

    # fallback – should never happen
    if best_seq is None:
        n0 = 3000
        best_seq = np.full(n0, np.sqrt(2.0 * n0) / n0)
        best_score = _max_convolution(best_seq)

    # --------------------------------------------------------------
    # Phase 2 – upscale and evaluate a few larger lengths (no optimisation)
    # --------------------------------------------------------------
    cand_lengths = [len(best_seq)]
    for factor in [1.2, 1.5, 1.8, 2.0, 2.5, 3.0, 4.0, 5.0]:
        L = int(len(best_seq) * factor)
        if 2000 <= L <= 25000:
            cand_lengths.append(L)
    cand_lengths = sorted(set(cand_lengths))

    cur_seq = best_seq
    cur_score = best_score
    cur_n = len(best_seq)

    for L in cand_lengths:
        if time.time() > end_time - 250.0:
            break
        a_scaled = _scale_sequence(cur_seq, L)
        a_scaled = np.clip(a_scaled, 0.0, None)
        S = np.sqrt(2.0 * L)
        a_scaled = a_scaled * (S / a_scaled.sum())
        peak = _max_convolution(a_scaled)

        if peak < cur_score:
            cur_score = peak
            cur_seq = a_scaled.copy()
            cur_n = L
            print(f"[scale] length={L:5d}  improved max={cur_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Adam optimisation (continuous refinement)
    # --------------------------------------------------------------
    remaining = end_time - time.time() - 150.0   # keep ~150 s for LP + hill
    if remaining > 20.0:
        n_restart = max(1, int(remaining // 70))   # roughly 70 s per restart
        per_run = remaining / n_restart
        for r in range(n_restart):
            if time.time() > end_time - 150.0:
                break
            noise = rng.normal(scale=0.02 *
                               np.sqrt(2.0 * cur_n) / cur_n, size=cur_n)
            a_start = np.maximum(cur_seq + noise, 0.0)
            a_start = a_start * (np.sqrt(2.0 * cur_n) / a_start.sum())
            a_opt, peak = _adam_optimize(a_start,
                                          time.time() + per_run,
                                          rng)
            if peak < cur_score:
                cur_score = peak
                cur_seq = a_opt.copy()
                print(f"[adam] restart={r} new best max={cur_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – LP‑based mixing refinement
    # --------------------------------------------------------------
    remaining = end_time - time.time() - 40.0    # leave ~40 s for final hill
    if remaining > 30.0:
        # If the current length is large, shrink to a manageable size
        if cur_n > 2500:
            small_n = 2500
            a_small = _scale_sequence(cur_seq, small_n)
            a_small = a_small * (np.sqrt(2.0 * small_n) / a_small.sum())
        else:
            small_n = cur_n
            a_small = cur_seq.copy()

        a_refined, refined_peak = _lp_refine(a_small,
                                            time.time() + remaining,
                                            rng)

        # Upscale back to the original length (if we down‑scaled)
        if small_n != cur_n:
            a_refined = _scale_sequence(a_refined, cur_n)
            a_refined = a_refined * (np.sqrt(2.0 * cur_n) / a_refined.sum())
            refined_peak = _max_convolution(a_refined)

        if refined_peak < cur_score:
            cur_score = refined_peak
            cur_seq = a_refined.copy()
            print(f"[lp] refined max={cur_score:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – cheap random hill‑climb (few seconds)
    # --------------------------------------------------------------
    while time.time() < end_time - 1.0:
        a = cur_seq.copy()
        n = a.shape[0]
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        frac = rng.uniform(0.02, 0.15)
        amount = a[i] * frac
        a[i] -= amount
        a[j] += amount
        a = a * (np.sqrt(2.0 * n) / a.sum())
        peak = _max_convolution(a)
        if peak + 1e-9 < cur_score:
            cur_score = peak
            cur_seq = a.copy()
            print(f"[hill] improved max={cur_score:.6f}")

    # --------------------------------------------------------------
    # Final sanity check via the provided evaluator
    # --------------------------------------------------------------
    final_seq = cur_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f}  internal={cur_score:.6f}  length={len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
