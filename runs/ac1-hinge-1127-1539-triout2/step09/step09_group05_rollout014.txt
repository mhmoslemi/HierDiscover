def _smooth_descent(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_val = _max_convolution(a)
    # momentum vector
    v = np.zeros_like(a)
    beta = 0.9
    eps_schedule = [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    for eps in eps_schedule:
        # set number of internal iterations based on epsilon
        if eps >= 0.1:
            max_iter = 300
        elif eps >= 0.02:
            max_iter = 800
        else:
            max_iter = 1500
        it = 0
        while time.time() < deadline and it < max_iter:
            it += 1
            # compute convolution
            b = _full_convolution_fft(a)
            maxb = b.max()
            # softmax weights
            shifted = b - maxb
            w = np.exp(shifted / eps)
            w_sum = w.sum()
            w_norm = w / w_sum
            # compute gradient: 2 * correlation of a and w_norm
            f_len = _next_pow2(len(w_norm) + n - 1)
            Fa = np.fft.rfft(a, n=f_len)
            Fw = np.fft.rfft(w_norm, n=f_len)
            corr = np.fft.irfft(Fa * np.conj(Fw), n=f_len)[:n]
            grad = 2.0 * corr
            # step size: maybe 0.1 * S / (norm+eps)
            norm = np.linalg.norm(grad)
            # ensure non-zero
            step = 0.1 * S / (norm + 1e-12)
            # Nesterov momentum update
            v = beta * v - step * grad
            a_new = a + v
            a_new = _project_onto_simplex(a_new, S)
            a = a_new
            cur_val = _max_convolution(a)
            if cur_val < best_val:

                best_a = a.copy()
            # occasional jitter
            if it % 200 == 0:
                jitter = np.random.normal(scale=0.0005 * S, size=n)
                a = _project_onto_simplex(a + jitter, S)
        # optional: after each epsilon reduce momentum to avoid large moves?
        v = beta * v  # maybe reduce
    return best_a, best_val

def _smooth_descent(a0, deadline, rng):
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_val = _max_convolution(a)
    best_f_eps = None
    eps_schedule = [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    for eps in eps_schedule:
        # decide max iterations for this epsilon
        max_iters = 2000 if eps <= 0.02 else 800
        for it in range(max_iters):
            if time.time() >= deadline:
                return best_a, best_val
            # convolution
            b = _full_convolution_fft(a)
            maxb = b.max()
            # f_eps and weights
            shifted = b - maxb
            w = np.exp(shifted / eps)
            w_sum = w.sum()
            w_norm = w / w_sum
            f_eps = maxb + eps * np.log(w_sum)
            if best_f_eps is None or f_eps < best_f_eps:
                best_f_eps = f_eps
            # gradient via correlation
            f_len = _next_pow2(len(w_norm) + n - 1)
            Fa = np.fft.rfft(a, n=f_len)
            Fw = np.fft.rfft(w_norm, n=f_len)
            corr = np.fft.irfft(Fa * np.conj(Fw), n=f_len)[:n]
            grad = 2.0 * corr
            # step size
            norm_sq = np.dot(grad, grad)
            if norm_sq > 0:
                step = (f_eps - best_f_eps) / (norm_sq + 1e-12)
                # Ensure positive step (if f_eps <= best_f_eps, step <=0)
                # we want non-negative step; if negative, set step small
                if step <= 0:
                    step = 0.01 * S / (np.sqrt(norm_sq) + 1e-12)

            else:
                step = 0.0
            a = _project_onto_simplex(a - step * grad, S)
            cur_max = _max_convolution(a)
            if cur_max < best_val:
                best_val = cur_max
                best_a = a.copy()

            if it % 250 == 0:
                jitter = rng.normal(scale=0.0005 * S, size=n)

"""
Optimiser for the step‑function max‑convolution problem.

The algorithm works in two stages:

1. **Smoothed gradient descent** – minimise a log‑sum‑exp
   approximation of the hard max ( maxₖ (a∗a)[k] ).  The gradient is
   obtained analytically by a correlation of the current vector with the
   soft‑max weights and the iterate is projected onto the ℓ₁‑simplex
   {a≥0, ∑a=√(2 n)}.  The smoothing parameter ε is decreased through a
   schedule (0.5 → 0.001) and a Polyak step size is used.

2. **Hard‑max sub‑gradient refinement** – after the smooth phase a few
   iterations of projected sub‑gradient descent on the true max
   objective are performed to tighten the bound.

Several random restarts with different lengths (3 000–25 000) and
initialisations (warm‑starting from the previous best, Dirichlet,
interpolated) are executed until the time budget (≈ 1000 s) is exhausted.
The best sequence seen is stored in the global variable
``height_sequence_1`` and returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT, convolution, projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a∗a."""
    return _full_convolution_fft(a).max()


def _project_onto_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex { w | w_i ≥ 0 , ∑ w_i = S }.
    Implements the O(n) algorithm of Duchi et al.
    """
    if S < 0:
        raise ValueError("Simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_and_gradient(a: np.ndarray):
    """
    Returns (grad, max_val) where grad is a sub‑gradient of
    maxₖ(a∗a)[k] w.r.t. a and max_val is that maximum.
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_val = b.max()
    mask = (b == max_val).astype(float)

    a_rev = a[::-1]
    conv_len = len(mask) + n - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad, max_val


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) gradient descent
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray, deadline: float, rng: np.random.Generator):
    """
    Projected gradient descent on the smooth approximation
        f_eps(a) = max(b) + eps * log( sum_k exp((b_k-max(b))/eps) )
    where b = a∗a.  The schedule of eps decreases to approach the true max.
    Returns the best vector seen and its hard max value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _max_convolution(a)
    best_f_eps = None

    eps_schedule = [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]

    for eps in eps_schedule:
        # iteration budget per epsilon (more for smaller eps)
        max_iters = 300 if eps > 0.05 else 800 if eps > 0.01 else 1500
        for it in range(max_iters):
            if time.time() >= deadline:
                return best_a, best_val

            # convolution and soft‑max weights
            b = _full_convolution_fft(a)
            maxb = b.max()
            shifted = b - maxb                     # ≤ 0
            w = np.exp(shifted / eps)              # in (0,1]
            w_sum = w.sum()
            w_norm = w / w_sum
            # smooth objective (not really needed for step size but convenient)
            f_eps = maxb + eps * np.log(w_sum)
            if best_f_eps is None or f_eps < best_f_eps:
                best_f_eps = f_eps

            # gradient = 2 * (a ★ w_norm)   (linear cross‑correlation)
            f_len = _next_pow2(len(w_norm) + n - 1)
            Fa = np.fft.rfft(a, n=f_len)
            Fw = np.fft.rfft(w_norm, n=f_len)
            corr = np.fft.irfft(Fa * np.conj(Fw), n=f_len)[:n]
            grad = 2.0 * corr

            # Polyak step size (ensure positivity)
            norm_sq = np.dot(grad, grad)
            if norm_sq > 0:
                step = (f_eps - best_f_eps) / (norm_sq + 1e-12)
                if step <= 0:
                    step = 0.01 * S / (np.sqrt(norm_sq) + 1e-12)
                step = np.clip(step, 1e-7, 0.3)
            else:
                step = 0.0

            # projected gradient step
            a = _project_onto_simplex(a - step * grad, S)

            # evaluate hard max
            cur_val = _max_convolution(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()

            # occasional jitter to escape plateaus
            if it % 250 == 0:
                jitter = rng.normal(scale=0.0005 * S, size=n)
                a = _project_onto_simplex(a + jitter, S)

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient refinement on the true (hard) max
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         global_best: float):
    """
    Projected sub‑gradient descent on f(a)=max(a∗a).
    Polyak step size is used and occasional jitter is added.
    Returns the best vector seen and its hard max value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_val = _max_convolution_and_gradient(a)

        norm_sq = np.dot(grad, grad)
        if norm_sq > 0:
            step = (cur_val - best_val) / (norm_sq + 1e-12)
            step = np.clip(step, 1e-7, 0.2)
        else:
            step = 0.0

        a = _project_onto_simplex(a - step * grad, S)

        cur_val = _max_convolution(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        if it % 300 == 0:
            jitter = rng.normal(scale=0.001 * S, size=n)
            a = _project_onto_simplex(a + jitter, S)

    return best_a, best_val
