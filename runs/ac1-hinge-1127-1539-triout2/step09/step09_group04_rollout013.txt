def _subgradient_descent(a0, target_sum, deadline):
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    best_val = _score(a)
    best = a.copy()
    it = 0
    while time.time() < deadline:
        # compute convolution
        fa = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
        k_max = int(conv.argmax())
        # compute gradient
        grad = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        grad[idx] = 2.0 * a[partner]
        norm = np.linalg.norm(grad)
        if norm == 0:
            break
        # step size schedule
        step = (target_sum * 0.5) / (norm * np.sqrt(it + 1))
        # optionally clip step
        a = a - step * grad
        # project onto simplex (nonneg + sum=target)
        a = _project_to_simplex(a, target_sum)
        # evaluate
        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it} score={best_val:.6f}")
        it += 1
    return best, best_val

"""
Search for a non‑negative step sequence that minimizes
    2·n·max(a∗a) / (sum(a)²)

Algorithm
---------
1.  Choose n according to the time budget (≈12000 for the full budget).
2.  Warm‑start from a power‑law or the previously stored best sequence.
3.  A very short parametric grid‑search on a reduced size refines the warm‑start.
4.  Long projected sub‑gradient descent on the exact maximal convolution entry.
5.  Deterministic mass‑transfer moves that directly lower that maximal entry.
6.  Stochastic hill‑climbing (random mass transfers) to escape any remaining
    local minima.
7.  Trim trailing zeros, store the sequence globally and return it.

All steps keep the ℓ¹‑norm fixed (Σa = √(2n)) by Euclidean projection onto the
simplex; the final list can be fed straight to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution / scoring utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact objective value (no rounding)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) shape, normalised to the required ℓ¹ norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Stage 1 – cheap parametric search on a reduced scale
# ----------------------------------------------------------------------


def _grid_param_search(small_n: int,
                       target_sum: float,
                       deadline: float) -> tuple[np.ndarray | None, float]:
    """Explore a two‑parameter family a_i = (i+1)^(-p)·exp(-λ·i/small_n)."""
    p_vals = np.linspace(0.2, 1.2, 17)
    lam_vals = np.linspace(0.0, 6.0, 13)

    best_seq = None
    best_score = np.inf
    i = np.arange(small_n, dtype=np.float64)

    for p in p_vals:
        for lam in lam_vals:
            if time.time() > deadline:
                break
            seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
            seq = seq * (target_sum / seq.sum())
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()
                print(f"[{time.time():.2f}] Grid best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")
        if time.time() > deadline:
            break

    rng = np.random.default_rng()
    while time.time() < deadline:
        p = rng.uniform(0.2, 1.2)
        lam = rng.exponential(1.0) * 3.0
        seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
        seq = seq * (target_sum / seq.sum())
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Random best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stage 2 – projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Long projected sub‑gradient descent on max(a∗a)."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    best_a = a.copy()
    best_val = _score(a)

    it = 0
    while time.time() < deadline:
        # full convolution via FFT
        fa = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]

        k_max = int(conv.argmax())

        # gradient of the active (max) entry: 2·a[k_max‑i] on the feasible indices
        grad = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        grad[idx] = 2.0 * a[partner]

        norm = np.linalg.norm(grad)
        if norm == 0.0:
            break

        # diminishing step size; the factor 0.5 is empirical
        step = (target_sum * 0.5) / (norm * np.sqrt(it + 1.0))
        a = a - step * grad
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 3 – deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _mass_transfer_refine(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Greedy transfer of a small fraction of mass from a high‑contributor to
    a low‑contributor (repeated until time runs out)."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    while time.time() < deadline:
        k_max = int(np.argmax(conv))

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        weight = a[idx] * a[partner]

        donor = idx[np.argmax(weight)]

        # pick a low‑contributing index (10‑percentile)
        low_thr = np.percentile(weight, 10)
        low_candidates = idx[weight <= low_thr]
        if low_candidates.size:
            receiver = low_candidates[np.argmin(weight[weight <= low_thr])]
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            # fallback to a random move
            donor = rng.choice(idx)
            receiver = rng.integers(n)
            if donor == receiver:
                continue

        max_frac = 0.04
        delta = a[donor] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta

        conv = _conv_fft(a)
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Transfer improve score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 4 – simple stochastic hill‑climbing (random mass transfers)
# ----------------------------------------------------------------------


def _hill_climbing_random(a0: np.ndarray,
                          target_sum: float,
                          deadline: float) -> tuple[np.ndarray, float]:
    """Randomised mass‑transfer moves; accept downhill moves and
    occasionally uphill moves (simulated‑annealing style)."""
    a = a0.copy()
    best_a = a.copy()
    best_val = _score(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    max_frac = 0.04

    start = time.time()
    total = deadline - start

    while time.time() < deadline:
        donor = rng.integers(n)
        if a[donor] <= 0.0:
            continue
        receiver = rng.integers(n)
        if receiver == donor:
            continue

        delta = a[donor] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[donor] -= delta
        cand[receiver] += delta

        val = _score(cand)

        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
            print(f"[{time.time():.2f}] HillClimb improve score={best_val:.6f}")
        else:
            # simple annealing: accept with a tiny probability that decays
            elapsed = time.time() - start
            temp = max(1e-8, (total - elapsed) / total) * 0.02
            if rng.random() < np.exp(-(val - best_val) / temp):
                a = cand

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Choose problem size
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Warm‑start
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = _resize_sequence(np.array(globals()["height_sequence_1"],
                                      dtype=np.float64), n)
        a = a * (target_sum / a.sum())
    else:
        a = _power_law_sequence(n, 0.55, target_sum)

    # ------------------------------------------------------------------
    # Small parametric sweep (≈10 % of budget, but at most 20 s)
    # ------------------------------------------------------------------
    if deadline - time.time() > 15.0:
        small_n = max(2000, n // 4)
        small_target = np.sqrt(2.0 * small_n)
        param_deadline = time.time() + min(12.0,
                                           (deadline - time.time()) * 0.10)
        best_small, _ = _grid_param_search(small_n, small_target,
                                           param_deadline)
        if best_small is not None:
            a = _resize_sequence(best_small, n)
            a = a * (target_sum / a.sum())
            print(f"[{time.time():.2f}] Parametric warm‑start ready (n={n})")

    # ------------------------------------------------------------------
    # Stage 2 – long sub‑gradient descent (≈50 % of remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + (deadline - time.time()) * 0.50
        a, best_val = _subgradient_descent(a, target_sum, sub_deadline)
        print(f"[{time.time():.2f}] After sub‑gradient best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – deterministic mass‑transfer refinement (≈30 % of remaining)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        mt_deadline = time.time() + (deadline - time.time()) * 0.30
        a, best_val = _mass_transfer_refine(a, target_sum, mt_deadline)
        print(f"[{time.time():.2f}] After mass‑transfer best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – stochastic hill‑climbing (use whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a, best_val = _hill_climbing_random(a, target_sum, deadline)
        print(f"[{time.time():.2f}] After hill‑climb best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up and final evaluation
    # ------------------------------------------------------------------
    arr = np.array(a, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in a subsequent call
    globals()["height_sequence_1"] = final_seq

    return final_seq
