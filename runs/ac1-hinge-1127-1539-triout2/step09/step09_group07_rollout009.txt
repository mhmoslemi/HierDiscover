Propose a non‑negative step‑function minimizing ``evaluate_sequence``.

Algorithm
---------
1. **Random exploration** – generate many candidates from a diverse set of
   parametric families (Dirichlet, Gamma, power‑law, exponential,
   sparse bumps, Gaussian, half‑sine).  Keep the best few.

   back with ``f`` (small mixing factor).  A few iterations quickly lower
   the peak.

4. **Final stochastic fine‑tuning** – a short simulated‑annealing walk
   followed by a greedy hill‑climber cleans up remaining tiny improvements.

All steps respect the 1000 s budget; the best construction is stored in the
global variable ``height_sequence_1`` and returned as a plain list.
"""

import time
import math
import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = rfft(a, n=f_len)
    conv = irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Peak value of a*a (full convolution)."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Exact (hard‑max) sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------
def _max_gradient(a: np.ndarray, tau: float):
    """
    Returns (grad, max_b) where ``grad = ∂ max(b) / ∂a`` and
    ``max_b = max(b)`` with ``b = a*a`` (full convolution).

    For very small ``tau`` a hard‑max is used: the gradient is the average
    of the gradients of all indices whose value is within a tiny tolerance
    of the maximum.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        eps = max_b * 1e-9 + 1e-12
        mask = (b >= max_b - eps).astype(float)
        s = mask.sum()
        mask = mask / s if s > 0 else np.ones_like(b) / b.size
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / exps.sum()

    a_rev = a[::-1]
    conv_len = mask.size + a.size - 1
    f_len = _next_pow2(conv_len)

    Fmask = rfft(mask, n=f_len)
    Fa_rev = rfft(a_rev, n=f_len)

    conv = irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    # slice to obtain length‑n gradient
    grad = 2.0 * conv[a.size - 1:a.size - 1 + a.size]
    return grad, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   lr0: float = 0.5,
                   tau_decay: float = 0.9995,
                   lr_decay: float = 0.99985,
                   jitter_every: int = 400) -> tuple[np.ndarray, float]:
    """
    Minimise the peak of the self‑convolution under the ℓ₁‑norm constraint
    (sum = sqrt(2 n)).  Returns the best vector seen and its (hard) peak.
    """
    n = a0.shape[0]
    S = math.sqrt(2.0 * n)                     # required ℓ₁‑norm

    # initialise logits from the feasible a0 (log‑softmax)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                           # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr0

    tau = max(_full_convolution_fft(a0).max(), 0.5)

    best_a = a0.copy()
    best_peak = _max_convolution(a0)
    it = 0

    while time.time() < deadline:
        it += 1

        # recover a from logits (log‑softmax)
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)               # stability shift
        soft = exp_p / exp_p.sum()
        a = soft * S

        # gradient of (soft‑)max convolution
        g, peak = _max_gradient(a, tau)

        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # chain rule: ∂L/∂p = a * (g - μ)   with μ = (a·g)/S
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # annealing schedules
        tau *= tau_decay
        lr *= lr_decay

        # occasional jitter to escape plateaus
        if jitter_every and (it % jitter_every) == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Random families for initial vectors
# ----------------------------------------------------------------------
def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.2, 5.0)
    return rng.gamma(alpha, 1.0, size=n)


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.1, 0.8)
    return rng.gamma(shape, 1.0, size=n)


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_left_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    p = rng.uniform(0.3, 3.0)
    offset = rng.uniform(0.0, 10.0)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * base + (1.0 - mix) * (idx + offset) ** (-p)


def _init_left_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    lam = rng.uniform(0.001, 0.02)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * base + (1.0 - mix) * np.exp(-lam * idx)


def _init_sparse_bumps(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = np.full(n, rng.uniform(0.0, 0.15))
    nbumps = rng.integers(1, max(2, n // 250))
    for _ in range(nbumps):
        center = rng.integers(0, n)
        width = rng.integers(3, max(5, n // 100))
        height = rng.uniform(0.8, 4.0)
        lo = max(0, center - width // 2)
        hi = min(n, lo + width)
        raw[lo:hi] = np.maximum(raw[lo:hi], height)
    return raw


def _init_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    mu = (n - 1) / 2.0
    sigma = rng.uniform(0.05, 0.30) * n
    x = np.arange(n)
    return np.exp(-0.5 * ((x - mu) / sigma) ** 2)


def _init_half_sine(n: int, rng: np.random.Generator) -> np.ndarray:
    x = np.arange(n)
    raw = np.sin(np.pi * x / (n - 1))
    raw = raw * rng.uniform(0.5, 2.0) + rng.uniform(0.0, 0.2)
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Occasionally warm‑starts from the global best; otherwise draws from a
    random parametric family.
    """
    S = math.sqrt(2.0 * n)

    # Warm start occasionally
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    style = rng.choice(['dirichlet', 'gamma', 'uniform',
                        'left_powerlaw', 'left_exponential',
                        'sparse', 'gaussian', 'half_sine'])
    if style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'left_powerlaw':
        raw = _init_left_powerlaw(n, rng)
    elif style == 'left_exponential':
        raw = _init_left_exponential(n, rng)
    elif style == 'sparse':
        raw = _init_sparse_bumps(n, rng)
    elif style == 'gaussian':
        raw = _init_gaussian(n, rng)
    else:  # half_sine
        raw = _init_half_sine(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    return raw * (S / total)


# ----------------------------------------------------------------------
# Simulated annealing / hill‑climbing fine‑tuner
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Mass‑transfer walk on the simplex with exponential acceptance."""
    n = len(seq)
    S = math.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score
    temperature = 0.15 * cur_score

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] == 0.0:
            continue
        delta = rng.uniform(0.0, 0.3) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())
        new_score = _max_convolution(new)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

        temperature *= 0.99995
        if temperature < 1e-6:
            temperature = 1e-6

    return best, best_score


def _hill_climb(seq: np.ndarray,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Very cheap greedy hill‑climber (strictly downhill moves only).
    """
    n = len(seq)
    S = math.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] < 1e-12:
            continue
        delta = rng.uniform(0.0, 0.15) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())
        new_score = _max_convolution(new)
        if new_score < cur_score:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score
    return best, best_score


# ----------------------------------------------------------------------
# Linear‑programming boost (max‑convolution active‑set)
# ----------------------------------------------------------------------
def _lp_boost(f: np.ndarray,
              max_M: float,
              rng: np.random.Generator,
              mix: float = 0.2,
              eps_rel: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    One LP‑based refinement step.

    ``f`` – current sequence, ``max_M`` – its current maximal convolution.
    The LP maximises the ℓ₁‑norm of a vector ``b`` under the constraints
    ``f * b ≤ max_M`` on the active indices (where the convolution of ``f`` with
    itself attains the maximum).  The solution ``b`` is rescaled to the required
    ℓ₁‑norm and mixed back into ``f``.
    Returns the new sequence and its new maximal convolution.
    """
    n = len(f)
    S = math.sqrt(2.0 * n)

    # active set (indices whose convolution value is within a tiny tolerance)
    conv_f = _full_convolution_fft(f)
    tol = max(1e-12, max_M * eps_rel)
    active_idx = np.where(conv_f >= max_M - tol)[0]
    if len(active_idx) == 0:
        return f, max_M

    # Build constraint matrix A (|active| × n)
    m = len(active_idx)
    A = np.zeros((m, n), dtype=float)
    for row, k in enumerate(active_idx):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = k - i_vals
        A[row, j_vals] = f[i_vals]

    # Linear program: maximise sum(b)  → minimise -sum(b)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c,
                  A_ub=A,
                  b_ub=np.full(m, max_M),
                  bounds=bounds,
                  method='highs',
                  options={'presolve': True, 'time_limit': 5.0})
    if not res.success:
        return f, max_M

    b = res.x
    if b.sum() == 0.0:
        return f, max_M

    # Rescale to the required ℓ₁‑norm
    b = b * (S / b.sum())

    # Convex combination with a small mixing factor
    new_f = (1.0 - mix) * f + mix * b
    new_M = _max_convolution(new_f)
    return new_f, new_M


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The result is stored in the global variable ``height_sequence_1`` and
    also returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    safety_margin = 5.0
    end_time = time.time() + budget_s - safety_margin
    if end_time <= time.time():
        raise RuntimeError("Budget too small after safety margin.")

    # ------------------------------------------------------------------
    # Warm‑start from a previously stored construction, if any
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # ------------------------------------------------------------------
    # Phase 1 – fast random exploration (≈15 % of budget)
    # ------------------------------------------------------------------
    explore_deadline = time.time() + 0.15 * budget_s
    top_k = 8
    top_candidates = []          # list of (score, seq)
    best_seq = None
    best_score = np.inf
    attempt = 0

    while time.time() < explore_deadline:
        attempt += 1
        # length distribution – mix of moderate and larger sizes
        if rng.random() < 0.65:
            n = int(rng.integers(2000, 8000))
        else:
            n = int(rng.integers(8000, 25000))

        a0 = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a0)

        if peak < best_score:
            best_score = peak
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempt:5d} n={n:5d}  new best={best_score:.6f}")

        # maintain top‑k list
        if len(top_candidates) < top_k:
            top_candidates.append((peak, a0.copy()))
            top_candidates.sort(key=lambda x: x[0])
        elif peak < top_candidates[-1][0]:
            top_candidates[-1] = (peak, a0.copy())
            top_candidates.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # Safety net – ensure at least one candidate exists
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 4000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n)
        best_score = _max_convolution(best_seq)
        best_n = n
        top_candidates = [(best_score, best_seq.copy())]

    # ------------------------------------------------------------------
    # Phase 2 – multi‑scale Adam refinement (≈70 % of budget)
    # ------------------------------------------------------------------
    max_len = min(100000, best_n * 15)  # allow a fairly large final length
    lengths = [best_n]
    cur_len = best_n
    while cur_len * 1.7 < max_len:
        cur_len = int(cur_len * 1.7)
        lengths.append(cur_len)
    lengths.append(max_len)
    lengths = sorted(set(lengths))

    for idx, (init_score, init_seq) in enumerate(top_candidates):
        cur_seq = init_seq.copy()
        for target_n in lengths:
            if time.time() >= end_time - 20.0:
                break
            # interpolate to new length and renormalise
            a_scaled = _scale_sequence(cur_seq, target_n)
            a_scaled = np.clip(a_scaled, 0.0, None)
            a_scaled = a_scaled * (math.sqrt(2.0 * target_n) / a_scaled.sum())

            # allocate a slice of remaining time (cap at 150 s)
            remaining = end_time - time.time()
            slice_budget = min(remaining * 0.3, 150.0)

            # randomise a few hyper‑parameters for diversity
            lr0 = rng.choice([0.45, 0.5, 0.55, 0.6])
            tau_decay = rng.choice([0.9993, 0.9995, 0.9997])
            refined, peak = _adam_optimize(a_scaled,
                                           time.time() + slice_budget,
                                           rng,
                                           lr0=lr0,
                                           tau_decay=tau_decay,
                                           lr_decay=0.99985,
                                           jitter_every=300)

            if peak < best_score:
                best_score = peak
                best_seq = refined.copy()
                best_n = target_n
                print(f"[refine] init#{idx+1} len={target_n:5d}  improved score={best_score:.6f}")

            cur_seq = refined
        if time.time() >= end_time - 20.0:
            break

    # ------------------------------------------------------------------
    # Phase 3 – linear‑programming boost (few iterations)
    # ------------------------------------------------------------------
    while time.time() < end_time - 10.0:
        cur_M = _max_convolution(best_seq)
        new_seq, new_M = _lp_boost(best_seq, cur_M, rng, mix=0.2, eps_rel=1e-9)
        if new_M < best_score - 1e-6:
            best_seq = new_seq
            best_score = new_M
            print(f"[lp] improved to {best_score:.6f}")
        else:
            break   # stop if no progress

    # ------------------------------------------------------------------
    # Phase 4 – cheap stochastic fine‑tuning (≈10 % of budget)
    # ------------------------------------------------------------------
    remaining = end_time - time.time()
    if remaining > 15.0:
        best_seq, best_score = _anneal(best_seq,
                                       time.time() + 0.6 * remaining,
                                       rng)
        remaining = end_time - time.time()
        if remaining > 5.0:
            best_seq, best_score = _hill_climb(best_seq,
                                               time.time() + remaining,
                                               rng)

    # ------------------------------------------------------------------
    # Final sanity check via the supplied black‑box evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  internal max = {best_score:.6f}"
          f"  length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
