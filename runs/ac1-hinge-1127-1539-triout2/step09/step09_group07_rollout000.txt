def _init_U_powerlaw(n, rng):
    p = rng.uniform(0.2, 3.0)
    offset = rng.uniform(0.0, 10.0)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    idx = np.arange(n)
    tail = (idx + offset) ** (-p) + (n-1 - idx + offset) ** (-p)
    raw = mix * base + (1-mix) * tail
    return raw
```

def _init_two_gaussians(n, rng):
    mu1 = rng.uniform(0, n/2)
    mu2 = rng.uniform(n/2, n-1)
    sigma = rng.uniform(0.05, 0.15) * n  # width moderate
    amp1 = rng.uniform(0.5, 2.0)
    amp2 = rng.uniform(0.5, 2.0)
    x = np.arange(n)
    raw = amp1 * np.exp(-0.5*((x - mu1) / sigma)**2) + amp2 * np.exp(-0.5*((x - mu2) / sigma)**2)
    # add small baseline
    raw += rng.uniform(0.0, 0.1)
    return raw

"""
Propose a non‑negative step‑function that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Random exploration** (≈50 % of the time budget):
   * Sample many lengths (4000–20000) and a rich set of parametric families,
     including new “U‑shaped’’ power‑law, cosine‑edge, and double‑Gaussian
     families that concentrate mass near the ends.
   * For each candidate compute the convolution peak via FFT and keep the
     best few (``top_k``) candidates.

2. **Refinement** (≈40 % of the time budget):
   * Each top candidate is refined by an Adam optimiser that works on the
     simplex (log‑softmax parametrisation) and uses the exact sub‑gradient
     of the hard max‑convolution (soft‑max temperature annealed to zero).
   * The optimiser respects the ℓ₁‑norm ``sum = sqrt(2·n)`` and non‑negativity.

3. **Final fine‑tuning** (≈10 % of the time budget):
   * A short simulated‑annealing mass‑transfer walk followed by a greedy
     hill‑climber cleans up any remaining tiny improvements.

The final sequence is stored in the global ``height_sequence_1`` and returned
as a plain list.  All numbers are non‑negative; the internal objective is
the peak of the self‑convolution, which equals the external evaluation
value (the normalisation cancels out).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions for convolution (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Peak value of a*a (full convolution)."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Exact sub‑gradient of the hard max‑convolution
# ----------------------------------------------------------------------
def _max_gradient(a: np.ndarray, tau: float):
    """
    Returns (grad, max_b) where grad = ∂ max(b) / ∂a,
    b = a*a (full convolution).  ``tau`` is the soft‑max temperature.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        eps = max_b * 1e-9 + 1e-12
        mask = (b >= max_b - eps).astype(float)
        s = mask.sum()
        if s == 0:
            mask = np.ones_like(b) / b.size
        else:
            mask /= s
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / exps.sum()

    # gradient = 2 * (mask * a_rev) convolved with a_rev
    a_rev = a[::-1]
    conv_len = mask.size + a.size - 1
    f_len = _next_pow2(conv_len)

    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)

    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    grad = 2.0 * conv[a.size - 1:a.size - 1 + a.size]
    return grad, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parameters)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   lr0: float = 0.6,
                   tau_decay: float = 0.9994,
                   lr_decay: float = 0.99985,
                   jitter_every: int = 400) -> tuple[np.ndarray, float]:
    """
    Minimise the peak of the self‑convolution under the ℓ₁‑norm constraint
    ``sum = sqrt(2·n)``.  Returns the best vector seen and its peak value.
    """
    n = a0.shape[0]
    S = math.sqrt(2.0 * n)                     # required ℓ₁‑norm

    # initialise logits from feasible a0 (log‑softmax)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                           # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr0

    # initialise temperature for τ‑soft‑max
    tau = max(_full_convolution_fft(a0).max(), 0.5)

    best_a = a0.copy()
    best_peak = _max_convolution(a0)
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- recover a from logits (log‑softmax) -----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)               # shift for stability
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ----- gradient of (soft‑)max convolution -----
        g, peak = _max_gradient(a, tau)

        # keep best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ----- gradient w.r.t. logits (chain rule) -----
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- schedules -----
        tau *= tau_decay
        lr *= lr_decay

        # occasional jitter to escape plateaus
        if jitter_every and (it % jitter_every) == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Random families (including new, more diverse families)
# ----------------------------------------------------------------------
def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.2, 5.0)
    raw = rng.gamma(alpha, 1.0, size=n)
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.1, 0.8)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_left_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    p = rng.uniform(0.3, 3.0)
    offset = rng.uniform(0.0, 10.0)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * (idx + offset) ** (-p)
    return raw


def _init_left_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    lam = rng.uniform(0.001, 0.02)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * np.exp(-lam * idx)
    return raw


def _init_sparse_bumps(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = np.full(n, rng.uniform(0.0, 0.15))
    nbumps = rng.integers(1, max(2, n // 250))
    for _ in range(nbumps):
        center = rng.integers(0, n)
        width = rng.integers(3, max(5, n // 100))
        height = rng.uniform(0.8, 4.0)
        lo = max(0, center - width // 2)
        hi = min(n, lo + width)
        raw[lo:hi] = np.maximum(raw[lo:hi], height)
    return raw


def _init_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    mu = (n - 1) / 2.0
    sigma = rng.uniform(0.05, 0.30) * n
    x = np.arange(n)
    raw = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    return raw


def _init_half_sine(n: int, rng: np.random.Generator) -> np.ndarray:
    x = np.arange(n)
    raw = np.sin(np.pi * x / (n - 1))
    raw = raw * rng.uniform(0.5, 2.0) + rng.uniform(0.0, 0.2)
    return raw


def _init_U_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """U‑shaped power‑law: high at both ends, low in the centre."""
    idx = np.arange(n)
    p = rng.uniform(0.2, 3.0)
    offset = rng.uniform(0.0, 10.0)
    mix = rng.uniform(0.0, 1.0)
    tail = (idx + offset) ** (-p) + (n - 1 - idx + offset) ** (-p)
    raw = mix * np.ones(n) + (1 - mix) * tail
    return raw


def _init_cos_edge(n: int, rng: np.random.Generator) -> np.ndarray:
    """Cosine‑edge shape – low in the centre, higher at the ends."""
    amp = rng.uniform(0.5, 2.0)
    offset = rng.uniform(0.0, 0.2)
    x = np.arange(n)
    raw = offset + amp * (1.0 - np.cos(np.pi * x / (n - 1)))
    return raw


def _init_two_gaussians(n: int, rng: np.random.Generator) -> np.ndarray:
    """Two Gaussian bumps placed far apart (centres uniformly random)."""
    mu1 = rng.uniform(0.0, n * 0.4)
    mu2 = rng.uniform(n * 0.6, n - 1.0)
    sigma = rng.uniform(0.04, 0.12) * n
    amp1 = rng.uniform(0.5, 2.0)
    amp2 = rng.uniform(0.5, 2.0)
    x = np.arange(n)
    raw = (amp1 * np.exp(-0.5 * ((x - mu1) / sigma) ** 2) +
           amp2 * np.exp(-0.5 * ((x - mu2) / sigma) ** 2))
    # small uniform background to keep positivity
    raw += rng.uniform(0.0, 0.05)
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Occasionally warm‑start from the global best; otherwise draw from a
    random parametric family.
    """
    S = math.sqrt(2.0 * n)

    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    families = [_init_dirichlet, _init_gamma, _init_uniform,
                _init_left_powerlaw, _init_left_exponential,
                _init_sparse_bumps, _init_gaussian, _init_half_sine,
                _init_U_powerlaw, _init_cos_edge, _init_two_gaussians]
    chosen = rng.choice(families)
    raw = chosen(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Simulated‑annealing / greedy hill‑climbing fine‑tuner (same as before)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Mass‑transfer walk with exponential acceptance."""
    n = len(seq)
    S = math.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score
    temperature = 0.15 * cur_score

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] == 0.0:
            continue
        delta = rng.uniform(0.0, 0.3) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())
        new_score = _max_convolution(new)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temperature, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

        temperature *= 0.99995
        if temperature < 1e-6:
            temperature = 1e-6

    return best, best_score


def _hill_climb(seq: np.ndarray,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Cheap greedy hill‑climber (downhill moves only)."""
    n = len(seq)
    S = math.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] < 1e-12:
            continue
        delta = rng.uniform(0.0, 0.15) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())
        new_score = _max_convolution(new)
        if new_score < cur_score:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The algorithm consists of random exploration, Adam refinement and a final
    stochastic fine‑tuning.  The resulting sequence is stored in the global
    variable ``height_sequence_1`` and also returned as a list.
    """
    rng = np.random.default_rng(seed)
    safety_margin = 5.0
    end_time = time.time() + budget_s - safety_margin
    if end_time <= time.time():
        raise RuntimeError("Budget too small after safety margin.")

    # ------------------------------------------------------------------
    # Warm‑start information (if any)
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # ------------------------------------------------------------------
    # Phase 1 – fast random exploration (≈50 % of budget)
    # ------------------------------------------------------------------
    explore_deadline = time.time() + 0.5 * budget_s
    top_k = 15
    top_candidates = []          # list of (peak, seq)
    best_seq = None
    best_score = np.inf
    attempt = 0

    while time.time() < explore_deadline:
        attempt += 1
        # length selection – favour medium‑large sizes
        if rng.random() < 0.75:
            n = int(rng.integers(4000, 12000))
        else:
            n = int(rng.integers(12000, 25000))

        a0 = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a0)

        if peak < best_score:
            best_score = peak
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempt:5d} n={n:5d}  new best={best_score:.6f}")

        # maintain top‑k list
        if len(top_candidates) < top_k:
            top_candidates.append((peak, a0.copy()))
            top_candidates.sort(key=lambda x: x[0])
        elif peak < top_candidates[-1][0]:
            top_candidates[-1] = (peak, a0.copy())
            top_candidates.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # Safety net – ensure at least one candidate exists
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 5000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n)
        best_score = _max_convolution(best_seq)
        best_n = n
        top_candidates = [(best_score, best_seq.copy())]

    # ------------------------------------------------------------------
    # Phase 2 – Adam refinement (≈40 % of budget)
    # ------------------------------------------------------------------
    remaining_time = end_time - time.time()
    per_candidate_budget = max(10.0, remaining_time * 0.4 / max(len(top_candidates), 1))

    for init_idx, (init_score, init_seq) in enumerate(top_candidates):
        if time.time() >= end_time - 30.0:
            break

        # allocate slice of time for this candidate
        slice_deadline = time.time() + per_candidate_budget

        refined, peak = _adam_optimize(init_seq,
                                       slice_deadline,
                                       rng,
                                       lr0=rng.choice([0.55, 0.60, 0.65]),
                                       tau_decay=rng.choice([0.9992, 0.9994, 0.9996]),
                                       lr_decay=0.99985,
                                       jitter_every=400)

        if peak < best_score:
            best_score = peak
            best_seq = refined.copy()
            best_n = len(refined)
            print(f"[refine] init#{init_idx+1} len={best_n:5d}  improved score={best_score:.6f}")

        # feed the refined vector as the warm start for the next candidates
        warm_seq = refined

    # ------------------------------------------------------------------
    # Phase 3 – final fine‑tuning (≈10 % of budget)
    # ------------------------------------------------------------------
    remaining = end_time - time.time()
    if remaining > 20.0:
        # simulated annealing
        best_seq, best_score = _anneal(best_seq,
                                       time.time() + 0.6 * remaining,
                                       rng)
        remaining = end_time - time.time()
        # greedy hill‑climber
        if remaining > 5.0:
            best_seq, best_score = _hill_climb(best_seq,
                                               time.time() + remaining,
                                               rng)

    # ------------------------------------------------------------------
    # Final sanity check via the supplied black‑box evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  internal max = {best_score:.6f}"
          f"  length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
