def max_sum_under_M(A, M, n):
    # Solve max sum(g) s.t. A @ g <= M*1, g >=0
    c = -np.ones(n)
    b = np.full(A.shape[0], M, dtype=float)
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs', options={'presolve': True})
    if not res.success:
        return None, None
    sum_g = -res.fun
    return sum_g, res.x

"""
LP‑based optimizer for the max‑self‑convolution problem.

The evaluation function is scale‑invariant:
after fixing the ℓ¹–norm to √(2·n) the score equals the maximal entry
of the self‑convolution (M = maxₖ (a*a)ₖ).  We therefore minimise M.

Algorithm
---------
1.   Initialise a feasible sequence (warm‑start if a previous result is
     available, otherwise uniform) with the required ℓ¹–norm.
2.   For the current sequence f compute M = max(f*f).
3.   Build the convolution matrix A_f where (A_f g)ₖ = (f*g)ₖ.
     For a target bound M' the linear programme

        maximise   Σ g_i
        subject to A_f g ≤ M'·1 ,   g_i ≥ 0

     returns the largest total mass that can be placed under the bound
     M'.  The function Σg(M') is monotone, so a simple bisection finds the
     smallest M' for which Σg ≥ √(2·n) (the required ℓ¹–norm).  The
     corresponding optimiser g is then scaled to exact ℓ¹–norm; because the
     scaling never increases any entry of the convolution, the new
     sequence has max‑convolution ≤ M'.
4.   Replace f with the scaled optimiser and repeat step 2–3 a few
     times (each repetition strictly reduces the maximal convolution).
5.   Finish with a short sub‑gradient descent / hill‑climb for fine
     polishing.
6.   Return the best sequence and store it in the global variable
     ``height_sequence_1`` for the next run.

The implementation respects the 1000 s budget, uses only NumPy,
SciPy (high‑level LP solver “highs”) and prints progress messages.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse

# ----------------------------------------------------------------------
# Convolution utilities (FFT – O(n log n))
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a∗a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = 1 << ((total_len - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a∗a."""
    return _full_convolution_fft(a).max()


# ----------------------------------------------------------------------
# Linear‑programming helpers
# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> scipy.sparse.csr_matrix:
    """
    Build the (2n‑1)×n matrix A where A@x = f∗x (linear convolution).
    Entry (k,i) equals f_{k‑i} if 0≤k‑i<n, otherwise 0.
    """
    n = f.shape[0]
    # row indices = i + j  (i from 0..n‑1, j from 0..n‑1)
    rows = (np.arange(n, dtype=np.int32)[:, None] + np.arange(n, dtype=np.int32)).ravel()
    cols = np.repeat(np.arange(n, dtype=np.int32), n)
    data = np.tile(f, n)                                   # f_i appears in each column
    return scipy.sparse.csr_matrix((data, (rows, cols)),
                                   shape=(2 * n - 1, n))


def _max_sum_under_M(A: scipy.sparse.csr_matrix,
                     M: float,
                     n: int):
    """
    Solve   maximise Σ g_i  subject to A g ≤ M·1 ,  g≥0 .
    Returns (sum_g, g) or (None, None) if the solver fails.
    """
    c = -np.ones(n)                         # minimise -Σg_i  ⇔ maximise Σg_i
    b = np.full(A.shape[0], M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    # ``highs`` is the default high‑performance simplex‑interior‑point hybrid.
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b,
                                bounds=bounds,
                                method='highs',
                                options={'presolve': True})
    if not res.success:
        return None, None
    sum_g = -res.fun
    return sum_g, res.x


def _find_min_feasible_M(f: np.ndarray,
                         S: float,
                         deadline: float):
    """
    For a fixed sequence f, find the smallest bound M such that there exists
    a non‑negative g with ℓ¹(g) ≥ S and (f∗g)ₖ ≤ M for all k.
    Returns (M_feasible, g_feasible).  The returned g satisfies Σg = S
    after the scaling performed by the caller.
    """
    n = f.shape[0]
    A = _build_conv_matrix(f)

    low = 0.0
    high = _max_convolution(f)               # feasible (choose g = f)
    best_g = None
    best_M = high
    # Relative tolerance – we stop when interval shrinks below 1e‑4·high
    tol = max(1e-5, 1e-4 * high)

    while high - low > tol and time.time() < deadline:
        mid = (low + high) * 0.5
        sum_g, g = _max_sum_under_M(A, mid, n)
        if sum_g is None:
            # LP failure – treat as infeasible and raise low bound
            low = mid
            continue
        if sum_g >= S:                     # feasible
            high = mid
            best_M = mid
            best_g = g
        else:
            low = mid

    # If the bisection never found a feasible point (should not happen)
    if best_g is None:
        best_g = f.copy()
        best_M = high
    return best_M, best_g


# ----------------------------------------------------------------------
# Simple sub‑gradient descent (fine polishing)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Hard‑max sub‑gradient descent with simplex projection."""
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    a = a0.copy()
    best_a = a.copy()
    best_val = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1
        b = _full_convolution_fft(a)
        k = int(b.argmax())
        val = b[k]

        if val < best_val:
            best_val = val
            best_a = a.copy()

        # sub‑gradient of (a*a)_k
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        jdx = k - idx
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[jdx]

        step = 0.8 * S / (np.sqrt(it) + 1.0)
        a = a - step * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, S / n)
        else:
            a = a * (S / s)

        # occasional jitter
        if it % 500 == 0:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                frac = rng.uniform(0.02, 0.12)
                amount = a[i] * frac
                a[i] -= amount
                a[j] += amount
                a = a * (S / a.sum())
    return best_a, best_val

    variable ``height_sequence_1`` for warm‑starts.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Warm‑start handling
    # ------------------------------------------------------------------
    warm_seq = globals().get('height_sequence_1')
    if warm_seq is not None:
        try:
            warm_arr = np.asarray(warm_seq, dtype=float)
            n = warm_arr.shape[0]
            S = np.sqrt(2.0 * n)
            a = warm_arr.copy()
            # re‑normalise just in case the stored sequence was clipped
            if a.sum() == 0.0:
                a = np.full(n, S / n)
            else:
                a = a * (S / a.sum())
        except Exception:
            warm_arr = None

    if warm_seq is None or warm_arr is None:
        # No usable warm‑start → pick a moderate length
        n = 2600
        S = np.sqrt(2.0 * n)
        a = np.full(n, S / n)                # uniform start

    best_seq = a.copy()
    best_score = _max_convolution(best_seq)
    print(f"[init] length={n}  init max={best_score:.6f}")

    # ------------------------------------------------------------------
    # Outer LP‑based refinement loop (usually 3‑5 iterations)
    # ------------------------------------------------------------------
    outer_iter = 0
    while time.time() < end_time - 30.0 and outer_iter < 6:
        outer_iter += 1
        S = np.sqrt(2.0 * a.shape[0])
        # Find the smallest feasible bound M and the corresponding g
        deadline_lp = time.time() + (end_time - time.time()) * 0.4   # up to 40 % of remaining time
        M_feas, g_raw = _find_min_feasible_M(a, S, deadline_lp)

        # Scale g to exact ℓ¹‑norm (never increases any convolution entry)
        sum_g = g_raw.sum()
        if sum_g == 0.0:
            break
        a_new = g_raw * (S / sum_g)

        new_score = _max_convolution(a_new)

        if new_score + 1e-9 < best_score:
            best_score = new_score
            best_seq = a_new.copy()
            a = a_new
            print(f"[lp  iter {outer_iter}] improved max={best_score:.6f}  bound={M_feas:.6f}")
        else:
            # No strict improvement – still replace to continue the descent
            a = a_new
            print(f"[lp  iter {outer_iter}] no improvement (new={new_score:.6f})")
        # tiny random perturbation to help escape plateaus in the next iteration
        if time.time() < end_time - 30.0:
            pert = rng.normal(scale=0.005 * S / a.shape[0], size=a.shape[0])
            a = np.maximum(a + pert, 0.0)
            a = a * (S / a.sum())

    # ------------------------------------------------------------------
    # Short sub‑gradient polishing phase
    # ------------------------------------------------------------------
    remaining = end_time - time.time() - 5.0
    if remaining > 10.0:
        a_ref, peak_ref = _subgradient_descent(best_seq,
                                               time.time() + remaining,
                                               rng)
        if peak_ref < best_score:
            best_score = peak_ref
            best_seq = a_ref.copy()
            print(f"[subgrad] final max={best_score:.6f}")

    # ------------------------------------------------------------------
    # Very cheap stochastic hill‑climb (few seconds)
    # ------------------------------------------------------------------
    while time.time() < end_time - 1.0:
        a = best_seq.copy()
        n = a.shape[0]
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        frac = rng.uniform(0.02, 0.12)
        amount = a[i] * frac
        a[i] -= amount
        a[j] += amount
        a = a * (np.sqrt(2.0 * n) / a.sum())
        peak = _max_convolution(a)
        if peak + 1e-9 < best_score:
            best_score = peak
            best_seq = a.copy()
            print(f"[hill] improved max={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f}  internal max={best_score:.6f}  length={len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
