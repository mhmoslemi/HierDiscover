"""
Propose a non‑negative step function that minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)**2)

The algorithm follows three phases within the given wall‑clock budget:

1.  Several short restarts of projected Adam on a smooth log‑sum‑exp
    surrogate of the max‑convolution.  The smoothing parameter β is
    increased quickly so that the gradient focuses on the true maximum.
2.  An aggressive deterministic hill‑climb.  At each iteration the
    current worst convolution entry is identified, a source index that
    contributes the most to it is chosen and a small amount of mass is
    transferred to a destination index with the smallest counterpart.
    Several candidate destinations are evaluated (full convolution via
    FFT) and the best improving move is accepted.  Random jitter is used
    to escape plateaus.
3.  A short polishing Adam run.

The routine respects the time budget (default 1000 s) and returns the
best sequence found.  If a previous construction is available in the
global variable ``height_sequence_1`` it is used as a seed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x ≥ 0, Σx = s}.
    """
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(af * af, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using an FFT (fast O(n log n))."""
    return float(_full_convolution(a).max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float,
                   deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max surrogate.
    Runs until the wall‑clock deadline.
    """
    a = a0.copy()
    n = a.size

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # learning‑rate (scaled with the simplex radius)
    base_lr = 0.1 * target_sum

    # smoothed‑max schedule
    beta = 1.0
    beta_max = 1e8
    beta_factor = 2.0

    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if t % 100 == 0:
            jitter = rng.normal(
                scale=target_sum * 1e-5 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # increase smoothing
        if t % 250 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a

# Random initialisers
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                  # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                  # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                  # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                  # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Aggressive hill‑climbing on the exact convolution
# ----------------------------------------------------------------------


def _hill_climb_extended(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climb that, for the current worst convolution entry,
    tries several destination indices and picks the best improving move.
    The full convolution is recomputed (FFT) after each accepted move.
    Returns (best_sequence, best_score).
    """
    a = a0.copy()
    n = a.size
    b = _full_convolution(a)
    best_a = a.copy()
    best_score = b.max()

    rng = np.random.default_rng()
    K = 6                # number of destination candidates examined
    move_factor = 0.8    # fraction of source mass moved when testing a move
    eps_improve = 1e-12

    while time.time() < deadline:
        d = int(np.argmax(b))                     # worst shift
        i_low = max(0, d - (n - 1))
        i_high = min(n - 1, d)
        idx = np.arange(i_low, i_high + 1, dtype=int)

        # source that contributes most to this shift
        contrib = a[idx] * a[d - idx]
        if contrib.size == 0:
            break
        i_star = idx[np.argmax(contrib)]

        if a[i_star] < 1e-14:
            # mass exhausted – jitter and recompute
            jitter = rng.normal(scale=1e-7, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, math.sqrt(2.0 * n))
            b = _full_convolution(a)
            continue

        # counterpart for each possible destination in the same range
        counterpart = a[d - idx]                     # a[d‑j] for j in idx
        order = np.argsort(counterpart)
        # select up to K destinations with smallest counterpart (excluding i_star)
        cand_js = []
        for jj in order:
            j = idx[jj]
            if j != i_star:
                cand_js.append(j)
            if len(cand_js) >= K:
                break
        if not cand_js:
            # fallback: any other index
            cand_js = [j for j in range(n) if j != i_star][:K]

        improved = False
        best_local_max = best_score
        best_local_j = None
        best_local_delta = None

        # try each candidate destination
        for j in cand_js:
            # try a few fractions of the source mass
            for frac in (move_factor, move_factor * 0.5, move_factor * 0.25):
                delta = a[i_star] * frac
                if delta <= 0.0:
                    continue
                a_test = a.copy()
                a_test[i_star] -= delta
                a_test[j] += delta
                # sum is preserved – no projection needed
                b_test = _full_convolution(a_test)
                new_max = b_test.max()
                if new_max < best_local_max - eps_improve:
                    best_local_max = new_max
                    best_local_j = j
                    best_local_delta = delta
                    improved = True

        if improved and best_local_j is not None:
            # accept the best move found
            a[i_star] -= best_local_delta
            a[best_local_j] += best_local_delta
            b = _full_convolution(a)

            if best_local_max < best_score - eps_improve:
                best_score = best_local_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill‑climb improved to {best_score:.6f}")

            continue

        # no improving move – perform a tiny random jitter
        jitter = rng.normal(scale=1e-6, size=n)
        a = np.maximum(a + jitter, 0.0)
        a = _project_to_simplex(a, math.sqrt(2.0 * n))
        b = _full_convolution(a)

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects the wall‑clock budget (default 1000 s) and
    returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    best_seq = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # 1)  Seed from a previous run if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            n0 = prev.size
            target0 = math.sqrt(2.0 * n0)
            prev = _project_to_simplex(prev, target0)
            best_seq = prev.copy()
            best_val = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] seeded from height_sequence_1 "
                  f"(n={n0}) score={best_val:.6f}")
        except Exception as e:
            print(f"seeding error: {e}")

    # ------------------------------------------------------------------
    # 2)  Length schedule – try a few lengths, allocating time dynamically
    # ------------------------------------------------------------------
    schedule = [2000, 4000, 8000]

    for n in schedule:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Initialise a candidate for this length
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)

        a_init = _project_to_simplex(a_init, target_sum)

        # --------------------------------------------------------------
        # Allocate a slice of the remaining budget
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 12.0:
            break

        gd_budget = remaining * 0.30
        hc_budget = remaining * 0.55
        extra_budget = remaining - gd_budget - hc_budget

        # --------------------------------------------------------------
        # Phase A – several short Adam restarts
        # --------------------------------------------------------------
        min_restart = 5.0                     # seconds per restart
        n_restarts = max(1, int(gd_budget // min_restart))
        per_restart = gd_budget / n_restarts if n_restarts else 0.0

        local_best = None
        local_best_score = np.inf

        for r in range(n_restarts):
            if time.time() >= deadline:
                break
            start_vec = a_init.copy() if r == 0 else _random_initial_sequence(n, rng)
            a_opt = _adam_optimize(start_vec, target_sum,
                                   time.time() + per_restart - 0.1)
            score = _max_convolution_fft(a_opt)
            if score < local_best_score:
                local_best_score = score
                local_best = a_opt.copy()

        if local_best is not None and local_best_score < best_val - 1e-12:
            best_seq = local_best.copy()
            best_val = local_best_score
            print(f"[{time.time():.1f}] Adam improvement (n={n}) score={best_val:.6f}")

        # --------------------------------------------------------------
        # Phase B – aggressive deterministic hill‑climbing
        # --------------------------------------------------------------
        if time.time() < deadline - 1.0:
            a_cur = best_seq.copy()
            a_cur = _project_to_simplex(a_cur, target_sum)
            a_hc, score_hc = _hill_climb_extended(
                a_cur, time.time() + hc_budget - 0.1)
            if score_hc < best_val - 1e-12:
                best_seq = a_hc.copy()
                best_val = score_hc
                print(f"[{time.time():.1f}] Hill‑climb improvement (n={n}) "
                      f"score={best_val:.6f}")

        # --------------------------------------------------------------
        # Phase C – short polishing Adam run
        # --------------------------------------------------------------
        if time.time() < deadline - 0.5 and extra_budget > 2.0:
            a_ref = _adam_optimize(best_seq, target_sum,
                                   deadline - 0.5)
            val_ref = _max_convolution_fft(a_ref)
            if val_ref < best_val - 1e-12:
                best_seq = a_ref.copy()
                best_val = val_ref
                print(f"[{time.time():.1f}] Polishing Adam (n={n}) score={best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Final polishing on whatever length we ended up with
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1.0:
        n = best_seq.size
        target = math.sqrt(2.0 * n)
        final_deadline = deadline - 0.5
        best_seq = _adam_optimize(best_seq, target, final_deadline)
        # one short hill‑climb pass
        b_final = _full_convolution(best_seq)
        best_seq, final_score = _hill_climb_extended(
            best_seq, deadline - 0.2)
        if final_score < best_val - 1e-12:
            best_val = final_score
            print(f"[{time.time():.1f}] Final polishing improvement score={best_val:.6f}")

    # ------------------------------------------------------------------
    # 4) Clean‑up: drop trailing near‑zeros
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution
        n = 2000
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]

    # ------------------------------------------------------------------

    try:

              f"(length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
