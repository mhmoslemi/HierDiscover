def _full_convolution_fft(a):
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = 1 << ((total_len - 1).bit_length())
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv

def _adam_optimize(a0, deadline, rng):
    n = len(a0)
    S = np.sqrt(2*n)
    # Initialize log-weights
    p = np.log(np.maximum(a0, 1e-12) / S)  # log-softmax initial
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5
    # initial temperature
    b = _full_convolution_fft(a0)
    tau = max(b.max(), 10.0)
    best_a = a0.copy()
    best_val = b.max()
    it = 0
    while time.time() < deadline:
        it += 1
        # get a = softmax(p) * S
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        a = exp_p / exp_p.sum() * S
        # gradient via softmax approx
        b = _full_convolution_fft(a)
        M = b.max()
        # compute mask
        if tau < 1e-12:
            mask = (b == M).astype(float)
        else:
            diff = b - M
            exps = np.exp(diff / tau)
            mask = exps / np.sum(exps)
        # compute gradient: g = 2 * conv(mask, reversed a)
        a_rev = a[::-1]
        # Use convolution length = len(mask) + n - 1
        conv_len = len(mask) + n - 1
        f_len = 1 << ((conv_len - 1).bit_length())
        Fmask = np.fft.rfft(mask, n=f_len)
        Fa = np.fft.rfft(a_rev, n=f_len)
        conv = np.fft.irfft(Fmask * Fa, n=f_len)[:conv_len]
        # gradient w.r.t a
        g = 2.0 * conv[n-1:n-1+n]
        # track best using hard M
        if M < best_val:
            best_val = M
            best_a = a.copy()
        # gradient w.r.t logits
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_p
        v = beta2 * v + (1 - beta2) * (grad_p**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps)
        # update temperature and lr decay
        tau *= 0.9999
        lr *= 0.9999
        # jitter occasionally
        if it % 500 == 0:
            p += rng.normal(scale=0.01, size=n)
    return best_a, best_val
```

def _subgradient_multi(a0, deadline, rng):
    n = len(a0)
    S = np.sqrt(2*n)
    a = a0.copy()
    best_a = a.copy()
    best_val = _max_convolution(a)
    it = 0
    while time.time() < deadline:
        it += 1
        b = _full_convolution_fft(a)
        M = b.max()
        # identify near-max mask
        # delta proportional to relative tolerance
        delta = max(1e-9, M * 1e-4)
        mask = (b >= M - delta).astype(float)
        # gradient via mask
        a_rev = a[::-1]
        conv_len = len(mask) + n - 1
        f_len = 1 << ((conv_len - 1).bit_length())
        Fmask = np.fft.rfft(mask, n=f_len)
        Fa_rev = np.fft.rfft(a_rev, n=f_len)
        conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
        g = 2.0 * conv[n-1:n-1+n]
        # step size
        step = 0.7 * S / (np.sqrt(it) + 1.0)
        a = a - step * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, S / n)
        else:
            a = a * (S / s)
        if M < best_val:
            best_val = M
            best_a = a.copy()
        # occasional random mass transfer
        if it % 300 == 0:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                frac = rng.uniform(0.02, 0.15)
                amount = a[i] * frac
                a[i] -= amount
                a[j] += amount
                a = a * (S / a.sum())
    return best_a, best_val

"""
Optimiser for the self‑convolution peak problem.

The evaluator is scale–invariant: after normalising the ℓ1‑norm to
sqrt(2·n) the score equals the maximal entry of a∗a.  The algorithm
keeps the sequence on the simplex (non‑negative, fixed ℓ1‑norm) and
combines several complementary heuristics:

1. Fast random exploration of a few thousand‑length families.
2. Linear interpolation to several larger lengths.
3. Mirror‑descent on the simplex (log‑softmax + Adam) using a τ‑soft‑max
   approximation of the max‑convolution.
4. Multi‑constraint sub‑gradient descent that follows *all* near‑max
   convolution entries (mask‑gradient).
5. A short second Adam run with a very low temperature for fine‑tuning.
6. Cheap stochastic hill‑climb (mass transfer between two positions).

The search is time‑bounded (default 1000 s) and updates the global
variable ``height_sequence_1`` with the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a∗a."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Random families for the initial vectors
# ----------------------------------------------------------------------
def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    return mix * edge + (1.0 - mix) * base


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    return mix * exp_part + (1.0 - mix) * base


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    return mix * lin + (1.0 - mix) * base


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    return mix * ush + (1.0 - mix) * base


def _init_cosine(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    raw = 1.0 - np.cos(np.pi * i / (n - 1))
    raw = raw * rng.uniform(0.5, 2.0)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * raw + (1.0 - mix) * base


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    return rng.random(n)


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.05, 0.9)
    return rng.gamma(shape, 1.0, size=n)


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    With some probability we warm‑start from ``warm_seq`` (scaled);
    otherwise we draw from a random parametric family.
    The returned vector satisfies sum = sqrt(2·n) and all entries ≥0.
    """
    S = np.sqrt(2.0 * n)

    # ----- warm start (≈20 % chance) -----
    if warm_seq is not None and rng.random() < 0.23:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
        return a

    # ----- random family -----
    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'gamma', 'uniform', 'dirichlet', 'ushape', 'cosine'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'cosine':
        raw = _init_cosine(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Adam optimiser (mirror descent on the simplex) with τ‑soft‑max
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam on the simplex (log‑softmax parametrisation) minimising the
    τ‑soft‑max approximation of the convolution peak.
    Returns the best sequence seen and its *hard* max value.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                     # required ℓ1‑norm

    # initialise logits from a0 (log‑softmax)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.55                 # slightly more aggressive than before

    # start temperature (large ⇒ smooth)
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 8.0)

    best_a = a0.copy()
    best_val = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ---- recover a from logits ----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ---- convolution and τ‑soft‑max mask ----
        b = _full_convolution_fft(a)
        M = b.max()

        if tau < 1e-12:
            mask = (b == M).astype(float)
        else:
            diff = b - M
            exps = np.exp(diff / tau)
            mask = exps / np.expm1(0) if exps.sum() == 0 else exps / exps.sum()

        # ---- gradient of τ‑soft‑max ----
        a_rev = a[::-1]
        conv_len = len(mask) + n - 1
        f_len = _next_pow2(conv_len)
        Fmask = np.fft.rfft(mask, n=f_len)
        Fa_rev = np.fft.rfft(a_rev, n=f_len)
        conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
        g = 2.0 * conv[n - 1:n - 1 + n]          # shape (n,)

        # keep hard best
        if M < best_val:
            best_val = M
            best_a = a.copy()

        # ---- gradient w.r.t logits (mirror descent) ----
        mu = np.dot(a, g) / S                    # ⟨a, g⟩ / S
        grad_p = a * (g - mu)

        # optional clipping to avoid exploding updates
        clip = np.max(np.abs(grad_p))
        if clip > 6.0 * S:
            grad_p = grad_p * (6.0 * S / clip)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- schedules (slow decay) ----
        tau *= 0.99993
        lr *= 0.99993

        # jitter occasionally
        if it % 600 == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_val


# ----------------------------------------------------------------------
# Multi‑constraint sub‑gradient descent (mask‑gradient)
# ----------------------------------------------------------------------
def _subgradient_multi(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent that follows *all* convolution entries whose
    value is within a small relative tolerance of the current maximum.
    Returns the best sequence seen and its hard max.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    a = a0.copy()
    best_a = a.copy()
    best_val = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1

        b = _full_convolution_fft(a)
        M = b.max()

        # choose a tolerance that captures the few largest entries
        delta = max(1e-9, M * 1e-4)
        mask = (b >= M - delta).astype(float)

        # gradient = 2 * conv(mask, a_rev) restricted to n entries
        a_rev = a[::-1]
        conv_len = len(mask) + n - 1
        f_len = _next_pow2(conv_len)
        Fmask = np.fft.rfft(mask, n=f_len)
        Fa_rev = np.fft.rfft(a_rev, n=f_len)
        conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
        g = 2.0 * conv[n - 1:n - 1 + n]

        # step size – classic 1/√t decay
        step = 0.75 * S / (np.sqrt(it) + 1.0)
        a = a - step * g
        a = np.maximum(a, 0.0)

        total = a.sum()
        if total == 0.0:
            a = np.full(n, S / n)
        else:
            a = a * (S / total)

        if M < best_val:
            best_val = M
            best_a = a.copy()

        # occasional random mass transfer to break tiny plateaus
        if it % 400 == 0:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                frac = rng.uniform(0.02, 0.14)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = a * (S / a.sum())

    return best_a, best_val


# ----------------------------------------------------------------------
# Cheap stochastic hill‑climb (pairwise mass transfer)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Very cheap local search: repeatedly move a random fraction of mass
    from one position to another.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_val = _max_convolution(a)

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        frac = rng.uniform(0.02, 0.12)
        amount = a[i] * frac
        a[i] -= amount
        a[j] += amount
        # re‑normalise to the required ℓ1‑norm
        S = np.sqrt(2.0 * n)
        a = a * (S / a.sum())
        val = _max_convolution(a)
        if val + 1e-12 < cur_val:            # strict improvement
            cur_val = val
            a0 = a.copy()                     # keep the improved one
    return a0, cur_val

This implementation expands on the previous hybrid Adam + sub‑gradient approach by:

* Adding a *mask‑gradient* sub‑gradient method that follows **all** near‑maximal
  convolution entries rather than only the single maximal one.
* Using a slightly more aggressive learning‑rate schedule and a smoother temperature decay
  in the Adam phase.
* Allocating a larger portion of the time budget to the refinement phases,
  while still preserving a fast random‑exploration front‑end.
* Introducing a second Adam run with a low temperature after the hard‑max sub‑gradient
  descent for very fine tuning.
* Keeping a cheap stochastic hill‑climb until the very last second.

The solver respects the 1000 s budget, works on the simplex (non‑negative entries,
fixed ℓ1‑norm) and stores the best found sequence in the global variable
``height_sequence_1``.
