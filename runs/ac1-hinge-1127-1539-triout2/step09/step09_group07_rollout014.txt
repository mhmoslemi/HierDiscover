"""
Propose a step‑function sequence minimizing ``evaluate_sequence``.

Algorithm
---------
1.  The objective (after the mandatory ℓ₁‑normalisation
    Σaᵢ=√(2n)) equals the peak of the linear self‑convolution
    ``b = a*a``.
    For a non‑negative decreasing exponential
        a[i] = exp(-λ·i)  (i=0,…,n‑1)
    the maximal entry of ``b`` is a₀², while the centre entry
    (inner product with the reversed vector) is
        n·a₀²·exp(-λ·(n‑1)).
    Choosing λ ≈ (log n)/n makes the centre term ≤ a₀², so the
    evaluation is essentially a₀² = (√(2n)·λ)² = 2·n·λ².

    Taking λ = c·log(n)/n with a small constant c>1 gives
    ``evaluate ≈ 2·c²·(log n)² / n`` which rapidly falls below 0.01
    for n≈10⁴.

2.  We scan a handful of lengths (2000 … 64000) and a small grid of
    integers ``c`` (≈⌈log n⌉ … ⌈log n⌉+7).  For each pair we
    build the exponential, renormalise to Σaᵢ=√(2n), compute the
    max‑convolution via an FFT, and keep the best candidate.

3.  With the remaining time we run a cheap stochastic “anneal”
    walk (mass‑transfer moves) and a strict greedy hill‑climber to
    capture any final improvement.

The whole search respects the 1000 s budget, prints progress and
stores the best sequence in ``height_sequence_1`` for future runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Peak value of a*a (full convolution)."""
    return _full_convolution_fft(a).max()


# ----------------------------------------------------------------------
# Simple stochastic mass‑transfer walk (simulated annealing)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Mass‑transfer walk on the simplex, accepting occasional uphill moves."""
    n = len(seq)
    S = math.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score
    temperature = 0.15 * cur_score

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] == 0.0:
            continue
        delta = rng.uniform(0.0, 0.3) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())
        new_score = _max_convolution(new)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temperature, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

        temperature *= 0.99995
        if temperature < 1e-6:
            temperature = 1e-6

    return best, best_score


def _hill_climb(seq: np.ndarray,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Strict downhill hill‑climber (no uphill moves)."""
    n = len(seq)
    S = math.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] < 1e-12:
            continue
        delta = rng.uniform(0.0, 0.15) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (S / new.sum())
        new_score = _max_convolution(new)

        if new_score < cur_score:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

    return best, best_score

    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    safety_margin = 5.0                      # keep a few seconds for final check
    end_time = time.time() + budget_s - safety_margin
    if end_time <= time.time():
        raise RuntimeError("Budget too small after safety margin.")

    best_val = float('inf')
    best_seq = None
    attempts = 0

    # ------------------------------------------------------------------
    # Phase 1 – exhaustive scan over exponential families
    # ------------------------------------------------------------------
    # candidate lengths (powers of two, limited by 80000 per problem statement)
    lengths = [2000, 4000, 8000, 16000, 32000, 64000]
    for n in lengths:
        if time.time() >= end_time:
            break

        # minimal integer c that ensures λ·n > log n
        c_min = max(1, int(math.ceil(math.log(n))))
        # try a few consecutive values of c
        for c in range(c_min, c_min + 8):
            lam = c / n                      # λ = c / n  (≈ log n / n)
            # build exponential vector
            idx = np.arange(n, dtype=float)
            raw = np.exp(-lam * idx)        # decreasing exponential
            total = raw.sum()
            if total == 0.0:                # should never happen
                continue
            scale = math.sqrt(2.0 * n) / total
            a = raw * scale                  # now Σa = √(2n)
            val = _max_convolution(a)       # equals evaluate_sequence for a feasible a

            attempts += 1
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[scan] n={n:5d} c={c:2d} λ={lam:.6e} val={val:.6f}")

            # early break if we already reach a very tiny value
            if best_val < 1e-4:
                break
        # optional early exit if already excellent
        if best_val < 1e-4:
            break

    # ------------------------------------------------------------------
    # Phase 2 – stochastic fine‑tuning of the current best candidate
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = end_time - time.time()
        if remaining > 20.0:
            # use simulated annealing for ~60 % of the remaining time
            best_arr, best_val2 = _anneal(best_seq,
                                          time.time() + 0.6 * remaining,
                                          rng)
            if best_val2 < best_val:
                best_val = best_val2
                best_seq = best_arr.copy()
            # final greedy hill‑climb for the rest
            remaining2 = end_time - time.time()
            if remaining2 > 5.0:
                best_arr2, best_val3 = _hill_climb(best_seq,
                                                   time.time() + remaining2,
                                                   rng)
                if best_val3 < best_val:
                    best_val = best_val3
                    best_seq = best_arr2.copy()

    # ------------------------------------------------------------------
    # Final sanity check via the supplied evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)     # type: ignore[name-defined]
    print(f"[final] external score = {external:.6f}  internal max = {best_val:.6f}"
          f"  length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
