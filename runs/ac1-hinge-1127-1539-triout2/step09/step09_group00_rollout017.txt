        n = 6000

    target_sum = np.sqrt(2.0 * n)

    # Phase 0 parametric search
    candidates = []
    # (pure power law)
    for p in np.linspace(0.30, 0.80, 13):
        seq = (np.arange(1,n+1, dtype=np.float64) ** (-p))
        seq *= target_sum / seq.sum()
        candidates.append(seq)

    # (breakpoint power+exp tail)
    for p in np.linspace(0.30, 0.80, 9):
        for q in np.linspace(0.0, 6.0, 7):
            for r in np.linspace(0.2, 0.8, 7):
                cut = int(r * n)
                i = np.arange(n, dtype=np.float64)
                seq = (i + 1) ** (-p)
                if cut < n:
                    tail_idx = np.arange(cut, n)
                    seq[tail_idx] *= np.exp(-q * (tail_idx - cut) / (n - cut))
                seq *= target_sum / seq.sum()
                candidates.append(seq)

def _subgradient_descent(a0, target_sum, deadline, step_factor=0.5, epsilon=1e-12):
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    conv = _conv_fft(a)
    max_val = conv.max()
    n = a.size
    avg_val = (target_sum * target_sum) / (2*n-1)  # average of convolution entries = sum_a^2/(2n-1) = 2n/(2n-1).
    # But not needed.
    best_val = max_val

    # for momentum we could use similar to Adam but for simplicity, use subgradient step.
    while time.time() < deadline:
        # compute max idx
        k = int(np.argmax(conv))
        max_val = conv[k]

        # compute subgradient g
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi+1)
        # contributions
        g[idx] = a[idx] + a[k - idx]  # includes both terms; ok.

        # compute step size based on gap to average maybe.
        # gap = max_val - avg_val
        gap = max_val - avg_val
        if gap < 1e-8:
            break
        norm_g = np.linalg.norm(g)
        if norm_g < epsilon:
            break

        alpha = step_factor * gap / (norm_g**2 + epsilon)

        # limit alpha to keep nonnegative
        pos_idx = g > 1e-12
        if np.any(pos_idx):
            alphas = a[pos_idx] / g[pos_idx]
            max_alpha = np.min(alphas) * 0.999
            if alpha > max_alpha:
                alpha = max_alpha

        a_new = a - alpha * g
        a_new = _project_to_simplex(a_new, target_sum)
        conv_new = _conv_fft(a_new)
        max_new = conv_new.max()
        if max_new < best_val:
            best_val = max_new
            best = a_new.copy()
            print(f"[{time.time():.2f}] Subgrad it best max={best_val:.6f}")

        # accept if improvement else maybe keep old with some probability
        # For simplicity always accept new if max_new <= max_val else revert
        if max_new <= max_val:
            a = a_new
            conv = conv_new
        else:
            # reduce step factor
            step_factor *= 0.7
            # optional: continue
            continue

        # optional jitter occasionally
        # Could also increase step_factor when not improvement.

    return best, best_val

    conv = _conv_fft(a)
    max_val = conv.max()

"""
Step‑function optimizer for the benchmark.

The objective (with the normalisation Σa = √(2 n)) is simply
    score = max_k (a * a)_k
i.e. minimise the largest entry of the autocorrelation.
The algorithm consists of four phases:

0. Very fast parametric sweep (pure power‑laws, power‑law + exponential tail
   and a breakpoint variant).  A few thousand candidates are generated,
   scored with a single FFT and only the best few are kept.

1. Projected sub‑gradient descent on the exact max‑convolution.
   The sub‑gradient is computed analytically (g_i = a_i + a_{k‑i} for the
   current maximal index k).  A simple diminishing step‑size rule and a
   projection onto the simplex Σa = √(2 n) are used.  Several random restarts
   are performed.

2. A short deterministic mass‑transfer refinement (the “DMT” from the
   reference code) to further attack the current maximal entry.

3. Final polishing with the soft‑max gradient descent from the reference
   implementation (annealed temperature up to 5 · 10⁶).

All phases respect the global time budget and the best sequence found is
returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple parametric families
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _break_power_exp_sequence(
    n: int, p: float, q: float, r: float, target_sum: float
) -> np.ndarray:
    """(i+1)^(-p) with an exponential tail after a break point r∈(0,1)."""
    cut = int(r * n)
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p)
    if cut < n:
        tail = np.arange(cut, n, dtype=np.float64)
        a[cut:] *= np.exp(-q * (tail - cut) / (n - cut))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max gradient (used for the final polishing)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_val = conv.max()
    exps = np.exp(beta * (conv - max_val))
    p = exps / np.sum(exps)                     # soft‑max weights

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, conv


def _annealed_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 2e5,
    step_factor: float = 0.6,
    gamma: float = 0.99,
) -> tuple[np.ndarray, float]:
    """Momentum projected GD on the soft‑max approximation."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    step_base = target_sum * step_factor
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur = _max_convolution_fft(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD β={beta:6.1f} best={best_val:.6f}")

        if it % 300 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on max‑convolution
# ----------------------------------------------------------------------


def _exact_subgradient_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    step_factor: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent using the exact max‑convolution gradient.
    Returns (best_vector, best_max).
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    n = a.shape[0]
    avg_val = (target_sum * target_sum) / (2 * n - 1)  # the theoretical average

    best_a = a.copy()
    conv = _conv_fft(a)
    best_max = conv.max()

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        # current max and its indices (allow a tiny tolerance)
        cur_max = conv.max()
        max_idxs = np.where(conv >= cur_max - 1e-12)[0]

        # build sub‑gradient (sum of gradients of all maximal entries)
        g = np.zeros_like(a)
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            g[idx] += a[idx] + a[k - idx]

        norm_g = np.linalg.norm(g)
        if norm_g < 1e-15:
            break

        # step size proportional to the gap to the average entry
        gap = cur_max - avg_val
        if gap < 1e-12:
            break
        alpha = step_factor * gap / (norm_g ** 2 + 1e-15)

        # keep non‑negativity (projected later, but we limit to avoid huge overshoot)
        pos = g > 1e-14
        if np.any(pos):
            max_alpha = np.min(a[pos] / g[pos]) * 0.999
            if alpha > max_alpha:
                alpha = max_alpha

        a_new = a - alpha * g
        a_new = _project_to_simplex(a_new, target_sum)
        conv_new = _conv_fft(a_new)
        new_max = conv_new.max()

        # simple acceptance rule
        if new_max <= cur_max:
            a, conv = a_new, conv_new
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time():.2f}] Sub‑grad it={it} best_max={best_max:.6f}")
        else:
            # if no improvement, shrink the step factor
            step_factor *= 0.7
            # occasional jitter to escape plateaus
            if it % 200 == 0:
                jitter = rng.normal(scale=target_sum * 0.001, size=n)
                a = _project_to_simplex(a + jitter, target_sum)
                conv = _conv_fft(a)

        it += 1

    return best_a, best_max


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (phase B from the reference)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    start_temp: float = 0.5,
    end_temp: float = 1e-6,
    low_frac: float = 0.05,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    cur_max = _max_convolution_fft(a)
    best_a = a.copy()
    best_val = cur_max
    rng = np.random.default_rng()
    start = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]
        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w.max() - w.min()
        if weight_diff > 0.0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0.0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_max = conv.max()

        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob = np.exp(-(new_max - cur_max) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_max = new_max
            if new_max < best_val:
                best_val = new_max
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it} new best={best_val:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_max = conv.max()

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine respects the time budget, prints progress information and
    returns the best sequence found (as a plain Python list of floats).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ---------------------------------------------------------------
    # Choose problem size (larger n gives more degrees of freedom)
    # ---------------------------------------------------------------
    if budget_s >= 800:
        n = 15000
    elif budget_s >= 400:
        n = 10000
    else:
        n = 6000

    target_sum = np.sqrt(2.0 * n)   # Σa is fixed to this value (scaling invariant)

    # ---------------------------------------------------------------
    # Phase 0 – fast parametric pre‑search
    # ---------------------------------------------------------------
    rng = np.random.default_rng()
    candidates = []

    # a) pure power‑law
    for p in np.linspace(0.30, 0.80, 13):
        candidates.append(_power_law_sequence(n, p, target_sum))

    # b) power‑law + exponential tail
    for p in np.linspace(0.30, 0.80, 9):
        for q in np.linspace(0.0, 6.0, 7):
            candidates.append(_exp_power_law_sequence(n, p, q, target_sum))

    # c) breakpoint + exponential tail
    for p in np.linspace(0.30, 0.80, 9):
        for q in np.linspace(0.0, 6.0, 7):
            for r in np.linspace(0.2, 0.8, 7):
                candidates.append(_break_power_exp_sequence(n, p, q, r, target_sum))

    # d) previous best (if available)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        candidates.append(_resize_sequence(prev, n))

    # e) jittered variants of everything above (one jitter per candidate)
    jitter_scale = target_sum * 0.003
    for seq in list(candidates):
        seq_j = seq + rng.normal(scale=jitter_scale, size=n)
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        candidates.append(seq_j)

    # ---------------------------------------------------------------
    # Keep only a manageable subset for the cheap scoring pass
    # ---------------------------------------------------------------
    max_initial = min(200, len(candidates))
    sample_idx = rng.choice(len(candidates), size=max_initial, replace=False)

    best_seq = None
    best_val = np.inf

    for i in sample_idx:
        seq = candidates[i]
        val = _max_convolution_fft(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
    print(f"[{time.time():.2f}] Initial best max = {best_val:.6f}")

    # ---------------------------------------------------------------
    # Phase 1 – exact sub‑gradient descent with random restarts
    # ---------------------------------------------------------------
    subgrad_deadline = time.time() + (deadline - time.time()) * 0.45
    while time.time() < subgrad_deadline - 2.0:
        jitter = rng.normal(scale=target_sum * 0.02, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _exact_subgradient_descent(
            start_vec, target_sum, subgrad_deadline, step_factor=0.8
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Sub‑grad improvement: {best_val:.6f}")

    # ---------------------------------------------------------------
    # Phase 2 – deterministic mass‑transfer refinement
    # ---------------------------------------------------------------
    if time.time() < deadline - 5.0:
        a_ref, val_ref = _deterministic_mass_transfer(
            best_seq, deadline, target_sum, start_temp=0.5, end_temp=1e-6, low_frac=0.05
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] DMT improvement: {best_val:.6f}")

    # ---------------------------------------------------------------
    # Phase 3 – final polishing (soft‑max GD with very high β)
    # ---------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _annealed_gradient_descent(
            best_seq,
            deadline,
            target_sum,
            max_beta=5e6,
            step_factor=0.5,
            gamma=0.99,
        )
        if val_fin < best_val:
            best_val = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best = {best_val:.6f}")

    # ---------------------------------------------------------------
    # Post‑processing: trim trailing zeros and return as Python list
    # ---------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-13)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final evaluation (harmless if evaluate_sequence is not defined)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
