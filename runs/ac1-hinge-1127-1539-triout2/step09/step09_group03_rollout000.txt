def _subgradient_descent(a0, target_sum, deadline):
    a = a0.copy()
    best = a.copy()
    best_val = _score(a)
    it=0
    rng = np.random.default_rng()
    step0 = target_sum * 0.8  # initial step magnitude
    while time.time() < deadline:
        b = _conv_fft(a)  # exact conv
        max_val = b.max()
        if max_val != best_val:
            # compute gradient of current max (if multiple indices, pick one randomly)
            k_idxs = np.where(b == max_val)[0]
            k = rng.choice(k_idxs)
            g = np.zeros_like(a)
            lo = max(0, k - (len(a)-1))
            hi = min(k, len(a)-1)
            # indices i from lo..hi inclusive correspond to j=k-i
            # g[i] = 2 * a[k-i]
            idx = np.arange(lo, hi+1)
            g[idx] = 2 * a[k - idx]
        else:
            # not necessary
            g = np.zeros_like(a)

        # step size schedule
        eta = step0 / np.sqrt(it + 1)
        # update
        a = a - eta * g
        a = np.maximum(a, 0.0)
        # project to simplex with sum target_sum
        a = _project_to_simplex(a, target_sum)

        # evaluate
        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
        # occasional jitter
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(np.maximum(a,0), target_sum)
        it += 1
    return best, best_val

    construction (global ``height_sequence_1``) exists it is also considered.

2️⃣  **Projected sub‑gradient descent on the true max‑convolution** – the
    objective  F(a)=maxₖ (a∗a)[k] is convex but non‑smooth.  The sub‑gradient
    at the current maximising index k∗ is  g_i = 2·a_{k∗‑i} (zero outside the
    support).  A diminishing step size α_t = α₀/√t is used and after each
    update the vector is projected onto the simplex {x≥0, Σx = √(2 n)}.
    Occasional Gaussian jitter prevents stagnation.

3️⃣  **Exact local refinement** – a simulated‑annealing style hill‑climb that
    transfers a small amount of mass from a “donor’’ index (large contribution
    to the current maximal entry) to a “receiver’’ index (small contribution).
    The convolution is updated in O(n) using slice arithmetic, so thousands of
    moves can be performed quickly.  The temperature linearly decays to zero,
    allowing uphill moves early on.

The three stages are timed to respect the 1000 s budget; the best vector
found is truncated (trailing zeros are removed) and returned as a plain
list of floats.  The result is stored in ``height_sequence_1`` for warm‑starts
in subsequent calls.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) / (sum(a)²)."""
    n = a.shape[0]
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic shape generators
# ----------------------------------------------------------------------
def _make_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam * i) (i = 0,…,n‑1)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a


def _make_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i = (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    return a


def _make_linear(n: int, height: float, slope: float) -> np.ndarray:
    """a_i = max(height - slope*i, 0)."""
    i = np.arange(n, dtype=np.float64)
    a = np.maximum(height - slope * i, 0.0)
    return a


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on max(a*a)."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_val = _score(a)

    alpha0 = target_sum * 0.8          # initial step magnitude
    it = 0
    while time.time() < deadline:
        # compute convolution and identify a maximising index
        b = _conv_fft(a)
        max_val = b.max()
        # pick one maximiser uniformly at random (helps when several tie)
        max_idxs = np.flatnonzero(b == max_val)
        k = rng.choice(max_idxs)

        # sub‑gradient of the max (only the selected k matters)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # diminishing step size
        eta = alpha0 / np.sqrt(it + 1)

        a = a - eta * grad
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        # evaluate
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional progress report
            if it % 200 == 0:
                print(f"[{time.time():.2f}] SubGD it={it:5d} score={best_val:.6f}")

        # jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact local mass‑transfer refinement (simulated annealing)
# ----------------------------------------------------------------------
def _local_refine(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """Hill‑climbing on the true max‑convolution using O(n) updates."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # current full convolution
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start

    # initial temperature (roughly 4 % of the current score)
    T0 = best_val * 0.04
    last_full = start

    while time.time() < deadline:
        # -------------------------------------------------------------
        # Identify donor (large contribution) and receiver (small)
        # -------------------------------------------------------------
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        prod = a[idx] * a[k_max - idx]               # contribution of each index

        # donor chosen with probability proportional to its product
        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver sampled from the low‑product tail (≤30‑th percentile)
        thresh = np.percentile(prod, 30)
        low_candidates = idx[prod <= thresh]
        if low_candidates.size:
            receiver = rng.choice(low_candidates)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # -------------------------------------------------------------
        # Propose a transfer of mass
        # -------------------------------------------------------------
        # allow somewhat larger moves than the previous version
        max_delta = min(a[donor] * 0.08, target_sum * 0.0016)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # -------------------------------------------------------------
        # Update convolution in O(n)
        # -------------------------------------------------------------
        b_new = b.copy()
        # linear term: -2·delta·a at donor, +2·delta·a at receiver
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic terms
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # -------------------------------------------------------------
        # Simulated‑annealing acceptance
        # -------------------------------------------------------------
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))

        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            # commit the move
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Local refine improved score={best_val:.6f}")

        # -------------------------------------------------------------
        # Periodic re‑normalisation (combat numerical drift)
        # -------------------------------------------------------------
        if time.time() - last_full > 12.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_full = time.time()
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑norm improved score={best_val:.6f}")

        # occasional tiny jitter
        if int(elapsed * 100) % 2500 == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.005, target_sum * 0.0004)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The function respects the given CPU‑time budget (seconds) and returns
    the best sequence found as a plain Python list.
    """
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin for final evaluation

    # -------------------------------------------------------------
    # Choose problem size (larger n gives a richer search space)
    # -------------------------------------------------------------
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000
    target_sum = np.sqrt(2.0 * n)

    # -------------------------------------------------------------
    # Warm‑start: generate analytic candidates and pick the best
    # -------------------------------------------------------------
    candidates = []

    # 1️⃣  use previous global best if available (resized)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        candidates.append(_resize_sequence(prev, n))

    # 2️⃣  exponential decay – sweep a few lambdas
    for lam in np.linspace(0.0001, 0.0015, 12):
        a = _make_exponential(n, lam)
        a = a * (target_sum / a.sum())
        candidates.append(a)

    # 3️⃣  power‑law – sweep exponents
    for p in np.linspace(0.35, 1.10, 12):
        a = _make_powerlaw(n, p)
        a = a * (target_sum / a.sum())
        candidates.append(a)

    # 4️⃣  simple linear drop (height, slope tuned)
    for height, slope in [(1.5, 0.00007), (2.0, 0.00009), (1.2, 0.00005)]:
        a = _make_linear(n, height, slope)
        a = a * (target_sum / a.sum())
        candidates.append(a)

    # evaluate all candidates (cheap – O(#candidates) FFTs)
    best_seq = None
    best_score = np.inf
    for idx, a in enumerate(candidates):
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
        print(f"[{time.time():.2f}] candidate #{idx:02d} score={sc:.6f}")

    print(f"[{time.time():.2f}] Warm‑start best score={best_score:.6f}")

    # -------------------------------------------------------------
    # Stage 2 – sub‑gradient descent on the true max‑convolution
    # -------------------------------------------------------------
    if time.time() < deadline:
        sd_deadline = time.time() + 0.35 * (deadline - time.time())
        a_sg, val_sg = _subgradient_descent(best_seq, target_sum, sd_deadline)
        if val_sg < best_score:
            best_seq = a_sg
            best_score = val_sg
        print(f"[{time.time():.2f}] After sub‑gradient best score={best_score:.6f}")

    # -------------------------------------------------------------
    # Stage 3 – exact local mass‑transfer refinement (remaining time)
    # -------------------------------------------------------------
    if time.time() < deadline:
        a_ref, val_ref = _local_refine(best_seq, target_sum, deadline)
        if val_ref < best_score:
            best_seq = a_ref
            best_score = val_ref
        print(f"[{time.time():.2f}] After local refinement best score={best_score:.6f}")

    # -------------------------------------------------------------
    # Final clean‑up: drop trailing near‑zeros and evaluate with the provided
    #                evaluate_sequence (the official benchmark)
    # -------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # official evaluation (may further clip values, but we already obey constraints)
    try:
        final_val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for warm‑starts in future runs
    globals()["height_sequence_1"] = final_seq

    return final_seq


# ----------------------------------------------------------------------
# Small utility used by the warm‑start (linear resizing)
# ----------------------------------------------------------------------
def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)
