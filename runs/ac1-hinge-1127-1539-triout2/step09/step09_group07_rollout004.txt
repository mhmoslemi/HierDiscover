"""
Propose a step‑function sequence that (approximately) minimises
``evaluate_sequence``.

The algorithm consists of three stages:

1. **Broad random exploration** – many cheap candidates of
   heterogeneous lengths are drawn from a richer set of parametric
   families (including edge‑heavy and double‑bump shapes).  The best
   few are kept.

2. **Progressive multi‑scale optimisation** – the current best candidate
   is repeatedly interpolated to a larger support (up to ~80 k points)
   and refined with an Adam optimiser on the simplex.  The optimiser
   works on log‑softmax parameters, uses the exact sub‑gradient of the
   hard‑max of the self‑convolution (via a tiny soft‑max temperature)
   and includes jitter, learning‑rate decay and a simple gradient clip.
   Several Adam restarts are performed at the final (largest) length.

3. **Fine‑tuning** – a simulated‑annealing mass‑transfer walk followed
   by a strict greedy hill‑climber is applied while time permits.
   Finally the external ``evaluate_sequence`` is called for verification.

All steps respect the 1000 s budget (a 5 s safety margin is kept).
The best construction is stored in the global variable
``height_sequence_1`` and returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT‑based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Peak value of a*a (full convolution)."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Resize a sequence by linear interpolation.
    ``seq`` is assumed to be 1‑D.
    """
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Exact (hard‑max) sub‑gradient of max(b) where b = a*a
# ----------------------------------------------------------------------
def _max_gradient(a: np.ndarray, tau: float):
    """
    Returns (grad, max_b) where

        grad = ∂ max(b) / ∂ a    and    max_b = max(b)  with b = a*a.

    If ``tau`` is very small a hard‑max is used (all indices whose value
    is within a tiny tolerance of the maximum are averaged).  Otherwise a
    soft‑max with temperature ``tau`` is used.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    # ---- weighting mask -----------------------------------------------
    if tau < 1e-12:
        eps = max_b * 1e-9 + 1e-12
        mask = (b >= max_b - eps).astype(float)
        s = mask.sum()
        if s == 0:
            mask = np.ones_like(b) / b.size
        else:
            mask /= s
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / exps.sum()

    # ---- gradient -------------------------------------------------------
    # grad = 2 * conv(mask, reverse(a)) * reverse(a)  (restricted to length n)
    a_rev = a[::-1]
    conv_len = mask.size + a.size - 1
    f_len = _next_pow2(conv_len)

    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)

    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    grad = 2.0 * conv[a.size - 1 : a.size - 1 + a.size]
    return grad, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   lr0: float = 0.5,
                   tau_decay: float = 0.9995,
                   lr_decay: float = 0.99985,
                   jitter_every: int = 400) -> tuple[np.ndarray, float]:
    """
    Minimise the peak of the self‑convolution under the ℓ₁‑norm constraint
    ``sum = sqrt(2·n)``.  Returns the best vector seen and the corresponding
    hard‑max value.
    """
    n = a0.shape[0]
    target_sum = math.sqrt(2.0 * n)          # required ℓ₁‑norm

    # initialise logits (log‑softmax) from a feasible a0
    eps = 1e-12
    tmp = np.maximum(a0 / target_sum, eps)
    tmp = tmp / tmp.sum()
    p = np.log(tmp)                          # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr0

    # start with a relatively high temperature (soft‑max≈hard‑max)
    tau = max(_full_convolution_fft(a0).max(), 0.5)

    best_a = a0.copy()
    best_peak = _max_convolution(a0)
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- recover a from logits (log‑softmax) -----------------------
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)               # stability shift
        soft = exp_p / exp_p.sum()
        a = soft * target_sum

        # ----- gradient of (soft‑)max convolution -----------------------
        g, peak = _max_gradient(a, tau)

        # keep the best hard‑max seen so far
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ----- chain rule: gradient w.r.t. logits -----------------------
        mu = np.dot(a, g) / target_sum
        grad_p = a * (g - mu)

        # optional clipping for stability
        max_abs = np.max(np.abs(grad_p))
        if max_abs > 5.0 * target_sum:
            grad_p = grad_p * (5.0 * target_sum / max_abs)

        # ----- Adam update ---------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- annealing of temperature and learning rate ---------------
        tau *= tau_decay
        lr *= lr_decay

        # ----- occasional jitter to escape plateaus --------------------
        if jitter_every and (it % jitter_every) == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak

def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.2, 5.0)
    return rng.gamma(alpha, 1.0, size=n)


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.1, 0.8)
    return rng.gamma(shape, 1.0, size=n)


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_left_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    p = rng.uniform(0.3, 3.0)
    offset = rng.uniform(0.0, 10.0)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * (idx + offset) ** (-p)
    return raw


def _init_left_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    lam = rng.uniform(0.001, 0.02)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * np.exp(-lam * idx)
    return raw


def _init_sparse_bumps(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = np.full(n, rng.uniform(0.0, 0.15))
    nbumps = rng.integers(1, max(2, n // 250))
    for _ in range(nbumps):
        center = rng.integers(0, n)
        width = rng.integers(3, max(5, n // 100))
        height = rng.uniform(0.8, 4.0)
        lo = max(0, center - width // 2)
        hi = min(n, lo + width)
        raw[lo:hi] = np.maximum(raw[lo:hi], height)
    return raw


def _init_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    mu = (n - 1) / 2.0
    sigma = rng.uniform(0.05, 0.30) * n
    x = np.arange(n)
    return np.exp(-0.5 * ((x - mu) / sigma) ** 2)


def _init_half_sine(n: int, rng: np.random.Generator) -> np.ndarray:
    x = np.arange(n)
    raw = np.sin(np.pi * x / (n - 1))
    raw = raw * rng.uniform(0.5, 2.0) + rng.uniform(0.0, 0.2)
    return raw


def _init_edge_heavy(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    High values at the two edges, low in the middle (U‑shape).
    """
    idx = np.arange(n, dtype=float)
    ctr = (n - 1) / 2.0
    scale = rng.uniform(0.05, 0.30) * n
    power = rng.uniform(1.0, 4.0)
    raw = 1.0 / (1.0 + ((np.abs(idx - ctr) / scale) ** power))
    raw = raw * rng.uniform(0.5, 2.0) + rng.uniform(0.0, 0.2)
    return raw


def _init_double_bump(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Two separated rectangular bumps (symmetrically placed).
    """
    raw = np.full(n, rng.uniform(0.0, 0.05))
    width = rng.integers(max(2, n // 20), max(3, n // 10))
    gap = rng.integers(width, max(width * 2, n // 4))
    start1 = rng.integers(0, n - 2 * width - gap)
    start2 = start1 + width + gap
    height1 = rng.uniform(0.8, 3.0)
    height2 = rng.uniform(0.8, 3.0)
    raw[start1 : start1 + width] = np.maximum(raw[start1 : start1 + width], height1)
    raw[start2 : start2 + width] = np.maximum(raw[start2 : start2 + width], height2)
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length ``n``.
    With probability ≈0.22 a warm‑start (interpolated previous best)
    is used; otherwise a random parametric family is drawn.
    """
    target_sum = math.sqrt(2.0 * n)

    # Warm start ≈22 % of the time
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * target_sum
        return a

    # Choose a family
    style = rng.choice(['dirichlet', 'gamma', 'uniform',
                        'left_powerlaw', 'left_exponential',
                        'sparse', 'gaussian', 'half_sine',
                        'edge_heavy', 'double_bump'])
    if style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'left_powerlaw':
        raw = _init_left_powerlaw(n, rng)
    elif style == 'left_exponential':
        raw = _init_left_exponential(n, rng)
    elif style == 'sparse':
        raw = _init_sparse_bumps(n, rng)
    elif style == 'gaussian':
        raw = _init_gaussian(n, rng)
    elif style == 'half_sine':
        raw = _init_half_sine(n, rng)
    elif style == 'edge_heavy':
        raw = _init_edge_heavy(n, rng)
    else:  # double_bump
        raw = _init_double_bump(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (target_sum / total)
    return a


# ----------------------------------------------------------------------
# Simulated‑annealing / greedy hill‑climbing fine‑tuner
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Mass‑transfer walk on the simplex with exponential acceptance."""
    n = seq.shape[0]
    target_sum = math.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score
    temperature = 0.15 * cur_score

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] == 0.0:
            continue
        # transfer a random fraction of the mass at i to j
        delta = rng.uniform(0.0, 0.3) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (target_sum / new.sum())
        new_score = _max_convolution(new)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temperature, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

        temperature *= 0.99995
        if temperature < 1e-6:
            temperature = 1e-6

    return best, best_score


def _hill_climb(seq: np.ndarray,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Strictly downhill greedy hill‑climber (fast, cheap)."""
    n = seq.shape[0]
    target_sum = math.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] < 1e-12:
            continue
        delta = rng.uniform(0.0, 0.15) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = new * (target_sum / new.sum())
        new_score = _max_convolution(new)
        if new_score < cur_score:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

    return best, best_score

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The result is stored in the global variable ``height_sequence_1`` and
    also returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    # keep a small safety margin (the harness will kill us if we exceed the budget)
    safety_margin = 5.0
    end_time = time.time() + budget_s - safety_margin
    if end_time <= time.time():
        raise RuntimeError("Budget too small after safety margin.")

    # ------------------------------------------------------------------
    # Warm‑start from the previously best construction, if any
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # ------------------------------------------------------------------
    # Phase 1 – fast random exploration (≈15 % of budget)
    # ------------------------------------------------------------------
    explore_deadline = time.time() + 0.15 * budget_s
    top_k = 20
    top_candidates = []                     # (peak, vector)
    best_seq = None
    best_peak = np.inf
    attempts = 0

    while time.time() < explore_deadline:
        attempts += 1
        # length distribution – mix of moderate and larger sizes
        if rng.random() < 0.75:
            n = int(rng.integers(3000, 8000))
        else:
            n = int(rng.integers(8000, 25000))

        a0 = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a0)

        if peak < best_peak:
            best_peak = peak
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempts:5d} n={n:5d}  new best={best_peak:.6f}")

        # keep a short list of the best candidates
        if len(top_candidates) < top_k:
            top_candidates.append((peak, a0.copy()))
            top_candidates.sort(key=lambda x: x[0])
        elif peak < top_candidates[-1][0]:
            top_candidates[-1] = (peak, a0.copy())
            top_candidates.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # Ensure we have at least one candidate
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 4000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n)
        best_peak = _max_convolution(best_seq)
        best_n = n
        top_candidates = [(best_peak, best_seq.copy())]

    # ------------------------------------------------------------------
    # Phase 2 – progressive multi‑scale refinement (≈70 % of budget)
    # ------------------------------------------------------------------
    max_len = min(80000, best_n * 15)   # allow a fairly aggressive growth
    # geometric schedule of lengths
    lengths = [best_n]
    cur_len = best_n
    while cur_len * 1.6 < max_len:
        cur_len = int(cur_len * 1.6)
        lengths.append(cur_len)
    lengths.append(max_len)
    lengths = sorted(set(lengths))

    # refine only the most promising candidate (the current global best)
    cur_seq = best_seq.copy()
    cur_len = best_n

    for target_n in lengths:
        if time.time() >= end_time - 30.0:
            break

        # interpolate to the new length and renormalise exactly
        a_scaled = _scale_sequence(cur_seq, target_n)
        a_scaled = np.clip(a_scaled, 0.0, None)
        a_scaled = a_scaled * (math.sqrt(2.0 * target_n) / a_scaled.sum())

        # allocate a time slice (max 180 s) – earlier lengths get a larger slice
        remaining = end_time - time.time()
        slice_budget = min(remaining * 0.45, 180.0)

        # perform a few Adam restarts at this scale, keep the best
        best_local = a_scaled.copy()
        best_local_peak = _max_convolution(a_scaled)
        restarts = 2 if target_n < 20000 else 1
        for r in range(restarts):
            # small random jitter around the current point to create diversity
            jitter = rng.normal(scale=0.02 * best_local.mean(), size=target_n)
            init = best_local + jitter
            init = np.clip(init, 0.0, None)
            init = init * (math.sqrt(2.0 * target_n) / init.sum())

            refined, peak = _adam_optimize(init,
                                           time.time() + slice_budget,
                                           rng,
                                           lr0=0.55,
                                           tau_decay=0.9993,
                                           lr_decay=0.99985,
                                           jitter_every=300)
            if peak < best_local_peak:
                best_local_peak = peak
                best_local = refined.copy()

        # update global best if improved
        if best_local_peak < best_peak:
            best_peak = best_local_peak
            best_seq = best_local.copy()
            best_n = target_n
            print(f"[refine] length={target_n:5d}  new best={best_peak:.6f}")

        cur_seq = best_local
        cur_len = target_n

    # ------------------------------------------------------------------
    # Phase 3 – final stochastic fine‑tuning (≈10 % of budget)
    # ------------------------------------------------------------------
    remaining = end_time - time.time()
    if remaining > 20.0:
        # simulated annealing walk
        best_seq, best_peak = _anneal(best_seq,
                                      time.time() + 0.6 * remaining,
                                      rng)
        remaining = end_time - time.time()
        # strict greedy hill‑climber
        if remaining > 5.0:
            best_seq, best_peak = _hill_climb(best_seq,
                                              time.time() + remaining,
                                              rng)

    # ------------------------------------------------------------------
    # Final verification with the provided black‑box evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  internal max = {best_peak:.6f}"
          f"  length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
