def _aggressive_local_refine(a0, target_sum, deadline):
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    best_a = a.copy()
    best_val = _score(a)
    best_max = b.max()
    start = time.time()
    rng = np.random.default_rng()
    recompute_interval = 10.0
    last_recalc = start
    # parameters
    # allow temperature annealing maybe.
    while time.time() < deadline:
        # compute current max index
        k_max = int(np.argmax(b))

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        # product contributions
        prod = a[idx] * a[k_max - idx]
        # donor: max product
        donor = idx[np.argmax(prod)]
        # receiver: min product (or pick index with smallest a)
        # we can consider searching indices with minimal product, but we also want low product with k_max offset
        receiver_candidates = idx[prod == prod.min()]  # may be many
        # choose one randomly
        receiver = rng.choice(receiver_candidates)
        if donor == receiver:
            continue
        # candidate delta: could move up to a[donor] * 0.2 (20%) but not exceed sum
        max_delta = min(a[donor] * 0.2, target_sum * 0.005)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # Update convolution efficiently
        b_new = b.copy()
        # linear parts
        # For donor: subtract 2*delta*a shifted
        b_new[donor:donor + n] -= 2.0 * delta * a
        # For receiver: add 2*delta*a shifted
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic parts
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        # Accept if lower
        if new_max < best_max or rng.random() < 0.01:   # occasional random acceptance
            # apply
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
                best_val = float(2.0 * n * best_max / (target_sum ** 2))
                print(f"[{time.time():.2f}] Local refine improved max={best_max:.6f}")
        # occasional recompute to fix numerical errors
        now = time.time()
        if now - last_recalc > recompute_interval:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_recalc = now
            cur_max = b.max()
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                best_val = float(2.0 * n * best_max / (target_sum ** 2))
                print(f"[{time.time():.2f}] Recompute improved max={best_max:.6f}")

                    best_val = float(2.0 * n * best_max / (target_sum ** 2))
                    print(f"[{time.time():.2f}] Jitter improved max={best_max:.6f}")

    return best_a, best_val

"""
Multi‑stage search for a low‑score non‑negative step function.

The algorithm works in four phases:

1. **Warm‑start / diverse initialisations** – reuse the best known
   construction (if any) and generate a few simple shapes
   (power‑law, exponential, triangular).  All are scaled to the target
   ℓ¹‑norm `√(2·n)`.

2. **Smooth optimisation** – a short Adam optimiser on the
   soft‑max approximation of `max(a*a)`.  This provides a good
   neighbourhood for the final refinement.

3. **Exponential (mirror) descent** – multiplicative updates on the same
   surrogate (helps when the optimum is on the simplex boundary).

4. **Aggressive local refinement** – repeated exact mass transfers from
   the index that hurts the current maximal convolution entry most
   (`donor = argmax a[i]·a[k_max‑i]`) to the index that hurts it least
   (`receiver = argmin a[j]·a[k_max‑j]`).  Updates are performed in O(n)
   by slicing the convolution array; occasional full recomputation
   avoids drift.  A tiny simulated‑annealing acceptance rule allows the
   search to escape shallow plateaus.

All steps respect non‑negativity and the ℓ¹‑norm constraint,
so the returned sequence can be fed directly to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using FFT (length 2·n‑1).
    Returns a real‑valued 1‑d array of length 2·n‑1.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv.real


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective  2·n·max(a*a) / (sum(a)²).  For Σa=√(2n) this equals max(a*a)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, Σx = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where conv is the full convolution a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()                     # soft‑max weights over the 2n‑1 entries

    # grad = 2 * (reverse(a) convolve p)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]    # keep only the n components that affect a
    return grad, b


# ----------------------------------------------------------------------
# Phase 2 – Adam (projected) descent on the soft‑max surrogate
# ----------------------------------------------------------------------


def _adam_descent(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation of the soft‑max surrogate.
    The ℓ¹‑norm is enforced after each step.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.2
    it = 0
    beta = 1.0
    max_beta = 65536.0
    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1
        g, _ = _gradient_softmax(a, beta)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = base_lr / (np.sqrt(v_hat) + eps)
        a -= step * m_hat

        # projection onto the simplex (keeps sum exactly target_sum)
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:5.1f} score={best_val:.6f}")

        # slowly sharpen the soft‑max surrogate
        if it % 250 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

    return best_a, best_val

def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (multiplicative) on the soft‑max surrogate.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.03
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.0007, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it:5d} β={beta:5.1f} score={best_val:.6f}")

        if it % 250 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

    return best_a, best_val

def _annealed_gradient_descent_v2(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent (Adam‑like momentum) on the soft‑max surrogate.
    Used inside the restart wrapper.
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    step_base = target_sum * 0.45
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.93
    it = 0

    while time.time() < deadline:
        it += 1
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.0012 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GDv2 it={it:5d} β={beta:5.1f} score={best_val:.6f}")

        if it % 250 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Multiple short GD restarts – keep the best result."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_v2(start,
                                               time.time() + run_budget,
                                               target_sum)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val

# Phase 4 – aggressive exact local refinement
# ----------------------------------------------------------------------


def _aggressive_local_refine(a0: np.ndarray,
                             target_sum: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climbing with larger mass transfers.
    In each iteration we move a small amount of mass from the index that
    contributes most to the current maximal convolution entry to the index
    that contributes the least.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    best_a = a.copy()
    best_val = _score(a)
    best_max = b.max()

    rng = np.random.default_rng()
    start = time.time()
    recompute_every = 12.0                # seconds
    last_recalc = start
    total_time = deadline - start
    T0 = best_val * 0.04                  # initial temperature for occasional uphill moves

    while time.time() < deadline:
        # --------------------------------------------------------------
        # locate the offending offset
        # --------------------------------------------------------------
        k_max = int(np.argmax(b))

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contributions a[i] * a[k_max-i]
        prod = a[idx] * a[k_max - idx]

        donor = idx[np.argmax(prod)]
        receiver = idx[np.argmin(prod)]
        if donor == receiver:
            # fallback to a random move if everything is equal
            donor, receiver = rng.choice(idx, size=2, replace=False)

        # --------------------------------------------------------------
        # propose a mass transfer
        # --------------------------------------------------------------
        max_delta = min(a[donor] * 0.15, target_sum * 0.0015)
        if max_delta <= 0:
            # nothing to move – skip this iteration
            continue
        delta = rng.random() * max_delta

        # --------------------------------------------------------------
        # update the full convolution in O(n) using slices
        # --------------------------------------------------------------
        b_new = b.copy()
        # linear parts (remove from donor, add to receiver)
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic self‑terms
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        # --------------------------------------------------------------
        # decision (simulated annealing)
        # --------------------------------------------------------------
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_max < best_max:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_max - best_max) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
                best_val = float(2.0 * n * best_max / (target_sum ** 2))
                print(f"[{time.time():.2f}] Local refine improved max={best_max:.6f}")

        # --------------------------------------------------------------
        # occasional full recomputation to keep numerical errors low
        # --------------------------------------------------------------
        if time.time() - last_recalc > recompute_every:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_recalc = time.time()
            cur_max = b.max()
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                best_val = float(2.0 * n * best_max / (target_sum ** 2))
                print(f"[{time.time():.2f}] Full recompute improved max={best_max:.6f}")

        # very occasional tiny jitter to shake out of plateaus
        if int(elapsed * 30) % 4000 == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.005, target_sum * 0.0005)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_max = b.max()
                if cur_max < best_max:
                    best_max = cur_max
                    best_a = a.copy()
                    best_val = float(2.0 * n * best_max / (target_sum ** 2))
                    print(f"[{time.time():.2f}] Jitter improved max={best_max:.6f}")

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose problem size based on the time budget
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 600:
        n = 14000
    else:
        n = 9000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Build a pool of initial sequences
    # ------------------------------------------------------------------
    init_pool = []

    # warm‑start (if available)
    if "height_sequence_1" in globals():
        init_pool.append(_resize_sequence(np.array(globals()["height_sequence_1"],
                                                  dtype=np.float64), n))
    else:
        # simple decreasing power‑law fallback
        init_pool.append((np.arange(1, n + 1, dtype=np.float64) ** (-0.55)))

    # exponential families with different decay rates
    for c in (0.5, 1.0, 1.5, 2.0):
        r = np.exp(-c / n)                     # geometric ratio
        geom = (1 - r) / (1 - r ** n) * target_sum * (r ** np.arange(n))
        init_pool.append(geom)

    # triangular shape – high at the edges, low in the centre
    x = np.linspace(0.0, 1.0, n)
    tri = np.abs(x - 0.5) * 2.0               # 0 in the middle, 1 at the borders
    tri = tri / tri.mean() * target_sum / n   # scale to have the right ℓ¹‑norm
    init_pool.append(tri)

    # ------------------------------------------------------------------
    # Phase 1 – short Adam optimisation on each start (≈10 % of budget)
    # ------------------------------------------------------------------
    best_a = None
    best_val = np.inf
    time_per_init = max(5.0, 0.08 * (deadline - time.time()) / len(init_pool))

    for idx, init in enumerate(init_pool):
        if time.time() >= deadline:
            break
        sub_deadline = time.time() + time_per_init
        a, val = _adam_descent(init, target_sum, sub_deadline)
        if val < best_val:
            best_val = val
            best_a = a.copy()
        print(f"[{time.time():.2f}] Init {idx} Adam best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exponential (mirror) descent (≈15 % of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        exp_deadline = time.time() + 0.15 * (deadline - time.time())
        a_exp, val_exp = _exponential_descent(best_a, target_sum, exp_deadline)
        if val_exp < best_val:
            best_val = val_exp
            best_a = a_exp.copy()
        print(f"[{time.time():.2f}] After exponential descent best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – projected GD with restarts (≈30 % of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        gd_deadline = time.time() + 0.30 * (deadline - time.time())
        a_gd, val_gd = _gradient_descent_with_restarts(best_a,
                                                       target_sum,
                                                       gd_deadline)
        if val_gd < best_val:
            best_val = val_gd
            best_a = a_gd.copy()
        print(f"[{time.time():.2f}] After GD restarts best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – aggressive exact local refinement (rest of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, val_ref = _aggressive_local_refine(best_a,
                                                  target_sum,
                                                  deadline)
        if val_ref < best_val:
            best_val = val_ref
            best_a = a_ref.copy()
        print(f"[{time.time():.2f}] After local refinement best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up, final evaluation and warm‑start storage
    # ------------------------------------------------------------------
    arr = np.array(best_a, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)          # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    globals()["height_sequence_1"] = final_seq

    return final_seq
