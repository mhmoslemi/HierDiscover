"""
Step‑function optimizer (binary mask + Adam / sub‑gradient fine‑tuning).

The evaluator is scale‑invariant, i.e. the objective depends only on the
shape of the sequence.  The algorithm works in three stages:

1. **Random binary mask search** – a large number of short trials are
   performed.  For a randomly chosen length `n` a random subset of
   positions of size `m` is filled with the same weight
   `w = sqrt(2·n)/m`.  The resulting piecewise‑constant step function has
   a very flat self‑convolution and often yields a score well below the
   previous best (≈1.5).  A cheap hill‑climb that swaps a “1’’ with a
   “0’’ is applied locally to each candidate.

2. **Continuous refinement** – the best sequence found so far is handed
   to a mirror‑descent Adam optimiser that minimizes a τ‑soft‑max
   approximation of the convolution peak, followed by a hard‑max
   sub‑gradient descent as in the reference implementation.

3. **Final stochastic hill‑climb** – a few random mass‑transfer moves
   are tried until the time budget expires.

All work is performed within the 1000 s budget; the best sequence is
stored in the global variable ``height_sequence_1`` and returned as a
plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a∗a."""
    return _full_convolution_fft(a).max()


def _objective(a: np.ndarray) -> float:
    """
    The value returned by ``evaluate_sequence`` for a given array.
    Equivalent to 2·n·max(a*a) / (sum(a)^2).
    """
    n = a.shape[0]
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Generation of binary‑mask sequences
# ----------------------------------------------------------------------
def _binary_seq(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """
    Return a length‑`n` non‑negative vector with exactly `m` equal
    positive entries and all others zero.  The entries are set so that
    the ℓ₁‑norm equals √(2·n), i.e. the objective reduces to the raw
    maximum of the self‑convolution.
    """
    S = np.sqrt(2.0 * n)
    weight = S / max(1, m)
    a = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=m, replace=False)
    a[idx] = weight
    return a


# ----------------------------------------------------------------------
# τ‑soft‑max gradient (used by Adam)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """Gradient of the τ‑soft‑max approximation of max(b) where b = a*a."""
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient: 2*(mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = mask.shape[0] + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Adam optimiser that minimizes the *hard* max of the convolution."""
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                     # target ℓ₁‑norm

    # initialise log‑softmax logits from a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                         # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.45

    # start temperature (smoothness)
    tau = max(_full_convolution_fft(a0).max(), 5.0)

    best_a = a0.copy()
    best_peak = _full_convolution_fft(a0).max()
    it = 0

    while time.time() < deadline:
        it += 1

        # recover a from logits
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # smoothed gradient
        g, peak = _softmax_gradient(a, tau)

        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # convert gradient to log‑softmax space
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # clip for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 6.0 * S:
            grad_p = grad_p * (6.0 * S / clip_val)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # schedules (very slow decay)
        tau *= 0.99995
        lr *= 0.99995

        # occasional jitter
        if it % 800 == 0:
            p += rng.normal(scale=0.006, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Sub‑gradient descent on the *hard* max (no smoothing)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Simple sub‑gradient descent that follows the current maximal entry."""
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    a = a0.copy()
    best_a = a.copy()
    best_val = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1

        b = _full_convolution_fft(a)
        k = int(b.argmax())
        val = b[k]

        if val < best_val:
            best_val = val
            best_a = a.copy()

        # sub‑gradient of (a*a)_k
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        jdx = k - idx
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[jdx]

        step = 0.8 * S / (np.sqrt(it) + 1.0)

        a = a - step * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, S / n)
        else:
            a = a * (S / s)

        # occasional random jitter
        if it % 500 == 0:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                frac = rng.uniform(0.02, 0.12)
                amount = a[i] * frac
                a[i] -= amount
                a[j] += amount
                a = a * (S / a.sum())

    return best_a, best_val

# ----------------------------------------------------------------------

        cur_obj = obj

                if cur_obj < best_obj:

            else:

                cur_mask[j] = False

    # ------------------------------------------------------------------
    # Phase 2 – continuous refinement with Adam + sub‑gradient
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30.0:

                                             time.time() + per_run,

                if obj < best_obj:

                    best_seq = a_opt.copy()
                    print(f"[adam] restart={r} obj={obj:.6f}")
