def _deterministic_flatten(a0: np.ndarray, target_sum: float, deadline: float) -> tuple[np.ndarray,float]:
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)   # exact convolution
    best_a = a.copy()
    best_val = _score(a)
    start = time.time()
    total = deadline - start
    init_factor = 0.25  # fraction of donor mass to move initially
    final_factor = 0.0005
    # Ensure we have enough time to run at least a few iterations.
    while time.time() < deadline:
        # Find current max convolution index
        k_max = int(np.argmax(b))
        # Determine overlapping indices where a[i] and a[k_max-i] both defined
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        # compute contributions
        prod = a[idx] * a[k_max - idx]
        # donor index (max contribution)
        donor = idx[np.argmax(prod)]
        receiver = idx[np.argmin(prod)]
        if donor == receiver:
            # cannot improve
            break
        # compute step factor based on elapsed time
        elapsed = time.time() - start
        factor = init_factor * (1 - elapsed / total) + final_factor * (elapsed / total)
        # bound factor between final_factor and init_factor
        factor = max(final_factor, min(init_factor, factor))
        max_delta = a[donor] * factor
        if max_delta <= 0:
            break
        delta = max_delta

        # Compute new convolution after transfer
        b_new = b.copy()
        # linear contributions
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic terms
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        if new_val < best_val - 1e-12:
            # Accept move
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            best_val = new_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Deterministic flatten improved score={best_val:.6f}")

        else:
            # try reduced delta
            improved = False
            for factor_red in (0.5, 0.25, 0.125, 0.0625):
                delta2 = delta * factor_red
                if delta2 <= 1e-15:
                    break
                b_new2 = b.copy()
                b_new2[donor:donor + n] -= 2.0 * delta2 * a
                b_new2[receiver:receiver + n] += 2.0 * delta2 * a
                b_new2[2 * donor] += delta2 * delta2
                b_new2[2 * receiver] += delta2 * delta2
                if donor != receiver:
                    b_new2[donor + receiver] -= 2.0 * delta2 * delta2
                new_max2 = b_new2.max()
                new_val2 = float(2.0 * n * new_max2 / (target_sum * target_sum))
                if new_val2 < best_val - 1e-12:
                    a[donor] -= delta2
                    a[receiver] += delta2
                    b = b_new2
                    best_val = new_val2
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Deterministic flatten (scaled) improved score={best_val:.6f}")
                    improved = True
                    break
            if not improved:
                # If no improvement possible with this donor/receiver pair, random jitter to break
                # we can break out to avoid infinite loops
                break

"""
Propose a step‑function minimizing 2·n·max(a*a)/(sum(a)²).

Algorithm
---------
1. **Warm‑start** – load the best sequence found in a previous run
   (`height_sequence_1`) and resize it, otherwise fall back to a simple power‑law.

2. **Multiplicative (exponential) descent** – mirror‑descent on a soft‑max
   surrogate of the maximal auto‑convolution.  The surrogate gets sharper
   (β ↑) while the iterate is projected onto the simplex Σa = √(2n).

3. **Projected gradient descent with restarts** – several short runs of
   momentum‑gradient descent on the same surrogate, each started from a
   jittered copy of the current best solution.

4. **Exact local refinement** – simulated‑annealing style mass transfers
   (donor ↔ receiver) with O(n) convolution updates; accepts improving or
   occasionally uphill moves.

5. **Deterministic flattening** – repeatedly identify the shift k where the
   convolution attains its maximum, then move a fraction of mass from the
   index that contributes most to this entry to the index that contributes
   least.  The convolution is updated exactly, guaranteeing monotone
   improvement (or a scaled back step).

6. **Final clean‑up** – truncate trailing zeros, evaluate the sequence with
   the supplied ``evaluate_sequence`` and store it for future warm‑starts.

The routine respects the 1000 s budget, the non‑negativity constraint and the
ℓ¹‑norm Σa = √(2n).  All heavy linear‑algebra uses NumPy/FFT; the deterministic
flattening runs in O(n) per iteration and often yields the final few
percent‑point improvements needed to beat the target score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  For Σa=√(2n) this equals max(a*a)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, Σx = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max surrogate utilities
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where conv is the full convolution a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Stage 2 – exponential (multiplicative) descent
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Mirror‑descent (multiplicative) on the soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

# Stage 3 – projected gradient descent with restarts
# ----------------------------------------------------------------------


def _annealed_gradient_descent_v2(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the soft‑max surrogate."""
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    step_base = target_sum * 0.45
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.93
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0015 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GDv2 it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several short GD restarts with random jitter; keep the best."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_v2(start,
                                               time.time() + run_budget,
                                               target_sum)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val

# ----------------------------------------------------------------------


def _local_refine(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """Exact hill‑climbing / simulated‑annealing on the true max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.04
    last_recalc = start

    while time.time() < deadline:
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        prod = a[idx] * a[k_max - idx]
        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        weight_all = np.zeros(n, dtype=a.dtype)
        weight_all[lo:hi + 1] = prod
        thresh = np.percentile(weight_all, 30)
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.04, target_sum * 0.0008)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        b_new = b.copy()
        b_new[donor:donor + n] += -2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Local refine improved score={best_val:.6f}")

        if time.time() - last_recalc > 12.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_recalc = time.time()
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑normalisation improved score={best_val:.6f}")

        if (int(elapsed * 100) % 2500) == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.005, target_sum * 0.0004)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 6 – deterministic flattening (exact maximal reduction)
# ----------------------------------------------------------------------


def _deterministic_flatten(a0: np.ndarray,
                           target_sum: float,
                           deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a fraction of mass from the index that contributes most
    to the current maximal convolution entry to the index that contributes
    least.  The convolution is updated exactly in O(n) per move.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    best_a = a.copy()
    best_val = _score(a)

    start = time.time()
    total = deadline - start
    init_factor = 0.25   # start moving up to 25 % of the donor mass
    final_factor = 0.0005  # shrink to a tiny step near the end

    while time.time() < deadline:
        k_max = int(b.argmax())
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contributions of each pair (i, k_max-i) to the maximal entry
        prod = a[idx] * a[k_max - idx]
        donor = idx[np.argmax(prod)]
        receiver = idx[np.argmin(prod)]
        if donor == receiver:
            break

        # step size that fades linearly with time
        elapsed = time.time() - start
        factor = init_factor * (1.0 - elapsed / total) + final_factor * (elapsed / total)
        factor = max(final_factor, min(init_factor, factor))
        max_delta = a[donor] * factor
        if max_delta <= 0:
            break
        delta = max_delta

        # exact update of the convolution after moving delta
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        if new_val < best_val - 1e-12:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            best_val = new_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Deterministic flatten improved score={best_val:.6f}")
        else:
            # try smaller scalings of the same move
            improved = False
            for scale in (0.5, 0.25, 0.125, 0.0625):
                delta2 = delta * scale
                if delta2 <= 1e-15:
                    break
                b_tmp = b.copy()
                b_tmp[donor:donor + n] -= 2.0 * delta2 * a
                b_tmp[receiver:receiver + n] += 2.0 * delta2 * a
                b_tmp[2 * donor] += delta2 * delta2
                b_tmp[2 * receiver] += delta2 * delta2
                if donor != receiver:
                    b_tmp[donor + receiver] -= 2.0 * delta2 * delta2
                val_tmp = float(2.0 * n * b_tmp.max() / (target_sum * target_sum))
                if val_tmp < best_val - 1e-12:
                    a[donor] -= delta2
                    a[receiver] += delta2
                    b = b_tmp
                    best_val = val_tmp
                    best_a = a.copy()
                    improved = True
                    print(f"[{time.time():.2f}] Deterministic flatten (scaled) improved score={best_val:.6f}")
                    break
            if not improved:
                # no improvement possible with this pair – give up on this shift
                break

        # periodic full recomputation to avoid numerical drift
        if time.time() - start > 40.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑normalisation during flatten improved score={best_val:.6f}")
            start = time.time()

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget and returns the best sequence found.
    """
    start = time.time()
    deadline = start + budget_s - 0.5        # safety margin for evaluator

    # --------------------------------------------------------------
    # Choose problem size (larger n gives a larger search space)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 500:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Warm‑start
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(
            np.array(globals()["height_sequence_1"], dtype=np.float64), n
        )
    else:
        init = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))

    init = init * (target_sum / init.sum())

    # --------------------------------------------------------------
    # Stage 2 – exponential (multiplicative) descent
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    exp_deadline = time.time() + 0.15 * remaining
    a_exp, val_exp = _exponential_descent(init, target_sum, exp_deadline)
    best_seq, best_score = a_exp, val_exp
    print(f"[{time.time():.2f}] After exponential descent best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – projected gradient descent with restarts
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        gd_deadline = time.time() + 0.35 * remaining
        a_gd, val_gd = _gradient_descent_with_restarts(
            best_seq, target_sum, gd_deadline
        )
        if val_gd < best_score:
            best_seq, best_score = a_gd, val_gd
        print(f"[{time.time():.2f}] After GD restarts best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 5 – exact local refinement
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        a_loc, val_loc = _local_refine(best_seq, target_sum, time.time() + 0.30 * remaining)
        if val_loc < best_score:
            best_seq, best_score = a_loc, val_loc
        print(f"[{time.time():.2f}] After local refinement best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 6 – deterministic flattening
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        a_flat, val_flat = _deterministic_flatten(best_seq, target_sum, deadline)
        if val_flat < best_score:
            best_seq, best_score = a_flat, val_flat
        print(f"[{time.time():.2f}] After deterministic flatten best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional final short local search (catch any remaining wiggle room)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        a_final, val_final = _local_refine(best_seq, target_sum, deadline)
        if val_final < best_score:
            best_seq, best_score = a_final, val_final
        print(f"[{time.time():.2f}] Final local refinement best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Clean up trailing zeros and final evaluation
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for warm‑starts in subsequent calls
    globals()["height_sequence_1"] = final_seq

    return final_seq
