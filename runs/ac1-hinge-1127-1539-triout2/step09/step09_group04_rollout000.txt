def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a (self‑convolution)."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²)."""
    n = a.size
    s = a.sum()
    if s < 0.01:
        return np.inf
    max_b = _max_convolution_fft(a)
    return float(2.0 * n * max_b / (s * s))


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Seq a_i = (i+1)^(-p) normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _grid_param_search(small_n: int,
                       target_sum: float,
                       deadline: float) -> tuple[np.ndarray, float]:

    Search the two‑parameter family
        a_i = (i+1)^(-p) * exp(-λ·i/small_n)

    """
    p_vals = np.linspace(0.2, 1.2, 17)

            seq = (i + 1) ** (-p) * np.exp(-lam * i / small_n)
            seq = seq * (target_sum / seq.sum())
            val = _score(seq)
            if val < best_score:

                best_seq = seq.copy()
                print(f"[{time.time():.2f}] Grid best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")
        if time.time() > deadline:

    while time.time() < deadline:

            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Random best p={p:.3f} λ={lam:.3f} score={best_score:.6f}")

def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(a*a).
    Returns (grad, conv) where conv = a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * reverse(a) convolve p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Stage 2 – exponential mirror descent (multiplicative updates)
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Mirror‑descent with multiplicative updates using a soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 16384.0
    base_eta = 0.04
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 3 – Adam‑style exponential descent
# ----------------------------------------------------------------------


def _adam_exponential_descent(a0: np.ndarray,
                             target_sum: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    """Adam‑like adaptive multiplicative updates, keeping positivity."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    beta = 1.0
    max_beta = 16384.0

    while time.time() < deadline:
        t += 1
        # soft‑max gradient
        g, _ = _gradient_softmax(a, beta)
        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1**t)
        v_hat = v / (1.0 - beta2**t)

        # learning rate schedule (decreasing)
        lr = 0.5 * target_sum / (math.sqrt(t) + 1.0)

        # multiplicative update
        a = a * np.exp(-lr * m_hat / (np.sqrt(v_hat) + eps))
        a = a * (target_sum / a.sum())

        # occasional jitter to avoid stagnation
        if t % 2000 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamExp it={t} β={beta:6.1f} score={best_val:.6f}")

        # increase β periodically (helps focus on the true max)
        if t % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 4 – projected gradient descent with momentum (soft‑max)
# ----------------------------------------------------------------------


def _proj_gradient_descent(a0: np.ndarray,
                          target_sum: float,
                          deadline: float) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the soft‑max surrogate with momentum."""
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 16384.0
    step_base = target_sum * 0.5
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.95
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.002 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] PGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 5 – exact sub‑gradient descent with simple back‑tracking line‑search
# ----------------------------------------------------------------------


def _subgradient_line_search(a0: np.ndarray,
                             target_sum: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true maximal convolution entry.
    Uses a tiny back‑tracking line‑search to guarantee monotone improvement.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())

        # sub‑gradient of the maximal entry
        grad = np.zeros_like(a)
        lo = max(0, k_max - (a.shape[0] - 1))
        hi = min(k_max, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad[idx] = 2.0 * a[k_max - idx]

        # safe step bound to preserve non‑negativity
        pos = grad > 0
        if np.any(pos):
            safe_eta = np.min(a[pos] / grad[pos])
        else:
            safe_eta = np.inf

        # try a few decreasing factors until improvement is seen
        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125):
            eta = min(0.5 * safe_eta, 0.5 / np.sqrt(it + 1)) * factor
            if eta <= 0:
                continue
            a_candidate = a - eta * grad
            a_candidate = _project_to_simplex(a_candidate, target_sum)
            val = _score(a_candidate)
            if val < best_val:
                best_val = val
                best_a = a_candidate.copy()
                a = a_candidate
                improved = True
                print(f"[{time.time():.2f}] Subgrad LS it={it:5d} factor={factor:.3f} score={best_val:.6f}")
                break

        if not improved:
            # occasional jitter to escape flat regions
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n gives more flexibility)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 14000
    else:
        n = 10000

    target_sum = math.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Initial sequence (warm–start from previous run if available)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
        init = init * (target_sum / init.sum())
    else:
        init = _power_law_sequence(n, p=0.55, target_sum=target_sum)

    # --------------------------------------------------------------
    # Stage 1 – cheap parametric search on a reduced scale
    # --------------------------------------------------------------
    small_n = max(2000, n // 5)
    small_target = math.sqrt(2.0 * small_n)
    param_deadline = time.time() + min(20.0, (deadline - time.time()) * 0.08)
    best_small, _ = _grid_param_search(small_n, small_target, param_deadline)
    if best_small is not None:
        init = _resize_sequence(best_small, n)
        init = init * (target_sum / init.sum())
        print(f"[{time.time():.2f}] Warm‑start from parametric search (n={n})")

    # --------------------------------------------------------------
    # Stage 2 – exponential mirror descent
    # --------------------------------------------------------------
    if time.time() < deadline:
        exp_deadline = time.time() + (deadline - time.time()) * 0.15
        a_exp, val_exp = _exponential_descent(init, target_sum, exp_deadline)
        best_seq = a_exp
        best_score = val_exp
        print(f"[{time.time():.2f}] After ExpGD best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – Adam‑style exponential descent
    # --------------------------------------------------------------
    if time.time() < deadline:
        adam_deadline = time.time() + (deadline - time.time()) * 0.25
        a_adam, val_adam = _adam_exponential_descent(best_seq, target_sum, adam_deadline)
        if val_adam < best_score:
            best_seq = a_adam
            best_score = val_adam
        print(f"[{time.time():.2f}] After AdamExp best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – projected gradient descent with momentum (soft‑max)
    # --------------------------------------------------------------
    if time.time() < deadline:
        pgd_deadline = time.time() + (deadline - time.time()) * 0.25
        a_pgd, val_pgd = _proj_gradient_descent(best_seq, target_sum, pgd_deadline)
        if val_pgd < best_score:
            best_seq = a_pgd
            best_score = val_pgd
        print(f"[{time.time():.2f}] After PGD best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 5 – exact sub‑gradient descent with line‑search
    # --------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = deadline
        a_sub, val_sub = _subgradient_line_search(best_seq, target_sum, sub_deadline)
        if val_sub < best_score:
            best_seq = a_sub
            best_score = val_sub
        print(f"[{time.time():.2f}] After Subgrad LS best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final clean‑up
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    # trim trailing near‑zero entries
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # Final evaluation (for logging)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # Store for warm‑starts in later calls
    globals()["height_sequence_1"] = final_seq

    return final_seq
