"""
Hybrid optimizer with LP‑refinement.

The evaluation is scale‑invariant, therefore we keep the ℓ1‑norm equal to
√(2·n) and minimise the maximal entry of the self‑convolution.
The algorithm consists of:

1.  Fast random exploration of several parametric families.
2.  Linear‑interpolation to longer lengths and cheap selection.
3.  Several short Adam runs on a τ‑soft‑max surrogate (mirror descent on
    the simplex).
4.  Sub‑gradient descent on the *hard* maximum.
5.  One or two iterations of a linear‑program that maximises the total
    mass while keeping the convolution with the current iterate bounded.
    The LP step pushes mass towards the edges, which is known to lower the
    peak.
6.  Very cheap stochastic hill‑climb.
The whole pipeline is bounded by a 1000 s budget and stores the final
candidate in the global variable ``height_sequence_1``.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, scaling, etc.)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a∗a."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# τ‑soft‑max gradient (used by Adam)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient: 2*(mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser (mirror descent on the simplex)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex (log‑softmax parametrisation).
    Minimises the *hard* maximum of the convolution by a slowly
    decaying τ‑soft‑max.
    Returns the best sequence found and its (hard) peak value.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                     # ℓ1‑norm we must keep

    # initialise logits from the feasible a0 (entropic parametrisation)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                         # logits (log‑softmax)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.45                                # aggressive start

    # temperature schedule – start relatively large for smoothness
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 5.0)                 # a few units of smoothing

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ---- recover a from logits (log‑softmax) ----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ---- convolution & gradient (τ‑soft‑max) ----
        g, peak = _softmax_gradient(a, tau)

        # keep the best hard peak seen so far
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ---- gradient w.r.t. logits (chain rule) ----
        mu = np.dot(a, g) / S                 # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)                 # shape (n,)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 6.0 * S:
            grad_p = grad_p * (6.0 * S / clip_val)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- schedules (very slow decay) ----
        tau *= 0.99995
        lr *= 0.99995

        # occasional jitter to escape plateaus
        if it % 800 == 0:
            p += rng.normal(scale=0.006, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Sub‑gradient descent on the *hard* max (no smoothing)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple sub‑gradient descent that always follows the entry of the
    convolution attaining the current maximum.  After each step the
    vector is projected back onto the simplex (positivity + fixed ℓ1‑norm).
    Returns the best sequence seen and its hard peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    a = a0.copy()
    best_a = a.copy()
    best_val = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # current convolution and its maximiser
        b = _full_convolution_fft(a)
        k = int(b.argmax())
        val = b[k]

        if val < best_val:
            best_val = val
            best_a = a.copy()

        # sub‑gradient of (a*a)_k :  g_i = 2 * a_{k-i} (if inside range)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        jdx = k - idx
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[jdx]

        # step size decreasing like 1/√t
        step = 0.8 * S / (np.sqrt(it) + 1.0)

        a = a - step * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, S / n)
        else:
            a = a * (S / s)

        # occasional random jitter to avoid tiny plateaus
        if it % 500 == 0:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                frac = rng.uniform(0.02, 0.12)
                amount = a[i] * frac
                a[i] -= amount
                a[j] += amount
                a = a * (S / a.sum())

    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑program refinement (maximise total mass under current bound)
# ----------------------------------------------------------------------
def _lp_refine(a0: np.ndarray,
               deadline: float,
               rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    One‑shot LP refinement: given the current shape a0, solve the LP

        maximise   Σ g_j
        subject to   (a0 * g)_k ≤ M   for all k,
                     g_j ≥ 0,

    where M = max (a0 * a0).  The optimal g is then mixed with a0.
    The procedure is repeated a few times (while time permits) to push
    mass towards the edges and lower the peak.
    Returns the best sequence seen and its peak value.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    a = a0.copy()
    best_a = a.copy()
    best_M = _max_convolution(a)

    it = 0
    while time.time() < deadline and it < 4:
        it += 1
        M = _max_convolution(a)

        # Build convolution matrix A (size (2n‑1) × n)  A @ g = a * g
        col = np.concatenate([a, np.zeros(n - 1, dtype=np.float64)])
        row = np.concatenate([np.array([a[0]], dtype=np.float64),
                              np.zeros(2 * n - 2, dtype=np.float64)])
        A = toeplitz(col, row)          # dense; acceptable for moderate n

        # Linear programme
        c = -np.ones(n)                 # maximise Σ g  → minimise -Σ g
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})

        if not res.success:
            print(f"[lp] iteration {it}: linprog failed ({res.message})")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            break

        # Scale to the correct ℓ1‑norm
        g = g * (S / sum_g)

        # Mix a little bit towards g (α chosen randomly)
        alpha = rng.uniform(0.25, 0.45)
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new * (S / a_new.sum())

        new_M = _max_convolution(a_new)
        if new_M < best_M - 1e-8:
            best_M = new_M
            best_a = a_new.copy()
            print(f"[lp] iteration {it} improved M={best_M:.6f}")

        a = a_new

    return best_a, best_M

def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    raw = mix * lin + (1.0 - mix) * base
    return raw

def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    With probability ≈0.2 we warm‑start from ``warm_seq`` (scaled).
    Otherwise we draw from one of the random families.
    The returned vector satisfies sum = sqrt(2·n) and all entries ≥0.
    """
    S = np.sqrt(2.0 * n)

    if warm_seq is not None and rng.random() < 0.23:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
        return a

    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'gamma', 'uniform', 'dirichlet', 'ushape', 'cosine'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'cosine':
        raw = _init_cosine(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence as a plain list and
    stores it in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Warm‑start handling
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # ------------------------------------------------------------------
    # Phase 1 – fast random exploration (≈ 20 % of budget)
    # ------------------------------------------------------------------
    explore_deadline = end_time - 380.0   # keep ~380 s for later phases
    best_seq = None
    best_score = np.inf
    best_n = None
    attempts = 0

    while time.time() < explore_deadline:
        attempts += 1
        # favour a few‑thousand entries but keep variety
        if rng.random() < 0.70:
            n = int(rng.integers(1800, 6000))
        else:
            n = int(rng.integers(6000, 12000))

        a0 = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a0)

        if peak < best_score:
            best_score = peak
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempts:4d} n={n:5d}  new best={best_score:.6f}")

    if best_seq is None:
        # fallback – uniform vector of moderate length
        best_n = 3000
        best_seq = np.full(best_n, np.sqrt(2.0 * best_n) / best_n)
        best_score = _max_convolution(best_seq)

    # ------------------------------------------------------------------
    # Phase 2 – upscale to several larger lengths (no optimisation)
    # ------------------------------------------------------------------
    cand_lengths = [best_n]
    for factor in [1.2, 1.4, 1.6, 2.0, 3.0]:
        cand = int(best_n * factor)
        if 2000 <= cand <= 25000:
            cand_lengths.append(cand)
    cand_lengths = sorted(set(cand_lengths))

    cur_seq = best_seq
    cur_score = best_score
    cur_n = best_n

    for L in cand_lengths:
        if time.time() > end_time - 260.0:
            break
        a_scaled = _scale_sequence(cur_seq, L)
        a_scaled = np.clip(a_scaled, 0.0, None)
        S = np.sqrt(2.0 * L)
        a_scaled = a_scaled * (S / a_scaled.sum())
        peak = _max_convolution(a_scaled)

        if peak < cur_score:
            cur_score = peak
            cur_seq = a_scaled.copy()
            cur_n = L
            print(f"[scale] length={L:5d}  improved max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam optimisation (several short restarts)
    # ------------------------------------------------------------------
    remaining = end_time - time.time() - 150.0   # reserve time for later phases
    if remaining > 20.0:
        n_restart = max(1, int(remaining // 70))   # ~1‑minute per restart
        per_run = remaining / n_restart
        for r in range(n_restart):
            if time.time() > end_time - 150.0:
                break
            # add tiny random jitter before each Adam run
            noise = rng.normal(scale=0.015 * np.sqrt(2.0 * cur_n) / cur_n,
                               size=cur_n)
            a_start = np.maximum(cur_seq + noise, 0.0)
            a_start = a_start * (np.sqrt(2.0 * cur_n) / a_start.sum())
            a_opt, peak = _adam_optimize(a_start,
                                          time.time() + per_run,
                                          rng)
            if peak < cur_score:
                cur_score = peak
                cur_seq = a_opt.copy()
                print(f"[adam] restart={r} new best max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – sub‑gradient descent (polishing)
    # ------------------------------------------------------------------
    remaining = end_time - time.time() - 70.0
    if remaining > 30.0:
        a_opt, peak = _subgradient_descent(cur_seq,
                                           time.time() + remaining,
                                           rng)
        if peak < cur_score:
            cur_score = peak
            cur_seq = a_opt.copy()
            print(f"[subgrad] polished max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – LP refinement (push mass to the edges)
    # ------------------------------------------------------------------
    remaining = end_time - time.time() - 20.0
    # apply LP only if n is modest (memory) and we still have time
    if remaining > 15.0 and cur_n <= 2000:
        a_opt, peak = _lp_refine(cur_seq,
                                 time.time() + remaining,
                                 rng)
        if peak < cur_score:
            cur_score = peak
            cur_seq = a_opt.copy()
            print(f"[lp] final refined max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – cheap random hill‑climb (few seconds)
    # ------------------------------------------------------------------
    while time.time() < end_time - 1.0:
        a = cur_seq.copy()
        n = a.shape[0]
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        frac = rng.uniform(0.02, 0.12)
        amount = a[i] * frac
        a[i] -= amount
        a[j] += amount
        peak = _max_convolution(a)
        if peak + 1e-9 < cur_score:   # strict improvement
            cur_score = peak
            cur_seq = a.copy()
            print(f"[hill] improved max={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    final_seq = cur_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f}  internal max={cur_score:.6f}  length={len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
