"""
Candidate sequence optimizer: combines multiplicative descent,
projected gradient descent, a greedy large‑step mass‑transfer,
and a fine‑grained local refinement.  The algorithm respects the
non‑negative, ℓ¹‑norm (Σa = √(2n)) constraints required by
``evaluate_sequence`` and aims to beat the previous best bound
≈1.508.
"""
import time
import numpy as np

# ---------- convolution utilities ----------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the convolution a*a."""
    return float(_conv_fft(a).max())

def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)².  For Σa = √(2n) this equals max(a*a)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))

# ---------- simplex projection ----------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx=s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq)

# ---------- soft‑max gradient ----------
def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where conv is the full convolution a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b

# ---------- phase 2 : exponential (mirror) descent ----------
def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 8e-4, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")
        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)
        it += 1
    return best_a, best_val

# ---------- phase 3 : projected gradient descent with restarts ----------
def _annealed_gradient_descent_v2(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    step_base = target_sum * 0.45
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.93
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0015 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GDv2 it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)
        it += 1
    return best_a, best_val

def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several short GD restarts with random jitter; keep the best."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_v2(start,
                                               time.time() + run_budget,
                                               target_sum)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val

# ---------- phase 4 : greedy large‑step refinement ----------
def _large_step_refine(a0: np.ndarray,
                      target_sum: float,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Deterministic “big move” refinement.
    Repeatedly selects a high‑contributing donor index and a low‑contributing
    receiver, then computes the maximal admissible transfer that does not push
    any convolution entry above the current second‑largest value.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    while time.time() < deadline:
        # identify current worst (max) convolution entry and the runner‑up
        k_max = int(np.argmax(b))
        M = b[k_max]

        # second largest value
        temp = b.copy()
        temp[k_max] = -np.inf
        M2 = temp.max()
        if np.isinf(M2):
            M2 = M   # all entries equal – we are done

        # build donor/receiver candidate sets based on contribution to k_max
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        contrib = a[idx] * a[k_max - idx]

        # donor candidates – weighted by contribution
        if contrib.sum() > 0:
            donor_cands = rng.choice(idx,
                                     size=min(5, idx.size),
                                     replace=False,
                                     p=contrib / contrib.sum())
        else:
            donor_cands = rng.choice(idx,
                                     size=min(5, idx.size),
                                     replace=False)

        # receiver candidates – low contribution tail
        thresh = np.percentile(contrib, 30)
        low_idx = idx[contrib <= thresh]
        if low_idx.size >= 5:
            recv_cands = rng.choice(low_idx,
                                    size=5,
                                    replace=False)
        else:
            extra = rng.choice(np.arange(n),
                               size=5 - low_idx.size,
                               replace=False)
            recv_cands = np.concatenate([low_idx, extra])

        improved = False
        for donor in donor_cands:
            for recv in recv_cands:
                if donor == recv:
                    continue
                # maximal amount we could move (cannot exceed donor's mass)
                max_delta = min(a[donor], target_sum * 0.02)
                if max_delta <= 0:
                    continue

                # effect vector d on convolution (linear part only)
                d = np.zeros_like(b)
                d[donor:donor + n] -= 2.0 * a
                d[recv:recv + n] += 2.0 * a

                # compute admissible delta so that no entry exceeds M2
                pos = d > 1e-14
                if pos.any():
                    # safe bound from each positive entry
                    delta_bound = np.min((M2 - b)[pos] / d[pos])
                    delta = min(max_delta, delta_bound, a[donor])
                else:
                    delta = max_delta

                if delta <= 1e-12:
                    continue

                # apply move to convolution (including quadratic terms)
                b_new = b.copy()
                b_new[donor:donor + n] -= 2.0 * delta * a
                b_new[recv:recv + n] += 2.0 * delta * a
                b_new[2 * donor] += delta * delta
                b_new[2 * recv] += delta * delta
                if donor != recv:
                    b_new[donor + recv] -= 2.0 * delta * delta

                new_max = b_new.max()
                new_val = float(2.0 * n * new_max / (target_sum * target_sum))

                if new_val < best_val - 1e-9:
                    # accept the move
                    a_old = a.copy()
                    a[donor] -= delta
                    a[recv] += delta
                    b = b_new
                    best_val = new_val
                    best_a = a.copy()
                    improved = True
                    print(f"[{time.time():.2f}] Large step donor={donor} recv={recv} delta={delta:.6f} val={best_val:.6f}")
                    break
            if improved:
                break

        if not improved:
            # no improving big move found – fall back to a small random jitter
            # This keeps the algorithm alive when we get stuck.
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.003, target_sum * 0.0005)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter after stall improved val={best_val:.6f}")
    return best_a, best_val

# ---------- phase 5 : fine‑grained local refinement ----------
def _local_refine(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """Exact hill‑climbing / simulated‑annealing on the true max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.04
    last_recalc = start

    while time.time() < deadline:
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        prod = a[idx] * a[k_max - idx]

        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        weight_all = np.zeros(n, dtype=a.dtype)
        weight_all[lo:hi + 1] = prod
        thresh = np.percentile(weight_all, 30)
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.04, target_sum * 0.0008)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        b_new = b.copy()
        b_new[donor:donor + n] += -2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()

"""
Candidate sequence optimizer.

The algorithm consists of four stages

1. **Exponential (mirror) descent** on a soft‑max surrogate of the maximal
   convolution entry (fast, multiplicative updates).

2. **Projected gradient descent with restarts** (the same surrogate,
   momentum, jitter, and a slowly increasing temperature β).

3. **Greedy large‑step refinement** – repeatedly picks a “donor’’ index that
   contributes heavily to the current maximal convolution entry and a “receiver’’
   index with a low contribution, then computes the maximal admissible mass
   transfer that keeps every convolution entry below the current second‑largest
   value.  The move is applied exactly (O(n) slice updates).

4. **Fine‑grained local refinement** – the original small‑step simulated‑annealing
   hill‑climber from the reference implementation.

All iterates respect non‑negativity and the ℓ¹‑norm constraint
∑a = √(2n) required by ``evaluate_sequence``.
The routine returns the best sequence found before the time budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)².  For Σa = √(2n) this equals max(a*a)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx=s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq)


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where conv is the full convolution a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase 2 – exponential (mirror) descent
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 8e-4, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

def _annealed_gradient_descent_v2(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the soft‑max surrogate."""
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    step_base = target_sum * 0.45
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.93
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0015 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GDv2 it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several short GD restarts with random jitter; keep the best."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_v2(start,
                                               time.time() + run_budget,
                                               target_sum)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val

def _large_step_refine(a0: np.ndarray,
                      target_sum: float,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Deterministic “big move’’ refinement.
    For the current maximal convolution entry we select a donor that
    contributes a lot and a receiver that contributes little, then move the
    largest admissible amount of mass without letting any entry exceed the
    current second‑largest value.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    while time.time() < deadline:
        k_max = int(np.argmax(b))
        M = b[k_max]

        # second‑largest convolution entry
        tmp = b.copy()
        tmp[k_max] = -np.inf
        M2 = tmp.max()
        if np.isinf(M2):
            M2 = M

        # indices that actually affect k_max
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        contrib = a[idx] * a[k_max - idx]

        # donor candidates (weighted by contribution)
        if contrib.sum() > 0:
            donor_cands = rng.choice(idx,
                                     size=min(5, idx.size),
                                     replace=False,
                                     p=contrib / contrib.sum())
        else:
            donor_cands = rng.choice(idx,
                                     size=min(5, idx.size),
                                     replace=False)

        # receiver candidates – low‑contribution tail
        thresh = np.percentile(contrib, 30)
        low_idx = idx[contrib <= thresh]
        if low_idx.size >= 5:
            recv_cands = rng.choice(low_idx, size=5, replace=False)
        else:
            extra = rng.choice(np.arange(n), size=5 - low_idx.size, replace=False)
            recv_cands = np.concatenate([low_idx, extra])

        improved = False
        for donor in donor_cands:
            for recv in recv_cands:
                if donor == recv:
                    continue
                max_delta = min(a[donor], target_sum * 0.02)
                if max_delta <= 0:
                    continue

                # linear effect of moving δ from donor → receiver on the convolution
                d = np.zeros_like(b)
                d[donor:donor + n] -= 2.0 * a
                d[recv:recv + n]   += 2.0 * a

                pos = d > 1e-14
                if pos.any():
                    delta_bound = np.min((M2 - b)[pos] / d[pos])
                    delta = min(max_delta, delta_bound, a[donor])
                else:
                    delta = max_delta

                if delta <= 1e-12:
                    continue

                # exact update of the convolution (including quadratic terms)
                b_new = b.copy()
                b_new[donor:donor + n] -= 2.0 * delta * a
                b_new[recv:recv + n]   += 2.0 * delta * a
                b_new[2 * donor] += delta * delta
                b_new[2 * recv]  += delta * delta
                if donor != recv:
                    b_new[donor + recv] -= 2.0 * delta * delta

                new_max = b_new.max()
                new_val = float(2.0 * n * new_max / (target_sum * target_sum))

                if new_val < best_val - 1e-9:
                    # accept the move
                    a[donor] -= delta
                    a[recv]   += delta
                    b = b_new
                    best_val = new_val
                    best_a = a.copy()
                    improved = True
                    print(f"[{time.time():.2f}] Large step donor={donor} recv={recv} "
                          f"δ={delta:.6f} val={best_val:.6f}")
                    break
            if improved:
                break

        if not improved:
            # tiny random jitter to avoid a dead‑end
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.003, target_sum * 5e-4)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved val={best_val:.6f}")

    return best_a, best_val

                                       deadline)
        if val_ref < best_score:
            best_seq = a_ref
            best_score = val_ref
        print(f"[{time.time():.2f}] After final local refinement best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Clean up and final evaluation
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
