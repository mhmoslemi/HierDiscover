def _fast_mass_transfer_refinement(a: np.ndarray,
                                  target_sum: float,
                                  deadline: float,
                                  low_frac: float = 0.05,
                                  jitter_interval: int = 5000,
                                  max_iter: int = None) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    conv = _conv_fft(a)
    conv = np.maximum(conv, 0.0)
    best_a = a.copy()
    best_max = conv.max()
    rng = np.random.default_rng()
    it = 0
    last_improve_it = 0
    while time.time() < deadline:
        # Determine donor/receiver
        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = kmax - idx
        contrib = a[idx] * a[partner]
        donor = idx[np.argmax(contrib)]
        # select receiver from low contributors
        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argpartition(contrib, low_cnt)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        # Determine delta
        max_frac = 0.5  # up to half of donor mass
        delta = a[donor] * rng.random() * max_frac
        if delta <= 0:
            it += 1
            continue

        # backup:
        a_old = a.copy()
        conv_old = conv.copy()
        cur_max = conv.max()

        # apply update using a_old reference
        di = -delta
        dj = delta
        # Update convolution slices using a_old
        conv[donor:donor + n] += 2.0 * di * a_old
        conv[receiver:receiver + n] += 2.0 * dj * a_old
        cross_idx = donor + receiver
        if 0 <= cross_idx < conv.shape[0]:
            conv[cross_idx] += 2.0 * di * dj
        conv[2 * donor] += di * di
        conv[2 * receiver] += dj * dj

        # update a
        a[donor] = a_old[donor] + di
        a[receiver] = a_old[receiver] + dj

        # Ensure non-negativity (should hold because di = -delta, a_old[donor] >= delta)
        # compute new max
        new_max = conv.max()

        if new_max < best_max:
            best_max = new_max
            best_a = a.copy()
            # optionally store best conv not needed
            last_improve_it = it

        if new_max < cur_max:
            # accept move (already updated)
            pass
        else:
            # reject: revert
            a[:] = a_old
            conv[:] = conv_old

        # occasional jitter if no improvement for many iterations
        if it - last_improve_it > jitter_interval:
            # jitter perturbation
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter

                best_a = a.copy()
            last_improve_it = it

        it += 1

    best_max_val = best_max

def propose_candidate(budget_s=1000, **kwargs):
    start = time.time()
    deadline = start + budget_s - 0.5

    if budget_s >= 800:
        n = 15000   # bigger
    elif budget_s >= 400:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)

    # Build init candidates
    init_seqs = []

    # power law list
    for p in (0.45, 0.5, 0.55, 0.6, 0.65):
        init_seqs.append(_power_law_sequence(n, p, target_sum))

    # previous best if exists
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        init_seqs.append(_resize_sequence(prev, n))

    # jittered variants
    rng = np.random.default_rng()
    for seq in list(init_seqs):
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        seq_jit = seq + jitter
        seq_jit = np.maximum(seq_jit, 0.0)
        seq_jit = _project_to_simplex(seq_jit, target_sum)
        init_seqs.append(seq_jit)

    # Phase 1: gradient descent restarts
    remaining = deadline - time.time()
    phase1_deadline = time.time() + 0.2 * budget_s

    best_seq = None
    best_score = np.inf

    for init in init_seqs:
        if time.time() >= phase1_deadline:
            break
        a_cand, val = _gradient_descent_with_restarts(init, target_sum, phase1_deadline)
        if val < best_score:
            best_score = val
            best_seq = a_cand.copy()
            print(f"[{time.time():.2f}] Phase1 new best score={best_score:.6f}")

    if best_seq is None:
        best_seq = _power_law_sequence(n, 0.55, target_sum)
        best_score = _score(best_seq)

    # Phase 2: fast mass transfer refinement
    if time.time() < deadline:
        phase2_deadline = time.time() + 0.6 * budget_s
        if phase2_deadline > deadline:
            phase2_deadline = deadline
        a_ref, max_val = _fast_mass_transfer_refinement(best_seq, target_sum, phase2_deadline)
        if max_val < best_score:
            best_score = max_val
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] Phase2 mass‑transfer improved best={best_score:.6f}")

    # Phase 3: final polishing gradient descent
    if time.time() < deadline:
        a_fin, val_fin = _annealed_gradient_descent_v2_ext(
            best_seq,
            deadline,
            target_sum,
            max_beta=3e5,
            step_factor=0.6,
            gamma=0.995,
        )
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Phase3 final polish best={best_score:.6f}")

    # Trim trailing zeros
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[:nz[-1]+1]

    final_seq = arr.tolist()
    try:
        val = evaluate_sequence(final_seq)
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as e:
        print(f"Evaluation exception: {e}")
    globals()["height_sequence_1"] = final_seq

    return final_seq

#!/usr/bin/env python
"""
Propose a low‑score step function.

The algorithm is a three‑phase pipeline:

1. **Projected gradient descent** – a few gradient‑descent restarts
   (soft‑max approximation of the max‑convolution) to obtain a
   decent starting point.
2. **Fast greedy mass‑transfer** – many cheap O(n) updates that move
   a small amount of mass from a “donor’’ index (large contribution
   to the current maximal convolution entry) to a “receiver’’ index
   (small contribution).  The convolution is updated incrementally
   and only improvements (or occasional jitter) are kept.
3. **Final polishing** – a high‑β projected gradient descent pass
   to clean up any remaining irregularities.

The whole search respects the 1000 s budget, stores the best sequence
in the global variable ``height_sequence_1`` and finally returns it as
a list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Benchmark objective: 2·n·max(a*a) / (Σa)² (with Σa=√(2n) this = max)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Gradient‑descent utilities (soft‑max approximation)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient: 2 * (a reversed)⋆p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent_v2_ext(a0: np.ndarray,
                                     deadline: float,
                                     target_sum: float,
                                     max_beta: float = 1e5,
                                     step_factor: float = 0.7,
                                     gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max approximation.
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD_ext it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Several short gradient‑descent runs with random jitter.
    Keeps the best result found before the given deadline.
    """
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_v2_ext(
            start,
            time.time() + run_budget,
            target_sum,
            max_beta=1e5,
            step_factor=0.8,
            gamma=0.99,
        )
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Fast greedy mass‑transfer refinement (O(n) updates)
# ----------------------------------------------------------------------


def _fast_mass_transfer_refinement(a: np.ndarray,
                                   target_sum: float,
                                   deadline: float,
                                   low_frac: float = 0.05,
                                   jitter_interval: int = 3000) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer moves that directly target the current maximal
    convolution entry.  Updates are performed with O(n) slice operations,
    allowing many more iterations than the FFT‑based version.
    """
    n = a.shape[0]
    conv = _conv_fft(a)
    conv = np.maximum(conv, 0.0)

    best_a = a.copy()
    best_max = conv.max()
    rng = np.random.default_rng()
    it = 0
    last_improve = 0

    while time.time() < deadline:
        cur_max = conv.max()
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = kmax - idx
        contrib = a[idx] * a[partner]

        donor = idx[np.argmax(contrib)]
        # pick a receiver among the lowest contributors
        receiver = idx[np.argmin(contrib)]

        if donor == receiver:
            it += 1
            continue

        # amount to move – larger early on, smaller later
        elapsed = time.time() - (deadline - (deadline - time.time()))
        fraction = (time.time() - (deadline - (deadline - time.time()))) / (deadline - (deadline - time.time()) + 1e-12)
        # simple schedule – cap at 0.5 of donor mass
        max_frac = 0.5 * (1.0 - fraction) + 0.01
        max_frac = min(max_frac, 0.5)
        delta = a[donor] * rng.random() * max_frac
        if delta <= 0.0:
            it += 1
            continue

        # backup copies (small, O(n))
        a_before = a.copy()
        conv_before = conv.copy()

        di = -delta
        dj = delta

        # incremental convolution update using a_before (the state before the move)
        conv[donor:donor + n] += 2.0 * di * a_before
        conv[receiver:receiver + n] += 2.0 * dj * a_before
        cross = donor + receiver
        if 0 <= cross < conv.shape[0]:
            conv[cross] += 2.0 * di * dj
        conv[2 * donor] += di * di
        conv[2 * receiver] += dj * dj

        # apply the mass transfer
        a[donor] = a_before[donor] + di
        a[receiver] = a_before[receiver] + dj

        new_max = conv.max()

        if new_max < best_max:
            best_max = new_max
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time():.2f}] MT iter={it:6d} new best={best_max:.6f}")

        if new_max <= cur_max:
            # keep the move
            pass
        else:
            # revert
            a[:] = a_before
            conv[:] = conv_before

        # occasional jitter to escape plateaus
        if it - last_improve > jitter_interval:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            conv = np.maximum(conv, 0.0)
            last_improve = it
            if conv.max() < best_max:
                best_max = conv.max()
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT jitter improved best={best_max:.6f}")

        it += 1

    return best_a, best_max


# ----------------------------------------------------------------------
# Simple re‑ordering heuristic (alternating large/small)
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Reorder by alternating largest‑left, second‑largest‑right, …."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left = 0
    right = n - 1
    left_turn = True
    for v in sorted_vals:
        if left_turn:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        left_turn = not left_turn
    return out
