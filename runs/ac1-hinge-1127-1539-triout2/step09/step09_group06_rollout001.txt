def _simulated_annealing(a, deadline, rng):
    best = a.copy()
    best_val = _max_convolution(a)
    cur = a.copy()
    cur_val = best_val
    # initial temperature
    T0 = best_val * 0.01  # small relative
    # We'll use exponential cooling
    start = time.time()
    while time.time() < deadline:
        # propose move
        n = cur.shape[0]
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if cur[i] < 1e-12:
            continue
        # fraction of mass to move
        frac = rng.uniform(0.001, 0.05)
        amount = cur[i] * frac
        cur[i] -= amount
        cur[j] += amount
        # ensure sum unchanged by construction
        # evaluate
        val = _max_convolution(cur)
        # acceptance
        if val < cur_val or rng.random() < np.exp(-(val - cur_val) / max(T0 * (1 - (time.time()-start)/(deadline-start)), 1e-12)):
            cur_val = val
            if val < best_val:
                best_val = val
                best = cur.copy()
        else:
            # revert
            cur[j] -= amount
            cur[i] += amount
        # occasionally random perturb
        if rng.random() < 0.001:
            # random jitter
            cur = best.copy()
            # small random noise
            noise = rng.normal(scale=0.01 * np.sqrt(2.0 * n)/n, size=n)
            cur = np.maximum(cur + noise, 0)
            cur = cur * (np.sqrt(2.0 * n)/cur.sum())
            cur_val = _max_convolution(cur)
    return best, best_val
```

"""
Step‑function optimizer – progressive length annealing + Adam + sub‑gradient +
simulated annealing.

The goal is to minimise 2·n·max(a*a) / (sum(a)^2), which after normalising the
ℓ1‑norm to √(2 n) equals the maximum entry of the self‑convolution.  The
algorithm works as follows:

1. **Random exploration** – generate many random feasible sequences of
   moderate size (2 k…15 k) and keep the best.
2. **Progressive refinement** – repeatedly up‑scale the current best
   sequence to a larger length and run a short Adam optimisation (soft‑max
   approximation of the max) followed by a few iterations of hard
   sub‑gradient descent.
3. **Simulated annealing** – with the remaining time perform a Metropolis
   walk that moves a small fraction of mass between two positions; uphill
   moves are occasionally accepted to escape shallow local minima.
4. **Final sanity check** – evaluate the obtained sequence with the provided
   ``evaluate_sequence`` and store it in the global variable
   ``height_sequence_1`` for warm‑starts in later runs.

All operations respect the ℓ₁‑norm constraint (sum = √(2·n)) and non‑negativity.
The code deliberately leaves a small safety margin before the budget expires
to avoid hard termination by the harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, scaling, etc.)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a∗a."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# τ‑soft‑max gradient (used by Adam)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser (mirror descent on the simplex)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex (log‑softmax parametrisation).
    Returns the best feasible vector seen and its (hard) peak value.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)               # target ℓ1‑norm
    # initialise logits from a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                   # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.45

    # initial temperature for the soft‑max
    tau = max(_full_convolution_fft(a0).max(), 5.0)

    best_a = a0.copy()
    best_peak = _max_convolution(a0)
    it = 0

    while time.time() < deadline:
        it += 1

        # recover a from logits (log‑softmax)
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # gradient of τ‑soft‑max proxy
        g, peak = _softmax_gradient(a, tau)

        # keep strict hard max improvement
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # chain rule to obtain gradient w.r.t. logits
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # optional gradient clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 6.0 * S:
            grad_p = grad_p * (6.0 * S / clip_val)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # schedules (very slow decay)
        tau *= 0.99995
        lr *= 0.99995

        # occasional jitter to escape plateaus
        if it % 800 == 0:
            p += rng.normal(scale=0.006, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Sub‑gradient descent on the *hard* max (no smoothing)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple sub‑gradient descent that follows the active entry of the
    convolution attaining the current maximum.  After each step the vector
    is projected back onto the simplex.
    Returns the best feasible vector seen and its (hard) peak value.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    a = a0.copy()
    best_a = a.copy()
    best_val = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1
        b = _full_convolution_fft(a)
        k = int(b.argmax())
        val = b[k]

        if val < best_val:
            best_val = val
            best_a = a.copy()

        # sub‑gradient of (a*a)_k
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        jdx = k - idx
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[jdx]

        # step size decreasing like 1/√t
        step = 0.8 * S / (np.sqrt(it) + 1.0)

        a = a - step * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, S / n)
        else:
            a = a * (S / s)

        # jitter every few iterations
        if it % 500 == 0:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                frac = rng.uniform(0.02, 0.12)
                amount = a[i] * frac
                a[i] -= amount
                a[j] += amount
                a = a * (S / a.sum())

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing on the hard max
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Metropolis walk that moves a small fraction of mass between two positions.
    Occasionally accepts uphill moves according to a slowly cooling temperature.
    Returns the best vector seen and its (hard) max convolution value.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    cur = a0.copy()
    cur_val = _max_convolution(cur)
    best = cur.copy()
    best_val = cur_val

    start = time.time()
    total = deadline - start
    # initial temperature set to a small fraction of the current peak
    T0 = cur_val * 0.04 if cur_val > 0 else 0.01

    while time.time() < deadline:
        # linear cooling
        elapsed = time.time() - start
        T = max(T0 * (1.0 - elapsed / total), 1e-12)

        i, j = rng.integers(0, n, size=2)
        if i == j or cur[i] <= 0.0:
            continue

        # move a random fraction of the mass at i to j
        frac = rng.uniform(0.001, 0.05)
        amount = cur[i] * frac
        cur[i] -= amount
        cur[j] += amount

        val = _max_convolution(cur)

        # Metropolis acceptance
        if val < cur_val or rng.random() < np.exp(-(val - cur_val) / T):
            cur_val = val
            if val < best_val:
                best_val = val
                best = cur.copy()
        else:
            # revert the move
            cur[j] -= amount
            cur[i] += amount

        # occasional global jitter to revive exploration
        if rng.random() < 0.001:
            cur = best.copy()
            noise = rng.normal(scale=0.015 * S / n, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur = cur * (S / cur.sum())
            cur_val = _max_convolution(cur)
            if cur_val < best_val:
                best_val = cur_val
                best = cur.copy()

    return best, best_val

def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    raw = mix * ush + (1.0 - mix) * base
    return raw


def _init_cosine(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    raw = 1.0 - np.cos(np.pi * i / (n - 1))
    raw = raw * rng.uniform(0.5, 2.0)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    return mix * raw + (1.0 - mix) * base


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.random(n)
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.05, 0.9)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    With probability ≈0.22 we warm‑start from ``warm_seq`` (scaled).
    Otherwise we draw from one of the random families.
    The returned vector satisfies sum = sqrt(2·n) and all entries ≥0.
    """
    S = np.sqrt(2.0 * n)

    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
        return a

    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'gamma', 'uniform', 'dirichlet', 'ushape', 'cosine'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'cosine':
        raw = _init_cosine(n, rng)
    elif style == 'uniform':
        raw = np.full(n, 1.0)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a
