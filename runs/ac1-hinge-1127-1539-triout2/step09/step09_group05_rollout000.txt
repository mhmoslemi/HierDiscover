"""
height_sequence_2 – “smooth‑max Adam” optimizer.

The evaluation function
    evaluate_sequence(seq) = max_k (seq * seq)[k]
when the sequence is normalised so that sum(seq)=sqrt(2·len(seq)).

Goal
    minimise the maximal self‑convolution value (the “peak” of the
    autocorrelation).  The code builds on the previous optimiser but
    replaces the hard‑max sub‑gradient by a smooth (log‑sum‑exp)
    surrogate and uses the Adam adaptive‑step optimiser.  A schedule
    of increasing β (the soft‑max temperature) drives the surrogate
    towards the true max while keeping the landscape smooth enough
    for fast progress.  After the Adam phase a short projected
    sub‑gradient refinement and a final simulated‑annealing hill‑climb
    are performed.

Main phases
-------------
1. Broad random exploration (multiple parametric families,
   lengths 3 000–20 000).  The best 5 candidates are kept.
2. For each retained candidate run Adam optimisation with a β‑schedule
   (β = 0.5 → 300).  After each β block the current best value is
   remembered.  The iterates are always projected onto the simplex
   {x≥0, Σx = √(2n)}.
3. One short projected sub‑gradient run (hard‑max) to polish the
   best solution.
4. A cheap simulated‑annealing mass‑transfer hill‑climb.
5. Return the final sequence and store it as ``height_sequence_2``.

All heavy work is done with FFT‑based convolutions (O(n log n)).
The algorithm respects the 1000 s time budget – the last few seconds
are reserved for the final sanity check.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a∗a."""
    return _full_convolution_fft(a).max()


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σx = S}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius S."""
    if S < 0:
        raise ValueError("Simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence scaling (linear interpolation)
# ----------------------------------------------------------------------
def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Parametric families for random start vectors
# ----------------------------------------------------------------------
def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)
    offset = rng.uniform(0.0, 20.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0003, 0.025)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    p = rng.uniform(0.4, 2.5)
    mix = rng.uniform(0.0, 1.0)
    ush = prod.astype(float) ** (-p)
    base = np.ones(n)
    raw = mix * ush + (1.0 - mix) * base
    return raw


def _init_sine(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n)
    vals = np.sin(np.pi * (i + 1) / (n + 1))
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * vals + (1.0 - mix) * base
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    return rng.random(n)


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Occasionally warm‑starts from the globally best sequence,
    otherwise draws from a random parametric family.
    """
    S = np.sqrt(2.0 * n)

    # Warm start with probability ≈0.22
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
        return a

    # Choose a family
    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'ushape', 'sine', 'uniform', 'dirichlet'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'ushape':
        raw = _init_ushape(n, rng)
    elif style == 'sine':
        raw = _init_sine(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) gradient -------------------------------------------------
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Compute the gradient of the soft‑max surrogate
        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)[k])
    and also return the exact hard max value of (a*a).
    """
    n = a.shape[0]

    # 1) self‑convolution (needed for both the value and the weights)
    conv = _full_convolution_fft(a)                     # length 2n‑1
    max_val = conv.max()                                # hard max for reporting

    # 2) soft‑max weights  w_k ∝ exp(β·conv_k)
    #   subtract max for numeric stability
    shifted = beta * (conv - max_val)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()                # length 2n‑1, sum = 1

    # 3) gradient = 2·(w ∗ a_rev)   (central window of length n)
    a_rev = a[::-1]
    conv_len = len(w) + n - 1          # = (2n‑1) + n – 1 = 3n‑2
    f_len = _next_pow2(conv_len)
    W = np.fft.rfft(w, n=f_len)
    A_rev = np.fft.rfft(a_rev, n=f_len)
    grad_full = np.fft.irfft(W * A_rev, n=f_len)[:conv_len]
    start = n - 1
    grad = 2.0 * grad_full[start:start + n]
    return grad, max_val


# ----------------------------------------------------------------------
# Adam optimiser (smooth‑max)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the soft‑max surrogate.
    A β‑schedule is used: start with a small β (smooth) and increase
    it gradually so that the final iterate approximates the true max.
    Every iterate is projected onto the simplex.
    Returns the best (lowest hard max) vector found.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam parameters
    lr = 0.04
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_val = _max_convolution(a)

    # β‑schedule (increasing smoothness)
    betas = [0.5, 2.0, 5.0, 15.0, 40.0, 120.0, 300.0]
    beta_idx = 0
    cur_beta = betas[beta_idx]

    # counters for adaptation
    last_improve_iter = 0

    while time.time() < deadline:
        t += 1
        # gradient of soft‑max at current β
        grad, cur_hard_max = _gradient_softmax(a, cur_beta)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a - step, S)

        # exact (hard) max for book‑keeping
        cur_val = _max_convolution(a)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            last_improve_iter = t

        # increase β if we have stalled for a while
        if t - last_improve_iter > 300:
            if beta_idx < len(betas) - 1:
                beta_idx += 1
                cur_beta = betas[beta_idx]
                last_improve_iter = t
                # small jitter to kick the optimiser out of a flat region
                jitter = rng.normal(scale=0.001 * S, size=n)
                a = _project_onto_simplex(a + jitter, S)

        # occasional tiny random perturbation (helps escaping plateaus)
        if t % 500 == 0:
            jitter = rng.normal(scale=0.001 * S, size=n)
            a = _project_onto_simplex(a + jitter, S)

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (hard‑max) – kept for final polishing
# ----------------------------------------------------------------------
def _max_convolution_and_gradient(a: np.ndarray):
    """
    Sub‑gradient of the hard max (mask of maximal entries) together
    with the hard‑max value.
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_val = b.max()
    mask = (b == max_val).astype(float)

    a_rev = a[::-1]
    conv_len = len(mask) + n - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad, max_val


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         global_best: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with Polyak step size."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_val = _max_convolution_and_gradient(a)
        norm_sq = np.dot(grad, grad)
        if norm_sq > 0:
            step = (cur_val - best_val) / (norm_sq + 1e-12)
            step = np.clip(step, 1e-7, 0.25)
        else:
            step = 0.0

        a = _project_onto_simplex(a - step * grad, S)

        cur_val = _max_convolution(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()

        if it % 400 == 0:
            jitter = rng.normal(scale=0.001 * np.mean(a), size=n)
            a = _project_onto_simplex(a + jitter, S)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb (mass transfer)
# ----------------------------------------------------------------------
def _anneal_local_search(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Mass‑transfer SA that accepts occasional worsening moves."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    cur_val = _max_convolution(a)
    best_a = a.copy()
    best_val = cur_val

    # temperature starts as ~5 % of current peak
    T = 0.05 * cur_val
    decay = 0.9999

    it = 0
    while time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        frac = rng.uniform(0.02, 0.18)
        delta = a[i] * frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _max_convolution(a_new)
        delta_val = new_val - cur_val
        if delta_val < 0 or rng.random() < np.exp(-delta_val / (T + 1e-12)):
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()

        T *= decay
        if it % 600 == 0:
            jitter = rng.normal(scale=0.001 * S, size=n)
            a = _project_onto_simplex(a + jitter, S)
            cur_val = _max_convolution(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine follows the multi‑stage strategy described in the
    module docstring and stores the final result in the global variable
    ``height_sequence_2``.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # a small safety margin

    # ------------------------------------------------------------------
    # Warm‑start handling
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_2' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_2'], dtype=float)
        except Exception:
            warm_seq = None
    elif 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # ------------------------------------------------------------------
    # Phase 1 – broad random exploration (≈ 55 % of budget)
    # ------------------------------------------------------------------
    explore_deadline = end_time - 340.0   # keep ~340 s for refinement
    top_k = 5
    top_candidates = []    # list of (seq, score)
    attempts = 0
    while time.time() < explore_deadline:
        attempts += 1
        # favour longer sequences but keep some diversity
        if rng.random() < 0.78:
            n = int(rng.integers(3000, 15000))
        else:
            n = int(rng.integers(15000, 25000))

        a0 = _initial_vector(n, rng, warm_seq)
        val = _max_convolution(a0)

        # keep the best few candidates
        if len(top_candidates) < top_k:
            top_candidates.append((a0, val))
            top_candidates.sort(key=lambda x: x[1])
        else:
            if val < top_candidates[-1][1]:
                top_candidates[-1] = (a0, val)
                top_candidates.sort(key=lambda x: x[1])

        if attempts % 250 == 0:
            best_current = top_candidates[0][1] if top_candidates else np.inf
            print(f"[explore] attempts={attempts:4d} best_sofar={best_current:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation on the retained candidates
    # ------------------------------------------------------------------
    # allocate roughly equal slices of remaining time to each candidate
    remaining_time = end_time - time.time() - 120.0   # reserve for final phases
    if remaining_time <= 0:
        remaining_time = 0.0

    per_candidate_time = remaining_time / max(1, len(top_candidates))

    best_seq = None
    best_score = np.inf

    for idx, (seq, seq_val) in enumerate(top_candidates):
        if time.time() >= end_time - 130.0:
            break
        print(f"[adam] candidate {idx + 1}/{len(top_candidates)} length={len(seq)} start_val={seq_val:.6f}")
        slice_deadline = time.time() + per_candidate_time

        # Adam optimisation
        a_opt, val = _adam_optimize(seq, slice_deadline, rng)

        # short hard‑max sub‑gradient polishing
        sub_deadline = time.time() + min(12.0, slice_deadline - time.time())
        a_opt, val = _subgradient_descent(a_opt, sub_deadline, rng, best_score)

        if val < best_score:
            best_score = val
            best_seq = a_opt.copy()
            print(f"[adam] NEW GLOBAL BEST = {best_score:.6f} (len {len(best_seq)})")

    # ------------------------------------------------------------------
    # Phase 3 – final long sub‑gradient run on the current best
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < end_time - 80.0:
        final_sub_deadline = time.time() + (end_time - time.time() - 80.0)
        a_opt, val = _subgradient_descent(best_seq, final_sub_deadline, rng, best_score)
        if val < best_score:
            best_score = val
            best_seq = a_opt.copy()
            print(f"[final subgrad] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated‑annealing refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < end_time - 10.0:
        anneal_deadline = time.time() + (end_time - time.time() - 10.0)
        a_opt, val = _anneal_local_search(best_seq, anneal_deadline, rng)
        if val < best_score:
            best_score = val
            best_seq = a_opt.copy()
            print(f"[anneal] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback to a uniform sequence of moderate size
        n = 4000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = _max_convolution(best_seq)

    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f}  internal={best_score:.6f}  length={len(final_seq)}")

    # store under a distinct global name (so the next run can warm‑start)
    globals()['height_sequence_2'] = final_seq
    return final_seq
