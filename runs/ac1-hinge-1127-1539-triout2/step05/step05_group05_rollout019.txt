def _subgradient_descent(a0, deadline):
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    step = 0
    best = a.copy()
    best_score = _max_convolution_fft(a)
    while time.time() < deadline:
        b = _full_convolution_fft(a) # shape length 2n-1
        k_max = int(np.argmax(b))
        # compute gradient
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[idx[mask]]
        # compute step size
        eta = 0.5 / np.sqrt(step+1)
        # update
        a = a - eta * g
        a = np.maximum(a, 0.0)  # optional
        # project to simplex sum = target_sum
        s = np.sum(a)
        if s > 0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)
        # evaluate improvement
        cur_score = b[k_max]   # max value
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
        step += 1
        # occasional random perturbation
        if step % 1000 == 0:
            # jitter
            jitter = np.random.default_rng().normal(scale=target_sum*0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            s = np.sum(a)
            if s > 0:
                a = a * (target_sum / s)
    return best

def generate_geometric(n, beta):
    a = np.exp(beta * np.arange(n, dtype=float))
    a = np.maximum(a, 0.0) # though always positive
    # Normalize to sum sqrt(2n)
    s = np.sqrt(2.0 * n)
    a = a * (s / a.sum())
    return a
```

def _subgradient_descent(a0, deadline):
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    best = a.copy()
    best_score = _max_convolution_fft(a)
    step = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        # compute convolution and max index
        b = _full_convolution_fft(a)
        k = int(np.argmax(b))
        # gradient
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[idx[mask]]
        # step size
        eta = 0.3 / np.sqrt(step + 1.0)
        a = a - eta * g
        # add random jitter occasionally
        if step % 500 == 0:
            if rng.random() < 0.2:
                jitter = rng.normal(scale=target_sum * 0.001, size=n)
                a += jitter
        # project to simplex
        s = np.sum(a)
        if s > 0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)
        cur_score = b[k]
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time():.1f}] Subgrad improvement: {best_score:.6f}")
        step += 1
    return best

        # gradient of variance-like objective

        # compute gradient = 4 * (a_rev * p) central
        a_rev = a[::-1]
        size = 1 << ((len(p) + n - 1).bit_length())

"""
Improved step‑function search.

The optimiser works in three stages under the global time budget.

1.  **Coarse parametric scan** – for a handful of lengths (up to 10 000)
    we generate families with a few parameters (geometric, power‑law,
    combined power×exponential, Beta‑shaped, Dirichlet).  The families are
    normalised to Σa_i = √(2 n) and the maximal self‑convolution entry is
    measured with a fast FFT.  The best candidate from this cheap scan is
    kept.

2.  **Variance‑minimising gradient descent** – we minimise the
    squared‑error between the whole convolution vector and its average
    value 2 n/(2 n‑1).  The gradient of
        J(a)=∑_k ( (a*a)[k] – μ )²
    is 4·(a ⋆ ((a*a)–μ)) and can be evaluated with two FFTs.  After each
    gradient step the vector is projected back onto the ℓ¹‑simplex
    (non‑negative, Σa_i = √(2 n)).  This directly flattens the convolution
    and quickly drives the maximal entry down.

3.  **Targeted simulated annealing** – the best sequence from the
    gradient phase is refined with the SA routine from the reference
    implementation (mass is moved from indices that heavily contribute
    to the current peak).  The temperature schedule is kept simple and
    a small amount of global jitter is added occasionally.

All helper functions are top‑level; no I/O is performed.  The routine
returns a plain ``list[float]`` that can be fed to the supplied
``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean, Duchi et al. 2008)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x_i≥0, Σx_i = s}."""
    if s <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random families (used in the coarse scan)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Generate a random non‑negative sequence on the ℓ¹‑simplex."""
    target_sum = np.sqrt(2.0 * n)
    kind = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if kind == 0:                     # pure power‑law (decreasing)
        p = rng.uniform(0.15, 1.0)
        a = (idx + 1.0) ** (-p)

    elif kind == 1:                 # exponential decay from the left edge
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif kind == 2:                 # mixed power‑law + exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif kind == 3:                 # Dirichlet (uniform simplex)
        a = rng.dirichlet(alpha=np.ones(n))

    elif kind == 4:                 # symmetric edge‑heavy
        p = rng.uniform(0.2, 1.2)
        left = (idx + 1.0) ** (-p)
        right = (n - idx) ** (-p)
        a = left + right

    else:                           # Beta‑shaped
        alpha = rng.uniform(0.2, 2.0)
        beta = rng.uniform(0.2, 2.0)
        x = (idx + 0.5) / n
        a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))

    a += rng.normal(scale=target_sum * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target_sum)
    return a


def _geometric_sequence(n: int, beta: float) -> np.ndarray:
    """Geometric (exponential) sequence a_i ∝ exp(beta·i), normalised."""
    target_sum = np.sqrt(2.0 * n)
    a = np.exp(beta * np.arange(n, dtype=float))
    a = a * (target_sum / a.sum())
    return a


def _powerlaw_sequence(n: int, p: float) -> np.ndarray:
    """Power‑law sequence a_i ∝ (i+1)^p, normalised."""
    target_sum = np.sqrt(2.0 * n)
    a = (np.arange(1, n + 1, dtype=float) ** p)
    a = a * (target_sum / a.sum())
    return a


def _combined_sequence(n: int, p: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^p · exp(beta·i) , normalised."""
    target_sum = np.sqrt(2.0 * n)
    idx = np.arange(n, dtype=float)
    a = (idx + 1.0) ** p * np.exp(beta * idx)
    a = a * (target_sum / a.sum())
    return a


# ----------------------------------------------------------------------
# Gradient descent that minimises variance of the convolution
# ----------------------------------------------------------------------


def _min_variance_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Gradient descent on J(a)=∑_k ( (a*a)[k] – μ )²,
    where μ = 2·n/(2·n‑1) is the average value of the convolution.
    After each step the iterate is projected onto the ℓ¹‑simplex.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    mu = 2.0 * n / (2.0 * n - 1.0)          # average convolution value
    step = 0
    best = a.copy()
    best_score = _max_convolution_fft(a)
    rng = np.random.default_rng()

    while time.time() < deadline:
        # full convolution and its deviation from the average
        b = _full_convolution_fft(a)               # length 2n‑1
        p = b - mu                                   # can be negative

        # gradient = 4·(a_rev ⋆ p) (central n entries)
        a_rev = a[::-1]
        size = 1 << ((len(p) + n - 1).bit_length())
        A_rev_fft = np.fft.rfft(a_rev, n=size)
        P_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(A_rev_fft * P_fft, n=size)[: len(p) + n - 1]
        g = 4.0 * conv_rev[n - 1:n - 1 + n]

        # diminishing step size
        eta = 0.5 / np.sqrt(step + 1.0)
        a -= eta * g

        # keep non‑negativity and enforce the ℓ¹‑norm
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        cur_score = b.max()
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time():.1f}] variance‑descent improvement: {best_score:.6f}")

        # occasional jitter to escape plateaus
        if step % 400 == 0 and rng.random() < 0.25:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0.0:
                a = a * (target_sum / s)

        step += 1

    return best


# ----------------------------------------------------------------------
# Targeted simulated‑annealing (unchanged from the reference code)
# ----------------------------------------------------------------------


def _targeted_simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    SA that repeatedly moves a fraction of mass from an index that
    heavily contributes to the current maximal convolution entry to a
    random destination.  Temperature decays multiplicatively.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_fft(best)

    T = best_score * 0.02 + 1e-8
    cooling = 0.9995
    rng = np.random.default_rng()
    max_transfer = 0.25

    while time.time() < deadline:
        b = _full_convolution_fft(a)
        k_max = int(np.argmax(b))

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idx_range = np.arange(lo, hi + 1)
        contrib = a[idx_range] * a[k_max - idx_range]
        weight_sum = np.sum(contrib)

        if weight_sum > 0:
            probs = contrib / weight_sum
            i = rng.choice(idx_range, p=probs)
        else:
            i = rng.integers(0, n)

        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        delta = a[i] * rng.random() * max_transfer
        if delta <= 0.0:
            continue
        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        accept = False
        if cur_score <= best_score:
            accept = True
        elif T > 0.0 and rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA improvement: {best_score:.6f}")
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling

        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a_sum = np.sum(a)
            if a_sum > 0:
                a = a * (np.sqrt(2.0 * n) / a_sum)
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA jitter improvement: {best_score:.6f}")

    return best

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative sequence that minimises evaluate_sequence().
    The routine scans a few lengths with parametric families, refines the
    best candidate with variance‑minimising gradient descent and finally
    applies a short simulated‑annealing polishing phase.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # If a hand‑crafted sequence exists, start from it.
    # ------------------------------------------------------------------
    global_seq = globals().get("height_sequence_1")
    if isinstance(global_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(global_seq, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        sc = _max_convolution_fft(a0)
        best_seq, best_score = a0.copy(), sc
        print(f"[{time.time():.1f}] Start from global height_sequence_1 "
              f"(n={n0})  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Coarse parametric scan over several lengths
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 4000, 5000, 6000, 8000, 10000]
    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # --- geometric family ------------------------------------------------
        for beta in np.linspace(-0.010, -0.0005, 30):
            a = _geometric_sequence(n, beta)
            score = _max_convolution_fft(a)
            if score < best_score:
                best_seq, best_score = a.copy(), score
                print(f"[{time.time():.1f}] New best (geom) n={n} β={beta:.5f}  "
                      f"score={score:.6f}")

        # --- power‑law family -------------------------------------------------
        for p in np.linspace(-2.0, 2.0, 30):
            a = _powerlaw_sequence(n, p)
            score = _max_convolution_fft(a)
            if score < best_score:
                best_seq, best_score = a.copy(), score
                print(f"[{time.time():.1f}] New best (pow) n={n} p={p:.3f}  "
                      f"score={score:.6f}")

        # --- combined power‑law × exponential --------------------------------
        for _ in range(40):
            p = rng.uniform(-2.0, 2.0)
            beta = rng.uniform(-0.010, -0.0005)
            a = _combined_sequence(n, p, beta)
            score = _max_convolution_fft(a)
            if score < best_score:
                best_seq, best_score = a.copy(), score
                print(f"[{time.time():.1f}] New best (combo) n={n} p={p:.3f} β={beta:.5f}  "
                      f"score={score:.6f}")

        # --- random diversifiers (same as the reference code) ---------------
        for _ in range(30):
            a = _random_initial_sequence(n, rng)
            score = _max_convolution_fft(a)
            if score < best_score:
                best_seq, best_score = a.copy(), score
                print(f"[{time.time():.1f}] New best (rand) n={n}  score={score:.6f}")

    # ------------------------------------------------------------------
    # If nothing succeeded, fall back to a uniform sequence of moderate size.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 3000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = _max_convolution_fft(best_seq)
        print(f"[{time.time():.1f}] Fallback uniform n={n}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Variance‑minimising gradient descent (major part of the remaining budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 15.0:
        gd_deadline = time.time() + remaining * 0.6
        a_gd = _min_variance_descent(best_seq, gd_deadline)
        gd_score = _max_convolution_fft(a_gd)
        if gd_score < best_score:
            best_seq, best_score = a_gd.copy(), gd_score
            print(f"[{time.time():.1f}] After variance‑descent score={gd_score:.6f}")

        # ------------------------------------------------------------------
        # 3️⃣  Targeted simulated‑annealing polishing (uses whatever time left)
        # ------------------------------------------------------------------
        a_sa = _targeted_simulated_annealing(best_seq, deadline)
        final_score = _max_convolution_fft(a_sa)
        if final_score < best_score:
            best_seq, best_score = a_sa.copy(), final_score
            print(f"[{time.time():.1f}] After SA polish score={final_score:.6f}")

    # ------------------------------------------------------------------
    # Remove trailing near‑zero entries and return a plain list
    # ------------------------------------------------------------------
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # sanity check with the official evaluator (optional)
    try:
        ev = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence = {ev:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best max‑convolution = "
          f"{best_score:.6f}   length = {best_seq.size}")

    return best_seq.tolist()
