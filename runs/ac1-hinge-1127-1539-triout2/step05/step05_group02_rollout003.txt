"""
Propose a low‑score step‑function.

The optimiser combines three ideas:

1. **Simplex initialisation** – start from a known good sequence
   (``height_sequence_1``) if available, otherwise a power‑law profile.
   All candidates are projected onto the ℓ¹‑simplex
   Σ a_i = √(2·n) (the scaling that makes the benchmark score equal to
   the maximum convolution entry).

2. **Frank‑Wolfe style LP step** – given the current sequence f,
   compute M = max(f∗f).  Solve the linear programme

        maximise   Σ g_i
        subject to  (f ∗ g)_k ≤ M   for every k
                    g_i ≥ 0

   The matrix (f ∗ g)_k ≤ M is a lower‑triangular Toeplitz matrix that can be
   built sparsely.  The solution g₀ is then scaled to the required ℓ¹‑norm.
   A convex combination f←(1‑t)f+t·g (with a back‑tracking line‑search) strictly
   reduces the objective.

3. **Soft‑max gradient descent** – occasional steps on a smooth
   approximation of the max‑operator (soft‑max with an increasing temperature
   β) provide fine‑grained refinements.

The routine runs several independent optimisations for different lengths
(1500 → 2500 → 3500) until the time budget (≈1000 s) is exhausted and returns
the best sequence found.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Convolution helpers (FFT based – fast and accurate for our sizes)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Benchmark score for a sequence that already satisfies the ℓ¹‑norm."""
    # the evaluator normalises the sum to √(2·n), so the score equals max conv
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto the simplex Σx_i = s,  x_i ≥ 0  (Duchi et al. 2008)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x | x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑operator (used for fine tuning)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient of the soft‑max w.r.t. a
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Linear programme used in the Frank‑Wolfe step
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1)×n matrix A with A[k,i] = f_i   if   k−i = f-index ≥ 0
    (i.e. the linear constraints (f∗g)_k ≤ M are written as A·g ≤ M·1).
    """
    n = f.size
    rows = 2 * n - 1
    A = sp.lil_matrix((rows, n), dtype=np.float64)

    # column i contributes f[i] to rows i … i+n‑1
    for i in range(n):
        A[i:i + n, i] = f[i]

    return A.tocsr()


def _solve_lp_direction(f: np.ndarray, M: float, target_sum: float):
    """
    Solve   maximise Σ g_i   s.t.   A·g ≤ M·1 , g ≥ 0,
    where A encodes the convolution with f.
    The returned vector is rescaled to have ℓ¹‑norm = target_sum.
    Returns None if the LP fails.
    """
    n = f.size
    A = _build_conv_matrix(f)
    b = np.full(A.shape[0], M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)               # minimise –Σ g_i  ↔ maximise Σ g_i
    bounds = [(0, None)] * n

    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                                 method="highs", options={"presolve": True})
    if not res.success:
        return None

    g0 = res.x
    s0 = g0.sum()
    if s0 <= 1e-12:
        return None
    # scale to the required ℓ¹‑norm
    g = g0 * (target_sum / s0)
    return g

    f = _project_to_simplex(f, target_sum)

    best_f = f.copy()
    best_score = _score(best_f)

    it = 0
    beta = 1.0          # temperature for the soft‑max gradient
    while time.time() < deadline:
        # ----- current max‑convolution -----
        M = _max_convolution_fft(f)

        # ----- Frank‑Wolfe LP step (every 8th iteration) -----
        if it % 8 == 0:
            g = _solve_lp_direction(f, M, target_sum)
            if g is not None:
                # simple back‑tracking line search for the best mixing weight t
                t = 0.5
                while t > 1e-4:
                    f_candidate = (1.0 - t) * f + t * g
                    cand_score = _score(f_candidate)
                    if cand_score < best_score:
                        best_score = cand_score
                        best_f = f_candidate.copy()
                        print(f"[{time.time():.1f}] n={n} it={it} LP improve → {best_score:.6f}")
                    if cand_score <= _score(f):
                        f = f_candidate
                        break
                    t *= 0.5

        # ----- occasional soft‑max gradient descent -----
        if it % 5 == 0:
            # increase temperature gradually
            beta = min(1024.0, beta * 1.05)
            grad, _ = _gradient_softmax(f, beta)
            gn = np.linalg.norm(grad)
            if gn > 0:
                # step size inversely proportional to gradient norm
                lr = 0.2 * target_sum / (gn + 1e-12)
                f_new = f - lr * grad
                f_new = np.maximum(f_new, 0.0)
                f_new = _project_to_simplex(f_new, target_sum)
                new_score = _score(f_new)
                if new_score < best_score:
                    best_score = new_score
                    best_f = f_new.copy()
                    print(f"[{time.time():.1f}] n={n} it={it} GD improve → {best_score:.6f}")
                if new_score <= _score(f):
                    f = f_new

        # ----- random jitter to escape plateaus (every 200 iters) -----
        if it % 200 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            f = f + jitter
            f = np.maximum(f, 0.0)
            f = _project_to_simplex(f, target_sum)

        it += 1

    return best_f, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the benchmark.
    The search runs several independent optimisations for different lengths,
    each employing a Frank‑Wolfe LP step together with occasional
    soft‑max gradient descent.  The best sequence found before the time budget
    expires is returned.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # a tiny safety margin

    best_seq = None
    best_score = float("inf")

    # lengths to explore – larger n gives a finer granularity
    lengths = [1500, 2500, 3500]

    # give each length a fair share of the remaining budget (≈300 s each)
    for n in lengths:
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # allocate at most 300 s or whatever is left
        max_time_n = min(300.0, remaining)
        seq, score = _optimise_length(n, max_time_n, rng)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] New global best (n={n}) → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Strip trailing zeros (they do not affect the score) and return
    # ------------------------------------------------------------------
    if best_seq is None:
        # fall‑back – a tiny uniform sequence (should never happen)
        best_seq = np.array([1.0], dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    print(f"[{time.time():.1f}] Final best score = {best_score:.6f}, length = {len(best_seq)}")
    return best_seq.tolist()
