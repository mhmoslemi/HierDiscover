"""
Hybrid annealed‑power gradient descent for the self‑convolution peak problem.

The objective is 2·n·max(conv(a,a)) / (∑a)².  This implementation
* works on the ℓ¹‑simplex Σa = √(2·n),
* uses an Adam optimiser on a smooth “p‑power’’ approximation of the
  max‑norm (weights w_i ∝ (c_i / max(c))^p, p growing from 2 to 128),
* projects onto the simplex after each step,
* occasionally performs random mass‑swaps for exploration,
* finishes each trial with a projected sub‑gradient phase that directly
  attacks the worst convolution entry,
* repeatedly restarts with different lengths and up‑samples the current
  best construction, keeping the global best throughout the time budget.

The search runs for the allotted budget (default 1000 s) and finally
returns the best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient term for a weighted max‑norm:
        g_j = 2 Σ_i w_{i+j} a_i .
    Both ``a`` (len n) and ``w`` (len 2·n‑1) are real vectors.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1           # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)           # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Weight computation (p‑power soft‑max)
# ----------------------------------------------------------------------


def _p_weights(conv: np.ndarray, p: float) -> np.ndarray:
    """
    Compute normalized weights w_i ∝ (c_i / max(c))^p.
    The weights sum to one and concentrate on the largest convolution entries.
    """
    maxc = np.max(conv)
    if maxc <= 0:
        return np.ones_like(conv) / conv.size
    rel = conv / maxc                     # values in [0,1]
    # power may underflow for large p; that is fine – it just makes the weight
    # of non‑max entries very small.
    pw = np.power(rel, p)
    s = pw.sum()
    if s == 0.0:                           # extreme under‑flow → fall back to uniform
        return np.ones_like(conv) / conv.size
    return pw / s


# ----------------------------------------------------------------------
# Upsampling helper (linear interpolation)
# ----------------------------------------------------------------------


def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a to length new_n."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=float)
    return np.interp(x_new, x_old, a)


# ----------------------------------------------------------------------
# Adam optimiser on the p‑power smooth max‑norm
# ----------------------------------------------------------------------


def _adam_optimize_power(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         p_schedule=(2, 4, 8, 16, 32, 64, 128)) -> tuple[np.ndarray, float]:
    """
    Adam optimisation where the loss is the weighted sum of the convolution
    entries with weights w_i ∝ (c_i / max(c))^p.
    The exponent p is increased according to ``p_schedule``.
    Returns the best vector seen and its worst‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.4
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    p_idx = 0
    p_cur = p_schedule[p_idx]
    steps_per_p = 500          # keep each p for about this many updates

    while time.time() < deadline:
        # ---------- power schedule ----------
        if t % steps_per_p == 0 and p_idx + 1 < len(p_schedule):
            p_idx += 1
            p_cur = p_schedule[p_idx]

        # ---------- gradient ----------
        conv = _full_convolution(a)
        w = _p_weights(conv, p_cur)          # shape (2n-1,)
        grad = _conv_with_weights(a, w)       # already contains factor 2

        # ---------- Adam update ----------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---------- projection ----------
        a = _project_simplex(a, S)

        # occasional random “mass‑swap’’ for exploration
        if rng.random() < 0.04:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.01
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # cheap early‑stop test
        if t % 4000 == 0 and best_score == cur_score:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient optimiser (acts on the true max‑norm)
# ----------------------------------------------------------------------


def _subgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that directly reduces the worst convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.25
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient: g_i = a_{k_max-i} (zero when index out of range)
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional random perturbation
        if rng.random() < 0.05:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.02
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = np.max(_full_convolution(cand))
                if cand_score < cur_score or rng.random() < 0.2:
                    a = cand
                    cur_score = cand_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()

        if it % 8000 == 0 and best_score == cur_score:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimiser (Adam → Sub‑gradient)
# ----------------------------------------------------------------------


def _combined_optimize(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Run the power‑Adam phase, then finish with a sub‑gradient refinement."""
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    # allocate ~60 % of the remaining budget to the Adam phase
    adam_deadline = now + 0.6 * remaining
    a_mid, sc_mid = _adam_optimize_power(a0, adam_deadline, rng)

    # if time permits, run a sub‑gradient sweep on the remainder
    if time.time() < deadline:
        a_fin, sc_fin = _subgrad_optimize(a_mid, deadline, rng)
        if sc_fin < sc_mid:
            return a_fin, sc_fin
    return a_mid, sc_mid


# ----------------------------------------------------------------------
# Entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function (vector) minimising the
    ``evaluate_sequence`` score.  The routine mixes an annealed p‑power Adam
    optimiser, a sub‑gradient fine‑tuner, progressive up‑sampling and random
    restarts.  The best sequence found within the time budget is returned as
    a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin for the final external evaluation
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_score = np.inf
    attempts = 0
    last_improve = start_time

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline - 30.0:   # reserve ~30 s for final polishing
        attempts += 1

        # ---- decide on problem size -------------------------------------------------
        if best_seq is None or (time.time() - last_improve > 45):
            # fresh random start, moderate size
            n = int(rng.integers(1500, 3500))
            a0 = rng.random(n)
        else:
            # sometimes up‑sample the current best to a larger arena
            if rng.random() < 0.30 and len(best_seq) < 8000:
                n = min(int(len(best_seq) * 1.5), 8000)
                a0 = _upsample(best_seq, n)
            else:
                n = len(best_seq)
                a0 = best_seq.copy()

            # tiny jitter to escape plateaus
            jitter = rng.normal(scale=a0.mean() * 1e-4, size=n)
            a0 = a0 + jitter
            a0 = np.clip(a0, 0.0, None)

        # ---- project onto the simplex (Σa = √(2·n)) -------------------------------
        S = np.sqrt(2.0 * n)
        total = a0.sum()
        if total == 0.0:
            a0 = np.full(n, 1.0)
            total = n
        a0 = a0 / total * S

        # ---- allocate a time slice for this trial ---------------------------------
        slice_deadline = min(time.time() + 20.0, deadline - 30.0)

        # ---- run the combined optimiser -------------------------------------------
        a_opt, score_opt = _combined_optimize(a0, slice_deadline, rng)

        # ---- keep the global best -------------------------------------------------
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            last_improve = time.time()
            print(f"[global] attempt={attempts} n={n} new best max‑conv={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing on the best candidate (use the remaining time)
    # ------------------------------------------------------------------
    if best_seq is not None:
        final_seq, final_score = _combined_optimize(best_seq, deadline, rng)
    else:
        # fallback – a tiny constant vector
        final_seq = np.array([1.0])
        final_score = np.max(_full_convolution(final_seq))

    # ------------------------------------------------------------------
    # External evaluation & return
    # ------------------------------------------------------------------
    final_list = final_seq.tolist()
    external_score = evaluate_sequence(final_list)
    print(f"[final] external score = {external_score:.6f} "
          f"(internal max‑conv = {final_score:.6f}) length = {len(final_list)}")
    return final_list
