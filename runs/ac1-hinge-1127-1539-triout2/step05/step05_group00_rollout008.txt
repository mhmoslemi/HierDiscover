def _targeted_hill_climbing(a0, b0, deadline):
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()
    best_b = b.copy()
    rng = np.random.default_rng()
    T = cur_score * 0.01 + 1e-8   # initial temperature
    cooling = 0.9999

    while time.time() < deadline:
        # find the index of max b
        k = int(np.argmax(b))
        # Compute contributions to this max b
        i_min = max(0, k - (n-1))
        i_max = min(n-1, k)
        # sample source i from contributions weighted by product a_i * a_{k-i}
        a_i = a[i_min:i_max+1]
        a_j = a[k - np.arange(i_min, i_max+1)]
        contrib = a_i * a_j
        sum_contrib = contrib.sum()
        if sum_contrib == 0:
            # fallback to random source
            i = rng.integers(0, n)
        else:
            probs = contrib / sum_contrib
            i = rng.choice(np.arange(i_min, i_max+1), p=probs)

        # choose destination j with small a[j]; sample biased
        # compute weight inverse of a + epsilon
        inv = 1.0 / (a + 1e-12)
        probs_dest = inv / inv.sum()
        j = rng.choice(n, p=probs_dest)
        if i == j:
            continue

        # delta up to e.g., 30% of source
        max_delta = a[i] * 0.3
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # Update convolution as before
        b_new = b.copy()
        b_new[i:i+n] -= 2.0 * delta * a
        b_new[j:j+n] += 2.0 * delta * a
        k2i = 2*i
        k2j = 2*j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()
        accept = False
        if new_score < cur_score:
            accept = True
        else:

                best_a = a.copy()
                best_b = b.copy()
                print(f"[{time.time():.1f}] targeted_hill improvement: {best_score:.6f}")

        T *= cooling

    return best_a, best_b

"""
Improved step‑function optimizer.

The search proceeds in several stages:

1.  Load a previously‑found construction (if any) and keep it as the current best.
2.  For a schedule of lengths (2000 → 5000 → 10000 → 20000) allocate a
    time slice and optimise that length.
    *   Initialise the candidate either by up‑sampling the current best
        or by a random Dirichlet / exponential family.
    *   Run a projected gradient descent on a smooth log‑sum‑exp
        approximation of the convolution maximum (the same scheme as in the
        reference code, but we also symmetrise the vector after the descent).
    *   Perform a *targeted* hill‑climbing phase:  the index `k` attaining the
        current maximum of the convolution is identified and mass is moved
        preferentially from indices that contribute heavily to `b_k` towards
        indices with low values.  The move is accepted with a simple
        simulated‑annealing rule.
    *   A short run of the generic hill‑climber from the reference code is
        executed as a final clean‑up for the current length.
3.  After the length schedule a final polishing phase runs the targeted
    hill‑climber for the remaining budget.
4.  Trailing near‑zero entries are stripped and the resulting list is
    returned.  Throughout the search the vector is kept non‑negative and
    projected onto the ℓ¹‑simplex Σa = √(2 n) so that the official
    `evaluate_sequence` sees a valid step‑function.

The algorithm respects the overall wall‑clock budget (default 1000 s) and
logs intermediate scores with timestamps.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a as a NumPy array."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force symmetry: (v + v[::-1]) / 2 (preserves sum)."""
    return (v + v[::-1]) * 0.5


# ----------------------------------------------------------------------
# Smoothed maximum (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of   (1/β)·log Σ exp(β·b)   where   b = a*a .
    Returns (g, b) with g = 2·(a ⋆ p) and
    pₖ ∝ exp(β·bₖ).  Correlation is performed via FFT.
    """
    n = a.shape[0]

    # b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]                          # reversed a
    conv_len = n + (2 * n - 1) - 1           # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smooth max.
    Uses momentum, an annealed inverse temperature β
    and occasional jitter.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 8192.0
    beta_factor = 1.4

    step_base = target_sum * 0.15
    momentum = np.zeros_like(a)
    rho = 0.90

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β annealing
        if it % 350 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative sequence on the ℓ¹‑simplex Σa = sqrt(2 n).
    Several families (power‑law, exponential, Dirichlet, piecewise) are
    sampled uniformly.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # pure power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length.
    The result is projected back onto the appropriate simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

def _hill_climbing(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Generic hill‑climbing (mass‑transfer) with a simple annealing rule.
    Returns the best (a, b) pair observed before the deadline.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()
    best_b = b.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.02 + 1e-9
    cooling = 0.9997

    while time.time() < deadline:
        # pick a source index proportionally to its mass
        if a.sum() == 0:
            break
        probs = a / a.sum()
        i = rng.choice(n, p=probs)

        if a[i] < 1e-14:
            continue

        # pick a destination uniformly
        j = rng.integers(0, n)
        if i == j:
            continue

        # amount of mass to move (up to 50 % of source)
        max_delta = a[i] * 0.5
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ---------- compute new convolution ----------
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a

        # quadratic correction terms
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # ---------- acceptance ----------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-(new_score - cur_score) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                best_b = b.copy()
                print(f"[{time.time():.1f}] hill_climb improvement: {best_score:.6f}")

        T *= cooling

    return best_a, best_b


def _targeted_hill_climbing(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Hill‑climbing that focuses on the index attaining the current maximum
    of the convolution.  Mass is moved preferentially from entries that
    contribute heavily to that maximum towards entries with small values.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()
    best_b = b.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.01 + 1e-8
    cooling = 0.9999

    while time.time() < deadline:
        # index of maximal convolution entry
        k = int(np.argmax(b))

        # range of i that actually participates in b_k
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)

        # contribution weights = a_i * a_{k-i}
        idx_range = np.arange(i_min, i_max + 1)
        contrib = a[idx_range] * a[k - idx_range]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs_src = contrib / total_contrib
            i = rng.choice(idx_range, p=probs_src)
        else:
            i = rng.integers(i_min, i_max + 1)

        # destination: bias towards small a[j]
        inv = 1.0 / (a + 1e-12)
        probs_dest = inv / inv.sum()
        j = rng.choice(n, p=probs_dest)

        if i == j:
            continue

        max_delta = a[i] * 0.3  # move up to 30 % of source mass
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # update convolution (same formula as generic hill climb)
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a

        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-(new_score - cur_score) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                best_b = b.copy()
                print(f"[{time.time():.1f}] targeted_hill improvement: {best_score:.6f}")

        T *= cooling

    return best_a, best_b

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock budget (default 1000 s) and returns
    the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Seed from a previously stored construction if available
    # ------------------------------------------------------------------
    earlier = globals().get("height_sequence_1")
    if isinstance(earlier, (list, tuple, np.ndarray)):
        a0 = np.asarray(earlier, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        best_seq = a0.copy()
        best_score = _max_convolution_fft(a0)
        print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Length schedule – allocate a slice of the budget to each length
    # ------------------------------------------------------------------
    # (length, allocated_seconds) – numbers tuned empirically
    schedule = [
        (2000, 180),
        (5000, 180),
        (10000, 260),
        (20000, max(0, deadline - time.time() - 5)),   # whatever is left
    ]

    for n, allocated in schedule:
        if time.time() >= deadline:
            break
        time_left = deadline - time.time()
        if time_left <= 0:
            break
        # Do not exceed the remaining budget
        allocated = min(allocated, time_left)
        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Initialise a candidate for this length
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a = _upsample_sequence(best_seq, n)
        else:
            a = _random_initial_sequence(n, rng) if best_seq is None else best_seq.copy()
        a = _project_to_simplex(a, target_sum)

        # --------------------------------------------------------------
        # Phase A – projected gradient descent (≈ half of allocated time)
        # --------------------------------------------------------------
        gd_deadline = time.time() + allocated * 0.5
        a_gd = _gradient_descent_opt(a, gd_deadline)
        # enforce symmetry (helps flatten the convolution)
        a_gd = _symmetrize(a_gd)
        a_gd = _project_to_simplex(a_gd, target_sum)

        score_gd = _max_convolution_fft(a_gd)
        if score_gd < best_score:
            best_score = score_gd
            best_seq = a_gd.copy()
            print(f"[{time.time():.1f}] GD improvement (n={n}) score={score_gd:.6f}")

        # --------------------------------------------------------------
        # Phase B – targeted hill climbing
        # --------------------------------------------------------------
        if time.time() < deadline:
            b_full = _full_convolution(a_gd)
            hc_deadline = time.time() + allocated * 0.5
            a_hc, b_hc = _targeted_hill_climbing(a_gd, b_full, hc_deadline)
            # optional extra symmetrisation
            a_hc = _symmetrize(a_hc)
            a_hc = _project_to_simplex(a_hc, target_sum)

            score_hc = b_hc.max()
            if score_hc < best_score:
                best_score = score_hc
                best_seq = a_hc.copy()
                print(f"[{time.time():.1f}] targeted hill improvement (n={n}) score={score_hc:.6f}")

            # --------------------------------------------------------------
            # Phase C – a short run of the original hill‑climber as cleanup
            # --------------------------------------------------------------
            if time.time() < deadline - 5:
                extra_dead = deadline - 5
                a_hc2, b_hc2 = _hill_climbing(best_seq, _full_convolution(best_seq), extra_dead)
                score_hc2 = b_hc2.max()
                if score_hc2 < best_score:
                    best_score = score_hc2
                    best_seq = a_hc2.copy()
                    print(f"[{time.time():.1f}] generic hill improvement (n={n}) score={score_hc2:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with any remaining budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5:
        a_final = best_seq.copy()
        b_final = _full_convolution(a_final)
        a_final, b_final = _targeted_hill_climbing(a_final, b_final, deadline)
        a_final = _symmetrize(a_final)
        a_final = _project_to_simplex(a_final, np.sqrt(2.0 * a_final.size))
        final_score = b_final.max()
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] final polishing improvement: {final_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros for neatness
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log the official evaluation (objective = max of the self‑convolution)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
