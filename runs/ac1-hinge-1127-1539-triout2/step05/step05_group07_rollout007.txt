def solve_lp(a, M, solver="ECOS"):
    n = len(a)
    C = np.zeros((2*n-1, n))
    for i in range(n):
        C[i:i+n, i] = a
    b = cvxpy.Variable(n, nonneg=True)
    constraints = [C @ b <= M]
    prob = cvxpy.Problem(cvxpy.Maximize(cvxpy.sum(b)), constraints)
    prob.solve(solver=solver, warm_start=True, verbose=False, max_iters=5000)
    if b.value is None:
        return None
    else:
        return np.array(b.value, dtype=float)

"""
adversarial step‑function optimiser – extended version

The optimiser searches for a non‑negative vector `a` of length `n` that
minimises

        2·n·max(a*a) / (sum(a)²)

(the denominator is invariant under scaling, so we keep the ℓ₁‑norm
fixed to `√(2·n)` and simply minimise the peak of the self‑convolution).

Main ideas beyond the reference implementation:

* **Longer runs** – each restart now receives a sizeable time slice
  (≈ 20 % of the remaining budget, but at least 6 s) allowing many more
  Adam steps.
* **Wider range of lengths** – we explore very moderate (≈ 2 k) and
  large (≈ 2·10⁴) sequences; the theoretical lower bound on the score
  improves with `n`.
* **More aggressive cooling** – the soft‑max temperature `τ` is driven
  to the neighbourhood of zero much faster, which yields a gradient
  that is closer to the true sub‑gradient of the max‑operator.
* **Stronger jitter** – every few hundred iterations a Gaussian perturb
  is added in log‑space to escape plateaus.
* **Best‑candidate tracking** – after every inner optimisation run the
  exact hard peak is computed and the global best is updated.
* **Reuse of the previously‑found construction** – if the global
  variable ``height_sequence_1`` exists it is interpolated to the new
  length and mixed in with a small probability, providing a warm start.

The code respects the 1000 s budget, prints progress information and
stores the best sequence in the global name ``height_sequence_1``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full (linear) self‑convolution of a real vector using FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize `seq` to length `new_n` by linear interpolation."""
    old_n = len(seq)
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.ceil(pos).astype(int)
    hi = np.clip(hi, 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Soft‑max approximation of the max‑convolution and its gradient
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Returns (g, max_b) where
        g = ∂/∂a  τ·log∑exp(b/τ) ,   b = a*a   (full convolution)
    and max_b is the exact (hard) maximum of b.
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = b.max()

    # soft‑max weights over the (2n‑1) convolution entries
    if tau < 1e-12:
        # exact sub‑gradient: equal weight on every active (max) entry
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b          # negative numbers → exp ≈ 0
        exps = np.exp(diff / tau)   # shape (2n‑1,)
        mask = exps / exps.sum()     # soft‑max distribution

    # gradient w.r.t a : 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = mask.shape[0] + n - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]   # slice of length n
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Minimise the soft‑max approximation of the convolution peak.
    Returns the best vector encountered (still normalised to sum √(2n))
    and the corresponding hard peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)          # target ℓ₁‑norm, makes the score = peak

    # initialise logits so that softmax(p) * S == a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                     # logits (no projection needed later)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.3

    # start with a temperature slightly larger than the initial peak
    tau = max(_full_convolution_fft(a0).max(), 0.5)

    best_a = a0.copy()
    best_peak = _full_convolution_fft(a0).max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- a from logits (soft‑max on the simplex) -----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)               # stability trick
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ----- convolution & gradient (soft‑max) -----
        g, peak = _softmax_gradient(a, tau)

        # Keep the best **hard** peak seen so far
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ----- gradient w.r.t. logits (mirror descent) -----
        mu = np.dot(a, g) / S                  # ⟨a,∇⟩ / S   (scalar)
        grad_p = a * (g - mu)                  # shape (n,)

        # gradient clipping – prevents occasional explosions
        clip_thr = 5.0 * S
        norm = np.max(np.abs(grad_p))
        if norm > clip_thr:
            grad_p = grad_p * (clip_thr / norm)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- schedules -----
        tau *= 0.999   # faster cooling than the reference code
        lr *= 0.9999   # gentle decay of the step size

        # ----- occasional log‑space jitter -----
        if it % 400 == 0:
            p += rng.normal(scale=0.03, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------

        globals()['height_sequence_1'] = final_seq
