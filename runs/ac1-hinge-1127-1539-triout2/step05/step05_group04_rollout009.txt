def _hill_climbing(a0, deadline):
    a = a0.copy()
    scores = _score(a)
    best_score = scores
    best_a = a.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    # parameters
    iterations = 0
    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        # define fraction up to maybe max_frac = 0.3
        frac = rng.random() * 0.3
        delta = a[i] * frac
        if delta < 1e-12:
            continue
        # propose move
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # compute score
        new_score = _score(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
        # accept if beneficial (or with some probability)
        if new_score <= scores:
            a = a_new
            scores = new_score
        else:
            # occasionally accept uphill with prob e^{- (new_score - scores)/T}
            # T decays over time
            # we can compute T = (initial_score - best_score) * 0.1 / (iterations+1)
            # but simpler: accept with small probability
            if rng.random() < 0.001:
                a = a_new
                scores = new_score
        iterations += 1
    return best_a

def _mass_balance_refine_enhanced(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    target = np.sum(a)   # should be sqrt(2*n)
    rng = np.random.default_rng()
    while time.time() < deadline:
        conv = _max_convolution_fft(a)
        k = int(np.argmax(conv))
        # compute contributions of each index to entry k
        idx = np.arange(n)
        other = k - idx
        mask = (other >=0) & (other < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[other[mask]]
        # i_high = index with max contrib
        i_high = int(np.argmax(contrib))
        # i_low = index with min contrib (prefer low mass)
        # Choose among those with small contribution, optionally random among lowest 20% to avoid stagnation
        low_mask = mask & (contrib > 0)  # ensure there is a counterpart? but can be zero contributions also fine
        if low_mask.any():
            # Use percentile
            threshold = np.percentile(contrib[mask], 20)
            low_candidates = np.where(contrib <= threshold)[0]
        else:
            low_candidates = np.arange(n)
        i_low = int(rng.choice(low_candidates))
        # move fraction
        # choose fraction between 0.08 and 0.25 randomly
        frac = 0.08 + 0.17 * rng.random()
        delta = a[i_high] * frac
        if delta < 1e-12:
            break
        a[i_high] -= delta
        a[i_low] += delta
        # re-project to keep sum (should be unchanged) and nonnegative
        a = np.maximum(a, 0.0)
        # Rebalance small numeric drift
        a_sum = np.sum(a)
        if abs(a_sum - target) > 1e-9:
            a = a * (target / a_sum)

def _hill_climbing(a0, deadline):

    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    cur_score = best_score

    T = T0
    cooling = 0.9999
    while time.time() < deadline:
        # choose i with mass > epsilon
        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        # fraction random up to 0.2
        frac = rng.random() * 0.2
        delta = a[i] * frac
        if delta < 1e-12:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _score(a_new)
        dE = new_score - cur_score
        accept = dE <= 0 or (T > 0 and rng.random() < np.exp(-dE / T))
        if accept:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        # update temperature
        T *= cooling
    return best_a

"""
Step‑function optimizer – diversified search + aggressive refinement.

* Stage 1 – deterministic sweep of simple analytic families (power‑law,
  exponential, power×exponential, Dirichlet, piecewise‑exponential) for
  several lengths.  All candidates are explicitly scaled to satisfy the
  ℓ¹‑normalisation Σa = √(2 n) and scored by the exact infinity‑norm of the
  self‑convolution (the official objective).

* Stage 2 – the best candidate from the sweep is refined until the wall‑clock
  budget expires.  Four complementary local‑search primitives are cycled:

    – smooth‑max (log‑sum‑exp) projected gradient descent,
    – exact sub‑gradient descent on the max‑norm,
    – an aggressive “mass‑balance” move that transfers a noticeable
      fraction of the heaviest contributor of the current peak to a light
      site,
    – a lightweight simulated‑annealing hill‑climber that moves a random
      fraction of mass between two positions.

  The loop stops when less than five seconds remain, guaranteeing that the
  final call to ``evaluate_sequence`` fits within the time limit.

All helper functions are top‑level; there are no closures or lambdas.
The routine respects the non‑negativity and ℓ¹‑normalisation constraints,
uses only NumPy (no external solvers), and prints progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1, dtype=float)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n−1) using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective after ℓ¹‑normalisation Σa = √(2 n): the supremum norm of a*a."""
    return float(_max_convolution_fft(a).max())

# Soft‑max gradient (log‑sum‑exp) – Phase 1 of the refinement
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) where
        g = ∇_a  ( 1/β·log Σ exp(β·b) )
          = 2·(a ⋆ p)          with   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities p
    max_b = float(np.max(b))
    exps = np.exp(beta * (b - max_b))          # shift for numerical stability
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1            # = 3 n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _smooth_max_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Annealed projected gradient descent on the smooth‑max.
    Uses momentum, an adaptive step and occasional Gaussian jitter.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 4096.0
    beta_factor = 1.4

    # a slightly larger step than the reference code – 0.30 of the ℓ¹‑norm
    step_base = target_sum * 0.30
    momentum = np.zeros_like(a)
    rho = 0.90

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        # projection onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # jitter occasionally
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal β
        if it % 350 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a


# ----------------------------------------------------------------------
# Exact max‑index sub‑gradient – Phase 2 of the refinement
# ----------------------------------------------------------------------


def _max_index_subgradient(a: np.ndarray):
    """Sub‑gradient of max(b) where b = a*a."""
    n = a.shape[0]
    conv = _max_convolution_fft(a)
    k = int(np.argmax(conv))

    g = np.zeros_like(a)
    if k < n:                         # left half of the convolution
        g[:k + 1] = 2.0 * a[k::-1]
    else:                             # right half
        d = k - (n - 1)               # shift amount (0 … n‑1)
        g[d:] = 2.0 * a[: n - d]
    return g, k


def _subgradient_descent_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Projected sub‑gradient descent on the true max‑norm."""
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        g, _ = _max_index_subgradient(a)

        # diminishing step size η_t = α / √t
        step = 0.8 * target_sum / (np.sqrt(it + 1) * np.linalg.norm(g) + 1e-12)
        a -= step * g

        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        # tiny jitter every now and then
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1
    return a


# ----------------------------------------------------------------------
# Aggressive mass‑balance move
# ----------------------------------------------------------------------


def _mass_balance_refine_enhanced(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Greedy move that removes a *moderate* fraction of the coefficient that
    contributes most to the current maximal convolution entry and places it
    at a location that contributes the least.  The fraction is chosen
    randomly between 8 % and 25 % of the heavy coefficient.
    """
    a = a0.copy()
    n = a.shape[0]
    target = np.sum(a)               # should be √(2 n)
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _max_convolution_fft(a)
        k = int(np.argmax(conv))

        # contributions of each index to entry k
        idx = np.arange(n)
        other = k - idx
        mask = (other >= 0) & (other < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[other[mask]]

        i_high = int(np.argmax(contrib))

        # pick a low‑contribution index (among the 20 % smallest)
        low_thr = np.percentile(contrib[mask], 20)
        low_cands = np.where(contrib <= low_thr)[0]
        if low_cands.size == 0:
            i_low = int(np.argmin(a))
        else:
            i_low = int(rng.choice(low_cands))

        # transfer a random fraction of the heavy mass
        frac = 0.08 + 0.17 * rng.random()        # between 0.08 and 0.25
        delta = a[i_high] * frac
        if delta < 1e-12:
            break

        a[i_high] -= delta
        a[i_low] += delta

        # re‑project only if numerical drift occurs
        a = np.maximum(a, 0.0)
        current_sum = np.sum(a)
        if abs(current_sum - target) > 1e-12:
            a = a * (target / current_sum)

    return a


# ----------------------------------------------------------------------
# Light simulated‑annealing hill climbing
# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Very light SA that repeatedly moves a random fraction of mass between two
    positions.  The temperature is initialised from the current score and
    decays geometrically.
    """
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    cur_score = _score(a)
    best_score = cur_score

    # temperature schedule
    T = cur_score * 0.02 + 1e-9
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-16:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        frac = rng.random() * 0.2          # up to 20 % of a[i]
        delta = a[i] * frac
        if delta < 1e-12:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = _score(a_new)
        dE = new_score - cur_score
        accept = dE <= 0.0 or (T > 0.0 and rng.random() < np.exp(-dE / T))

        if accept:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

        T *= cooling
    return best_a


# ----------------------------------------------------------------------
# Analytic families for the fast sweep (Stage 1)
# ----------------------------------------------------------------------


def _shape_powerlaw(n: int, p: float) -> np.ndarray:
    idx = np.arange(n, dtype=float) + 1.0
    raw = idx ** (-p)
    target = np.sqrt(2.0 * n)
    return raw * (target / raw.sum())


def _shape_exponential(n: int, lam: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.exp(-lam * idx)
    target = np.sqrt(2.0 * n)
    return raw * (target / raw.sum())


def _shape_powerexp(n: int, p: float, lam: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = (idx + 1.0) ** (-p) * np.exp(-lam * idx)
    target = np.sqrt(2.0 * n)
    return raw * (target / raw.sum())


def _shape_dirichlet(rng: np.random.Generator, n: int) -> np.ndarray:
    raw = rng.dirichlet(np.ones(n))
    target = np.sqrt(2.0 * n)
    return raw * target   # already sums to 1 → just scale


def _shape_piecewise_exp(rng: np.random.Generator, n: int) -> np.ndarray:
    # split point somewhere in the middle
    b = rng.integers(int(0.3 * n), int(0.7 * n))
    lam1 = rng.exponential(scale=0.02)
    lam2 = rng.exponential(scale=0.02)
    a1 = np.exp(-lam1 * np.arange(b))
    a2 = np.exp(-lam2 * np.arange(n - b))
    raw = np.concatenate([a1, a2])
    target = np.sqrt(2.0 * n)
    return raw * (target / raw.sum())


# ----------------------------------------------------------------------
# Stage 1 – deterministic sweep of families
# ----------------------------------------------------------------------


def _parameter_sweep(rng: np.random.Generator,
                     best_seq: np.ndarray | None,
                     best_score: float,
                     deadline: float):
    """Explore simple analytic families for several lengths."""
    lengths = [2000, 4000, 6000, 8000, 10000]

    for n in lengths:
        if time.time() >= deadline:
            break
        # ---------- power‑law ----------
        for p in np.linspace(0.20, 2.00, 15):
            if time.time() >= deadline:
                break
            a = _shape_powerlaw(n, p)
            sc = _score(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] power‑law p={p:.3f} n={n} score={sc:.6f}")

        # ---------- pure exponential ----------
        for lam in np.logspace(-3.0, -0.30, 12):
            if time.time() >= deadline:
                break
            a = _shape_exponential(n, lam)
            sc = _score(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] exponential λ={lam:.5f} n={n} score={sc:.6f}")

        # ---------- power × exponential ----------
        for p in np.linspace(0.40, 1.20, 5):
            for lam in np.logspace(-3.0, -0.30, 5):
                if time.time() >= deadline:
                    break
                a = _shape_powerexp(n, p, lam)
                sc = _score(a)
                if sc < best_score:
                    best_score, best_seq = sc, a.copy()
                    print(f"[{time.time():.1f}] pow×exp p={p:.2f} λ={lam:.5f} n={n} score={sc:.6f}")

        # ---------- Dirichlet ----------
        for _ in range(3):
            if time.time() >= deadline:
                break
            a = _shape_dirichlet(rng, n)
            sc = _score(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] Dirichlet n={n} score={sc:.6f}")

        # ---------- piecewise exponential ----------
        for _ in range(3):
            if time.time() >= deadline:
                break
            a = _shape_piecewise_exp(rng, n)
            sc = _score(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] piecewise‑exp n={n} score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stage 2 – iterative refinement until time runs out
# ----------------------------------------------------------------------


def _refine_sequence(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Cycle through the four local‑search primitives until time expires."""
    a = a0.copy()
    best_a = a.copy()
    best_sc = _score(a)

    while True:
        now = time.time()
        if now >= deadline - 5.0:            # keep a safety margin for the final call
            break
        remaining = deadline - now

        # ---- smooth‑max gradient descent ----
        sub_dead = min(deadline, now + min(20.0, remaining * 0.40))
        a = _smooth_max_descent(a, sub_dead)
        sc = _score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[{time.time():.1f}] smooth‑max improvement → {sc:.6f}")

        # ---- exact sub‑gradient descent ----
        now = time.time()
        if now >= deadline - 5.0:
            break
        remaining = deadline - now
        sub_dead = min(deadline, now + min(5.0, remaining * 0.20))
        a = _subgradient_descent_max(a, sub_dead)
        sc = _score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[{time.time():.1f}] sub‑gradient improvement → {sc:.6f}")

        # ---- aggressive mass‑balance ----
        now = time.time()
        if now >= deadline - 5.0:
            break
        remaining = deadline - now
        sub_dead = min(deadline, now + min(5.0, remaining * 0.20))
        a = _mass_balance_refine_enhanced(a, sub_dead)
        sc = _score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[{time.time():.1f}] mass‑balance improvement → {sc:.6f}")

        # ---- hill climbing (light SA) ----
        now = time.time()
        if now >= deadline - 5.0:
            break
        remaining = deadline - now
        sub_dead = min(deadline, now + min(5.0, remaining * 0.20))
        a = _hill_climbing(a, sub_dead)
        sc = _score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[{time.time():.1f}] hill‑climbing improvement → {sc:.6f}")

        # next cycle starts from the best found so far
        a = best_a.copy()

    return best_a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimizes
    ``evaluate_sequence``.  The routine respects a wall‑clock budget,
    performs a fast analytical sweep followed by intensive local refinement,
    and finally returns the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Seeding from a known construction, if available
    # ------------------------------------------------------------------
    earlier = globals().get("height_sequence_1")
    if isinstance(earlier, (list, tuple, np.ndarray)):
        arr = np.asarray(earlier, dtype=float)
        if arr.ndim == 1 and arr.size > 0:
            n0 = arr.size
            target0 = np.sqrt(2.0 * n0)
            arr = arr * (target0 / arr.sum())
            sc0 = _score(arr)
            best_seq, best_score = arr.copy(), sc0
            print(f"[{time.time():.1f}] seeded from height_sequence_1 n={n0} score={sc0:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – deterministic sweep of simple families
    # ------------------------------------------------------------------
    stage1_deadline = min(deadline,
                          time.time() + max(60.0, (deadline - time.time()) * 0.20))
    best_seq, best_score = _parameter_sweep(rng, best_seq, best_score, stage1_deadline)

    # ------------------------------------------------------------------
    # Stage 2 – intensive iterative refinement
    # ------------------------------------------------------------------
    if best_seq is not None:
        best_seq = _refine_sequence(best_seq, deadline)
        best_score = _score(best_seq)
        print(f"[{time.time():.1f}] after refinement best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation and cleanup
    # ------------------------------------------------------------------
    if best_seq is None:
        # very last resort – uniform distribution of length 1000
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    # Trim trailing almost‑zeros for a tidy output
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Report the official metric (the platform will also call it)
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:                     # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
