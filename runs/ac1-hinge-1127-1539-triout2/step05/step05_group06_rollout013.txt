   a = a0.copy()
   n = len(a); S = sqrt(2*n)
   best_a = a.copy(); best_score = max(_full_convolution(a))
   alpha = 5.0; # smoothing temperature
   iteration = 0
   while time.time() < deadline:
       iteration += 1
       conv = _full_convolution(a)
       # compute softmax weight
       w = _smooth_weights(conv, alpha)
       grad = _conv_with_weights(a, w) # gradient of smoothed max
       # learning rate schedule
       eta = 0.5 / np.sqrt(iteration)  # maybe 0.5 is base
       # EG update
       a = a * np.exp(-eta * grad)  # elementwise
       # renorm to simplex
       a_sum = a.sum()
       if a_sum == 0:

           a_sum = a.sum()

       if cur_score < best_score:
           best_score = cur_score
           best_a = a.copy()

           # optional early stop

"""
Optimisation of a non‑negative step sequence that minimises the
self‑convolution peak (the value returned by ``evaluate_sequence``).

Key ideas
---------
* Work on the L¹‑simplex  Σ a_i = √(2·n).  The objective is the maximum
  entry of the full convolution a*a.
* A two‑phase optimiser is used for every trial
    1. **Exponentiated‑gradient (mirror‑descent) on a smooth soft‑max
       approximation** of the max‑norm.  The soft‑max temperature is
       annealed from a smooth regime (α≈0.5) up to a very sharp one
       (α≈6000) so that the algorithm eventually targets the true peak.
       The updates are performed in log‑space, guaranteeing positivity
       and an exact Euclidean projection onto the simplex after each
       step.
    2. **Projected sub‑gradient descent** that directly attacks the
       worst convolution entry.  This refines the solution obtained by
       the first phase.
* The outer loop performs many short trials with different problem
  sizes.  Successful trials are up‑sampled (linear interpolation) to
  larger ``n`` – this gives the optimiser more degrees of freedom while
  preserving the shape that already works well.
* Random “mass‑swap’’ and jitter moves are injected regularly to escape
  plateaus.
* The routine runs for the full allotted budget (≈ 1000 s) and returns
  the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max approx. of max(b) w.r.t. a.
    Implements  g_j = 2 Σ_i w_{i+j} a_i .
    Both ``a`` (len n) and ``w`` (len 2·n‑1) are real vectors.
    """
    n = a.shape[0]
    # reverse w because convolution with reversed weights is needed
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights (smooth approximation of max)
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """w_i = exp(alpha·conv_i) / Σ_j exp(alpha·conv_j) (stable)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Upsampling helper (linear interpolation)
# ----------------------------------------------------------------------


def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a to length new_n."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=float)
    return np.interp(x_new, x_old, a)


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (soft‑max)
# ----------------------------------------------------------------------


def _optimize_exponential(a0: np.ndarray,
                          deadline: float,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent / exponentiated‑gradient optimisation of a soft‑max
    approximation of the peak of the self‑convolution.
    Returns the best vector found and its (non‑soft) peak value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # log‑domain representation guarantees positivity
    eps = 1e-12
    log_a = np.log(np.maximum(a, eps))

    best_a = None
    best_score = np.inf

    iter_cnt = 0
    alpha = 0.5        # start very smooth

    while time.time() < deadline:
        iter_cnt += 1

        # re‑normalise from log space (Euclidean projection onto the simplex)
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, S / n)
        else:
            a = a * (S / a_sum)

        # evaluate current peak
        conv = _full_convolution(a)
        cur_score = np.max(conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # compute soft‑max weights and gradient
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # step size (decreasing)
        eta = 0.4 / np.sqrt(iter_cnt)

        # multiplicative update in log‑space
        log_a -= eta * grad

        # occasional random mass‑swap (exploration)
        if rng.random() < 0.03:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                # transfer a tiny fraction of mass from i to j
                a_tmp = np.exp(log_a) * (S / np.exp(log_a).sum())
                delta = a_tmp[i] * rng.random() * 0.01
                a_tmp[i] -= delta
                a_tmp[j] += delta
                a_tmp = a_tmp * (S / a_tmp.sum())
                log_a = np.log(np.maximum(a_tmp, eps))

        # temperature schedule – make the soft‑max sharper
        alpha = min(alpha * 1.03, 6000.0)

        # cheap early‑stop
        if iter_cnt % 5000 == 0 and cur_score >= best_score * 0.9999:
            break

    # final projection of the best vector onto the simplex
    if best_a is None:
        best_a = np.full(n, S / n)
    else:
        best_a = best_a * (S / best_a.sum())
    final_peak = np.max(_full_convolution(best_a))
    return best_a, final_peak


# ----------------------------------------------------------------------
# Projected sub‑gradient optimiser (acts on the true max‑norm)
# ----------------------------------------------------------------------


def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that directly reduces the worst entry."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.25
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of max entry:
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional random perturbation
        if rng.random() < 0.05:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.02
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = np.max(_full_convolution(cand))
                if cand_score < cur_score or rng.random() < 0.2:
                    a = cand
                    cur_score = cand_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()

        # cheap early‑stop
        if it % 8000 == 0 and cur_score >= best_score * 0.9999:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimiser (exponential → sub‑gradient)
# ----------------------------------------------------------------------


def _optimize_combined(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run the exponentiated‑gradient phase for roughly 60 % of the time
    and finish with a projected sub‑gradient refinement.
    """
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    smooth_deadline = now + 0.6 * remaining
    a_mid, sc_mid = _optimize_exponential(a0, smooth_deadline, rng)

    if time.time() < deadline:
        a_final, sc_final = _optimize_subgrad(a_mid, deadline, rng)
        if sc_final < sc_mid:
            return a_final, sc_final
    return a_mid, sc_mid
