def _adam_optimize(a0, deadline, target_sum):

    a = a0.copy()

    m = np.zeros_like(a); v = np.zeros_like(a)

    beta1 = 0.9; beta2 = 0.999

    eps = 1e-8

    lr_base = 0.5 # scaling factor

    t = 0

    beta = 1.0

    max_beta = 1e6

    while time.time() < deadline:

        g, _ = _gradient_softmax(a, beta)

        t += 1

        m = beta1 * m + (1 - beta1) * g

        v = beta2 * v + (1 - beta2) * (g * g)

        m_hat = m / (1 - beta1 ** t)

        v_hat = v / (1 - beta2 ** t)

        # compute step size relative to sum magnitude; adapt lr
        step = lr_base / (np.sqrt(v_hat) + eps) * m_hat

        a -= step

        # projection onto simplex
        a = _project_to_simplex(a, target_sum)

        # adjust beta schedule occasionally
        if t % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.2, max_beta)

        # occasional jitter: every 500 steps add small noise
        if t % 500 == 0:
            jitter = np.random.default_rng().normal(scale=target_sum * 1e-4, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

    return a

"""
Improved search for a non‑negative step function that minimizes
``evaluate_sequence``.  The algorithm consists of

1. **Candidate generation** – several parametric families (power‑law,
   exponential, triangular, symmetric power‑law) and random Dirichlet
   vectors are produced for a handful of lengths (4000…10000).  The
   candidates are projected onto the ℓ¹‑simplex Σa = √(2 n).

2. **Adam‑style projected optimisation** – each elite candidate is
   refined by minimising a smooth log‑sum‑exp surrogate of the maximal
   convolution entry.  The optimiser uses Adam updates, an annealed
   temperature β (smoothness) and simplex projection after every step.

3. **Simulated‑annealing polishing** – a short SA phase with mass‑transfer
   moves finishes the refinement.

The whole search respects the given time budget (default 1000 s) and
always returns the best sequence found.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex Σx = s, x≥0
# ------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection onto {x_i ≥ 0, Σx_i = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Smoothed (log‑sum‑exp) surrogate of max(b) and its gradient
# ------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b) where
        b = a*a (full convolution)
        g = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a ⋆ p) ,  p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]

    # ----- convolution a*a -----
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # ----- soft‑max probabilities -----
    max_b = np.max(b)
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)

    # ----- gradient g = 2·(a ⋆ p) -----
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ------------------------------------------------------------
# Adam‑style projected optimisation on the smoothed maximum
# ------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   target_sum: float) -> np.ndarray:
    """Projected Adam on the log‑sum‑exp surrogate."""
    a = a0.copy()
    n = a.size

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow = 1.0
    beta2_pow = 1.0

    # learning‑rate (relative to target_sum)
    lr = 0.1 * target_sum

    # smoothness parameter β
    beta = 1.0
    max_beta = 1e6
    beta_mul = 1.12          # growth factor per schedule step

    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # gradient of smooth max
        g, _ = _gradient_softmax(a, beta)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        beta1_pow *= beta1
        beta2_pow *= beta2

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep the step from being too aggressive
        norm_step = np.linalg.norm(step)
        max_norm = target_sum * 0.5
        if norm_step > max_norm:
            step = step * (max_norm / norm_step)

        a -= step
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # increase smoothness slowly
        if it % 150 == 0 and beta < max_beta:
            beta = min(beta * beta_mul, max_beta)

        it += 1
    return a


# ------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Random mass‑transfer SA with multiplicative cooling."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # temperature initialised from current objective
    T = best_score * 0.02 + 1e-9
    cooling = 0.9997
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        # transfer a fraction (up to 15 %) of mass from i to j
        frac = rng.random() * 0.15
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        accept = False
        if cur_score <= best_score:
            accept = True
        elif T > 0.0 and rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
    return best


# ------------------------------------------------------------
# Parametric families
# ------------------------------------------------------------


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    target_sum = np.sqrt(2.0 * n)
    scale = target_sum / raw.sum()
    a = raw * scale
    return np.maximum(a, 0.0)


def _symmetric_powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    left = (idx + shift) ** (-p)
    right = (n - 1 - idx + shift) ** (-p)
    raw = left + right
    target_sum = np.sqrt(2.0 * n)
    scale = target_sum / raw.sum()
    a = raw * scale
    return np.maximum(a, 0.0)


def _exponential_sequence(n: int, lam: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.exp(-lam * idx)
    target_sum = np.sqrt(2.0 * n)
    scale = target_sum / raw.sum()
    a = raw * scale
    return np.maximum(a, 0.0)


def _triangular_sequence(n: int) -> np.ndarray:
    """Symmetric triangle: peak at both ends, valley in the middle."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(half, 0, -1, dtype=float)
        right = np.arange(1, half + 1, dtype=float)
        raw = np.concatenate([left, right])
    else:
        left = np.arange(half + 1, 0, -1, dtype=float)
        right = np.arange(1, half + 1, dtype=float)
        raw = np.concatenate([left, right])
    target_sum = np.sqrt(2.0 * n)
    scale = target_sum / raw.sum()
    a = raw * scale
    return np.maximum(a, 0.0)


def _interpolate_sequence(seq: np.ndarray, n_new: int) -> np.ndarray:
    """Resize a sequence by linear interpolation and re‑project onto the simplex."""
    old_n = seq.size
    if old_n == n_new:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, n_new)
    raw = np.interp(x_new, x_old, seq)
    target_sum = np.sqrt(2.0 * n_new)
    return _project_to_simplex(np.maximum(raw, 0.0), target_sum)

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises the target
    ``evaluate_sequence`` objective.  The routine respects the allotted
    time budget, generates a diverse set of parametric seeds, refines the
    elite ones with Adam‑style optimisation and a short simulated‑annealing
    phase, and finally returns the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    deadline = time.time() + budget_s - 0.5  # safety margin

    best_seq = None
    best_obj = np.inf
    elite: list[tuple[float, np.ndarray]] = []   # (obj, vector)

    # ------------------------------------------------------------------
    # 1. Candidate generation (different families, several lengths)
    # ------------------------------------------------------------------
    lengths = [4000, 5000, 6000, 8000, 10000]   # modest set of lengths

    # try to reuse a previously found construction if available
    prev_best = globals().get('height_sequence_1', None)
    if isinstance(prev_best, (list, tuple, np.ndarray)):
        prev_best = np.asarray(prev_best, dtype=float)

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)

        # ---- deterministic families -------------------------------------------------
        # power‑law
        for p in np.linspace(0.45, 0.80, 9):
            for shift in np.linspace(0.5, 5.0, 7):
                a = _powerlaw_sequence(n, p, shift)
                obj = _max_convolution_fft(a)
                if obj < best_obj:
                    best_obj, best_seq = obj, a.copy()
                    print(f"[{time.time():.1f}] New global best (pow) n={n} p={p:.3f} shift={shift:.2f} obj={obj:.6f}")

                # symmetric variant
                a_sym = _symmetric_powerlaw_sequence(n, p, shift)
                obj_sym = _max_convolution_fft(a_sym)
                if obj_sym < best_obj:
                    best_obj, best_seq = obj_sym, a_sym.copy()
                    print(f"[{time.time():.1f}] New global best (sym) n={n} p={p:.3f} shift={shift:.2f} obj={obj_sym:.6f}")

                # keep elite
                if len(elite) < 12 or obj < elite[-1][0]:
                    elite.append((obj, a))
                    elite.sort(key=lambda x: x[0])
                    elite = elite[:12]
                if len(elite) < 12 or obj_sym < elite[-1][0]:
                    elite.append((obj_sym, a_sym))
                    elite.sort(key=lambda x: x[0])
                    elite = elite[:12]

        # exponential family
        for lam in np.linspace(5e-4, 2e-2, 9):
            a = _exponential_sequence(n, lam)
            obj = _max_convolution_fft(a)
            if obj < best_obj:
                best_obj, best_seq = obj, a.copy()
                print(f"[{time.time():.1f}] New global best (exp) n={n} λ={lam:.5f} obj={obj:.6f}")
            if len(elite) < 12 or obj < elite[-1][0]:
                elite.append((obj, a))
                elite.sort(key=lambda x: x[0])
                elite = elite[:12]

        # triangular shape
        a_tri = _triangular_sequence(n)
        obj_tri = _max_convolution_fft(a_tri)
        if obj_tri < best_obj:
            best_obj, best_seq = obj_tri, a_tri.copy()
            print(f"[{time.time():.1f}] New global best (tri) n={n} obj={obj_tri:.6f}")
        if len(elite) < 12 or obj_tri < elite[-1][0]:
            elite.append((obj_tri, a_tri))
            elite.sort(key=lambda x: x[0])
            elite = elite[:12]

        # random Dirichlet‑like candidates (few per length)
        for _ in range(5):
            a_rand = rng.random(n)
            a_rand = a_rand / a_rand.sum() * target_sum
            obj_rand = _max_convolution_fft(a_rand)
            if obj_rand < best_obj:
                best_obj, best_seq = obj_rand, a_rand.copy()
                print(f"[{time.time():.1f}] New global best (rand) n={n} obj={obj_rand:.6f}")
            if len(elite) < 12 or obj_rand < elite[-1][0]:
                elite.append((obj_rand, a_rand))
                elite.sort(key=lambda x: x[0])
                elite = elite[:12]

        # reuse previous construction (if any) by interpolation
        if prev_best is not None:
            a_interp = _interpolate_sequence(prev_best, n)
            obj_interp = _max_convolution_fft(a_interp)
            if obj_interp < best_obj:
                best_obj, best_seq = obj_interp, a_interp.copy()
                print(f"[{time.time():.1f}] New global best (interp) n={n} obj={obj_interp:.6f}")
            if len(elite) < 12 or obj_interp < elite[-1][0]:
                elite.append((obj_interp, a_interp))
                elite.sort(key=lambda x: x[0])
                elite = elite[:12]

    # ------------------------------------------------------------------
    # 2. Refinement of elite candidates (Adam + SA)
    # ------------------------------------------------------------------
    # sort elite one more time (just in case)
    elite.sort(key=lambda x: x[0])
    for rank, (_, seed_vec) in enumerate(elite):
        if time.time() >= deadline:
            break
        n = seed_vec.size
        target_sum = np.sqrt(2.0 * n)

        # allocate at most 40 s per candidate, respecting the global deadline
        remaining = deadline - time.time()
        per = min(40.0, remaining / max(1, len(elite) - rank))
        sub_dead = time.time() + per

        # Adam optimisation
        a_adam = _adam_optimize(seed_vec, sub_dead, target_sum)

        # Simulated‑annealing (uses the global deadline, may finish earlier)
        a_sa = _simulated_annealing(a_adam, deadline)

        obj = _max_convolution_fft(a_sa)
        if obj < best_obj:
            best_obj, best_seq = obj, a_sa.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL BEST (refined) n={n} rank={rank} obj={obj:.6f}")

    # ------------------------------------------------------------------
    # 3. Final polishing if time remains
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        target_sum = np.sqrt(2.0 * n)
        final_dead = deadline
        best_seq = _adam_optimize(best_seq, final_dead, target_sum)
        best_seq = _simulated_annealing(best_seq, deadline)
        best_obj = _max_convolution_fft(best_seq)
        print(f"[{time.time():.1f}] Final polishing result obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 4. Post‑processing and return
    # ------------------------------------------------------------------
    if best_seq is None:
        # fall‑back: uniform distribution of length 4000
        n = 4000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)

    # Trim trailing near‑zero entries for neatness
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {final_val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective = {best_obj:.6f}, length = {best_seq.size}")
    return best_seq.tolist()
