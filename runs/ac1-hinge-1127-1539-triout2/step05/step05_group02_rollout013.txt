    n = f.size
    offsets = np.arange(n, dtype=int)
    data = [np.full(n, f_i) for f_i in f]  # list of arrays

    for j in range(n):
        A[j:j+n, j] = f
    # constraints: A @ g <= M (vector)

    return res.x

"""
Propose a step‑function (non‑negative height sequence) with a low benchmark score.
The routine combines three optimisation phases:

1.  Warm‑start with a power‑law shape and a few seconds of projected‑gradient
    descent on a soft‑max approximation of the max‑convolution.
2.  A sequence of linear‑programming refinements (the “LP‑step” described in the
    problem statement).  Each LP maximises the total mass of a new candidate
    g under the constraint ‖f * g‖∞ ≤ M, where M is the current max‑convolution.
    The candidate is scaled to the required ℓ¹‑norm and then merged with the
    current best f via a back‑tracking line search that guarantees a monotone
    decrease of the max‑convolution.
3.  A final simulated‑annealing stage for fine‑grained improvement.

All steps respect the ℓ¹‑normalisation Σa = √(2·n), which makes the benchmark
score equal to the maximum entry of the self‑convolution.  The algorithm is
designed to finish well inside the 1000 s budget and to return the best
sequence discovered.
"""

import time
import math
import numpy as np
import scipy.optimize
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Fast convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (using FFT)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (using FFT)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Benchmark score for a (here just the max convolution, because Σa²=2n)."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – fixes the ℓ¹‑norm
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple initial families
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Gradient descent on a soft‑max approximation of max‑convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient of soft‑max w.r.t. a
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _annealed_gradient_descent(a0: np.ndarray, deadline: float):
    """Projected gradient descent on the soft‑max approximation."""
    a = a0.copy()
    n = a.size
    target_sum = math.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 2048.0          # a sharper approximation than the original code
    step_base = target_sum * 0.20
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            lr = step_base / (norm_g * (beta + 1.0))
            a -= lr * g

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] GD improvement: β={beta:.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement (the “LP‑step”)
# ----------------------------------------------------------------------


def _lp_optimize(f: np.ndarray, M: float, target_sum: float):
    """
    Solve  max Σ g_j   s.t.  (f * g)_k ≤ M  ∀k,  g ≥ 0.
    Returns the optimal g (un‑scaled).  Returns None if the solver fails.
    """
    n = f.size
    rows = 2 * n - 1

    # Build the (rows × n) convolution matrix A such that A @ g = f * g.
    # Column j of A is f placed starting at row j.
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f

    b = np.full(rows, M, dtype=np.float64)          # upper bound for each convolution entry
    c = -np.ones(n, dtype=np.float64)              # maximise Σg → minimise –Σg
    bounds = [(0.0, None)] * n

    # Use the high‑performance HiGHS interior‑point solver.
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                                method='highs', options={'presolve': True, 'time_limit': 30})
    if not res.success:
        return None
    return res.x

# Simulated‑annealing refinement (identical to the previous version)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float,
                         start_temp: float = 0.5,

    cur_score = _score(a)

    best_score = cur_score

    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        max_b = conv.max()
        k_max = int(np.argmax(conv))

        # sensitivities for the active (maximum) convolution entry
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]

        if rng.random() < 0.70:      # mass transfer

                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        elif rng.random() < 0.90:    # local smoothing
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                old_seg = a[start:end].copy()
                avg = old_seg.mean()
                a[start:end] = avg
                move_type = "smooth"
                saved_seg = old_seg
                saved_start, saved_end = start, end

        else:                         # random swap
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # evaluate -----------------------------------------------------------
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = _score(a)

        # Metropolis acceptance -----------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA new best = {best_score:.6f}")
        else:
            # revert change
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional jitter to escape deep minima
        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = a * (a.sum() / a.sum())   # renormalise (no change, just defensive)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point – orchestrates the whole search
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (height sequence) that minimises the
    benchmark score.  The function respects the 1000 s time budget and returns
    the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0            # small safety margin

    # ------------------------------------------------------------------
    # Parameters – we keep n moderate (1500) because larger n raises LP size.
    # ------------------------------------------------------------------
    n = 1500
    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Initialise – use the globally supplied sequence if it matches the length,
    # otherwise fall back to a power‑law start.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if a.size != n:
            a = _power_law_sequence(n, 0.55, target_sum)
    else:
        a = _power_law_sequence(n, 0.55, target_sum)

    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_M = _max_convolution_fft(a)
    print(f"[{time.time():.1f}] initial max‑conv = {best_M:.6f}")

    # ------------------------------------------------------------------
    # 1) Short projected‑gradient descent phase (up to ~30 % of the budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 60.0:
        gd_deadline = time.time() + min(150.0, remaining * 0.35)
        a_gd, gd_M = _annealed_gradient_descent(a, gd_deadline)
        if gd_M < best_M:
            best_M = gd_M
            best_a = a_gd.copy()
            print(f"[{time.time():.1f}] after GD best max‑conv = {best_M:.6f}")
        a = a_gd

    # ------------------------------------------------------------------
    # 2) Linear‑programming refinement loop (aim for monotone descent)
    # ------------------------------------------------------------------
    while time.time() < deadline - 20.0:
        cur_M = _max_convolution_fft(a)
        g0 = _lp_optimize(a, cur_M, target_sum)
        if g0 is None:
            break
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            break

        # scale g0 to the required ℓ¹‑norm
        g = g0 * (target_sum / sum_g0)

        # back‑tracking line search on the convex combination parameter t
        t = 0.5
        improved = False
        while t > 1e-4:
            a_new = (1.0 - t) * a + t * g
            M_new = _max_convolution_fft(a_new)
            # accept if we do not increase the max‑convolution
            if M_new <= cur_M * (1.0 + 1e-12):
                a = a_new
                improved = True
                if M_new < best_M:
                    best_M = M_new
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] LP step accepted (t={t:.3f}) max‑conv={best_M:.6f}")
                break
            t *= 0.5

        if not improved:
            # If no improvement, inject a tiny random jitter and continue
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)
            print(f"[{time.time():.1f}] jitter after failed LP step")

        if time.time() > deadline - 20.0:
            break

    # ------------------------------------------------------------------
    # 3) Final simulated‑annealing refinement (uses remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_sa, sa_M = _simulated_annealing(best_a, deadline)
        if sa_M < best_M:
            best_M = sa_M
            best_a = a_sa.copy()
            print(f"[{time.time():.1f}] SA final best max‑conv = {best_M:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing zeros and return the best sequence
    # ------------------------------------------------------------------
    nz = np.where(best_a > 1e-12)[0]
    if nz.size:
        best_a = best_a[: nz[-1] + 1]

    print(f"[{time.time():.1f}] final best max‑conv = {best_M:.6f}, length = {len(best_a)}")
    return best_a.tolist()
