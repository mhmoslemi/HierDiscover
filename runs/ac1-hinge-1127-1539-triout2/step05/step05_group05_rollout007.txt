"""
Propose a low‑max‑convolution step function.

The algorithm works in three stages under a user‑supplied time budget
(default 1000 s):

1. **Exploratory scan** – for several candidate lengths (800 … 3000)
   a large number of random sequences are generated from diverse
   families (power‑law, exponential, Dirichlet, symmetric edge‑heavy,
   Beta‑shaped, Gaussian, linear, U‑shaped).  Each sequence is projected
   onto the ℓ¹‑simplex Σa_i = √(2 n) and its exact maximal self‑convolution
   entry is recorded.  The best sequence for each length is kept.

2. **Adam‑based smoothed‑max optimisation** – the best candidates from
   stage 1 (and a few fresh random restarts) are refined by a
   projected Adam optimiser.  The optimiser minimises the smooth
   approximation  (1/β)·log Σ exp(β·b_k)  of the ℓ∞‑norm of the
   self‑convolution, where b = a∗a.  The smoothing parameter β is
   increased geometrically (β←β·1.07 every 200 steps) up to 10⁷,
   so the optimisation converges to the true max.  After each Adam
   step the iterate is projected back onto the non‑negative simplex.

3. **Targeted simulated annealing (SA)** – the current best sequence is
   further polished by a specialised SA that repeatedly moves a
   fraction of mass from an index that heavily contributes to the
   current maximal convolution entry to an index that is unlikely to
   increase it.  Temperature decays multiplicatively; uphill moves are
   accepted with the Metropolis probability.  The SA runs for the
   remaining time budget.

All helper functions are top‑level, no file I/O is performed and the
search respects the supplied time budget.  The returned value is a
plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x_i ≥ 0, Σx_i = s}
# (Duchi et al. 2008)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x_i≥0, Σx_i = s}."""
    if s <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) of the self‑convolution and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Return (g, b) where
        b = a*a (full convolution, length 2·n‑1)
        g = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a ⋆ p)   with   p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]

    # ----- b = a*a ----------------------------------------------------
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ----- soft‑max probabilities ---------------------------------------
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))          # stable exp
    p = exps / np.sum(exps)

    # ----- gradient g = 2·(a ⋆ p) --------------------------------------
    a_rev = a[::-1]                            # reversed a
    conv_len = n + (2 * n - 1) - 1              # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]        # central n entries
    return g, b


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed max (projected onto the simplex)
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected Adam optimiser for the smoothed max of the self‑convolution.
    The iterate stays on the ℓ¹‑simplex Σa_i = √(2·n) and a_i ≥ 0.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.05                     # base learning rate
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # smoothing schedule for the soft‑max
    beta = 0.5
    beta_max = 1e7

    step = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        step += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)

        # adaptive step (Adam)
        eta = lr0 / np.sqrt(step)          # mild decay for stability
        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # enforce non‑negativity
        a = np.maximum(a, 0.0)

        # simplex projection (keeps Σa = √(2 n))
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # increase smoothing parameter geometrically
        if step % 200 == 0 and beta < beta_max:
            beta = min(beta * 1.07, beta_max)

    return a


# ----------------------------------------------------------------------
# Targeted simulated annealing (identical to the previous implementation,
# slightly tweaked for speed)
# ----------------------------------------------------------------------


def _targeted_simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    SA that repeatedly moves a fraction of mass from an index that
    heavily contributes to the current maximal convolution entry to a
    random destination.  Temperature decays multiplicatively.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # initial temperature proportional to current objective
    T = best_score * 0.02 + 1e-8
    cooling = 0.9995
    rng = np.random.default_rng()
    max_transfer = 0.25  # up to 25 % of the selected mass

    while time.time() < deadline:
        # current convolution and its maximal index
        b = _full_convolution_fft(a)
        k_max = int(np.argmax(b))

        # indices that actually affect b[k_max]
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idx_range = np.arange(lo, hi + 1)                     # possible i
        contrib = a[idx_range] * a[k_max - idx_range]        # a_i * a_{k‑i}
        weight_sum = np.sum(contrib)

        if weight_sum > 0.0:
            probs = contrib / weight_sum
            i = rng.choice(idx_range, p=probs)
        else:
            i = rng.integers(0, n)

        # pick a destination j ≠ i uniformly
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        # transfer a random fraction of the mass at i to j
        delta = a[i] * rng.random() * max_transfer
        if delta <= 0.0:
            continue
        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score

        accept = False
        if cur_score <= best_score:
            accept = True
        elif T > 0.0 and rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA improvement: {best_score:.6f}")
        else:
            # revert the move
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # occasional global jitter to escape deep minima
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 0:
                a = a * (np.sqrt(2.0 * n) / a_sum)
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA jitter improvement: {best_score:.6f}")

    return best


# ----------------------------------------------------------------------
# Random initialisers (a richer collection of families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Generate a random non‑negative sequence on the ℓ¹‑simplex
    Σ a_i = √(2 n).  Several families are mixed to increase diversity:
    – pure power‑law (decreasing)
    – exponential decay
    – mixed power‑law + exponential
    – Dirichlet (uniform simplex)
    – symmetric edge‑heavy
    – Beta‑shaped (U‑ or bell‑shaped)
    – Gaussian bump (center heavy)
    – linear increase
    – U‑shaped (inverse Gaussian)
    """
    target_sum = np.sqrt(2.0 * n)
    kind = rng.integers(0, 9)  # 0 … 8
    idx = np.arange(n, dtype=float)

    if kind == 0:                     # pure power‑law (decreasing)
        p = rng.uniform(0.15, 1.0)
        a = (idx + 1.0) ** (-p)

    elif kind == 1:                   # exponential decay from the left edge
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif kind == 2:                   # mixed power‑law + exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif kind == 3:                   # Dirichlet (uniform simplex)
        a = rng.dirichlet(alpha=np.ones(n))

    elif kind == 4:                   # symmetric edge‑heavy
        p = rng.uniform(0.2, 1.2)
        left = (idx + 1.0) ** (-p)
        right = (n - idx) ** (-p)
        a = left + right

    elif kind == 5:                   # Beta‑shaped
        alpha = rng.uniform(0.2, 2.0)
        beta = rng.uniform(0.2, 2.0)
        x = (idx + 0.5) / n
        a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))

    elif kind == 6:                   # Gaussian bump (center heavy)
        sigma = rng.uniform(0.08, 0.25) * n
        centre = (n - 1) / 2.0
        a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)

    elif kind == 7:                   # linear increase (edge heavy on right)
        a = idx + 1.0

    else:                             # U‑shaped (inverse quadratic)
        centre = (n - 1) / 2.0
        a = 1.0 / (1.0 + ((idx - centre) / (0.4 * n)) ** 2)

    # small Gaussian perturbation and projection onto the simplex
    a += rng.normal(scale=target_sum * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target_sum)
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative sequence) that minimises
    ``evaluate_sequence`` under the supplied time budget.  The routine
    performs a quick multi‑length scan, refines promising candidates with
    a projected Adam optimiser, and finishes with targeted simulated
    annealing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_obj = np.inf
    # keep per‑length bests during the scan phase
    per_len_best = {}

    # ------------------------------------------------------------------
    # Phase 1 – quick random scan over several lengths
    # ------------------------------------------------------------------
    lengths = [800, 1000, 1200, 1500, 1800, 2000, 2500, 3000]
    scan_time_per_len = 5.0  # seconds dedicated to each length

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)
        end_phase = min(deadline, time.time() + scan_time_per_len)

        while time.time() < end_phase:
            a = _random_initial_sequence(n, rng)
            # exact max self‑convolution (objective before normalisation)
            obj = _max_convolution_fft(a)
            if obj < best_obj:
                best_seq, best_obj = a.copy(), obj
                print(f"[{time.time():.1f}] New global best (scan) n={n}  obj={obj:.6f}")

            # record best per length
            if n not in per_len_best or obj < per_len_best[n][0]:
                per_len_best[n] = (obj, a.copy())

    # ------------------------------------------------------------------
    # If nothing was generated (unlikely), fall back to uniform.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1200
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_obj = _max_convolution_fft(best_seq)

    # ------------------------------------------------------------------
    # Phase 2 – intensive Adam optimisation on promising lengths
    # ------------------------------------------------------------------
    # Choose the three lengths with the lowest recorded obj for deeper work
    promising = sorted(per_len_best.items(), key=lambda kv: kv[1][0])[:3]
    # allocate roughly half of the remaining budget to Adam
    remaining = deadline - time.time()
    if remaining > 20.0:
        adam_budget = remaining * 0.55
        # time for each Adam run (seconds)
        time_per_adam = 6.0
        while time.time() < deadline - 5.0:
            # pick a promising length (cycling)
            n = promising[rng.integers(0, len(proposing := promising))][0]
            # initialise either from the stored best of that length or from a fresh random start
            if rng.random() < 0.6 and n in per_len_best:
                a0 = per_len_best[n][1].copy()
            else:
                a0 = _random_initial_sequence(n, rng)

            sub_deadline = min(deadline, time.time() + time_per_adam)
            a_opt = _adam_optimize(a0, sub_deadline)
            obj = _max_convolution_fft(a_opt)

            if obj < best_obj:
                best_seq, best_obj = a_opt.copy(), obj
                print(f"[{time.time():.1f}] New best after Adam n={n}  obj={obj:.6f}")

            # keep the best per‑length record for potential later use
            if n not in per_len_best or obj < per_len_best[n][0]:
                per_len_best[n] = (obj, a_opt.copy())

    # ------------------------------------------------------------------
    # Phase 3 – targeted simulated annealing refinement (uses leftover time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        a_sa = _targeted_simulated_annealing(best_seq, deadline)
        final_obj = _max_convolution_fft(a_sa)
        if final_obj < best_obj:
            best_seq, best_obj = a_sa.copy(), final_obj
            print(f"[{time.time():.1f}] New best after SA  obj={final_obj:.6f}")

    # ------------------------------------------------------------------
    # Strip trailing near‑zero entries and return plain Python list
    # ------------------------------------------------------------------
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Optional sanity check with the official evaluator
    try:
        ev = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence = {ev:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective (max‑conv) = "
          f"{best_obj:.6f}   length = {best_seq.size}")

    return best_seq.tolist()
