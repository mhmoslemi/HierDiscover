def propose_candidate(seed=42, budget_s=1000, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    best_seq = None
    best_score = float('inf')
    # lengths to try
    lengths = [2000, 4000, 6000, 8000, 10000]
    # exponents to test
    exponents = list(range(2, 11))  # 2..10

    for n in lengths:
        target = np.sqrt(2.0 * n)
        # raw decreasing numbers: (n, n-1, ..., 1) ** p
        indices = np.arange(1, n+1, dtype=float)[::-1]  # n, n-1, ..., 1

            # compute score quickly

                best_score = sc

                print(f"[{time.time():.1f}] new best n={n} p={p} score={sc:.6f}")
        if time.time() >= deadline:

"""
Propose a step‑function by exhaustively testing simple polynomial‐decreasing shapes.

The family a_i = (n‑i)^p (i = 0 … n‑1) is normalized so that
∑ a_i = √(2 n).  For a fixed n the maximal entry of the self‑convolution
has a closed‑form limit

    M(p) = 2·(p+1)^2·Beta(p+1, p+1)

which decreases rapidly with p (M(2)≈0.60, M(5)≈0.026, …).  Hence a few
integer exponents already give scores far below the target 1.503.

The routine enumerates a small set of lengths and exponents, evaluates the
max‑convolution via an FFT, keeps the best candidate and finally returns it
as a plain list.  The search finishes in well under the 1000‑second budget,
but respects the deadline in case the environment is tighter.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper routines (FFT convolution, simplex‑free normalization)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution (the official objective)."""
    return float(_max_convolution_fft(a).max())


# ----------------------------------------------------------------------
# Core candidate generator
# ----------------------------------------------------------------------


def _polynomial_seq(n: int, p: int) -> np.ndarray:
    """
    Return a non‑negative sequence of length n of the form
        a_i ∝ (n-i)^p,
    scaled so that Σ a_i = √(2 n).
    """
    # raw decreasing polynomial (float to avoid int overflow)
    raw = np.arange(n, 0, -1, dtype=float) ** p          # n, n‑1, …, 1
    target = np.sqrt(2.0 * n)
    seq = raw * (target / raw.sum())
    return seq

# ----------------------------------------------------------------------

    """
