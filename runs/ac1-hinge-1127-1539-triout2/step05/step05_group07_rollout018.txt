def propose_candidate(seed=42, budget_s=1000):
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5

    best_seq = None
    best_score = np.inf

    # maybe seed from global height_sequence_1
    base_seq = np.asarray(globals().get('height_sequence_1', []), dtype=float)
    if base_seq.size > 0:
        # Evaluate base candidate
        base_score = evaluate_sequence(base_seq.tolist())
        best_seq = base_seq.tolist()
        best_score = base_score

    # define list of possible lengths
    lengths = [1500, 2500, 4000, 6000, 8000, 12000, 16000, 20000]

    #_main loop
    while time.time() < end_time:
        # choose n randomly or round-robin among lengths
        n = rng.choice(lengths)
        # random approach among shape families
        shape_type = rng.choice(['edge_power', 'sqrt_series', 'uniform', 'random'])
        if shape_type == 'edge_power':
            # pick p from uniform range
            p = rng.uniform(0.3, 1.8)
            a0 = generate_edge_power(n, p)
        elif shape_type == 'sqrt_series':
            a0 = generate_sqrt_series(n)
        elif shape_type == 'uniform':
            a0 = np.full(n, 1.0)
        else:
            a0 = rng.random(n)

        # scale to sum sqrt(2n)
        a0 = scale_sequence_sum(a0, np.sqrt(2.0 * n))

        # optionally jitter / random mixture
        if base_seq.size > 0 and rng.random() < 0.1:
            # mix with base via interpolation
            a0 = (a0 + _scale_sequence(base_seq, n)) / 2

        # now run optimizer
        # allocate time slice for this restart (e.g., 5-15 seconds)
        slice_deadline = min(time.time() + rng.uniform(4, 12), end_time)

        a_opt, peak = _adam_optimize(a0, slice_deadline, rng)

        # compute final score
        score = 2.0 * n * peak / (np.sum(a_opt) ** 2)

        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[improve] n={n} shape={shape_type} p={p if shape_type=='edge_power' else ''} score={score:.6f}")

    # final sanity check
    if best_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()
        external = evaluate_sequence(final_seq)
        print(f"[final] external score = {external:.6f} internal score = {best_score:.6f} length = {len(final_seq)}")
        globals()['height_sequence_1'] = final_seq

    return final_seq

"""
adversarial step‑function optimizer (v2)

The optimizer searches for a non‑negative vector a of length n that minimizes

    score = 2·n·max(a*a) / (∑ a)²

(the same quantity that ``evaluate_sequence`` returns).  The code combines

* a fast FFT based convolution,
* a τ‑soft‑max surrogate for the ∞‑norm,
* an Adam optimiser on the simplex (log‑softmax parametrisation),
* a geometric cooling of τ and the learning‑rate,
* a mirrored‑vector constraint (a[i] = a[n‑1‑i]) – all optimal solutions are
  symmetric, this reduces the search space,
* cheap heuristic initialisations (edge‑heavy power law, square‑root series,
  random Dirichlet, uniform) and a short random‑jitter phase,
* a multi‑restart loop that spends the available time budget (≈1000 s)
  on many independent attempts and finally refines the best candidate.

The routine returns a Python list of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic building blocks
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _scale_to_sum(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale a non‑negative vector so that its ℓ₁‑norm equals target_sum."""
    cur = np.sum(seq)
    if cur == 0.0:
        return seq
    return seq * (target_sum / cur)


def _edge_power_seq(n: int, p: float) -> np.ndarray:
    """a_i ∝ (dist_to_edge+1)^(-p)   (symmetrised)."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1          # 1‑based distance to nearest edge
    a = d.astype(float) ** (-p)
    return a


def _sqrt_series_seq(n: int) -> np.ndarray:
    """
    Coefficients of (1‑x)^(-1/2) = Σ (2k choose k)/4^k x^k,
    truncated at length n.  These are the classic
    a_k = C(2k, k)/4^k  ≃ 1/√(π k).
    """
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2 * k - 1) / (2 * k)
    return a


def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Returns (g, max_b) where g = ∂/∂a  τ‑soft‑max(b)  and max_b is the
    (hard) maximum of b = a*a.  The computation uses FFTs and works in O(n log n).
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        # Exact sub‑gradient: give weight 1 to every entry attaining the maximum.
        mask = (b == max_b).astype(float)
    else:
        # τ‑soft‑max weights (the exponential form of log‑sum‑exp).
        diff = b - max_b                     # ≤ 0
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)           # shape (2n‑1,)

    # ∂b/∂a = 2 * (mask * a) ⊛ a_rev    (⊛ = linear convolution)
    a_rev = a[::-1]
    conv_len = mask.shape[0] + n - 1
    f_len = _next_pow2(conv_len)

    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]

    g = 2.0 * conv[n - 1:n - 1 + n]        # gradient has length n
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex; symmetric sequences are enforced.
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Minimise the τ‑soft‑max surrogate of max(a*a) while keeping a on the
    probability simplex (∑a = √(2n)).  Returns the best vector found and its
    hard peak value.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    # Initialise logits from a0 (so that softmax(p)·S == a0)
    eps = 1e-12
    init = np.maximum(a0 / S, eps)
    init = init / init.sum()
    p = np.log(init)                       # logits, unrestricted

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.25                               # a bit larger than the old code

    # start temperature τ with the current hard maximum
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 1.0)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ---------- obtain a from logits (softmax) ----------
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)            # subtract max for numeric stability
        soft = exp_p / exp_p.sum()
        a = soft * S

        # enforce symmetry (optimal solutions are symmetric)
        a = 0.5 * (a + a[::-1])

        # ---------- convolution and τ‑soft‑max gradient ----------
        g, peak = _softmax_gradient(a, tau)

        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ---------- gradient w.r.t logits (entropy mirror) ----------
        mu = np.dot(a, g) / S                # scalar ensures the simplex constraint
        grad_p = a * (g - mu)                # (∂L/∂a)·(∂a/∂p) ; a = S·softmax(p)

        # optional clipping for stability
        clip = np.max(np.abs(grad_p))
        if clip > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---------- schedules ----------
        tau *= 0.997                         # accelerate cooling a little
        lr *= 0.9999

        # occasional jitter in log‑space to escape plateaus
        if it % 800 == 0:
            p += rng.normal(scale=0.015, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Multi‑restart optimiser that searches for the best step‑function.
    The routine respects the time budget, keeps the best vector seen,
    stores it in the global variable ``height_sequence_1`` and returns it.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0      # small safety margin

    # ------------------------------------------------------------------
    # Re‑use a previous construction if it exists
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    attempt = 0

    # ------------------------------------------------------------------
    # Helper to build an initial feasible vector of length n
    # ------------------------------------------------------------------
    def _initial_vector(n):
        S = np.sqrt(2.0 * n)
        choice = rng.choice(['edge', 'sqrt', 'uniform', 'random'])
        if choice == 'edge':
            p_exp = rng.uniform(0.3, 2.2)           # power‑law exponent
            a = _edge_power_seq(n, p_exp)
        elif choice == 'sqrt':
            a = _sqrt_series_seq(n)
        elif choice == 'uniform':
            a = np.full(n, 1.0)
        else:                                      # random Dirichlet‑like start
            a = rng.random(n) + 1e-3

        # ensure symmetry (important for later optimisation)
        a = 0.5 * (a + a[::-1])

        # scale to the required ℓ₁ norm (the optimiser works on the simplex)
        a = _scale_to_sum(a, S)

        # occasionally blend with the best known construction
        if base_seq is not None and rng.random() < 0.2:
            a = _scale_to_sum(_scale_to_sum(base_seq, S), S)   # simple interpolation
        return a

    # ------------------------------------------------------------------
    # Main loop: independent restarts until time runs out
    # ------------------------------------------------------------------
    while time.time() < end_time:
        attempt += 1

        # pick length (mix moderate and larger sizes)
        if rng.random() < 0.65:
            n = rng.integers(1200, 5000)   # moderate
        else:
            n = rng.integers(5000, 20000)  # larger

        a_start = _initial_vector(n)

        # allocate a slice of the remaining budget (≈ 14 s per restart)
        slice_deadline = min(time.time() + rng.uniform(10.0, 16.0), end_time)

        a_opt, peak = _adam_optimize(a_start, slice_deadline, rng)

        # compute the exact score (max of the hard convolution)
        score = 2.0 * n * peak / (np.sum(a_opt) ** 2)

        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[improve] attempt={attempt} n={n}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final refinement of the best candidate (a short extra optimisation)
    # ------------------------------------------------------------------
    if best_seq is not None:
        n = len(best_seq)
        # give the current best a few extra Adam steps with a very low τ
        extra_deadline = min(time.time() + 8.0, end_time)
        best_seq, _ = _adam_optimize(best_seq, extra_deadline, rng)

    # ------------------------------------------------------------------
    # Sanity check via the black‑box evaluator and store the result
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()
        external = evaluate_sequence(final_seq)
        print(f"[final] external score = {external:.6f}   internal approx = {best_score:.6f}   length = {len(final_seq)}")
        globals()['height_sequence_1'] = final_seq

    return final_seq
