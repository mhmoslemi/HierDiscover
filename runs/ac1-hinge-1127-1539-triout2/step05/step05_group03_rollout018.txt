"""
Search for a non‑negative step function (list of heights) that minimises
the peak of its self‑convolution as measured by ``evaluate_sequence``.
The method repeatedly draws diverse initial vectors, then refines them
with a two‑phase optimiser:

* A smooth Adam phase that minimises a log‑sum‑exp approximation of the
  max‑norm.  The temperature (soft‑max sharpness) is increased up to a
  large value (≈80) so the gradient eventually focuses on the worst
  convolution entries.
* A sub‑gradient phase that directly pushes down the current worst shift.

The optimiser works on the simplex Σ a_i = √(2·n) (scale‑invariant) and
projects onto it after each update.  Random “mass‑swap’’ moves and
periodic rejuvenation of the starting shape help escape plateaus.
The whole search runs for the allocated time budget (≈1000 s) and
returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of (1/α)·log‑sum‑exp(α·(a*a)) w.r.t. a.
    The vector ``w`` must have length 2·n‑1 and sum to 1.
    Returns a vector of length n (the gradient).
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    # pick the slice that corresponds to the gradient
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights (smooth approximation of max)
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """w_i = exp(alpha*conv_i) / Σ_j exp(alpha*conv_j)  (stable)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Initial sequence generators (many families for diversity)
# ----------------------------------------------------------------------
def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Resize an existing sequence to length n using linear interpolation."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n with ℓ¹‑norm √(2·n).
    Several random families are mixed to diversify the search.
    """
    S = np.sqrt(2.0 * n)

    typ = rng.integers(0, 7)      # 0 … 6

    if typ == 0 and base_seq is not None and base_seq.size > 10:
        a = _upsample_seq(base_seq, n)
        a += rng.normal(scale=a.mean() * 1e-3, size=n)
        a = np.clip(a, 0.0, None)

    elif typ == 1:
        # pure Dirichlet(1) (uniform random)
        a = rng.random(n)

    elif typ == 2:
        # symmetric geometric decay
        rate = rng.uniform(0.95, 0.9999)
        left = rate ** np.arange(n)
        right = left[::-1]
        a = (left + right) / 2.0

    elif typ == 3:
        # edge‑heavy reciprocal shape
        idx = np.arange(n, dtype=float)
        a = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.8)
        a = a ** power

    elif typ == 4:
        # random convex combination of the three basic shapes
        cand = []
        cand.append(rng.random(n))                # Dirichlet
        rate = rng.uniform(0.95, 0.9999)
        left = rate ** np.arange(n)
        right = left[::-1]
        cand.append((left + right) / 2.0)         # geometric
        idx = np.arange(n, dtype=float)
        rec = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.8)
        cand.append(rec ** power)                 # reciprocal
        w = rng.random(len(cand))
        w /= w.sum()
        a = sum(wi * ci for wi, ci in zip(w, cand))

    elif typ == 5:
        # two uniform blocks at the ends, zeros in the middle
        block_len = rng.integers(int(0.1 * n), int(0.3 * n) + 1)
        left = np.full(block_len, 1.0)
        right = np.full(block_len, 1.0)
        middle = np.zeros(n - 2 * block_len)
        a = np.concatenate([left, middle, right])

    else:   # typ == 6
        # centred Gaussian bell (may be very flat)
        sigma_frac = rng.uniform(0.08, 0.25)    # fraction of n
        sigma = sigma_frac * n
        x = np.arange(n) - (n - 1) / 2.0
        a = np.exp(-0.5 * (x / sigma) ** 2)

    # Normalise to required ℓ¹‑norm
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0)
        a_sum = a.sum()
    a = a / a_sum * S
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smooth log‑sum‑exp max‑approximation
# ----------------------------------------------------------------------
def _optimize_adam_v2(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected Adam minimising the smoothed max‑norm (more aggressive)."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.30
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # soft‑max temperature schedule
    alpha = 0.8
    alpha_mul = 1.015
    max_alpha = 80.0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = np.max(conv)
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # soft‑max weights (focus on large entries)
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update (projected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a, S)

        # occasional random “mass‑swap’’ moves
        if rng.random() < 0.05:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.12 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < best_score or rng.random() < 0.2:
                    a = cand

        # temperature increase
        alpha = min(alpha * alpha_mul, max_alpha)
        # gentle learning‑rate decay
        lr *= 0.9995

        # early exit if nothing improved for a while
        if it % 4000 == 0 and best_score == cur_max:
            break

    return best_a, best_score

# Sub‑gradient phase (directly addresses the worst shift)

def _optimize_subgrad_v2(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Simple sub‑gradient descent on the true max‑norm."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.15
    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient for shift k_max
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]
        g *= 2.0                     # accounts for the two symmetric terms

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # occasional random perturbation (helps escape plateaux)
        if rng.random() < 0.06:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.07 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.25:
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()

        if it % 5000 == 0 and best_score == cur_max:

    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimiser (Adam → sub‑gradient) with a longer smooth phase
# ----------------------------------------------------------------------
def _optimize_combined_v2(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run Adam for roughly 55 % of the allotted time,
    then a sub‑gradient refinement for the remainder.
    """
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    smooth_deadline = now + 0.55 * remaining
    a_s, score_s = _optimize_adam_v2(a0, smooth_deadline, rng)

    if time.time() < deadline:
        a_f, score_f = _optimize_subgrad_v2(a_s, deadline, rng)
        if score_f < score_s:
            return a_f, score_f
    return a_s, score_s

        # if we have been idle for a while, forget the old shape
        if time.time() - last_improve > 45.0:
            base_seq = None   # force fresh random restarts

        # keep the latest good shape as a warm‑start for the next round
        if best_overall_seq is not None:
            base_seq = best_overall_seq.copy()

    # ------------------------------------------------------------------
    # Fallback in the unlikely event no candidate was produced
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        n_fallback = 5000
        a_fallback = np.full(n_fallback, np.sqrt(2.0 * n_fallback) / n_fallback)
        best_overall_seq = a_fallback

    final_seq = best_overall_seq.tolist()

    # final sanity check via the public evaluator
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} "
          f"(internal best = {best_overall_score:.6f}) "
          f"length = {len(final_seq)}")
    return final_seq
