def _adam_mirror_descent_opt(a0, deadline):
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    beta = 0.5
    beta_max = 2e5
    lr = 0.5
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    t_iter = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        t_iter += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**t_iter)
        v_hat = v / (1 - beta2**t_iter)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # ensure nonnegativity and projection onto simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)
        # beta schedule
        if t_iter % 200 == 0:
            beta = min(beta * 1.07, beta_max)
    return a

"""
Hybrid multistart + Adam‑mirrored descent + fine‑grained SA.

* Randomly generate many candidates on the ℓ¹‑simplex for several
  lengths (2k‒10k). Keep the best according to max(self‑convolution).

* Refine the best candidate with a smoothed‑max (log‑sum‑exp) surrogate
  using an Adam‑style projected gradient descent.  The smoothing
  parameter β is slowly increased so the surrogate approaches the true
  ℓ∞ norm.

* Finish with a targeted simulated‑annealing phase that moves a tiny
  fraction of mass from indices that heavily contribute to the current
  maximal convolution entry to random destinations.  The temperature
  decays geometrically.

All steps respect the supplied time budget (default 1000 s) and the
returned sequence consists of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using an FFT (linear convolution)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex Σ x_i = s, x_i ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x ≥ 0, Σ x = s} (Duchi et al. 2008)."""
    if s <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, v.shape[0] + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) of a*a and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b) where
        b = a*a (full convolution)
        g = ∇_a  ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a ⋆ p),  p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]

    # full convolution a*a
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))    # keep numbers stable
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p) ; we need the central n entries
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1         # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on the smoothed max
# ----------------------------------------------------------------------


def _adam_mirror_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Adam optimiser on the log‑sum‑exp surrogate.
    Keeps the iterate on the ℓ¹‑simplex (via Euclidean projection) after
    each gradient step.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # smoothing parameter schedule
    beta = 1.0
    beta_max = 2e5
    beta_inc = 1.06

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # gradient step
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # projection onto the simplex (keeps non‑negativity)
        a = _project_to_simplex(a, target_sum)

        # increase smoothing (slowly) – approximates the true max
        if t % 250 == 0 and beta < beta_max:
            beta = min(beta * beta_inc, beta_max)

    return a


# ----------------------------------------------------------------------
# Targeted simulated‑annealing refinement
# ----------------------------------------------------------------------


def _targeted_simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    SA that repeatedly moves a tiny fraction of mass from an index that
    contributes heavily to the current maximal convolution entry to a
    random destination.  Temperature decays geometrically.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # temperature proportional to current objective, then cool
    T = best_score * 0.018 + 1e-9
    cooling = 0.9996
    rng = np.random.default_rng()
    max_transfer = 0.12      # up to 12 % of the selected mass per move

    while time.time() < deadline:
        b = _full_convolution_fft(a)
        k_max = int(np.argmax(b))

        # indices that actually contribute to b[k_max]
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idx_range = np.arange(lo, hi + 1)
        contrib = a[idx_range] * a[k_max - idx_range]
        weight_sum = np.sum(contrib)

        # pick source proportionally to its contribution
        if weight_sum > 0:
            probs = contrib / weight_sum
            i = rng.choice(idx_range, p=probs)
        else:
            i = rng.integers(0, n)

        # pick a distinct destination uniformly
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        # move a random fraction of the mass at i to j
        delta = a[i] * rng.random() * max_transfer
        if delta <= 0.0:
            continue
        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score

        accept = False
        if cur_score <= best_score:
            accept = True
        elif T > 0.0 and rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA improvement: {best_score:.6f}")
        else:
            # revert move
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # occasional global jitter to escape plateaus
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.0009, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, np.sqrt(2.0 * n))
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA jitter improvement: {best_score:.6f}")

    return best


# ----------------------------------------------------------------------
# Random initialisers – diverse families on the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Generate a random non‑negative sequence with Σ a_i = √(2 n)."""
    target_sum = np.sqrt(2.0 * n)
    kind = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if kind == 0:                           # power‑law (decreasing)
        p = rng.uniform(0.12, 1.2)
        a = (idx + 1.0) ** (-p)

    elif kind == 1:                         # left‑edge exponential
        lam = rng.exponential(scale=0.018)
        a = np.exp(-lam * idx)

    elif kind == 2:                         # mixed power‑law & exponential
        p = rng.uniform(0.12, 1.2)
        lam = rng.exponential(scale=0.018)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif kind == 3:                         # Dirichlet (uniform simplex)
        a = rng.dirichlet(alpha=np.ones(n))

    elif kind == 4:                         # symmetric edge‑heavy
        p = rng.uniform(0.15, 1.3)
        left = (idx + 1.0) ** (-p)
        right = (n - idx) ** (-p)
        a = left + right

    else:                                   # Beta‑shaped (U‑ or bell‑shaped)
        alpha = rng.uniform(0.2, 2.5)
        beta = rng.uniform(0.2, 2.5)
        x = (idx + 0.5) / n
        a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))

    # tiny Gaussian noise then project onto the simplex
    a += rng.normal(scale=target_sum * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target_sum)
    return a


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` under the given time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_obj = np.inf

    # ------------------------------------------------------------------
    # 1) If a previously discovered sequence exists, use it as a warm start
    # ------------------------------------------------------------------
    global_seq = globals().get("height_sequence_1")
    if isinstance(global_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(global_seq, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        cur_obj = 2 * n0 * _max_convolution_fft(a0) / (np.sqrt(2.0 * n0) ** 2)
        best_seq, best_obj = a0.copy(), cur_obj
        print(f"[{time.time():.1f}] Warm‑start from height_sequence_1 "
              f"(n={n0})  obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 2) Scan several lengths with cheap random samples
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000, 10000]
    per_len_time = 9.0           # seconds per length for the cheap scan

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)
        end_this_len = min(deadline, time.time() + per_len_time)

        while time.time() < end_this_len:
            a = _random_initial_sequence(n, rng)
            maxb = _max_convolution_fft(a)
            obj = 2 * n * maxb / (target_sum ** 2)
            if obj < best_obj:
                best_obj = obj
                best_seq = a.copy()
                print(f"[{time.time():.1f}] New global best (scan) n={n}  obj={obj:.6f}")

    # ------------------------------------------------------------------
    # 3) If nothing was generated, fall back to a uniform sequence
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 3000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_obj = 2 * n * _max_convolution_fft(best_seq) / (np.sqrt(2.0 * n) ** 2)
        print(f"[{time.time():.1f}] Fallback uniform seq n={n}  obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 4) Refine the current best with Adam‑mirrored descent (if time permits)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        md_deadline = time.time() + remaining * 0.45
        a_md = _adam_mirror_descent_opt(best_seq, md_deadline)
        n_md = a_md.size
        obj_md = 2 * n_md * _max_convolution_fft(a_md) / (np.sqrt(2.0 * n_md) ** 2)
        if obj_md < best_obj:
            best_obj = obj_md
            best_seq = a_md.copy()
            print(f"[{time.time():.1f}] After Adam‑MD improvement: obj={obj_md:.6f}")

    # ------------------------------------------------------------------
    # 5) Final targeted simulated‑annealing (use whatever time left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_sa = _targeted_simulated_annealing(best_seq, deadline)
        n_sa = a_sa.size
        obj_sa = 2 * n_sa * _max_convolution_fft(a_sa) / (np.sqrt(2.0 * n_sa) ** 2)
        if obj_sa < best_obj:
            best_obj = obj_sa
            best_seq = a_sa.copy()
            print(f"[{time.time():.1f}] After SA improvement: obj={obj_sa:.6f}")

    # ------------------------------------------------------------------
    # 6) Trim trailing near‑zero entries and return as a plain list
    # ------------------------------------------------------------------
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Optional sanity check with the official evaluator (if imported)
    try:
        ev = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence = {ev:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective = {best_obj:.6f} "
          f"   length = {best_seq.size}")

    return best_seq.tolist()
