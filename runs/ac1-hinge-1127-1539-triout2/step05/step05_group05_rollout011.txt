def _max_subgradient_opt(a0, deadline):
    a = a0.copy()
    target_sum = np.sqrt(2.0 * a0.size)
    step=0
    eta0 = 0.5
    eps=1e-12
    while time.time() < deadline:
        # compute b and max
        b = _full_convolution_fft(a)
        max_val = np.max(b)
        # compute active indices
        active = np.where(b >= max_val - 1e-12)[0]
        # compute gradient
        g = np.zeros_like(a)
        n = a.size
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx_range = np.arange(lo, hi + 1)
            # compute a_{k - i}
            g[idx_range] += 2.0 * a[k - idx_range]
        # update via exponentiated gradient (mirror descent style)
        eta = eta0 / np.sqrt(step+1.0)
        a = a * np.exp(-eta * g)
        a = np.maximum(a, eps)
        s = np.sum(a)
        a = a * (target_sum / s) if s>0 else np.full_like(a, target_sum / n)
        step += 1
    return a

"""
Optimal step–function via a three‑phase scheme:

1. **Coarse multi‑length scan** – a handful of random families (Dirichlet,
   power‑law, exponential, edge‑heavy, β‑shaped) are projected onto the
   ℓ¹‑simplex Σa = √(2 n).  For each candidate length a quick “max‑convolution”
   score is computed; the best length and seed are kept.

2. **Refinement** – two complementary mirror‑descent optimisers are run on the
   chosen candidate:
      * a *smooth* optimiser that minimises the log‑sum‑exp surrogate of the
        convolution (very similar to the previous code but with a faster
        β‑annealing schedule);
      * a *sub‑gradient* optimiser that works directly on the true ℓ∞‑norm
        (the gradient is a convex combination of the gradients of all
        active (maximal) convolution entries).  Both updates stay on the
        simplex automatically via an exponentiated‑gradient step.

3. **Deterministic greedy refinement** – the current best vector is repeatedly
   “mass‑shaved”: the index that contributes most to the current maximal
   convolution entry loses a fraction of its weight, which is transferred to
   the index whose addition increases the maximal entry the least.  The move
   is accepted only if it strictly improves the objective; occasional small
   jitter (Gaussian noise) helps escape shallow minima.

The whole routine respects the supplied time budget (default 1000 s) and
returns the best non‑negative sequence found (as a plain Python list).  All
helper functions are top‑level; no I/O is performed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean, Duchi et al. 2008)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x_i≥0, Σx_i = s}."""
    if s <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) of the convolution and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Return (g, b) where
        b = a*a (full convolution, length 2·n‑1)
        g = ∇_a  ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a ⋆ p)   with   p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]

    # ----- b = a*a ----------------------------------------------------
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ----- soft‑max probabilities ---------------------------------------
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))          # stable exponentiation
    p = exps / np.sum(exps)

    # ----- gradient g = 2·(a ⋆ p) --------------------------------------
    a_rev = a[::-1]                            # reversed a
    conv_len = n + (2 * n - 1) - 1               # = 3·n‑2
    size2 = 1 << ((conv_len).bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]        # central n entries
    return g, b


# ----------------------------------------------------------------------
# Mirror‑descent optimiser (smooth surrogate)
# ----------------------------------------------------------------------


def _smooth_mirror_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Exponentiated‑gradient descent on the log‑sum‑exp surrogate.
    Keeps the iterate on the ℓ¹‑simplex automatically.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 1.0
    beta_max = 1e5
    eta0 = 0.6
    step = 0
    eps = 1e-12

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        eta = eta0 / np.sqrt(step + 1.0)
        a = a * np.exp(-eta * g)          # element‑wise exponentiated update
        a = np.maximum(a, eps)            # avoid exact zeros

        # renormalise to the simplex
        s = np.sum(a)
        if s > 0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # anneal β more aggressively
        if step % 100 == 0 and beta < beta_max:
            beta = min(beta * 1.12, beta_max)

        step += 1

    return a


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑norm (exponentiated‑gradient)
# ----------------------------------------------------------------------


def _max_subgradient_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Mirror‑descent (exponentiated‑gradient) using a genuine sub‑gradient of
    the ℓ∞‑norm of the convolution (i.e. the max entry).  The update direction
    is based on all currently active (maximal) convolution slots.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    eta0 = 0.8
    step = 0
    eps = 1e-12

    while time.time() < deadline:
        b = _full_convolution_fft(a)
        max_val = np.max(b)
        # indices where the maximum is attained (within a tiny tolerance)
        active = np.where(abs(b - max_val) < 1e-12)[0]

        # sub‑gradient: g_i = 2 * Σ_{k∈active} a_{k-i}  (zero when out of range)
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] += 2.0 * a[k - idx]

        # exponentiated‑gradient step
        eta = eta0 / np.sqrt(step + 1.0)
        a = a * np.exp(-eta * g)
        a = np.maximum(a, eps)

        # renormalise onto the simplex
        s = np.sum(a)
        if s > 0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        step += 1

    return a


# ----------------------------------------------------------------------
# Deterministic greedy refinement (mass‑shifting)
# ----------------------------------------------------------------------


def _greedy_refine(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Repeatedly move a fraction of the mass from the index that contributes
    most to the current maximal convolution entry to a destination that
    causes the smallest increase of that entry.  Moves are accepted only
    when they strictly improve the current best score; occasional tiny
    Gaussian jitter helps escape local minima.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # temperature‑free greedy search
    rng = np.random.default_rng()
    max_transfer = 0.30       # up to 30 % of the selected mass

    while time.time() < deadline:
        b = _full_convolution_fft(a)
        k_max = int(np.argmax(b))

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idx_range = np.arange(lo, hi + 1, dtype=int)
        contrib = a[idx_range] * a[k_max - idx_range]   # a_i * a_{k-i}
        if contrib.sum() == 0:
            # degenerate – pick a random i
            i = rng.integers(0, n)
        else:
            i = idx_range[np.argmax(contrib)]

        # pick a destination j that is far from the support of k_max
        # (i.e. minimise a[j] * a[k_max-j]); we simply try a few random
        # candidates and keep the best.
        best_j = None
        best_increase = np.inf
        for _ in range(5):
            cand = rng.integers(0, n)
            if cand == i:
                continue
            # estimated increase of the maximal entry if we add δ mass to cand:
            inc = a[cand] * a[k_max - cand] if (0 <= k_max - cand < n) else 0.0
            if inc < best_increase:
                best_increase = inc
                best_j = cand
        if best_j is None:
            continue
        j = best_j

        delta = a[i] * rng.random() * max_transfer
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time():.1f}] Greedy improvement: {best_score:.6f}")
        else:
            # revert – greedy step only keeps improvements
            a[j] -= delta
            a[i] += delta

        # occasional global jitter (≈0.1 % of iterations)
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            s = np.sum(a)
            if s > 0:
                a = a * (np.sqrt(2.0 * n) / s)
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] Jitter improvement: {best_score:.6f}")

    return best


# ----------------------------------------------------------------------
# Random initialisers (diverse families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Generate a random non‑negative sequence on the ℓ¹‑simplex
    Σ a_i = √(2 n).  Several families are mixed to increase diversity.
    """
    target_sum = np.sqrt(2.0 * n)
    kind = rng.integers(0, 7)
    idx = np.arange(n, dtype=float)

    if kind == 0:                     # decreasing power‑law
        p = rng.uniform(0.15, 1.0)
        a = (idx + 1.0) ** (-p)

    elif kind == 1:                 # exponential decay from the left
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif kind == 2:                 # mixed power‑law + exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif kind == 3:                 # Dirichlet (uniform simplex)
        a = rng.dirichlet(alpha=np.ones(n))

    elif kind == 4:                 # symmetric edge‑heavy
        p = rng.uniform(0.2, 1.2)
        left = (idx + 1.0) ** (-p)
        right = (n - idx) ** (-p)
        a = left + right

    elif kind == 5:                 # Beta‑shaped (U‑ or bell‑shaped)
        alpha = rng.uniform(0.2, 2.0)
        beta = rng.uniform(0.2, 2.0)
        x = (idx + 0.5) / n
        a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))

    else:                           # simple cosine bump (smooth)
        a = 0.5 * (1.0 - np.cos(2.0 * np.pi * idx / (n - 1)))

    # small Gaussian perturbation and projection onto the simplex
    a += rng.normal(scale=target_sum * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target_sum)
    return a
