    S = np.sqrt(2 * n)
    NP = 20 if n < 2000 else 12  # smaller for larger n
    # Initialize population
    pop = []
    scores = []
    for i in range(NP):
        if i == 0:
            a = initial_a.copy()
        else:
            a = rng.random(n)
        a = a / a.sum() * S
        pop.append(a)
        scores.append(np.max(_full_convolution(a)))
    # Best tracking
    best_idx = int(np.argmin(scores))
    best_a = pop[best_idx].copy()
    best_score = scores[best_idx]
    while time.time() < deadline:
        # For each individual
        for i in range(NP):
            # Choose parents
            idxs = list(range(NP))
            idxs.remove(i)
            p1, p2 = rng.choice(idxs, size=2, replace=False)
            # Crossover: convex combination
            alpha = rng.random()
            child = alpha * pop[p1] + (1 - alpha) * pop[p2]
            # Mutation: add small noise
            if rng.random() < 0.3:
                noise = rng.normal(scale=child.mean() * 0.05, size=n)
                child = child + noise
            # Clip negative
            child = np.maximum(child, 0.0)
            # Project onto simplex
            child = _project_simplex(child, S)
            # Evaluate
            child_score = np.max(_full_convolution(child))
            # Selection: if child better replace
            if child_score < scores[i]:
                pop[i] = child
                scores[i] = child_score
                # Update global best
                if child_score < best_score:
                    best_score = child_score
                    best_a = child.copy()
        # Occasionally introduce new random individuals to maintain diversity
        if rng.random() < 0.1:
            # replace worst

"""
Propose a step‑function (non‑negative vector) that minimises
the maximum entry of its self‑convolution, i.e. the value returned
by ``evaluate_sequence``.

Algorithm
---------
* Work on the L¹‑simplex Σ a_i = √(2·n).  The objective is exactly
  max(conv(a,a)), so we minimise the infinity‑norm of the convolution.
* For each problem size we run a two‑phase optimiser:
    1. Adam on a smooth soft‑max approximation of the max‑norm.
       The gradient is obtained by a weighted convolution (FFT) and the
       vector is projected onto the simplex after each update.
    2. Projected sub‑gradient descent that attacks the true worst entry.
* After the gradient phase we run a lightweight genetic algorithm
  (convex recombination + mutation + selection) to escape possible
  local minima.
* The optimisation proceeds through three increasing sizes
  (800 → 2000 → 4000).  Results are up‑sampled between stages and the
  best sequence found is returned.
* The code respects the 1000 s budget (leaving a 5 s safety margin)
  and prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient term for the smooth max‑norm:
        g_j = 2 Σ_i w_{i+j} a_i .
    Both ``a`` (len n) and ``w`` (len 2·n‑1) are real vectors.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights (smooth approximation of max)
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """w_i = exp(alpha·conv_i) / Σ_j exp(alpha·conv_j)  (stable)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Upsampling helper (linear interpolation)
# ----------------------------------------------------------------------


def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a to length new_n."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=float)
    return np.interp(x_new, x_old, a)


# ----------------------------------------------------------------------
# Adam optimiser on smooth max‑norm
# ----------------------------------------------------------------------


def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Adam optimiser on a soft‑max approximation of max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters (tuned for this problem)
    lr = 0.4
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    alpha = 0.5          # temperature – starts smooth
    alpha_max = 1500.0   # final sharpness

    while time.time() < deadline:
        t += 1
        # gradient of smooth max‑norm
        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex
        a = _project_simplex(a, S)

        # occasional random mass swap (exploration)
        if rng.random() < 0.03:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.02
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # temperature schedule (make the soft‑max sharper)
        alpha = min(alpha * 1.05, alpha_max)

        # cheap early‑stop test
        if t % 5000 == 0 and cur_score >= best_score * (1 + 1e-9):
            # continue, but could break if stagnating – we keep running
            pass

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient optimiser (acts on the true max‑norm)
# ----------------------------------------------------------------------


def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that directly reduces the worst entry."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.5  # initial step size constant
    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # construct sub‑gradient: g_i = a_{k_max-i} (zero outside overlap)
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional random perturbation
        if rng.random() < 0.07:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.03
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = np.max(_full_convolution(cand))
                if cand_score < cur_score or rng.random() < 0.2:
                    a = cand
                    cur_score = cand_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()

        # cheap early‑stop
        if it % 8000 == 0 and cur_score >= best_score * (1 + 1e-9):
            pass

    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimiser (Adam → sub‑gradient) with configurable split
# ----------------------------------------------------------------------


def _optimize_combined_adam(a0: np.ndarray,
                            deadline: float,
                            rng: np.random.Generator,
                            fraction_adam: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Run Adam for a fraction of the allocated time, then finish with sub‑gradient.
    """
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    adam_deadline = now + fraction_adam * remaining
    a_s, sc_s = _optimize_adam(a0, adam_deadline, rng)

    if time.time() < deadline:
        a_f, sc_f = _optimize_subgrad(a_s, deadline, rng)
        if sc_f < sc_s:
            return a_f, sc_f
    return a_s, sc_s


# ----------------------------------------------------------------------
# Genetic algorithm optimiser (convex recombination + mutation)
# ----------------------------------------------------------------------


def _run_genetic_optimizer(initial_a: np.ndarray,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple steady‑state GA that works on the simplex.
    Returns the best vector found before ``deadline``.
    """
    n = initial_a.shape[0]
    S = np.sqrt(2.0 * n)

    # population size (smaller for larger problems to keep time low)
    pop_size = 20 if n <= 2000 else 12

    # initialise population (first individual is the provided one)
    pop = []
    scores = []
    for i in range(pop_size):
        if i == 0:
            a = initial_a.copy()
        else:
            a = rng.random(n)
        a = _project_simplex(a, S)
        pop.append(a)
        scores.append(np.max(_full_convolution(a)))
    pop = np.array(pop)
    scores = np.array(scores)

    best_idx = int(np.argmin(scores))
    best_a = pop[best_idx].copy()
    best_score = scores[best_idx]

    # main loop
    while time.time() < deadline:
        for i in range(pop_size):
            # select two distinct parents (different from target)
            idxs = list(range(pop_size))
            idxs.remove(i)
            p1, p2 = rng.choice(idxs, size=2, replace=False)

            # convex recombination
            alpha = rng.random()
            child = alpha * pop[p1] + (1.0 - alpha) * pop[p2]

            # mutation (Gaussian noise)
            if rng.random() < 0.35:
                noise = rng.normal(scale=child.mean() * 0.05 + 1e-12, size=n)
                child += noise

            # enforce non‑negativity and simplex
            child = np.maximum(child, 0.0)
            child = _project_simplex(child, S)

            child_score = np.max(_full_convolution(child))

            # replace if better than current target
            if child_score < scores[i]:
                pop[i] = child
                scores[i] = child_score
                if child_score < best_score:
                    best_score = child_score
                    best_a = child.copy()

        # occasional diversity injection
        if rng.random() < 0.12:
            worst_idx = int(np.argmax(scores))
            a_new = rng.random(n)
            a_new = _project_simplex(a_new, S)
            new_score = np.max(_full_convolution(a_new))
            pop[worst_idx] = a_new
            scores[worst_idx] = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

        # terminate early if no progress for a while
        if time.time() + 0.1 >= deadline:
            break

    return best_a, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function (non‑negative vector) that minimises
    ``evaluate_sequence``.  The routine mixes Adam, sub‑gradient,
    and a lightweight genetic algorithm, and proceeds through
    three increasing problem sizes.
    Returns the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin for the external evaluation call
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered construction if available
    # ------------------------------------------------------------------
    best_seq = None
    if 'height_sequence_1' in globals():
        try:
            best_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            best_seq = None

    best_score = np.inf
    last_improve = start_time

    # ------------------------------------------------------------------
    # Progressive optimisation over three problem sizes
    # ------------------------------------------------------------------
    sizes = [800, 2000, 4000]          # increasing number of coefficients
    # Allocate time fractions (sum < 1 to keep a safety buffer)
    time_fractions = [0.17, 0.28, 0.45]   # 90 % of the budget is used here

    total_budget = deadline - start_time
    for stage, n in enumerate(sizes):
        if time.time() >= deadline:
            break

        # allocate time for this stage
        stage_time = total_budget * time_fractions[stage]
        stage_deadline = min(time.time() + stage_time, deadline)

        # initialise the starting vector for this stage
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        else:
            a0 = rng.random(n)

        # tiny jitter to avoid exact repetitions
        jitter = rng.normal(scale=a0.mean() * 1e-4 + 1e-12, size=n)
        a0 = np.clip(a0 + jitter, 0.0, None)

        # simplex projection (Σa = √(2·n))
        S = np.sqrt(2.0 * n)
        if a0.sum() == 0.0:
            a0 = np.full(n, 1.0)
        a0 = a0 / a0.sum() * S

        # 1️⃣  Gradient phase (Adam + sub‑gradient)
        a_opt, sc_opt = _optimize_combined_adam(a0, stage_deadline, rng,
                                                fraction_adam=0.78)

        # 2️⃣  Short genetic‑algorithm refinement (if time remains)
        remaining = stage_deadline - time.time()
        if remaining > 1.5:
            ga_deadline = time.time() + min(10.0, remaining)
            a_ga, sc_ga = _run_genetic_optimizer(a_opt, ga_deadline, rng)
            if sc_ga < sc_opt:
                a_opt, sc_opt = a_ga, sc_ga

        # update global best
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            last_improve = time.time()
            print(f"[stage] n={n} new global best score={best_score:.6f}")

        # shrink remaining total budget for later stages
        total_budget = deadline - time.time()

    # ------------------------------------------------------------------
    # Final sanity check & external evaluation
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()

    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} "
          f"(internal best = {best_score:.6f}) length = {len(final_seq)}")
    return final_seq
