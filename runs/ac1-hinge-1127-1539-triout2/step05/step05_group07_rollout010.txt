def simplex_projection(v, z):
    # v is a numpy array
    # projects onto {x >=0, sum x = z}
    # algorithm: sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
adversarial step‑function optimizer – improved version

The optimizer searches for a non‑negative vector `a` of length `n` whose
ℓ₁‑norm `S = sqrt(2·n)` minimises the peak of its self‑convolution.
It combines three phases:

1. **Multi‑restart Adam** on a τ‑soft‑max approximation of the ∞‑norm.
   Longer runs, slower cooling and a larger initial learning rate give
   a deeper local optimisation than the previous baseline.

2. **Hard‑max sub‑gradient refinement** – after the Adam phase we switch
   to the exact max (τ≈0) and perform projected sub‑gradient descent on
   the simplex to further reduce the peak.

3. **Linear‑programming “active‑set” refinement** – the current vector
   `f` is used as a constraint matrix for the LP
       maximise  Σ g_i   s.t.   f * g ≤ M   and g ≥ 0,
   where `M` is the current convolution peak.
   The obtained `g` is blended with `f` (mixing parameter `t`) and the
   process is repeated a few times.  This step implements the idea
   described in the problem statement and often yields a noticeable
   “jump” in the objective.

All phases respect a global time‑budget (default 1000 s).  The best
sequence found is stored in the global variable ``height_sequence_1``
and returned.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.ceil(pos).astype(int)
    hi = np.clip(hi, 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Returns (g, max_b) where g = ∂/∂a  τ‑soft‑max(b) and max_b = max(a*a).
    The gradient is expressed in the original 'a' space.
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        # exact sub‑gradient (all active entries)
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        w = exps / np.sum(exps)          # soft‑max weights, shape (2n‑1,)
        mask = w

    # gradient w.r.t a: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + n - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]
    return g, max_b


def _simplex_projection(v: np.ndarray, z: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex {x ≥ 0, Σ x = z}.
    Implementation of Duchi et al. (2008).
    """
    if z < 0:
        raise ValueError("z must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Phase 1 – Adam optimisation on the soft‑max of the convolution peak
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   lr_initial: float = 0.5,
                   tau_factor: float = 0.9998,
                   lr_factor: float = 0.99995,
                   jitter_interval: int = 400) -> tuple[np.ndarray, float]:
    """
    Minimises a smoothed version of max(a*a) using Adam on the log‑softmax
    parametrisation.  Returns the best vector found and its hard peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    # initialise logits so that softmax(p)*S == a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                     # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr_initial

    # initial smoothing parameter
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 1.0)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ------- a from logits (log‑softmax) -------
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ------- gradient of the τ‑soft‑max -------
        g, peak = _softmax_gradient(a, tau)

        # keep best (hard) peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ------- gradient w.r.t. logits -------
        mu = np.dot(a, g) / S                     # scalar
        grad_p = a * (g - mu)                     # shape (n,)

        # optional gradient clipping
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ------- Adam update -------
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ------- schedules -------
        tau *= tau_factor               # tighten the soft‑max
        lr *= lr_factor                 # decay the learning rate

        # occasional jitter in log‑space to escape plateaus
        if it % jitter_interval == 0:
            p += rng.normal(scale=0.02, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Phase 2 – Hard‑max sub‑gradient descent (projection onto simplex)
# ----------------------------------------------------------------------
def _subgradient_optimize(a0: np.ndarray,
                          deadline: float,
                          rng: np.random.Generator,
                          lr_initial: float = 0.03,
                          decay: float = 0.9999,
                          jitter_interval: int = 200) -> tuple[np.ndarray, float]:
    """
    Performs projected sub‑gradient descent on the exact max(a*a).
    Returns the best vector encountered and its peak.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    lr = lr_initial
    best_a = a.copy()
    best_peak = _full_convolution_fft(a).max()
    it = 0

    while time.time() < deadline:
        it += 1

        # full convolution and hard max mask
        b = _full_convolution_fft(a)
        max_b = b.max()
        mask = (b == max_b).astype(float)

        # gradient of hard max (average over active positions)
        a_rev = a[::-1]
        conv_len = len(mask) + n - 1
        f_len = _next_pow2(conv_len)
        Fmask = np.fft.rfft(mask, n=f_len)
        Fa_rev = np.fft.rfft(a_rev, n=f_len)
        conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
        g = 2.0 * conv[n - 1:n - 1 + n]   # sub‑gradient of max

        # descent step on the simplex
        a = a - lr * g
        a = _simplex_projection(a, S)

        # keep best
        cur_peak = _full_convolution_fft(a).max()
        if cur_peak < best_peak:
            best_peak = cur_peak
            best_a = a.copy()

        lr *= decay

        # occasional jitter to escape flat regions
        if it % jitter_interval == 0:
            jitter = rng.normal(scale=0.01 * S, size=n)
            a = a + jitter
            a = _simplex_projection(a, S)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Phase 3 – Linear‑programming refinement (active‑set style)
# ----------------------------------------------------------------------
def _lp_refine_seq(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   t_initial: float = 0.2,
                   max_iters: int = 6) -> tuple[np.ndarray, float]:
    """
    Repeatedly solves a linear program:
        maximise Σ g_i   s.t.   a * g ≤ M , g ≥ 0
    where M = max(a*a).  The solution g is scaled to the required ℓ₁‑norm
    and mixed with the current vector a.
    Returns the best sequence seen and its peak.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_peak = _full_convolution_fft(a).max()
    t = t_initial
    it = 0

    # pre‑allocate space for the convolution matrix (dense)
    total_rows = 2 * n - 1
    while time.time() < deadline and it < max_iters:
        it += 1
        M = _full_convolution_fft(a).max()

        # Build the convolution constraint matrix A (dense)
        # A_{k,i} = a_{k-i}  if 0 ≤ k-i < n else 0
        A = np.zeros((total_rows, n), dtype=np.float64)
        for i in range(n):
            A[i:i + n, i] = a

        # Linear program: minimise -sum(g)  (i.e. maximise sum(g))
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n
        res = linprog(c, A_ub=A, b_ub=M * np.ones(total_rows),
                      bounds=bounds, method='highs', options={'disp': False})
        if res.status != 0:
            # LP failed – abort refinement
            break
        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 < 1e-12:
            break

        # Scale to the required ℓ₁‑norm and blend
        g = g0 * (S / sum_g0)
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)            # non‑negativity
        a_new = a_new / a_new.sum() * S           # project onto simplex

        new_peak = _full_convolution_fft(a_new).max()
        if new_peak < best_peak:
            best_peak = new_peak
            best_a = a_new.copy()
            a = a_new
            # keep mixing step relatively large after a successful improvement
            t = min(0.4, t * 1.1)
            # optional small jitter to keep diversity
            if rng.random() < 0.2:
                a += rng.normal(scale=0.005 * S, size=n)
                a = _simplex_projection(a, S)
            print(f"[LP] iter={it} improved peak={best_peak:.6f}")
        else:
            # no improvement – shrink the mixing factor
            t *= 0.6
            # optionally break if t becomes very small
            if t < 1e-4:
                break

    return best_a, best_peak

# Initial vector generator
# ----------------------------------------------------------------------
def _initial_vector(n: int,
                    rng: np.random.Generator,
                    base_seq: np.ndarray | None = None) -> np.ndarray:
    """
    Produce a feasible starting vector of length n with sum sqrt(2·n).
    Several shapes are tried (uniform, edge‑heavy, random Dirichlet).
    Occasionally the previous best solution is interpolated.
    """
    S = np.sqrt(2.0 * n)
    shape = rng.choice(['uniform', 'edge', 'random'])
    if shape == 'uniform':
        a = np.full(n, S / n)
    elif shape == 'edge':
        p_exp = rng.uniform(0.7, 2.0)
        idx = np.arange(n)
        d = np.minimum(idx, n - 1 - idx) + 1          # distance to nearest edge (1‑based)
        a = (d.astype(float) ** (-p_exp))
        a = a / a.sum() * S
    else:   # random Dirichlet‑like
        a = rng.random(n)
        a = a / a.sum() * S

    # blend with the global best occasionally
    if base_seq is not None and rng.random() < 0.2:
        a = _scale_sequence(base_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
    return a

            best_score = score
            best_seq = a_sg.copy()
            print(f"[restart] attempt={attempt} n={n}  new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3: linear‑programming refinement (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and (end_time - time.time()) > 5.0:
        # limit the refinement to a manageable size (dense matrix)
        if len(best_seq) > 2500:
            # down‑sample to keep the LP matrix reasonably small
            n_ref = 2500
            best_seq = _scale_sequence(best_seq, n_ref)
            S_ref = np.sqrt(2.0 * n_ref)
            best_seq = best_seq / best_seq.sum() * S_ref

        try:
            best_seq, best_score = _lp_refine_seq(best_seq, end_time, rng)
            print(f"[final LP] improved score={best_score:.6f}")
        except Exception as exc:
            print("[final LP] exception:", exc)

    # ------------------------------------------------------------------
    # Final sanity check and return
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()
        # External evaluation (for logging only – the harness will also call it)
        try:
            external = evaluate_sequence(final_seq)
            print(f"[final] external score = {external:.6f}  internal best = {best_score:.6f}  length = {len(final_seq)}")
        except Exception:
            print("[final] error during external evaluation.")
        globals()['height_sequence_1'] = final_seq

    return final_seq
