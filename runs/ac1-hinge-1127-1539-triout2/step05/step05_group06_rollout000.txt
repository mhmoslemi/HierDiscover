def propose_candidate(seed=12345, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 5.0

    best_seq = None
    best_score = np.inf

    # If existing height_sequence_1 present
    if 'height_sequence_1' in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=float)
            if np.any(base < 0):
                base = np.clip(base,0,None)
            best_seq = base.copy()
            best_score = np.max(_full_convolution(best_seq))
        except Exception:
            best_seq = None

    # Define parameters
    min_n = 2000
    max_n = 8000

    # Main loop
    attempts = 0
    while time.time() < deadline:
        attempts += 1

        # Determine n
        if best_seq is None or (time.time() - start) < 200 and attempts % 5 == 0:
            # Random start
            n = int(rng.integers(min_n, max_n+1))
            a0 = rng.random(n)
        else:
            # Use current best possibly upsampled
            if rng.random() < 0.3 and len(best_seq) < max_n:
                n = min(int(len(best_seq) * 1.5), max_n)
                a0 = _upsample(best_seq, n)
            else:
                n = len(best_seq)
                a0 = best_seq.copy()
                # jitter
                jitter_scale = a0.mean() * 1e-4
                a0 = a0 + rng.normal(scale=jitter_scale, size=n)
                a0 = np.clip(a0, 0.0, None)

        # Normalize onto simplex sum = sqrt(2*n)
        S = np.sqrt(2.0 * n)
        if a0.sum() == 0:
            a0 = np.full(n, S/n)
        else:
            a0 = a0 / a0.sum() * S

        # allocate slice time
        time_left = deadline - time.time()
        slice_time = min(30.0, time_left)  # allocate up to 30 sec per attempt
        slice_deadline = time.time() + slice_time

        # Run optimizer
        a_opt, score_opt = _optimize_combined(a0, slice_deadline, rng)

        # keep if better
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[best] attempt {attempts} n={n} score={best_score:.6f}")

    # After loop, maybe extra refinement on best_seq if time remains
    if time.time() < deadline and best_seq is not None and len(best_seq) < max_n:
        # try final upsample to max_n and optimize longer
        n = max_n
        a0 = _upsample(best_seq, n)
        a0 = a0 / a0.sum() * np.sqrt(2.0*n)
        final_deadline = deadline
        a_opt, score_opt = _optimize_combined(a0, final_deadline, rng)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()

Define _optimize_combined(a0, deadline, rng):

   time_now = time.time(); total_rem = deadline - time_now.

   stage1_end = time_now + 0.6 * total_rem

   a = a0.copy()

   best_a = a.copy(); best_score = np.max(_full_convolution(a))

   t = 0

   # Adam parameters
   lr = 0.4
   beta1=0.9; beta2=0.999; eps=1e-8

   m = np.zeros_like(a); v = np.zeros_like(a)

   # temperature schedule
   alpha = 1.0

   while time.time() < deadline:
       t +=1

       if time.time() < stage1_end:
           # compute conv and weighted gradient
           conv = _full_convolution(a)
           w = _smooth_weights(conv, alpha)
           grad = _conv_with_weights(a, w)  # same as earlier
           # Adam
           m = beta1*m + (1-beta1)*grad
           v = beta2*v + (1-beta2)*(grad**2)
           m_hat = m/(1-beta1**t)
           v_hat = v/(1-beta2**t)
           a = a - lr * m_hat/(np.sqrt(v_hat)+eps)
           # projection
           a = _project_simplex(a, np.sqrt(2.0*a.shape[0]))
           # temperature update
           alpha = min(alpha*1.02, 300.0)
       else:
           # subgradient
           conv = _full_convolution(a)
           kmax = int(conv.argmax())
           # compute subgradient using overlapping indices
           n = a.shape[0]
           g = np.zeros_like(a)
           i_low = max(0, kmax-(n-1))
           i_high = min(n-1, kmax)
           idx_i = np.arange(i_low, i_high+1)
           idx_j = kmax - idx_i
           g[idx_i] = a[idx_j]  # Since gradient is a[j] for each overlapping
           eta = 0.6 / np.sqrt(t)  # step size
           a = a - eta * g
           a = _project_simplex(a, np.sqrt(2.0*a.shape[0]))

       # occasional random mass swap
       if rng.random() < 0.05:
           i, j = rng.integers(0, a.shape[0], size=2)
           if i != j and a[i] > 0:
               delta = a[i] * rng.random() * 0.02
               a[i] -= delta
               a[j] += delta

"""
Propose a non‑negative step sequence that minimises the peak of its
self‑convolution (the value returned by ``evaluate_sequence``).

Algorithm
---------
* Multiple random restarts with lengths n ∈ [2000, 8000].
* Each restart runs a two‑stage optimiser:
    1. Adam on a soft‑max (temperature‑controlled) approximation of the
       max‑norm.  The gradient of the smooth objective is obtained by a
       weighted convolution (FFT) and the iterate is projected onto the
       simplex Σ a_i = √(2 n).
    2. Projected sub‑gradient descent on the true max‑norm.
* Occasional random “mass‑swap” moves and Gaussian jitter help escape
  plateaus.
* Whenever a better vector is found it is stored globally.  The current
  best vector is periodically up‑sampled (linear interpolation) to a
  larger size and refined further, allowing the optimiser to exploit
  more degrees of freedom.
* The whole search respects the 1000 s budget; the final vector is
  returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of Σ_i w_i (a*a)_i w.r.t. a (FFT implementation)."""
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max (smooth max) utilities
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights w_i ∝ exp(alpha·conv_i) (numerically stable)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Upsampling (linear interpolation)
# ----------------------------------------------------------------------


def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a to length new_n."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=float)
    return np.interp(x_new, x_old, a)


# ----------------------------------------------------------------------
# Two‑stage optimiser (Adam → sub‑gradient)
# ----------------------------------------------------------------------


def _optimize_combined(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run Adam on a smooth approximation of max(conv) for ~60 % of the
    allocated time, then finish with projected sub‑gradient descent.
    Returns the best vector encountered and its max‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # best bookkeeping
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    # Adam parameters
    lr = 0.4
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature for soft‑max
    alpha = 0.5

    # split the budget
    now = time.time()
    total_rem = max(0.0, deadline - now)
    stage1_deadline = now + 0.6 * total_rem
    it = 0

    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # Stage 1 – Adam on smooth max
        # --------------------------------------------------------------
        if time.time() < stage1_deadline:
            conv = _full_convolution(a)
            w = _smooth_weights(conv, alpha)
            grad = _conv_with_weights(a, w)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # projection
            a = _project_simplex(a, S)

            # temperature schedule (sharpen the soft‑max)
            alpha = min(alpha * 1.02, 300.0)

        # --------------------------------------------------------------
        # Stage 2 – sub‑gradient on the true max‑norm
        # --------------------------------------------------------------
        else:
            conv = _full_convolution(a)
            k_max = int(np.argmax(conv))

            # sub‑gradient of max entry (only entries that overlap)
            g = np.zeros_like(a)
            i_low = max(0, k_max - (n - 1))
            i_high = min(n - 1, k_max)
            idx_i = np.arange(i_low, i_high + 1)
            idx_j = k_max - idx_i
            g[idx_i] = a[idx_j]

            eta = 0.6 / np.sqrt(it)          # diminishing step size
            a = a - eta * g
            a = _project_simplex(a, S)

        # --------------------------------------------------------------
        # Exploration moves (mass‑swap & jitter)
        # --------------------------------------------------------------
        if rng.random() < 0.05:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.02
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        # --------------------------------------------------------------
        # Book‑keeping
        # --------------------------------------------------------------
        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()

        # cheap early stop (no improvement for many steps)
        if it % 8000 == 0 and best_score == cur_score:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function (non‑negative vector) that minimises
    ``evaluate_sequence``.  The routine runs many random restarts,
    progressively upsamples the current best solution and refines it
    with the combined Adam / sub‑gradient optimiser described above.
    Returns the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Keep a small safety margin for the external evaluation
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Initialise from a previously found construction if available
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    if 'height_sequence_1' in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            cand = np.clip(cand, 0.0, None)
            if cand.sum() > 0:

                n0 = cand.shape[0]
                cand = cand / cand.sum() * np.sqrt(2.0 * n0)
                best_seq = cand.copy()
                best_score = np.max(_full_convolution(best_seq))
                print(f"[warm‑start] loaded previous seq, score={best_score:.6f}")
        except Exception:
            best_seq = None

    # ------------------------------------------------------------------
    # Parameter ranges
    # ------------------------------------------------------------------
    min_n = 2000
    max_n = 8000
    attempts = 0

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    while time.time() < deadline:
        attempts += 1

        # --------------------------------------------------------------
        # Choose problem size and initial vector
        # --------------------------------------------------------------
        if best_seq is None or attempts % 7 == 0:
            # Fresh random start
            n = int(rng.integers(min_n, max_n + 1))
            a0 = rng.random(n)
        else:
            # Occasionally upscale the current best
            if rng.random() < 0.3 and len(best_seq) < max_n:
                n = min(int(len(best_seq) * 1.5), max_n)
                a0 = _upsample(best_seq, n)
            else:
                n = len(best_seq)
                a0 = best_seq.copy()
                # tiny jitter to escape flat regions
                jitter = rng.normal(scale=a0.mean() * 1e-4, size=n)
                a0 = a0 + jitter

        # Normalise onto the simplex Σ a_i = √(2·n)
        S = np.sqrt(2.0 * n)
        if a0.sum() == 0.0:
            a0 = np.full(n, S / n)
        else:
            a0 = a0 / a0.sum() * S

        # --------------------------------------------------------------
        # Allocate a time slice for this trial (≈30 s)
        # --------------------------------------------------------------
        slice_deadline = min(time.time() + 30.0, deadline)

        # Run the optimiser
        a_opt, score_opt = _optimize_combined(a0, slice_deadline, rng)

        # Keep the global best

            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[global] attempt={attempts} n={n} new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final refinement on the best candidate (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        # Try to push to the maximum allowed length for a final push
        if len(best_seq) < max_n:
            n_target = max_n
            a0 = _upsample(best_seq, n_target)
            a0 = a0 / a0.sum() * np.sqrt(2.0 * n_target)
        else:
            a0 = best_seq.copy()
            a0 = _project_simplex(a0, np.sqrt(2.0 * a0.shape[0]))

        a_opt, score_opt = _optimize_combined(a0, deadline, rng)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[final‑refine] length={len(best_seq)} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Return result

    return final_seq
