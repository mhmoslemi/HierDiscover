The objective after the mandatory ℓ¹‑normalisation (`∑a_i = √(2 n)`) is simply  

```
M(a) = max_k (a * a)[k]                (self‑convolution)
```

We minimise `M(a)` by alternating two complementary steps:

1. **LP‑boosting** –  
   For the current sequence `f` we solve the linear programme  

   ```
   maximise   Σ g_i
   subject to  (f ⧖ g)_k ≤ M(f)   for all k          (cross‑convolution)
               g_i ≥ 0
   ```

   The solution `g` puts all mass where the entries of `f` are *small*,
   i.e. where they contribute little to the maximal convolution peak.
   After scaling `g` to the required ℓ¹‑norm we replace a fraction `t`
   of `f` by `g` (`f ← (1‑t) f + t g`).  Repeating this step
   strictly decreases the peak `M`.

2. **Targeted simulated annealing** –  
   With the sequence already close to a local optimum we perform a
   specialised SA that moves mass from indices that heavily contribute
   to the current peak to indices that contribute little.  The move is
   accepted with a temperature‑controlled Metropolis rule, allowing the
   algorithm to escape shallow minima.

Both phases respect the time budget.  The LP‑boost is fast (a single
`linprog` call on a dense matrix of size `(2n‑1) × n`), while the SA
phase provides fine‑grained local optimisation.

The whole pipeline is wrapped in `propose_candidate`.  If a global
variable `height_sequence_1` is present it is used as the starting
point, otherwise a random edge‑heavy sequence is generated.

```python
"""
Propose a step‑function (non‑negative sequence) that minimises
evaluate_sequence().  The algorithm consists of

    1. LP‑boosting (convex combination with a solution of a linear program)
    2. Targeted simulated annealing (mass transfers guided by the current peak)

Both phases keep the sequence on the ℓ¹‑simplex Σ a_i = √(2 n).
The routine respects a 1000 s time budget and prints intermediate scores.
"""

import time
import numpy as np
from scipy import optimize as opt


# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (FFT, length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())     # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x_i≥0, Σx_i = s} (Duchi et al. 2008)."""
    if s <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear‑program solver for the boosting step
# ----------------------------------------------------------------------


def _solve_lp_max_sum(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve  maximise Σ g_i
            s.t.   (a * g)_k ≤ M   for all k = 0 … 2n‑2
                   g_i ≥ 0

    Returns the optimal g (may be all‑zero on failure).
    """
    eps = 1e-12                 # avoid zero columns – keep LP bounded
    a_pos = np.maximum(a, eps)

    n = a_pos.size
    rows = 2 * n - 1

    # Build the convolution matrix A where   (A·g)_k = (a * g)_k
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a_pos          # column j is shifted copy of a

    b = np.full(rows, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)   # maximise → minimise –c·g

    # Use the high‑performance HiGHS solver (available in recent SciPy)
    res = opt.linprog(c, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True})

    if res.success and res.x is not None:
        return res.x
    # fallback – keep the current sequence (no improvement possible)
    return a.copy()


# ----------------------------------------------------------------------
# LP‑boosting phase
# ----------------------------------------------------------------------


def _boost_sequence(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Repeatedly improve the sequence by convex combinations with the LP
    solution.  Stops when time expires or no further improvement is
    observed.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_M = _max_convolution_fft(a)

    t = 0.5                     # initial mixing weight
    it = 0

    while time.time() < deadline and it < 200:
        M = _max_convolution_fft(a)

        # LP step
        g = _solve_lp_max_sum(a, M)

        # scale g to the required ℓ¹‑norm
        sum_g = g.sum()
        if sum_g < 1e-12:
            break
        g = g * (target_sum / sum_g)

        # convex combination
        a_new = (1.0 - t) * a + t * g
        # re‑normalise (numerical drift)
        a_new = a_new * (target_sum / a_new.sum())

        new_M = _max_convolution_fft(a_new)

        if new_M < best_M - 1e-8:
            best_M = new_M
            best_a = a_new.copy()
            a = a_new
            # successful step – try a slightly larger mixing weight
            t = min(t * 1.05, 0.9)
        else:
            # step did not improve – shrink mixing weight
            t = max(t * 0.6, 0.01)

        it += 1

    return best_a


# ----------------------------------------------------------------------
# Targeted simulated‑annealing refinement (slightly tuned)
# ----------------------------------------------------------------------


def _targeted_simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Move mass from indices that heavily contribute to the current maximal
    convolution entry to indices that contribute little.  Accept uphill
    moves with a decreasing temperature.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # start temperature proportional to the current objective
    T = best_score * 0.05 + 1e-12
    cooling = 0.9997
    rng = np.random.default_rng()
    max_transfer = 0.30          # up to 30 % of the selected mass

    while time.time() < deadline:
        # current convolution and its maximal index
        b = _full_convolution_fft(a)
        k_max = int(np.argmax(b))

        # indices that actually affect b[k_max]
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idx = np.arange(lo, hi + 1)                     # possible i
        contrib = a[idx] * a[k_max - idx]               # a_i·a_{k‑i}
        weight_sum = contrib.sum()

        if weight_sum > 0.0:
            probs = contrib / weight_sum
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        # destination j ≠ i uniformly
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        # transfer a random fraction of the mass at i to j
        delta = a[i] * rng.random() * max_transfer
        if delta <= 0.0:
            continue
        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score

        accept = False
        if cur_score <= best_score:
            accept = True
        elif T > 0.0 and rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA improvement: {best_score:.6f}")
        else:
            # revert move
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # occasional global jitter to escape deep minima
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = a * (np.sqrt(2.0 * n) / a.sum())
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA jitter improvement: {best_score:.6f}")

    return best


# ----------------------------------------------------------------------
# Random initialisers (diverse families) – used only if no global seed
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Generate a random non‑negative sequence on the ℓ¹‑simplex."""
    target_sum = np.sqrt(2.0 * n)
    kind = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if kind == 0:                         # pure power‑law (decreasing)
        p = rng.uniform(0.15, 1.0)
        a = (idx + 1.0) ** (-p)

    elif kind == 1:                       # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif kind == 2:                       # mix power‑law & exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif kind == 3:                       # Dirichlet (uniform simplex)
        a = rng.dirichlet(alpha=np.ones(n))

    elif kind == 4:                       # symmetric edge‑heavy
        p = rng.uniform(0.2, 1.2)
        left = (idx + 1.0) ** (-p)
        right = (n - idx) ** (-p)
        a = left + right

    else:                                 # Beta‑shaped (U‑shaped or bell‑shaped)
        alpha = rng.uniform(0.2, 2.0)
        beta = rng.uniform(0.2, 2.0)
        x = (idx + 0.5) / n
        a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))

    # tiny Gaussian perturbation and projection onto the simplex
    a += rng.normal(scale=target_sum * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target_sum)
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` under the supplied time budget.
    The algorithm:
        * starts from the global variable ``height_sequence_1`` if present,
          otherwise from a random edge‑heavy sequence,
        * refines it with the LP‑boosting phase,
        * finishes with targeted simulated annealing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Initialise – try to reuse a previously found construction
    # ------------------------------------------------------------------
    global_seq = globals().get("height_sequence_1")
    if isinstance(global_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(global_seq, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        best_seq = a0.copy()
        best_score = _max_convolution_fft(best_seq)
        print(f"[{time.time():.1f}] Start from global height_sequence_1 "
              f"(n={n0})  max‑conv={best_score:.6f}")
    else:
        # fallback – generate a random edge‑heavy start
        n0 = 1200
        best_seq = _random_initial_sequence(n0, rng)
        best_score = _max_convolution_fft(best_seq)
        print(f"[{time.time():.1f}] No global seed – using random start "
              f"(n={n0})  max‑conv={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – LP‑boosting (use roughly half of the time budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        boost_deadline = time.time() + remaining * 0.55
        boosted = _boost_sequence(best_seq, boost_deadline)
        boosted_score = _max_convolution_fft(boosted)
        if boosted_score < best_score:
            best_seq, best_score = boosted, boosted_score
            print(f"[{time.time():.1f}] After boosting: max‑conv={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – targeted simulated annealing (use the rest of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        refined = _targeted_simulated_annealing(best_seq, deadline)
        refined_score = _max_convolution_fft(refined)
        if refined_score < best_score:
            best_seq, best_score = refined, refined_score
            print(f"[{time.time():.1f}] After SA: max‑conv={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final clean‑up – drop trailing near‑zero entries and enforce exact sum
    # ------------------------------------------------------------------
    eps = 1e-12
    nz = np.where(best_seq > eps)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # final projection onto the ℓ¹‑simplex (eliminates accumulated drift)
    target_sum = np.sqrt(2.0 * best_seq.size)
    best_seq = _project_to_simplex(best_seq, target_sum)

    # ------------------------------------------------------------------
    # Sanity check with the official evaluator
    # ------------------------------------------------------------------
    try:
        ev = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence = {ev:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best max‑conv = {best_score:.6f}   "
          f"length = {best_seq.size}")

    return best_seq.tolist()
```
