"""
Search for a non‑negative step sequence that minimises the self‑convolution
peak returned by ``evaluate_sequence``.

The algorithm builds on the earlier Adam + sub‑gradient scheme but adds
several enhancements:

* **Symmetrisation** – after each projection the vector is mirrored,
  which keeps the search in the (empirically) favourable symmetric sub‑space.
* **Dynamic temperature** – the soft‑max temperature grows to a larger
  maximum (500) for a sharper approximation of the true max.
* **Adaptive learning‑rate** – the Adam step size scales with √n so that
  updates stay effective for larger problem sizes.
* **Progressive up‑sampling** – start from moderate lengths (≈1‑2 k),
  repeatedly interpolate the current best vector to a larger size,
  re‑normalise and re‑optimise.  Larger n give more degrees of freedom and
  typically a lower peak.
* **Longer optimisation windows** – a larger fraction of the remaining
  budget is spent on the Adam phase (80 %), followed by a sub‑gradient
  refinement.
* **Final local‑search polishing** – a cheap hill‑climbing stage that
  shuffles small amounts of mass between random positions while the time
  budget permits.

All steps respect the non‑negativity and simplex constraints, and the
best sequence found before the deadline is returned as a plain Python
list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean, O(n log n) sorting)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper – symmetrisation
# ----------------------------------------------------------------------


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force symmetry v[i] == v[-1-i] without changing the sum."""
    return (v + v[::-1]) * 0.5


# ----------------------------------------------------------------------
# FFT‑based full convolution a * a (length 2·n‑1)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Gradient of the smooth max (soft‑max) – used by Adam
# ----------------------------------------------------------------------


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g_j = 2 Σ_i w_{i+j} a_i efficiently via FFT.
    a – length n, w – length 2n‑1
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1            # = 3n‑2
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)            # extracts the needed slice
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights (smooth approximation of max)
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """w_i = exp(alpha·conv_i) / Σ_j exp(alpha·conv_j)   (numerically stable)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Linear‑interpolation up‑sampling
# ----------------------------------------------------------------------


def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a to length new_n (inclusive of endpoints)."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=float)
    return np.interp(x_new, x_old, a)


# ----------------------------------------------------------------------
# Adam optimisation on a smooth‑max surrogate
# ----------------------------------------------------------------------


def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max approximation of max(conv).
    Returns the best vector (on the simplex) and its max‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)                 # target L1‑norm

    # Adam hyper‑parameters – step size scaled with √n
    lr = 0.5 * np.sqrt(n / 1000.0)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    # temperature (soft‑max sharpness)
    alpha = 0.7
    max_alpha = 500.0

    while time.time() < deadline:
        t += 1

        # compute convolution and soft‑max weights
        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # project onto simplex and enforce symmetry
        a = _project_simplex(a, S)
        a = _symmetrize(a)

        # occasional random mass‑swap to diversify
        if rng.random() < 0.04:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.015
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)
                a = _symmetrize(a)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # increase temperature (sharpen the approximation)
        alpha = min(alpha * 1.025, max_alpha)

        # cheap early‑stop test
        if t % 6000 == 0 and best_score == cur_score:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑norm
# ----------------------------------------------------------------------


def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent that directly reduces the current worst convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.3
    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # build sub‑gradient (see discussion in the original code)
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)
        a = _symmetrize(a)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # random jitter to escape plateaus
        if rng.random() < 0.06:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.025
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand = _symmetrize(cand)
                cand_score = np.max(_full_convolution(cand))
                if cand_score < cur_score or rng.random() < 0.2:
                    a = cand
                    cur_score = cand_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()

        # cheap early‑stop test
        if it % 9000 == 0 and best_score == cur_score:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimiser: Adam → sub‑gradient
# ----------------------------------------------------------------------


def _optimize_combined(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run Adam for the first ~80 % of the allocated time,
    then finish with a sub‑gradient phase.
    """
    now = time.time()
    remain = deadline - now
    if remain <= 0.0:
        return a0, np.max(_full_convolution(a0))

    # Adam phase
    adam_deadline = now + 0.78 * remain
    a_mid, sc_mid = _optimize_adam(a0, adam_deadline, rng)

    # Sub‑gradient phase (if time left)
    if time.time() < deadline:
        a_fin, sc_fin = _optimize_subgrad(a_mid, deadline, rng)
        if sc_fin < sc_mid:
            return a_fin, sc_fin
    return a_mid, sc_mid


# ----------------------------------------------------------------------
# Light‑weight local refinement (hill‑climbing)
# ----------------------------------------------------------------------


def _local_refine(a: np.ndarray,
                  deadline: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Very cheap random mass‑transfer hill‑climbing while time permits.
    """
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    best = a.copy()
    best_score = np.max(_full_convolution(best))

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or best[i] <= 0.0:
            continue
        delta = best[i] * rng.random() * 0.03
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        cand = _symmetrize(cand)
        cand_score = np.max(_full_convolution(cand))
        if cand_score < best_score:
            best = cand
            best_score = cand_score
    return best, best_score

    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        attempts += 1

        # --------------------------------------------------------------
        # Choose problem size
        # --------------------------------------------------------------
        if best_seq is None or (time.time() - last_improve > 45):
            # fresh start – moderate size to explore diverse shapes
            n = int(rng.integers(900, 2500))
            a0 = rng.random(n)
        else:
            # try to enlarge the current best (up‑sampling) or tweak it
            if rng.random() < 0.33 and len(best_seq) < 10000:
                # up‑sample – factor between 1.4 and 2.0
                factor = rng.uniform(1.4, 2.0)
                n = min(int(len(best_seq) * factor), 10000)
                a0 = _upsample(best_seq, n)
            else:
                n = len(best_seq)
                a0 = best_seq.copy()

            # small random jitter before re‑optimisation
            jitter = rng.normal(scale=a0.mean() * 1e-4, size=n)
            a0 = a0 + jitter
            a0 = np.clip(a0, 0.0, None)

        # --------------------------------------------------------------
        # Normalise onto the simplex Σa = √(2·n)
        # --------------------------------------------------------------
        S = np.sqrt(2.0 * n)
        a_sum = a0.sum()
        if a_sum == 0.0:
            a0 = np.full(n, 1.0)
            a_sum = a0.sum()
        a0 = a0 / a_sum * S
        a0 = _symmetrize(a0)                     # start from a symmetric point

        # --------------------------------------------------------------
        # Allocate a time slice for this trial (≈20 s, respecting the global deadline)
        # --------------------------------------------------------------
        slice_deadline = min(time.time() + 20.0, deadline)

        # --------------------------------------------------------------
        # Run the two‑phase optimiser
        # --------------------------------------------------------------
        a_opt, sc_opt = _optimize_combined(a0, slice_deadline, rng)

        # --------------------------------------------------------------
        # Keep the global best
        # --------------------------------------------------------------
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            last_improve = time.time()
            print(f"[global] attempt={attempts} n={n} new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing if time permits (≈10‑15 s)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        polish_deadline = min(deadline, time.time() + 12.0)
        best_seq, best_score = _local_refine(best_seq, polish_deadline, rng)

    # ------------------------------------------------------------------
    # Return the result in plain Python list form
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]                     # fallback – never happens in practice
    else:
        final_seq = best_seq.tolist()

    # One last external evaluation (for logging only – the harness will do its own)
    try:
        external = evaluate_sequence(final_seq)
        print(f"[final] external score = {external:.6f} (internal best = {best_score:.6f}) "
              f"length = {len(final_seq)}")
    except Exception as e:
        print(f"[final] evaluation failed: {e}")

    return final_seq
