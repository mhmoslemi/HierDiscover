def _adam_opt(a0, deadline):
    a = a0.copy()
    n = len(a)
    target_sum = np.sqrt(2.0*n)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = target_sum * 0.2  # step size scaling
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    beta = 0.5
    beta_max = 8192.0
    beta_factor = 1.4
    rng = np.random.default_rng()
    while time.time() < deadline:
        t += 1
        # compute gradient based on softmax with current beta
        g, b = _gradient_softmax(a, beta)
        # optional: update beta periodically
        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)
        # Adam update
        m = beta1*m + (1-beta1)*g
        v = beta2*v + (1-beta2)*(g**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)
        # occasional jitter
        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
    return a

    target = np.sqrt(2.0*n)

    beta_max = 1e5
    beta_factor = 1.3
    step_size = target * 0.18
    momentum = np.zeros_like(a)
    rho = 0.9
    step = 0
    while time.time() < deadline:

def _subgradient_max(a):
    n = a.shape[0]
    b = _full_convolution(a)
    max_b = b.max()
    # boolean mask of entries not far from max
    mask = (max_b - b) < 1e-12 * max_b
    # if zero mask due to tolerance too high, fallback to equality using tolerance absolute
    if not np.any(mask):
        mask = (b == max_b)
    p = mask.astype(float)
    p_sum = p.sum()
    if p_sum == 0:
        # fallback uniform
        p = np.ones_like(b) / b.size
    else:
        p /= p_sum
    # compute correlation gradient
    a_rev = a[::-1]
    conv_len = n + (2*n - 1) - 1
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n-1: n - 1 + n]
    return g

"""
Propose a candidate step‑function that minimizes ``evaluate_sequence``.
The algorithm combines several ideas:

1. **Simplex projection** – we always keep the L1‑norm equal to
   ``sqrt(2·n)`` (the objective is scale–invariant, but fixing the norm
   stabilises the optimiser).

2. **Smooth gradient descent (Adam‑style)** – we minimise a smooth
   log‑sum‑exp approximation of the maximum convolution.  The inverse
   temperature β is increased exponentially; after β becomes large we
   switch to the exact sub‑gradient (β → ∞).

3. **Targeted hill‑climbing** – after the gradient phase we repeatedly
   move a small amount of mass from an index that contributes heavily to
   the current peak (the index i with large a[i]·a[k_max‑i]) to a
   destination that does **not** belong to the peak.  The convolution is
   updated analytically in O(n) per move.  An annealed temperature
   permits occasional uphill moves.

4. **Length schedule & restarts** – the search is performed for several
   lengths (up to 30 000).  For each length we try a few random restarts,
   keep the best candidate, and optionally up‑sample the current best
   when moving to a larger length.

The whole routine respects the wall‑clock budget (default 1000 s) and
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_full_convolution(a).max())


# ----------------------------------------------------------------------
# Gradient of the smooth max (log‑sum‑exp) and sub‑gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with g = 2·(a ⋆ p) and p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # --- compute b = a*a ---
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # --- soft‑max probabilities ---
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # --- gradient g = 2·(a ⋆ p) ---
    a_rev = a[::-1]                     # reversed a
    conv_len = n + (2 * n - 1) - 1       # = 3·n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _subgradient_max(a: np.ndarray):
    """
    Sub‑gradient of the (non‑smooth) max‑convolution.
    Returns gradient g of size n.
    """
    n = a.size
    b = _full_convolution(a)
    max_b = b.max()

    # indices that are essentially active
    mask = (max_b - b) < 1e-12 * max_b
    if not np.any(mask):
        mask = (b == max_b)

    p = mask.astype(float)
    s = p.sum()
    if s == 0:
        p = np.ones_like(b) / b.size
    else:
        p /= s

    # correlation a⋆p = (reverse(a)) * p
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g

def _adam_like_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Adam‑style projected optimisation on the smoothed max.
    β is annealed; after β exceeds a threshold we switch to the exact
    sub‑gradient.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # optimiser hyper‑parameters
    beta = 0.5
    beta_max = 1e5
    beta_factor = 1.3

    lr_base = target_sum * 0.12
    eps = 1e-8
    beta1, beta2 = 0.9, 0.999
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    rng = np.random.default_rng()
    t = 0

    while time.time() < deadline:
        t += 1

        if beta < 1e4:
            g, _ = _gradient_softmax(a, beta)
        else:
            g = _subgradient_max(a)

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = lr_base * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape tiny plateaus
        if t % 260 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β annealing
        if t % 340 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


def _targeted_hill_climbing(a0: np.ndarray, b0: np.ndarray,
                           deadline: float):
    """
    Hill climbing that preferentially removes mass from indices that
    contribute to the current peak of the convolution.
    Returns (best_a, best_score) where best_score = max(b).
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    # temperature proportional to current peak, cools slowly
    T = cur_score * 0.1 + 1e-12
    cooling = 0.9999

    while time.time() < deadline:
        # --- locate the (one) peak index ---
        k_max = int(np.argmax(b))

        # --- compute contributions of each index to this peak ---
        low = max(0, k_max - (n - 1))
        high = min(n - 1, k_max)
        idx = np.arange(low, high + 1, dtype=int)
        jdx = k_max - idx
        contrib = np.zeros(n, dtype=float)
        contrib[idx] = a[idx] * a[jdx]

        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs_src = contrib / total_contrib
        else:
            # fallback: choose source proportional to current mass
            probs_src = a / a.sum() if a.sum() > 0 else np.full(n, 1.0 / n)

        i = rng.choice(n, p=probs_src)

        # --- pick destination that does NOT recreate the peak ---
        # simple strategy: uniform sampling with a few re‑tries
        attempts = 0
        while True:
            j = rng.integers(0, n)
            if j != i and (i + j) != k_max:
                break
            attempts += 1
            if attempts > 10:
                break

        # --- amount of mass to move ---
        max_delta = a[i] * 0.5          # move at most half of source
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        # --- update convolution analytically (O(n)) ---
        b_new = b.copy()
        # removal from i
        b_new[i:i + n] -= 2.0 * delta * a
        # addition to j
        b_new[j:j + n] += 2.0 * delta * a

        # quadratic correction terms
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # --- Metropolis acceptance ---
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-(new_score - cur_score) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill‑climb improvement: {best_score:.6f}")

        T *= cooling

    return best_a, best_score

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Generate a random non‑negative vector on the simplex Σa = √(2 n)."""
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                               # pure power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                             # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                             # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                             # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                             # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )
    else:                                          # near‑uniform jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a onto a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

    # ------------------------------------------------------------------

        best_score = _max_convolution_fft(a0)

        # Give longer runs to larger n but keep a minimum

        # --------------------------------------------------------------

                    print(f"[{time.time():.1f}] Hill‑climb improvement (n={n}) score={score_hc:.6f}")

            # Use the newly found best as start for next restart

            a_start = _project_to_simplex(a_start, target_sum)

            a_final, b_final, deadline

    # Clean‑up: drop trailing near‑zeros for neatness
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Official evaluation (objective = max of the self‑convolution)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
