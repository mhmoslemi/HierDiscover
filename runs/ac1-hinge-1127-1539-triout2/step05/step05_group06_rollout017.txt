"""
Two‑phase optimiser for the self‑convolution peak.

Phase‑1: Adam optimiser on a smooth log‑sum‑exp surrogate of the peak.
Phase‑2: Greedy zero‑sum sub‑gradient step with a back‑tracking line search.
Multiple restarts (random and a few deterministic window shapes) are tried
and the best result is up‑sampled to larger dimensions.
The routine respects the supplied time budget (default ≈ 1000 s) and
prints progress.
"""

import time
import numpy as np

def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x)=s}."""
    if s < 0:
        raise ValueError('Simplex sum must be non‑negative')
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    # find the rho index
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv

def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate.
    g_j = 2 * Σ_i w_{i+j} * a_i   (full convolution of a with reversed w).
    """
    n = a.shape[0]
    # reverse w for convolution
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    # slice that corresponds to gradient index
    idx = 2 * n - 2 - np.arange(n)          # descending slice of length n
    grad = 2.0 * full[idx]
    return grad

def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights w_i = exp(alpha * conv_i) / Σ_j exp(alpha * conv_j)."""
    # use stable exponentiation
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shift = np.exp(scaled - max_scaled)
    w = exp_shift / exp_shift.sum()
    return w

def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a to a new length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=float)
    return np.interp(x_new, x_old, a)

def _initial_sequences(n: int, rng: np.random.Generator):
    """
    Generate a small list of promising initial vectors for size n.
    Returns a list of numpy arrays already projected onto the simplex
    Σ a_i = sqrt(2·n).
    """
    S = np.sqrt(2.0 * n)
    seqs = []

    # 1. Random dense vector
    a = rng.random(n)
    a = a / a.sum() * S
    seqs.append(a)

    # 2. Exponential decaying shape (approximately geometric)
    # Choose decay rate so that the effective support covers the whole vector.
    # We parametrize by a factor c in (0, 5) controlling sharpness.
    c = rng.uniform(0.5, 3.0)   # a moderate range
    idx = np.arange(n, dtype=float)
    center = (n - 1) / 2.0
    a = np.exp(-c * np.abs(idx - center))
    a = a / a.sum() * S
    seqs.append(a)

    # 3. Half‑sine window (positive bump in the middle)
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    a = a / a.sum() * S
    seqs.append(a)

    # 4. Hann window (zero at borders, peak in the centre)
    a = 0.5 * (1 - np.cos(2 * np.pi * np.arange(n) / (n - 1)))
    a = a / a.sum() * S
    seqs.append(a)

    # 5. Linear ramp from 0 up to centre and symmetrically back (triangular)
    ramp_up = np.linspace(0.0, 1.0, n // 2 + 1, endpoint=False)
    ramp_down = ramp_up[::-1]
    a = np.concatenate([ramp_up, ramp_down[:n - len(ramp_up)]])
    a = a / a.sum() * S
    seqs.append(a)

    # 6. Random sparse vector (few non‑zero entries)
    k = max(2, int(np.sqrt(n)))          # about sqrt(n) non‑zeros
    a = np.zeros(n)
    idxs = rng.choice(n, size=k, replace=False)
    a[idxs] = rng.random(k)
    a = a / a.sum() * S
    seqs.append(a)

    return seqs

def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   max_iters: int = 5000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the log‑sum‑exp surrogate.
    Returns the best vector and its peak convolution (exact max).
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.35
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    # initialise temperature (soft‑max sharpness)
    alpha = 0.8

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        # compute convolution and soft‑max weights
        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)

        # gradient of the smooth objective
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex
        a = _project_simplex(a, S)

        # occasional random mass‑exchange to escape plateaus
        if rng.random() < 0.03:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = a[i] * rng.random() * 0.02
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # increase temperature (sharpen soft‑max)
        alpha = min(alpha * 1.035, 5000.0)

        # cheap early‑stop: if no improvement for a while
        if t % 1500 == 0 and best_score == cur_score:
            break

    return best_a, best_score

def _subgrad_line_search(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """
    Greedy sub‑gradient descent with a back‑tracking line search.
    Works directly on the exact max‑norm.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        conv = _full_convolution(a)
        max_val = conv.max()
        # get all indices that are (almost) maximal
        idxs = np.where(conv >= max_val - 1e-12)[0]

        # aggregate sub‑gradient over all maximal positions
        g = np.zeros_like(a)
        for k in idxs:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i_arr = np.arange(lo, hi + 1)
            j_arr = k - i_arr
            g[i_arr] += a[j_arr]

        # construct a zero‑sum search direction
        sum_g = g.sum()
        d = -g + sum_g / n

        # maximum feasible step respecting non‑negativity
        neg = d < -1e-14
        if np.any(neg):
            t_max = np.min(a[neg] / -d[neg])
        else:
            t_max = np.inf

        # try a geometric schedule of step size
        step = min(t_max * 0.9, 1.0)   # never jump more than 1.0 (very safe)
        if step < 1e-12:
            # inject a tiny random jitter to regain mobility
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = a[i] * rng.random() * 0.02
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)
            continue

        a_new = a + step * d
        a_new = _project_simplex(a_new, S)

        new_score = np.max(_full_convolution(a_new))

        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
        else:
            # If no improvement, shrink the step and retry once
            step = step * 0.5
            if step < 1e-12:
                # fallback to random jitter
                i, j = rng.integers(0, n, size=2)
                if i != j and a[i] > 0:
                    delta = a[i] * rng.random() * 0.02
                    a[i] -= delta
                    a[j] += delta
                    a = _project_simplex(a, S)
                continue
            a_tmp = a + step * d
            a_tmp = _project_simplex(a_tmp, S)
            tmp_score = np.max(_full_convolution(a_tmp))
            if tmp_score < best_score:
                best_score = tmp_score
                best_a = a_tmp.copy()
                a = a_tmp

        # occasional random perturbation to escape flat regions
        if rng.random() < 0.04:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = a[i] * rng.random() * 0.015
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = np.max(_full_convolution(cand))
                if cand_score < best_score or rng.random() < 0.2:
                    a = cand
                    best_score = cand_score
                    best_a = cand.copy()

    return best_a, best_score

def _optimize_combined(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run Adam for a fraction of the allocated time then finish with the
    zero‑sum sub‑gradient line‑search.
    """
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    # split time (≈60 % Adam, 40 % sub‑gradient)
    adam_deadline = now + 0.6 * remaining
    a_mid, score_mid = _adam_optimize(a0, adam_deadline, rng)

    if time.time() < deadline:
        a_fin, score_fin = _subgrad_line_search(a_mid, deadline, rng)
        if score_fin < score_mid:
            return a_fin, score_fin
    return a_mid, score_mid

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Main entry required by the platform.
    Uses many short trials with diverse initialisations, keeps the global
    best and finally returns it as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a short margin for the final external evaluation
    deadline = start_time + budget_s - 4.0

    best_seq = None
    best_score = np.inf
    attempts = 0
    last_improve = start_time

    # Preferred sizes (growing) – we start relatively small and let the
    # optimiser decide to up‑sample the current best when it proves useful.
    size_choices = [800, 1500, 3000, 5000, 8000]

    while time.time() < deadline:
        attempts += 1

        # decide problem size for this trial
        if best_seq is None or (time.time() - last_improve > 30.0):
            n = int(rng.choice(size_choices))
            init_vectors = _initial_sequences(n, rng)
        else:
            # occasionally try up‑sampling the current champion
            if rng.random() < 0.35 and len(best_seq) < 10000:
                factor = rng.uniform(1.2, 2.0)
                n = min(int(len(best_seq) * factor), 10000)
                init_vectors = [_upsample(best_seq, n)]
            else:
                n = len(best_seq)
                # add a tiny jitter to the champion before optimisation

"""
Two‑phase optimiser for the self‑convolution peak.

Phase‑1: Adam optimiser on a smooth log‑sum‑exp surrogate of the peak.
Phase‑2: Greedy zero‑sum sub‑gradient steps with a simple back‑tracking line search.
Several diverse initialisations (random, exponential, sinusoidal windows, etc.)
are tried, and the best result is up‑sampled to larger dimensions when it shows
promise.  The routine respects a supplied time budget (default ≈1000 s) and
prints progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x ≥ 0, Σ x = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate.
    g_j = 2 Σ_i w_{i+j} a_i .
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights (smooth approximation of max)
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """w_i = exp(alpha·conv_i) / Σ_j exp(alpha·conv_j)  (stable)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Upsampling helper (linear interpolation)
# ----------------------------------------------------------------------


def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a to length new_n."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=float)
    return np.interp(x_new, x_old, a)


# ----------------------------------------------------------------------
# Initial sequences (various shapes)
# ----------------------------------------------------------------------


def _initial_sequences(n: int, rng: np.random.Generator):
    """
    Produce a few diverse starter vectors for size n.
    Each vector is projected onto the simplex Σ a_i = √(2·n).
    """
    S = np.sqrt(2.0 * n)
    seqs = []

    # (1) dense random
    a = rng.random(n)
    a = a / a.sum() * S
    seqs.append(a)

    # (2) exponential decay centred in the middle
    c = rng.uniform(0.5, 3.0)
    idx = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    a = np.exp(-c * np.abs(idx - centre))
    a = a / a.sum() * S
    seqs.append(a)

    # (3) half‑sine window
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    a = a / a.sum() * S
    seqs.append(a)

    # (4) Hann window
    a = 0.5 * (1 - np.cos(2 * np.pi * np.arange(n) / (n - 1)))
    a = a / a.sum() * S
    seqs.append(a)

    # (5) triangular ramp
    ramp_up = np.linspace(0.0, 1.0, n // 2 + 1, endpoint=False)
    ramp_down = ramp_up[::-1]
    a = np.concatenate([ramp_up, ramp_down[:n - len(ramp_up)]])
    a = a / a.sum() * S
    seqs.append(a)

    # (6) sparse random (≈√n non‑zeros)
    k = max(2, int(np.sqrt(n)))
    a = np.zeros(n)
    idxs = rng.choice(n, size=k, replace=False)
    a[idxs] = rng.random(k)
    a = a / a.sum() * S
    seqs.append(a)

    return seqs


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max‑norm
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   max_iters: int = 5000) -> tuple[np.ndarray, float]:
    """Adam optimisation of the log‑sum‑exp surrogate."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.35
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    # temperature for the soft‑max
    alpha = 0.8
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex
        a = _project_simplex(a, S)

        # occasional random mass‑swap (exploration)
        if rng.random() < 0.03:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = a[i] * rng.random() * 0.02
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # sharpen the soft‑max
        alpha = min(alpha * 1.035, 5000.0)

        if t % 1500 == 0 and best_score == cur_score:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient with line‑search (direct max norm)
# ----------------------------------------------------------------------


def _subgrad_line_search(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """
    Greedy sub‑gradient descent on max(conv) with a simple back‑tracking
    line search that respects non‑negativity.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        conv = _full_convolution(a)
        max_val = conv.max()
        # all (near) maximal positions – helps when there are ties
        idxs = np.where(conv >= max_val - 1e-12)[0]

        # aggregate sub‑gradient over those positions
        g = np.zeros_like(a)
        for k in idxs:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i_arr = np.arange(lo, hi + 1)
            j_arr = k - i_arr
            g[i_arr] += a[j_arr]

        # zero‑sum search direction
        sum_g = g.sum()
        d = -g + sum_g / n

        # largest feasible step (keep a ≥ 0)
        neg = d < -1e-14
        if np.any(neg):
            t_max = np.min(a[neg] / -d[neg])
        else:
            t_max = np.inf

        step = min(t_max * 0.9, 1.0)   # safe upper bound
        if step < 1e-12:
            # inject tiny jitter if we get stuck
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = a[i] * rng.random() * 0.02
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)
            continue

        a_new = a + step * d
        a_new = _project_simplex(a_new, S)
        new_score = np.max(_full_convolution(a_new))

        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            continue

        # if no improvement, try a smaller step once
        step *= 0.5
        if step < 1e-12:
            # fallback jitter
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = a[i] * rng.random() * 0.02
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)
            continue
        a_tmp = a + step * d
        a_tmp = _project_simplex(a_tmp, S)
        tmp_score = np.max(_full_convolution(a_tmp))
        if tmp_score < best_score:
            best_score = tmp_score
            best_a = a_tmp.copy()
            a = a_tmp

        # occasional random perturbation
        if rng.random() < 0.04:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = a[i] * rng.random() * 0.015
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = np.max(_full_convolution(cand))
                if cand_score < best_score or rng.random() < 0.2:
                    a = cand
                    best_score = cand_score
                    best_a = cand.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimiser (Adam → sub‑gradient)
# ----------------------------------------------------------------------


def _optimize_combined(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Run Adam for ≈60 % of the allocated time, then finish with sub‑gradient."""
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    adam_deadline = now + 0.6 * remaining
    a_mid, score_mid = _adam_optimize(a0, adam_deadline, rng)

    if time.time() < deadline:
        a_fin, score_fin = _subgrad_line_search(a_mid, deadline, rng)
        if score_fin < score_mid:
            return a_fin, score_fin
    return a_mid, score_mid
