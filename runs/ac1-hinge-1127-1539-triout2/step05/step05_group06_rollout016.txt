def _optimize(a0, deadline, rng):
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))
    # Adam phase
    t = 0; m = np.zeros_like(a); v = np.zeros_like(a)
    lr = 0.3; beta1=0.9; beta2=0.999; eps=1e-8
    alpha=0.5
    alpha_max = 3000.0
    now = time.time()
    adam_deadline = now + 0.65 * (deadline - now)
    while time.time() < adam_deadline:
        t += 1
        conv = _full_convolution(a)
        cur_score = conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)
        # Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = (a + a[::-1]) * 0.5  # symmetrize
        a = _project_simplex(a, S)

        if rng.random() < 0.03:

                delta = a[i] * rng.random() * 0.01
                a[i] -= delta
                a[j] += delta
                a = (a + a[::-1]) * 0.5
                a = _project_simplex(a, S)
        # update temperature
        alpha = min(alpha * 1.02, alpha_max)
    # Subgradient phase
    t = 0
    eta0 = 0.5
    while time.time() < deadline:
        t += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]
        eta = eta0 / np.sqrt(t)
        a = a - eta * g
        a = (a + a[::-1]) * 0.5
        a = _project_simplex(a, S)

"""
Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence``.
The algorithm is a multi‑scale projected‑gradient optimiser:

* several cheap initial shapes (uniform, edge‑peaked, exponential, power‑law)
  are projected onto the ℓ1‑simplex Σa = √(2·n).
* each shape is refined by an Adam optimiser on a smooth approximation of
  max_i (a∗a)[i] (soft‑max with a temperature that is gradually increased).
  After every Adam update the vector is symmetrised (a←(a+a[::-1])/2) and
  re‑projected onto the simplex.
* a short sub‑gradient phase follows the Adam phase to drive the true
  maximum of the convolution down.
* the best vector is repeatedly up‑sampled (linear interpolation) to a
  larger problem size; each up‑sampled vector is further refined.
* the whole search runs for the allotted budget (≈1000 s) and finally
  returns the best sequence found.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max weighted sum of the convolution.
    Returns   g_j = 2 Σ_i w_{i+j} a_i .
    """
    n = a.shape[0]
    # reverse the weight vector for convolution
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    # slice that yields the gradient (see original implementation)
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights (smooth approximation of max)
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """w_i = exp(alpha·conv_i) / Σ_j exp(alpha·conv_j)   (numerically stable)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Upsampling helper (linear interpolation)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a to length new_n."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=float)
    return np.interp(x_new, x_old, a)


# ----------------------------------------------------------------------
# Core optimiser (Adam → sub‑gradient) with symmetry enforcement
# ----------------------------------------------------------------------
def _optimize(a0: np.ndarray, deadline: float, rng: np.random.Generator):
    """Run Adam on a soft‑max surrogate then a short sub‑gradient phase."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    # ----------------------- Adam phase -------------------------------
    lr = 0.30
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = 0.5          # start smooth
    alpha_max = 2500.0

    now = time.time()
    adam_deadline = now + 0.65 * (deadline - now)

    while time.time() < adam_deadline:
        t += 1
        conv = _full_convolution(a)
        cur_score = conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # enforce symmetry and project back onto the simplex
        a = (a + a[::-1]) * 0.5
        a = _project_simplex(a, S)

        # occasional random mass‑swap to improve exploration
        if rng.random() < 0.03:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.01
                a[i] -= delta
                a[j] += delta
                a = (a + a[::-1]) * 0.5
                a = _project_simplex(a, S)

        # temperature schedule – make the soft‑max sharper
        alpha = min(alpha * 1.02, alpha_max)

    # --------------------- Sub‑gradient phase -------------------------
    t = 0
    eta0 = 0.35
    while time.time() < deadline:
        t += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the true max entry (see original code)
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(t)
        a = a - eta * g
        a = (a + a[::-1]) * 0.5
        a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional jitter to get out of tiny plateaus
        if rng.random() < 0.02:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.015
                a[i] -= delta
                a[j] += delta
                a = (a + a[::-1]) * 0.5
                a = _project_simplex(a, S)

    return best_a, best_score

# Main entry point used by the judge
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence that minimises
    ``evaluate_sequence``.  The routine builds several simple initial
    shapes, refines each of them with the optimiser above, and then
    repeatedly upsamples the current best solution to a larger size.
    The best vector found before the time limit is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final external evaluation
    deadline = start_time + budget_s - 5.0

    max_n = 15000          # absolute upper bound for the length
    init_n = 1800          # size of the first round of searches

    # ------------------------------------------------------------------
    # Produce a collection of simple initial shapes
    # ------------------------------------------------------------------
    idx = np.arange(init_n, dtype=float)
    init_shapes = []

    # 1. uniform random
    init_shapes.append(rng.random(init_n))

    # 2. edge‑peaked shape  a_i = 1/(i+1) + 1/(n-i)
    edge_peaked = 1.0 / (idx + 1.0) + 1.0 / (init_n - idx)
    init_shapes.append(edge_peaked)

    # 3. exponential decay from the centre
    c = 0.006
    exp_decay = np.exp(-c * np.abs(idx - (init_n - 1) / 2.0))
    init_shapes.append(exp_decay)

    # 4–6. random power‑law shapes  a_i = (i+1)^p + (n-i)^p
    for _ in range(3):
        p = rng.uniform(-2.5, 2.5)
        pow_shape = (idx + 1) ** p + (init_n - idx) ** p
        init_shapes.append(pow_shape)

    best_seq = None
    best_score = np.inf
    attempts = 0

    # ------------------------------------------------------------------
    # Short optimisation of each initial shape (≈15 s each)
    # ------------------------------------------------------------------
    per_shape_time = min(15.0, (deadline - time.time()) / max(1, len(init_shapes)))
    for shape in init_shapes:
        if time.time() >= deadline:
            break
        a0 = np.maximum(shape, 0.0)
        S = np.sqrt(2.0 * init_n)
        if a0.sum() == 0.0:
            a0 = np.full(init_n, 1.0)
        a0 = a0 / a0.sum() * S
        slice_deadline = min(time.time() + per_shape_time, deadline)
        a_opt, sc_opt = _optimize(a0, slice_deadline, rng)
        attempts += 1
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[init] attempt={attempts} n={init_n} new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Progressive up‑sampling + refinement
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 5.0:
            break

        # decide next problem size
        if best_seq is None:
            n = rng.integers(1500, 2500)
            a0 = rng.random(n)
        else:
            curr_n = best_seq.shape[0]
            # grow while we still have time and we have not reached max_n
            if curr_n < max_n and remaining > 30.0:
                n = min(max_n, int(curr_n * 1.45))
                a0 = _upsample(best_seq, n)
            else:
                n = curr_n
                a0 = best_seq.copy()

        # tiny jitter before re‑optimisation
        jitter = rng.normal(scale=a0.mean() * 5e-4, size=n)
        a0 = a0 + jitter
        a0 = np.clip(a0, 0.0, None)

        # re‑scale to the proper ℓ1‑norm
        S = np.sqrt(2.0 * n)
        a0 = a0 / a0.sum() * S

        slice_deadline = min(time.time() + min(20.0, remaining * 0.35), deadline)
        a_opt, sc_opt = _optimize(a0, slice_deadline, rng)
        attempts += 1
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[up] attempt={attempts} n={n} new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check & return
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()

    external_score = evaluate_sequence(final_seq)
    print(f"[final] external={external_score:.6f} internal={best_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")
    return final_seq
