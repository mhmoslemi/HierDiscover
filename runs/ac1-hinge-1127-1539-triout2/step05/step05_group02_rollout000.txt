"""
search for a low‑score non‑negative step function.

The algorithm is a hybrid of three ideas:

1.  Multiple cheap initialisations (global warm‑start, power‑law,
    random).
2.  An LP‑based refinement (the “linear‑programming step” from the
    reference):  
    given the current f, maximise Σg under the linear constraints
    (f * g)(k) ≤ M where M = max(f*f).  The optimiser returns a new g
    which is then mixed with f (convex combination).  Repeating this
    reduces the maximal convolution.
3.  A short projected‑gradient descent (soft‑max) finish for fine
    tuning.

All work is bounded by the supplied ``budget_s`` seconds; the routine
periodically checks the deadline and returns the best sequence found.
"""

import time
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of appropriate size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a (computed via FFT)."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Compute the objective 2·n·max(a*a) / (sum(a)^2)
    (identical to ``evaluate_sequence`` for valid inputs).
    """
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (ℓ¹–norm = target)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Construction helpers
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Return (i+1)^(-p) normalised to have ℓ¹‑norm = target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1)×n convolution matrix A such that A @ g = f * g.
    The matrix contains constant diagonals: row‑index – col‑index = i
    has value f[i] for all admissible positions.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    diags = []
    offsets = []
    # each diagonal i (offset i) has length n-i and constant value f[i]
    for i, val in enumerate(f):
        length = n - i
        if length <= 0:
            break
        diags.append(np.full(length, val, dtype=np.float64))
        offsets.append(i)
    A = sp.diags(diags, offsets, shape=(rows, n), format='csr')
    return A


# ----------------------------------------------------------------------
# Gradient‑descent (soft‑max) – small, fast finetune
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(A_fft * A_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exp_term = np.exp(beta * (b - max_b))
    p = exp_term / exp_term.sum()

    # Convolution of reversed a with p (needed for gradient)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


def _gradient_descent_finetune(a0: np.ndarray, deadline: float):
    """Projected gradient descent on the soft‑max surrogate."""
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)

    best = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 1024.0
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)
        gn = np.linalg.norm(g)
        if gn > 0:
            # step inversely proportional to norm and temperature
            lr = 0.1 * target / (gn * (beta + 1.0))
            a -= lr * g

        a = _project_to_simplex(a, target)

        # occasional jitter to escape plateaus
        if it % 200 == 0:
            jitter = rng.normal(scale=target * 0.001 / (beta + 1.0), size=n)
            a = a + jitter
            a = _project_to_simplex(a, target)

        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.1f}] GD finetune iteration {it} β={beta:.1f} score={best_val:.6f}")

        # anneal temperature
        if it % 100 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1
    return best, best_val


# ----------------------------------------------------------------------
# LP‑based refinement
# ----------------------------------------------------------------------


def _lp_refine(seq: np.ndarray, deadline: float):
    """
    Repeatedly solve the LP (max Σg) subject to (seq * g) ≤ M,
    mix the result with the current sequence and continue.
    """
    n = seq.size
    target = np.sqrt(2.0 * n)
    seq = np.maximum(seq, 0.0)
    if seq.sum() == 0:
        seq = np.full(n, target / n)
    else:
        seq = seq * (target / seq.sum())

    best_seq = seq.copy()
    best_score = _score(seq)

    t = 0.3                     # initial mixing weight
    rng = np.random.default_rng()
    it = 0
    max_iter = 30               # safeguard against overrunning the time budget

    while time.time() < deadline and it < max_iter:
        # 1) current max convolution M
        conv = _conv_fft(seq)
        M = conv.max()

        # 2) build the convolution matrix A (sparse)
        A = _conv_matrix(seq)

        # 3) linear program: maximise Σg  ⇔ minimise -Σg
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(A.shape[0], M, dtype=np.float64)
        bounds = [(0, None)] * n
        res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})

        if not res.success:
            # LP failed – abort further LP refinements
            break

        g0 = res.x
        s0 = g0.sum()
        if s0 <= 0:
            break

        # 4) renormalise to the required ℓ¹ norm
        scale = target / s0
        g = g0 * scale

        # 5) convex combination
        new_seq = (1.0 - t) * seq + t * g
        new_seq = np.maximum(new_seq, 0.0)
        new_seq = new_seq * (target / new_seq.sum())

        new_score = _score(new_seq)
        if new_score < best_score:
            best_score = new_score
            best_seq = new_seq.copy()
            print(f"[{time.time():.1f}] LP iter {it} t={t:.3f} score={best_score:.6f}")

        # update for next iteration
        seq = new_seq
        it += 1
        t = max(t * 0.95, 0.01)          # slowly shrink the mixing weight

        # occasional random jitter to avoid stagnation
        if it % 5 == 0:
            jitter = rng.normal(scale=target * 0.0005, size=n)
            seq = np.maximum(seq + jitter, 0.0)
            seq = seq * (target / seq.sum())

    return best_seq, best_score


# ----------------------------------------------------------------------
# Full refinement pipeline for a single candidate sequence
# ----------------------------------------------------------------------


def _refine_candidate(init_seq: np.ndarray, deadline: float):
    """
    Apply LP refinement followed by a brief gradient‑descent finetune.
    Returns the best sequence found and its score.
    """
    # 1) LP‑based improvement
    seq, lp_score = _lp_refine(init_seq, deadline)

    # 2) Gradient‑descent finetune on the remaining time, if any
    remaining = deadline - time.time()
    if remaining > 2.0:
        seq, gd_score = _gradient_descent_finetune(seq, time.time() + remaining)
        final_score = min(lp_score, gd_score)
    else:
        final_score = lp_score

    return seq, final_score
