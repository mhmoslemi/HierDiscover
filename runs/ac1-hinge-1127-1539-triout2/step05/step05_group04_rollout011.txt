def _mass_transfer_sa(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    a = _project_to_simplex(a, target_sum)  # ensure on simplex.
    b = _max_convolution_fft(a)
    cur_max = float(b.max())
    best_a = a.copy()
    best_score = cur_max

    rng = np.random.default_rng()
    T = cur_max * 0.15 + 1e-9
    max_frac = 0.5

    it = 0
    while time.time() < deadline:
        # choose heavy
        i = int(np.argmax(a))
        # choose light candidates
        low_thresh = np.percentile(a, 20)
        low_candidates = np.where(a <= low_thresh)[0]
        if low_candidates.size == 0:
            j = int(np.argmin(a))
        else:
            j = int(rng.choice(low_candidates))
        if i == j:
            # pick another j
            j = int(rng.integers(0, n))
            if j == i:
                continue

        # choose delta
        max_possible = a[i]
        if max_possible <= 1e-12:
            # heavy mass zero, break
            break
        delta = max_possible * rng.random() * max_frac

        # compute new convolution
        b_tmp = b.copy()
        # diff
        b_tmp[i:i+n] -= 2.0 * delta * a
        b_tmp[j:j+n] += 2.0 * delta * a
        # compute new max
        new_max = float(b_tmp.max())
        dE = new_max - cur_max

        accept = dE <= 0.0 or (T > 0.0 and rng.random() < math.exp(-dE / T))
        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_tmp
            cur_max = new_max
            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()
                # print progress
                print(f"[{time.time():.1f}] SA improvement score={best_score:.6f}")
        # cooling
        T *= 0.99997
        # Reduce max_frac over time
        if it % 2000 == 0 and max_frac > 0.03:
            max_frac *= 0.97
        # periodic re-projection to fix numeric drift
        if it % 5000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _max_convolution_fft(a)
            cur_max = float(b.max())
        it += 1
    return best_a

"""
Step‑function optimizer – aggressive mass‑transfer simulated‑annealing.

The official score after the ℓ¹‑normalisation Σa = √(2 n) equals the
maximum entry of the self‑convolution a*a.  The optimiser works as follows:

1.  Several random starts are generated for a number of lengths.
    A short “soft‑max” projected‑gradient descent (smooth approximation of
    the ∞‑norm) quickly brings each start into a good basin.

2.  The core routine ``_mass_transfer_sa`` performs a custom simulated‑annealing
    that moves a small fraction of the “heavy” coefficient to a “light” one.
    The change in the convolution can be evaluated analytically with two
    O(n) slice updates, so millions of moves can be performed inside the
    1000 s budget.

3.  The best sequence found is returned as a plain list.  All numbers are
    non‑negative; the ℓ¹‑sum is forced to √(2 n) via a Euclidean projection
    onto the simplex.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1, dtype=float)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·n−1) using an FFT.
    Returned array is real‑valued.
    """
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Soft‑max gradient (log‑sum‑exp) – Phase 1 (quick basin‑finding)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) where
        g = ∇_a  ( 1/β·log Σ exp(β·b) )
          = 2·(a ⋆ p)          with   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = float(np.max(b))
    exps = np.exp(beta * (b - max_b))          # numerical stability
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1            # = 3 n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _smooth_max_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Annealed projected gradient descent on the smooth max.
    Uses momentum, an adaptive step and occasional jitter.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 4096.0
    beta_factor = 1.4

    step_base = target_sum * 0.15          # fraction of the ℓ¹‑norm
    momentum = np.zeros_like(a)
    rho = 0.90                               # momentum decay

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        # simplex projection
        a = _project_to_simplex(a, target_sum)

        # jitter every few hundred iterations
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal β
        if it % 350 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a


# ----------------------------------------------------------------------
# Random initialisation helpers
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative point on the simplex Σa = √(2 n).
    Several families (power‑law, exponential, Dirichlet, piecewise) are
    sampled to increase diversity.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                                 # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                               # pure exponential
        lam = rng.exponential(scale=0.02)
        a = np.exp(-lam * idx)

    elif family == 2:                               # power × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.02)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                               # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n))

    elif family == 4:                               # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                             np.exp(-lam2 * np.arange(b, n))])

    else:                                           # almost uniform with jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


# ----------------------------------------------------------------------
# Core mass‑transfer simulated annealing (Phase 2)
# ----------------------------------------------------------------------


def _mass_transfer_sa(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Fast greedy SA that repeatedly moves a small fraction of the heaviest
    coefficient to a light location.  The change in the convolution can be
    evaluated with two O(n) slice updates, so many moves fit into the budget.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    # tiny projection to be safe
    a = _project_to_simplex(a, target_sum)

    b = _max_convolution_fft(a)          # full convolution
    cur_max = float(b.max())
    best_a = a.copy()
    best_score = cur_max

    rng = np.random.default_rng()
    T = cur_max * 0.15 + 1e-12           # initial temperature
    max_frac = 0.5                       # maximal fraction of mass moved

    it = 0
    while time.time() < deadline:
        # ------------------------------------------------------------------
        # pick a heavy index i (largest coefficient)
        # ------------------------------------------------------------------
        i = int(np.argmax(a))

        # ------------------------------------------------------------------
        # pick a light index j (among the 20 % smallest coefficients)
        # ------------------------------------------------------------------
        low_thresh = np.percentile(a, 20)
        low_candidates = np.where(a <= low_thresh)[0]
        if low_candidates.size == 0:
            j = int(np.argmin(a))
        else:
            j = int(rng.choice(low_candidates))

        if i == j:
            # fall back to a completely random partner
            j = int(rng.integers(0, n))
            if j == i:
                it += 1
                continue

        # ------------------------------------------------------------------
        # amount of mass to transfer
        # ------------------------------------------------------------------
        max_possible = a[i]
        if max_possible <= 1e-15:
            # nothing left to move – give up on this i
            it += 1
            continue
        delta = max_possible * rng.random() * max_frac

        # ------------------------------------------------------------------
        # evaluate the move using slice updates (linearised change)
        # ------------------------------------------------------------------
        b_tmp = b.copy()
        # contribution of index i (removed)
        b_tmp[i:i + n] -= 2.0 * delta * a
        # contribution of index j (added)
        b_tmp[j:j + n] += 2.0 * delta * a

        new_max = float(b_tmp.max())
        dE = new_max - cur_max

        # ------------------------------------------------------------------
        # Metropolis acceptance
        # ------------------------------------------------------------------
        accept = dE <= 0.0 or rng.random() < math.exp(-dE / T)

        if accept:
            # commit the move
            a[i] -= delta
            a[j] += delta
            b = b_tmp
            cur_max = new_max

            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA improvement → {best_score:.6f} (n={n})")

        # ------------------------------------------------------------------
        # schedule updates
        # ------------------------------------------------------------------
        T *= 0.99997                     # exponential cooling
        if it % 2000 == 0 and max_frac > 0.03:
            max_frac *= 0.97            # slowly tighten step size

        # occasional re‑projection to keep the ℓ¹‑norm exact and numerically stable
        if it % 5000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _max_convolution_fft(a)
            cur_max = float(b.max())

        it += 1

    # final projection for safety
    best_a = _project_to_simplex(best_a, target_sum)
    return best_a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` metric.  The routine respects the given
    wall‑clock budget, explores several lengths, applies a short
    smooth‑max gradient descent for basin‑finding and then a massive
    mass‑transfer simulated annealing.  The best sequence found is returned
    as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Seed from a previously discovered construction if it exists
    # ------------------------------------------------------------------
    earlier = globals().get("height_sequence_1")
    if isinstance(earlier, (list, tuple, np.ndarray)):
        a0 = np.asarray(earlier, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        sc0 = float(_max_convolution_fft(a0).max())
        best_seq = a0.copy()
        best_score = sc0
        print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) score={sc0:.6f}")

    # ------------------------------------------------------------------
    # Lengths to investigate – larger n gives more freedom but is slower
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000, 10000]

    # iterate over lengths, allocating a slice of the remaining budget to each
    for n in lengths:
        if time.time() >= deadline:
            break

        # how much time do we have left?
        time_left = deadline - time.time()
        if time_left < 15.0:      # not enough time for a full run on this length
            break

        # allocate roughly 30 % of the remaining budget to this length,
        # but guarantee a minimum of 20 s.
        allocated = max(20.0, 0.30 * time_left)

        # ------------------------------------------------------------------
        # Phase A – quick random restarts + soft‑max gradient descent
        # ------------------------------------------------------------------
        phase_a_dead = time.time() + min(allocated * 0.15, 8.0)   # ≤ 8 s
        while time.time() < phase_a_dead:
            a_start = _random_initial_sequence(n, rng)

            # short gradient run (≈2 s)
            sub_dead = min(deadline, time.time() + 2.5)
            a_gd = _smooth_max_descent(a_start, sub_dead)

            cur_score = float(_max_convolution_fft(a_gd).max())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a_gd.copy()
                print(f"[{time.time():.1f}] New global best (rand) n={n} score={cur_score:.6f}")

        # ------------------------------------------------------------------
        # Phase B – intensive mass‑transfer SA on the current best of this n
        # ------------------------------------------------------------------
        if best_seq is not None and best_seq.size == n:
            # give the remainder of the allocated budget to SA
            phase_b_dead = min(deadline, time.time() + allocated * 0.85)
            a_ref = _mass_transfer_sa(best_seq, phase_b_dead)

            ref_score = float(_max_convolution_fft(a_ref).max())
            if ref_score < best_score:
                best_score = ref_score
                best_seq = a_ref.copy()
                print(f"[{time.time():.1f}] SA improvement n={n} score={ref_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing on whatever time is left (if any)
    # ------------------------------------------------------------------
    if best_seq is not None:
        left = deadline - time.time()
        if left > 5.0:
            final_seq = _mass_transfer_sa(best_seq, time.time() + left)
            final_score = float(_max_convolution_fft(final_seq).max())
            if final_score < best_score:
                best_score = final_score
                best_seq = final_seq.copy()
                print(f"[{time.time():.1f}] Final SA polishing score={final_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up trailing zeros and report the official evaluation
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform distribution of length 1000 (should never happen)
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:   # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
