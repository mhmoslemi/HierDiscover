def subgrad_opt(a, target_sum, deadline):
    best_a = a.copy()
    best_val = _score(a)
    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(np.argmax(conv))
        bmax = conv[k_max]
        # gradient
        lo = max(0, k_max - (len(a) - 1))
        hi = min(k_max, len(a) - 1)
        idx = np.arange(lo, hi+1)
        j = k_max - idx
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[j]
        if k_max % 2 == 0:
            mid = k_max // 2
            if lo <= mid <= hi:
                g[mid] = a[mid]   # correct self term
        # step size
        norm_sq = np.dot(g, g) + 1e-12
        step = 0.5 * bmax / norm_sq
        # update
        a_new = a - step * g
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_to_simplex(a_new, target_sum)
        val = _score(a_new)
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
            # print improvement

        # occasional jitter

            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
        it += 1
    return best_a, best_val

""" 
Algorithm: generate diverse initial step‑function candidates (power‑law,
U‑shaped, exponential, random), pick the best by a fast max‑convolution
score, then improve it with a custom projected sub‑gradient descent that
directly targets the current maximal convolution entry, followed by a
lightweight simulated‑annealing refinement.  All operations respect the
ℓ¹‑simplex ∑ a_i = √(2 n) and non‑negativity, matching the
evaluate_sequence security checks.
"""

import time
import numpy as np

def _max_convolution_fft(a: np.ndarray) -> float:
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())

def _conv_fft(a: np.ndarray, b: np.ndarray = None) -> np.ndarray:
    """Full linear convolution using FFT. If b is None, returns a*a."""
    if b is None:
        b = a
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: 2 * n - 1]
    return conv

def _score(a: np.ndarray) -> float:
    """Objective value for a sequence already normalised to the required ℓ¹‑simplex."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))

def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x_i ≥ 0, Σ x_i = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a

def _u_shaped_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    prod = i * (n - i + 1)
    a = prod ** (-p)
    a *= target_sum / a.sum()
    return a

def _exp_sequence(n: int, alpha: float, target_sum: float) -> np.ndarray:
    a = np.exp(-alpha * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a

def _subgradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Projected sub‑gradient descent that directly minimises the current maximal convolution."""
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)          # full convolution a*a
        k_max = int(np.argmax(conv))
        bmax = conv[k_max]

        # gradient of the active entry (sub‑gradient)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        rev_idx = k_max - idx
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[rev_idx]
        if k_max % 2 == 0:
            mid = k_max // 2
            if lo <= mid <= hi:
                g[mid] = a[mid]          # correct the double count on the diagonal

        # step size – classic diminishing rule with a small random factor
        norm_sq = np.dot(g, g) + 1e-12
        step = 0.6 * bmax / norm_sq
        step *= rng.uniform(0.5, 1.5)

        a_new = a - step * g
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_to_simplex(a_new, target_sum)

        val = _score(a_new)
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
            print(f"[{time.time():.1f}] subgrad iter {it} → score={best_val:.6f}")

        # occasional jitter to escape plateaus
        if it % 800 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a_new = a_new + jitter
            a_new = np.maximum(a_new, 0.0)
            a_new = _project_to_simplex(a_new, target_sum)

        a = a_new
        it += 1
    return best_a, best_val

def _simulated_annealing(a0: np.ndarray, deadline: float,
                         start_temp: float = 0.4,
                         end_temp: float = 1e-6,
                         max_frac_start: float = 0.35,
                         max_frac_end: float = 0.007,
                         max_window: int = 30) -> tuple[np.ndarray, float]:
    """A thin SA loop; moves are mass‑transfer, short‑window smoothing and swaps."""
    a = a0.copy()
    n = a.size
    cur_score = _score(a)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)
    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # identify a sensitive index (part of the current maximum entry)
        k_max = int(np.argmax(conv))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx_active = np.arange(lo, hi + 1, dtype=np.int64)

        # weights proportional to contribution a_i * a_{k_max-i}
        contrib = a[idx_active] * a[k_max - idx_active]
        weight = contrib
        if weight.sum() == 0:
            weight = a[idx_active] + 1e-12

        move_type = None
        if rng.random() < 0.70:                      # mass transfer
            donor = rng.choice(idx_active, p=weight / weight.sum())
            receiver = rng.integers(0, n)
            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        elif rng.random() < 0.90:                    # smoothing
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                old_seg = a[start:end].copy()
                avg = old_seg.mean()
                a[start:end] = avg
                move_type = "smooth"
                saved_seg = old_seg
                saved_start, saved_end = start, end

        else:                                         # swap
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        conv_new = _conv_fft(a)
        new_score = _score(a)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA new best = {best_score:.6f}")
        else:
            # revert
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter
        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = a * (target_sum / a.sum())
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1
    return best_a, best_score

    best_score = float('inf')

        seq = seq / seq.sum() * target

    # --------------------------------------------------------------

        if val < best_score:

    # 3.  Simulated‑annealing fine‑tuning
    # --------------------------------------------------------------

        if val < best_score:

            print(f"[{time.time():.1f}] after SA → score={val:.6f}")

    # final trimming of trailing zeros (insignificant entries)
    if best_seq is None:
        return []

    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    print(f"[{time.time():.1f}] final best score={best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()

"""
Optimise a non‑negative step‑function (height sequence) for the benchmark
score defined by ``evaluate_sequence``.

The workflow is:

1. **Exploratory families** – generate a few parametric candidates
   (power‑law, U‑shaped product‑inverse, exponential decay and random
   Dirichlet seeds).  All candidates are projected onto the ℓ¹‑simplex
   Σa = √(2·n) – this is exactly the normalisation used by
   ``evaluate_sequence`` – and the best one is kept.

2. **Projected sub‑gradient descent** – repeatedly
   * compute the full convolution a*a,
   * locate the index of the current maximal entry,
   * take a sub‑gradient of that entry (which is simply a shifted copy of
     ``a``), and
   * move a in the opposite direction with a diminishing step size.
   After every step the vector is projected back onto the simplex.

   Small Gaussian jitter is added periodically to escape plateaus.

3. **Simulated‑annealing fine‑tuning** – a lightweight SA stage
   (mass transfer, local smoothing and swaps) with an exponential
   temperature schedule further polishes the solution.

All steps respect the non‑negativity and ℓ¹‑simplex constraints, thus the
returned list can be fed directly to ``evaluate_sequence``.  The routine
runs until the provided time budget (default 1000 s) expires and returns
the best sequence discovered.
"""

import time
import numpy as np
from scipy.optimize import linprog  # optional, not used in the final version


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray, b: np.ndarray = None) -> np.ndarray:
    """Full linear convolution of a with b (or a with itself if b is None)."""
    if b is None:
        b = a
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Fast evaluation of the benchmark (max‑convolution) for a normalised sum."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x_i ≥ 0, Σx_i = s}.
    Implements Duchi et al. (2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i = (i+1)^(-p)  (i = 0 … n‑1) normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _u_shaped_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i = [(i+1)*(n‑i)]^(-p)  (U‑shaped) normalised to target_sum."""
    i = np.arange(1, n + 1, dtype=np.float64)
    prod = i * (n - i + 1)
    a = prod ** (-p)
    a *= target_sum / a.sum()
    return a


def _exp_sequence(n: int, alpha: float, target_sum: float) -> np.ndarray:
    """a_i = exp(-alpha·i)  (i = 0 … n‑1) normalised to target_sum."""
    a = np.exp(-alpha * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _subgradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Projected sub‑gradient descent that directly minimises the current maximal convolution."""
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)                     # a*a
        k_max = int(np.argmax(conv))
        bmax = conv[k_max]

        # sub‑gradient of the active entry
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        rev_idx = k_max - idx
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[rev_idx]
        if k_max % 2 == 0:                      # correct the double count on the diagonal
            mid = k_max // 2
            if lo <= mid <= hi:
                g[mid] = a[mid]

        norm_sq = np.dot(g, g) + 1e-12
        step = 0.6 * bmax / norm_sq
        step *= rng.uniform(0.5, 1.5)            # modest randomness

        a_new = a - step * g
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_to_simplex(a_new, target_sum)

        val = _score(a_new)
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
            print(f"[{time.time():.1f}] sub‑grad iter {it} → score={best_val:.6f}")

        # occasional jitter to escape flat regions
        if it % 800 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a_new = a_new + jitter
            a_new = np.maximum(a_new, 0.0)
            a_new = _project_to_simplex(a_new, target_sum)

        a = a_new
        it += 1

    return best_a, best_val


def _simulated_annealing(a0: np.ndarray, deadline: float,
                         start_temp: float = 0.4,
                         end_temp: float = 1e-6,
                         max_frac_start: float = 0.35,
                         max_frac_end: float = 0.007,
                         max_window: int = 30) -> tuple[np.ndarray, float]:
    """Lightweight simulated‑annealing refinement."""
    a = a0.copy()
    n = a.size
    cur_score = _score(a)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # active indices for the current maximal convolution entry
        k_max = int(np.argmax(conv))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx_active = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution‑based weights (a_i * a_{k‑max‑i})
        contrib = a[idx_active] * a[k_max - idx_active]
        weight = contrib
        if weight.sum() == 0:
            weight = a[idx_active] + 1e-12

        move_type = None
        # --------------------------------------------------------------
        # 70 % mass‑transfer moves
        # --------------------------------------------------------------
        if rng.random() < 0.70:
            donor = rng.choice(idx_active, p=weight / weight.sum())
            receiver = rng.integers(0, n)
            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        # --------------------------------------------------------------
        # 20 % local smoothing moves
        # --------------------------------------------------------------
        elif rng.random() < 0.90:
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                seg = a[start:end].copy()
                avg = seg.mean()
                a[start:end] = avg
                move_type = "smooth"
                saved_seg, saved_start, saved_end = seg, start, end

        # --------------------------------------------------------------
        # 10 % random swaps
        # --------------------------------------------------------------
        else:
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        conv_new = _conv_fft(a)
        new_score = _score(a)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA new best = {best_score:.6f}")
        else:
            # revert the move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter to shake out deep minima
        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = a * (target_sum / a.sum())
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


def propose_candidate(budget_s: int = 1000, seed: int = 123) -> list[float]:
    """
    Search for a non‑negative step function with a low benchmark score.
    The routine builds several parametric families, picks the best
    seed, refines it with projected sub‑gradient descent and finally
    runs a short simulated‑annealing phase.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem size – 1500 gives a good trade‑off between richness and speed
    # --------------------------------------------------------------
    n = 1500
    target = np.sqrt(2.0 * n)

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 1)  Scan a few parametric families
    # --------------------------------------------------------------
    # Power‑law family
    for p in np.linspace(0.20, 1.00, 17):
        seq = _power_law_sequence(n, p, target)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()
            print(f"[{time.time():.1f}] power‑law p={p:.3f} → score={val:.6f}")

    # U‑shaped (product‑inverse) family
    for p in np.linspace(0.10, 0.80, 15):
        seq = _u_shaped_sequence(n, p, target)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()
            print(f"[{time.time():.1f}] u‑shape p={p:.3f} → score={val:.6f}")

    # Exponential decay family
    for alpha in np.linspace(5e-4, 5e-3, 20):
        seq = _exp_sequence(n, alpha, target)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()
            print(f"[{time.time():.1f}] exp α={alpha:.5f} → score={val:.6f}")

    # Random Dirichlet seeds
    for _ in range(12):
        seq = rng.random(n)
        seq = seq / seq.sum() * target
        seq = _project_to_simplex(seq, target)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()
            print(f"[{time.time():.1f}] random seed → score={val:.6f}")

    # Optional warm start from the globally supplied solution, if present
    if "height_sequence_1" in globals():
        seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if seq.size > n:
            seq = seq[:n]
        else:
            seq = np.pad(seq, (0, n - seq.size), "constant")
        seq = _project_to_simplex(seq, target)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()
            print(f"[{time.time():.1f}] global warm start → score={val:.6f}")

    # --------------------------------------------------------------
    # 2) Projected sub‑gradient descent
    # --------------------------------------------------------------
    if time.time() < deadline:
        seq, val = _subgradient_descent(best_seq.copy(), target, deadline)
        if val < best_score:
            best_score, best_seq = val, seq.copy()
            print(f"[{time.time():.1f}] after sub‑gradient → score={val:.6f}")

    # --------------------------------------------------------------
    # 3) Simulated‑annealing fine‑tuning
    # --------------------------------------------------------------
    if time.time() < deadline:
        seq, val = _simulated_annealing(best_seq.copy(), deadline)
        if val < best_score:
            best_score, best_seq = val, seq.copy()
            print(f"[{time.time():.1f}] after SA → score={val:.6f}")

    # --------------------------------------------------------------
    # Final housekeeping – strip trailing near‑zero entries
    # --------------------------------------------------------------
    if best_seq is None:
        return []

    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    print(f"[{time.time():.1f}] final best score={best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
