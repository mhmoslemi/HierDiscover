def subgradient_descent_max(a, deadline, step0=0.1):
    t=0
    n = len(a)
    while time.time() < deadline:
        b = max_convolution_fft(a)
        # Compute gradient at max
        # Find first index of max, maybe random among ties.
        conv_full = _full_convolution_fft(a) # returns length 2n-1
        M = conv_full.max()
        k = np.argmax(conv_full)
        # gradient: g_i = 2*a[k-i] if 0 <= k-i < n else 0
        # due to reversed indices
        # Implementation: shift a with padding.
        g = np.zeros_like(a)
        lo = max(0, k-(n-1))
        hi = min(n-1, k)
        # For i in [lo,hi], compute j = k - i
        # g[i] = 2*a[j]
        i_range = np.arange(lo, hi+1)
        j_range = k - i_range
        g[i_range] = 2 * a[j_range]
        eta = step0 / np.sqrt(t+1)
        a = a - eta * g
        a = _project_to_simplex(a, 1.0)
        t += 1
        # optional early break if M is less than best known threshold.
    return a
```

def _gradient_softmax(a,beta):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    b = np.fft.irfft(A*A, n=size)[:2*n-1]
    max_b = b.max()
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)
    # gradient as 2*(a_rev * p)
    a_rev = a[::-1]
    conv_len = n + (2*n - 1) - 1
    size2 = 1 << (conv_len.bit_length())
    Ar = np.fft.rfft(a_rev, n=size2)
    Pr = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar * Pr, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n-1:n-1+n]
    return g, b

def _subgradient_descent_max(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    target_sum = 1.0
    step0 = 0.5   # initial step size
    iteration = 0
    while time.time() < deadline:
        # compute convolution full to find max index
        size = 1 << ((2*n - 1).bit_length())
        Fa = np.fft.rfft(a, n=size)
        b_full = np.fft.irfft(Fa * Fa, n=size)[:2*n-1]
        k = int(np.argmax(b_full))
        # gradient: g_i = 2 * a_{k-i} for valid i
        g = np.zeros_like(a)
        # compute bounds
        lo = max(0, k-(n-1))
        hi = min(n-1, k)
        i_range = np.arange(lo, hi+1, dtype=int)
        j_range = k - i_range
        g[i_range] = 2.0 * a[j_range]
        # step size schedule
        iteration += 1
        eta = step0 / np.sqrt(iteration)
        a = a - eta * g
        # project onto simplex
        a = _project_to_simplex(a, target_sum)
        # optional early break if max not decreasing
        # Could check improved objective less than best.
    return a

"""
Propose a non‑negative step function (sequence) that minimises
``evaluate_sequence``.  The algorithm proceeds in three stages:

1.  **Broad candidate generation** – deterministic families
    (power‑law, symmetric, and an asymmetric two‑exponent family) and
    random simplex samples.  The best few (elite) are kept.

2.  **Refinement** – each elite is first improved by a smoothed
    “log‑sum‑exp” surrogate gradient descent (projected onto the
    ℓ¹‑simplex).  Afterwards a pure sub‑gradient descent on the true
    max‑convolution is performed, followed by a short simulated‑annealing
    mass‑transfer phase.

3.  **Final greedy hill‑climb** – a greedy local search that moves
    small fractions of mass between indices while preserving the sum.

All steps respect the supplied time budget (default 1000 s).  The
returned list is a plain Python list of floats; its ℓ¹‑norm is 1.0,
which is scale‑invariant for the evaluation function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex Σx = s, x≥0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection onto {x_i ≥ 0, Σx_i = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) surrogate of max(b) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b) where
        b = a*a (full convolution)
        g = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a ⋆ p) ,  p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]

    # ----- convolution a*a -----
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # ----- soft‑max probabilities -----
    max_b = np.max(b)
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)

    # ----- gradient g = 2·(a ⋆ p) -----
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Projected gradient descent on the smoothed maximum
# ----------------------------------------------------------------------


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Annealed projected gradient descent on the log‑sum‑exp smooth max.
    The ℓ¹‑norm is kept equal to 1.0.
    """
    a = a0.copy()
    n = a.size
    target_sum = 1.0

    beta = 1.0
    max_beta = 1e7
    momentum = np.zeros_like(a)
    rho = 0.9                     # momentum decay
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum step
        momentum = rho * momentum + (1.0 - rho) * g
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            lr = 0.5 * target_sum / (norm * (beta + 1.0))
            a -= lr * momentum

        # simplex projection
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 8e-4 / (beta + 1.0),
                               size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        # exponential annealing of β
        if it % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.06, max_beta)

        it += 1
    return a


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simple sub‑gradient descent on  M(a)=max_k (a*a)_k .
    Step size η_t = η₀ / sqrt(t).  Projection onto the ℓ¹‑simplex (sum=1).
    """
    a = a0.copy()
    n = a.size
    target_sum = 1.0
    eta0 = 0.5
    t = 0

    while time.time() < deadline:
        # compute full convolution and the location of the current maximum
        size = 1 << ((2 * n - 1).bit_length())
        A = np.fft.rfft(a, n=size)
        b_full = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
        k = int(np.argmax(b_full))

        # gradient g_i = 2 * a_{k-i} for valid indices
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i_range = np.arange(lo, hi + 1, dtype=int)
        j_range = k - i_range
        g[i_range] = 2.0 * a[j_range]

        t += 1
        eta = eta0 / np.sqrt(t)
        a -= eta * g
        a = _project_to_simplex(a, target_sum)

    return a


# ----------------------------------------------------------------------
# Simulated annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Random mass‑transfer SA with multiplicative cooling."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # temperature initialised from current objective
    T = best_score * 0.02 + 1e-8
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        # move up to 15 % of mass from i to j
        frac = rng.random() * 0.15
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        if cur_score <= best_score or (T > 0.0 and rng.random() < np.exp(-dE / T)):
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
    return best


# ----------------------------------------------------------------------
# Greedy local hill‑climb (accept only improvements)
# ----------------------------------------------------------------------


def _local_hill_climb(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Greedy mass‑transfer hill climbing; never accepts worsening moves."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] < 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        # move a small fraction (≤5 %) of mass from i to j
        frac = rng.random() * 0.05
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
        else:
            # revert
            a[j] -= delta
            a[i] += delta

        # occasional global jitter to escape tiny plateaus
        if rng.random() < 0.001:
            jitter = rng.normal(scale=0.001, size=n)
            a += jitter
            a = _project_to_simplex(a, 1.0)
    return best


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    """a_i = (i+shift)^{-p}, projected onto ℓ¹‑simplex (sum=1)."""
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    raw = np.maximum(raw, 0.0)
    a = raw / raw.sum()
    return a


def _symmetric_powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    """Symmetric version: a_i = (i+shift)^{-p} + (n-1-i+shift)^{-p}."""
    idx = np.arange(n, dtype=float)
    left = (idx + shift) ** (-p)
    right = (n - 1 - idx + shift) ** (-p)
    raw = left + right
    raw = np.maximum(raw, 0.0)
    a = raw / raw.sum()
    return a


def _asymmetric_powerlaw_sequence(n: int, p_left: float,
                                  p_right: float, shift: float) -> np.ndarray:
    """
    Two‑exponent power‑law: left half uses p_left, right half uses p_right.
    The vector is projected onto the ℓ¹‑simplex (sum=1).
    """
    idx = np.arange(n, dtype=float)
    mid = (n - 1) / 2.0
    left_mask = idx <= mid
    raw = np.empty_like(idx)
    raw[left_mask] = (idx[left_mask] + shift) ** (-p_left)
    raw[~left_mask] = ((n - 1 - idx[~left_mask]) + shift) ** (-p_right)
    raw = np.maximum(raw, 0.0)
    a = raw / raw.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the allotted time budget, generates a diverse pool of
    candidates, refines a handful of elites, and finally returns the best sequence.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    deadline = time.time() + budget_s - 0.5  # safety margin

    best_seq = None
    best_obj = np.inf

    # ------------------------------------------------------------------
    # 0. Seed with any known construction (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        hs1 = globals().get("height_sequence_1")
        if isinstance(hs1, list) and len(hs1) > 0:
            a = np.array(hs1, dtype=float)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            best_seq = a.copy()
            best_obj = 2.0 * a.size * _max_convolution_fft(a)
            print(f"[{time.time():.1f}] Seeded with height_sequence_1, obj={best_obj:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Broad deterministic / random candidate sweep
    # ------------------------------------------------------------------
    candidates = []          # list of np.ndarray (already normalised to sum=1)
    elite = []               # list of (obj, array), kept sorted

    # basic uniform baseline
    n_uniform = 5000
    candidates.append(np.full(n_uniform, 1.0 / n_uniform))

    # parametric families
    lengths = [2500, 3000, 3500, 4000, 4500, 5000]
    p_grid = np.linspace(0.3, 0.9, 13)          # exponent grid
    shift_grid = np.linspace(0.2, 5.0, 12)      # shift grid

    for n in lengths:
        if time.time() >= deadline:
            break
        for p in p_grid:
            for shift in shift_grid:
                # plain power‑law
                a = _powerlaw_sequence(n, p, shift)
                candidates.append(a)
                # symmetric power‑law
                a2 = _symmetric_powerlaw_sequence(n, p, shift)
                candidates.append(a2)
                # asymmetric (two‑exponent) – limited sweep to keep time reasonable
                for p_r in (p, p + 0.1, max(p - 0.1, 0.2)):
                    a3 = _asymmetric_powerlaw_sequence(n, p, p_r, shift)
                    candidates.append(a3)

        # quick random sampling for the current length
        rnd_end = min(deadline, time.time() + 2.0)
        while time.time() < rnd_end:
            a_rand = rng.random(n)
            a_rand = a_rand / a_rand.sum()
            candidates.append(a_rand)

    # ------------------------------------------------------------------
    # 2. Evaluate candidates and keep elite list (size ≤ 10)
    # ------------------------------------------------------------------
    for a in candidates:
        if time.time() >= deadline:
            break
        n = a.size
        obj = 2.0 * n * _max_convolution_fft(a)   # sum is 1 → objective = 2 n M
        if obj < best_obj:
            best_obj = obj
            best_seq = a.copy()
            print(f"[{time.time():.1f}] New global best from sweep, n={n}, obj={best_obj:.6f}")

        # keep top 10 elites
        if len(elite) < 10 or obj < elite[-1][0]:
            elite.append((obj, a))
            elite.sort(key=lambda x: x[0])
            elite = elite[:10]

    # ------------------------------------------------------------------
    # 3. Refine each elite candidate
    # ------------------------------------------------------------------
    for base_obj, a0 in elite:
        if time.time() >= deadline:
            break
        n = a0.size
        # gradient descent on smoothed surrogate (≈30 s per candidate)
        gd_dead = min(deadline, time.time() + 30.0)
        a_gd = _gradient_descent_opt(a0, gd_dead)

        # sub‑gradient descent on true max (≈5 s)
        sg_dead = min(deadline, time.time() + 5.0)
        a_sg = _subgradient_descent_max(a_gd, sg_dead)

        # simulated‑annealing refinement using remaining time
        a_sa = _simulated_annealing(a_sg, deadline)

        obj = 2.0 * a_sa.size * _max_convolution_fft(a_sa)
        if obj < best_obj:
            best_obj = obj
            best_seq = a_sa.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL BEST after refinement, n={n}, obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 4. Final greedy hill‑climb (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a_final = _local_hill_climb(best_seq, deadline)
        obj = 2.0 * a_final.size * _max_convolution_fft(a_final)
        if obj < best_obj:
            best_obj = obj
            best_seq = a_final
            print(f"[{time.time():.1f}] FINAL hill‑climb best obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 5. Return result
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform distribution of length 2000
        n_fallback = 2000
        best_seq = np.full(n_fallback, 1.0 / n_fallback)

    # Trim trailing near‑zero entries for neatness
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Optional sanity check with the provided evaluator (does not affect the returned value)
    try:
        eval_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence = {eval_val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective = {best_obj:.6f}, length = {best_seq.size}")
    return best_seq.tolist()
