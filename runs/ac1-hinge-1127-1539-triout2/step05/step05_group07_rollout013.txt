def _project_simplex(v, s):
    # s positive
    n = len(v)
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    # find rho = max {i | u_i - cssv_i/(i+1) > 0}
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def _adam_subgrad_optimize(a0, deadline, rng):
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    # init logits from a0
    soft = np.maximum(a0 / S, 1e-12)
    soft = soft / soft.sum()
    p = np.log(soft)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1=0.9
    beta2=0.999
    eps_adam=1e-8
    lr = 0.05   # initial learning rate in logit space

    best_a = a0.copy()
    # compute initial peak
    b0 = _full_convolution_fft(a0)
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1
        # compute a from logits
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # convolution
        b = _full_convolution_fft(a)
        peak = b.max()

        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # compute mask for active indices
        # use tolerance relative to peak
        eps = max(peak * 1e-6, 1e-8)
        mask = (b >= peak - eps).astype(float)

        # gradient wrt a: g = 2 * conv(mask, a_rev)
        n = a.shape[0]
        a_rev = a[::-1]
        conv_len = len(mask) + n - 1
        f_len = _next_pow2(conv_len)
        Fmask = np.fft.rfft(mask, n=f_len)
        Frev = np.fft.rfft(a_rev, n=f_len)
        conv = np.fft.irfft(Fmask * Frev, n=f_len)[:conv_len]
        g = 2.0 * conv[n - 1:n - 1 + n]

        # map to gradient wrt logits
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_p
        v = beta2 * v + (1 - beta2) * (grad_p ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # schedule: decay lr
        lr *= 0.9999

        # jitter occasionally
        if it % 400 == 0:
            p += rng.normal(scale=0.01, size=n)

    return best_a, best_peak

"""
Sub‑gradient Adam optimiser for the step‑function (self‑convolution) problem.

The bound to be minimised equals the peak of the full convolution `a*a`
when the ℓ₁‑norm of the non‑negative vector `a` is normalised to
`S = sqrt(2·n)`.  The code works on the simplex by representing
`a = softmax(p) * S`.  In each iteration a hard sub‑gradient of the
∞‑norm is used: all indices whose convolution value is within a tiny
tolerance of the current maximum are collected into a binary mask,
the gradient of the corresponding linear constraints is obtained via a
single FFT‑based convolution, and an Adam update is performed on the
log‑softmax parameters `p`.  A small amount of Gaussian jitter is added
occasionally to escape plateaus.

Multiple restarts with different lengths and initial shapes (uniform,
edge‑heavy, Dirichlet‑sampled, exponential, linear, sinusoidal) are
performed until the time budget expires.  If a previous best construction
(`height_sequence_1`) exists it is used as a seed with a certain
probability.  The best sequence seen is stored globally and returned.

The routine respects the 1000 s time limit, prints progress, and finally
calls the provided ``evaluate_sequence`` to verify the external score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear (non‑circular) self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.ceil(pos).astype(int)
    hi = np.clip(hi, 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Projection onto the probability simplex (sum = S, non‑negative)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x) = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser that uses a hard sub‑gradient of the ∞‑norm.
# ----------------------------------------------------------------------
def _adam_subgrad_optimize(a0: np.ndarray,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Minimise the peak of the convolution using Adam on the logits.
    Returns the best vector encountered and its hard convolution peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    # initialise logits from a0 (softmax(p) * S == a0)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.07                     # step‑size in log‑space

    # keep the best seen point
    b0 = _full_convolution_fft(a0)
    best_a = a0.copy()
    best_peak = b0.max()

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- reconstruct a from logits -----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ----- convolution and peak -----
        b = _full_convolution_fft(a)
        peak = b.max()

        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ----- build mask of active (near‑max) constraints -----
        # tolerance is absolute; 1e‑6 works well for the scale of b (≈1‑2)
        tol = max(peak * 1e-6, 1e-8)
        mask = (b >= peak - tol).astype(float)     # shape (2n‑1,)

        # ----- gradient wrt a for the active constraints -----
        a_rev = a[::-1]                     # reversed a
        conv_len = len(mask) + n - 1
        f_len = _next_pow2(conv_len)
        Fmask = np.fft.rfft(mask, n=f_len)
        Frev = np.fft.rfft(a_rev, n=f_len)
        conv = np.fft.irfft(Fmask * Frev, n=f_len)[:conv_len]
        # gradient component (2 * correlation)
        g = 2.0 * conv[n - 1:n - 1 + n]

        # ----- map to gradient wrt logits (softmax parametrisation) -----
        mu = np.dot(a, g) / S               # scalar Lagrange multiplier for sum‑constraint
        grad_p = a * (g - mu)               # shape (n,)

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- schedules -----
        lr *= 0.9999                        # slow decay of learning‑rate
        if it % 400 == 0:
            # occasional jitter to escape flat regions
            p += rng.normal(scale=0.01, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Initial vector generator (no closures – all arguments explicit)
# ----------------------------------------------------------------------
def _initial_vector(n: int,
                    rng: np.random.Generator) -> np.ndarray:
    """
    Produce a feasible start vector of length n with ℓ₁‑norm sqrt(2·n).
    Several families are sampled, optionally blended with a stored
    construction if one is available.
    """
    S = np.sqrt(2.0 * n)

    # decide which shape to use
    shape = rng.choice(['uniform', 'edge', 'random', 'exp', 'linear', 'sin'])
    if shape == 'uniform':
        a = np.full(n, S / n)
    elif shape == 'edge':
        # edge‑heavy power‑law
        p_exp = rng.uniform(0.6, 2.2)
        idx = np.arange(n)
        d = np.minimum(idx, n - 1 - idx) + 1          # distance to nearest edge, 1‑based
        a = d.astype(float) ** (-p_exp)
        a = a / a.sum() * S
    elif shape == 'random':
        a = rng.random(n)
        a = a / a.sum() * S
    elif shape == 'exp':
        lam = rng.uniform(0.001, 0.015)
        a = np.exp(-lam * np.arange(n))
        a = a / a.sum() * S
    elif shape == 'linear':
        a = np.linspace(1.0, 0.01, n)
        a = a / a.sum() * S
    else:  # 'sin'
        a = np.sin(np.pi * np.arange(n) / (n - 1)) + 0.05
        a = a / a.sum() * S

    # occasionally mix with the previously stored construction
    base = globals().get('height_sequence_1')
    if base is not None and rng.random() < 0.2:
        try:
            base_arr = np.asarray(base, dtype=float)
            a = _scale_sequence(base_arr, n)
            a = np.clip(a, 0.0, None)
            a = a / a.sum() * S
        except Exception:
            pass

    return a

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The function runs many short Adam‑subgradient
    restarts, keeps the best result, refines it for the remaining time,
    stores the final sequence in ``height_sequence_1`` and returns it.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # safety margin for final checks

    # ------------------------------------------------------------------
    # Start from a previously found construction if it exists.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    if 'height_sequence_1' in globals():
        try:
            candidate = np.asarray(globals()['height_sequence_1'], dtype=float)
            # sanity check – evaluate externally
            external = evaluate_sequence(candidate.tolist())
            if external < best_score:
                best_score = external
                best_seq = candidate.copy()
                print(f"[init] loaded existing sequence: score={external:.6f} len={len(candidate)}")
        except Exception:
            pass

    attempt = 0

    # ------------------------------------------------------------------
    # Helper to compute the “hard” objective from a numpy vector.
    # ------------------------------------------------------------------
    def _objective(a: np.ndarray) -> float:
        n = a.shape[0]
        b = _full_convolution_fft(a)
        peak = b.max()
        # because sum(a) ≈ sqrt(2n) the evaluation simplifies to the peak
        return peak

    # ------------------------------------------------------------------
    # Main loop – independent restarts until time runs out.
    # ------------------------------------------------------------------
    while time.time() < end_time:
        attempt += 1

        # pick a length; favour moderate sizes that allow many iterations
        if rng.random() < 0.75:
            n = rng.integers(1500, 6000)   # moderate
        else:
            n = rng.integers(6000, 15000)  # larger, used less often

        # initialise a feasible vector
        a_start = _initial_vector(n, rng)

        # allocate a slice of the remaining budget (≈9 s per restart)
        slice_deadline = min(time.time() + 9.0, end_time)

        # run the sub‑gradient Adam optimiser
        a_opt, peak = _adam_subgrad_optimize(a_start, slice_deadline, rng)

        # compute (hard) score – this is exactly the peak when the ℓ₁‑norm is correct
        score = _objective(a_opt)

        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[restart] attempt={attempt:<3d} n={n:<5d}  new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final refinement of the best sequence with the remaining time.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < end_time - 2.0:
        final_deadline = end_time
        a_refined, peak = _adam_subgrad_optimize(best_seq, final_deadline, rng)
        refined_score = _objective(a_refined)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = a_refined.copy()
            print(f"[refine] improved final score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best vector as a plain list of floats.
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]                 # fallback – never used unless everything failed
    else:
        final_seq = best_seq.tolist()
        external = evaluate_sequence(final_seq)
        print(f"[final] external score = {external:.6f}  internal peak = {best_score:.6f}  length = {len(final_seq)}")
        globals()['height_sequence_1'] = final_seq

    return final_seq
