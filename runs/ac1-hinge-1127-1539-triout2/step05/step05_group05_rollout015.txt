"""
Propose a step‑function sequence that minimizes ``evaluate_sequence``.

Algorithmic outline
-------------------
1. **Diversified random seeding** – generate many candidates of several
   families (Dirichlet, Beta, power‑law, exponential, edge‑heavy,
   mixed) and keep the one with the smallest maximal self‑convolution.
2. **Projected sub‑gradient descent** – repeatedly move the probability
   vector away from the currently maximal convolution entry and project
   back onto the ℓ¹‑simplex (sum = 1).  A diminishing step‑size guarantees
   convergence to a local minimum.
3. **Linear‑programming refinement** – for the current vector `f`,
   solve the LP  ``max Σ b_i   s.t.   (f * b)_k ≤ M  (∀k) , b ≥ 0`` where
   ``M`` is the current maximal convolution value.  The LP solution
   `b` can be normalised to a probability vector `g` and mixed with `f`
   via a convex combination.  This step provably lowers the maximal
   entry whenever a feasible `b` with Σb_i > 1 exists.
4. **Targeted simulated annealing** – a fast SA that moves a small
   fraction of mass from indices that contribute heavily to the current
   peak towards random destinations, with a cooling schedule.
5. **Return** – truncate trailing zeros, perform a final sanity check
   with the official evaluator and output the list of floats.

All helper functions are top‑level; the routine respects the supplied
time budget (default 1000 s) and prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x_i≥0, Σx_i = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex Σx_i=s, x_i≥0."""
    if s <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random families for initialisation (sum = 1)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Generate a random non‑negative probability vector (Σa_i = 1).
    Several families are mixed to increase diversity.
    """
    kind = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if kind == 0:  # pure power‑law (decreasing)
        p = rng.uniform(0.15, 1.0)
        a = (idx + 1.0) ** (-p)

    elif kind == 1:  # exponential decay from the left edge
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif kind == 2:  # mixed power‑law + exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif kind == 3:  # Dirichlet (uniform simplex)
        a = rng.dirichlet(alpha=np.ones(n))

    elif kind == 4:  # symmetric edge‑heavy
        p = rng.uniform(0.2, 1.2)
        left = (idx + 1.0) ** (-p)
        right = (n - idx) ** (-p)
        a = left + right

    else:  # Beta‑shaped (U‑shaped or bell‑shaped)
        alpha = rng.uniform(0.2, 2.0)
        beta = rng.uniform(0.2, 2.0)
        x = (idx + 0.5) / n
        a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))

    # small perturbation + projection onto the simplex
    a += rng.normal(scale=0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, 1.0)
    return a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on max‑convolution (sum = 1)
# ----------------------------------------------------------------------


def _subgradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected sub‑gradient descent that targets the entry of the
    self‑convolution with maximal value.  The iterate always stays on the
    ℓ¹‑simplex Σa_i = 1.
    """
    a = a0.copy()
    n = a.size
    # record current max for monitoring
    cur_max = _max_convolution_fft(a)
    step = 0
    eta0 = 0.5
    eps = 1e-12

    while time.time() < deadline:
        # compute full convolution and locate the maximal index
        b = _full_convolution_fft(a)
        k = int(np.argmax(b))

        # gradient of the maximal entry (sub‑gradient of the max)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx_range = np.arange(lo, hi + 1, dtype=int)
        g = np.zeros_like(a)
        g[idx_range] = 2.0 * a[k - idx_range]

        # diminishing step size
        step += 1
        eta = eta0 / np.sqrt(step)

        # gradient step followed by projection onto the simplex
        a_new = a - eta * g
        a_new = _project_to_simplex(a_new, 1.0)

        new_max = _max_convolution_fft(a_new)

        # accept only if the max improves (or stays the same)
        if new_max <= cur_max:
            a = a_new
            cur_max = new_max
        # else keep the old iterate (the reduced step size will eventually
        # allow a descent)

    return a


# ----------------------------------------------------------------------
# Linear‑programming refinement (requires SciPy)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:          # pragma: no cover
    _SCIPY_AVAILABLE = False


def _lp_step(f: np.ndarray, M: float, mix: float):
    """
    One LP‑based refinement step for a probability vector ``f``.
    Returns a new vector (convex combination with the LP solution) and
    the LP objective Σb_i (or None on failure).
    """
    n = f.size
    rows = 2 * n - 1

    # Build the constraint matrix:   (f * b)_k ≤ M
    # A_ub[k, j] = f_{k-j}  for 0 ≤ k-j < n
    A = np.zeros((rows, n), dtype=float)
    for j in range(n):
        A[j:j + n, j] = f  # vectorised slice assignment

    # Linear programme:  maximise Σb_i  ⇔ minimise -Σb_i
    c = -np.ones(n)
    bounds = [(0.0, None) for _ in range(n)]
    res = linprog(
        c,
        A_ub=A,
        b_ub=np.full(rows, M),
        bounds=bounds,
        method="highs",
        options={"presolve": True, "time_limit": 5.0},
    )
    if not res.success:
        return None, None
    b = res.x
    sum_b = b.sum()
    if sum_b <= 0:
        return None, None
    # Normalise to a probability vector and mix with the current f
    g = b / sum_b
    new_f = (1.0 - mix) * f + mix * g
    return new_f, sum_b


def _lp_refinement_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Repeated LP refinements.  Starts from ``a0`` (Σa_i = 1, a_i > 0).
    Uses a decreasing mixing coefficient if an improvement is not seen.
    """
    if not _SCIPY_AVAILABLE:
        return a0  # fall back to the input unchanged

    a = np.maximum(a0, 1e-12)
    a = _project_to_simplex(a, 1.0)

    mix = 0.3
    while time.time() < deadline:
        M = _max_convolution_fft(a)
        new_a, _ = _lp_step(a, M, mix)
        if new_a is None:
            break
        new_M = _max_convolution_fft(new_a)

        if new_M < M:
            a = new_a
            # keep the same mixing factor
        else:
            # no progress – shrink the mixing factor
            mix *= 0.5
            if mix < 1e-5:
                break
    return a

def _targeted_simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    SA that moves a random fraction of mass from an index that heavily
    contributes to the current maximal convolution entry to a random
    destination.  Temperature decays multiplicatively.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # initial temperature proportional to current objective
    T = best_score * 0.02 + 1e-8
    cooling = 0.9995
    rng = np.random.default_rng()
    max_transfer = 0.25  # up to 25 % of the selected mass

    while time.time() < deadline:
        b = _full_convolution_fft(a)
        k_max = int(np.argmax(b))

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idx_range = np.arange(lo, hi + 1)
        contrib = a[idx_range] * a[k_max - idx_range]
        weight_sum = contrib.sum()

        if weight_sum > 0:
            probs = contrib / weight_sum
            i = rng.choice(idx_range, p=probs)
        else:
            i = rng.integers(0, n)

        # pick a distinct destination
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        delta = a[i] * rng.random() * max_transfer
        if delta <= 0.0:
            continue
        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        accept = False
        if cur_score <= best_score:
            accept = True
        elif T > 0.0 and rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA improvement: {best_score:.6f}")
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, 1.0)
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA jitter improvement: {best_score:.6f}")

    return best

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (probability vector) that
    minimises ``evaluate_sequence`` under the supplied time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    best_seq = None
    best_max = np.inf

    # ------------------------------------------------------------------
    # 0) optionally start from a provided global construction
    # ------------------------------------------------------------------
    global_seq = globals().get("height_sequence_1")
    if isinstance(global_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(global_seq, dtype=float)
        a0 = np.maximum(a0, 0.0)
        a0 = _project_to_simplex(a0, 1.0)   # normalise to probability vector
        cur_max = _max_convolution_fft(a0)
        best_seq, best_max = a0, cur_max
        print(f"[{time.time():.1f}] Start from global height_sequence_1 "
              f"(n={a0.size})  max‑conv={best_max:.6f}")

    # ------------------------------------------------------------------
    # 1) Random scanning over several lengths
    # ------------------------------------------------------------------
    lengths = [1000, 1200, 1500, 1800, 2000, 2500]
    scan_time_per_len = 6.0  # seconds per length

    for n in lengths:
        if time.time() >= deadline:
            break
        end_len = min(deadline, time.time() + scan_time_per_len)
        while time.time() < end_len:
            a = _random_initial_sequence(n, rng)
            m = _max_convolution_fft(a)
            if m < best_max:
                best_seq, best_max = a.copy(), m
                print(f"[{time.time():.1f}] New global best (scan) n={n}  max‑conv={m:.6f}")

    # ------------------------------------------------------------------
    # Fallback to a uniform vector if nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1200
        best_seq = np.full(n, 1.0 / n)
        best_max = _max_convolution_fft(best_seq)

    # ------------------------------------------------------------------
    # 2) Projected sub‑gradient descent (takes roughly half remaining time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        sub_deadline = time.time() + remaining * 0.45
        a_sub = _subgradient_descent_opt(best_seq, sub_deadline)
        m_sub = _max_convolution_fft(a_sub)
        if m_sub < best_max:
            best_seq, best_max = a_sub.copy(), m_sub
            print(f"[{time.time():.1f}] After sub‑gradient descent: max‑conv={m_sub:.6f}")

    # ------------------------------------------------------------------
    # 3) Linear‑programming based refinement (if SciPy is available)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0 and _SCIPY_AVAILABLE:
        lp_deadline = time.time() + remaining * 0.30
        a_lp = _lp_refinement_opt(best_seq, lp_deadline)
        m_lp = _max_convolution_fft(a_lp)
        if m_lp < best_max:
            best_seq, best_max = a_lp.copy(), m_lp
            print(f"[{time.time():.1f}] After LP refinement: max‑conv={m_lp:.6f}")

    # ------------------------------------------------------------------
    # 4) Targeted simulated annealing (remaining budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        a_sa = _targeted_simulated_annealing(best_seq, deadline)
        m_sa = _max_convolution_fft(a_sa)
        if m_sa < best_max:
            best_seq, best_max = a_sa.copy(), m_sa
            print(f"[{time.time():.1f}] After SA: max‑conv={m_sa:.6f}")

    # ------------------------------------------------------------------
    # 5) Clean‑up and final evaluation
    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Final sanity check with the official evaluator (if available)
    try:
        ev = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence = {ev:.6f}")
    except Exception:   # pragma: no cover
        pass

    print(f"[{time.time():.1f}] Finished. Best max‑conv = {best_max:.6f}   length = {best_seq.size}")

    return best_seq.tolist()
