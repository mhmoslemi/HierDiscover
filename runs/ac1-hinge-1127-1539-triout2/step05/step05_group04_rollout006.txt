def _exp_gradient_descent(p0, deadline):
    p = p0.copy()
    n = len(p)
    target = 1.0
    rng = np.random.default_rng()
    it = 0
    while time.time()<deadline:
        # compute conv
        conv = _max_convolution_fft(p)
        k = np.argmax(conv)

        # gradient = 2 * shifted p
        g = np.zeros_like(p)
        if k < n:
            g[:k+1] = 2 * p[k::-1]
        else:
            d = k - (n - 1)
            g[d:] = 2 * p[:n - d]
        # step size
        eta = 0.5 / np.sqrt(it+1)   # maybe adjust
        # exponentiated update
        p = p * np.exp(-eta * g)
        # renormalize
        s = p.sum()
        if s == 0:
            # fallback to uniform
            p = np.full(n, 1.0 / n)
        else:
            p /= s
        # occasional jitter
        if it % 200 == 0:
            jitter_mask = rng.random(n) < 0.05
            # add small random noise via Dirichlet mixing maybe
            p[jitter_mask] = (p[jitter_mask] + rng.random(jitter_mask.sum()) * 0.01) 
            p = np.maximum(p, 0)
            p /= p.sum()
        it += 1
    return p
```

"""
Propose a non‑negative step‑function (list of heights) that minimizes the
official ``evaluate_sequence`` metric.

The algorithm combines several complementary optimisation ideas:

* *Random families*: a diverse set of initial probability distributions
  (exponential, power‑law, Dirichlet, piecewise, etc.) is sampled.

* *Exponentiated‑gradient descent* on the ℓ¹‑simplex: the objective is the
  infinity‑norm of the self‑convolution.  For the currently maximal shift
  ``k*`` the sub‑gradient is ``2·p_{k*‑i}``.  An exponentiated‑gradient update
  (multiplicative weights) preserves non‑negativity and the ℓ¹‑norm exactly,
  which often yields a smoother descent than Euclidean projection.

* *Exact max‑index sub‑gradient* and a cheap *mass‑balance* greedy move
  (identical to the reference implementation) are applied as refinements.

* *Final simulated‑annealing* polishing moves a small fraction of mass between
  two random sites while accepting occasional uphill steps.

Time is managed automatically: the total wall‑clock budget (default 1000 s)
is split among a sequence of lengths (2000 … 12000).  For each length a few
short exponentiated‑gradient runs are performed, the best candidate is
refined with the sub‑gradient / mass‑balance stages and finally polished with
SA.  The best sequence found overall is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·n−1) using an FFT.
    Returned array is real‑valued.
    """
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _metric(a: np.ndarray) -> float:
    """Official metric for a given (non‑negative) sequence a."""
    n = a.size
    max_b = np.max(_max_convolution_fft(a))
    sum_a = np.sum(a)
    # sum_a is never zero – the optimiser always keeps a positive total mass
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Random initial distributions (sum = 1)
# ----------------------------------------------------------------------


def _random_initial_distribution(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random probability vector (sum = 1) of length n.
    Several families are sampled to increase diversity.
    """
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:  # power‑law
        p = (idx + 1.0) ** (-rng.uniform(0.15, 1.1))
    elif family == 1:  # pure exponential
        lam = rng.exponential(scale=0.02)
        p = np.exp(-lam * idx)
    elif family == 2:  # power × exponential
        p = (idx + 1.0) ** (-rng.uniform(0.15, 1.0)) * np.exp(-rng.exponential(scale=0.02) * idx)
    elif family == 3:  # Dirichlet (uniform on simplex)
        p = rng.dirichlet(alpha=np.ones(n))
    elif family == 4:  # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        p = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])
    else:  # almost uniform with tiny jitter
        p = np.full(n, 1.0 / n)

    # add a whisper of Gaussian noise and re‑normalise
    p += rng.normal(scale=1e-3, size=n)
    p = np.maximum(p, 0.0)
    total = p.sum()
    if total == 0.0:
        p = np.full(n, 1.0 / n)
    else:
        p /= total
    return p


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the max‑convolution (Phase 0)
# ----------------------------------------------------------------------


def _exp_gradient_descent(p0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Multiplicative‑weights (exponentiated) descent for the objective
    ``max_t (p * p)_t`` under the ℓ¹‑simplex.  ``p0`` must sum to 1.
    """
    p = p0.copy()
    n = p.size
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        conv = _max_convolution_fft(p)
        k = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(p)
        if k < n:
            # left half
            g[:k + 1] = 2.0 * p[k::-1]
        else:
            # right half
            d = k - (n - 1)          # 0 … n‑1
            g[d:] = 2.0 * p[: n - d]

        # step‑size schedule η_t = η₀ / √(t+1)
        eta0 = 0.5
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated update
        # to avoid overflow/underflow we shift by the maximum exponent
        exp_arg = -eta * g
        max_arg = np.max(exp_arg)
        factors = np.exp(exp_arg - max_arg)
        p *= factors
        # renormalise to stay on the simplex
        s = p.sum()
        if s == 0.0:
            p = np.full(n, 1.0 / n)
        else:
            p /= s

        # occasional tiny jitter prevents stagnation
        if it % 200 == 0:
            eps = rng.random() * 0.01
            jitter = rng.random(n) * eps
            p = (1 - eps) * p + eps * jitter
            p = np.maximum(p, 0.0)
            p /= p.sum()

        it += 1
    return p


# ----------------------------------------------------------------------
# Exact max‑index sub‑gradient (Phase 1) – same as reference code
# ----------------------------------------------------------------------


def _max_index_subgradient(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max_t (a*a)_t."""
    n = a.size
    conv = _max_convolution_fft(a)
    k = int(np.argmax(conv))

    g = np.zeros_like(a)
    if k < n:
        g[:k + 1] = 2.0 * a[k::-1]
    else:
        d = k - (n - 1)
        g[d:] = 2.0 * a[: n - d]
    return g, k


def _subgradient_descent_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Projected sub‑gradient descent on the true max‑norm."""
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        g, _ = _max_index_subgradient(a)

        # diminishing step size η_t = α / √t
        step = 0.8 * target_sum / (np.sqrt(it + 1) * np.linalg.norm(g) + 1e-12)
        a -= step * g

        # enforce feasibility
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1
    return a


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1, dtype=float)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Cheap mass‑balance refinement (Phase 2)
# ----------------------------------------------------------------------


def _mass_balance_refine(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Greedy redistribution that reduces the contribution of the maximal convolution entry."""
    a = a0.copy()
    n = a.size
    target_sum = np.sum(a)  # should already be √(2 n)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        conv = _max_convolution_fft(a)
        k = int(np.argmax(conv))

        # contributions of each index to that entry
        idx = np.arange(n)
        other = k - idx
        mask = (other >= 0) & (other < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[other[mask]]

        i_high = int(np.argmax(contrib))

        # pick a relatively empty position – among the 20 % smallest masses
        low_candidates = np.where(a <= np.percentile(a, 20))[0]
        if low_candidates.size == 0:
            i_low = int(np.argmin(a))
        else:
            i_low = int(rng.choice(low_candidates))

        # move a modest fraction of the heavy mass
        delta = a[i_high] * 0.03
        if delta < 1e-12:
            break

        a[i_high] -= delta
        a[i_low] += delta

        # re‑project onto the simplex
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        if it % 100 == 0 and rng.random() < 0.02:
            jitter = rng.normal(scale=target_sum * 1e-3, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1
    return a


# ----------------------------------------------------------------------
# Very light simulated‑annealing (Phase 3)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Simple SA that swaps a random fraction of mass between two sites."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    cur_score = _metric(a)
    best_score = cur_score

    T = cur_score * 0.02 + 1e-9
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-16:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        frac = rng.random() * 0.15
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta

        new_score = _metric(a)
        dE = new_score - cur_score
        accept = dE <= 0.0 or (T > 0.0 and rng.random() < np.exp(-dE / T))

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling
    return best


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step–function that minimises the official
    ``evaluate_sequence`` metric.  The routine respects a wall‑clock budget,
    tries several lengths, uses exponentiated‑gradient descent followed by
    sub‑gradient, mass‑balance and simulated‑annealing refinements, and finally
    returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5  # tiny safety margin

    best_seq = None
    best_metric = np.inf

    # --------------------------------------------------------------
    # Seed from a previously discovered construction if available
    # --------------------------------------------------------------
    earlier = globals().get("height_sequence_1")
    if isinstance(earlier, (list, tuple, np.ndarray)):
        a0 = np.asarray(earlier, dtype=float)
        if a0.size > 0:
            n0 = a0.size
            target0 = np.sqrt(2.0 * n0)
            a0 = a0 * (target0 / np.sum(a0))        # ℓ¹‑normalise to √(2 n)
            metric0 = _metric(a0)
            best_seq = a0.copy()
            best_metric = metric0
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) metric={metric0:.6f}")

    # --------------------------------------------------------------
    # Lengths to investigate – larger n gives more freedom but costs more
    # --------------------------------------------------------------
    lengths = [2000, 3000, 4000, 6000, 8000, 10000, 12000]

    # Iterate over lengths, allocating a slice of the remaining budget to each
    for idx_len, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)

        # ----------------------------------------------------------
        # Phase 0 – many cheap exponentiated‑gradient runs
        # ----------------------------------------------------------
        # Allocate at most 30 % of the remaining time to random restarts
        remaining = deadline - time.time()
        max_restart_time = min(30.0, remaining * 0.3)
        per_restart = 3.0                      # seconds per EG run
        max_restarts = int(max_restart_time // per_restart)

        for r in range(max_restarts):
            if time.time() >= deadline - 10.0:
                break
            # random initial distribution (sum = 1)
            p0 = _random_initial_distribution(n, rng)

            # short EG descent
            sub_dead = min(deadline - 5.0, time.time() + per_restart)
            p = _exp_gradient_descent(p0, sub_dead)

            # scale to the required ℓ¹‑norm for evaluation
            a = p * target_sum
            metric = _metric(a)

            if metric < best_metric:
                best_metric = metric
                best_seq = a.copy()
                print(f"[{time.time():.1f}] New global best (restart) n={n} metric={metric:.6f}")

        # ----------------------------------------------------------
        # Phase 1 – deeper refinement if we have a candidate of this size
        # ----------------------------------------------------------
        if best_seq is not None and best_seq.size == n:
            # Sub‑gradient descent (exact max‑norm)
            sub_dead = min(deadline, time.time() + 30.0)
            a_ref = _subgradient_descent_max(best_seq, sub_dead)
            metric = _metric(a_ref)
            if metric < best_metric:
                best_metric = metric
                best_seq = a_ref.copy()
                print(f"[{time.time():.1f}] Sub‑gradient improvement n={n} metric={metric:.6f}")

            # Mass‑balance fine‑tuning
            sub_dead = min(deadline, time.time() + 20.0)
            a_ref2 = _mass_balance_refine(best_seq, sub_dead)
            metric = _metric(a_ref2)
            if metric < best_metric:
                best_metric = metric
                best_seq = a_ref2.copy()
                print(f"[{time.time():.1f}] Mass‑balance refinement n={n} metric={metric:.6f}")

            # Optional brief EG pass (helps escape shallow local minima)
            if time.time() < deadline - 5.0:
                p_cur = best_seq / np.sum(best_seq)
                sub_dead = min(deadline, time.time() + 5.0)
                p = _exp_gradient_descent(p_cur, sub_dead)
                a_ref3 = p * np.sum(best_seq)   # keep original ℓ¹‑norm
                metric = _metric(a_ref3)
                if metric < best_metric:
                    best_metric = metric
                    best_seq = a_ref3.copy()
                    print(f"[{time.time():.1f}] Brief EG refinement n={n} metric={metric:.6f}")

        # ----------------------------------------------------------
        # Phase 2 – (optional) simulated‑annealing polishing if time left
        # ----------------------------------------------------------
        left = deadline - time.time()
        if left > 8.0 and best_seq is not None and best_seq.size == n:
            a_final = _simulated_annealing(best_seq, deadline)
            metric = _metric(a_final)
            if metric < best_metric:
                best_metric = metric
                best_seq = a_final.copy()
                print(f"[{time.time():.1f}] Final SA improvement (n={n}) metric={metric:.6f}")

    # ------------------------------------------------------------------
    # Clean up tiny trailing zeros and report the official evaluation
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback – uniform distribution of length 2000 (should never happen)
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
