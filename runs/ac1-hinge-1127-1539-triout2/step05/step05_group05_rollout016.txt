"""
Algorithm:
1. Randomly sample sequences of several lengths; keep the best few.
2. For each promising candidate run a two‑stage continuous optimisation:
   a) Adam optimiser on a log‑sum‑exp smooth approximation of the
      maximal convolution entry, slowly raising the smoothing parameter.
   b) Optional linear‑programming (LP) improvement step (only for moderate
      lengths) that solves max‑sum under the current maximal convolution bound.
3. Final refinement with targeted simulated‑annealing moves.
All steps respect a global time budget (default 1000 s) and the final sequence
is projected onto the ℓ¹‑simplex Σ a_i = √(2 n).
"""
import time
import numpy as np

# optional import of SciPy for LP step
try:
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:
    _SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())

def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

# ----------------------------------------------------------------------
# Simplex projection (Euclidean, Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x_i≥0, Σx_i = s}."""
    if s <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) of the convolution and its gradient
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Return (g, b) where
        b = a*a (full convolution, length 2·n‑1)
        g = ∇_a  ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a ⋆ p)   with   p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]

    # b = a*a using FFT
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    # stable exponentials
    exps = np.exp(beta * (b - max_b))
    denom = np.sum(exps)
    if denom == 0.0:
        p = np.full_like(b, 1.0 / b.size)
    else:
        p = exps / denom

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1               # = 3·n‑2
    size2 = 1 << ((conv_len).bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]        # central n entries
    return g, b

# ----------------------------------------------------------------------
# Sub‑gradient of the (non‑smooth) max‑convolution
# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray):
    """Return (g, b_max) where g is a sub‑gradient of max_k (a*a)[k]."""
    n = a.shape[0]
    b = np.convolve(a, a)                     # length 2n‑1, plain python loops (fast enough)
    k = int(np.argmax(b))
    # build gradient for that k
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    grad = np.zeros_like(a)
    if lo <= hi:
        # grad[i] = 2 * a[k - i] for i in [lo, hi]
        idx = np.arange(lo, hi + 1)
        grad[idx] = 2.0 * a[k - idx]
    return grad, b[k]

# ----------------------------------------------------------------------
# Adam optimiser on the smoothed max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Adam optimiser on the log‑sum‑exp surrogate of max‑convolution."""
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    step = 0
    beta = 0.5            # smoothing parameter
    beta_max = 1e7
    beta_mult = 1.6
    eta0 = 0.05           # base learning rate (on the a‑space)

    while time.time() < deadline:
        step += 1
        # compute gradient of smooth objective
        g, _ = _gradient_softmax(a, beta)

        # Adam update on the a‑vector (additive)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        lr = eta0 / (np.sqrt(step) + 1.0)   # simple decay
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep non‑negative and re‑project to simplex
        a = np.maximum(a, 0.0)
        s = np.sum(a)
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # anneal smoothing
        if step % 400 == 0 and beta < beta_max:
            beta = min(beta * beta_mult, beta_max)

    return a

# ----------------------------------------------------------------------
# Linear‑programming improvement (only for moderate lengths)
# ----------------------------------------------------------------------
def _lp_improvement(a: np.ndarray, target_sum: float, deadline: float) -> np.ndarray:
    """One LP step that tries to increase the ℓ¹‑mass under the current max‑conv bound."""
    if not _SCIPY_AVAILABLE:
        return a
    n = a.size
    if n > 2400:   # matrix would be too large; skip
        return a
    # current bound
    M = _max_convolution_fft(a)

    # build convolution matrix C where C @ x = a * x
    m = 2 * n - 1
    C = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        C[i:i + n, i] = a

    # objective: maximize sum(x)  <=> minimize -sum(x)
    c = -np.ones(n, dtype=np.float64)

    # inequality constraints: C @ x <= M
    b_ub = np.full(m, M, dtype=np.float64)

    # bounds for variables
    bounds = [(0.0, None)] * n

    # solve
    res = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        return a

    x = res.x
    # renormalise to the ℓ¹‑simplex
    s = np.sum(x)
    if s > 0:
        x = x * (target_sum / s)
    else:
        x = np.full_like(x, target_sum / n)

    # mix the new point a little (tuning parameter)
    t = 0.4
    a_new = (1 - t) * a + t * x
    return a_new

# ----------------------------------------------------------------------
# Targeted simulated‑annealing refinement
# ----------------------------------------------------------------------
def _targeted_simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Simple SA that moves a fraction of mass from a “heavy” index to a random destination."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # temperature schedule
    T = best_score * 0.02 + 1e-8
    cooling = 0.9993
    rng = np.random.default_rng()
    max_transfer = 0.3

    while time.time() < deadline:
        # compute convolution and locate a maximising index
        b = _full_convolution_fft(a)
        k_max = int(np.argmax(b))

        # pick an index that heavily participates in b[k_max]
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idx_range = np.arange(lo, hi + 1)
        contrib = a[idx_range] * a[k_max - idx_range]
        total = contrib.sum()
        if total > 0:
            probs = contrib / total
            i = rng.choice(idx_range, p=probs)
        else:
            i = rng.integers(0, n)

        # destination
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        delta = a[i] * rng.random() * max_transfer
        if delta <= 0.0:
            continue
        a[i] -= delta
        a[j] += delta

        cur = _max_convolution_fft(a)
        dE = cur - best_score

        accept = False
        if cur <= best_score:
            accept = True
        elif rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur < best_score:
                best_score = cur
                best = a.copy()
                print(f"[{time.time():.1f}] SA improvement: {best_score:.6f}")
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # occasional jitter to escape plateaus
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0:
                a = a * (np.sqrt(2.0 * n) / s)
            cur = _max_convolution_fft(a)
            if cur < best_score:
                best_score = cur
                best = a.copy()
                print(f"[{time.time():.1f}] SA jitter improvement: {best_score:.6f}")

    return best

# Random initialisers (diverse families)
# ----------------------------------------------------------------------
def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Generate a random non‑negative sequence on the ℓ¹‑simplex Σ a_i = √(2 n)."""
    target_sum = np.sqrt(2.0 * n)
    kind = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if kind == 0:                     # decreasing power‑law
        p = rng.uniform(0.15, 1.0)
        a = (idx + 1.0) ** (-p)

    elif kind == 1:                 # exponential decay from left
        lam = rng.exponential(scale=0.02)
        a = np.exp(-lam * idx)

    elif kind == 2:                 # mixture
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.02)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif kind == 3:                 # Dirichlet (uniform simplex)
        a = rng.dirichlet(alpha=np.ones(n))

    elif kind == 4:                 # symmetric edge‑heavy
        p = rng.uniform(0.2, 1.2)
        left = (idx + 1.0) ** (-p)
        right = (n - idx) ** (-p)
        a = left + right

    else:                           # Beta‑shaped
        alpha = rng.uniform(0.2, 2.0)
        beta = rng.uniform(0.2, 2.0)
        x = (idx + 0.5) / n
        a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))

    a += rng.normal(scale=target_sum * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target_sum)
    return a

"""
Algorithm overview
------------------

1. **Diversified random seeding** – generate many candidates of several
   lengths (800 … 5000) using Dirichlet, power‑law, exponential,
   symmetric edge‑heavy and Beta families.  Each candidate is projected onto
   the ℓ¹‑simplex Σ a_i = √(2 n) and its maximal self‑convolution entry is
   evaluated with an FFT.  The best candidate overall is kept.

2. **Optional LP improvement** – for moderate lengths (≤ 2400) we solve a
   linear programme

        maximise   Σ_j x_j
        subject to  a * x ≤ M   (component‑wise)
                    x_j ≥ 0

   where * denotes linear convolution and M is the current maximal
   convolution value of *a*.  The LP gives a direction that can increase the
   ℓ¹‑mass without raising the bound M.  The solution is mixed with the
   current iterate and renormalised.  This step is repeated while time
   permits.

   log‑sum‑exp approximation of the max‑convolution value

        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)

    else:
        p = exps / denom

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = 3 * n - 2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Adam optimiser on the log‑sum‑exp surrogate of the maximal convolution.
    The smoothing parameter beta is increased slowly so that the surrogate
    converges to the true ℓ∞‑norm.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    step = 0
    beta = 0.5            # initial smoothing
    beta_max = 1e7
    beta_mult = 1.6
    eta0 = 0.05           # base learning rate

    while time.time() < deadline:
        step += 1
        g, _ = _gradient_softmax(a, beta)

        # Adam update (additive)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        lr = eta0 / (np.sqrt(step) + 1.0)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the ℓ¹‑simplex
        a = np.maximum(a, 0.0)
        s = np.sum(a)
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # slowly raise the smoothing parameter
        if step % 400 == 0 and beta < beta_max:
            beta = min(beta * beta_mult, beta_max)

    return a


# ----------------------------------------------------------------------
# Linear‑programming improvement (moderate lengths only)
# ----------------------------------------------------------------------
def _lp_improvement(a: np.ndarray, target_sum: float, deadline: float) -> np.ndarray:
    """
    Solve a linear programme that maximises the ℓ¹‑mass under the current
    maximal convolution bound  M = max(a*a).  The new point is mixed with the
    old one (mixing weight t) and renormalised to the simplex.
    """
    if not _SCIPY_AVAILABLE or a.size > 2400:
        return a

    n = a.size
    M = _max_convolution_fft(a)

    # Build the convolution matrix C such that C @ x = a * x
    m = 2 * n - 1
    C = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        C[i:i + n, i] = a

    # maximise Σ x_j  ↔  minimise -Σ x_j
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(m, M, dtype=np.float64)
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return a

    x = res.x
    s = np.sum(x)
    if s > 0.0:
        x = x * (target_sum / s)
    else:
        x = np.full_like(x, target_sum / n)

    # mix a little to retain stability
    t = 0.4
    return (1.0 - t) * a + t * x


# ----------------------------------------------------------------------
# Targeted simulated‑annealing refinement
# ----------------------------------------------------------------------
def _targeted_simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Refine a candidate by moving a random fraction of mass from an index
    heavily involved in the maximal convolution entry to a random destination.
    Uphill moves are accepted with a temperature schedule; occasional jitter
    helps escape deep minima.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # temperature schedule
    T = best_score * 0.02 + 1e-8
    cooling = 0.9993
    rng = np.random.default_rng()
    max_transfer = 0.3

    while time.time() < deadline:
        b = _full_convolution_fft(a)
        k_max = int(np.argmax(b))

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idx_range = np.arange(lo, hi + 1)
        contrib = a[idx_range] * a[k_max - idx_range]
        tot = contrib.sum()

        if tot > 0.0:
            probs = contrib / tot
            i = rng.choice(idx_range, p=probs)
        else:
            i = rng.integers(0, n)

        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        delta = a[i] * rng.random() * max_transfer
        if delta <= 0.0:
            continue
        a[i] -= delta
        a[j] += delta

        cur = _max_convolution_fft(a)
        dE = cur - best_score

        accept = cur <= best_score or rng.random() < np.exp(-dE / T)
        if accept:
            if cur < best_score:
                best_score = cur
                best = a.copy()
                print(f"[{time.time():.1f}] SA improvement: {best_score:.6f}")
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0.0:
                a = a * (np.sqrt(2.0 * n) / s)
            cur = _max_convolution_fft(a)
            if cur < best_score:
                best_score = cur
                best = a.copy()
                print(f"[{time.time():.1f}] SA jitter improvement: {best_score:.6f}")

    return best


# ----------------------------------------------------------------------
# Random initialisers (diverse families)
# ----------------------------------------------------------------------
def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Generate a random non‑negative sequence on the ℓ¹‑simplex Σ a_i = √(2 n).
    Several families are mixed to promote diversity.
    """
    target_sum = np.sqrt(2.0 * n)
    kind = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if kind == 0:                     # decreasing power‑law
        p = rng.uniform(0.15, 1.0)
        a = (idx + 1.0) ** (-p)

    elif kind == 1:                 # exponential decay
        lam = rng.exponential(scale=0.02)
        a = np.exp(-lam * idx)

    elif kind == 2:                 # mixed power‑law + exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.02)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif kind == 3:                 # Dirichlet (uniform simplex)
        a = rng.dirichlet(alpha=np.ones(n))

    elif kind == 4:                 # symmetric edge‑heavy
        p = rng.uniform(0.2, 1.2)
        left = (idx + 1.0) ** (-p)
        right = (n - idx) ** (-p)
        a = left + right

    else:                           # Beta‑shaped
        alpha = rng.uniform(0.2, 2.0)
        beta = rng.uniform(0.2, 2.0)
        x = (idx + 0.5) / n
        a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))

    a += rng.normal(scale=target_sum * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target_sum)
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative sequence) that minimises the
    ``evaluate_sequence`` objective under the supplied time budget.
    The routine combines random seeding, an optional LP mass‑reallocation
    step, an Adam optimiser on a smoothed max‑convolution surrogate and a
    final simulated‑annealing refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Start from a pre‑existing global candidate if available
    # ------------------------------------------------------------------
    global_seq = globals().get("height_sequence_1")
    if isinstance(global_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(global_seq, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        sc = _max_convolution_fft(a0)
        best_seq, best_score = a0.copy(), sc
        print(f"[{time.time():.1f}] Start from global height_sequence_1 "
              f"(n={n0})  score={sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – diversified random scan (several lengths)
    # ------------------------------------------------------------------
    lengths = [800, 1200, 1600, 2000, 2500, 3000, 3500, 4000, 5000]
    scan_time_per_len = 6.0
    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)
        end_phase = min(deadline, time.time() + scan_time_per_len)
        while time.time() < end_phase:
            a = _random_initial_sequence(n, rng)
            score = _max_convolution_fft(a)
            if score < best_score:
                best_seq, best_score = a.copy(), score
                print(f"[{time.time():.1f}] New global best (scan) n={n}  "
                      f"score={score:.6f}")

    # fallback: uniform candidate if nothing was generated
    if best_seq is None:
        n = 1200
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = _max_convolution_fft(best_seq)

    # ------------------------------------------------------------------
    # Phase 2 – optional LP‑based improvement (moderate length only)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and _SCIPY_AVAILABLE and best_seq.size <= 2400:
        lp_deadline = time.time() + remaining * 0.25
        a_lp = best_seq.copy()
        target_sum = np.sqrt(2.0 * a_lp.size)
        while time.time() < lp_deadline:
            a_lp = _lp_improvement(a_lp, target_sum, lp_deadline)
            sc = _max_convolution_fft(a_lp)
            if sc < best_score:
                best_seq, best_score = a_lp.copy(), sc
                print(f"[{time.time():.1f}] LP improvement: score={sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam optimisation on smoothed max‑convolution
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        opt_deadline = time.time() + remaining * 0.55
        a_opt = _adam_optimize(best_seq, opt_deadline)
        sc = _max_convolution_fft(a_opt)
        if sc < best_score:
            best_seq, best_score = a_opt.copy(), sc
            print(f"[{time.time():.1f}] Adam optimisation: score={sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – targeted simulated‑annealing refinement
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        a_sa = _targeted_simulated_annealing(best_seq, deadline)
        sc = _max_convolution_fft(a_sa)
        if sc < best_score:
            best_seq, best_score = a_sa.copy(), sc
            print(f"[{time.time():.1f}] SA final improvement: score={sc:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing zeros and return a plain Python list
    # ------------------------------------------------------------------
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # sanity check with the official evaluator (if it exists)
    try:
        ev = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence -> {ev:.6f}")
    except Exception:                     # pragma: no cover
        pass

    print(f"[{time.time():.1f}] Finished. Best objective (max‑conv) = "
          f"{best_score:.6f}   length = {best_seq.size}")

    return best_seq.tolist()
