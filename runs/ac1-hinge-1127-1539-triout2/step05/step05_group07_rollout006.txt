def compute_max_conv(seq):
    n = seq.shape[0]
    total_len = 2*n - 1
    f_len = 1 << ((total_len - 1).bit_length())  # next pow2
    A = np.fft.rfft(seq, n=f_len)
    conv = np.fft.irfft(A*A, n=f_len)[:total_len]
    return conv.max()
```

"""
adversarial step‑function optimizer (new version)

This implementation searches a parameterised family of “edge‑heavy” shapes
(distance‑to‑edge raised to a power or an exponential decay) for many
different lengths, keeps the best candidate and finally refines it with a
soft‑max Adam optimisation on the simplex (log‑softmax parametrisation).
The whole process respects a 1000 s time‑budget and stores the best
sequence in the global variable ``height_sequence_1`` for reuse by later
runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based full linear self‑convolution (real inputs)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def compute_max_conv(seq: np.ndarray) -> float:
    """Convenient wrapper: maximum entry of the self‑convolution."""
    return _full_convolution_fft(seq).max()


# ----------------------------------------------------------------------
# Families of candidate shapes
# ----------------------------------------------------------------------
def edge_power_seq(n: int, p: float) -> np.ndarray:
    """a_i ∝ (dist_to_edge+1)^{-p}   (dist = min(i,n‑1‑i))."""
    idx = np.arange(n, dtype=float)
    dist = np.minimum(idx, n - 1 - idx) + 1.0
    return dist ** (-p)


def edge_exp_seq(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ exp(-alpha * dist_to_edge)   (dist = min(i,n‑1‑i))."""
    idx = np.arange(n, dtype=float)
    dist = np.minimum(idx, n - 1 - idx)
    return np.exp(-alpha * dist)


# ----------------------------------------------------------------------
# Gradient of a τ‑soft‑max approximation of max(b) where b = a*a
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Returns (g, max_b) where g = ∂/∂a τ‑soft‑max(b) and max_b = max(a*a).
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / exps.sum()

    # gradient: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = mask.size + n - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Minimise a τ‑soft‑max approximation of the convolution peak.
    Returns the best vector seen and the corresponding (hard) peak value.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    # initialise logits from a0 (ensure they produce a0 after softmax)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                     # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.18

    # start smoothing parameter – a few times the current peak
    tau = max(_full_convolution_fft(a0).max(), 1.0)

    best_a = a0.copy()
    best_peak = _full_convolution_fft(a0).max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- a from logits (log‑softmax) -----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ----- convolution and gradient -----
        g, peak = _softmax_gradient(a, tau)

        # keep record of the best *hard* peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ----- gradient wrt logits -----
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # optional clipping for stability
        clip = np.max(np.abs(grad_p))
        if clip > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- schedules -----
        tau *= 0.9999           # keep the soft‑max fairly soft for longer
        lr *= 0.9997            # gentle decay

        # occasional jitter to hop out of plateaus
        if it % 400 == 0:
            p += rng.normal(scale=0.015, size=n)

    return best_a, best_peak
