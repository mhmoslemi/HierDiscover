"""
Propose a step‑function that minimizes ``evaluate_sequence``.

Algorithm
---------
1. **Initialisation** – generate a random non‑negative vector on the
   simplex Σ a = √(2·n) (several families are mixed, the previous best
   solution is up‑sampled when available).

2. **Smooth Adam phase** – minimise a smooth log‑sum‑exp approximation of
   ``max(conv)``.  The gradient is obtained by an FFT‑based convolution.
   The temperature (soft‑max sharpness) is increased slowly up to a high
   value (α≈80) so that the smooth objective approaches the true
   ``max``.  After each step the vector is projected onto the simplex.

3. **Sub‑gradient refinement** – focus on the worst shift and perform a
   classic sub‑gradient step with a diminishing step‑size (√t schedule).

4. **Greedy local refinement** – repeatedly move a small amount of mass
   from the index that contributes most to the current worst
   convolution entry to an index that does **not** participate in that
   entry (typically an extreme index).  This directly reduces the peak
   without harming other shifts.

5. **Iterative up‑sampling** – when a better shape is found it is
   up‑sampled to a larger problem size and the whole optimisation is
   repeated.  Larger n gives more freedom to approach the continuous
   optimum.

The whole search runs for the allocated time budget (≤1000 s).  The
best sequence seen is returned.  All numbers are non‑negative and the
implementation respects the ‑‑infra restrictions (no I/O, single CPU,
pure‑Python + NumPy).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  (1/α)·log‑sum‑exp(α·(a*a)) wrt a.
    Returns a vector of length n.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights (smooth approximation of max)
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """w_i = exp(alpha*conv_i) / Σ_j exp(alpha*conv_j)  (stable)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Initial sequence generators
# ----------------------------------------------------------------------
def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Resize an existing sequence to length n using linear interpolation."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n with ℓ¹‑norm √(2·n).
    Several random families are mixed to diversify the search.
    """
    S = np.sqrt(2.0 * n)

    typ = rng.integers(0, 5)

    if typ == 0 and base_seq is not None and base_seq.size > 10:
        a = _upsample_seq(base_seq, n)
        a += rng.normal(scale=a.mean() * 1e-3, size=n)
        a = np.clip(a, 0.0, None)
    elif typ == 1:
        a = rng.random(n)                       # Dirichlet(1)
    elif typ == 2:
        rate = rng.uniform(0.9, 0.9995)
        left = rate ** np.arange(n)
        right = left[::-1]
        a = (left + right) / 2.0
    elif typ == 3:
        idx = np.arange(n, dtype=float)
        a = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.8)
        a = a ** power
    else:
        # convex combination of a few bases
        cand = []
        cand.append(rng.random(n))
        rate = rng.uniform(0.9, 0.9995)
        left = rate ** np.arange(n)
        cand.append((left + left[::-1]) / 2.0)
        idx = np.arange(n, dtype=float)
        rec = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.8)
        cand.append(rec ** power)
        w = rng.random(len(cand))
        w /= w.sum()
        a = sum(wi * ci for wi, ci in zip(w, cand))

    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0)
        a_sum = a.sum()
    a = a / a_sum * S
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smooth log‑sum‑exp max‑approximation
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected Adam minimising the smoothed max‑norm."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.25
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # smooth‑max temperature schedule
    alpha = 0.5
    alpha_mul = 1.025          # gradual tightening
    alpha_max = 80.0           # final sharpness

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        cur_max = np.max(conv)
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # soft‑max weighting
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # tentative step
        a_next = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_next = _project_simplex(a_next, S)

        # simple back‑track if the moving max rises dramatically
        new_max = np.max(_full_convolution(a_next))
        if new_max > cur_max * 1.02:        # too bad – halve the learning rate
            lr *= 0.5
        else:
            a = a_next
            lr *= 0.9999                     # gentle decay

        # occasional random mass‑swap (helps escape plateaus)
        if rng.random() < 0.015:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.05 * a[i] * rng.random()
                cand = a.copy()
                cand[i] = max(0.0, cand[i] - delta)
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < best_score or rng.random() < 0.12:
                    a = cand

        # temperature update
        if alpha < alpha_max:
            alpha = min(alpha * alpha_mul, alpha_max)

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient phase (focus on the worst shift)
# ----------------------------------------------------------------------
def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the true max‑norm (only the worst entry)."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.15                # initial step‑size constant
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # gradient of the worst entry
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # occasional random perturbation
        if rng.random() < 0.05:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.04 * a[i] * rng.random()
                cand = a.copy()
                cand[i] = max(0.0, cand[i] - delta)
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.25:
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy local refinement (mass moves that directly shrink the worst peak)
# ----------------------------------------------------------------------
def _greedy_refine(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a small amount of mass from the index that contributes most
    to the current worst convolution entry to an index that does not participate
    in that entry (typically an extreme index).  This directly attacks the peak.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best = a.copy()
    best_score = np.max(_full_convolution(a))

    while time.time() < deadline:
        conv = _full_convolution(a)
        k = int(np.argmax(conv))

        low = max(0, k - (n - 1))
        high = min(n - 1, k)

        if low > high:          # should not happen, but guard
            break

        idx = np.arange(low, high + 1)
        contrib = a[idx] * a[k - idx]
        if contrib.size == 0:
            break

        i_src = idx[np.argmax(contrib)]

        # choose a destination outside the overlapping interval
        if low > 0:
            j_dest = 0
        elif high < n - 1:
            j_dest = n - 1
        else:
            # fallback: random index far enough
            j_dest = rng.integers(0, n)
            while low <= j_dest <= high:
                j_dest = rng.integers(0, n)

        delta = 0.03 * a[i_src] * rng.random()
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[i_src] = max(0.0, cand[i_src] - delta)
        cand[j_dest] += delta
        cand = _project_simplex(cand, S)

        cand_score = np.max(_full_convolution(cand))
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()

        # accept if we lowered the current peak, otherwise keep a unchanged
        if cand_score < np.max(conv):
            a = cand
        else:
            # occasional uphill move to escape local minima
            if rng.random() < 0.03:
                a = cand

    return best, best_score


# ----------------------------------------------------------------------
# Combined optimiser (Adam → Sub‑gradient → Greedy)
# ----------------------------------------------------------------------
def _optimize_combined_refine(a0: np.ndarray,
                              deadline: float,
                              rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run a smooth Adam phase, a sub‑gradient phase and finish with a greedy
    mass‑swap refinement.  The time budget is split roughly 0.5 / 0.3 / 0.2.
    """
    now = time.time()
    if now >= deadline:
        return a0, np.max(_full_convolution(a0))

    # 1) Adam phase
    t_adam = now + 0.5 * (deadline - now)
    a_adam, s_adam = _optimize_adam(a0, t_adam, rng)

    now = time.time()
    if now >= deadline:
        # pick the best of the two candidates
        s0 = np.max(_full_convolution(a0))
        if s_adam < s0:
            return a_adam, s_adam
        return a0, s0

    # 2) Sub‑gradient phase
    t_sub = now + 0.3 * (deadline - now)
    a_sub, s_sub = _optimize_subgrad(a_adam, t_sub, rng)

    now = time.time()
    if now >= deadline:
        candidates = [(a0, np.max(_full_convolution(a0))),
                      (a_adam, s_adam),
                      (a_sub, s_sub)]
        best = min(candidates, key=lambda x: x[1])
        return best

    # 3) Greedy refinement
    a_greedy, s_greedy = _greedy_refine(a_sub, deadline, rng)

    # Choose the best among all stages
    candidates = [(a0, np.max(_full_convolution(a0))),
                  (a_adam, s_adam),
                  (a_sub, s_sub),
                  (a_greedy, s_greedy)]
    best_seq, best_score = min(candidates, key=lambda x: x[1])
    return best_seq, best_score

    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin for the final external evaluation
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Retrieve previous construction if available (used for up‑sampling)
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_overall_seq = None
    best_overall_score = np.inf
    attempts = 0
    last_improve_time = start_time

    # Problem size boundaries – larger sizes give more flexibility
    MIN_N = 5000
    MAX_N = 25000

    while time.time() < deadline:
        attempts += 1

        # --------------------------------------------------------------
        # Choose problem size
        # --------------------------------------------------------------
        n = int(rng.integers(MIN_N, MAX_N + 1))

        a0 = _initial_sequence(base_seq, n, rng)

        # --------------------------------------------------------------
        # Allocate a time slice for this attempt (≈ 20 s or the remaining time)
        # --------------------------------------------------------------
        slice_deadline = min(time.time() + 20.0, deadline)

        # --------------------------------------------------------------
        # Optimise
        # --------------------------------------------------------------
        a_opt, score = _optimize_combined_refine(a0, slice_deadline, rng)

        if score < best_overall_score:

    # ------------------------------------------------------------------

    print(f"[final] external score = {external_score:.6f} "

    return final_seq
