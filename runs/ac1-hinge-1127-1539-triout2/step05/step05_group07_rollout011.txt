"""
Improved adversarial step‑function optimizer.

Strategy
--------
1. *Exploratory grid search* over a few simple families of
   non‑negative sequences (uniform, symmetric “edge‑heavy’’ power‑law,
   asymmetric two‑sided power‑law) for several lengths.  The grid is
   tiny (∼10⁴ candidates) and therefore cheap – each score is obtained
   from a single FFT‑based convolution.

2. The best candidate from the grid is refined by a **soft‑max Adam**
   optimisation on the simplex (log‑softmax parametrisation).  The
   loss is a τ‑soft‑max approximation of the ‑∞‑norm of the
   self‑convolution; τ is annealed aggressively so the optimiser
   focuses on the true peak early.  A small amount of Gaussian jitter
   in log‑space helps escape plateaus.

3. The optimiser runs many short restarts (≈8 s each) until the
   overall budget (≈ 1000 s) is exhausted.  The best sequence seen so
   far is stored in the global variable ``height_sequence_1`` and
   returned.

The algorithm respects the 1000 s time limit, keeps all entries
non‑negative, and returns the sequence achieving the lowest
`evaluate_sequence` value found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Full (linear) self‑convolution a*a using a real FFT.
    Returns a 1‑D array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Resize a sequence to a new length by linear interpolation.
    """
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.ceil(pos).astype(int)
    hi = np.clip(hi, 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Soft‑max approximation of the max‑convolution and its gradient
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Returns (g, max_b) where

        g = ∂/∂a   τ‑soft‑max( a * a )

    and max_b is the exact hard maximum of the self‑convolution.
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        # exact sub‑gradient (binary mask on all active maxima)
        mask = (b == max_b).astype(float)
    else:
        # soft‑max weights over the 2·n‑1 entries
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / exps.sum()

    # gradient of soft‑max term:
    #   g_i = 2 * Σ_m mask_m * a_{m-i}
    a_rev = a[::-1]
    conv_len = len(mask) + n - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Refine ``a0`` (already satisfying the ℓ₁‑norm constraint)
    by minimising the soft‑max of its self‑convolution.
    Returns the best vector encountered and its *hard* peak value.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                     # the required ℓ₁‑norm

    # initialise logits so that softmax(logits) * S == a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                         # logits (unconstrained)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.5                                 # relatively aggressive start

    # smoothing schedule – start from the current hard maximum
    tau = max(_full_convolution_fft(a0).max(), 1.0)

    best_a = a0.copy()
    best_peak = _full_convolution_fft(a0).max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- obtain a from logits (simple soft‑max) -----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)             # stabilised exponentials
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ----- convolution and gradient -----
        g, peak = _softmax_gradient(a, tau)

        # keep the best hard peak seen so far
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ----- gradient w.r.t. logits (projected onto the simplex) -----
        mu = np.dot(a, g) / S                 # scalar to keep sum constant
        grad_p = a * (g - mu)                 # shape (n,)

        # optional clipping for stability
        clip_thr = 0.5 * S
        clip_val = np.max(np.abs(grad_p))
        if clip_val > clip_thr:
            grad_p = grad_p * (clip_thr / clip_val)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- schedules -----
        tau = max(tau * 0.995, 1e-9)          # tighten the soft‑max rapidly
        lr *= 0.9999                         # slowly decay the step size

        # occasional jitter in log‑space to escape flat regions
        if it % 400 == 0:
            p += rng.normal(scale=0.015, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Families of candidate sequences used in the grid search
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    """Uniform distribution (will be renormalised later)."""
    return np.ones(n, dtype=float)


def _make_sym_edge(n: int, p: float) -> np.ndarray:
    """
    Symmetric edge‑heavy shape:
        a_i ∝ (min(i, n‑1‑i) + 1)^{-p}
    """
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1
    a = d.astype(float) ** (-p)
    return a


def _make_asym_edge(n: int, p_left: float, p_right: float) -> np.ndarray:
    """
    Asymmetric two‑sided power‑law:
        a_i ∝ (i + 1)^{-p_left} + (n - i)^{-p_right}
    """
    idx = np.arange(n)
    left = (idx + 1.0) ** (-p_left)
    right = (n - idx) ** (-p_right)      # n‑idx = (n‑1‑idx)+1
    a = left + right
    return a


def _normalise_to_sum(a: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale ``a`` so that its ℓ₁‑norm equals ``target_sum``."""
    s = a.sum()
    if s == 0:
        return np.full_like(a, target_sum / a.size)
    return a * (target_sum / s)
