def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]

    # convolution a*a
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]

    # softmax probabilities
    max_b = np.max(b)
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1   # = 3·n-2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b
```

def _exp_gradient_descent(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    beta = 1.0
    max_beta = 5e5
    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        # step size schedule
        eta = 0.5 / np.sqrt(it + 1)
        # exponentiated update
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s > 0:
            a *= (target_sum / s)
        # anneal beta
        if it % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.07, max_beta)
        # optionally add jitter
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum*1e-3, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0:
                a *= (target_sum / s)
        it += 1
    return a

"""
Search for a non‑negative step function that minimises the
``evaluate_sequence`` objective.

The algorithm consists of three stages:

1. **Coarse parametric sweep** – generate many candidate vectors from a
   few low‑dimensional families (power‑law, exponential, mixed) for a
   range of lengths.  Keep the best few (the “elite” set).

2. **Exponentiated‑gradient refinement** – each elite candidate is further
   optimised by minimising a smooth log‑sum‑exp surrogate of the maximal
   convolution entry.  The update is multiplicative (exponentiated
   gradient) and the vector is repeatedly projected onto the ℓ¹‑simplex
   Σa = √(2 n).

3. **Short simulated‑annealing phase** – the refined vectors are fine‑tuned
   with a simple mass‑transfer SA that preserves the ℓ¹‑norm.

All steps respect the supplied time budget (default 1000 s).  The final
sequence is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) surrogate of max(b) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b) where
        b = a*a (full convolution)
        g = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a ⋆ p) ,  p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]

    # ----- convolution a*a -----
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # ----- soft‑max probabilities -----
    max_b = np.max(b)
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)

    # ----- gradient g = 2·(a ⋆ p) -----
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex Σx = s, x≥0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection onto {x_i ≥ 0, Σx_i = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the smooth surrogate
# ----------------------------------------------------------------------


def _exp_gradient_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Multiplicative (exponentiated) gradient descent on the log‑sum‑exp
    approximation of max(b).  After each update the vector is renormalised
    onto the ℓ¹‑simplex Σa = √(2 n).
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    beta = 1.0
    max_beta = 5e5
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # learning‑rate schedule (1/√t)
        eta = 0.5 / np.sqrt(it + 1.0)

        # exponentiated update – clip to avoid overflow
        exp_factor = np.exp(np.clip(-eta * g, -700, 700))
        a *= exp_factor

        # renormalise onto the simplex
        s = a.sum()
        if s > 0.0:
            a *= (target_sum / s)

        # anneal the temperature of the softmax
        if it % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.07, max_beta)

        # occasional small jitter to escape shallow plateaus
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 1e-3, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0.0:
                a *= (target_sum / s)

        it += 1
    return a


# ----------------------------------------------------------------------
# Simple simulated‑annealing refinement
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Random mass‑transfer SA with multiplicative cooling."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # initial temperature loosely based on current objective
    T = best_score * 0.02 + 1e-8
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # move a fraction of mass from i to j
        frac = rng.random() * 0.15
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        accept = False
        if cur_score <= best_score:
            accept = True
        elif T > 0.0 and rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # revert
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # very occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            # re‑normalise to keep the ℓ¹‑sum unchanged
            s = a.sum()
            if s > 0.0:
                a *= (np.sqrt(2.0 * n) / s)
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()

    return best


# ----------------------------------------------------------------------
# Parametric families (power‑law, exponential, mixed)
# ----------------------------------------------------------------------


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    """a_i = (i+shift)^(-p)   (projected onto the ℓ¹‑simplex)."""
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    target_sum = np.sqrt(2.0 * n)
    scale = target_sum / raw.sum()
    a = raw * scale
    a = np.maximum(a, 0.0)
    return a


def _exp_decay_sequence(n: int, c: float) -> np.ndarray:
    """a_i = exp(-c·i)   (projected onto the ℓ¹‑simplex)."""
    idx = np.arange(n, dtype=float)
    raw = np.exp(-c * idx)
    target_sum = np.sqrt(2.0 * n)
    a = raw * (target_sum / raw.sum())
    return a


def _mixed_sequence(n: int, p: float, shift: float, c: float) -> np.ndarray:
    """a_i = (i+shift)^(-p) * exp(-c·i)   (projected)."""
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p) * np.exp(-c * idx)
    target_sum = np.sqrt(2.0 * n)
    a = raw * (target_sum / raw.sum())
    a = np.maximum(a, 0.0)
    return a

        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # --- power‑law grid ---
        for p in p_grid:
            for shift in shift_grid:
                a = _powerlaw_sequence(n, p, shift)
                obj = _max_convolution_fft(a)
                if obj < best_obj:
                    best_obj, best_seq = obj, a.copy()
                    print(f"[{time.time():.1f}] New best (power) n={n} p={p:.3f} "
                          f"shift={shift:.2f} obj={obj:.6f}")
                # keep elite
                if len(elite) < 10 or obj < elite[-1][0]:
                    elite.append((obj, a))
                    elite.sort(key=lambda x: x[0])
                    elite = elite[:10]

                # mixed power‑law / exponential candidates
                for c in c_grid:
                    a_mix = _mixed_sequence(n, p, shift, c)
                    obj_mix = _max_convolution_fft(a_mix)
                    if obj_mix < best_obj:
                        best_obj, best_seq = obj_mix, a_mix.copy()
                        print(f"[{time.time():.1f}] New best (mix) n={n} p={p:.3f} "
                              f"shift={shift:.2f} c={c:.5f} obj={obj_mix:.6f}")
                    if len(elite) < 10 or obj_mix < elite[-1][0]:
                        elite.append((obj_mix, a_mix))
                        elite.sort(key=lambda x: x[0])
                        elite = elite[:10]

        # --- pure exponential decay candidates ---
        for c in c_grid:
            a = _exp_decay_sequence(n, c)
            obj = _max_convolution_fft(a)
            if obj < best_obj:
                best_obj, best_seq = obj, a.copy()
                print(f"[{time.time():.1f}] New best (exp) n={n} c={c:.5f} obj={obj:.6f}")
            if len(elite) < 10 or obj < elite[-1][0]:
                elite.append((obj, a))
                elite.sort(key=lambda x: x[0])
                elite = elite[:10]

        # --- a few random Dirichlet samples for diversity ---
        rnd_end = min(deadline, time.time() + 2.0)
        while time.time() < rnd_end:
            a = rng.dirichlet(np.ones(n)) * target_sum
            obj = _max_convolution_fft(a)
            if obj < best_obj:
                best_obj, best_seq = obj, a.copy()
                print(f"[{time.time():.1f}] New best (rand) n={n} obj={obj:.6f}")
            if len(elite) < 10 or obj < elite[-1][0]:
                elite.append((obj, a))
                elite.sort(key=lambda x: x[0])
                elite = elite[:10]

    # ------------------------------------------------------------------
    # 2. Refinement of elite candidates (exp‑gradient + SA)
    # ------------------------------------------------------------------
    for idx, (seed_score, seed_seq) in enumerate(elite):
        if time.time() >= deadline:
            break
        n = seed_seq.size

        # allocate roughly half of the remaining budget to gradient descent
        time_left = deadline - time.time()
        if time_left <= 0:
            break
        grad_dead = time.time() + min(0.5 * time_left, 30.0)

        a_refined = _exp_gradient_descent(seed_seq, grad_dead)

        # remaining time for SA (if any)
        a_final = _simulated_annealing(a_refined, deadline)

        final_obj = _max_convolution_fft(a_final)
        if final_obj < best_obj:
            best_obj, best_seq = final_obj, a_final.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL BEST after refinement "
                  f"idx={idx} n={n} obj={final_obj:.6f}")

    # ------------------------------------------------------------------
    # 3. Final sanity fallback
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 3000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_obj = _max_convolution_fft(best_seq)

    # Trim negligible trailing entries (purely cosmetic)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # ------------------------------------------------------------------
    # Report using the supplied evaluate_sequence (if available)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective = {best_obj:.6f} "
          f"length = {best_seq.size}")
    return best_seq.tolist()
