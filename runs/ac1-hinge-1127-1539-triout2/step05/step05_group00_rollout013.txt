def _adam_optimize(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    lr = 0.03 * target_sum  # base learning rate; may adjust
    eps = 1e-8
    t = 1
    beta = 0.5
    beta_factor = 1.5
    max_beta = 8192.0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = _project_to_simplex(a, target_sum)
        # anneal beta occasionally
        if t % 300 == 0 and beta < max_beta:
            beta = min(beta * beta_factor, max_beta)
        # jitter occasional
        if t % 500 == 0:
            jitter = np.random.default_rng().normal(scale=target_sum * 0.0005 / (beta + 1.0), size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)
        t += 1
    return a
```

def _targeted_hill_climb(a0, b0, deadline):
    a = a0.copy()
    b = b0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_b = b.copy()
    best_score = b.max()
    cur_score = best_score
    rng = np.random.default_rng()
    # temperature schedule
    T = cur_score * 0.02 + 1e-9
    cooling = 0.9998
    while time.time() < deadline:
        # compute k_max index
        k_max = int(np.argmax(b))
        # compute contributions w_i for i indices that affect b_kmax
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        # vectorized compute w_i
        i_indices = np.arange(lo, hi + 1)
        w = a[i_indices] * a[k_max - i_indices]
        total_w = w.sum()
        if total_w <= 0 or not np.isfinite(total_w):
            # fallback to simple selection
            prob_i = a / a.sum()
            i = rng.choice(n, p=prob_i)
        else:
            prob_i = w / total_w
            i = rng.choice(i_indices, p=prob_i)
        # choose destination j random not equal i
        j = rng.integers(0, n)
        if j == i:
            continue
        # delta: up to a[i]*0.5
        max_delta = a[i] * 0.5
        if max_delta <= 1e-12:
            continue
        delta = rng.random() * max_delta
        # update b_new using analytic
        b_new = b.copy()
        # subtract contributions from source
        b_new[i:i+n] -= 2.0 * delta * a
        # add contributions from dest
        b_new[j:j+n] += 2.0 * delta * a
        # quadratic corrections
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta
        new_score = b_new.max()
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if rng.random() < np.exp(-(new_score - cur_score) / T):
                accept = True
        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                best_b = b.copy()
                print(f"[{time.time():.1f}] targeted hill improvement: {best_score:.6f}")
        T *= cooling
    return best_a, best_b
```

            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final polish score: {final_score:.6f}")

    # Remove trailing near-zero entries and return list
    best_seq = np.asarray(best_seq, dtype=float)

"""
Step‑function optimizer (new version).

The search proceeds in three phases:

1. **Initialisation** – a set of random vectors on the ℓ¹‑simplex
   Σa = sqrt(2 n) is generated (several parametric families).

2. **Smooth optimisation** – each candidate is refined by Adam
   optimisation of a soft‑max approximation of the peak of the
   self‑convolution a*a.  The gradient is obtained analytically via FFTs.

3. **Targeted hill‑climbing** – from the best vector we repeatedly move
   a random amount of mass from a source index *i* that contributes a lot
   to the current maximal convolution entry (k_max) to a destination index
   *j* that contributes little.  The convolution is updated in O(n) time
   using the exact formula, and a simulated‑annealing schedule accepts
   occasional uphill steps.

The algorithm explores a few sequence lengths (here 4000, 8000,
12000) and keeps the best sequence found before a final polishing run.
All operations respect the non‑negativity and ℓ¹‑norm constraints required
by ``evaluate_sequence``.
"""

import time
import numpy as np
from math import sqrt

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a as a NumPy array."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of   (1/β)·log Σ exp(β·b)   where   b = a*a .
    Returns (g, b) with g = 2·(a ⋆ p) and
    pₖ ∝ exp(β·bₖ).  Correlation is performed via FFT.
    """
    n = a.shape[0]

    # b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]                          # reversed a
    conv_len = n + (2 * n - 1) - 1           # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed max
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected Adam optimiser for the soft‑max approximation of the L∞‑norm.
    """
    a = a0.copy()
    n = a.size
    target_sum = sqrt(2.0 * n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    lr0 = 0.02 * target_sum          # base learning rate
    beta = 0.5
    beta_max = 8192.0
    beta_factor = 1.5

    it = 1
    rng = np.random.default_rng()
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # project back onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # anneal beta (smoothness)
        if it % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        # occasional jitter to escape plateaus
        if it % 500 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=n
            )
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return a


# ----------------------------------------------------------------------
# Random initialisers (same families as the previous code)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Generate a random non‑negative vector on the simplex Σa = √(2 n)."""
    target = sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # pure power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )

    else:                               # near uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


# ----------------------------------------------------------------------
# Linear interpolation for up‑sampling a sequence
# ----------------------------------------------------------------------


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a onto new length `new_n` followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    a_new = _project_to_simplex(a_new, sqrt(2.0 * new_n))
    return a_new


# ----------------------------------------------------------------------
# Targeted mass‑transfer hill climbing
# ----------------------------------------------------------------------


def _targeted_hill_climb(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Hill‑climbing that moves mass from indices that most contribute to the
    current maximal convolution entry to indices that contribute little.
    The convolution is updated analytically in O(n) time.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()
    best_b = b.copy()

    rng = np.random.default_rng()

    # start with a relatively warm temperature
    T = cur_score * 0.2 + 1e-12
    cooling = 0.99985

    while time.time() < deadline:
        # location of the current peak
        k_max = int(np.argmax(b))

        # indices that actually participate in b[k_max]
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)

        idx = np.arange(lo, hi + 1)
        w = a[idx] * a[k_max - idx]                     # contribution of each index
        total_w = w.sum()

        if total_w > 0.0 and np.isfinite(total_w):
            probs = w / total_w
            i = rng.choice(idx, p=probs)                # source biased towards large contribution
        else:
            # fallback – pick source proportional to its mass
            probs = a / a.sum()
            i = rng.choice(n, p=probs)

        # random destination (avoid self‑move)
        j = rng.integers(0, n)
        if j == i:
            continue

        # amount of mass to move – up to 30 % of source to keep steps modest
        max_delta = a[i] * 0.3
        if max_delta <= 1e-14:
            continue
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ----- compute updated convolution -----
        b_new = b.copy()
        # contribution from removing `delta` at i
        b_new[i:i + n] -= 2.0 * delta * a
        # contribution from adding `delta` at j
        b_new[j:j + n] += 2.0 * delta * a

        # quadratic correction terms (exact)
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # ----- acceptance (simulated annealing) -----
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if rng.random() < np.exp(-(new_score - cur_score) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                best_b = b.copy()
                print(f"[{time.time():.1f}] hill‑climb improvement: {best_score:.6f}")

        # temperature decay
        T *= cooling

    return best_a, best_b

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects the wall‑clock budget (default 1000 s) and returns
    the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # If a previous construction is available, use it as a seed.
    # ------------------------------------------------------------------
    earlier = globals().get("height_sequence_1")
    if isinstance(earlier, (list, tuple, np.ndarray)):
        a0 = np.asarray(earlier, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, sqrt(2.0 * n0))
        best_seq = a0.copy()
        best_score = _max_convolution_fft(a0)
        print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Length schedule – we try a few different problem sizes.
    # ------------------------------------------------------------------
    lengths = [4000, 8000, 12000]

    for n in lengths:

        target_sum = sqrt(2.0 * n)

        # Decide how much time we can still spend on this length.
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        # reserve a tiny amount for the final polishing phase
        if remaining < 30.0:

        # split the remaining time: half for optimisation, half for hill‑climb
        phase_time = min(remaining * 0.45, 300.0)   # cap per length to keep schedule sane

        # --------------------------------------------------------------
        # Initialise a candidate for this length.

        if best_seq is not None and best_seq.size != n:

    # ------------------------------------------------------------------

        b_full = _full_convolution(a)
        a_final, b_final = _targeted_hill_climb(a, b_full, deadline)

        if final_score < best_score:

            print(f"[{time.time():.1f}] Final polishing improvement: {final_score:.6f}")

    # ------------------------------------------------------------------
    if best_seq is None:
