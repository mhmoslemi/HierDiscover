def compute_gradient_wrt_max(a, k_max):
    n = a.shape[0]
    g = np.zeros_like(a)
    i_low = max(0, k_max - (n-1))
    i_high = min(n-1, k_max)
    idx_i = np.arange(i_low, i_high+1)
    idx_j = k_max - idx_i
    g[idx_i] = a[idx_j]
    return g
```

"""
Proposes a step sequence minimizing the peak of its self-convolution.
The algorithm mixes
* Adam optimisation of a soft‑max approximation of the max‑norm,
* a greedy local mass‑swap that directly attacks the worst convolution entry,
* progressive up‑sampling and many random restarts.
It runs for the allotted time (default 1000 s) and returns the best sequence found.
"""
import time, numpy as np

# functions: project_simplex, full_convolution, smooth_weights, grad_softmax, compute_max_convolution
# optimizer functions: opt_adam, opt_local_swap, combined_opt

def propose_candidate(seed=12345, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 5.0  # safety margin

    best_seq = None
    best_score = np.inf
    attempts = 0

    # optionally load warm start
    if 'height_sequence_1' in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=float)
            if base.ndim == 1 and np.all(base >= 0):
                best_seq = base.copy()
                best_score = np.max(_full_convolution(best_seq))
        except Exception:
            pass

    # main loop: random restarts
    min_trial_time = 12.0
    while time.time() < deadline - min_trial_time:
        attempts += 1
        # decide size n
        if best_seq is None or rng.random() < 0.3:
            n = int(rng.integers(1200, 5000))
            a0 = rng.random(n)
        else:
            # upsample best
            n = min(int(len(best_seq) * rng.uniform(1.0, 1.8)), 8000)
            a0 = _upsample(best_seq, n)

        # add jitter
        jitter = rng.normal(scale=a0.mean()*1e-4, size=n)
        a0 = np.clip(a0 + jitter, 0, None)

        # project onto simplex Σ a = sqrt(2n)
        S = np.sqrt(2.0 * n)
        total = a0.sum()
        if total == 0:
            a0 = np.full(n, 1.0)
            total = a0.sum()
        a0 = a0 / total * S

        sub_deadline = min(time.time() + min_trial_time, deadline)

        # run combined optimizer
        a_opt, sc_opt = _optimize_combined(a0, sub_deadline, rng)

        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[global] attempt={attempts} n={n} new best={best_score:.6f}")

    # final refinement on best_seq
    if best_seq is not None:
        final_deadline = deadline
        a_refined, sc_refined = _optimize_combined(best_seq, final_deadline, rng)
        if sc_refined < best_score:
            best_seq = a_refined
            best_score = sc_refined

    # final check
    if best_seq is None:
        best_seq = np.ones(1, dtype=float) * np.sqrt(2.0)
    result = best_seq.tolist()
    external = evaluate_sequence(result)
    print(f"[final] external score = {external:.6f} (internal best = {best_score:.6f}) length = {len(result)}")

def _optimize_combined(a0, deadline, rng):
    now = time.time()
    remaining = deadline - now
    if remaining <= 0: return a0, np.max(_full_convolution(a0))

    # allocate about half to Adam
    adam_deadline = now + remaining * 0.55
    a_adam, sc_adam = _optimize_adam(a0, adam_deadline, rng)

    if time.time() < deadline:
        a_local, sc_local = _optimize_local_swap(a_adam, deadline, rng)
        if sc_local < sc_adam:
            return a_local, sc_local
    return a_adam, sc_adam

def _optimize_adam(a0, deadline, rng):
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    lr = 0.2
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    t = 0
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    alpha = 0.5  # temperature for softmax

    while time.time() < deadline:
        t += 1
        conv = _full_convolution(a)
        # softmax weights more stable with log-sum-exp
        w = _smooth_weights(conv, alpha)
        # gradient of smoothed max with respect to a (2 * convolution of a and w)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # schedule alpha growing
        alpha = min(alpha * 1.02, 500.0)

    return best_a, best_score

def _optimize_local_swap(a0, deadline, rng):
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    # parameters
    max_no_improve = 2000
    no_improve = 0
    while time.time() < deadline:
        # compute conv and max index
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))
        cur_score = conv[k_max]

        # compute gradient for max index
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        # source: index with largest g where a > tiny
        src_candidates = np.where(a > 0)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(g[src_candidates])]

        # destination: index with smallest g
        dst = np.argmin(g)

        if src == dst:
            # random fallback
            i, j = rng.integers(0, n, size=2)
            src, dst = i, j

        # choose delta
        delta = min(a[src] * rng.uniform(0.01, 0.2), S * rng.uniform(0.0005, 0.005))
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta
        # ensure nonnegative (should be)
        # projection optional but maintain sum automatically; just ensure nonneg
        # Compute new max
        new_score = np.max(_full_convolution(a_new))

        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # occasionally accept a worsening move to escape
            if rng.random() < 0.02:
                a = a_new
                cur_score = new_score
                no_improve = 0
            else:
                no_improve += 1

        if no_improve > max_no_improve:
            # random perturbation
            i, j = rng.integers(0, n, size=2)
            if i != j:
                delta2 = a[i] * rng.uniform(0.01, 0.05)
                a[i] -= delta2
                a[j] += delta2
                a = _project_simplex(a, S)
            no_improve = 0

    return best_a, best_score

"""
Propose a non‑negative step sequence that minimises the peak of its
self‑convolution (the value returned by ``evaluate_sequence``).

Algorithm
---------
* Multiple short trials with random initialisations (or up‑sampled
  best‑so‑far).  Each trial runs a three‑phase optimiser:
    1. Adam on a smooth soft‑max approximation of the max‑norm.
    2. Projected sub‑gradient descent that directly attacks the current
       worst convolution entry.
    3. A greedy “mass‑swap’’ local search that repeatedly moves a small
       amount of mass from a source index (large contribution to the worst
       entry) to a destination index (small contribution).  The swap is
       kept only if it reduces the peak.
* The best sequence found is refined for the remaining budget.
* All vectors are kept on the ℓ₁‑simplex Σ aᵢ = √(2·n) (the normalisation
  used by ``evaluate_sequence``).  Non‑negativity is enforced by a
  Euclidean simplex projection.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full linear convolution a * a (length 2·n‑1)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Soft‑max weights (smooth approximation of max)
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """w_i = exp(alpha·conv_i) / Σ_j exp(alpha·conv_j) (stable)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Gradient of the smooth max‑norm (used in Adam)
# ----------------------------------------------------------------------
def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Computes 2·(a * w_rev) restricted to the first n entries,
    i.e. the gradient of Σ_i w_i·(a*a)_i w.r.t. a.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = 3 * n - 2                     # length of a ⊛ w_rev
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)            # selects the needed slice
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Linear up‑sampling (interpolation) helper
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a to length new_n."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=float)
    return np.interp(x_new, x_old, a)


# ----------------------------------------------------------------------
# Adam optimiser on a smooth max‑norm approximation
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray, deadline: float, rng: np.random.Generator):
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.2
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    alpha = 0.5               # temperature – starts smooth
    while time.time() < deadline:
        t += 1
        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Simplex projection (maintains Σa = S, a ≥ 0)
        a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # temperature schedule – sharpen the soft‑max
        alpha = min(alpha * 1.02, 500.0)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (direct max‑norm)
# ----------------------------------------------------------------------
def _optimize_subgrad(a0: np.ndarray, deadline: float, rng: np.random.Generator):
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.5   # initial step‑size constant
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional random jitter to escape plateaus
        if rng.random() < 0.03:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.02
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = np.max(_full_convolution(cand))
                if cand_score < cur_score or rng.random() < 0.2:
                    a = cand
                    cur_score = cand_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy local “mass‑swap’’ optimiser (hits the worst entry directly)
# ----------------------------------------------------------------------
def _optimize_local_swap(a0: np.ndarray, deadline: float, rng: np.random.Generator):
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    no_improve = 0
    max_no_improve = 3000

    while time.time() < deadline:
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))
        cur_score = conv[k_max]

        # gradient of the active entry (g_i = a_{k_max-i})
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        # source = index with largest contribution (and positive mass)
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(g[src_candidates])]

        # destination = index with smallest contribution
        dst = int(np.argmin(g))

        if src == dst:
            # fallback to a random pair
            src, dst = rng.integers(0, n, size=2)

        # propose a modest mass transfer
        delta = min(a[src] * rng.uniform(0.01, 0.15),
                    S * rng.uniform(5e-4, 5e-3))
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        new_score = np.max(_full_convolution(a_new))

        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            no_improve = 0
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
        else:
            # occasional uphill move (small probability) to escape local minima
            if rng.random() < 0.02:
                a = a_new
                cur_score = new_score
                no_improve = 0
            else:
                no_improve += 1

        if no_improve > max_no_improve:
            # random perturbation to diversify
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta2 = a[i] * rng.uniform(0.01, 0.05)
                a[i] -= delta2
                a[j] += delta2
                a = _project_simplex(a, S)
            no_improve = 0

    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimiser: Adam → sub‑gradient → local swap
# ----------------------------------------------------------------------
def _optimize_combined(a0: np.ndarray, deadline: float, rng: np.random.Generator):
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    # 50 % Adam, 30 % sub‑gradient, rest local swap
    t_adam = now + remaining * 0.5
    a_adam, sc_adam = _optimize_adam(a0, t_adam, rng)

    now2 = time.time()
    if now2 >= deadline:
        return a_adam, sc_adam

    t_sub = now2 + (deadline - now2) * 0.3
    a_sub, sc_sub = _optimize_subgrad(a_adam, t_sub, rng)

    now3 = time.time()
    if now3 >= deadline:
        # return best of the two phases
        if sc_sub < sc_adam:
            return a_sub, sc_sub
        return a_adam, sc_adam

    # remaining time → greedy local swaps
    a_sw, sc_sw = _optimize_local_swap(a_sub, deadline, rng)

    # pick the best among the three
    best_a, best_sc = a_adam, sc_adam
    if sc_sub < best_sc:
        best_a, best_sc = a_sub, sc_sub
    if sc_sw < best_sc:
        best_a, best_sc = a_sw, sc_sw
    return best_a, best_sc

                a0 = _upsample(best_seq, n)
            else:
                n = len(best_seq)
                a0 = best_seq.copy()

            # small jitter to avoid exact repetition
            jitter = rng.normal(scale=a0.mean() * 1e-4, size=n)
            a0 = np.clip(a0 + jitter, 0.0, None)

        # ------------------------------------------------------------------
        # Project onto the ℓ₁‑simplex Σ a = √(2·n)
        # ------------------------------------------------------------------
        S = np.sqrt(2.0 * n)
        total = a0.sum()
        if total == 0.0:
            a0 = np.full(n, 1.0)
            total = a0.sum()
        a0 = a0 / total * S

        # ------------------------------------------------------------------
        # Run the combined optimiser for the allotted slice
        # ------------------------------------------------------------------
        slice_deadline = min(time.time() + min_trial_time, deadline)
        a_opt, sc_opt = _optimize_combined(a0, slice_deadline, rng)

        # ------------------------------------------------------------------
        # Update global best
        # ------------------------------------------------------------------
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            last_improve = time.time()
            print(f"[global] attempt={attempts} n={n} new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final refinement on the best sequence with whatever time is left
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_refined, sc_refined = _optimize_combined(best_seq, deadline, rng)
        if sc_refined < best_score:
            best_seq = a_refined
            best_score = sc_refined

    # ------------------------------------------------------------------
    # Prepare output
    # ------------------------------------------------------------------
    if best_seq is None:
        # fall back to a trivial feasible vector
        n0 = 1
        best_seq = np.full(n0, np.sqrt(2.0 * n0), dtype=float)

    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f} "
          f"(internal best = {best_score:.6f}) length = {len(final_seq)}")
    return final_seq
