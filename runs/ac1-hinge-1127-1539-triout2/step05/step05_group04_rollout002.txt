"""
Step‑function optimizer – hierarchical length scaling + Adam‑based smooth‑max
descent + conditional‑gradient (sub‑gradient) polishing + mass‑balancing +
short simulated‑annealing.

The algorithm proceeds through a list of increasing lengths (2000 → 20000).
For each length it

* builds an initial point (either by up‑sampling the best sequence from a
  smaller size or by a random draw on the ℓ¹‑simplex),
* performs many short Adam‑descent runs on a smooth approximation of the
  infinity‑norm of the self‑convolution,
* keeps the best candidate encountered,
* refines it with a long Adam run, a sub‑gradient descent on the true max,
  a cheap greedy mass‑balance move, and finally a brief simulated‑annealing
  polishing phase.

All updates are projected back onto the simplex {x≥0, Σx = √(2 n)}.
The code respects a wall‑clock budget (default 1000 s) and prints
progress information.  The returned list is the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0, Σ x_i = s }.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1, dtype=float)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


def _scale_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Faster, approximate projection: clip negatives, then renormalise to sum s.
    Used in the inner loops where many iterations are performed.
    """
    v = np.maximum(v, 0.0)
    total = v.sum()
    if total > 0.0:
        v = v * (s / total)
    else:
        v = np.full_like(v, s / v.size)
    return v


def _max_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n−1) via FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution (the official objective)."""
    return float(_max_convolution_fft(a).max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) gradient – Phase 1
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient ∇_a  ( 1/β·log Σ exp(β·b) ) where b = a*a.
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = float(np.max(b))
    exps = np.exp(beta * (b - max_b))          # shift for stability
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1            # = 3 n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _smooth_max_descent_adam(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Adam‑based projected descent on the smooth‑max approximation.
    The (inverse) temperature β is annealed exponentially.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 4096.0
    beta_factor = 1.4

    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    step_scale = target_sum * 0.20      # base step size

    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = step_scale / (np.sqrt(v_hat) + eps) / (beta + 1.0)
        a -= lr * m_hat

        # fast projection (clip + renormalise)
        a = _scale_to_simplex(a, target_sum)

        # small jitter every few hundred iterations
        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = _scale_to_simplex(a, target_sum)

        # anneal β
        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Exact max‑index sub‑gradient – Phase 2
# ----------------------------------------------------------------------


def _max_index_subgradient(a: np.ndarray):
    """
    Sub‑gradient of the true max‑norm of the convolution.
    Returns (g, k) where k = argmax convolution.
    """
    n = a.shape[0]
    conv = _max_convolution_fft(a)
    k = int(np.argmax(conv))

    g = np.zeros_like(a)
    if k < n:                         # left half
        g[: k + 1] = 2.0 * a[k::-1]
    else:                             # right half
        d = k - (n - 1)
        g[d:] = 2.0 * a[: n - d]
    return g, k


def _subgradient_descent_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Projected sub‑gradient descent on the true max‑norm."""
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        g, _ = _max_index_subgradient(a)

        # diminishing step size η_t = α / √t
        step = 0.8 * target_sum / (np.sqrt(it + 1) * np.linalg.norm(g) + 1e-12)
        a -= step * g

        a = _scale_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter
            a = _scale_to_simplex(a, target_sum)

        it += 1
    return a


# ----------------------------------------------------------------------
# Greedy mass‑balance refinement
# ----------------------------------------------------------------------


def _mass_balance_refine(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Move a small fraction of the heaviest contributor of the current
    maximal convolution entry to a light position, repeatedly."""
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sum(a)               # should already be √(2 n)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _max_convolution_fft(a)
        k = int(np.argmax(conv))

        # contributions to entry k
        idx = np.arange(n)
        other = k - idx
        mask = (other >= 0) & (other < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[other[mask]]

        i_high = int(np.argmax(contrib))

        # pick a relatively empty slot (20 % smallest masses)
        low_candidates = np.where(a <= np.percentile(a, 20))[0]
        if low_candidates.size == 0:
            i_low = int(np.argmin(a))
        else:
            i_low = int(rng.choice(low_candidates))

        delta = a[i_high] * 0.03
        if delta < 1e-15:
            break

        a[i_high] -= delta
        a[i_low]  += delta

        a = _scale_to_simplex(a, target_sum)

        if it % 100 == 0 and rng.random() < 0.02:
            jitter = rng.normal(scale=target_sum * 1e-3, size=n)
            a += jitter
            a = _scale_to_simplex(a, target_sum)

        it += 1

    return a


# ----------------------------------------------------------------------
# Random initialisation helpers
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative point on the simplex Σa = √(2 n).
    Various families (power‑law, exponential, Dirichlet, piecewise) are
    sampled to increase diversity.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                                 # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                               # pure exponential
        lam = rng.exponential(scale=0.02)
        a = np.exp(-lam * idx)

    elif family == 2:                               # power × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.02)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                               # Dirichlet (uniform)
        a = rng.dirichlet(alpha=np.ones(n))

    elif family == 4:                               # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                             np.exp(-lam2 * np.arange(b, n))])

    else:                                           # almost uniform with jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


# ----------------------------------------------------------------------
# Very light Simulated Annealing (final polishing)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Light SA that moves a random fraction of mass between two sites."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    cur_score = _score(a)
    best_score = cur_score

    T = cur_score * 0.02 + 1e-9
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-16:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        frac = rng.random() * 0.15
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta

        new_score = _score(a)
        dE = new_score - cur_score
        accept = dE <= 0.0 or (T > 0.0 and rng.random() < np.exp(-dE / T))

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling

    return best

            # Upsample previous best sequence linearly
            old_n = best_seq.size
            x_old = np.linspace(0.0, 1.0, old_n)
            x_new = np.linspace(0.0, 1.0, n)
            a_init = np.interp(x_new, x_old, best_seq)
            a_init = np.maximum(a_init, 0.0)
            a_init = _project_to_simplex(a_init, target_sum)
        else:
            a_init = _random_initial_sequence(n, rng)

        # ------------------------------------------------------------------
        # Phase A: many short Adam runs from random restarts
        # ------------------------------------------------------------------
        phase_a_dead = min(deadline, time.time() + time_for_this * 0.4)
        while time.time() < phase_a_dead:
            a_start = _random_initial_sequence(n, rng)
            # each short run gets ~2 s (adjusted to remaining budget)
            sub_dead = min(deadline, time.time() + 2.0)
            a_mid = _smooth_max_descent_adam(a_start, sub_dead)
            sc = _score(a_mid)
            if sc < best_score:
                best_score = sc
                best_seq = a_mid.copy()
                print(f"[{time.time():.1f}] New global best (rand) n={n} score={sc:.6f}")

        # ------------------------------------------------------------------
        # Phase B: deeper refinement of the best sequence of this size
        # ------------------------------------------------------------------
        if best_seq is not None and best_seq.size == n:
            # a) longer Adam smooth‑max descent
            sub_dead = min(deadline, time.time() + time_for_this * 0.3)
            a_ref = _smooth_max_descent_adam(best_seq, sub_dead)
            sc = _score(a_ref)
            if sc < best_score:
                best_score = sc
                best_seq = a_ref.copy()
                print(f"[{time.time():.1f}] Refined (Adam) n={n} score={sc:.6f}")

            # b) exact sub‑gradient descent on the true max‑norm
            sub_dead = min(deadline, time.time() + 30.0)
            a_ref2 = _subgradient_descent_max(best_seq, sub_dead)
            sc = _score(a_ref2)
            if sc < best_score:
                best_score = sc
                best_seq = a_ref2.copy()
                print(f"[{time.time():.1f}] Sub‑gradient improvement n={n} score={sc:.6f}")

            # c) cheap greedy mass‑balance move
            sub_dead = min(deadline, time.time() + 20.0)
            a_ref3 = _mass_balance_refine(best_seq, sub_dead)
            sc = _score(a_ref3)
            if sc < best_score:
                best_score = sc
                best_seq = a_ref3.copy()
                print(f"[{time.time():.1f}] Mass‑balance refinement n={n} score={sc:.6f}")

        # ------------------------------------------------------------------
        # Small time left for this length – move on to the next one
        # ------------------------------------------------------------------
        # (nothing else; the loop will continue)

    # ------------------------------------------------------------------
    # Final simulated‑annealing polishing with whatever time is left
    # ------------------------------------------------------------------
    if best_seq is not None:
        left = deadline - time.time()
        if left > 5.0:
            a_final = _simulated_annealing(best_seq, deadline)
            sc = _score(a_final)
            if sc < best_score:
                best_score = sc
                best_seq = a_final.copy()
                print(f"[{time.time():.1f}] Final SA improvement: score={sc:.6f}")

    # ------------------------------------------------------------------
    # Clean up trailing near‑zero entries and report the official evaluation
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution of length 1000
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
