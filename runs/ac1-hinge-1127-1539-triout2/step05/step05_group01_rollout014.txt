"""
Propose a non‑negative step‑function (a list of floats) that minimises
the provided ``evaluate_sequence`` objective.

Algorithm
---------

1. **Deterministic families** – generate a few thousand candidates from
   three analytically‑motivated families:
   * plain power‑law   a_i = (i+shift)^(-p)
   * symmetric power‑law a_i = (i+shift)^(-p) + (n‑1‑i+shift)^(-p)
   * edge‑power‑law    a_i = min(i,n‑1‑i+shift)^(-p)

   The parameters (length, exponent *p*, shift) are swept on a coarse
   grid, the best few candidates (the “elite”) are retained.

2. **Projected gradient descent on a smooth max** – each elite candidate
   is refined using the log‑sum‑exp surrogate of the maximal
   convolution entry.  The smoothing parameter β is annealed up to
   1e9 so the gradient eventually aligns with the true max.  The update
   uses momentum and projects back onto the ℓ¹‑simplex
   Σa_i = √(2 n).

3. **Simulated annealing** – the refined vectors are further tweaked
   by random mass‑transfer moves (up to 30 % of the source mass) with
   a slowly decreasing temperature.  The best vector seen during SA is
   kept.

4. **Final fine‑tuning** – the current global best is once more run
   through the gradient‑descent routine to squeeze out any remaining
   improvement.

All steps respect a user‑supplied time budget (default 1000 s) and
periodically check the deadline.  The routine returns the best sequence
found as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """
    Maximum entry of the linear convolution a * a (full length 2·n‑1).
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a (length 2·n‑1).
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex Σx = s, x ≥ 0  (Duchi et al. 2008)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) surrogate of max(b) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b) where
        b = a*a (full convolution)
        g = ∇_a  ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a ⋆ p) ,   p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]

    # ----- convolution a*a -----
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # ----- soft‑max probabilities -----
    max_b = np.max(b)
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)

    # ----- gradient g = 2·(a ⋆ p) -----
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Projected gradient descent on the smoothed maximum
# ----------------------------------------------------------------------


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Annealed projected gradient descent on the log‑sum‑exp smooth max.
    The smoothing parameter β is increased up to 1e9, which effectively
    targets the true max.  The routine stops when the global deadline is
    reached.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 1.0
    max_beta = 1e9
    momentum = np.zeros_like(a)
    rho = 0.9                     # momentum decay
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum step
        momentum = rho * momentum + (1.0 - rho) * g
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            # simple normalised step size
            lr = 0.5 * target_sum / (norm * (beta + 1.0))
            a -= lr * momentum

        # keep on the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow plateaus
        if it % 200 == 0:
            jitter = rng.normal(
                scale=target_sum * 5e-4 / (beta + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # exponential annealing of β (more aggressive than the baseline)
        if it % 100 == 0 and beta < max_beta:
            beta = min(beta * 1.12, max_beta)

        it += 1

    return a


# ----------------------------------------------------------------------
# Simple simulated‑annealing refinement
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Random mass‑transfer SA with a relatively slow cooling schedule.
    The temperature is initialised from the current objective; each
    iteration tries to move a random fraction (≤30 %) of the mass from
    one index to another.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_fft(best)

    T = best_score * 0.05 + 1e-8      # a hotter start than the baseline
    cooling = 0.9998
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # move up to 30 % of mass from i to j
        frac = rng.random() * 0.30
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        if cur_score <= best_score or (T > 0.0 and rng.random() < np.exp(-dE / T)):
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # reject – roll back
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # very occasional global jitter to keep diversity
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()

    return best


# ----------------------------------------------------------------------
# Parametric families (power‑law / symmetric / edge‑power‑law)
# ----------------------------------------------------------------------


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    target = np.sqrt(2.0 * n)
    a = raw * (target / raw.sum())
    return np.maximum(a, 0.0)


def _symmetric_powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    left = (idx + shift) ** (-p)
    right = (n - 1 - idx + shift) ** (-p)
    raw = left + right
    target = np.sqrt(2.0 * n)
    a = raw * (target / raw.sum())
    return np.maximum(a, 0.0)


def _edge_powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    """
    a_i = ( min(i, n‑1‑i) + shift )^(‑p), i = 0,…,n‑1.
    U‑shaped (large at the edges, small in the centre).  The vector is
    projected onto the ℓ¹‑simplex Σa = √(2 n).
    """
    idx = np.arange(n, dtype=float)
    dist = np.minimum(idx, n - 1 - idx) + shift
    raw = dist ** (-p)
    target = np.sqrt(2.0 * n)
    a = raw * (target / raw.sum())
    return np.maximum(a, 0.0)

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the target
    ``evaluate_sequence`` objective.  The routine respects the allotted
    time budget, explores several parametric families, refines the most
    promising candidates with gradient descent + SA, and finally returns
    the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # small safety margin so we do not overshoot the allotted time
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_obj = np.inf
    elite: list[tuple[float, np.ndarray]] = []      # (objective, vector)
    max_elite = 10

    # ------------------------------------------------------------------
    # 1. Deterministic parametric sweep (including the new edge‑law family)
    # ------------------------------------------------------------------
    lengths = [2500, 3000, 3500, 4000, 4500]
    p_grid = np.linspace(0.45, 0.85, 15)           # exponent grid
    shift_grid = np.linspace(0.5, 5.0, 10)         # shift grid
    generators = [_powerlaw_sequence,
                  _symmetric_powerlaw_sequence,
                  _edge_powerlaw_sequence]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Exploring length {n}")

        for p in p_grid:
            for shift in shift_grid:
                for gen in generators:
                    a = gen(n, p, shift)
                    score = _max_convolution_fft(a)
                    if score < best_obj:
                        best_obj = score
                        best_seq = a.copy()
                        print(f"[{time.time():.1f}] New global best (grid) "
                              f"{gen.__name__} n={n} p={p:.3f} shift={shift:.2f}  obj={score:.6f}")

                    # keep a short list of elite vectors
                    if len(elite) < max_elite or score < elite[-1][0]:
                        elite.append((score, a))
                        elite.sort(key=lambda x: x[0])
                        elite = elite[:max_elite]

                # a little random sampling for diversity (few per n)
                if time.time() >= deadline:
                    break
                if rng.random() < 0.02:   # roughly 2 % chance per loop
                    a_rand = rng.random(n)
                    a_rand = a_rand / a_rand.sum() * np.sqrt(2.0 * n)
                    score = _max_convolution_fft(a_rand)
                    if score < best_obj:
                        best_obj = score
                        best_seq = a_rand.copy()
                        print(f"[{time.time():.1f}] New global best (rand) n={n}  obj={score:.6f}")
                    if len(elite) < max_elite or score < elite[-1][0]:
                        elite.append((score, a_rand))
                        elite.sort(key=lambda x: x[0])
                        elite = elite[:max_elite]

    # ------------------------------------------------------------------
    # 2. Include the previously discovered construction if available
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            # re‑normalise to the required ℓ¹‑norm (just in case)
            target = np.sqrt(2.0 * prev.size)
            prev = prev * (target / prev.sum())
            score = _max_convolution_fft(prev)
            if score < best_obj:
                best_obj = score
                best_seq = prev.copy()
                print(f"[{time.time():.1f}] New global best (previous) obj={score:.6f}")
            if len(elite) < max_elite or score < elite[-1][0]:
                elite.append((score, prev))
                elite.sort(key=lambda x: x[0])
                elite = elite[:max_elite]
    except NameError:
        pass  # height_sequence_1 does not exist – ignore

    # ------------------------------------------------------------------
    # 3. Refinement of elite candidates (gradient descent + SA)
    # ------------------------------------------------------------------
    for _, seed_seq in elite:
        if time.time() >= deadline:
            break
        n = seed_seq.size
        # allocate up to 30 s for gradient descent (or whatever is left)
        gd_deadline = min(deadline, time.time() + 30.0)
        a_gd = _gradient_descent_opt(seed_seq, gd_deadline)

        # if time permits, run a second short GD pass (helps when the first
        # pass stopped early because of the global deadline)
        if time.time() < deadline:
            a_gd = _gradient_descent_opt(a_gd, deadline)

        # simulated annealing on the GD‑refined vector
        a_sa = _simulated_annealing(a_gd, deadline)

        obj = _max_convolution_fft(a_sa)
        if obj < best_obj:
            best_obj = obj
            best_seq = a_sa.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL BEST (refined) n={n}  obj={obj:.6f}")

    # ------------------------------------------------------------------
    # 4. Final fine‑tuning on the global best (if we still have time)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        best_seq = _gradient_descent_opt(best_seq, deadline)
        obj = _max_convolution_fft(best_seq)
        if obj < best_obj:
            best_obj = obj
            print(f"[{time.time():.1f}] Final gradient tweak improved obj={obj:.6f}")

    # ------------------------------------------------------------------
    # 5. Clean‑up and final evaluation
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 3000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)

    # Trim negligible trailing entries for a tidy output
    best_seq = np.maximum(best_seq, 0.0)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.1f}] Evaluation failed: {exc}")

    print(f"[{time.time():.1f}] Finished. Best objective = {best_obj:.6f}  length = {best_seq.size}")
    return best_seq.tolist()
