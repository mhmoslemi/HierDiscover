def propose_candidate(seed=42, budget_s=1000, **kwargs):

    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5

    best_f = None
    best_M = np.inf
    attempt = 0

    while time.time() < end_time:
        attempt += 1
        # Choose length
        n = rng.integers(400, 1500)   # moderate
        S = np.sqrt(2.0 * n)

        # Initial vector: random Dirichlet with random concentration alpha
        alpha = rng.uniform(0.2, 5.0)
        a = rng.gamma(alpha, 1.0, size=n)
        a = a / a.sum() * S

        # optionally edge heavy pattern
        if rng.random() < 0.3:
            idx = np.arange(n)
            d = np.minimum(idx, n-1-idx) + 1
            p = rng.uniform(0.5, 2.0)
            a = (d.astype(float) ** (-p))
            a = a / a.sum() * S

        # compute M
        b = np.convolve(a, a)
        M = b.max()
        # some iterations
        t = 0.2
        for it in range(12):
            # build conv matrix
            offsets = np.arange(n)
            A = sp.diags([a] * n, offsets, shape=(2*n-1, n), format='csr')
            # solve LP
            c = -np.ones(n)
            ub = np.full(2*n-1, M)
            res = opt.linprog(c, A_ub=A, b_ub=ub, bounds=(0, None), method='highs')
            if not res.success:
                break
            g = res.x
            sum_g = g.sum()
            if sum_g <= 0:
                break
            # scale to maintain sum S
            scale = S / sum_g
            g = g * scale

            # mixing
            a = (1 - t) * a + t * g
            # recompute M
            b = np.convolve(a, a)
            new_M = b.max()
            if new_M >= M * 0.9999:
                # no improvement, break
                break
            M = new_M

            # optionally reduce t gradually
            t *= 0.9

        # final evaluation
        score = evaluate_sequence(a.tolist())
        if score < best_M:
            best_M = score
            best_f = a.copy()
            print(f"[restart] attempt={attempt} n={n} score={score:.6f}")

    if best_f is None:
        best_f = np.ones(1)

    final_seq = best_f.tolist()
    globals()['height_sequence_1'] = final_seq
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}")

"""
adversarial step‑function optimizer – LP refinement + Adam fine‑tuning

The optimizer works in two stages.

1.  LP refinement (core of the original “linear‑program” idea):
    - keep the L1‑norm fixed to S = sqrt(2·n) (the scale does not affect the
      evaluation score because the objective is homogeneous of degree 0);
    - given a current candidate f, compute M = max(f*f);
    - solve a linear program that *maximises* the total mass of a new
      vector g under the *same* peak constraint  f⋆g ≤ M  (the convolution
      is linear in g).  The LP therefore finds a complementary direction that
      carries as much weight as possible without increasing the peak.
    - rescale g so that its sum equals S and mix it with the current f.
      Repeating this step drives the peak M downward.

2.  Short Adam run (soft‑max approximation of the ∞‑norm)
    to polish the solution obtained by the LP stage.

Both stages respect the 1000 s budget, keep the best sequence found,
store it in the global variable ``height_sequence_1`` and finally return it.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Convolution helpers
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full mode)."""
    return np.convolve(a, a)          # O(n log n) for typical sizes is fine


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return _full_convolution(a).max()


# ----------------------------------------------------------------------
# Linear‑program refinement
# ----------------------------------------------------------------------
def _lp_refine(a: np.ndarray,
               target_sum: float,
               max_iters: int = 15,
               t_start: float = 0.4) -> np.ndarray:
    """
    Perform a few LP‑based refinement steps.
    Returns a vector with the same L1‑norm as ``target_sum`` that (usually)
    has a lower convolution peak.
    """
    n = a.size
    t = t_start

    for it in range(max_iters):
        # current peak
        M = _max_convolution(a)

        # build the convolution matrix A (size (2n‑1) × n)
        # column j is the vector a shifted by j positions
        offsets = np.arange(n)                     # 0,…,n-1
        # each diagonal receives a copy of the current a (the LP solver copies the data)
        A = sp.diags([a] * n, offsets, shape=(2 * n - 1, n), format='csr')

        # linear program:
        #   maximise   sum(g)
        #   s.t.       A @ g <= M   (each entry of f⋆g is bounded by the current peak)
        #              g >= 0
        c = -np.ones(n)                           # minimise –sum(g)  == maximise sum(g)
        ub = np.full(2 * n - 1, M)

        res = opt.linprog(c,
                          A_ub=A,
                          b_ub=ub,
                          bounds=(0, None),
                          method='highs',
                          options={'presolve': True, 'time_limit': 5.0})

        if not res.success or res.x is None:
            break

        g = res.x
        if g.sum() == 0.0:
            break

        # rescale so that the L1‑norm stays exactly target_sum
        g = g * (target_sum / g.sum())

        # convex combination – the mixing factor decays geometrically
        a = (1.0 - t) * a + t * g
        t *= 0.85                                 # slowly reduce the step size

    return a


# ----------------------------------------------------------------------
# Adam optimiser (soft‑max approximation of the ℓ∞‑norm)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """FFT‑based full convolution a*a (real inputs)."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _softmax_gradient(a: np.ndarray, tau: float):
    """τ‑soft‑max gradient of max(a*a). Returns (grad, hard_max)."""
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        w = exps / exps.sum()
        mask = w

    # gradient = 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = mask.shape[0] + n - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]
    return g, max_b


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   max_iters: int = 5000) -> np.ndarray:
    """
    Adam optimisation on the simplex (log‑softmax parametrisation).
    Stops when time runs out.
    Returns the best vector encountered.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    # initialise logits from a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)

    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.3

    # initial temperature based on the current peak
    tau = max(_full_convolution_fft(a0).max(), 1.0)

    best_a = a0.copy()
    best_peak = _max_convolution(a0)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # reconstruct a from logits
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # gradient of the soft‑max approximation
        g, peak = _softmax_gradient(a, tau)

        # keep the best hard peak seen so far
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # gradient w.r.t. logits (projected onto simplex)
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # schedules (slow cooling / decay)
        tau *= 0.9996
        lr *= 0.9999

        if it % 800 == 0:
            p += rng.normal(scale=0.02, size=n)

    return best_a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimizes ``evaluate_sequence``.
    The method consists of many short restarts, each performing a few LP‑refinement
    steps followed by a brief Adam optimisation.  The best candidate seen within
    the time budget is returned.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0      # safety margin

    # try to reuse a previously discovered construction
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    attempt = 0

    # ------------------------------------------------------------------
    # Helper to create a feasible start vector of length n
    # ------------------------------------------------------------------
    def _initial_vector(n):
        S = np.sqrt(2.0 * n)
        # a random Dirichlet (concentration parameter drawn uniformly)
        alpha = rng.uniform(0.3, 4.0)
        a = rng.gamma(alpha, 1.0, size=n)
        a = a / a.sum() * S

        # occasionally bias towards the edges (known good heuristic)
        if rng.random() < 0.4:
            idx = np.arange(n)
            d = np.minimum(idx, n - 1 - idx) + 1          # distance to nearest edge (1‑based)
            p = rng.uniform(0.8, 2.5)                     # strength of edge decay
            a = (d.astype(float) ** (-p))
            a = a / a.sum() * S

        # optionally blend with the global best if it exists
        if base_seq is not None and rng.random() < 0.15:
            # resize the stored best to the new length by linear interpolation
            old = base_seq
            pos = np.linspace(0, old.size - 1, n)
            lo = np.floor(pos).astype(int)
            hi = np.ceil(pos).astype(int)
            hi = np.clip(hi, 0, old.size - 1)
            w = pos - lo
            a = (1.0 - w) * old[lo] + w * old[hi]
            a = np.clip(a, 0.0, None)
            a = a / a.sum() * S
        return a

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < end_time:
        attempt += 1

        # pick a length – moderate sizes are cheap for the LP,
        # larger ones give a finer discretisation.
        if rng.random() < 0.7:
            n = rng.integers(600, 1800)
        else:
            n = rng.integers(1800, 3000)

        a = _initial_vector(n)
        target_sum = np.sqrt(2.0 * n)      # the scale required by the objective

        # --------------------------------------------------------------
        # 1) LP refinement (fast, strong reduction of the peak)
        # --------------------------------------------------------------
        a = _lp_refine(a, target_sum,
                       max_iters=12,
                       t_start=0.45)

        # --------------------------------------------------------------
        # 2) short Adam fine‑tuning (adds a smooth refinement)
        # --------------------------------------------------------------
        per_restart_budget = min(8.0, (end_time - time.time()) * 0.5)
        if per_restart_budget > 1.0:
            a = _adam_optimize(a,
                               deadline=time.time() + per_restart_budget,
                               rng=rng,
                               max_iters=3000)

        # --------------------------------------------------------------
        # evaluate
        # --------------------------------------------------------------
        candidate = a.tolist()
        try:
            score = evaluate_sequence(candidate)
        except Exception:
            continue
        if score < best_score:
            best_score = score
            best_seq = candidate.copy()
            print(f"[restart] attempt={attempt} n={n}  new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check and output
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0, 1.0]                     # fallback (never happens)
    globals()['height_sequence_1'] = best_seq
    final_score = evaluate_sequence(best_seq)
    print(f"[final] external score = {final_score:.6f}  length = {len(best_seq)}")
    return best_seq
