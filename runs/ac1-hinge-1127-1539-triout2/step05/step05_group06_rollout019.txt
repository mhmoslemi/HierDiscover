def _simulated_annealing(a_init, deadline, rng):
    a = a_init.copy()
    n = a.shape[0]
    # ensure sum 1
    a = a / a.sum()
    cur_max = np.max(_full_convolution(a))
    best_a = a.copy()
    best_max = cur_max

    # set initial temperature relative to cur_max
    T = 0.1 * cur_max  # maybe 10% of current max
    decay = 0.99998
    step_factor = 0.2  # proportion of mass moved

    it = 0
    last_improve_time = time.time()
    while time.time() < deadline:
        it += 1
        # random move
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        # move a fraction of a[i] to j
        delta = a[i] * rng.random() * step_factor
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_max = np.max(_full_convolution(a_new))
        dE = new_max - cur_max
        accept = False
        if dE <= 0:
            accept = True
        else:
            # acceptance prob
            if rng.random() < np.exp(-dE / T):
                accept = True
        if accept:
            a = a_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
                last_improve_time = time.time()
                print(f"[SA] iter={it} new best max={best_max:.6f}")

        # temperature decay
        T = max(T * decay, 1e-8)
        # occasional larger step factor adaptation
        if rng.random() < 0.005:
            # random jitter across 3 indices
            idx = rng.choice(n, size=3, replace=False)
            total = a[idx].sum()
            # random Dirichlet distribution
            new_vals = rng.random(3)
            new_vals = new_vals / new_vals.sum() * total
            a[idx] = new_vals
            cur_max = np.max(_full_convolution(a))
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[SA] jitter improved max={best_max:.6f}")

        # reduce step_factor if stagnation
        if it % 5000 == 0:
            if time.time() - last_improve_time > 30:
                step_factor *= 0.7
    return best_a, best_max

"""
Propose a step‑function that minimises the self‑convolution peak.
The algorithm works in two phases:

1. **Smooth optimisation** – a few minutes of Adam + sub‑gradient
   optimisation (the same machinery that produced the previous 1.514‑score).
   The optimiser works on the simplex Σa = √(2·n) and uses a soft‑max
   approximation of the max‑norm.

2. **Fine‑grained simulated‑annealing** – starting from the best vector
   found above, a temperature‑controlled random walk on the simplex
   (mass swaps) is performed.  The objective used during this phase is
   the raw maximum of the full convolution; the evaluation function
   itself is scale‑invariant, so we keep the vector normalised to sum = 1.
   The annealing quickly decorrelates the sequence and often pushes the
   score a few‑hundredths below the previous best.

Both phases respect the non‑negativity and simplex constraints,
use FFT‑based convolutions for speed and stay well within the 1000 s
budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the weighted sum Σ w_k·conv_k w.r.t. a.
    Implements g_j = 2 Σ_i w_{i+j} a_i.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max (smooth max) utilities
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Stable soft‑max weights w_i = exp(alpha·conv_i) / Σ exp(alpha·conv_i)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Upsampling helper (linear interpolation)
# ----------------------------------------------------------------------


def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a to length new_n."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=float)
    return np.interp(x_new, x_old, a)


# ----------------------------------------------------------------------
# Adam optimiser on a smooth max‑norm
# ----------------------------------------------------------------------


def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Adam optimisation on a soft‑max approximation of max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # hyper‑parameters
    lr = 0.3
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    alpha = 0.5
    while time.time() < deadline:
        t += 1
        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_simplex(a, S)

        # tiny random mass‑swap for extra exploration
        if rng.random() < 0.04:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.01
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # sharpen the soft‑max
        alpha = min(alpha * 1.02, 250.0)

        # cheap early‑stop
        if t % 5000 == 0 and best_score == cur_score:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient optimiser (max‑norm)
# ----------------------------------------------------------------------


def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that directly attacks the worst entry."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.25
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient for the active shift
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional random perturbation
        if rng.random() < 0.05:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.02
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = np.max(_full_convolution(cand))
                if cand_score < cur_score or rng.random() < 0.2:
                    a = cand
                    cur_score = cand_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()

        if it % 8000 == 0 and best_score == cur_score:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimiser (Adam → sub‑gradient)
# ----------------------------------------------------------------------


def _optimize_combined_adam(a0: np.ndarray,
                            deadline: float,
                            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Run Adam for ≈60 % of the allocated time, then finish with sub‑gradient."""
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    smooth_deadline = now + 0.6 * remaining
    a_s, sc_s = _optimize_adam(a0, smooth_deadline, rng)

    if time.time() < deadline:
        a_f, sc_f = _optimize_subgrad(a_s, deadline, rng)
        if sc_f < sc_s:
            return a_f, sc_f
    return a_s, sc_s


# ----------------------------------------------------------------------
# Simulated annealing on the raw max‑norm (sum‑normalised vectors)
# ----------------------------------------------------------------------


def _simulated_annealing(a_init: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Mass‑swap annealing on the simplex (Σa = 1).  Returns the best vector
    seen and its raw maximum convolution entry.
    """
    a = a_init.copy()
    a = a / a.sum()                     # enforce Σa = 1
    n = a.shape[0]

    cur_max = np.max(_full_convolution(a))
    best_a = a.copy()
    best_max = cur_max

    # temperature proportional to the current objective
    T = max(1e-6, 0.05 * cur_max)
    decay = 0.99999
    step_factor = 0.30
    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        # move a small fraction of mass from i to j
        delta = a[i] * rng.random() * step_factor
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_max = np.max(_full_convolution(a_new))
        dE = new_max - cur_max

        accept = False
        if dE <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / T):
                accept = True

        if accept:
            a = a_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
                last_improve = time.time()
                print(f"[SA] iter={it} new best max={best_max:.8f}")

        # temperature decay
        T = max(T * decay, 1e-12)

        # occasional larger jitter (3‑way redistribution)
        if rng.random() < 0.004:
            idx = rng.choice(n, size=3, replace=False)
            total = a[idx].sum()
            new_vals = rng.random(3)
            new_vals = new_vals / new_vals.sum() * total
            a[idx] = new_vals
            cur_max = np.max(_full_convolution(a))
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                last_improve = time.time()
                print(f"[SA] jitter improved max={best_max:.8f}")

        # adapt step size if we have been stagnant
        if it % 5000 == 0:
            if time.time() - last_improve > 30.0:
                step_factor = max(step_factor * 0.7, 0.02)

    return best_a, best_max
