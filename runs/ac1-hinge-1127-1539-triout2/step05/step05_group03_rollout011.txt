def _optimize_adam(a0, deadline, rng):
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    lr = 0.4
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    alpha = 0.5
    alpha_mul = 1.02

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)

        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            # print progress
            if it % 2000 == 0:
                print(f"[adam] it={it} best={best_score:.6f}")

        # weights
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a, S)

        # random mass swap
        if rng.random() < 0.01:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.05 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max:
                    a = cand

        # schedule adjustments
        alpha = min(alpha * alpha_mul, 200.0)  # sharper
        lr *= 0.99996  # slower decay

    return best_a, best_score

"""Minimise the self‑convolution peak (the value returned by evaluate_sequence).

The optimiser works in three stages:

1. **Small‑scale search** – a modest length vector is optimised with a
   long Adam run (smooth log‑sum‑exp max‑approximation) followed by a
   sub‑gradient phase.  This quickly discovers a promising shape.

2. **Upsampling** – the best small vector is linearly interpolated to a
   larger length and re‑optimised for the remaining budget, again with
   Adam → sub‑gradient.

3. **Polishing** – a few fast “mix‑in” moves with simple reference
   shapes (uniform, edge‑heavy) are attempted to shave off the last
   fractional amount of the peak.

All updates keep the coefficients on the simplex
{a_i ≥ 0, Σ a_i = √(2·n)}; the objective is scale‑invariant, so this
normalisation does not affect the score.  The code runs for the full
budget (≈ 1000 s) and returns the best sequence it found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of (1/α)·log‑sum‑exp(α·(a*a)) w.r.t. a.
    w must be a probability vector (sum = 1) of the same length as
    the full convolution of a with itself.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Smooth‑max utilities
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights w_i = exp(alpha*conv_i) / Σ_j exp(alpha*conv_j)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------
def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Resize an existing sequence to length n using linear interpolation."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n with ℓ¹‑norm √(2·n).
    Several families are mixed to diversify the starting points.
    """
    S = np.sqrt(2.0 * n)

    typ = rng.integers(0, 5)

    if typ == 0 and base_seq is not None and base_seq.size > 10:
        a = _upsample_seq(base_seq, n)
        a += rng.normal(scale=a.mean() * 1e-3, size=n)
        a = np.clip(a, 0.0, None)
    elif typ == 1:
        a = rng.random(n)                     # Dirichlet(1)
    elif typ == 2:
        rate = rng.uniform(0.9, 0.9995)
        left = rate ** np.arange(n)
        right = left[::-1]
        a = (left + right) / 2.0
    elif typ == 3:
        idx = np.arange(n, dtype=float)
        a = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.1, 0.9)
        a = a ** power
    else:
        # random convex combination of a few bases
        cand = []
        cand.append(rng.random(n))
        rate = rng.uniform(0.9, 0.9995)
        left = rate ** np.arange(n)
        cand.append((left + left[::-1]) / 2.0)
        idx = np.arange(n, dtype=float)
        rec = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        cand.append(rec ** rng.uniform(0.2, 0.8))
        w = rng.random(len(cand))
        w /= w.sum()
        a = sum(wi * ci for wi, ci in zip(w, cand))

    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0)
        a_sum = a.sum()
    a = a / a_sum * S
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smooth log‑sum‑exp max‑approximation
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected Adam minimising the smoothed max‑norm."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.4
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule
    alpha = 0.5
    alpha_mul = 1.02
    max_alpha = 200.0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[adam] it={it} best={best_score:.6f}")

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a, S)

        # occasional random mass‑swap
        if rng.random() < 0.01:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = 0.05 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max:
                    a = cand

        # schedule updates
        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99996

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient phase (focus on the worst shift)
# ----------------------------------------------------------------------
def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the true max‑norm (only the worst entry)."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the active shift
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = 2.0 * a[idx_j]

        eta = 0.25 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            if it % 1000 == 0:
                print(f"[subgrad] it={it} best={best_score:.6f}")

        # occasional random perturbation
        if rng.random() < 0.03:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max:
                    a = cand

    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimiser (Adam → sub‑gradient)
# ----------------------------------------------------------------------
def _optimize_combined(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run a smooth Adam phase for ~60 % of the remaining budget,
    then a sub‑gradient refinement for the rest.
    """
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    # 60 % Adam, 40 % sub‑gradient
    smooth_deadline = now + 0.6 * remaining
    a_s, sc_s = _optimize_adam(a0, smooth_deadline, rng)

    if time.time() < deadline:
        a_f, sc_f = _optimize_subgrad(a_s, deadline, rng)
        if sc_f < sc_s:
            return a_f, sc_f
    return a_s, sc_s


# ----------------------------------------------------------------------
# Polishing stage – short mixes with simple reference shapes
# ----------------------------------------------------------------------
def _polish(a: np.ndarray,
            deadline: float,
            rng: np.random.Generator) -> np.ndarray:
    """Attempt cheap mix‑ins with uniform and edge‑heavy shapes."""
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    best = a.copy()
    best_val = np.max(_full_convolution(a))

    attempts = 0
    while time.time() < deadline and attempts < 60:
        attempts += 1

        # reference shapes
        uniform = np.full(n, 1.0)
        uniform = uniform / uniform.sum() * S

        idx = np.arange(n, dtype=float)
        edge = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        edge = edge ** rng.uniform(0.2, 0.8)
        edge = edge / edge.sum() * S

        for ref in (uniform, edge):
            for t in (0.05, 0.1, 0.2):
                cand = (1.0 - t) * a + t * ref
                cand = _project_simplex(cand, S)
                val = np.max(_full_convolution(cand))
                if val < best_val - 1e-5:
                    best = cand
                    best_val = val
                    a = cand    # keep the improvement for further tries

    return best


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Multi‑scale optimisation for the step‑function.
    Returns the best sequence found before the time limit.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # keep a safety margin for the final external evaluation
    deadline = start + budget_s - 5.0

    # ------------------------------------------------------------------
    # Stage 1 – small vector optimisation
    # ------------------------------------------------------------------
    n_small = int(rng.integers(1800, 3000))
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    a0 = _initial_sequence(base_seq, n_small, rng)
    t1_deadline = min(time.time() + 0.15 * (deadline - start), deadline)
    a_small, score_small = _optimize_combined(a0, t1_deadline, rng)
    print(f"[stage 1] n={n_small} best={score_small:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – up‑sample and large‑scale optimisation
    # ------------------------------------------------------------------
    n_large = int(rng.integers(6000, 12000))
    a_large = _upsample_seq(a_small, n_large)
    S_large = np.sqrt(2.0 * n_large)
    a_large = a_large / a_large.sum() * S_large

    a_large, score_large = _optimize_combined(a_large, deadline, rng)
    print(f"[stage 2] n={n_large} best={score_large:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – polishing with cheap mixes
    # ------------------------------------------------------------------
    polish_deadline = min(deadline, time.time() + 4.0)
    a_final = _polish(a_large, polish_deadline, rng)

    final_seq = a_final.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external={external_score:.6f} internal={score_large:.6f} length={len(final_seq)}")
    return final_seq
