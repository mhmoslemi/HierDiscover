def _reduce_max_convolution(a, deadline):
    n = len(a); target = np.sqrt(2*n)
    best = a.copy(); best_score = _score(a)
    rng = np.random.default_rng()
    # hyperparameters:
    base_lr = 0.1 # adjust
    # Determine schedule for learning_rate decaying
    while time.time() < deadline:
        # Compute conv and max
        conv = _conv_fft(a)
        k_max = int(np.argmax(conv))
        # Determine sensitivities:
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi+1)
        s = a[k_max - idx]  # length hi-lo+1
        # compute weight = a_i * s_i (for indices idx)
        weight = np.zeros(n)
        weight[idx] = a[idx] * s
        # choose donor and receiver
        total_w = weight.sum()
        if total_w > 0:
            donor = rng.choice(n, p=weight/total_w)
        else:
            donor = rng.integers(n)
        # receivers: maybe choose from low weight
        low_weight_indices = np.where(weight <= np.percentile(weight, 20))[0]
        if low_weight_indices.size == 0:
            receiver = rng.integers(n)
        else:
            receiver = rng.choice(low_weight_indices)
        if donor != receiver:
            # Transfer fraction
            delta = a[donor] * 0.1 * rng.random()  # up to 10%
            a[donor] -= delta
            a[receiver] += delta
        # Project onto simplex? Not required because sum unchanged, nonnegatives maintained; but rounding errors might accumulate; we can occasionally renormalize.
        if (a < 0).any(): # fix negative
            a = np.maximum(a, 0.0)
        # Occasionally reproject to maintain sum target exactly
        if (np.abs(a.sum() - target) > 1e-8) or (np.random.random() < 0.001):
            a = _project_to_simplex(a, target)

        # Evaluate score
        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time():.1f}] Reduce step new best {best_score:.6f}")
        # maybe also implement smooth step: average over short windows
        if rng.random() < 0.05:
            start = rng.integers(0, n)
            length = rng.integers(2, min(50, n-start)+1)
            segment = a[start:start+length]
            avg = segment.mean()
            a[start:start+length] = avg
    return best, best_score

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import time, numpy as np

    start = time.time()
    deadline = start + budget_s - 1.0

    rng = np.random.default_rng(seed)

    # Stage 1: quick search on small n
    n_small = 2000
    target_small = np.sqrt(2.0 * n_small)
    best_small = None
    best_small_score = np.inf

    # try power law with varying exponent
    for exp in np.linspace(0.2, 1.0, 9):
        seq = _power_law_sequence(n_small, exp, target_small)
        seq = _project_to_simplex(seq, target_small)
        score = _score(seq)
        if score < best_small_score:
            best_small_score = score
            best_small = seq.copy()
    # random Dirichlet attempts
    while time.time() < deadline - 150:  # reserve for later stage
        seq = rng.random(n_small)
        seq = seq / seq.sum() * target_small
        score = _score(seq)
        if score < best_small_score:
            best_small_score = score
            best_small = seq.copy()
            print(f"[{time.time():.1f}] Stage1 new best {best_small_score:.6f}")

    # Stage 2: upsample to large n
    n_large = 5000
    target_large = np.sqrt(2.0 * n_large)

    # upsample
    x_small = np.linspace(0, 1, n_small)
    x_large = np.linspace(0, 1, n_large)
    seq_large = np.interp(x_large, x_small, best_small)
    seq_large = _project_to_simplex(seq_large, target_large)

    # Run gradient descent
    gd_deadline = min(deadline, time.time() + 300.0)  # 5 minutes
    seq_large, gd_score = _annealed_gradient_descent(seq_large, gd_deadline, max_beta=4096.0)
    best_large = seq_large.copy()
    best_large_score = gd_score

    # Run simulated annealing until deadline
    if time.time() < deadline:
        seq_large, sa_score = _simulated_annealing(best_large, deadline)
        if sa_score < best_large_score:
            best_large = seq_large.copy()
            best_large_score = sa_score

    # Also attempt a second run with fresh random initialize if still time
    if time.time() < deadline - 30:
        seq2 = rng.random(n_large)
        seq2 = seq2 / seq2.sum() * target_large
        # quick gradient descent 30 sec
        gd_deadline2 = time.time() + 30
        seq2, _ = _annealed_gradient_descent(seq2, gd_deadline2, max_beta=4096.0)
        # SA for remaining
        seq2, score2 = _simulated_annealing(seq2, deadline)
        if score2 < best_large_score:
            best_large = seq2.copy()
            best_large_score = score2

    # strip trailing zeros
    nz = np.where(best_large > 1e-12)[0]
    if nz.size:
        best_large = best_large[:nz[-1]+1]

    print(f"[{time.time():.1f}] final best score={best_large_score:.6f}, length={len(best_large)}")
    return best_large.tolist()

"""
Hybrid optimisation for the step‑function benchmark.

The routine works in three phases:

1️⃣  **Broad exploration (small size)** – a few power‑law shapes and many
    random Dirichlet vectors of length ~2000 are projected onto the
    ℓ¹‑simplex Σa = √(2 n).  The best candidate is kept.

2️⃣  **Upscaling & smooth optimisation** – the best small candidate is
    linearly interpolated to a larger dimension (≈8000).  A projected
    gradient descent on a soft‑max approximation of the ‑max‑convolution
    (β raised up to 4096) quickly lowers the smooth objective while
    staying on the simplex.

3️⃣  **Fine‑grained simulated‑annealing** – on the large‑scale problem a
    bespoke SA loop performs mass‑transfer, local smoothing and random
    swaps.  The move‑size and temperature decay exponentially, allowing
    the search to escape shallow minima and settle near a low‑maximum
    self‑convolution.

All steps respect the non‑negativity and ℓ¹‑norm constraints, so the
final sequence can be evaluated directly with ``evaluate_sequence``.
The function returns the best sequence discovered before the time budget
expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())           # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Fast evaluation of the benchmark: 2·n·max(conv)/(sum²)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).

    Returns:
        g   – gradient w.r.t. a (shape n)
        b   – full convolution (length 2n‑1)
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)                     # soft‑max distribution over indices

    # gradient = 2 * (a * reverse(p))  (via convolution)
    a_rev = a[::-1]
    Pr_fft = np.fft.rfft(p, n=size)
    Ar_fft = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]        # keep only the n relevant components
    return g, b


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               max_beta: float = 1024.0) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the soft‑max approximation of max(conv).

    Parameters
    ----------
    a0 : np.ndarray
        Starting point (already on the simplex Σa = √(2 n)).
    deadline : float
        Unix timestamp at which the optimisation must stop.
    max_beta : float, optional
        Maximum temperature for the soft‑max (higher → sharper).

    Returns
    -------
    best_a, best_val
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    step_base = target_sum * 0.20
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            lr = step_base / (norm_g * (beta + 1.0))
            a -= lr * g

        # keep feasibility
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to shake out of plateaus
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] GD beta={beta:.1f} best={best_val:.6f}")

        # increase temperature (sharpen the max)
        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return best_a, best_val


def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_temp: float = 0.8,
                         end_temp: float = 1e-6,
                         max_frac_start: float = 0.40,
                         max_frac_end: float = 0.01,
                         max_window: int = 120) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing refinement on a large problem.
    Moves: mass transfer, local averaging, random swaps.
    """
    a = a0.copy()
    n = a.size
    cur_score = _score(a)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)

    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()
    conv = _conv_fft(a)                     # current convolution

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # active index (where conv attains its maximum)
        max_b = conv.max()
        k_max = int(np.argmax(conv))

        # sensitivity for each coordinate caused by the active entry
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]

        weight = a * sens                     # contribution of each coordinate
        move_type = None

        # ---------- mass transfer (most frequent) ----------
        if rng.random() < 0.70:
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight / total_w)
            else:
                donor = rng.integers(n)

            # pick a receiver that currently carries little weight
            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            receiver = rng.choice(low_idx) if low_idx.size else rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        # ---------- short‑window smoothing ----------
        elif rng.random() < 0.90:
            start = rng.integers(0, n)
            length = rng.integers(2, min(max_window, n - start) + 1)
            end = start + length
            if end - start >= 2:
                old_seg = a[start:end].copy()
                avg = old_seg.mean()
                a[start:end] = avg
                move_type = "smooth"
                saved_seg = old_seg
                saved_start, saved_end = start, end

        # ---------- random swap ----------
        else:
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # evaluate the new state (single FFT)
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = _score(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA new best = {best_score:.6f}")
        else:
            # revert the move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter to escape deep minima
        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=cur_score * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = a * (cur_score / a.sum())
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

    print(f"[{time.time():.1f}] final best score={best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
