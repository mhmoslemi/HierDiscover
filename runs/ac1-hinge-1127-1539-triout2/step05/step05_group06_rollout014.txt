def _lp_improve(a, M, S, rng):
    n = len(a)
    # Build constraints matrix A as described.
    data = []
    offsets = []
    for i in range(n):
        length = n - i
        data.append(np.full(length, a[i]))
        offsets.append(i)
    A = diags(data, offsets, shape=(2*n-1, n), format='csc')
    # bounds
    bounds = [(0, None)] * n
    # objective: maximize sum(b) => minimize -sum(b)
    c = -np.ones(n)
    # RHS
    b_ub = np.full(2*n-1, M)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return a
    b0 = res.x
    sum_b = b0.sum()
    if sum_b == 0:
        return a
    g = b0 * (S / sum_b)
    # combine
    t = 0.1
    a_new = (1 - t) * a + t * g
    # ensure symmetric
    a_new = (a_new + a_new[::-1]) / 2
    a_new = _project_simplex(a_new, S)
    return a_new

"""
Propose a low‑peak non‑negative step sequence.

The algorithm works in three stages:
1. **Random / up‑sampled restarts** – generate many candidate vectors of
   different lengths and bring them onto the L¹‑simplex Σ a_i = √(2·n).
2. **Symmetric Adam + sub‑gradient refinement** – minimise a smooth
   soft‑max approximation of the convolution peak while enforcing symmetry
   a_i = a_{n‑1‑i}.  After a few seconds the temperature is raised to make the
   approximation very sharp, then a true sub‑gradient descent on the worst
   convolution entry finishes the local optimisation.
3. **Linear‑programming boost** – for the current best vector solve a small
   LP (maximise Σ b_j subject to f * b ≤ current peak) and mix the solution
   back.  This follows the “LP‑update” described in the statement and often
   yields a noticeable drop in the peak.

The whole search runs until the supplied budget expires, always keeping the
best sequence seen.  Finally the external `evaluate_sequence` is called and
the best list is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – keeps Σ a_i = s and a_i ≥ 0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution helpers
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient term for a smooth approximation of the max‑norm:
        g_j = 2 Σ_i w_{i+j} a_i .
    Both ``a`` (len n) and ``w`` (len 2·n‑1) are real vectors.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) weights – smooth approximation of max
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """w_i = exp(alpha·conv_i) / Σ_j exp(alpha·conv_j) (stable)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Linear‑up‑sampling (simple linear interpolation)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a to length new_n."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=float)
    return np.interp(x_new, x_old, a)


# ----------------------------------------------------------------------
# Enforce mirror symmetry a_i = a_{n-1-i}
# ----------------------------------------------------------------------
def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Replace a by its symmetric version (average with its reverse)."""
    return (a + a[::-1]) * 0.5


# ----------------------------------------------------------------------
# Adam optimiser (smooth max) – symmetric version
# ----------------------------------------------------------------------
def _optimize_adam_sym(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Adam on a soft‑max of the autocorrelation, keeping the vector symmetric."""
    a = a0.copy()
    n = a.shape[0]
    S = math.sqrt(2.0 * n)

    # hyper‑parameters
    lr = 0.4
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    alpha = 0.5                     # start very smooth
    while time.time() < deadline:
        t += 1
        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

                a[j] += delta

                a = _symmetrize(a)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:

            best_a = a.copy()

        # sharpen the soft‑max
        alpha = min(alpha * 1.05, 2000.0)

    return best_a, best_score

# ----------------------------------------------------------------------
# Linear‑programming improvement (LP boost)
# ----------------------------------------------------------------------
def _lp_improve(a: np.ndarray,
                M: float,
                S: float,
                rng: np.random.Generator) -> np.ndarray:
    """
    Solve   maximise Σ b_j    subject to   a * b ≤ M   and   b ≥ 0,
    then mix the solution back with a small step.
    """
    try:
        from scipy.sparse import diags
        from scipy.optimize import linprog
    except Exception:                     # SciPy not available – skip LP step
        return a

    n = a.shape[0]

    # Build the Toeplitz‑like matrix that maps b → a * b  (full convolution)
    data = []
    offsets = []
    for i in range(n):
        length = n - i                     # number of non‑zero entries on this diagonal
        if length <= 0:
            continue
        data.append(np.full(length, a[i]))
        offsets.append(i)
    A = diags(data, offsets, shape=(2 * n - 1, n), format='csc')

    # Linprog solves a *minimise* problem → minimise –∑b_j
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)            # a * b ≤ M
    bounds = [(0, None)] * n                # b ≥ 0

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success or res.x is None:
        return a

    b0 = res.x
    total = b0.sum()
    if total == 0.0:
        return a

    # Scale so that Σ g_j = S (the same L¹ norm as a)
    g = b0 * (S / total)

    # Mix a little towards g
    t = 0.1
    a_new = (1.0 - t) * a + t * g

    a_new = _symmetrize(a_new)
    a_new = _project_simplex(a_new, S)
    return a_new

            S_big = math.sqrt(2.0 * len(seq_big))
            M_big = np.max(_full_convolution(seq_big))
            seq_big = _lp_improve(seq_big, M_big, S_big, rng)
            final_score = np.max(_full_convolution(seq_big))

        best_seq = seq_big
        best_score = final_score

    # ------------------------------------------------------------------
    # Return the best sequence as a plain Python list
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()

    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} "
          f"(internal best = {best_score:.6f}) length = {len(final_seq)}")
    return final_seq
