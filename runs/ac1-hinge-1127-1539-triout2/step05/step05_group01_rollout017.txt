"""
Improved search for a step‑function (non‑negative sequence) that minimises
``evaluate_sequence``.  The algorithm consists of four phases:

1. **Broad parametric sweep** – power‑law, exponential‑decay and random
   monotone candidates are generated for several lengths.  The best few
   (elite) are kept.

2. **Projected gradient descent on a log‑sum‑exp smooth approximation**
   of the maximal convolution entry.  The update uses momentum and an
   adaptive step size and projects onto the ℓ¹‑simplex
   Σa_i = √(2 n).

3. **Simulated‑annealing (SA)** – mass‑transfer moves with a geometric
   cooling schedule are applied to each refined elite vector.

4. **Greedy flattening** – after SA the index k where the convolution
   is maximal is identified; a small amount of mass is transferred from
   the pair (i,k‑i) that contributes most to that entry to a low‑impact
   location.  The move is kept if it does not increase the current maximum.
   This cheap deterministic local search often pushes the objective a
   little further down.

All phases respect a global time budget (default 1000 s).  The final
sequence is returned as a plain list of floats; its ℓ¹‑norm is
√(2 n) so the value returned by ``evaluate_sequence`` equals the maximal
convolution entry of the sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex Σx = s, x≥0  (Duchi et al. 2008)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) surrogate of max(b) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of   (1/β)·log Σ_k exp(β·b_k)   where b = a*a (linear convolution).
    Returns   g = ∇_a (…)   and the full convolution b.
    """
    n = a.shape[0]

    # ---- a*a (full convolution) ----
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # ---- soft‑max probabilities ----
    max_b = np.max(b)
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)

    # ---- gradient g = 2·(a ⋆ p) ----
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Projected gradient descent on the smoothed maximum
# ----------------------------------------------------------------------


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Momentum‑based projected gradient descent on the log‑sum‑exp smooth max.
    Stops when the time budget expires.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 1.0
    max_beta = 1e7
    momentum = np.zeros_like(a)
    rho = 0.9                     # momentum decay
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum step -------------------------------------------------
        momentum = rho * momentum + (1.0 - rho) * g
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            # adaptive step size – inversely proportional to gradient norm
            lr = 0.5 * target_sum / (norm * (beta + 1.0))
            a -= lr * momentum

        # projection onto the ℓ¹‑simplex ---------------------------------
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation -------------------------
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4 / (beta + 1.0),
                               size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal β -------------------------------------------------------
        if it % 150 == 0 and beta < max_beta:
            beta = min(beta * 1.07, max_beta)

        it += 1
    return a


# ----------------------------------------------------------------------
# Simple simulated‑annealing refinement
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Random mass‑transfer SA with geometric cooling.
    The move transfers a random fraction (≤15 %) of mass from one index to another.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    T = best_score * 0.02 + 1e-8
    cooling = 0.9994
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-15:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        frac = rng.random() * 0.15
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        accept = False
        if cur_score <= best_score:
            accept = True
        elif T > 0.0 and rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # very occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
    return best


# ----------------------------------------------------------------------
# Greedy flattening of the convolution peak
# ----------------------------------------------------------------------


def _greedy_flatten(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Deterministic local search: repeatedly identify the convolution index k
    where the maximum occurs and transfer a small fraction of mass from the
    most “responsible’’ pair (i, k‑i) to a location that hardly participates
    in that peak.  The move is kept only if it does not increase the current
    maximum.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_val = _max_convolution_fft(best)

    rng = np.random.default_rng()
    while time.time() < deadline:
        conv = _full_convolution_fft(a)
        cur_max = float(conv.max())
        max_k = int(conv.argmax())

        # contributions of each index to the max entry
        idx = np.arange(n)
        j = max_k - idx
        mask = (j >= 0) & (j < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[j[mask]]

        # pick donor index – the one with biggest contribution
        donor = int(np.argmax(contrib))
        if a[donor] <= 1e-12:
            # nothing left to move – stop
            break

        # pick a receiver that has (almost) zero contribution to this peak
        # cheap heuristic: choose a random index far from donor and with tiny contrib
        receiver = donor
        attempts = 0
        while (receiver == donor or contrib[receiver] > 1e-12) and attempts < 15:
            receiver = rng.integers(0, n)
            attempts += 1
        if receiver == donor:
            # fall back to a random small‑value index
            receiver = int(np.argmin(a + 1e-12 * rng.random(n)))

        # amount to move – up to 20 % of donor's mass
        delta = a[donor] * 0.2
        a[donor] -= delta
        a[receiver] += delta

        new_max = _max_convolution_fft(a)
        if new_max <= best_val:
            best_val = new_max
            best = a.copy()
        else:
            # revert
            a[receiver] -= delta
            a[donor] += delta
            # if we cannot improve, do a tiny random jitter and continue
            if rng.random() < 0.02:
                jitter = rng.normal(scale=a.sum() * 5e-4, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                a = _project_to_simplex(a, a.sum())
    return best


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    """a_i = (i + shift)^{-p}, projected onto Σa = √(2 n)."""
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    target = np.sqrt(2.0 * n)
    scale = target / raw.sum()
    a = raw * scale
    return np.maximum(a, 0.0)


def _exponential_sequence(n: int, c: float) -> np.ndarray:
    """
    a_i = A * (1 - ε)^{i},   ε = c / n.
    A is chosen so that Σ a_i = √(2 n).  The sequence is non‑negative and decreasing.
    """
    eps = c / n
    r = 1.0 - eps
    # finite‑sum normalisation
    rn = r ** n
    target = np.sqrt(2.0 * n)
    A = target * eps / (1.0 - rn)
    idx = np.arange(n, dtype=float)
    a = A * (r ** idx)
    return np.maximum(a, 0.0)


def _random_monotone_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Generate a random decreasing sequence: draw uniform numbers,
    sort them decreasingly and project onto the simplex.
    """
    raw = rng.random(n)
    raw = np.sort(raw)[::-1]
    target = np.sqrt(2.0 * n)
    scale = target / raw.sum()
    a = raw * scale
    return np.maximum(a, 0.0)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the allotted time budget, explores a wide parametric
    grid, refines elite vectors with gradient descent + SA + greedy flattening,
    and finally returns the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # small safety margin – we do not want to overrun the budget
    deadline = time.time() + budget_s - 0.2

    best_seq = None
    best_obj = np.inf
    elite = []                      # list of (max_conv, array)

    # ------------------------------------------------------------------
    # Phase 1 – deterministic/random sweep
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 4000, 5000, 6000]          # keep within a few thousand
    p_grid = np.linspace(0.45, 0.80, 13)             # power‑law exponent
    shift_grid = np.linspace(0.5, 4.5, 9)            # shift for power‑law
    c_grid = np.linspace(0.6, 3.0, 12)               # exponential decay constant c

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)
        print(f"[{time.time():.1f}] Exploring length {n}")

        # ---- power‑law candidates ----
        for p in p_grid:
            for shift in shift_grid:
                a = _powerlaw_sequence(n, p, shift)
                score = _max_convolution_fft(a)
                if score < best_obj:
                    best_obj = score
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] New global best (grid) n={n} p={p:.3f} shift={shift:.2f} obj={best_obj:.6f}")

                # keep top‑10 elite candidates
                if len(elite) < 10 or score < elite[-1][0]:
                    elite.append((score, a))
                    elite.sort(key=lambda x: x[0])
                    elite = elite[:10]

        # ---- exponential‑decay candidates ----
        for c in c_grid:
            a = _exponential_sequence(n, c)
            score = _max_convolution_fft(a)
            if score < best_obj:
                best_obj = score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] New global best (exp) n={n} c={c:.2f} obj={best_obj:.6f}")

            if len(elite) < 10 or score < elite[-1][0]:
                elite.append((score, a))
                elite.sort(key=lambda x: x[0])
                elite = elite[:10]

        # ---- random monotone candidates (quick burst) ----
        rnd_end = min(deadline, time.time() + 2.0)
        while time.time() < rnd_end:
            a = _random_monotone_sequence(n, rng)
            score = _max_convolution_fft(a)
            if score < best_obj:
                best_obj = score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] New global best (rand) n={n} obj={best_obj:.6f}")

            if len(elite) < 10 or score < elite[-1][0]:
                elite.append((score, a))
                elite.sort(key=lambda x: x[0])
                elite = elite[:10]

    # ------------------------------------------------------------------
    # Phase 2 – refinement of elite candidates
    # ------------------------------------------------------------------
    for rank, (init_score, seed_seq) in enumerate(elite):
        if time.time() >= deadline:
            break
        n = seed_seq.size
        print(f"[{time.time():.1f}] Refining elite #{rank+1} (n={n}, init_obj={init_score:.6f})")

        # allocate a sub‑budget (max 30 s) for gradient descent; remaining time goes to SA+flatten
        sub_dead_gd = min(deadline, time.time() + 30.0)
        a_gd = _gradient_descent_opt(seed_seq, sub_dead_gd)

        # SA – give it the rest of the global budget
        a_sa = _simulated_annealing(a_gd, deadline)

        # Greedy flattening – a cheap deterministic polishing step
        a_final = _greedy_flatten(a_sa, deadline)

        obj = _max_convolution_fft(a_final)
        if obj < best_obj:
            best_obj = obj
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL BEST (refined) n={n} obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final sanity / fallback
    # ------------------------------------------------------------------
    if best_seq is None:
        # uniform fallback – should never happen
        n = 2500
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_obj = _max_convolution_fft(best_seq)

    # Trim trailing near‑zero entries for neatness
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # final sanity check with the provided evaluation routine
    try:
        eval_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {eval_val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective = {best_obj:.6f}   length = {best_seq.size}")
    return best_seq.tolist()
