"""
Search algorithm for a non‑negative step function minimizing the target
``evaluate_sequence``.  The method consists of:

1. **Broad parametric sampling** – random two‑term power‑law/exponential
   families (and their symmetric variants) are generated for several
   sequence lengths.  The best few candidates are kept.

2. **Projected gradient descent on a smooth max‑convolution surrogate** –
   the log‑sum‑exp smoothing of the convolution maximum is minimized while
   keeping the vector on the ℓ¹‑simplex Σa = √(2 n).

3. **Deterministic mass‑transfer (sub‑gradient) descent** – for the current
   maximum convolution entry (and nearby entries) the sub‑gradient w.r.t.
   the sequence is computed and a small amount of mass is moved from the
   most harmful index to the least harmful one.  This directly reduces
   the true maximum.

4. **Simulated annealing** – a short SA phase refines the result.

The routine respects a user‑supplied time budget (default 1000 s) and
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex Σx = s, x≥0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection onto {x_i ≥ 0, Σx_i = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) surrogate of max(b) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b) where
        b = a*a (full convolution)
        g = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a ⋆ p) ,  p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]

    # ----- convolution a*a -----
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # ----- soft‑max probabilities -----
    max_b = np.max(b)
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)

    # ----- gradient g = 2·(a ⋆ p) -----
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Projected gradient descent on the smoothed maximum
# ----------------------------------------------------------------------


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Projected gradient descent on the log‑sum‑exp smooth max."""
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 1.0
    max_beta = 1e5
    momentum = np.zeros_like(a)
    rho = 0.9                     # momentum decay
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum step
        momentum = rho * momentum + (1.0 - rho) * g
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            lr = 0.5 * target_sum / (norm * (beta + 1.0))
            a -= lr * momentum

        # simplex projection
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 8e-4 / (beta + 1.0),
                               size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # exponential annealing of β
        if it % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.06, max_beta)

        it += 1
    return a


# ----------------------------------------------------------------------
# Simple simulated‑annealing refinement
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Random mass‑transfer SA with multiplicative cooling."""
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # temperature initialised from current objective
    T = best_score * 0.02 + 1e-8
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        # move up to 15 % of mass from i to j
        frac = rng.random() * 0.15
        delta = a[i] * frac
        if delta <= 0.0:
            continue
        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        accept = False
        if cur_score <= best_score:
            accept = True
        elif T > 0.0 and rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # very occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
    return best


# ----------------------------------------------------------------------
# Parametric families (power‑law / exponential / multi‑term)
# ----------------------------------------------------------------------


def _powerlaw_exponential_sequence(n: int, p: float, shift: float,
                                   lam: float) -> np.ndarray:
    """a_i = (i+shift)^{-p}·exp(-lam·i), projected onto Σa = √(2 n)."""
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p) * np.exp(-lam * idx)
    target = np.sqrt(2.0 * n)
    a = raw * (target / raw.sum())
    a = np.maximum(a, 0.0)
    return a


def _symmetric_powerlaw_exponential_sequence(n: int, p: float, shift: float,
                                             lam: float) -> np.ndarray:
    """Symmetric version: a_i = (i+shift)^{-p}e^{-lam i} + (n-1-i+shift)^{-p}e^{-lam (n-1-i)}."""
    idx = np.arange(n, dtype=float)
    left = (idx + shift) ** (-p) * np.exp(-lam * idx)
    right = (n - 1 - idx + shift) ** (-p) * np.exp(-lam * (n - 1 - idx))
    raw = left + right
    target = np.sqrt(2.0 * n)
    a = raw * (target / raw.sum())
    a = np.maximum(a, 0.0)
    return a


def _multi_term_sequence(n: int, terms) -> np.ndarray:
    """
    Sum of weighted power‑law·exponential terms.
    ``terms`` – iterable of (p, shift, lam, weight).
    Result is projected onto the ℓ¹‑simplex Σa = √(2 n).
    """
    idx = np.arange(n, dtype=float)
    raw = np.zeros(n, dtype=float)
    for p, shift, lam, w in terms:
        raw += w * (idx + shift) ** (-p) * np.exp(-lam * idx)
    target = np.sqrt(2.0 * n)
    a = raw * (target / raw.sum())
    a = np.maximum(a, 0.0)
    return a


# ----------------------------------------------------------------------
# Deterministic sub‑gradient mass‑transfer descent
# ----------------------------------------------------------------------


def _mass_transfer_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Directly reduces the true maximum entry of the convolution.
    Repeatedly moves a small fraction of mass from the index with the
    largest sub‑gradient (most harmful) to the one with the smallest.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    rng = np.random.default_rng()
    while time.time() < deadline:
        b = _full_convolution_fft(a)
        max_b = b.max()
        # consider indices whose value is within 0.1 % of the maximum
        tol = max_b * 0.001
        idxs = np.where(b >= max_b - tol)[0]
        if idxs.size == 0:
            break

        # sub‑gradient for the selected indices
        g = np.zeros(n)
        for k in idxs:
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            for i in range(i_min, i_max + 1):
                j = k - i
                g[i] += a[j]

        i_src = int(np.argmax(g))
        i_tgt = int(np.argmin(g))
        if i_src == i_tgt:
            break

        # move a modest fraction of the source mass
        delta = a[i_src] * 0.08  # 8 %
        if delta < 1e-12:
            break
        a[i_src] -= delta
        a[i_tgt] += delta

        # keep feasibility
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        # occasional random jitter to escape plateaus
        if rng.random() < 0.01:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimizes
    ``evaluate_sequence``.  The routine respects the allotted time budget,
    explores a rich parametric family, refines elite candidates with
    projected gradient descent, a deterministic sub‑gradient mass‑transfer
    phase and a short simulated‑annealing finish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_obj = np.inf
    elite = []        # list of (obj, array), at most 10 elements

    # ------------------------------------------------------------------
    # 0.  Incorporate a known good construction if available
    # ------------------------------------------------------------------
    try:
        prev = globals().get("height_sequence_1")
        if isinstance(prev, (list, tuple)) and len(prev) > 0:
            a = np.asarray(prev, dtype=float)
            if a.ndim == 1:
                a = np.maximum(a, 0.0)
                target = np.sqrt(2.0 * a.size)
                a = a * (target / a.sum())
                cur_obj = _max_convolution_fft(a)
                best_seq, best_obj = a, cur_obj
                elite.append((cur_obj, a.copy()))
                print(f"[{time.time():.1f}] Seeded with height_sequence_1, obj={cur_obj:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Broad parametric sweep (random + grid)
    # ------------------------------------------------------------------
    lengths = [1800, 2200, 2600, 3000, 3500, 4000, 4500]
    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # ---- random two‑term samples ----
        for _ in range(200):
            if time.time() >= deadline:
                break
            # draw random parameters
            p1 = rng.uniform(0.30, 1.00)
            shift1 = rng.uniform(0.0, 12.0)
            lam1 = rng.uniform(0.0, 0.025)
            w1 = rng.random()
            p2 = rng.uniform(0.30, 1.00)
            shift2 = rng.uniform(0.0, 12.0)
            lam2 = rng.uniform(0.0, 0.025)
            w2 = 1.0 - w1
            terms = [(p1, shift1, lam1, w1), (p2, shift2, lam2, w2)]
            a = _multi_term_sequence(n, terms)
            # occasional symmetric mirroring
            if rng.random() < 0.5:
                a = (a + a[::-1]) * 0.5
                a = a * (target_sum / a.sum())
            obj = _max_convolution_fft(a)
            if obj < best_obj:
                best_obj, best_seq = obj, a.copy()
                print(f"[{time.time():.1f}] New global best (rnd) n={n} obj={obj:.6f}")
            if len(elite) < 10 or obj < elite[-1][0]:
                elite.append((obj, a.copy()))
                elite.sort(key=lambda x: x[0])
                elite = elite[:10]

        # ---- deterministic grid over power‑law + exponential ----
        p_grid = np.linspace(0.30, 1.00, 9)
        shift_grid = np.linspace(0.0, 12.0, 7)
        lam_grid = [0.0, 0.008, 0.016]    # three exponential strengths
        for p in p_grid:
            for shift in shift_grid:
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    a = _powerlaw_exponential_sequence(n, p, shift, lam)
                    obj = _max_convolution_fft(a)
                    if obj < best_obj:
                        best_obj, best_seq = obj, a.copy()
                        print(f"[{time.time():.1f}] New global best (grid) n={n} p={p:.3f} shift={shift:.2f} lam={lam:.3f} obj={obj:.6f}")
                    a_sym = _symmetric_powerlaw_exponential_sequence(n, p, shift, lam)
                    obj2 = _max_convolution_fft(a_sym)
                    if obj2 < best_obj:
                        best_obj, best_seq = obj2, a_sym.copy()
                        print(f"[{time.time():.1f}] New global best (sym) n={n} p={p:.3f} shift={shift:.2f} lam={lam:.3f} obj={obj2:.6f}")
                    # store elite
                    for cand_obj, cand in ((obj, a), (obj2, a_sym)):
                        if len(elite) < 10 or cand_obj < elite[-1][0]:
                            elite.append((cand_obj, cand.copy()))
                            elite.sort(key=lambda x: x[0])
                            elite = elite[:10]

    # ------------------------------------------------------------------
    # 2. Refinement of elite candidates
    # ------------------------------------------------------------------
    for _, seed_seq in elite:
        if time.time() >= deadline:
            break
        n = seed_seq.size
        target_sum = np.sqrt(2.0 * n)

        # ---- gradient descent (smooth max) ----
        gd_deadline = min(deadline, time.time() + 22.0)   # about 22 s
        a_gd = _gradient_descent_opt(seed_seq, gd_deadline)

        # ---- deterministic mass‑transfer descent ----
        mt_deadline = min(deadline, time.time() + 22.0)
        a_mt = _mass_transfer_descent(a_gd, mt_deadline)

        # ---- short simulated annealing ----
        a_sa = _simulated_annealing(a_mt, deadline)

        obj = _max_convolution_fft(a_sa)
        if obj < best_obj:
            best_obj, best_seq = obj, a_sa.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL BEST (refined) n={n} obj={obj:.6f}")

    # ------------------------------------------------------------------
    # 3. Final sanity / fallback
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        target = np.sqrt(2.0 * n)
        best_seq = np.full(n, target / n)

    # Trim trailing near‑zero entries for neatness
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Verify with the official evaluator (optional)
    try:
        ev = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {ev:.6f}")
    except Exception as e:
        print("Evaluation failed:", e)

    print(f"[{time.time():.1f}] Finished. Best objective = {best_obj:.6f} "
          f"length = {best_seq.size}")
    return best_seq.tolist()
