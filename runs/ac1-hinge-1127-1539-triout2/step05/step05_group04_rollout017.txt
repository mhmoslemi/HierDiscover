"""
Step‑function optimizer – heavy multi‑restart projected descent + aggressive
mass‑balancing.  The routine explores several lengths (2000 … 20000), creates
many random Dirichlet‑type starts, runs a long smooth‑max (log‑sum‑exp)
gradient descent, follows with exact sub‑gradient steps on the true
ℓ∞‑norm and ends with repeated “mass‑transfer” moves that directly reduce the
convolution peak.  The best candidate is finally polished by a short
simulated‑annealing run.  All operations respect the ℓ¹‑simplex
Σa = √(2 n) so the official score equals the maximal entry of a*a.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0, Σ x_i = s } (Duchi et al. 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1, dtype=float)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a (full length 2·n−1) via FFT.
    Returned array is real‑valued.
    """
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective after ℓ¹‑normalisation Σa = √(2 n): max(a*a)."""
    return float(_max_convolution_fft(a).max())


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) gradient – Phase 1
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ_k exp(β·b_k) where b = a*a.

    Returns (g, b) with
        g = 2·(a ⋆ p),  p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = float(b.max())
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3 n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _smooth_max_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Annealed projected gradient descent on the smooth max.
    The schedule is deliberately aggressive to drive the solution close
    to a true ℓ∞‑minimiser.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 8192.0                # larger than before → tighter approximation
    beta_factor = 1.5                # a slightly faster annealing

    # step size is a fraction of the ℓ¹‑norm (empirically good)
    step_base = target_sum * 0.25
    momentum = np.zeros_like(a)
    rho = 0.90

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        # projection onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal β
        if it % 350 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a


# ----------------------------------------------------------------------
# Exact max‑index sub‑gradient – Phase 2
# ----------------------------------------------------------------------


def _max_index_subgradient(a: np.ndarray):
    """
    Sub‑gradient of max(b) where b = a*a.
    Returns (g, k) with g_i = 2·a_{k‑i} for indices that contribute
    to the maximal convolution entry k.
    """
    n = a.shape[0]
    conv = _max_convolution_fft(a)
    k = int(np.argmax(conv))

    g = np.zeros_like(a)
    if k < n:                         # left side of the convolution
        g[:k + 1] = 2.0 * a[k::-1]
    else:                             # right side
        d = k - (n - 1)
        g[d:] = 2.0 * a[: n - d]
    return g, k


def _subgradient_descent_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected sub‑gradient descent on the true ℓ∞‑norm.  The diminishing
    step size η_t = α / √t works well in practice.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _max_index_subgradient(a)

        # diminishing step size
        step = 0.9 * target_sum / (np.sqrt(it + 1) *
                                   np.linalg.norm(g) + 1e-12)
        a -= step * g

        # feasibility
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        # tiny jitter to avoid cycles
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 4e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1
    return a

# Targeted mass‑balance refinement
# ----------------------------------------------------------------------


def _mass_balance_refine(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Greedy move that reduces the contribution of the presently maximal
    convolution entry.  A modest fraction of the “heaviest” coefficient
    (the one that participates most in that entry) is transferred to a
    “light” location.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sum(a)                 # should already be √(2 n)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _max_convolution_fft(a)
        k = int(np.argmax(conv))

        # contribution of each index to that entry
        idx = np.arange(n)
        other = k - idx
        mask = (other >= 0) & (other < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[other[mask]]

        i_high = int(np.argmax(contrib))

        # pick a relatively empty position – among the 20 % smallest masses
        low_candidates = np.where(a <= np.percentile(a, 20))[0]
        if low_candidates.size == 0:
            i_low = int(np.argmin(a))
        else:
            i_low = int(rng.choice(low_candidates))

        # move a modest fraction (3 %) of the heavy mass
        delta = a[i_high] * 0.03
        if delta < 1e-12:
            break

        a[i_high] -= delta
        a[i_low] += delta

        # keep feasibility
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        # occasional global jitter
        if it % 100 == 0 and rng.random() < 0.02:
            jitter = rng.normal(scale=target_sum * 1e-3, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return a

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Sample a diverse point on the ℓ¹‑simplex Σa = √(2 n).
    Several families (power‑law, exponential, Dirichlet, piecewise)
    are mixed to increase diversity.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                                 # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)
    elif family == 1:                               # pure exponential
        lam = rng.exponential(scale=0.02)
        a = np.exp(-lam * idx)
    elif family == 2:                               # power×exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.02)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)
    elif family == 3:                               # Dirichlet (uniform)
        a = rng.dirichlet(alpha=np.ones(n))
    elif family == 4:                               # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])
    else:                                           # almost uniform with jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a

def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Very lightweight SA that moves a random fraction of mass between two sites."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    cur_score = _score(a)
    best_score = cur_score

    T = cur_score * 0.02 + 1e-9
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-16:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        frac = rng.random() * 0.15
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta

        new_score = _score(a)
        dE = new_score - cur_score
        accept = dE <= 0.0 or (T > 0.0 and rng.random() < np.exp(-dE / T))

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling

    return best

    # Lengths to investigate – larger n gives more freedom but costs more
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000, 10000, 15000, 20000]

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)

        # How many seconds do we allocate for this length?
        # Roughly split the remaining budget evenly among remaining lengths.
        remaining_lengths = len(lengths) - lengths.index(n)
        time_per_len = (deadline - time.time()) / max(remaining_lengths, 1)

        # ------------------------------------------------------------------
        # Phase A – many cheap random restarts with short smooth‑max GD
        # ------------------------------------------------------------------
        phase_a_dead = min(deadline,
                           time.time() + min(30.0, time_per_len * 0.4))
        while time.time() < phase_a_dead:
            a_start = _random_initial_sequence(n, rng)

            # short smooth‑max run (≈2 s) to get into a decent basin
            sub_dead = min(deadline, time.time() + 2.5)
            a_gd = _smooth_max_descent(a_start, sub_dead)

            sc = _score(a_gd)
            if sc < best_score:
                best_score = sc
                best_seq = a_gd.copy()
                print(f"[{time.time():.1f}] Random restart n={n}  score={sc:.6f}")

        # ------------------------------------------------------------------
        # Phase B – deeper refinement of the current best of this length
        # ------------------------------------------------------------------
        if best_seq is not None and best_seq.size == n:
            # allocate the bulk of the remaining budget for this length
            phase_b_dead = min(deadline, time.time() + time_per_len * 0.5)

            # 1) long smooth‑max gradient descent
            a_ref = _smooth_max_descent(best_seq, phase_b_dead)
            sc = _score(a_ref)
            if sc < best_score:
                best_score = sc
                best_seq = a_ref.copy()
                print(f"[{time.time():.1f}] Smooth‑max refinement n={n}  "
                      f"score={sc:.6f}")

            # 2) exact max‑index sub‑gradient descent
            sub_dead = min(deadline, time.time() + min(40.0, time_per_len * 0.2))
            a_ref2 = _subgradient_descent_max(best_seq, sub_dead)
            sc = _score(a_ref2)
            if sc < best_score:
                best_score = sc
                best_seq = a_ref2.copy()
                print(f"[{time.time():.1f}] Sub‑gradient improvement n={n}  "
                      f"score={sc:.6f}")

            # 3) aggressive mass‑balance fine‑tuning
            sub_dead = min(deadline, time.time() + min(30.0, time_per_len * 0.15))
            a_ref3 = _mass_balance_refine(best_seq, sub_dead)
            sc = _score(a_ref3)
            if sc < best_score:
                best_score = sc
                best_seq = a_ref3.copy()
                print(f"[{time.time():.1f}] Mass‑balance refinement n={n}  "
                      f"score={sc:.6f}")

        # ------------------------------------------------------------------
        # Phase C – try to enlarge the sequence (more degrees of freedom)
        # ------------------------------------------------------------------
        # Occasionally embed the current best into a longer vector.
        if best_seq is not None and n < lengths[-1]:
            nxt = n + 2000
            if nxt in lengths:
                a_padded = np.concatenate([best_seq, np.zeros(nxt - n)])
                a_padded = _project_to_simplex(a_padded,
                                               np.sqrt(2.0 * nxt))
                # quick smooth‑max on the padded version
                sub_dead = min(deadline, time.time() + 3.0)
                a_padded = _smooth_max_descent(a_padded, sub_dead)
                sc = _score(a_padded)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_padded.copy()
                    print(f"[{time.time():.1f}] Padded to n={nxt}  "
                          f"score={sc:.6f}")

    # ----------------------------------------------------------------------
    # Final simulated‑annealing polishing with whatever time is left
    # ----------------------------------------------------------------------
    if best_seq is not None:
        left = deadline - time.time()
        if left > 5.0:
            a_final = _simulated_annealing(best_seq, deadline)
            sc = _score(a_final)
            if sc < best_score:
                best_score = sc
                best_seq = a_final.copy()
                print(f"[{time.time():.1f}] Final SA improvement: score={sc:.6f}")

    # ----------------------------------------------------------------------
    # Clean up tiny trailing zeros and report the official evaluation
    # ----------------------------------------------------------------------
    if best_seq is None:
        # Fallback – uniform distribution of length 2000 (should never happen)
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    # remove trailing values that are essentially zero
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Clip to the limits enforced by ``evaluate_sequence`` just to be safe
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final official score (for logging)
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:      # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
