"""
Step‑function optimizer – “Adam‑softmax + sub‑gradient + mass‑balance”.
The routine searches over several lengths, starting from a known seed,
from scaled‑up versions of the current best, and from diverse random
initialisations.  For each length it runs a fast Adam optimiser on a
log‑sum‑exp (smooth‑max) surrogate of the self‑convolution peak, then
refines the solution with an exact max‑index sub‑gradient step, a greedy
mass‑balancing move, and a brief simulated‑annealing polish.  Time is
budgeted so that the total wall‑clock does not exceed the supplied limit.
The best sequence (as a plain Python list) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1, dtype=float)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·n−1) using an FFT.
    Returned array is real‑valued.
    """
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Objective after ℓ¹‑normalisation Σa = √(2 n):
    simply the supremum norm of the self‑convolution.
    """
    return float(_max_convolution_fft(a).max())


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) gradient – Phase 1 (Adam optimiser)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) where
        g = ∇_a  ( 1/β·log Σ exp(β·b) )
          = 2·(a ⋆ p)          with   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # ---- b = a*a -------------------------------------------------------
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ---- soft‑max probabilities p --------------------------------------
    max_b = float(np.max(b))
    exps = np.exp(beta * (b - max_b))          # shift for numerical stability
    p = exps / np.sum(exps)

    # ---- gradient g = 2·(a ⋆ p) ----------------------------------------
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1            # = 3 n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _adam_softmax_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Adam optimiser on the smooth‑max surrogate.
    The temperature β is increased geometrically; after each few
    thousand steps a tiny Gaussian jitter is added to escape flat regions.
    The iterate is projected onto the ℓ¹‑simplex after every update.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 2048.0
    beta_factor = 1.5

    # Adam hyper‑parameters
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    rng = np.random.default_rng()

    # a modest base learning‑rate (as a fraction of the ℓ¹‑norm)
    base_lr = 0.02 * target_sum

    while time.time() < deadline:
        it += 1
        g, _ = _gradient_softmax(a, beta)

        # Adam updates
        m = b1 * m + (1.0 - b1) * g
        v = b2 * v + (1.0 - b2) * (g * g)
        m_hat = m / (1.0 - b1 ** it)
        v_hat = v / (1.0 - b2 ** it)

        step = base_lr / (np.sqrt(v_hat) + eps)
        a -= step * m_hat

        # keep feasibility
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 3e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal temperature
        if beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Exact max‑index sub‑gradient descent – Phase 2
# ----------------------------------------------------------------------


def _max_index_subgradient(a: np.ndarray):
    """
    Sub‑gradient for the true max‑norm of the self‑convolution.
    Returns (g, k) where k is the arg‑max index.
    """
    n = a.shape[0]
    conv = _max_convolution_fft(a)
    k = int(np.argmax(conv))

    g = np.zeros_like(a)
    if k < n:                         # left half
        g[:k + 1] = 2.0 * a[k::-1]
    else:                             # right half
        d = k - (n - 1)               # shift amount (0 … n‑1)
        g[d:] = 2.0 * a[: n - d]
    return g, k


def _subgradient_descent_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected sub‑gradient descent on the true max‑norm.
    Diminishing step‑size η_t = α / √t is used.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        g, _ = _max_index_subgradient(a)

        step = 0.8 * target_sum / (np.sqrt(it + 1) *
                                   (np.linalg.norm(g) + 1e-12))
        a -= step * g

        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1
    return a


# ----------------------------------------------------------------------
# Mass‑balance greedy fine‑tuning
# ----------------------------------------------------------------------


def _mass_balance_refine(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Greedy move that transfers a small fraction of the “heavy” coefficient
    (the one that contributes most to the current maximal convolution entry)
    to a relatively empty location.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sum(a)               # should already be √(2 n)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _max_convolution_fft(a)
        k = int(np.argmax(conv))

        # contribution of each index to that entry: a_i * a_{k‑i}
        idx = np.arange(n)
        other = k - idx
        mask = (other >= 0) & (other < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[other[mask]]

        i_high = int(np.argmax(contrib))

        # pick a location among the 20 % smallest masses
        low_candidates = np.where(a <= np.percentile(a, 20))[0]
        if low_candidates.size == 0:
            i_low = int(np.argmin(a))
        else:
            i_low = int(rng.choice(low_candidates))

        delta = a[i_high] * 0.03
        if delta < 1e-12:
            break

        a[i_high] -= delta
        a[i_low] += delta

        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        if it % 100 == 0 and rng.random() < 0.02:
            jitter = rng.normal(scale=target_sum * 1e-3, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return a


# ----------------------------------------------------------------------
# Simple simulated‑annealing polishing (lightweight)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Very light SA that moves a random fraction of mass between two sites."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    cur_score = _score(a)
    best_score = cur_score

    T = cur_score * 0.02 + 1e-9
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-16:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        frac = rng.random() * 0.15
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta

        new_score = _score(a)
        dE = new_score - cur_score
        accept = dE <= 0.0 or (T > 0.0 and rng.random() < np.exp(-dE / T))

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling

    return best


# ----------------------------------------------------------------------
# Random initialisation helpers
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative point on the simplex Σa = √(2 n).
    Several families (power‑law, exponential, Dirichlet, piecewise) are
    sampled to increase diversity.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                                 # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                               # pure exponential
        lam = rng.exponential(scale=0.02)
        a = np.exp(-lam * idx)

    elif family == 2:                               # power × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.02)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                               # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n))

    elif family == 4:                               # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                             np.exp(-lam2 * np.arange(b, n))])

    else:                                           # almost uniform with jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _scale_sequence(a: np.ndarray, n_new: int) -> np.ndarray:
    """
    Linear interpolation of an existing sequence to a new length,
    followed by projection onto the appropriate ℓ¹‑simplex.
    """
    n0 = a.shape[0]
    if n0 == n_new:
        return a.copy()
    x_old = np.arange(n0)
    x_new = np.linspace(0, n0 - 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target_sum = np.sqrt(2.0 * n_new)
    a_new = _project_to_simplex(a_new, target_sum)
    return a_new


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` metric within the provided time budget.
    Returns the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Seed from a previously discovered construction if available
    # ------------------------------------------------------------------
    earlier = globals().get("height_sequence_1")
    if isinstance(earlier, (list, tuple, np.ndarray)):
        a0 = np.asarray(earlier, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        sc0 = _score(a0)
        best_seq = a0.copy()
        best_score = sc0
        print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) score={sc0:.6f}")

    # ------------------------------------------------------------------
    # Lengths to explore – larger n gives more freedom but costs more
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000, 12000, 16000, 20000]

    prev_best_seq = best_seq
    prev_n = best_seq.size if best_seq is not None else None

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        print(f"[{time.time():.1f}] ==== Optimising length n={n} ====")

        best_n_seq = None
        best_n_score = np.inf

        # ---- 1. Scaled‑up version of the previous best -----------------
        if prev_best_seq is not None:
            init = _scale_sequence(prev_best_seq, n)
            sub_dead = time.time() + min(15.0, max(0.0, deadline - time.time()))
            cand = _adam_softmax_descent(init, sub_dead)
            sc = _score(cand)
            if sc < best_n_score:
                best_n_score = sc
                best_n_seq = cand.copy()
                print(f"[{time.time():.1f}] Scaled init from n={prev_n} → n={n}, score={sc:.6f}")

        # ---- 2. Random restarts ----------------------------------------
        restarts = 4 if n <= 8000 else 2
        for r in range(restarts):
            if time.time() >= deadline:
                break
            a0 = _random_initial_sequence(n, rng)
            sub_dead = time.time() + min(12.0, max(0.0, deadline - time.time()))
            cand = _adam_softmax_descent(a0, sub_dead)
            sc = _score(cand)
            if sc < best_n_score:
                best_n_score = sc
                best_n_seq = cand.copy()
                print(f"[{time.time():.1f}] Random restart n={n} r={r}, score={sc:.6f}")

        if best_n_seq is None:
            continue  # shouldn't happen, but guard against it

        # ---- 3. Sub‑gradient descent on the true max --------------------
        sub_dead = time.time() + min(25.0, max(0.0, deadline - time.time()))
        cand = _subgradient_descent_max(best_n_seq, sub_dead)
        sc = _score(cand)
        if sc < best_n_score:
            best_n_score = sc
            best_n_seq = cand.copy()
            print(f"[{time.time():.1f}] Sub‑gradient refinement n={n}, score={sc:.6f}")

        # ---- 4. Mass‑balance greedy fine‑tuning -------------------------
        sub_dead = time.time() + min(15.0, max(0.0, deadline - time.time()))
        cand = _mass_balance_refine(best_n_seq, sub_dead)
        sc = _score(cand)
        if sc < best_n_score:
            best_n_score = sc
            best_n_seq = cand.copy()
            print(f"[{time.time():.1f}] Mass‑balance refinement n={n}, score={sc:.6f}")

        # ---- 5. Light simulated‑annealing polish (if time permits) -----
        if deadline - time.time() > 5.0:
            cand = _simulated_annealing(best_n_seq, deadline)
            sc = _score(cand)
            if sc < best_n_score:
                best_n_score = sc
                best_n_seq = cand.copy()
                print(f"[{time.time():.1f}] SA polish n={n}, score={sc:.6f}")

        # ---- Update global best -----------------------------------------
        if best_n_score < best_score:
            best_score = best_n_score
            best_seq = best_n_seq.copy()
            print(f"[{time.time():.1f}] ** Global best updated: n={n} score={best_score:.6f}")

        # Early exit if target reached
        if best_score <= 1.503:
            print(f"[{time.time():.1f}] Target reached (≤1.503), stopping early.")
            break

        # Store for scaling to the next length
        prev_best_seq = best_n_seq.copy()
        prev_n = n

    # ------------------------------------------------------------------
    # Final simulated‑annealing polishing with any remaining budget
    # ------------------------------------------------------------------
    if best_seq is not None and deadline - time.time() > 5.0:
        left = deadline - time.time()
        if left > 5.0:
            cand = _simulated_annealing(best_seq, deadline)
            sc = _score(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time():.1f}] Final SA improvement: score={sc:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up tiny trailing zeros and report the official evaluation
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution of length 1000 (should never happen)
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
