def adam_opt(a0, deadline):
    a = a0.copy()
    m = np.zeros_like(a); v = np.zeros_like(a)
    t = 0
    beta1=0.9; beta2=0.999; eps=1e-8
    target_sum = sqrt(2*n)
    beta = 1.0; max_beta = 1e6
    while time.time()<deadline:
        grad,_ = gradient_softmax(a,beta)
        t+=1
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        step = 0.1/(np.sqrt(v_hat)+eps)  # adapt
        a = a - step * m_hat
        # project onto simplex
        a = project_to_simplex(a, target_sum)
        # anneal beta periodically after some iterations
        if t%200 ==0:
            beta = min(beta*1.1, max_beta)
    return a

"""
Propose a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.  The algorithm works in three stages
   1.  A fast parametric sweep (power‑law, exponential & random Dirichlet)
       generates a short elite list.
   2.  Each elite candidate is refined by a long Adam‑style projected
       gradient descent on a soft‑max surrogate of the maximal convolution.
   3.  The best refined vector is polished by a short simulated‑annealing
       mass‑transfer phase.

All operations respect the total time budget (default 1000 s) and keep the
ℓ¹‑simplex constraint Σa = √(2 n) – under this normalisation the
evaluate_sequence value equals the maximal entry of the self‑convolution,
so the optimiser directly minimises that quantity.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection onto the simplex using the algorithm of Duchi et al. (2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) surrogate of the max entry of a*a
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (grad, conv) where
        conv = a*a (full linear convolution)
        grad = ∇_a ( (1/β)·log Σ_k exp(β·conv_k) )
              = 2·(a ⋆ p) ,   p_k ∝ exp(β·conv_k)
    """
    n = a.shape[0]

    # ----- a*a via FFT -----
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # ----- soft‑max probabilities -----
    max_c = np.max(conv)
    exp_shifted = np.exp(beta * (conv - max_c))
    p = exp_shifted / np.sum(exp_shifted)

    # ----- gradient = 2·(a ⋆ p)  (cross‑correlation) -----
    # cross‑correlation a with p is convolution of reversed a with p
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    A_fft = np.fft.rfft(a_rev, n=size2)
    P_fft = np.fft.rfft(p, n=size2)
    cross = np.fft.irfft(A_fft * P_fft, n=size2)[:conv_len]
    grad = 2.0 * cross[n - 1: n - 1 + n]

    return grad, conv


# ----------------------------------------------------------------------
# Projected Adam‑style gradient descent on the smooth max
# ----------------------------------------------------------------------


def _adam_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Adam optimizer with simplex projection.
    The smoothness parameter β is slowly increased to approximate the true max.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # schedule for the soft‑max temperature
    beta = 0.5
    max_beta = 1e7

    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.1 / (np.sqrt(v_hat) + eps)      # base learning rate 0.1
        a -= step * m_hat

        # projection onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape tiny plateaus
        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4 / (beta + 1.0), size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        # anneal β (smoothness) slowly upward
        if it % 250 == 0:
            beta = min(beta * 1.05, max_beta)

    return a


# ----------------------------------------------------------------------
# Simple simulated‑annealing polishing
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Mass‑transfer SA with geometric cooling.
    The sum of the vector stays constant, so no projection is needed.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_val = _max_convolution_fft(best)

    # initial temperature – a fraction of the current objective
    T = best_val * 0.02 + 1e-12
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        # pick two distinct indices
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # transfer a random fraction of the mass from i to j
        frac = rng.random() * 0.15          # up to 15 %
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta

        cur_val = _max_convolution_fft(a)
        dE = cur_val - best_val
        accept = cur_val <= best_val or (rng.random() < np.exp(-dE / T))

        if accept:
            if cur_val < best_val:
                best_val = cur_val
                best = a.copy()
        else:
            # revert the move
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # very rare global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            # keep sum exact
            a = _project_to_simplex(a, a.sum())
            cur_val = _max_convolution_fft(a)
            if cur_val < best_val:
                best_val = cur_val
                best = a.copy()

    return best


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    """a_i = (i+shift)^{-p}, projected onto the ℓ¹‑simplex."""
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    target = np.sqrt(2.0 * n)
    a = raw * (target / raw.sum())
    return np.maximum(a, 0.0)


def _symmetric_powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    """Symmetric version: a_i = (i+shift)^{-p} + (n‑1‑i+shift)^{-p}."""
    idx = np.arange(n, dtype=float)
    left = (idx + shift) ** (-p)
    right = (n - 1 - idx + shift) ** (-p)
    raw = left + right
    target = np.sqrt(2.0 * n)
    a = raw * (target / raw.sum())
    return np.maximum(a, 0.0)


def _exponential_sequence(n: int, r: float) -> np.ndarray:
    """a_i = r**i, projected onto the ℓ¹‑simplex."""
    idx = np.arange(n, dtype=float)
    raw = r ** idx
    target = np.sqrt(2.0 * n)
    a = raw * (target / raw.sum())
    return np.maximum(a, 0.0)


def _symmetric_exponential_sequence(n: int, r: float) -> np.ndarray:
    """a_i = r**i + r**(n‑1‑i), projected onto the ℓ¹‑simplex."""
    idx = np.arange(n, dtype=float)
    raw = r ** idx + r ** (n - 1 - idx)
    target = np.sqrt(2.0 * n)
    a = raw * (target / raw.sum())
    return np.maximum(a, 0.0)
