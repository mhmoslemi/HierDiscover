def max_conv_and_grad(a):
    n = a.shape[0]
    # compute convolution via FFT
    size = 1 << ((2*n-1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    c = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
    max_val = c.max()
    # get the index of max (if multiple, pick random among them)
    idxs = np.where(np.abs(c - max_val) < 1e-14)[0]
    k = np.random.choice(idxs)
    # compute gradient subvector
    grad = np.zeros_like(a)
    start = max(0, k - (n - 1))
    end = min(n - 1, k)
    # i runs start..end inclusive
    i = np.arange(start, end + 1)
    j = k - i
    grad[i] = a[j] + a[i]   # vectorized
    return max_val, grad
```

def _max_conv_and_subgrad(a):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    c = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]
    max_val = c.max()
    # find indices of max (choose one)
    idxs = np.where(c == max_val)[0]  # could be multiple
    k = np.random.choice(idxs)
    # compute gradient
    grad = np.zeros_like(a)
    start = max(0, k - (n - 1))
    end = min(n - 1, k)
    if start <= end:
        i = np.arange(start, end + 1)
        j = k - i
        grad[i] = a[j] + a[i]    # a[i] + a[j]
    return max_val, grad
```

Note: When j = i (i.e., k even and i = k/2), gradient includes a[i] twice (since a[i] + a[i]) which matches derivative of a_i^2 term (2 a_i). Good.

def _adam_optimize(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    target_sum = 1.0  # sum constraint
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_a = a.copy()
    best_obj = np.inf
    while time.time() < deadline:
        max_val, grad = _max_conv_and_subgrad(a)
        obj = 2.0 * n * max_val / (target_sum * target_sum)  # target_sum^2 =1
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = 0.01 * m_hat / (np.sqrt(v_hat) + eps)  # Choose α = 0.01
        a = a - step
        # project onto simplex (sum 1)
        a = _project_to_simplex(a, target_sum)
        # optional jitter every some iterations
        if t % 2000 == 0:
            jitter = np.random.default_rng().normal(scale=1e-3, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)
    return best_a, best_obj
```

"""
Propose a step‑function (non‑negative sequence) that minimises
``evaluate_sequence``.  

The search consists of three stages:

1. **Broad deterministic sweep** – generate several parametric families
   (power‑law, exponential, linear, random) for a few different lengths,
   keep the elite few (lowest max‑convolution).

2. **Exact sub‑gradient descent (Adam) on the true max** – for each elite
   candidate we minimise  max (a * a)  subject to a ≥ 0 and Σa = 1.
   The gradient of max (a * a) is obtained analytically, an Adam
   optimiser is used and each step is projected onto the ℓ¹‑simplex.

3. **Short random‑restart refinement** – while time permits we sample a
   fresh random start, run a quick Adam optimisation and keep the best
   result.

All steps respect the supplied time budget (default 1000 s).  The final
sequence is scaled to Σa = √(2 n) – this does not affect the objective but
keeps the values well below the hard limit of 1000 used by
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex Σx = s , x ≥ 0   (Duchi et al. 2008)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact gradient of the max‑convolution (sub‑gradient)
# ----------------------------------------------------------------------


def _max_conv_and_subgrad(a: np.ndarray):
    """
    Returns (max_val, grad) where
        max_val = max_k (a*a)_k
        grad_i   = ∂/∂a_i max_k (a*a)_k   (a sub‑gradient)
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_val = float(conv.max())
    # pick one of the maximising shifts uniformly at random
    idxs = np.where(conv == max_val)[0]
    k = np.random.choice(idxs)

    grad = np.zeros_like(a)
    start = max(0, k - (n - 1))
    end = min(n - 1, k)
    if start <= end:
        i = np.arange(start, end + 1)
        j = k - i
        # derivative of a_i*a_j plus derivative of a_j*a_i (when i≠j)
        grad[i] = a[j] + a[i]
    return max_val, grad


# ----------------------------------------------------------------------
# Adam optimiser for the non‑smooth objective  max(a*a)
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray, deadline: float):
    """
    Adam optimisation of  max(a*a)  with simplex projection.
    Returns the best vector seen (sum = 1) and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = 1.0

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr_const = 0.05          # base learning‑rate factor

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_obj = np.inf

    while time.time() < deadline:
        max_val, grad = _max_conv_and_subgrad(a)

        obj = 2.0 * n * max_val            # because Σa = 1 -> obj = 2 n max
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        t += 1
        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # adaptive step size (scale with gradient norm)
        step = lr_const * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow plateaus
        if t % 2000 == 0:
            jitter = np.random.default_rng().normal(scale=1e-3, size=n)
            a = _project_to_simplex(a + jitter, target_sum)

    return best_a, best_obj


# ----------------------------------------------------------------------
# Parametric families (raw, then normalised to sum = 1)
# ----------------------------------------------------------------------


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    raw = np.maximum(raw, 0.0)
    prob = raw / raw.sum()
    return prob


def _exponential_sequence(n: int, lam: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.exp(-lam * idx)
    prob = raw / raw.sum()
    return prob


def _linear_decreasing_sequence(n: int) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = (n - idx)
    prob = raw / raw.sum()
    return prob


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    best_seq = None
    best_obj = np.inf

    # ------------------------------------------------------------------
    # 1. Broad deterministic sweep (few lengths, many families)
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 4000, 5000, 6000]
    elite = []                     # (obj, array)  – we keep the top‑5
    elite_size = 5

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = 1.0

        # ----- power‑law grid -------------------------------------------------
        p_grid = np.linspace(0.45, 0.80, 13)
        shift_grid = np.linspace(0.5, 5.0, 9)
        for p in p_grid:
            for shift in shift_grid:
                a = _powerlaw_sequence(n, p, shift)
                max_val = _max_convolution_fft(a)
                obj = 2.0 * n * max_val
                if obj < best_obj:
                    best_obj, best_seq = obj, a.copy()
                    print(f"[{time.time():.1f}] New global best (pow) n={n} p={p:.3f}"
                          f" shift={shift:.2f} obj={obj:.6f}")
                # keep elite
                if len(elite) < elite_size or obj < elite[-1][0]:
                    elite.append((obj, a))
                    elite.sort(key=lambda x: x[0])
                    elite = elite[:elite_size]

        # ----- exponential / linear families ----------------------------------
        for lam in [0.001, 0.002, 0.005, 0.01, 0.02]:
            a = _exponential_sequence(n, lam)
            max_val = _max_convolution_fft(a)
            obj = 2.0 * n * max_val
            if obj < best_obj:
                best_obj, best_seq = obj, a.copy()
                print(f"[{time.time():.1f}] New global best (exp) n={n} lam={lam:.4f}"
                      f" obj={obj:.6f}")
            if len(elite) < elite_size or obj < elite[-1][0]:
                elite.append((obj, a))
                elite.sort(key=lambda x: x[0])
                elite = elite[:elite_size]

        a = _linear_decreasing_sequence(n)
        max_val = _max_convolution_fft(a)
        obj = 2.0 * n * max_val
        if obj < best_obj:
            best_obj, best_seq = obj, a.copy()
            print(f"[{time.time():.1f}] New global best (lin) n={n} obj={obj:.6f}")
        if len(elite) < elite_size or obj < elite[-1][0]:
            elite.append((obj, a))
            elite.sort(key=lambda x: x[0])
            elite = elite[:elite_size]

        # ----- random Dirichlet draws (few per length) ------------------------
        rnd_end = min(deadline, time.time() + 2.0)
        while time.time() < rnd_end:
            a = rng.random(n)
            a = a / a.sum()
            max_val = _max_convolution_fft(a)
            obj = 2.0 * n * max_val
            if obj < best_obj:
                best_obj, best_seq = obj, a.copy()
                print(f"[{time.time():.1f}] New global best (rand) n={n} obj={obj:.6f}")
            if len(elite) < elite_size or obj < elite[-1][0]:
                elite.append((obj, a))
                elite.sort(key=lambda x: x[0])
                elite = elite[:elite_size]

    # ------------------------------------------------------------------
    # 2. Adam optimisation of elite candidates
    # ------------------------------------------------------------------
    for idx, (_, seed_seq) in enumerate(elite):
        if time.time() >= deadline:
            break
        n = seed_seq.size
        sub_dead = min(deadline, time.time() + 35.0)   # ~35 s per elite
        a_opt, obj_opt = _adam_optimize(seed_seq, sub_dead)
        if obj_opt < best_obj:
            best_obj, best_seq = obj_opt, a_opt.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL BEST (adam) elite#{idx} n={n}"
                  f" obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 3. Random‑restart short Adam runs while time permits
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # pick a random length from the earlier set (biased to larger n)
        n = int(rng.choice([3000, 4000, 5000, 6000]))
        a0 = rng.random(n)
        a0 = a0 / a0.sum()
        sub_dead = min(deadline, time.time() + 6.0)   # quick 6 s optimisation
        a_opt, obj_opt = _adam_optimize(a0, sub_dead)
        if obj_opt < best_obj:
            best_obj, best_seq = obj_opt, a_opt.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL BEST (rand‑restart) n={n}"
                  f" obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 4. Post‑processing – scale to Σa = sqrt(2 n) as used by the challenge
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution of length 2000
        n = 2000
        best_seq = np.full(n, 1.0 / n)

    n = best_seq.size
    scale = np.sqrt(2.0 * n) / best_seq.sum()
    best_seq = best_seq * scale

    # tiny cleanup – drop trailing zeros
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # sanity check with the provided evaluator
    try:
        eval_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {eval_val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective = {best_obj:.6f}"
          f" length = {best_seq.size}")
    return best_seq.tolist()
