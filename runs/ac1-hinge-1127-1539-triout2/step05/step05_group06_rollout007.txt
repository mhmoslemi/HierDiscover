"""
Non‑negative step‑function optimisation.

Goal:
    minimise  max_k (a * a)[k]   subject to   a_i ≥ 0 ,  Σ a_i = √(2·n).

Method:
    * Work on the L¹‑simplex (projection on each iteration).
    * Smooth optimisation of the Lp‑norm of the full convolution
      with a schedule p = 2 → 4 → 8 → 16 → 32.
      The gradient of the Lp‑norm is obtained by a weighted convolution
      (FFT) and Adam is used for fast convergence.
    * The smooth phase consumes roughly 60 % of the allocated time,
      the remaining time is spent on a projected sub‑gradient method
      that directly attacks the current worst convolution entry.
    * Many short trials with random restarts are performed.
      Successful trials are up‑sampled to larger problem sizes
      (more degrees of freedom) and refined again.
    * Random “mass‑swap’’ and jitter moves are injected to escape
      plateaus.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient term for a weighted Lp‑norm of the convolution:
        g_j = 2 Σ_i w_{i+j} a_i .
    Both ``a`` (len n) and ``w`` (len 2·n‑1) are real vectors.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Linear interpolation up‑sampling
# ----------------------------------------------------------------------


def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a to length new_n."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=float)
    return np.interp(x_new, x_old, a)


# ----------------------------------------------------------------------
# Adam optimisation on a smooth Lp‑norm of the full convolution
# ----------------------------------------------------------------------


def _optimize_adam_pnorm(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises a smooth Lp‑norm (p increases from 2 to 32).
    Returns the best vector found and its peak‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = math.sqrt(2.0 * n)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # schedule of p‑values
    p_seq = [2, 4, 8, 16, 32]
    p_idx = 0
    steps_per_p = 2000          # after this many iterations we increase p

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        p = p_seq[p_idx]

        # convolution of the current iterate
        conv = _full_convolution(a)

        # weights for the Lp‑norm gradient: w_i ∝ conv_i^{p‑1}
        safe = np.maximum(conv, 1e-12)
        w = safe ** (p - 1)

        # normalise to keep the magnitude of the gradient bounded
        w_sum = w.sum()
        if w_sum > 0:
            w = w / w_sum

        # gradient (via weighted convolution)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning rate slowly shrinks with p
        lr = 0.6 / math.sqrt(p)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex
        a = _project_simplex(a, S)

        # occasional random mass‑swap (exploration)
        if rng.random() < 0.04:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.015
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # move to the next p‑value after a fixed number of iterations
        if it % steps_per_p == 0 and p_idx < len(p_seq) - 1:
            p_idx += 1

        # cheap early stop
        if it % 5000 == 0 and best_score == cur_score:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient optimiser (refines the true max‑norm)
# ----------------------------------------------------------------------


def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent that directly reduces the worst convolution
    entry.  Uses a diminishing step size η_t = η0 / √t.
    """
    a = a0.copy()
    n = a.shape[0]
    S = math.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.4          # initial step size
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the worst entry
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / math.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # random perturbation to escape plateaus
        if rng.random() < 0.05:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.02
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = np.max(_full_convolution(cand))
                if cand_score < cur_score or rng.random() < 0.2:
                    a = cand
                    cur_score = cand_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()

        # cheap early stop
        if it % 8000 == 0 and best_score == cur_score:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimiser (smooth → sub‑gradient)
# ----------------------------------------------------------------------


def _optimize_combined_adam(a0: np.ndarray,
                            deadline: float,
                            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run the smooth Lp‑norm Adam for ≈60 % of the time, then finish with a
    sub‑gradient phase.
    """
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    smooth_deadline = now + 0.6 * remaining
    a_s, sc_s = _optimize_adam_pnorm(a0, smooth_deadline, rng)

    if time.time() < deadline:
        a_f, sc_f = _optimize_subgrad(a_s, deadline, rng)
        if sc_f < sc_s:
            return a_f, sc_f
    return a_s, sc_s


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step sequence that minimises ``evaluate_sequence``.
    The routine performs many short Adam/Lp‑norm trials, progressively upsamples
    the best solution and finishes with a sub‑gradient refinement.
    Returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # reserve a few seconds for the final external evaluation
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_score = np.inf
    attempts = 0
    last_improve = start_time

    # ------------------------------------------------------------------
    # Main optimisation loop (short independent trials)
    # ------------------------------------------------------------------
    while time.time() < deadline - 12.0:      # keep a safety window for final work
        attempts += 1

        # decide on the problem size for this trial
        if best_seq is None or (time.time() - last_improve > 45.0):
            # fresh random start, moderate size
            n = int(rng.integers(800, 2500))
            a0 = rng.random(n)
        else:
            # sometimes up‑sample the current best to a larger resolution
            if rng.random() < 0.30 and len(best_seq) < 8000:
                n = min(int(len(best_seq) * 1.5), 8000)
                a0 = _upsample(best_seq, n)
            else:
                n = len(best_seq)
                a0 = best_seq.copy()

            # small jitter to avoid stagnation
            jitter = rng.normal(scale=a0.mean() * 1e-4, size=n)
            a0 = a0 + jitter
            a0 = np.clip(a0, 0.0, None)

        # project onto the correct simplex Σa = √(2·n)
        S = math.sqrt(2.0 * n)
        a0_sum = a0.sum()
        if a0_sum == 0.0:
            a0 = np.full(n, 1.0)
            a0_sum = a0.sum()
        a0 = a0 / a0_sum * S

        # time slice for this trial (~20 s)
        slice_deadline = min(time.time() + 20.0, deadline)

        a_opt, score_opt = _optimize_combined_adam(a0, slice_deadline, rng)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            last_improve = time.time()
            print(f"[global] attempt={attempts} n={n} new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final refinement on the largest feasible size
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 6.0:
        target_n = min(8000, max(len(best_seq), 4000))
        if target_n > len(best_seq):
            best_seq = _upsample(best_seq, target_n)

        # tiny random perturbation before the final run
        jitter = rng.normal(scale=best_seq.mean() * 1e-5, size=target_n)
        best_seq = np.clip(best_seq + jitter, 0.0, None)

        # re‑project (numerical safety)
        S = math.sqrt(2.0 * target_n)
        best_seq = _project_simplex(best_seq, S)

        final_deadline = deadline
        best_seq, best_score = _optimize_combined_adam(best_seq, final_deadline, rng)

    # ------------------------------------------------------------------
    # Final sanity check & return
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
    else:
        # final projection onto the simplex for the exact sum constraint
        n = len(best_seq)
        final_seq_np = _project_simplex(best_seq, math.sqrt(2.0 * n))
        final_seq = final_seq_np.tolist()

    # evaluate with the provided (secure) function
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external_score={external_score:.6f} "
          f"(internal best={best_score:.6f}) length={len(final_seq)}")
    return final_seq
