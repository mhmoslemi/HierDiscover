def propose_candidate(seed=12345, budget_s=1000, **kwargs):

    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 5

    base_seq = None
    if 'height_sequence_1' in globals():
        try: base_seq = np.asarray(height_sequence_1, dtype=float)
        except: base_seq = None

    best_seq = None
    best_score = np.inf
    attempts = 0
    last_improve = start

    # Reserve time for final refinement
    final_refine_time = 200.0  # seconds
    explore_deadline = deadline - final_refine_time
    if explore_deadline < start:
        explore_deadline = deadline

    while time.time() < explore_deadline:
        attempts += 1
        n = int(rng.integers(2000, 40000))  # bigger
        a0 = _initial_sequence(base_seq, n, rng)

        slice_deadline = min(time.time() + 15.0, explore_deadline)
        a_opt, score = _optimize_combined(a0, slice_deadline, rng)

        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            last_improve = time.time()
            print(f"[global] attempt={attempts} n={n} new best max={best_score:.6f}")

        # Occasionally discard base_seq for diversification
        if time.time() - last_improve > 60.0:
            base_seq = None

    # Final refinement on best_seq
    if best_seq is not None:
        # Maybe upsample if not too large? Already large.
        # continuous refine for remaining time
        refine_deadline = deadline
        a_refined, score_ref = _optimize_combined(best_seq, refine_deadline, rng)
        if score_ref < best_score:
            best_seq = a_refined.copy()
            best_score = score_ref
            print(f"[refine] improved to max={best_score:.6f}")

    # Convert to list and evaluate final external score
    final_seq = best_seq.tolist() if best_seq is not None else [1.0]
    external = evaluate_sequence(final_seq)
    print(f"[final] external score={external:.6f} internal max={best_score:.6f} length={len(final_seq)}")
    return final_seq

"""
Propose a step‑function that minimizes the self‑convolution peak.
Algorithm
---------
* Work on the ℓ¹‑simplex Σ a = √(2·n) (objective is scale‑invariant).
* Randomly sample problem sizes up to ~40 k and generate a wide
  variety of initial shapes – uniform, geometric, reciprocal,
  cosine “U‑shape”, exponential edge‑heavy, and mixtures thereof.
* Optimise each start with a two‑phase method:
    – 60 % of the allocated time: Adam on a smooth log‑sum‑exp
      approximation of the max‑norm (soft‑max temperature is raised
      slowly).  The iterate is projected onto the simplex after every
      update.
    – 40 % of the time: sub‑gradient descent on the true max‑norm.
* The inner optimiser runs for a configurable slice (≈15 s) and
  records the best max‑convolution found.
* After a number of exploratory attempts (leaving ~200 s for final
  refinement) the current global best is further refined for the
  remaining budget.
* The final sequence is returned as a plain list of floats; the black‑box
  ``evaluate_sequence`` is called once for sanity checking.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  (1/α)·log‑sum‑exp(α·(a*a)) w.r.t. a.
    Returns a vector of length n.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights (smooth approximation of max)
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """w_i = exp(alpha*conv_i) / Σ_j exp(alpha*conv_j) (stable)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Initial sequence generators (diverse families)
# ----------------------------------------------------------------------
def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Resize an existing sequence to length n using linear interpolation."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n with ℓ¹‑norm √(2·n).
    Several families are mixed to diversify the search.
    """
    S = np.sqrt(2.0 * n)

    # 0 – up‑sampled previous best (if any)
    # 1 – uniform Dirichlet
    # 2 – geometric decay (symmetrised)
    # 3 – reciprocal “edge‑heavy”
    # 4 – random convex mixture of 1–3
    # 5 – cosine U‑shape
    # 6 – exponential edge‑heavy
    typ = rng.integers(0, 7)

    if typ == 0 and base_seq is not None and base_seq.size > 10:
        a = _upsample_seq(base_seq, n)
        a += rng.normal(scale=a.mean() * 1e-3, size=n)   # tiny jitter
        a = np.clip(a, 0.0, None)

    elif typ == 1:
        a = rng.random(n)               # Dirichlet(1)

    elif typ == 2:
        rate = rng.uniform(0.9, 0.9995)
        left = rate ** np.arange(n)
        right = left[::-1]
        a = (left + right) / 2.0

    elif typ == 3:
        idx = np.arange(n, dtype=float)
        a = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.8)
        a = a ** power

    elif typ == 4:
        cand = []
        cand.append(rng.random(n))
        rate = rng.uniform(0.9, 0.9995)
        left = rate ** np.arange(n)
        right = left[::-1]
        cand.append((left + right) / 2.0)
        idx = np.arange(n, dtype=float)
        rec = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.8)
        cand.append(rec ** power)
        w = rng.random(len(cand))
        w /= w.sum()
        a = sum(wi * ci for wi, ci in zip(w, cand))

    elif typ == 5:
        # cosine based U‑shape: edges high, centre low
        beta = rng.uniform(0.2, 0.9)               # amplitude
        a = 1.0 + beta * np.cos(np.pi * np.arange(n) / (n - 1))
        # ensure positivity (cos ranges -1..1)
        a = np.clip(a, 1e-6, None)

    else:   # typ == 6
        # exponential decay from each edge towards the centre
        decay = rng.uniform(0.001, 0.02)
        mindist = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a = np.exp(-decay * mindist)
        power = rng.uniform(0.5, 2.0)
        a = a ** power

    # Normalise to the required ℓ¹‑norm √(2·n)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0)
        a_sum = a.sum()
    a = a / a_sum * S
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed log‑sum‑exp max‑approximation
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected Adam minimising the smoothed max‑norm."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.5                # a bit larger than before
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # soft‑max temperature schedule
    alpha = 0.3
    alpha_mul = 1.018

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        cur_max = np.max(conv)

        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # smooth weights and gradient
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a, S)

        # occasional mass‑swap across a few entries
        if rng.random() < 0.03:
            idxs = rng.choice(n, size=5, replace=False)
            # move a small fraction from the largest of the chosen to the smallest
            i_max = idxs[np.argmax(a[idxs])]
            i_min = idxs[np.argmin(a[idxs])]
            delta = 0.03 * a[i_max] * rng.random()
            cand = a.copy()
            cand[i_max] -= delta
            cand[i_min] += delta
            cand = _project_simplex(cand, S)
            cand_max = np.max(_full_convolution(cand))
            if cand_max < best_score or rng.random() < 0.1:
                a = cand

        # temperature & learning‑rate adaptation
        alpha = min(alpha * alpha_mul, 30.0)
        lr *= 0.9997

        # early stopping if stagnant
        if it % 2500 == 0 and best_score == cur_max:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient phase (focus on the true max‑norm)
# ----------------------------------------------------------------------
def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the true max‑norm (only the worst entry)."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.15                # a slightly larger base step
    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient: reversed copy aligned with the worst shift
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # occasional random perturbation (swap mass between two indices)
        if rng.random() < 0.05:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.04 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.2:
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()

        if it % 5000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimiser (Adam → sub‑gradient)
# ----------------------------------------------------------------------
def _optimize_combined(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run a smooth Adam phase for ~60 % of the remaining budget,
    then a sub‑gradient refinement for the rest.
    """
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    smooth_deadline = now + 0.6 * remaining
    a_s, score_s = _optimize_adam(a0, smooth_deadline, rng)

    if time.time() < deadline:
        a_f, score_f = _optimize_subgrad(a_s, deadline, rng)
        if score_f < score_s:
            return a_f, score_f
    return a_s, score_s
