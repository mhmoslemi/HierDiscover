"""

self‑convolution peak returned by ``evaluate_sequence``.

Algorithm
---------
1.  Randomly pick a problem size ``n`` (2000‑8000) and initialise a
    candidate on the L¹‑simplex Σaᵢ = √(2 n).  Initialisations are
    * uniform,
    * biased (gamma with shape <1 → sparse),

    approximation of ``max(conv)``.  The temperature (soft‑max sharpness)
    is increased exponentially (annealing) so that the optimisation
    gradually focuses on the true L∞‑norm.
3.  **Phase B – true sub‑gradient descent** – projected sub‑gradient on
    the exact max‑entry (the standard Chebyshev sub‑gradient).  The step
    size follows a 1/√t rule with a small back‑off.
4.  **Phase C – local random mass‑swaps** – a cheap hill‑climbing loop
    that moves a tiny amount of mass from one index to another and
    accepts the move if it lowers the peak.  This refines the solution
    found by the first two phases.
5.  The whole procedure is wrapped in many short trials (≈20 s each)
    within the total budget (≈1000 s).  Whenever a trial improves the
    best score the solution is stored and optionally up‑sampled for the
    next trials, giving the optimiser more degrees of freedom.

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector *v* onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth max ‑norm.
    Returns 2 · ∑_i w_{i+j} a_i for each j (size = len(a)).
    """
    n = a.shape[0]
    # reverse weights because convolution with reversed kernel gives the required cross‑correlation
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1            # = 3 n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    # pick the slice that corresponds to the gradient entries
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights (smooth approximation of max)
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    w_i = exp(alpha·conv_i) / Σ_j exp(alpha·conv_j)
    with a stable implementation.
    """
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Linear up‑sampling (interpolation) – used when we enlarge a vector
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate vector *a* linearly to length *new_n*."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=float)
    return np.interp(x_new, x_old, a)


# ----------------------------------------------------------------------
# Phase A – Adam on a smooth approximation of the L∞‑norm
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Adam optimiser on a soft‑max approximation of max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    # Adam hyper‑parameters (tuned for this problem)
    lr = 0.4
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # start with a very smooth soft‑max
    alpha = 0.3
    # store best seen
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    while time.time() < deadline:
        t += 1
        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # simplex projection
        a = _project_simplex(a, target_sum)

        # occasional small random mass‑swap for extra exploration
        if rng.random() < 0.03:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.008
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, target_sum)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # increase temperature – the soft‑max becomes sharper
        alpha = min(alpha * 1.03, 500.0)

        # cheap early‑stop
        if t % 4000 == 0 and np.abs(best_score - cur_score) < 1e-7:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase B – Projected sub‑gradient descent on the true max‑norm
# ----------------------------------------------------------------------
def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent that directly minimises max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.3  # base step size
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient for the max entry
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, target_sum)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional random jitter to escape plateaus
        if rng.random() < 0.04:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.015
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, target_sum)
                cand_score = np.max(_full_convolution(cand))
                if cand_score < cur_score or rng.random() < 0.2:
                    a = cand
                    cur_score = cand_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()

        if it % 6000 == 0 and np.abs(best_score - cur_score) < 1e-7:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase C – cheap local random mass‑swap hill‑climbing
# ----------------------------------------------------------------------
def _local_search(a0: np.ndarray,
                  cur_score: float,
                  deadline: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Perform many tiny random moves (move a small amount of mass from i to j).
    Accept only if the peak improves.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_sc = cur_score

    while time.time() < deadline:
        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if a[i] == 0.0:
            continue
        # move a *very* small fraction of mass
        delta = a[i] * rng.random() * 0.005
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # projection is not needed – we kept the sum and non‑negativity
        cand_sc = np.max(_full_convolution(cand))
        if cand_sc < best_sc:
            best_sc = cand_sc
            best = cand.copy()
            a = cand   # continue from the improved point
    return best, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step sequence that minimises
    ``evaluate_sequence``.  The routine combines
    * randomised initialisation,
    * Adam optimisation of a smoothed max‑norm,
    * true sub‑gradient descent,
    * short local random‑swap refinements,
    * progressive up‑sampling.
    Returns the best vector as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # keep a small safety margin for the final external call
    deadline = start + budget_s - 5.0

    best_seq = None
    best_score = np.inf
    last_improve = start
    attempts = 0

    # ------------------------------------------------------------------
    # Helper to build a fresh random start (different flavours)
    # ------------------------------------------------------------------
    def _random_start(n):
        mode = rng.integers(0, 3)
        if mode == 0:          # plain uniform
            a0 = rng.random(n)
        elif mode == 1:        # sparse gamma (shape < 1)
            a0 = rng.gamma(shape=0.4, scale=1.0, size=n)
        else:                  # constant (useful when up‑sampling)
            a0 = np.full(n, 1.0)
        # scale to the correct simplex
        target = np.sqrt(2.0 * n)
        if a0.sum() == 0.0:
            a0 = np.full(n, 1.0)
        a0 = a0 / a0.sum() * target
        return a0

    # ------------------------------------------------------------------
    # Main optimisation loop (many independent trials)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        attempts += 1

        # ------------------------------------------------------------------
        # Choose problem size and start point
        # ------------------------------------------------------------------
        if best_seq is None or (time.time() - last_improve > 35):
            # fresh random trial
            n = int(rng.integers(1800, 6200))
            a_start = _random_start(n)
        else:
            # sometimes enlarge the current best
            if rng.random() < 0.4 and len(best_seq) < 8000:
                n = min(int(len(best_seq) * 1.4), 8000)
                a_start = _upsample(best_seq, n)
                # add minute jitter to break perfect symmetry
                jitter = rng.normal(scale=0.001 * np.mean(best_seq), size=n)
                a_start = np.clip(a_start + jitter, 0.0, None)
            else:
                n = len(best_seq)
                a_start = best_seq.copy()
                # random local jitter before optimisation
                jitter = rng.normal(scale=0.0015 * np.mean(best_seq), size=n)
                a_start = np.clip(a_start + jitter, 0.0, None)

            # re‑normalise after any manipulation
            target = np.sqrt(2.0 * n)
            if a_start.sum() == 0.0:
                a_start = np.full(n, 1.0)
            a_start = a_start / a_start.sum() * target

        # ------------------------------------------------------------------
        # Give each trial a time slice (≈20 s) and run the three phases
        # ------------------------------------------------------------------
        slice_deadline = min(time.time() + 20.0, deadline)

        # Phase A – Adam on smooth max
        a_mid, sc_mid = _optimize_adam(a_start, slice_deadline, rng)

        # Phase B – true sub‑gradient (if time permits)
        if time.time() < slice_deadline:
            a_sub, sc_sub = _optimize_subgrad(a_mid, slice_deadline, rng)
        else:
            a_sub, sc_sub = a_mid, sc_mid

        # Phase C – lightweight local search (few seconds)
        if time.time() < deadline:
            local_deadline = min(time.time() + 4.0, deadline)
            a_local, sc_local = _local_search(a_sub, sc_sub, local_deadline, rng)
        else:
            a_local, sc_local = a_sub, sc_sub

        # ------------------------------------------------------------------
        # Keep the globally best solution
        # ------------------------------------------------------------------
        if sc_local < best_score:
            best_score = sc_local
            best_seq = a_local.copy()
            last_improve = time.time()
            print(f"[global] attempt={attempts} n={n} new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check & return
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – a single unit step (guaranteed to be valid)
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()

    # final external evaluation (the harness will also call it)
    external_score = evaluate_sequence(final_seq)
    print(f"[final] internal best = {best_score:.6f}, external = {external_score:.6f}, "
          f"length = {len(final_seq)}")
    return final_seq
