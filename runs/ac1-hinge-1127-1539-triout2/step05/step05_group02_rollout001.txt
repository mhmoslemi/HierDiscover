def _softmax_gradient(a, beta):
    n = a.size
    size = 1 << ((2*n-1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()
    # compute gradient: sum_k p_k * derivative of conv(k) w.r.t a

def _adam_optimize(a0, deadline, beta_initial=1.0, beta_max=1024.0):
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_score = _score(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = target_sum * 0.1  # base learning rate
    beta = beta_initial
    iteration = 0
    while time.time() < deadline:
        # optionally adjust beta based on iteration or time
        # increase schedule
        if iteration % 1000 == 0 and beta < beta_max:
            beta = min(beta_max, beta * 1.5)
        g, _ = _softmax_gradient(a, beta)
        # Adam step
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**(iteration + 1))
        v_hat = v / (1 - beta2**(iteration + 1))
        step = lr0 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        # project onto simplex
        a = _project_to_simplex(a, target_sum)
        # evaluate
        score = _score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{time.time():.1f}] Adam iteration {iteration}, beta={beta:.2f}, best={best_score:.6f}")
        # occasional jitter
        if iteration % 5000 == 0 and iteration > 0:
            # add small noise
            jitter = np.random.default_rng().normal(scale=target_sum * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0)
            a = _project_to_simplex(a, target_sum)
        iteration += 1
    return best_a, best_score

def _param_search(n, deadline, rng):
    best_seq = None
    best_score = float('inf')
    target = np.sqrt(2.0 * n)
    # define ranges
    p_vals = np.arange(0.30, 1.20+1e-9, 0.02)  # inclusive
    lam_vals = np.array([0.0, 0.0005, 0.001, 0.002, 0.005])
    for p in p_vals:
        # precompute power vector for speed
        power = (np.arange(1, n+1, dtype=np.float64) ** (-p))
        for lam in lam_vals:
            # compute shape
            a = power * np.exp(-lam * np.arange(n, dtype=np.float64))
            # normalize to target sum
            a = a * (target / a.sum())
            score = _score(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
    return best_seq, best_score

"""
Step‑function optimiser.

The algorithm works in three stages:

1. **Coarse parametric sweep** – for several lengths `n` we evaluate
   simple monotone families
       a_i ∝ (i+1)^(-p)·exp(-λ·i)
   over a grid of exponents `p` and decay rates `λ`.  The best shape
   (after projection onto the ℓ¹‑simplex Σa = √(2·n)) is kept.

2. **Projected Adam optimisation** – starting from the best candidate
   we run a projected Adam optimiser on a soft‑max approximation of the
   convolution maximum.  The temperature `β` is increased gradually up to
   1024, so the approximation converges to the true max.  Feasibility is
   enforced by Euclidean projection onto the simplex after each update.

3. **Light simulated‑annealing refinement** – a final short SA pass
   (mass‑transfer, local smoothing, random swaps) removes any remaining
   local irregularities.

All heavy work uses FFT‑based convolutions, which makes the approach fast
even for sequences of several thousand entries.  The routine respects
the 1000 s budget and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic building blocks
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s} (Duchi et al., 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (full linear convolution)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact objective (same as evaluate_sequence up to clipping)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    Returns (gradient, soft‑max probabilities) for the current beta.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * (a_rev * p)_{central n entries}
    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    p_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[: 2 * n - 1]
    g = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# 1️⃣  Coarse parametric sweep
# ----------------------------------------------------------------------


def _parametric_sweep(n: int, deadline: float, rng: np.random.Generator):
    """
    Search the 2‑parameter family
        a_i ∝ (i+1)^(-p) * exp(-λ·i)
    over a modest grid.  Returns the best sequence found for length n.
    """
    target = np.sqrt(2.0 * n)
    best_seq = None
    best_val = float("inf")

    # grid (chosen to be cheap yet expressive)
    p_vals = np.arange(0.30, 1.25 + 1e-12, 0.02)   # exponent of the power law
    lam_vals = np.array([0.0, 0.0005, 0.001, 0.002, 0.005])

    idx = np.arange(n, dtype=np.float64)
    i_plus = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        if time.time() >= deadline:
            break
        base = i_plus ** (-p)          # power‑law part (vectorised)
        for lam in lam_vals:
            if time.time() >= deadline:
                break
            a = base * np.exp(-lam * idx)
            # projection onto the ℓ¹‑simplex Σa = √(2·n)
            a = _project_to_simplex(a, target)
            val = _score(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# 2️⃣  Projected Adam optimiser on the soft‑max surrogate
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the soft‑max surrogate of max(conv).
    The temperature β is raised gradually up to 1024.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = target_sum * 0.02         # base learning rate
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    pow1, pow2 = beta1, beta2      # bias‑correction denominators

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    beta_max = 1024.0
    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        # increase temperature slowly
        if it % 600 == 0 and beta < beta_max:
            beta = min(beta_max, beta * 1.5)

        # gradient of the soft‑max surrogate
        g, _ = _softmax_gradient(a, beta)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # bias‑corrected moments
        m_hat = m / (1.0 - pow1)
        v_hat = v / (1.0 - pow2)

        # parameter update + projection
        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_to_simplex(a, target_sum)

        # evaluate
        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] Adam it={it} β={beta:.1f} best={best_val:.6f}")

        # occasional jitter to escape plateaus
        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        # update bias‑correction denominators
        pow1 *= beta1
        pow2 *= beta2
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# 3️⃣  Light simulated‑annealing refinement (optional)
# ----------------------------------------------------------------------


def _simulated_annealing(
    a0: np.ndarray,
    deadline: float,
    start_temp: float = 0.5,
    end_temp: float = 1e-5,
    max_frac_start: float = 0.40,
    max_frac_end: float = 0.005,
    max_window: int = 30,
) -> tuple[np.ndarray, float]:
    """
    A fast SA pass that moves mass between indices, performs local smoothing,
    and occasionally swaps entries.  The schedule is exponential in both
    temperature and maximal transferred fraction.
    """
    a = a0.copy()
    n = a.size
    cur_score = _score(a)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)

    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()
    conv = _conv_fft(a)  # current convolution

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # locate current maximum of the convolution
        max_b = conv.max()
        k_max = int(np.argmax(conv))

        # sensitivity for the active entry (how much each coordinate contributes)
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]

        weight = a * sens

        move_type = None
        # ---- mass transfer -------------------------------------------------
        if rng.random() < 0.70:
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight / total_w)
            else:
                donor = rng.integers(n)

            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            receiver = (
                rng.choice(low_idx) if low_idx.size else rng.integers(n)
            )
            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        # ---- local smoothing -----------------------------------------------
        elif rng.random() < 0.90:
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                saved_seg = a[start:end].copy()
                avg = saved_seg.mean()
                a[start:end] = avg
                move_type = "smooth"
                saved_start, saved_end = start, end

        # ---- random swap ---------------------------------------------------
        else:
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # evaluate new state (single FFT)
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = _score(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA new best = {best_score:.6f}")
        else:
            # revert move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter
        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (a.sum() / a.sum())  # renormalise (does nothing but keeps pattern)
            a = _project_to_simplex(a, np.sqrt(2.0 * n))
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    The routine respects the given time budget (default 1000 s) and
    returns the best sequence discovered.
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – parametric sweep over several lengths
    # ------------------------------------------------------------------
    lengths = [500, 1000, 2000, 4000, 8000]   # richer families for larger time
    for n in lengths:
        if time.time() >= deadline:
            break

        seq, val = _parametric_sweep(n, deadline, rng)
        if seq is not None and val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] Sweep n={n} → best={best_score:.6f}")

        # also try the provided global construction if available
        if "height_sequence_1" in globals():
            glob = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if glob.size > n:
                glob_n = glob[:n]
            else:
                glob_n = np.pad(glob, (0, n - glob.size), "constant")
            target = np.sqrt(2.0 * n)
            glob_n = _project_to_simplex(glob_n, target)
            valg = _score(glob_n)
            if valg < best_score:
                best_score = valg
                best_seq = glob_n.copy()
                print(f"[{time.time():.1f}] Global seq n={n} → best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam refinement on the best shape found
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        seq_opt, val_opt = _adam_optimize(best_seq, deadline)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = seq_opt.copy()
            print(f"[{time.time():.1f}] Adam refinement → best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – optional light simulated‑annealing pass (if time left)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        seq_sa, val_sa = _simulated_annealing(best_seq, deadline)
        if val_sa < best_score:
            best_score = val_sa
            best_seq = seq_sa.copy()

    # ------------------------------------------------------------------
    # Final tidy‑up
    # ------------------------------------------------------------------
    if best_seq is None:
        # Defensive fallback – should never happen
        return []

    # Remove trailing near‑zero entries (they do not affect the score)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    print(f"[{time.time():.1f}] Final best score={best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
