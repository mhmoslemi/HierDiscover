def _lp_step(f, M, S):
    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)
    conv = cp.conv(cp.Constant(f), g)  # length 2n-1
    constraints = [conv <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    # Solve with ECOS
    prob.solve(solver=cp.ECOS, abstol=1e-7, reltol=1e-7, max_iters=2000, verbose=False)
    if g.value is None:
        # fallback fallback: return f (no improvement)
        return f
    g_val = np.array(g.value, dtype=float)

def _lp_step(f, M, S):
    n = int(f.shape[0])
    g = cp.Variable(n, nonneg=True)
    conv = cp.conv(cp.Constant(f), g)  # shape 2n-1
    constraints = [conv <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    # Solve
    try:
        prob.solve(solver=cp.ECOS, abstol=1e-7, reltol=1e-7, max_iters=2000, verbose=False)
    except Exception as e:
        # fallback: return original f
        return f
    if g.value is None:
        return f
    g_val = np.maximum(np.array(g.value, dtype=float), 0.0)
    sum_g = g_val.sum()
    if sum_g == 0:
        return f
    factor = S / sum_g
    # Since factor <= 1 (sum_g >= S), this reduces magnitude
    g_scaled = g_val * factor
    return g_scaled

"""
Hybrid optimizer for the step‑function attack.

The algorithm works in three stages for each random restart:

1. **Warm‑up** – an Adam optimiser on a τ‑soft‑max approximation of the
   ∞‑norm of the self‑convolution (the same machinery used in the
   previous state‑of‑the‑art).  This quickly finds a decent shape.

2. **Linear‑programming boost** – with the current sequence `f` we compute
   `M = max(f*f)`.  Keeping `M` fixed we solve the LP

        maximise   Σ g_j
        subject to  (f * g)_k ≤ M   for all k
                    g_j ≥ 0

   (the convolution is linear in `g`).  The optimiser (cvxpy/ECOS) returns a
   vector `g` with a larger ℓ₁‑norm than `f`.  After scaling `g` back to the
   required total mass we mix it with `f` (50 % weight).  The mixing strictly
   reduces the peak of the self‑convolution, and the step is repeated a few
   times.

3. **Fine‑tuning** – a short second Adam run cleans up any remaining
   irregularities.

Multiple independent restarts with random lengths (≈ 2 k–3.5 k) and random
edge‑heavy initial shapes are performed until the time budget expires.
The best sequence seen is stored in the global variable `height_sequence_1`
and returned as a plain Python list.
"""

import time
import math
import numpy as np

# optional linear‑programming backend
try:
    import cvxpy as cp
except Exception:            # pragma: no cover
    cp = None

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of `a`."""
    return float(_full_convolution_fft(a).max())


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Returns (grad, max_b) where
        grad = d/da τ‑soft‑max(b) ,  b = a*a,
        max_b = max(b) (hard maximum).
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        w = exps / np.sum(exps)          # soft‑max weights, length 2n‑1
        mask = w

    # gradient w.r.t. a:
    a_rev = a[::-1]
    conv_len = len(mask) + n - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Minimise the τ‑soft‑max of the self‑convolution.
    Returns the best vector encountered and the corresponding hard peak.
    """
    n = a0.shape[0]
    S = math.sqrt(2.0 * n)

    # initialise logits from a0 (they will stay on the simplex)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                     # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.2

    # smoothing schedule
    tau = max(_full_convolution_fft(a0).max(), 1.0)

    best_a = a0.copy()
    best_peak = _max_conv(a0)

    it = 0
    while time.time() < deadline:
        it += 1

        # ------- a from logits (simplex) -------
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ------- convolution and gradient -------
        g, peak = _softmax_gradient(a, tau)

        # keep the best (hard) peak seen
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ------- gradient wrt logits -------
        mu = np.dot(a, g) / S            # scalar
        grad_p = a * (g - mu)            # shape (n,)

        # optional clipping (stability)
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ------- Adam update -------
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ------- schedules -------
        tau *= 0.9995               # tighten soft‑max
        lr *= 0.9999                # learning‑rate decay

        # occasional jitter in log‑space
        if it % 500 == 0:
            p += rng.normal(scale=0.02, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Edge‑heavy initialisation
# ----------------------------------------------------------------------
def generate_initial(n: int,
                     rng: np.random.Generator,
                     base_seq: np.ndarray | None = None) -> np.ndarray:
    """
    Produce a feasible start vector of length `n`.
    Mixes an edge‑heavy power‑law shape with a few random alternatives.
    The result is scaled to the required ℓ₁‑norm S = √(2n).
    """
    S = math.sqrt(2.0 * n)

    shape = rng.choice(['edge', 'uniform', 'random'])
    if shape == 'uniform':
        a = np.full(n, 1.0)
    elif shape == 'random':
        a = rng.random(n) + 0.1
    else:  # edge‑heavy
        p_exp = rng.uniform(0.7, 2.0)
        idx = np.arange(n)
        d = np.minimum(idx, n - 1 - idx) + 1          # distance to nearest edge (≥1)
        a = (d.astype(float) ** (-p_exp))

    # 20 % chance to re‑use the global best (if same length)
    if base_seq is not None and base_seq.shape[0] == n and rng.random() < 0.2:
        a = base_seq.copy()

    # project to the simplex with prescribed ℓ₁‑norm
    a = np.clip(a, 0.0, None)
    if a.sum() == 0:
        a = np.full(n, 1.0)
    a = a / a.sum() * S
    return a


# ----------------------------------------------------------------------
# Linear‑programming step (maximise ℓ₁‑norm under convolution bound)
# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray,
             M: float,
             S: float,
             rng: np.random.Generator) -> np.ndarray:
    """
    Solve   maximise Σ g_j   subject to   (f * g)_k ≤ M,   g ≥ 0.
    The returned vector is scaled back to ℓ₁‑norm S (S = √(2n)).
    If cvxpy is unavailable the original `f` is returned unchanged.
    """
    if cp is None:          # pragma: no cover
        return f

    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)

    # full linear convolution (constant kernel f)
    conv = cp.conv(cp.Constant(f), g)          # length 2n‑1

    prob = cp.Problem(cp.Maximize(cp.sum(g)),
                      [conv <= M])
    try:
        prob.solve(solver=cp.ECOS, abstol=1e-7, reltol=1e-7,
                   max_iters=2000, verbose=False)
    except Exception:        # pragma: no cover
        return f

    if g.value is None:      # pragma: no cover
        return f

    g_val = np.maximum(np.asarray(g.value, dtype=float), 0.0)
    sum_g = g_val.sum()
    if sum_g == 0.0:
        return f

    # scaling factor ≤ 1 (since sum_g ≥ S for any feasible point)
    factor = S / sum_g
    g_scaled = g_val * factor
    return g_scaled


# ----------------------------------------------------------------------
# LP‑driven improvement loop
# ----------------------------------------------------------------------
def improve_via_lp(f_start: np.ndarray,
                   S: float,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Repeatedly apply the LP step and mix the result with the current
    sequence.  Stops when the time budget expires or when no improvement
    is observed for a few iterations.
    Returns the best sequence and its peak convolution.
    """
    best_f = f_start.copy()
    best_M = _max_conv(best_f)
    stagnation = 0
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        g = _lp_step(best_f, best_M, S, rng)

        # convex combination (50 % weight) and renormalisation
        new_f = 0.5 * best_f + 0.5 * g
        new_f = np.clip(new_f, 0.0, None)
        new_f = new_f / new_f.sum() * S

        new_M = _max_conv(new_f)
        if new_M < best_M - 1e-12:
            best_f, best_M = new_f, new_M
            stagnation = 0
            print(f"[LP] iter={iteration} improved M={best_M:.6f}")
        else:
            stagnation += 1
            if stagnation >= 3:
                break
            # tiny random jitter to escape possible plateaus
            jitter = rng.normal(scale=0.01, size=best_f.shape)
            best_f = best_f + jitter
            best_f = np.clip(best_f, 0.0, None)
            best_f = best_f / best_f.sum() * S
            best_M = _max_conv(best_f)

    return best_f, best_M


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence().  The routine performs many short independent
    restarts, each consisting of:
        * Adam smoothing,
        * LP‑based peak reduction,
        * a final short Adam polish.
    The best candidate found is stored in ``height_sequence_1`` and
    returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0      # safety margin

    # Load previously discovered construction if it exists
    base_seq = globals().get('height_sequence_1')
    if base_seq is not None:
        try:
            base_seq = np.asarray(base_seq, dtype=float)
        except Exception:  # pragma: no cover
            base_seq = None

    best_seq = None
    best_score = np.inf
    restart = 0

    while time.time() < end_time:
        restart += 1

        # pick a length (2 k – 3.5 k gives a good trade‑off)
        n = int(rng.integers(2000, 3500))
        S = math.sqrt(2.0 * n)

        # initial feasible vector
        a0 = generate_initial(n, rng, base_seq)

        # --- stage 1: Adam warm‑up (≈6 s) ---
        t1_dead = min(end_time, time.time() + 6.0)
        a1, peak1 = _adam_optimize(a0, t1_dead, rng)

        # --- stage 2: LP boost (≈6 s) ---
        t2_dead = min(end_time, time.time() + 6.0)
        a2, peak2 = improve_via_lp(a1, S, t2_dead, rng)

        # --- stage 3: short Adam polish (≈2 s) ---
        t3_dead = min(end_time, time.time() + 2.0)
        a3, peak3 = _adam_optimize(a2, t3_dead, rng)

        # final score (with the normalisation the evaluate() value equals the peak)
        score = 2.0 * n * peak3 / (np.sum(a3) ** 2)   # = peak3
        if score < best_score:
            best_score = score
            best_seq = a3.copy()
            print(f"[candidate] restart={restart} n={n} score={best_score:.6f}")

    # fallback in the unlikely case that nothing was produced
    if best_seq is None:
        n = 2500
        S = math.sqrt(2.0 * n)
        best_seq = np.full(n, S / n)

    # final sanity check with the black‑box evaluator
    external = evaluate_sequence(best_seq.tolist())
    print(f"[final] external={external:.6f} internal={best_score:.6f} length={len(best_seq)}")
    globals()['height_sequence_1'] = best_seq.tolist()
    return best_seq.tolist()
