"""
Propose a step‑function (non‑negative sequence) that minimizes the official
``evaluate_sequence`` metric.

The optimiser combines three ideas:

1. **Diverse initialisation** – a mixture of random draws (exponential,
   power‑law, Dirichlet, piecewise, etc.) together with a few deterministic
   families (pure exponential, pure power‑law, linear decay).  Every candidate
   is scaled to the canonical ℓ¹‑norm  Σa = √(2 n)  (the objective is
   homogeneous, so this normalisation does not affect the score).

2. **Two‑phase descent** –
   * Phase 1 : a smooth‑max (log‑sum‑exp) projected gradient descent that
     approximates the max‑norm of the self‑convolution.
   * Phase 2 : exact sub‑gradient descent on the true max‑norm together with
     a cheap “mass‑balancing’’ move that shifts mass from indices contributing
     most to the current peak to lightly‑filled indices.

   Both phases respect the simplex constraints (non‑negativity and fixed
   ℓ¹‑norm).

3. **Final polishing** – a short simulated‑annealing run that swaps small
   fractions of mass between two positions; this helps to escape tiny local
   minima.

The whole search respects a 1000 s wall‑clock budget.  Time is split among
different sequence lengths (2000…10000) and the various phases; the routine
keeps the best sequence ever seen and returns it as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1, dtype=float)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·n−1) using an FFT.
    Returned array is real‑valued.
    """
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Objective after ℓ¹‑normalisation Σa = √(2 n):
    simply the supremum norm of the self‑convolution.
    """
    return float(_max_convolution_fft(a).max())


# ----------------------------------------------------------------------
# Random initialisation helpers (same families as in the reference solution)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative point on the simplex Σa = √(2 n).
    Several families (power‑law, exponential, Dirichlet, piecewise) are
    sampled to increase diversity.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                                 # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                               # pure exponential
        lam = rng.exponential(scale=0.02)
        a = np.exp(-lam * idx)

    elif family == 2:                               # power × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.02)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                               # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n))

    elif family == 4:                               # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                             np.exp(-lam2 * np.arange(b, n))])

    else:                                           # almost uniform with jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


# ----------------------------------------------------------------------
# Deterministic seed families (useful as strong starting points)
# ----------------------------------------------------------------------


def _seed_exponential(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(-c·i)   (monotone decreasing)."""
    idx = np.arange(n, dtype=float)
    a = np.exp(-c * idx)
    return a


def _seed_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}   (monotone decreasing)."""
    idx = np.arange(n, dtype=float)
    a = (idx + 1.0) ** (-p)
    return a


def _seed_linear_decay(n: int) -> np.ndarray:
    """a_i ∝ (n-i)   (simple linear drop)."""
    return np.arange(n, 0, -1, dtype=float)


# ----------------------------------------------------------------------
# Soft‑max gradient (log‑sum‑exp) – Phase 1
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) where
        g = ∇_a  ( 1/β·log Σ exp(β·b) )
          = 2·(a ⋆ p)          with   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # ---- b = a*a -------------------------------------------------------
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ---- soft‑max probabilities p --------------------------------------
    max_b = float(np.max(b))
    exps = np.exp(beta * (b - max_b))          # shift for numerical stability
    p = exps / np.sum(exps)

    # ---- gradient g = 2·(a ⋆ p) ----------------------------------------
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1            # = 3 n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _smooth_max_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Annealed projected gradient descent on the smooth max.
    Uses momentum, an adaptive step and occasional Gaussian jitter.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 4096.0
    beta_factor = 1.4

    step_base = target_sum * 0.15          # fraction of the ℓ¹‑norm
    momentum = np.zeros_like(a)
    rho = 0.90                               # momentum decay

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        # simplex projection (keeps Σa = target_sum)
        a = _project_to_simplex(a, target_sum)

        # a little jitter every few hundred iterations
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal β
        if it % 350 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a


# ----------------------------------------------------------------------
# Exact max‑index sub‑gradient + cheap mass‑balancing – Phase 2
# ----------------------------------------------------------------------


def _max_index_subgradient(a: np.ndarray):
    """
    Gradient of max(b) where b = a*a.
    The active index k* = argmax b is used; the sub‑gradient is
        g_i = 2·a_{k*‑i} (if the index is inside the support, else 0).
    """
    n = a.shape[0]
    conv = _max_convolution_fft(a)
    k = int(np.argmax(conv))

    g = np.zeros_like(a)
    if k < n:                         # left half of the convolution
        g[:k + 1] = 2.0 * a[k::-1]
    else:                             # right half
        d = k - (n - 1)               # shift amount (0 … n‑1)
        g[d:] = 2.0 * a[: n - d]
    return g, k


def _subgradient_descent_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected sub‑gradient descent on the true max‑norm (no smoothing).
    A diminishing step‑size  η_t = α / √t is used.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        g, _ = _max_index_subgradient(a)

        # diminishing step size
        step = 0.8 * target_sum / (np.sqrt(it + 1) * np.linalg.norm(g) + 1e-12)
        a -= step * g

        # enforce feasibility
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        # occasional tiny jitter to avoid cycling
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1
    return a


def _mass_balance_refine(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Fast greedy move that explicitly reduces the contribution of the
    currently maximal convolution entry.  At each step a small fraction
    of the “heavy’’ coefficient (the one that participates most in that
    entry) is transferred to a “light’’ location.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sum(a)               # should already be √(2 n)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        # full convolution (needed to locate the maximal entry)
        conv = _max_convolution_fft(a)
        k = int(np.argmax(conv))

        # contribution of each index to that entry: a_i * a_{k‑i}
        idx = np.arange(n)
        other = k - idx
        mask = (other >= 0) & (other < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[other[mask]]

        i_high = int(np.argmax(contrib))                # most offending index
        # pick a relatively empty position – among the 20 % smallest masses
        low_candidates = np.where(a <= np.percentile(a, 20))[0]
        if low_candidates.size == 0:
            i_low = int(np.argmin(a))
        else:
            i_low = int(rng.choice(low_candidates))

        # move a modest fraction (3 %) of the heavy mass
        delta = a[i_high] * 0.03
        if delta < 1e-12:
            break

        a[i_high] -= delta
        a[i_low]  += delta

        # keep feasibility (non‑negativity + ℓ¹‑norm)
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        if it % 100 == 0:
            # very occasional global jitter
            if rng.random() < 0.02:
                jitter = rng.normal(scale=target_sum * 1e-3, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                a = _project_to_simplex(a, target_sum)

        it += 1

    return a


# ----------------------------------------------------------------------
# Simple simulated‑annealing (kept from the reference solution)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Very light SA that moves a random fraction of mass between two sites."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    cur_score = _score(a)
    best_score = cur_score

    T = cur_score * 0.02 + 1e-9
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-16:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        frac = rng.random() * 0.15
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta

        new_score = _score(a)
        dE = new_score - cur_score
        accept = dE <= 0.0 or (T > 0.0 and rng.random() < np.exp(-dE / T))

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling

    return best

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` metric.  The routine respects the wall‑clock budget,
    tries several lengths, many random initialisations, deterministic seeds,
    and refines the best candidate with a two‑phase projected descent,
    a cheap mass‑balancing loop and a short simulated‑annealing polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # If a previous construction is present, use it as a seed.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        prev_arr = np.asarray(previous, dtype=float)
        n0 = prev_arr.size
        prev_arr = _project_to_simplex(prev_arr, np.sqrt(2.0 * n0))
        sc0 = _score(prev_arr)
        best_seq = prev_arr.copy()
        best_score = sc0
        print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) score={sc0:.6f}")

    # ------------------------------------------------------------------
    # Lengths to explore – larger n gives more freedom but costs more.
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000, 10000]

    # Helper to scale a raw (non‑negative) vector onto the ℓ¹‑simplex
    def _scale_to_target(a_raw, target):
        s = np.sum(a_raw)
        if s == 0:
            return np.full_like(a_raw, target / a_raw.size)
        return a_raw * (target / s)

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)
        print(f"[{time.time():.1f}] ==== Exploring length {n} (target ℓ¹={target_sum:.3f}) ====")

        # --------------------------------------------------------------
        # 1) Deterministic seeds (few, but strong)
        # --------------------------------------------------------------
        deterministic_families = [
            (_seed_exponential, {"c": 0.015}),
            (_seed_exponential, {"c": 0.020}),
            (_seed_exponential, {"c": 0.025}),
            (_seed_power_law, {"p": 0.6}),
            (_seed_power_law, {"p": 0.9}),
            (_seed_linear_decay, {}),
        ]

        for func, params in deterministic_families:
            a_raw = func(n, **params) if params else func(n)
            a = _scale_to_target(a_raw, target_sum)
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Deterministic seed improved → score={sc:.6f}")

        # --------------------------------------------------------------
        # 2) Random restarts + short smooth‑max GD
        # --------------------------------------------------------------
        # allocate a modest slice of time for cheap exploration
        phase_a_dead = min(deadline,
                           time.time() + min(30.0, (deadline - time.time()) * 0.25))

        while time.time() < phase_a_dead:
            a_start = _random_initial_sequence(n, rng)

            # a very short smooth‑max run (≈2 s) – just to get into a basin
            sub_dead = min(deadline, time.time() + 2.5)
            a_gd = _smooth_max_descent(a_start, sub_dead)
            sc = _score(a_gd)
            if sc < best_score:
                best_score = sc
                best_seq = a_gd.copy()
                print(f"[{time.time():.1f}] Random restart improvement → score={sc:.6f}")

        # --------------------------------------------------------------
        # 3) Deep refinement on the current best of this length
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size == n:
            # --- longer smooth‑max GD (≈120 s or whatever remains) ---
            phase_b_dead = min(deadline, time.time() + min(120.0, deadline - time.time()))
            a_ref = _smooth_max_descent(best_seq, phase_b_dead)
            sc = _score(a_ref)
            if sc < best_score:
                best_score = sc
                best_seq = a_ref.copy()
                print(f"[{time.time():.1f}] Smooth‑max refinement → score={sc:.6f}")

            # --- sub‑gradient descent (≈30 s) ---
            sub_dead = min(deadline, time.time() + 30.0)
            a_ref2 = _subgradient_descent_max(best_seq, sub_dead)
            sc = _score(a_ref2)
            if sc < best_score:
                best_score = sc
                best_seq = a_ref2.copy()
                print(f"[{time.time():.1f}] Sub‑gradient descent → score={sc:.6f}")

            # --- mass‑balance polishing (≈20 s) ---
            sub_dead = min(deadline, time.time() + 20.0)
            a_ref3 = _mass_balance_refine(best_seq, sub_dead)
            sc = _score(a_ref3)
            if sc < best_score:
                best_score = sc
                best_seq = a_ref3.copy()
                print(f"[{time.time():.1f}] Mass‑balance refinement → score={sc:.6f}")

    # ------------------------------------------------------------------
    # Final simulated‑annealing polishing with whatever time remains.
    # ------------------------------------------------------------------
    if best_seq is not None:
        left = deadline - time.time()
        if left > 5.0:
            a_final = _simulated_annealing(best_seq, deadline)
            sc = _score(a_final)
            if sc < best_score:
                best_score = sc
                best_seq = a_final.copy()
                print(f"[{time.time():.1f}] Final SA improvement → score={sc:.6f}")

    # ------------------------------------------------------------------
    # Post‑process: strip trailing zeros and report the official evaluation.
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution of length 1000 (should never happen)
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # official evaluation (includes the security clipping)
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
