def _hill_climbing_improved(a0, b0, deadline):
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()
    best_b = b.copy()
    rng = np.random.default_rng()
    # initial temperature
    T = cur_score * 0.02 + 1e-9
    cooling = 0.9998
    iter_count = 0
    while time.time() < deadline:
        # choose source i proportional to mass
        if a.sum() == 0:
            break
        # compute probabilities maybe use cumulative
        probs = a / a.sum()
        i = rng.choice(n, p=probs)
        if a[i] < 1e-14:
            continue

        # choose destination
        if rng.random() < 0.15:
            # neighbor
            j = (i + (1 if rng.random() < 0.5 else -1)) % n
        else:
            # biased to low a
            if rng.random() < 0.5:
                # uniform
                j = rng.integers(0, n)
            else:
                inv = np.max(a) - a + 1e-12
                probs_j = inv / inv.sum()
                j = rng.choice(n, p=probs_j)

        if i == j:
            continue

        max_delta = a[i] * 0.5
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        # compute new convolution efficiently
        # start with copy
        b_new = b.copy()
        # subtract contribution from i
        b_new[i:i+n] -= 2.0 * delta * a
        # add contribution from j
        b_new[j:j+n] += 2.0 * delta * a

        # quadratic terms
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # Simulated annealing
            if T > 1e-12 and rng.random() < np.exp(-(new_score - cur_score) / T):
                accept = True
        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                best_b = b.copy()
                # print improvement
                print(f"[{time.time():.1f}] hill_climb improvement: {best_score:.6f}")
        T *= cooling
        iter_count += 1
    return best_a, best_b

"""
Step‑function optimizer (new version).

The algorithm works in three phases, repeatedly for a growing sequence
length:

1. **Randomised Adam‑PGD** – a projected Adam optimiser on a smooth
   log‑sum‑exp approximation of the max‑convolution.  The inverse
   temperature β is annealed quickly up to 8192 so that the smooth loss
   converges to the true max.

2. **Mass‑transfer hill‑climbing** – an improved Simulated‑Annealing
   routine that moves a random fraction of mass from a heavy entry to a
   light one (sometimes to a neighbour).  The linear convolution update
   is performed in O(n) using the analytic formula, and an exponential
   temperature schedule accepts occasional uphill moves.

3. **Polishing & restarts** – after the main passes the current best
   sequence is smoothed, a short PGD is applied again and a few random
   restarts are tried within the remaining budget.

All operations keep the vector on the ℓ¹‑simplex
`Σ a_i = sqrt(2·n)` (the normalisation used by the official
`evaluate_sequence`).  The routine respects a wall‑clock budget (default
1000 s) and returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) via FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b)   where   b = a*a .
    Returns (g, b) with g = 2·(a ⊛ p) and pₖ ∝ exp(β·bₖ).
    """
    n = a.shape[0]

    # b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]                       # reversed a
    conv_len = n + (2 * n - 1) - 1        # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Adam‑based projected gradient descent on the smoothed max
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected Adam optimiser on the smooth log‑sum‑exp approximation of
    the max‑convolution.  The algorithm runs until ``deadline``.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature (inverse soft‑max scale)
    beta = 0.5
    beta_max = 8192.0
    beta_factor = 1.4

    it = 1
    rng = np.random.default_rng()

    while time.time() < deadline:
        # gradient of the smooth max
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = target_sum * 0.08                     # base learning‑rate
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        # β annealing (make the smooth loss closer to the true max)
        if it % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1

    return a


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative vector on the ℓ¹‑simplex
    Σa = sqrt(2·n).  Several families (power‑law, exponential,
    mixed, Dirichlet, piecewise, near‑uniform) are sampled.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # pure power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length.
    The result is projected back onto the appropriate simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Simple smoothing helper
# ----------------------------------------------------------------------


def _smooth_sequence(a: np.ndarray) -> np.ndarray:
    """Apply a small 3‑point smoothing kernel and re‑project to the simplex."""
    kernel = np.array([0.25, 0.5, 0.25])
    a_s = np.convolve(a, kernel, mode="same")
    a_s = np.maximum(a_s, 0.0)
    target = np.sqrt(2.0 * a_s.size)
    a_s = _project_to_simplex(a_s, target)
    return a_s


# ----------------------------------------------------------------------
# Mass‑transfer hill climbing (exact convolution updates)
# ----------------------------------------------------------------------


def _hill_climbing_improved(a0: np.ndarray,
                            b0: np.ndarray,
                            deadline: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Simulated‑annealing hill climbing that moves a random fraction of mass
    from a heavy entry to a lighter one (sometimes to a neighbour).
    The full auto‑convolution ``b`` is updated in O(n) using the analytic
    formula, allowing thousands of cheap moves.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()
    best_b = b.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.02 + 1e-9          # initial temperature
    cooling = 0.9998

    while time.time() < deadline:
        if a.sum() == 0:
            break

        # source index – proportional to its mass (big entries donate more)
        i = rng.choice(n, p=a / a.sum())
        if a[i] < 1e-14:
            continue

        # destination index – mixture of uniform, bias to light entries,
        # and occasional neighbour moves.
        rnd = rng.random()
        if rnd < 0.15:                               # neighbour
            j = (i + (1 if rng.random() < 0.5 else -1)) % n
        elif rnd < 0.55:                             # bias to low entries
            inv = np.max(a) - a + 1e-12
            probs = inv / inv.sum()
            j = rng.choice(n, p=probs)
        else:                                        # uniform
            j = rng.integers(0, n)

        if i == j:
            continue

        # transfer up to 50 % of the source mass
        max_delta = a[i] * 0.5
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ------- update convolution analytically -------
        b_new = b.copy()
        # contribution of -δ at i
        b_new[i:i + n] -= 2.0 * delta * a
        # contribution of +δ at j
        b_new[j:j + n] += 2.0 * delta * a

        # quadratic correction terms
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # ------- acceptance test -------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-(new_score - cur_score) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                best_b = b.copy()
                print(f"[{time.time():.1f}] hill_climb improvement: {best_score:.6f}")

        T *= cooling

    return best_a, best_b


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock budget (default 1000 s) and returns
    the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5         # tiny safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Seed from a previously stored construction if available
    # ------------------------------------------------------------------
    earlier = globals().get("height_sequence_1")
    if isinstance(earlier, (list, tuple, np.ndarray)):
        a0 = np.asarray(earlier, dtype=float)
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * a0.size))
        best_seq = a0.copy()
        best_score = _max_convolution_fft(a0)
        print(f"[{time.time():.1f}] Seeded from height_sequence_1 "
              f"(n={a0.size}) score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Length schedule – start modest, then grow
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000, 10000, 12000, 15000]

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # ----- initialise candidate for this length -----
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)
        a_init = _project_to_simplex(a_init, target_sum)

        # ----- allocate time for this length -----
        remaining = deadline - time.time()
        if remaining <= 0:
            break

        # Phase A – Adam‑PGD (≈40 % of the remaining budget, at most 120 s)
        gd_budget = min(remaining * 0.4, 120.0)
        gd_end = time.time() + gd_budget
        a_gd = _adam_gradient_descent(a_init, gd_end)
        score_gd = _max_convolution_fft(a_gd)
        if score_gd < best_score:
            best_score = score_gd
            best_seq = a_gd.copy()
            print(f"[{time.time():.1f}] GD improvement (n={n}) score={score_gd:.6f}")

        # Phase B – hill climbing (≈30 % of the remaining budget)
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        hc_budget = min(remaining * 0.3, 120.0)
        hc_end = time.time() + hc_budget
        b_full = _full_convolution(a_gd)
        a_hc, b_hc = _hill_climbing_improved(a_gd, b_full, hc_end)
        score_hc = b_hc.max()
        if score_hc < best_score:
            best_score = score_hc
            best_seq = a_hc.copy()
            print(f"[{time.time():.1f}] Hill‑climb improvement (n={n}) "
                  f"score={score_hc:.6f}")

        # Phase C – short smooth + PGD (use a tiny slice of the budget)
        remaining = deadline - time.time()
        if remaining > 15.0:
            a_s = _smooth_sequence(best_seq)
            a_s = _project_to_simplex(a_s, np.sqrt(2.0 * a_s.size))
            pgd_end = time.time() + min(10.0, remaining / 2)
            a_s2 = _adam_gradient_descent(a_s, pgd_end)
            score_s = _max_convolution_fft(a_s2)
            if score_s < best_score:
                best_score = score_s
                best_seq = a_s2.copy()
                print(f"[{time.time():.1f}] Post‑smooth GD improvement "
                      f"score={score_s:.6f}")

    # ------------------------------------------------------------------
    # Random restarts for the final length (if time permits)
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        n = best_seq.size if best_seq is not None else 5000
        a_rand = _random_initial_sequence(n, rng)
        a_rand = _project_to_simplex(a_rand, np.sqrt(2.0 * n))
        # short PGD
        pgd_end = time.time() + min(8.0, (deadline - time.time()) * 0.2)
        a_gd = _adam_gradient_descent(a_rand, pgd_end)
        b_full = _full_convolution(a_gd)
        hc_end = deadline
        a_hc, b_hc = _hill_climbing_improved(a_gd, b_full, hc_end)
        score = b_hc.max()
        if score < best_score:
            best_score = score
            best_seq = a_hc.copy()
            print(f"[{time.time():.1f}] Restart hill‑climb improvement "
                  f"score={score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing (remaining budget)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        a_final = best_seq.copy()
        b_final = _full_convolution(a_final)
        a_final, b_final = _hill_climbing_improved(a_final, b_final, deadline)
        final_score = b_final.max()
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final polishing improvement "
                  f"score={final_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros for a compact representation
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback – uniform distribution (should never be needed)
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log the official evaluation (objective = max of the self‑convolution)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
