def exponential_score(n, c):
    λ = c / n
    r = np.exp(-λ)
    # compute sum
    sum_a = (1 - r**n) / (1 - r)
    # scaling factor α
    α = np.sqrt(2.0 * n) / sum_a
    # compute vector m:
    k = np.arange(2*n-1)
    m = np.minimum(k+1, 2*n-1 - k)
    # compute r^k:
    # Use np.exp(k * np.log(r)) for speed
    log_r = -λ
    r_pow_k = np.exp(k * log_r)  # may underflow for large k but fine
    max_val = np.max(m * r_pow_k)
    score = α**2 * max_val
    return score

"""
Simple exponential‑shape construction.

The objective is   2·n·max(a*a) / (sum(a)²) .
If we enforce Σa = √(2 n) the factor 2 n / Σa² equals 1,
so the score is exactly the maximal entry of the self‑convolution.
For a geometric progression a_i = α·e^{‑λ i} we can compute the
convolution analytically:

    (a*a)_k = α²·r^k·m(k) ,   r = e^{‑λ},
    m(k) = min(k+1, 2 n‑1‑k).

Choosing λ = (ln n)/n (i.e. c = ln n) balances the two
potentially dominant entries – the leftmost term α²
and the central term α²·n·r^{n‑1} – and yields the asymptotically
optimal bound ≈ 2·(ln n)² / n.
For n = 20 000 this gives a score ≈ 0.01, far below the target 1.503.

The routine builds such a sequence, fine‑tunes λ by a tiny
grid search and returns the best sequence found before the time

def _exponential_score(n: int, lam: float) -> float:
    """
    Returns the objective value for a pure exponential sequence
    a_i = α·exp(-lam·i) (scaled to Σa = √(2 n)).
    The score is max_k (a*a)_k, because the ℓ¹‑normalisation makes the
    official metric equal to this maximum.
    """
    r = np.exp(-lam)
    # geometric series sum = (1‑rⁿ)/(1‑r)
    # use np.exp for stability when lam is tiny
    sum_geom = (1.0 - r ** n) / (1.0 - r)
    alpha = np.sqrt(2.0 * n) / sum_geom          # scaling factor
    # compute the maximal entry of the convolution:
    #   conv_k = α² * r^k * min(k+1,2n‑1‑k)
    # we can evaluate this vectorised (size 2n‑1 is modest)
    k = np.arange(2 * n - 1, dtype=np.float64)
    m = np.minimum(k + 1.0, (2 * n - 1) - k)
    # use log‐space to avoid underflow for large k
    log_vals = 2.0 * np.log(alpha) + k * np.log(r) + np.log(m)
    max_log = np.max(log_vals)
    return float(np.exp(max_log))


def _build_exponential(n: int, lam: float) -> np.ndarray:
    """
    Build the sequence a_i = α·exp(-lam·i) with Σa = √(2 n).

    r = np.exp(-lam)

    sum_geom = (1.0 - r ** n) / (1.0 - r)
    alpha = np.sqrt(2.0 * n) / sum_geom
    a *= alpha
    return a
