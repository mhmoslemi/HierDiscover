    t=0
    best_a = a.copy()

    while time.time() < deadline:
        t += 1
        max_b, g = _subgradient_max(a)
        if max_b < best_obj:

        v = beta2 * v + (1 - beta2) * (g**2)

"""
Implementation of a refined search for the step‑function that minimises
evaluate_sequence.  The approach consists of

1. a quick deterministic sweep over a few parametric families
   (power‑law, symmetric power‑law, exponential) for several lengths;
   the best ten candidates are kept.

2. each elite candidate is refined by a projected Adam optimiser that
   directly minimises the true maximum of the self‑convolution (sub‑gradient
   of the max).  The optimisation runs until the global deadline is reached
   or a per‑candidate time‑budget expires.

3. the refined vectors are finally polished by a short simulated‑annealing
   mass‑transfer phase.

The algorithm respects the 1000 s time budget, prints progress and finally
returns the best sequence found as a plain ``list[float]``.
"""
import time
import numpy as np

# ---------- Fast convolution utilities ----------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())

def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

# ---------- Projection onto the ℓ¹‑simplex ----------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection onto {x_i ≥ 0, Σx_i = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ---------- Parametric families ----------
def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    """a_i = (i + shift)^{-p}, projected onto the ℓ¹‑simplex."""
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    target = np.sqrt(2.0 * n)
    scale = target / raw.sum()
    a = raw * scale
    return np.maximum(a, 0.0)

def _symmetric_powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    """a_i = (i+shift)^{-p} + (n-1-i+shift)^{-p}, projected onto the ℓ¹‑simplex."""
    idx = np.arange(n, dtype=float)
    left = (idx + shift) ** (-p)
    right = (n - 1 - idx + shift) ** (-p)
    raw = left + right
    target = np.sqrt(2.0 * n)
    scale = target / raw.sum()
    a = raw * scale
    return np.maximum(a, 0.0)

def _exponential_sequence(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam·i), projected onto the ℓ¹‑simplex."""
    idx = np.arange(n, dtype=float)
    raw = np.exp(-lam * idx)
    target = np.sqrt(2.0 * n)
    scale = target / raw.sum()
    a = raw * scale
    return np.maximum(a, 0.0)

# ---------- Sub‑gradient of the max‑convolution ----------
def _subgradient_max(a: np.ndarray):
    """
    Returns (max_b, g) where
        max_b = max_k (a*a)_k
        g      = ∇ max_k (a*a)_k   (any sub‑gradient).
    The sub‑gradient is 2·(a ⋆ m) with m the indicator of the
    maximising index/indices.
    """
    n = a.size
    # convolution a*a
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]
    max_b = float(b.max())
    # mask of maximising entries (tolerance to avoid losing multiple ties)
    tol = max_b * 1e-12 + 1e-15
    mask = (b >= max_b - tol).astype(float)

    # gradient = 2 * conv_rev(a_rev, mask)[n-1:n-1+n]
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    A_rev_fft = np.fft.rfft(a_rev, n=size2)
    M_fft = np.fft.rfft(mask, n=size2)
    conv_rev = np.fft.irfft(A_rev_fft * M_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return max_b, g

# ---------- Adam on the simplex ----------
def _adam_descent_on_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected Adam optimisation that directly minimises max(a*a).
    Stops when time runs out and returns the best vector seen.
    """
    a = a0.copy()
    n = a.shape[0]
    target = np.sqrt(2.0 * n)

    # ensure proper simplex position at start
    a = _project_to_simplex(a, target)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_obj = np.inf
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        max_b, g = _subgradient_max(a)

        # keep best
        if max_b < best_obj:
            best_obj = max_b
            best_a = a.copy()

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size – a modest baseline that scales with the target sum
        base_lr = 0.5 * target
        step = base_lr / (np.sqrt(v_hat) + eps)

        # gradient step
        a -= step * m_hat

        # projection back onto the simplex (keeps non‑negativity)
        a = _project_to_simplex(a, target)

        # occasional small jitter to escape flat regions
        if t % 250 == 0:
            jitter = rng.normal(scale=target * 5e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

    return best_a

# ---------- Simple simulated‑annealing ----------
def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Mass‑transfer SA that respects the simplex constraint."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)
    rng = np.random.default_rng()

    # initialise temperature from current objective
    T = best_score * 0.02 + 1e-8
    cooling = 0.9995

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        # move a fraction of mass from i to j
        frac = rng.random() * 0.15
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta

        cur = _max_convolution_fft(a)
        dE = cur - best_score
        accept = False
        if cur <= best_score:
            accept = True
        elif T > 0.0 and rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur < best_score:
                best_score = cur
                best = a.copy()
        else:
            # revert move
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # very occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=best.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur = _max_convolution_fft(a)
            if cur < best_score:
                best_score = cur
                best = a.copy()

    return best

"""
Implementation of a refined search for the step‑function that minimises
evaluate_sequence.  The algorithm proceeds in three stages:

1. **Parametric sweep** – explore three simple families (power‑law,
   symmetric power‑law, exponential) for several lengths.  The ten
   best candidates are kept.

2. **Projected Adam optimisation** – each elite candidate is refined by a
   sub‑gradient method that directly minimises the true maximum of the
   self‑convolution (the quantity returned by evaluate_sequence when the
   ℓ¹‑norm is normalised).  The optimiser runs until a per‑candidate time
   budget expires or the global deadline is reached.

3. **Simulated‑annealing polishing** – a short random‑mass‑transfer SA
   phase further lowers the peak convolution while always staying on the
   ℓ¹‑simplex.

All helper functions are top‑level; no I/O is performed.  The routine
obeys the 1000 s budget, prints progress information and finally returns
the best sequence as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex Σx = s, x≥0
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection onto {x_i ≥ 0, Σx_i = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (power‑law / symmetric power‑law / exponential)
# ----------------------------------------------------------------------
def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    """a_i = (i + shift)^{-p}, projected onto the ℓ¹‑simplex."""
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    target = np.sqrt(2.0 * n)
    scale = target / raw.sum()
    a = raw * scale
    return np.maximum(a, 0.0)


def _symmetric_powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    """a_i = (i+shift)^{-p} + (n‑1‑i+shift)^{-p}, projected onto the ℓ¹‑simplex."""
    idx = np.arange(n, dtype=float)
    left = (idx + shift) ** (-p)
    right = (n - 1 - idx + shift) ** (-p)
    raw = left + right
    target = np.sqrt(2.0 * n)
    scale = target / raw.sum()
    a = raw * scale
    return np.maximum(a, 0.0)


def _exponential_sequence(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam·i), projected onto the ℓ¹‑simplex."""
    idx = np.arange(n, dtype=float)
    raw = np.exp(-lam * idx)
    target = np.sqrt(2.0 * n)
    scale = target / raw.sum()
    a = raw * scale
    return np.maximum(a, 0.0)
