        total = vec.sum()

"""
Propose a step‑function (non‑negative vector) that minimizes
``evaluate_sequence``.  The method builds on the previous Adam+sub‑gradient
pipeline but adds a few new ideas that usually give a modest but
reliable improvement:

* **Symmetrisation** – the sequence is forced to be symmetric around its
  centre (a_i = a_{n‑1‑i}).  The optimal solution of the max‑convolution
  problem is known to be symmetric, and the extra projection reduces
  the search space and the peak of the autocorrelation.

* **Two‑phase schedule** – a short warm‑up phase with many random restarts
  quickly finds a decent shape.  Afterwards a long fine‑tuning phase
  runs Adam on a soft‑max approximation of the max‑convolution, with a
  slowly increasing temperature (α) and a reduced learning‑rate, finally
  finished by a few sub‑gradient steps.

* **Progressive up‑sampling** – once a good shape for a moderate length
  (≈3‑4 k) is obtained, it is linearly up‑sampled to a larger length
  (up to 8000) where the extra degrees of freedom allow a flatter
  autocorrelation.  The fine‑tuner is run again on the up‑sampled vector.

* **Simplex projection** – the vector is always projected onto the
  L¹‑simplex {x ≥ 0, Σx = √(2·n)}; because the evaluation function is
  scale‑invariant this guarantees that the internal score equals the
  true max‑convolution value.

The whole routine respects the 1000 s time budget, prints progress and
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient term for a soft‑max weighted max‑norm:
        g_j = 2 Σ_i w_{i+j} a_i .
    Both ``a`` (len n) and ``w`` (len 2·n‑1) are real vectors.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights (smooth approximation of max)
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """w_i = exp(alpha·conv_i) / Σ_j exp(alpha·conv_j)  (stable)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Upsampling helper (linear interpolation)
# ----------------------------------------------------------------------


def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a to length new_n."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=float)
    return np.interp(x_new, x_old, a)


# ----------------------------------------------------------------------
# Symmetrisation helper
# ----------------------------------------------------------------------


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force symmetry v[i] = v[-1-i]."""
    return (v + v[::-1]) * 0.5


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max approximation of max(conv)
# ----------------------------------------------------------------------


def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   symmetrize: bool = True) -> tuple[np.ndarray, float]:
    """Adam on a smooth max‑norm approximation with optional symmetry."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters (tuned for fine‑tuning)
    lr = 0.07
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    # start with a fairly smooth soft‑max and increase temperature
    alpha = 0.4

    while time.time() < deadline:
        t += 1
        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # optional symmetry projection
        if symmetrize:
            a = _symmetrize(a)

        # simplex projection
        a = _project_simplex(a, S)

        # occasional random mass‑swap for exploration
        if rng.random() < 0.03:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.015
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # raise temperature slowly (sharper soft‑max)
        alpha = min(alpha * 1.015, 400.0)

        # cheap early stop when no improvement for a while
        if t % 6000 == 0 and best_score == cur_score:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient optimiser (acts on the true max‑norm)
# ----------------------------------------------------------------------


def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      symmetrize: bool = True) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that directly reduces the worst entry."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.23
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # true sub‑gradient of (a*a)[k_max] (ordered pairs)
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        # each ordered pair contributes a[j]; we count both orders
        g[idx_i] = a[idx_j] + a[idx_i]   # the +a[idx_i] accounts for the diagonal term

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        if symmetrize:
            a = _symmetrize(a)
        a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional random perturbation
        if rng.random() < 0.04:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.025
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                if symmetrize:
                    cand = _symmetrize(cand)
                cand = _project_simplex(cand, S)
                cand_score = np.max(_full_convolution(cand))
                if cand_score < cur_score or rng.random() < 0.2:
                    a = cand
                    cur_score = cand_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()

        if it % 8000 == 0 and best_score == cur_score:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimiser (Adam → sub‑gradient) with symmetry flag
# ----------------------------------------------------------------------


def _optimize_combined(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       symmetrize: bool = True) -> tuple[np.ndarray, float]:
    """Run Adam for ~60 % of the time, finish with sub‑gradient."""
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    # 60 % time for Adam, 40 % for sub‑gradient
    smooth_deadline = now + 0.6 * remaining
    a_s, sc_s = _optimize_adam(a0, smooth_deadline, rng, symmetrize)

    if time.time() < deadline:
        a_f, sc_f = _optimize_subgrad(a_s, deadline, rng, symmetrize)
        if sc_f < sc_s:
            return a_f, sc_f
    return a_s, sc_s

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,

    """
    Search for a non‑negative step function (vector) that minimises
    ``evaluate_sequence``.  The algorithm consists of:

    1. a short warm‑up phase with many random restarts,
    2. a long fine‑tuning phase that repeatedly optimises the current
       best candidate (with symmetry enforced) using Adam on a smooth
       max‑norm followed by a sub‑gradient clean‑up,
    3. a progressive up‑sampling to a larger length (up to 8000) and
       another refinement pass.

    The best sequence found before the deadline is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a few seconds for the final external evaluation
    deadline = start_time + budget_s - 4.0

    # ------------------------------------------------------------------
    # Helper: project a random vector onto the simplex {x≥0, Σx = √(2·n)}
    # ------------------------------------------------------------------
    def _rand_on_simplex(n):
        v = rng.random(n)
        S = np.sqrt(2.0 * n)
        return _project_simplex(v, S)

    # ------------------------------------------------------------------
    # Initialise from the global construction if it exists
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    if 'height_sequence_1' in globals():
        try:

            tmp = np.clip(tmp, 0.0, None)
            n0 = tmp.shape[0]
            tmp = _project_simplex(tmp, np.sqrt(2.0 * n0))
            best_seq = tmp.copy()
            best_score = np.max(_full_convolution(best_seq))
            print(f"[init] loaded height_sequence_1, n={n0}, score={best_score:.6f}")
        except Exception as e:
            print(f"[init] could not use height_sequence_1: {e}")

    # ------------------------------------------------------------------
    # Warm‑up: many short random trials (≈30 s total)
    # ------------------------------------------------------------------
    warmup_deadline = min(deadline, start_time + 30.0)
    attempts = 0
    while time.time() < warmup_deadline:
        attempts += 1
        n = int(rng.integers(800, 3000))
        a0 = _rand_on_simplex(n)
        # 5 s slice for a quick optimisation
        slice_dead = time.time() + 5.0
        a_opt, sc_opt = _optimize_combined(a0, slice_dead, rng, symmetrize=True)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[warmup] attempt={attempts} n={n} new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Main refinement loop (uses the current best as starting point)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        # decide whether to up‑sample for a bigger search space

            a0 = _upsample(best_seq, n_target)
        else:

            # tiny jitter to avoid stagnation

            a0 = _project_simplex(a0, np.sqrt(2.0 * a0.shape[0]))

        slice_deadline = time.time() + min(remaining * 0.7, 30.0)
        a_opt, sc_opt = _optimize_combined(a0, slice_deadline, rng, symmetrize=True)

        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[refine] improved: score={best_score:.6f} n={a_opt.shape[0]}")

    # ------------------------------------------------------------------
    # Final sanity check and return

    print(f"[final] external_score={external_score:.6f} internal_best={best_score:.6f} length={len(final_seq)}")
    return final_seq
