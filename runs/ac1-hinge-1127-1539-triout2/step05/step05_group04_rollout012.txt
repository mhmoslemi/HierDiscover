def _grid_initial_candidates(n, target_sum):
    candidates = []
    idx = np.arange(n)+1
    # power-law exponents p in [0.1,2.0] step 0.1
    for p in np.linspace(0.1, 2.0, 20):

* **Exact max‑index sub‑gradient descent** on the true peak of the
  self‑convolution.
* **Targeted mass‑balance moves** that actively steal mass from the pair of
  indices contributing most to the current worst convolution entry.
* **A short simulated‑annealing polish**.

All steps respect the ℓ¹‑simplex constraint ``Σ a_i = √(2 n)`` and keep the
coefficients non‑negative.  The routine respects the supplied wall‑clock
budget (default 1000 s) and returns the best sequence found as a plain list
of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0, Σ x_i = s }.

    Implementation of Duchi et al. (2008).  Returns a new array.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1, dtype=float)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·n−1) via FFT.
    Returns a real‑valued NumPy array.
    """
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Objective after ℓ¹‑normalisation Σa = √(2 n): the supremum norm of the
    self‑convolution a*a (no extra scaling).
    """
    return float(_max_convolution_fft(a).max())


# ----------------------------------------------------------------------
# Random initialiser (several families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative point on the simplex Σa = √(2 n).
    A few families are sampled to increase diversity.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                         # power‑law
        p = rng.uniform(0.2, 1.2)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                       # pure exponential
        lam = rng.exponential(scale=0.02)
        a = np.exp(-lam * idx)

    elif family == 2:                       # power × exponential
        p = rng.uniform(0.2, 1.0)
        lam = rng.exponential(scale=0.02)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                       # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n))

    elif family == 4:                       # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                             np.exp(-lam2 * np.arange(b, n))])

    else:                                   # almost uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) gradient utilities – Phase 1
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σₖ exp(β·bₖ) where b = a*a.
    Returns a tuple (g, b) where

        g = 2·(a ⋆ p)    with   pₖ ∝ exp(β·bₖ)
        b = a*a (full convolution)
    """
    n = a.shape[0]

    # b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities p ∝ exp(β·b)
    max_b = float(np.max(b))
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1        # = 3·n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _smooth_max_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Annealed projected gradient descent on the smooth max.
    Uses a simple momentum scheme and occasional Gaussian jitter.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 2 ** 16          # ≈ 65536
    beta_factor = 1.3

    step_base = target_sum * 0.12
    momentum = np.zeros_like(a)
    rho = 0.92                    # momentum decay

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        # enforce feasibility (simple ℓ¹‑renormalisation)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a *= target_sum / s

        # occasional jitter
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a *= target_sum / a.sum()

        # anneal β
        if it % 350 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a


# ----------------------------------------------------------------------
# Exact max‑index sub‑gradient – Phase 2
# ----------------------------------------------------------------------


def _max_index_subgradient(a: np.ndarray):
    """
    Sub‑gradient of max(b) where b = a*a.
    Returns (g, k) where k = argmax b and
        g_i = 2·a_{k‑i}  for indices inside the support.
    """
    n = a.shape[0]
    conv = _max_convolution_fft(a)
    k = int(np.argmax(conv))

    g = np.zeros_like(a)
    if k < n:                         # left half
        g[:k + 1] = 2.0 * a[k::-1]
    else:                             # right half
        d = k - (n - 1)               # shift amount (0 … n‑1)
        g[d:] = 2.0 * a[: n - d]
    return g, k


def _subgradient_descent_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected sub‑gradient descent on the true max‑norm.
    Uses a diminishing step size ηₜ = α / √t.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        g, _ = _max_index_subgradient(a)

        step = 0.7 * target_sum / (np.sqrt(it + 1) *
                                   (np.linalg.norm(g) + 1e-12))
        a -= step * g

        # feasibility
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a *= target_sum / s

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 4e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a *= target_sum / a.sum()

        it += 1
    return a


# ----------------------------------------------------------------------
# Targeted mass‑balance refinement
# ----------------------------------------------------------------------


def _mass_balance_refine(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Greedy move that reduces the contribution of the currently maximal
    convolution entry.  At each step the index (i, j) that contributes most
    to the peak is identified; a small fraction of the larger of the two
    masses is transferred to a light location.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sum(a)               # already ≈ √(2 n)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _max_convolution_fft(a)
        k = int(np.argmax(conv))

        # contribution of each index to that entry:
        idx = np.arange(n)
        other = k - idx
        mask = (other >= 0) & (other < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[other[mask]]

        i_high = int(np.argmax(contrib))
        j_high = k - i_high
        # pick the heavier of the pair as donor
        donor = i_high if a[i_high] >= a[j_high] else j_high
        partner = j_high if donor == i_high else i_high

        # choose a light target (among the lower 20 % masses)
        low_candidates = np.where(a <= np.percentile(a, 20))[0]
        if low_candidates.size == 0:
            i_low = int(np.argmin(a))
        else:
            i_low = int(rng.choice(low_candidates))

        # move a modest fraction (≈3 %) of donor's mass
        delta = a[donor] * 0.03
        if delta < 1e-14:
            break

        a[donor] -= delta
        a[i_low] += delta

        # renorm to simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a *= target_sum / s

        if it % 150 == 0:
            # occasional global jitter
            if rng.random() < 0.02:
                jitter = rng.normal(scale=target_sum * 8e-4, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                a *= target_sum / a.sum()

        it += 1
    return a


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (identical to reference)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Very light SA moving a random fraction of mass between two sites."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    cur_score = _score(a)
    best_score = cur_score

    T = cur_score * 0.02 + 1e-9
    cooling = 0.9996
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-16:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        frac = rng.random() * 0.14
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta

        new_score = _score(a)
        dE = new_score - cur_score
        accept = dE <= 0.0 or (T > 0.0 and rng.random() < np.exp(-dE / T))

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling
    return best


# ----------------------------------------------------------------------
# Parametric sweep helpers (quick seeds)
# ----------------------------------------------------------------------


def _parametric_seeds(n: int, target_sum: float):
    """
    Yield a handful of deterministic parameterised candidates:
      – pure power‑law      a_i ∝ (i+1)^{-p}
      – pure exponential    a_i ∝ exp(-λ·i)
      – mixed (power × exp)
    The candidates are projected onto the ℓ¹‑simplex.
    """
    idx = np.arange(n, dtype=float) + 1.0

    # power‑law
    for p in np.linspace(0.1, 2.0, 15):
        a = idx ** (-p)
        yield _project_to_simplex(a, target_sum)

    # exponential
    for lam in np.linspace(0.0005, 0.025, 15):
        a = np.exp(-lam * (idx - 1.0))
        yield _project_to_simplex(a, target_sum)

    # mixed
    for p in np.linspace(0.2, 1.2, 8):
        for lam in np.linspace(0.001, 0.02, 8):
            a = (idx ** (-p)) * np.exp(-lam * (idx - 1.0))
            yield _project_to_simplex(a, target_sum)


# ----------------------------------------------------------------------
# Core optimisation per length
# ----------------------------------------------------------------------


def _optimise_length(n: int, deadline: float, rng) -> tuple[np.ndarray, float]:
    """
    Search for the best sequence of length ``n`` within the supplied
    sub‑deadline.  Returns a tuple ``(best_seq, best_score)``.  The routine
    combines a cheap parametric sweep, many short stochastic runs, and a
    final refined descent on the current best.
    """
    target_sum = np.sqrt(2.0 * n)
    best_seq = None
    best_score = np.inf

    # 1) quick parametric sweep ------------------------------------------------
    for a in _parametric_seeds(n, target_sum):
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
    if best_seq is not None:
        print(f"[{time.time():.1f}] length {n} – sweep best score {best_score:.6f}")

    # 2) short stochastic restarts -------------------------------------------
    # allocate ~30 % of the remaining budget to cheap restarts
    restart_deadline = min(deadline, time.time() + 0.30 *
                           (deadline - time.time()))
    while time.time() < restart_deadline:
        a0 = _random_initial_sequence(n, rng)

        # a) smooth‑max GD (≈4 s)
        sub_dead = min(deadline, time.time() + 4.0)
        a = _smooth_max_descent(a0, sub_dead)

        # b) brief max‑index sub‑gradient (≈2 s)
        sub_dead = min(deadline, time.time() + 2.0)
        a = _subgradient_descent_max(a, sub_dead)

        # c) quick mass‑balance (≈1 s)
        sub_dead = min(deadline, time.time() + 1.0)
        a = _mass_balance_refine(a, sub_dead)

        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time():.1f}] length {n} – new best from restart {sc:.6f}")

    # 3) deep refinement of the current best (if any) -------------------------
    if best_seq is not None:
        # smooth‑max GD (longer)
        sub_dead = min(deadline, time.time() + 0.25 *
                       (deadline - time.time()))
        a = _smooth_max_descent(best_seq, sub_dead)
        sc = _score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time():.1f}] length {n} – refined (smooth) {sc:.6f}")

        # exact max‑index sub‑gradient
        sub_dead = min(deadline, time.time() + 0.15 *
                       (deadline - time.time()))
        a = _subgradient_descent_max(best_seq, sub_dead)
        sc = _score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time():.1f}] length {n} – refined (sub‑grad) {sc:.6f}")

        # aggressive mass‑balance
        sub_dead = min(deadline, time.time() + 0.10 *
                       (deadline - time.time()))
        a = _mass_balance_refine(best_seq, sub_dead)
        sc = _score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time():.1f}] length {n} – refined (mass) {sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` metric.  The routine respects the given
    wall‑clock budget, explores a handful of lengths, applies the two‑phase
    descent described above and finally hands the best sequence back as a
    plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 0) Seed from a previously discovered construction, if present
    # ------------------------------------------------------------------
    earlier = globals().get("height_sequence_1")
    if isinstance(earlier, (list, tuple, np.ndarray)):
        a0 = np.asarray(earlier, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        sc0 = _score(a0)
        best_seq = a0.copy()
        best_score = sc0
        print(f"[{time.time():.1f}] Seeded from height_sequence_1 "
              f"(n={n0}) – score={sc0:.6f}")

    # ------------------------------------------------------------------
    # 1) Explore several lengths
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000, 10000, 12000]

    for n in lengths:
        if time.time() >= deadline:
            break
        # allocate a portion of the remaining budget to this length
        remaining = deadline - time.time()
        # give each length at most 180 s but never exceed the remaining budget
        sub_dead = time.time() + min(180.0, remaining * 0.30)
        sub_dead = min(sub_dead, deadline)

        seq_n, score_n = _optimise_length(n, sub_dead, rng)

        if seq_n is not None and score_n < best_score:
            best_score = score_n
            best_seq = seq_n.copy()
            print(f"[{time.time():.1f}] Global best updated: length={n} "
                  f"score={score_n:.6f}")

    # ------------------------------------------------------------------
    # 2) Final polishing of the global best (if any)
    # ------------------------------------------------------------------
    if best_seq is not None:
        left = deadline - time.time()
        if left > 10.0:
            # a) long smooth‑max GD
            sub_dead = time.time() + min(0.25 * left, 120.0)
            best_seq = _smooth_max_descent(best_seq, sub_dead)
            best_score = _score(best_seq)
            print(f"[{time.time():.1f}] Polished (smooth) – score={best_score:.6f}")

        left = deadline - time.time()
        if left > 8.0:
            # b) sub‑gradient descent
            sub_dead = time.time() + min(0.20 * left, 80.0)
            best_seq = _subgradient_descent_max(best_seq, sub_dead)
            best_score = _score(best_seq)
            print(f"[{time.time():.1f}] Polished (sub‑grad) – score={best_score:.6f}")

        left = deadline - time.time()
        if left > 5.0:
            # c) mass‑balance refinement
            sub_dead = time.time() + min(0.15 * left, 60.0)
            best_seq = _mass_balance_refine(best_seq, sub_dead)
            best_score = _score(best_seq)
            print(f"[{time.time():.1f}] Polished (mass) – score={best_score:.6f}")

        left = deadline - time.time()
        if left > 4.0:
            # d) Simulated‑annealing polishing
            best_seq = _simulated_annealing(best_seq, deadline)
            best_score = _score(best_seq)
            print(f"[{time.time():.1f}] Polished (SA) – score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Final clean‑up and official evaluation
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution of length 1000 (should never happen)
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # official metric (may be slower – called only once)
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
