"""
adversarial step‑function optimizer – enhanced version

The optimizer searches for a non‑negative vector `a` of length `n`
with ℓ₁‑norm `S = sqrt(2·n)` that minimises the peak of the
self‑convolution `a*a`.  The search consists of three stages:

1. **Diverse initialisation** – random lengths and a family of
   parametrised “edge‑heavy” shapes (distance‑to‑edge power law
   with an offset and optional uniform mixing).  This supplies many
   feasible starting points for the local optimiser.

2. **Projected Adam on the simplex** – the same Adam routine as the
   reference implementation, but with a longer budget per restart
   and a slightly tighter cooling schedule.  Gradients are obtained
   from a τ‑soft‑max approximation of the convolution peak.

3. **Final refinement** – the current best candidate receives the
   remaining time budget in a dedicated Adam run.

The algorithm keeps the best sequence seen and stores it in the
global variable ``height_sequence_1``.  It aims to push the
evaluation score below the previous best of 1.5149.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.ceil(pos).astype(int)
    hi = np.clip(hi, 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Gradient of the τ‑soft‑max approximation of max(b) where b = a*a
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """Return (grad, max_b) where grad = ∂ τ‑soft‑max(b) / ∂a."""
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        # exact sub‑gradient: equally weight all active entries
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        w = exps / np.sum(exps)          # soft‑max weights, shape (2n‑1,)
        mask = w

    # gradient w.r.t a: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Minimise the τ‑soft‑max approximation of the convolution peak.
    Returns the best vector encountered and the corresponding hard peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)          # required ℓ₁‑norm

    # initialise logits so that softmax(p) * S == a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)               # logits (no explicit projection needed later)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.3                        # a bit larger than the reference value

    # initialise smoothing temperature
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 1.0)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- recover a from logits (log‑softmax) -----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ----- convolution & gradient (soft‑max) -----
        g, peak = _softmax_gradient(a, tau)

        # keep best hard peak (not the softened one)
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ----- gradient w.r.t. logits (chain rule) -----
        mu = np.dot(a, g) / S           # ⟨a,∂b/∂a⟩/S
        grad_p = a * (g - mu)           # shape (n,)

        # optional gradient clipping (helps stability for large n)
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- schedules -----
        tau *= 0.9997               # tighter soft‑max
        lr *= 0.9999                # very slow decay

        # occasional jitter to escape plateaus
        if it % 800 == 0:
            p += rng.normal(scale=0.015, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Parameterised edge‑heavy initial shapes
# ----------------------------------------------------------------------
def _edge_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Build a feasible start vector of length n.
    The shape is (d + offset)^{-p} optionally mixed with a uniform
    background; finally it is scaled to the required ℓ₁‑norm.
    """
    S = np.sqrt(2.0 * n)

    # distance to nearest edge, 1‑based
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0

    # random parameters
    p = rng.uniform(0.2, 1.0)         # power‑law exponent
    offset = rng.uniform(0.0, 5.0)    # denominator shift
    mix = rng.uniform(0.0, 1.0)       # proportion of edge‑heavy component

    edge_part = (d + offset) ** (-p)
    base = np.ones(n)                # uniform background
    raw = mix * edge_part + (1.0 - mix) * base

    raw = np.clip(raw, 0.0, None)
    raw = raw / raw.sum() * S
    return raw
