"""
Propose a step‑function (non‑negative vector) that minimizes
``evaluate_sequence``.

Algorithm
---------
* Work on the simplex Σ a = √(2·n) (the objective is scale‑invariant:
  the score is exactly the maximum of the self‑convolution).
* For each attempt we pick a problem size n (typically a few‑ten
  thousand) and initialise a from a diverse family (uniform Dirichlet,
  geometric, reciprocal, Gaussian, up‑sampled previous best).
* Optimisation consists of two stages:
  1. **Smooth Adam** – minimise a soft‑max approximation of the
     max‑norm.  The temperature (α) is increased slowly so the loss
     becomes tighter.
  2. **Active‑set descent** – focus exactly on the currently
     maximal convolution entries.  The gradient of the sum of those
     entries is computed with an FFT‑based convolution, a diminishing
     step‑size is used and the vector is projected back onto the simplex.
* Random “mass‑swap’’ moves are interleaved to escape plateaus.
* The best sequence seen over all attempts is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ w_k·(a*a)[k] w.r.t a.
    Returns a vector of length n.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights (smooth approximation of max)
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Stable soft‑max over the convolution entries."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Initial sequence generators
# ----------------------------------------------------------------------
def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Resize an existing sequence to length n using linear interpolation."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """Produce a non‑negative vector of length n with ℓ¹‑norm √(2·n)."""
    S = np.sqrt(2.0 * n)

    choice = rng.integers(0, 6)
    if choice == 0 and base_seq is not None and base_seq.size >= 10:
        a = _upsample_seq(base_seq, n)
        a += rng.normal(scale=a.mean() * 1e-4, size=n)   # tiny jitter
        a = np.clip(a, 0.0, None)
    elif choice == 1:                     # Dirichlet(1)
        a = rng.random(n)
    elif choice == 2:                     # symmetric geometric decay
        rate = rng.uniform(0.9, 0.9995)
        left = rate ** np.arange(n)
        right = left[::-1]
        a = (left + right) / 2.0
    elif choice == 3:                     # edge‑heavy reciprocal
        idx = np.arange(n, dtype=float)
        a = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.8)
        a = a ** power
    elif choice == 4:                     # centred Gaussian
        sigma = rng.uniform(0.12, 0.3) * n
        x = np.arange(n) - (n - 1) / 2.0
        a = np.exp(-0.5 * (x / sigma) ** 2)
    else:                                 # random convex combination of a few bases
        cand = []
        cand.append(rng.random(n))
        rate = rng.uniform(0.9, 0.9995)
        left = rate ** np.arange(n)
        cand.append((left + left[::-1]) / 2.0)
        idx = np.arange(n, dtype=float)
        rec = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.8)
        cand.append(rec ** power)
        sigma = rng.uniform(0.12, 0.3) * n
        x = np.arange(n) - (n - 1) / 2.0
        cand.append(np.exp(-0.5 * (x / sigma) ** 2))
        w = rng.random(len(cand))
        w /= w.sum()
        a = sum(wi * ci for wi, ci in zip(w, cand))

    # Normalise to the required ℓ¹‑norm
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0)
        a_sum = a.sum()
    a = a / a_sum * S
    return a

def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected Adam on a log‑sum‑exp (soft‑max) surrogate of the max."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    lr = 0.25                     # a bit larger than the original
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    alpha = 0.4
    alpha_mul = 1.018
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a, S)

        # occasional mass‑swap
        if rng.random() < 0.025:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.05 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < best_score or rng.random() < 0.12:
                    a = cand

        alpha = min(alpha * alpha_mul, 80.0)   # push to a harder max‑approx.
        lr *= 0.99995                         # very gentle decay

        if it % 4000 == 0 and best_score == cur_max:
            break
    return best_a, best_score


def _optimize_active(a0: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Descent on the true max‑norm.  At each iteration the gradient of the
    (normalized) sum of all entries that are within a tiny tolerance of the
    current maximum is used.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        M = conv.max()
        # active set – keep a few entries that are almost maximal
        tol = max(1e-8, M * 1e-5)
        act = np.where(conv >= M - tol)[0]
        # weight vector for the active entries (averaged)
        w = np.zeros_like(conv)
        if act.size:
            w[act] = 1.0 / act.size
        else:
            w[np.argmax(conv)] = 1.0

        grad = _conv_with_weights(a, w)

        # diminishing step‑size
        eta = 0.6 / np.sqrt(it)
        a = a - eta * grad
        a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # random jiggle to avoid tiny stalling
        if rng.random() < 0.018:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.04 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = np.max(_full_convolution(cand))
                if cand_score < cur_score or rng.random() < 0.15:
                    a = cand

        if it % 3000 == 0 and best_score == cur_score:
            break
    return best_a, best_score


def _optimize_combined(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Run a smooth Adam phase (≈ 40 % of the budget) then the active set phase."""
    now = time.time()
    remain = deadline - now
    if remain <= 0.0:
        return a0, np.max(_full_convolution(a0))

    # 40 % for Adam, the rest for the active set
    split = now + 0.4 * remain
    a_mid, _ = _optimize_adam(a0, split, rng)
    a_opt, score_opt = _optimize_active(a_mid, deadline, rng)
    return a_opt, score_opt

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function minimising ``evaluate_sequence``.
    Returns the best sequence (list of floats) found before the time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the external evaluator
    deadline = start_time + budget_s - 5.0

    # ----------- previous best (if any) -----------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    attempts = 0
    last_improve = start_time

    # main loop – many short attempts
    while time.time() < deadline:
        attempts += 1

        # choose a problem size (larger n gives a tighter theoretical bound)
        n = int(rng.integers(3000, 18000))
        a0 = _initial_sequence(base_seq, n, rng)

        # give each attempt roughly 9 s (or whatever is left)
        slice_deadline = min(time.time() + 9.0, deadline)

        a_opt, score = _optimize_combined(a0, slice_deadline, rng)

        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            last_improve = time.time()
            print(f"[global] attempt={attempts} n={n} new best={best_score:.6f}")

        # early stop if we have already crossed a useful threshold
        if best_score <= 1.48:          # a comfortable safety margin
            break

        # if we have been idle for a while, drop the current base_seq
        if time.time() - last_improve > 45.0:
            base_seq = None

    # ----------- final polishing on the overall best (if time permits) -----
    if best_seq is not None and time.time() < deadline - 1.0:
        # give the current best a final refinement using the full remaining budget
        final_seq, final_score = _optimize_combined(best_seq, deadline, rng)
        if final_score < best_score:
            best_seq = final_seq
            best_score = final_score

    if best_seq is None:
        final = [1.0]
    else:
        final = best_seq.tolist()

    # final sanity check (the harness will also evaluate)
    try:
        external = evaluate_sequence(final)
        print(f"[final] external={external:.6f} internal={best_score:.6f} "
              f"len={len(final)} attempts={attempts}")
    except Exception as e:
        # in the unlikely event evaluate_sequence is unavailable here,
        # just report the internal best.
        print(f"[final] internal best={best_score:.6f} len={len(final)} attempts={attempts}")

    return final
