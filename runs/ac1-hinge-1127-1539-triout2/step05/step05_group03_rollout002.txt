def lp_step(f, M, S):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    f_param = cp.Parameter(n, nonneg=True)
    f_param.value = f
    conv_expr = cp.conv(f_param, g)   # returns length 2n-1
    constraints = [conv_expr <= M]   # vector bound
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    if g.value is None:
        return None  # failed
    sum_g = np.sum(g.value)
    if sum_g <= 1e-12:
        return None
    # scale g to have L1=S
    g_scaled = g.value * (S / sum_g)
    return g_scaled
```

def _solve_lp(f, M, S):
    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)
    f_param = cp.Parameter(n, nonneg=True)
    f_param.value = f
    conv = cp.conv(f_param, g)
    constraints = [conv <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000)
    except cp.error.SolverError:
        return None
    if g.value is None:
        return None
    sum_g = np.sum(g.value)
    if sum_g <= 1e-12:
        return None
    # scale to sum = S

"""
Propose a step‑function that minimizes evaluate_sequence().
The algorithm combines a linear‑programming direction step (as in the
paper sketch) with a simple convex mixing scheme and a final up‑sampling
pass.  The LP solves:

    max   ∑ g_i
    s.t.  (f * g)_k ≤ M   for all k,
          g_i ≥ 0,

where f is the current sequence, M = max(f*f).  Solving this LP yields a
direction g that places mass where the convolution with f is small.
We then replace f by a convex combination (1‑t)·f + t·g (both are
normalised to the simplex Σa = √(2n)).  Repeating the step quickly
lowers the peak of the self‑convolution.  After the LP phase we
up‑sample the best shape to larger lengths and keep the best result
found before the time budget expires.
"""

import time
import numpy as np
import cvxpy as cp


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a shorter sequence to length n."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n with L¹‑norm √(2·n).
    Several random families are mixed to provide diversity.
    """
    S = np.sqrt(2.0 * n)

    typ = rng.integers(0, 5)

    if typ == 0 and base_seq is not None and base_seq.size > 10:
        a = _upsample_seq(base_seq, n)
        a += rng.normal(scale=a.mean() * 1e-3, size=n)
        a = np.clip(a, 0.0, None)
    elif typ == 1:
        a = rng.random(n)                     # Dirichlet(1)
    elif typ == 2:
        rate = rng.uniform(0.9, 0.9995)
        left = rate ** np.arange(n)
        right = left[::-1]
        a = (left + right) / 2.0
    elif typ == 3:
        idx = np.arange(n, dtype=float)
        a = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.8)
        a = a ** power
    else:
        # blend a few bases
        cand = []
        cand.append(rng.random(n))
        rate = rng.uniform(0.9, 0.9995)
        left = rate ** np.arange(n)
        cand.append((left + left[::-1]) / 2.0)
        idx = np.arange(n, dtype=float)
        rec = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.8)
        cand.append(rec ** power)
        w = rng.random(len(cand))
        w /= w.sum()
        a = sum(wi * ci for wi, ci in zip(w, cand))

    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0)
        a_sum = a.sum()
    a = a / a_sum * S
    return a


# ----------------------------------------------------------------------
# LP direction step
# ----------------------------------------------------------------------
def _solve_lp_direction(a: np.ndarray, M: float, S: float) -> np.ndarray | None:
    """
    Solve  max ∑ g_i   s.t.  (a * g)_k ≤ M  ∀k,  g ≥ 0.
    Returns g scaled to have ℓ¹‑norm S, or None on failure.
    """
    n = a.shape[0]
    # Build the dense convolution matrix A (shape (2n‑1) × n)
    # Row k contains the coefficients a_j where j contributes to (a * g)_k.
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    for i in range(n):
        A[i:i + n, i] = a                     # column i is shifted copy of a

    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]                # element‑wise ≤
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                   max_iters=2000, abstol=1e-7, reltol=1e-7, feastol=1e-7)
    except Exception:
        return None

    if g.value is None:
        return None
    sum_g = np.sum(g.value)
    if sum_g <= 1e-12:
        return None
    return g.value * (S / sum_g)             # scale to the required ℓ¹‑norm


def _lp_optimize(a0: np.ndarray,
                 deadline: float,
                 rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Iteratively improve the sequence using the LP direction step.
    Returns the best vector found and its convolution peak.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_max = np.max(np.convolve(a, a))

    # mixing weight – start relatively aggressive and decay
    t = 0.6
    it = 0
    while time.time() < deadline:
        it += 1
        M = np.max(np.convolve(a, a))
        g = _solve_lp_direction(a, M, S)
        if g is None:
            break

        # convex combination (both have ℓ¹‑norm S)
        a = (1.0 - t) * a + t * g
        t *= 0.96            # slowly reduce step size

        cur_max = np.max(np.convolve(a, a))
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()

        # occasional random jitter to escape tiny plateaus
        if rng.random() < 0.03:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.02 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        # simple termination test
        if it % 7 == 0 and cur_max >= best_max * 0.9999:
            break

    return best_a, best_max

def propose_candidate(seed: int = 12345,

    rng = np.random.default_rng(seed)
