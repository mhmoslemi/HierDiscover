"""
Propose candidate step-function sequence via alternating
linear-programming refinement and a final projected‑gradient
phase. The LP solves:
    max   Σ g_i
    s.t.  (a * g)_k ≤ M   ∀k,   g_i ≥ 0
where a is the current sequence and M = max(a*a). The solution
pushes mass to indices that are “light’’ under the current
convolution, flattening the autocorrelation. A small convex
mixing step updates a, preserving Σ a_i = √(2n).  After a few
LP iterations we run a short projected‑gradient descent on a
soft‑max approximation of the max‑operator to polish the result.
The routine runs within the allotted time budget (default 1000 s)
and returns the best sequence found.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Core utilities (same as in the reference implementation)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Fast evaluation of the objective (max convolution) for a normalised sum."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σ x = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------


def _conv_constraint_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1)×n matrix A such that (A @ g) == a * g (linear convolution).
    Row k corresponds to (a*g)_k.
    """
    n = a.shape[0]
    # column index for each entry (repeated n times per column)
    col = np.repeat(np.arange(n, dtype=np.int32), n)
    # row index = col + offset where offset = 0..n‑1 repeated for each column
    row = np.tile(np.arange(n, dtype=np.int32), n) + col
    data = np.tile(a, n)
    A = sp.csr_matrix((data, (row, col)), shape=(2 * n - 1, n))
    return A


def _lp_refine(a: np.ndarray, deadline: float, mix: float = 0.2,
               max_iter: int = 30) -> tuple[np.ndarray, float]:
    """
    Alternating linear‑programming refinement.
    ``mix`` is the convex‑mixing factor (1‑t) a + t·g when a new
    feasible vector g has been obtained.
    The function stops when the deadline is reached or when the
    improvement stalls.
    Returns the best sequence and its score seen during the run.
    """
    n = a.size
    target = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_val = _score(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        M = _max_convolution_fft(a)

        # Build constraint matrix for the current a
        A = _conv_constraint_matrix(a)

        # Linear program: maximise sum(g)  ⇔ minimise -sum(g)
        c = np.full(n, -1.0)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                    bounds=[(0, None)] * n,
                                    method='highs')
        if not res.success:
            # LP failed – fall back to gradient descent
            break

        g = res.x
        sum_g = g.sum()
        if sum_g == 0:
            break
        # Scale to keep the ℓ¹‑norm exactly target
        g_scaled = g * (target / sum_g)

        # Convex mixing (keeps the ℓ¹‑norm unchanged)
        a = (1.0 - mix) * a + mix * g_scaled

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] LP iter {it:02d}  score={best_val:.6f}")

        # slowly reduce mixing to stabilise
        mix = max(0.02, mix * 0.95)
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected gradient descent (soft‑max) – used as a polishing step
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient w.r.t. a
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _annealed_gradient_descent(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the soft‑max approximation of the max.
    The temperature β is increased up to 1024.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 1024.0
    step_base = target_sum * 0.20
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            lr = step_base / (norm_g * (beta + 1.0))
            a = a - lr * g
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] GD improvement β={beta:.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Orchestrator – tries a few problem sizes and returns the best.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with a low benchmark score.
    The search runs three stages:
      1.  Initialise a power‑law candidate of several lengths.
      2.  Refine each candidate with the LP‑mixing procedure.
      3.  Optional short projected‑gradient polishing.
    The best sequence seen before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # small safety margin

    best_seq = None
    best_score = float("inf")

    # Problem sizes we will attempt – they fit comfortably into memory.
    sizes = [800, 1500, 2500]

    for n in sizes:
        if time.time() >= deadline:
            break

        target = np.sqrt(2.0 * n)

        # initialise – either from the global warm‑start or a fresh power law
        if "height_sequence_1" in globals():
            seq0 = np.array(globals()["height_sequence_1"], dtype=np.float64)
            if seq0.size > n:
                seq0 = seq0[:n]
            else:
                seq0 = np.pad(seq0, (0, n - seq0.size), "constant")
        else:
            # power‑law exponent tuned per size (a bit larger for bigger n)
            exp = 0.55 + 0.05 * (n // 1000)  # 0.55 for 800, 0.6 for 1500, 0.65 for 2500
            seq0 = _power_law_sequence(n, exp, target)

        seq0 = _project_to_simplex(seq0, target)

        # ----- LP refinement -------------------------------------------------
        lp_deadline = min(deadline, time.time() + 200.0)  # at most 200 s per size
        seq_lp, score_lp = _lp_refine(seq0, lp_deadline,
                                      mix=0.3,
                                      max_iter=25)

        if score_lp < best_score:
            best_score = score_lp
            best_seq = seq_lp.copy()
            print(f"[{time.time():.1f}] size {n} after LP best score={best_score:.6f}")

        # ----- short gradient polishing ------------------------------------
        if time.time() < deadline:
            gd_deadline = min(deadline, time.time() + 30.0)
            seq_gd, score_gd = _annealed_gradient_descent(seq_lp, gd_deadline)
            if score_gd < best_score:
                best_score = score_gd
                best_seq = seq_gd.copy()
                print(f"[{time.time():.1f}] size {n} after GD best score={best_score:.6f}")

"""
Propose candidate step‑function sequence via alternating
linear‑programming refinement and a final projected‑gradient
polish.  

The LP solves
    max   Σ g_i
    s.t. (a * g)_k ≤ M   ∀k ,   g_i ≥ 0
where a is the current sequence, M = max(a*a).  The solution
pushes mass to indices that are “light’’ under the current
convolution, flattening the autocorrelation.  A convex mixing step
updates a while preserving Σ a_i = √(2·n).  After a few LP passes a
short soft‑max gradient descent cleans up the remaining irregularities.
The routine stays within the 1000 s budget and returns the best
sequence found.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Core utilities (fast convolution, scoring, projection)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Evaluate the benchmark score (max‑convolution objective)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Linear‑programming refinement utilities
# ----------------------------------------------------------------------


def _conv_constraint_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1)×n matrix A such that (A @ g) == a * g (linear convolution).
    Row k corresponds to (a*g)_k.
    """
    n = a.shape[0]
    # column index for each entry (repeated n times per column)
    col = np.repeat(np.arange(n, dtype=np.int32), n)
    # row index = col + offset where offset = 0..n‑1 repeated for each column
    row = np.tile(np.arange(n, dtype=np.int32), n) + col
    data = np.tile(a, n)
    A = sp.csr_matrix((data, (row, col)), shape=(2 * n - 1, n))
    return A


def _lp_refine(a: np.ndarray, deadline: float,
               mix: float = 0.2, max_iter: int = 30) -> tuple[np.ndarray, float]:
    """
    Alternating LP refinement.

    Parameters
    ----------
    a : np.ndarray
        Initial sequence (already projected onto the ℓ¹‑simplex).
    deadline : float
        Unix timestamp after which the routine must stop.
    mix : float
        Initial convex‑mixing factor (1‑mix)·a + mix·g.
    max_iter : int
        Upper bound on the number of LP iterations.

    Returns
    -------
    best_a : np.ndarray
        Best sequence seen during the run.
    best_val : float
        Corresponding benchmark score.
    """
    n = a.size
    target = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        M = _max_convolution_fft(a)

        A = _conv_constraint_matrix(a)

        # maximise Σ g_i  ⇔  minimise -Σ g_i
        c = np.full(n, -1.0)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                    bounds=[(0, None)] * n,
                                    method='highs')
        if not res.success:
            # If the LP fails (numerical issues) we stop the LP phase.
            break

        g = res.x
        sum_g = g.sum()
        if sum_g == 0:
            break

        # Rescale to keep the ℓ¹‑norm exactly target.
        g_scaled = g * (target / sum_g)

        # Convex mixing – preserves the sum.
        a = (1.0 - mix) * a + mix * g_scaled

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] LP iter {it:02d}  score={best_val:.6f}")

        # Gradually shrink the mixing factor to stabilise later.
        mix = max(0.02, mix * 0.95)
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected gradient descent (soft‑max) – polishing stage
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient w.r.t. a
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _annealed_gradient_descent(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on a soft‑max approximation of the max‑operator.
    Temperature β is increased up to 1024.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 1024.0
    step_base = target_sum * 0.20
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            lr = step_base / (norm_g * (beta + 1.0))
            a = a - lr * g
        a = _project_to_simplex(a, target_sum)

        # small jitter every few hundred iterations
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] GD β={beta:.1f}  score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with a low benchmark score.

    The routine runs three stages:
      1. Initialise power‑law candidates of several lengths.
      2. Refine each candidate with the LP‑mixing procedure.
      3. Optionally polish the best candidate with a short projected‑gradient descent.

    The best sequence seen before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5      # safety margin

    best_seq = None
    best_score = float("inf")

    # Problem sizes we attempt – they comfortably fit into memory.
    sizes = [800, 1500, 2500]

    for n in sizes:
        if time.time() >= deadline:
            break

        target = np.sqrt(2.0 * n)

        # ------------------------------------------------------------------
        # Initialise a candidate.
        # ------------------------------------------------------------------
        if "height_sequence_1" in globals():
            seq0 = np.array(globals()["height_sequence_1"], dtype=np.float64)
            if seq0.size > n:
                seq0 = seq0[:n]
            else:
                seq0 = np.pad(seq0, (0, n - seq0.size), "constant")
        else:
            # Slightly larger exponent for larger n (empirically useful)
            exp = 0.55 + 0.05 * (n // 1000)   # 0.55, 0.60, 0.65 for the three sizes
            seq0 = _power_law_sequence(n, exp, target)

        seq0 = _project_to_simplex(seq0, target)

        # ------------------------------------------------------------------
        # LP refinement (takes at most ~200 s per size)
        # ------------------------------------------------------------------
        lp_deadline = min(deadline, time.time() + 200.0)
        seq_lp, score_lp = _lp_refine(seq0, lp_deadline,
                                      mix=0.30,
                                      max_iter=25)

        if score_lp < best_score:
            best_score = score_lp
            best_seq = seq_lp.copy()
            print(f"[{time.time():.1f}] size {n} after LP  best={best_score:.6f}")

        # ------------------------------------------------------------------
        # Short projected‑gradient polishing (≈30 s)
        # ------------------------------------------------------------------
        if time.time() < deadline:
            gd_deadline = min(deadline, time.time() + 30.0)
            seq_gd, score_gd = _annealed_gradient_descent(seq_lp, gd_deadline)
            if score_gd < best_score:
                best_score = score_gd
                best_seq = seq_gd.copy()
                print(f"[{time.time():.1f}] size {n} after GD  best={best_score:.6f}")

    # ----------------------------------------------------------------------
    # Final housekeeping – strip trailing zeros and expose global variable.
    # ----------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.array([], dtype=float)

    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # make the best sequence available for warm‑starts in future calls
    globals()["height_sequence_1"] = best_seq.tolist()

    print(f"[{time.time():.1f}] final best score={best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
