def _gradient_descent_adam(a0, deadline):
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    lr0 = target_sum * 0.1   # base learning rate

    beta = 0.5
    beta_max = 8192.0
    beta_factor = 1.4
    iter = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        iter += 1
        # Adam updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** iter)
        v_hat = v / (1.0 - beta2 ** iter)
        lr = lr0 / (np.sqrt(v_hat) + epsilon)
        a -= lr * m_hat

        # projection
        a = _project_to_simplex(a, target_sum)

        # jitter
        if iter % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        # beta schedule
        if iter % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a
```

def _gd_simple(a0, deadline):
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2*n)
    beta = 0.5
    beta_max = 8192.0
    beta_factor = 1.4
    step_scale = target_sum * 0.05  # step magnitude
    momentum = np.zeros_like(a)
    rho = 0.9
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = rho * momentum + (1 - rho) * g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_scale / norm
            a -= lr * momentum
        # projection
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            a *= target_sum / s
        # jitter, beta schedule
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005 / (beta+1), size=n)
            a += jitter
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 0:
                a *= target_sum / s
        if it % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)
        it += 1
    return a

def _hill_climb_targeted(a0, b0, deadline):
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max
    best_score = cur_max
    rng = np.random.default_rng()
    T = cur_max * 0.02 + 1e-8
    cooling = 0.9995

    while time.time() < deadline:
        k = int(b.argmax())
        # compute contributions
        idx = np.arange(n)
        idx2 = k - idx
        mask = (idx2 >= 0) & (idx2 < n)
        contrib = np.zeros(n)
        contrib[mask] = a[mask] * a[idx2[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs_src = contrib / total_contrib
            i = rng.choice(n, p=probs_src)
        else:
            i = rng.choice(n, p=a / a.sum())
        # destination j: choose low a indices (lowest 30%)
        threshold = np.percentile(a, 30)
        low_idxs = np.where(a <= threshold)[0]
        if low_idxs.size > 0:
            j = rng.choice(low_idxs)
        else:
            j = rng.integers(n)
        if i == j:
            continue
        max_delta = a[i] * 0.5
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta
        # compute new convolution
        a_old = a.copy()
        b_new = b.copy()
        b_new[i:i+n] -= 2.0 * delta * a_old
        b_new[j:j+n] += 2.0 * delta * a_old
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta
        new_max = b_new.max()
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True
        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                # optionally store best b?
        T *= cooling
    return best_a, b  # but b corresponds to best ? we may recompute.

"""
Advanced step‑function optimiser.

The optimiser works in three loosely coupled phases
(1)   initialise a candidate (optionally from the historic
      ``height_sequence_1``), up‑sample it to a large length and
      normalise it on the ℓ¹‑simplex Σa = √(2 n);
(2)   a fast projected gradient descent that minimises a
      log‑sum‑exp approximation of max(a∗a) – the optimiser uses
      momentum, a simple ℓ¹‑projection (scale‑and‑clip) and an
      annealed inverse‑temperature β;
(3)   a directed stochastic hill‑climber that repeatedly moves a
      random fraction of mass from a source index (chosen proportionally
      to its contribution to the current maximal convolution entry) to
      a destination with low mass.  The convolution is updated in O(n)
      time using the exact analytical Δb formula, and the move is
      accepted either if it improves the max‑convolution or with a
      temperature‑controlled probability.

The whole search respects the wall‑clock budget (default 1000 s) and
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (size 2 n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simple ℓ¹‑simplex projection (scale‑and‑clip)
# ----------------------------------------------------------------------


def _simplex_scale_clip(v: np.ndarray, target: float) -> np.ndarray:
    """
    Fast projection onto {x ≥ 0, Σx = target}.
    This is NOT the Euclidean projection but is sufficient for the
    stochastic optimiser and is O(n).
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        v = v * (target / s)
    else:
        # fall back to uniform distribution
        v = np.full_like(v, target / v.size)
    return v


# ----------------------------------------------------------------------
# Smoothed maximum (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of   (1/β)·log Σ exp(β·b)   where   b = a*a .
    Returns (g, b) with
        g = 2·(a ⋆ p)           (⋆ = correlation)
        pₖ ∝ exp(β·bₖ)
    The correlation is performed with FFTs.
    """
    n = a.shape[0]

    # --- b = a*a -------------------------------------------------------
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # --- soft‑max probabilities ----------------------------------------
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # --- gradient g = 2·(a ⋆ p) ----------------------------------------
    a_rev = a[::-1]                     # reversed a
    conv_len = n + (2 * n - 1) - 1      # = 3n‑2
    size2 = _next_pow2(conv_len)
    ar_fft = np.fft.rfft(a_rev, n=size2)
    pr_fft = np.fft.rfft(p, n=size2)
    corr_rev = np.fft.irfft(ar_fft * pr_fft, n=size2)[:conv_len]
    g = 2.0 * corr_rev[n - 1: n - 1 + n]   # slice centred on the centre index
    return g, b


# ----------------------------------------------------------------------
# Projected gradient descent (momentum + simple projection)
# ----------------------------------------------------------------------


def _gradient_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Fast projected gradient descent on the smoothed max.
    Uses momentum, an annealed inverse temperature β and occasional jitter.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 8192.0
    beta_factor = 1.4

    step_scale = target_sum * 0.15      # controls absolute step size
    rho = 0.90                          # momentum decay
    momentum = np.zeros_like(a)

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        # gradient of the smooth max
        g, _ = _gradient_softmax(a, beta)

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * g
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            lr = step_scale / norm
            a -= lr * momentum

        # projection onto the ℓ¹‑simplex
        a = _simplex_scale_clip(a, target_sum)

        # jitter to escape plateaus (every few hundred iterations)
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = _simplex_scale_clip(a, target_sum)

        # β‑annealing
        if it % 350 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a


# ----------------------------------------------------------------------
# Random initialisers (various families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Return a random non‑negative vector on the simplex Σa = √(2 n)."""
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # pure power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n) + 1)
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = _simplex_scale_clip(a, target)
    return a


# ----------------------------------------------------------------------
# Linear up‑sampling (interpolation) preserving the ℓ¹‑norm
# ----------------------------------------------------------------------


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a onto a new length, re‑projected onto the simplex."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = _simplex_scale_clip(a_new, np.sqrt(2.0 * new_n))
    return a_new


# ----------------------------------------------------------------------
# Targeted hill‑climbing (exact convolution updates)
# ----------------------------------------------------------------------


def _hill_climb_targeted(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Stochastic hill‑climbing that preferentially moves mass away from
    entries that contribute most to the current maximal convolution entry.
    Exact O(n) updates of the convolution are used.
    Returns the best sequence observed and its full convolution.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    # Temperature proportional to the current peak; slowly cooled.
    T = cur_max * 0.02 + 1e-9
    cooling = 0.9997

    while time.time() < deadline:
        # index of the current maximal convolution entry
        k = int(b.argmax())

        # contribution of each a_i to b_k  (i + j = k)
        idx = np.arange(n)
        idx2 = k - idx
        mask = (idx2 >= 0) & (idx2 < n)
        contrib = np.zeros(n, dtype=float)
        contrib[mask] = a[mask] * a[idx2[mask]]

        # source index – weighted by its contribution to the peak
        total_c = contrib.sum()
        if total_c > 0.0:
            probs_src = contrib / total_c
            i = rng.choice(n, p=probs_src)
        else:
            # fallback: choose proportionally to mass
            probs_src = a / a.sum()
            i = rng.choice(n, p=probs_src)

        # destination – pick among the 30 % smallest entries
        thresh = np.percentile(a, 30)
        low_idx = np.where(a <= thresh)[0]
        if low_idx.size:
            j = rng.choice(low_idx)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        max_delta = a[i] * 0.5          # never make a[i] negative
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ---- exact convolution update ---------------------------------
        a_old = a.copy()
        b_new = b.copy()

        # linear part  (−2·δ·a at i, +2·δ·a at j)
        b_new[i:i + n] -= 2.0 * delta * a_old
        b_new[j:j + n] += 2.0 * delta * a_old

        # quadratic correction terms
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        # ---- acceptance (simulated‑annealing style) -------------------
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                # optional diagnostic
                print(f"[{time.time():.1f}] hill‑climb improvement: {best_max:.6f}")

        T *= cooling

    # recompute the full convolution for the best sequence (cheap)
    best_b = _full_convolution(best_a)
    return best_a, best_b


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock budget (default 1000 s) and returns
    the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5      # small safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 1️⃣  Seed from historic solution if available
    # ------------------------------------------------------------------
    earlier = globals().get("height_sequence_1")
    if isinstance(earlier, (list, tuple, np.ndarray)):
        a0 = np.asarray(earlier, dtype=float)
        n0 = a0.size
        a0 = _simplex_scale_clip(a0, np.sqrt(2.0 * n0))
        best_seq = a0.copy()
        best_score = _max_convolution_fft(a0)
        print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Choose a target length and allocate the budget
    # ------------------------------------------------------------------
    # Larger n usually yields a smaller supremum; we target 15–20 k.
    target_lengths = [12000, 15000, 20000]

    for n in target_lengths:
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)

        # initialise (upsample previous best or random)
        if best_seq is not None and best_seq.size != n:
            a = _upsample_sequence(best_seq, n)
        else:
            a = _random_initial_sequence(n, rng)

        a = _simplex_scale_clip(a, target_sum)

        # ------------------------------------------------------------------
        # 2A – Gradient descent (majority of the remaining budget)
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        gd_deadline = time.time() + min(remaining * 0.55, 250.0)   # cap to keep time for hill‑climb
        a = _gradient_descent(a, gd_deadline)
        score = _max_convolution_fft(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] GD improvement (n={n}) score={score:.6f}")

        # ------------------------------------------------------------------
        # 2B – Targeted hill‑climbing
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        hc_deadline = time.time() + min(remaining * 0.35, 180.0)
        b_full = _full_convolution(a)
        a_hc, b_hc = _hill_climb_targeted(a, b_full, hc_deadline)
        score_hc = b_hc.max()
        if score_hc < best_score:
            best_score = score_hc
            best_seq = a_hc.copy()
            print(f"[{time.time():.1f}] Hill‑climb improvement (n={n}) score={score_hc:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final polishing with any left‑over time
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a_last = best_seq.copy()
        b_last = _full_convolution(a_last)
        a_last, b_last = _hill_climb_targeted(a_last, b_last, deadline)
        final_score = b_last.max()
        if final_score < best_score:
            best_score = final_score
            best_seq = a_last.copy()
            print(f"[{time.time():.1f}] Final polishing score={final_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Clean‑up & official evaluation
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # Log the official evaluation (ratio = max of self‑convolution after normalisation)
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
