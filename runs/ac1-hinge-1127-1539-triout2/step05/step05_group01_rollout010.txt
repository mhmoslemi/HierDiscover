"""
Propose a non‑negative step function (list of heights) that minimizes
the objective used in ``evaluate_sequence``.  The routine works in three
phases within the given time budget:

1. **Deterministic sweep** – a fine grid of power‑law sequences
   a_i = (i+shift)^(-p) for several lengths (3000‑5000) and a narrow
   range of p around the theoretically optimal 0.5.  The best few
   candidates are kept.

2. **Adam‑based projected gradient descent** – each elite candidate is
   refined by minimizing a smooth log‑sum‑exp surrogate of the maximal
   convolution entry.  The update uses the Adam optimiser and projects
   onto the ℓ¹‑simplex Σa = √(2 n) after each step.  The smoothing
   parameter β is annealed up to 1e6.

3. **Simulated annealing** – the best refined vector is further tweaked
   with random mass‑transfer moves.  A multiplicative cooling schedule
   and occasional global jitter are used to escape local minima.

All intermediate results are logged with ``print``; the final list

budget (default 1000 s) and never performs I/O other than printing.

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

# Projection onto the ℓ¹‑simplex Σx = s, x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection onto {x_i ≥ 0, Σx_i = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) surrogate of max(b) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b) where
        b = a*a (full convolution)
        g = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a ⋆ p) ,  p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]

    # ----- convolution a*a -----
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # ----- soft‑max probabilities -----
    max_b = np.max(b)
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)

    # ----- gradient g = 2·(a ⋆ p) -----
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Adam‑based projected gradient descent on the smoothed maximum
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(a*a).
    Returns the best vector seen (projected onto the ℓ¹‑simplex).
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # smoothing schedule
    beta = 1.0
    max_beta = 1e6

    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    t = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        g, b = _gradient_softmax(a, beta)
        t += 1

        # Adam updates
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # per‑coordinate adaptive step
        step = 0.5 * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat

        # proj onto simplex
        a = _project_to_simplex(a, target_sum)

        # update best (using exact max)
        cur_val = b.max()          # exact max of a*a (already computed)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # anneal smoothing parameter
        if t % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.07, max_beta)

        # occasional jitter to avoid stagnation
        if t % 500 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4 / (beta + 1.0), size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing refinement
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Random mass‑transfer annealing with a multiplicative cooling schedule."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # start temperature as a small fraction of current score
    T = best_score * 0.02 + 1e-12
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        # pick source and destination indices
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # move a random fraction (up to 20 %) of mass from i to j
        frac = rng.random() * 0.20
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        accept = False
        if cur_score <= best_score:
            accept = True
        elif T > 0.0 and rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # reject – revert
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # very occasional global jitter (helps escape plateaus)
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()

    return best


# ----------------------------------------------------------------------
# Parametric families (power‑law, symmetric)
# ----------------------------------------------------------------------


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    """
    a_i = (i + shift)^{-p},   i = 0,…,n‑1 .
    Returned vector is projected onto the ℓ¹‑simplex Σa = √(2 n).
    """
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    target_sum = np.sqrt(2.0 * n)
    scale = target_sum / raw.sum()
    a = raw * scale
    a = np.maximum(a, 0.0)
    return a


def _symmetric_powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    """
    Symmetric version: a_i = (i+shift)^{-p} + (n-1-i+shift)^{-p}.
    Afterwards projected onto the ℓ¹‑simplex.
    """
    idx = np.arange(n, dtype=float)
    left = (idx + shift) ** (-p)
    right = (n - 1 - idx + shift) ** (-p)
    raw = left + right
    target_sum = np.sqrt(2.0 * n)
    scale = target_sum / raw.sum()
    a = raw * scale
    a = np.maximum(a, 0.0)
    return a

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The algorithm runs three stages (grid sweep → Adam descent → SA) while
    respecting the supplied time budget.  Returns a list of floats.
    """
    rng = np.random.default_rng(seed)

    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_obj = np.inf

    # ------------------------------------------------------------------
    # 0. Optional warm‑start from a previously found construction
    # ------------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)
        # project onto the correct simplex (might have been altered)
        n0 = init.size
        init = _project_to_simplex(init, np.sqrt(2.0 * n0))
        best_seq = init.copy()
        best_obj = _max_convolution_fft(best_seq)
        print(f"[{time.time():.1f}] Warm‑start from height_sequence_1 n={n0}  obj={best_obj:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Deterministic parametric sweep (power‑law families)
    # ------------------------------------------------------------------
    lengths = [3000, 3500, 4000, 4500, 5000]
    p_grid = np.linspace(0.48, 0.52, 9)          # focus on the theoretically optimal region
    shift_grid = np.linspace(0.1, 2.0, 12)       # modest shifts

    elite = []  # list of (score, a) – we keep the top 8 candidates

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Grid sweep, length {n}")

        for p in p_grid:
            for shift in shift_grid:
                # plain power‑law
                a = _powerlaw_sequence(n, p, shift)
                score = _max_convolution_fft(a)
                if score < best_obj:
                    best_obj = score
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] New global best (grid) n={n} p={p:.4f} "
                          f"shift={shift:.2f}  obj={best_obj:.6f}")

                # symmetric variant
                a_sym = _symmetric_powerlaw_sequence(n, p, shift)
                score_sym = _max_convolution_fft(a_sym)
                if score_sym < best_obj:
                    best_obj = score_sym
                    best_seq = a_sym.copy()
                    print(f"[{time.time():.1f}] New global best (sym) n={n} p={p:.4f} "
                          f"shift={shift:.2f}  obj={best_obj:.6f}")

                # keep up to 8 elite entries
                def _maybe_push(val, vec):
                    if len(elite) < 8 or val < elite[-1][0]:
                        elite.append((val, vec))
                        elite.sort(key=lambda x: x[0])
                        del elite[8:]

                _maybe_push(score, a)
                _maybe_push(score_sym, a_sym)

        # add a few random candidates for this length (adds diversity)
        rnd_end = min(deadline, time.time() + 1.5)
        while time.time() < rnd_end:
            a_rand = rng.random(n)
            a_rand = a_rand / a_rand.sum() * np.sqrt(2.0 * n)
            score = _max_convolution_fft(a_rand)
            if score < best_obj:
                best_obj = score
                best_seq = a_rand.copy()
                print(f"[{time.time():.1f}] New global best (rand) n={n}  obj={best_obj:.6f}")
            _maybe_push(score, a_rand)

    # ------------------------------------------------------------------
    # 2. Adam‑based refinement of elite candidates
    # ------------------------------------------------------------------
    # allocate roughly half of the remaining budget to this phase
    remaining = deadline - time.time()
    if remaining > 0:
        per_candidate = remaining / max(1, len(elite))
    else:
        per_candidate = 0

    for idx, (_, seed_vec) in enumerate(elite):
        if time.time() >= deadline:
            break
        n = seed_vec.size
        sub_dead = min(deadline, time.time() + per_candidate)
        print(f"[{time.time():.1f}] Adam refinement of elite #{idx+1} (n={n})")
        a_refined = _adam_gradient_descent(seed_vec, sub_dead)

        # one short SA pass after the Adam stage (no extra deadline handling)
        a_final = _simulated_annealing(a_refined, deadline)

        cur_obj = _max_convolution_fft(a_final)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL BEST (refined) n={n}  obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 3. Final simulated‑annealing polish on the current best vector
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print(f"[{time.time():.1f}] Final SA polish, length {best_seq.size}")
        best_seq = _simulated_annealing(best_seq, deadline)

    # ------------------------------------------------------------------
    # 4. Clean‑up and final evaluation
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 3000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)

    # drop trailing entries that are numerically zero (just for neatness)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # sanity check with the provided evaluation routine
    try:
        ev = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {ev:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective = {best_obj:.6f} "
          f"length = {best_seq.size}")
    return best_seq.tolist()
