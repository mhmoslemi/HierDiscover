"""
Propose a step‑function (non‑negative sequence) that minimises
``evaluate_sequence``.

Algorithm outline
-----------------
1. **Exploratory parametric sweep** – generate many candidate families:
   * power‑law            a_i = (i+shift)^{-p}
   * symmetric power‑law a_i = (i+shift)^{-p} + (n‑1‑i+shift)^{-p}
   * exponential          a_i = r^{i}
   * Gaussian             a_i = exp(-0.5*((i‑µ)/σ)^2)
   For a grid of parameters (n, p, shift, r, σ) we evaluate the
   maximal self‑convolution (the objective after normalisation)
   and keep the best few (elite) candidates.

2. **Deterministic sub‑gradient descent** – the objective after normalising
   Σa = √(2n) is simply ``max (a⋆a)``.  The sub‑gradient of the max is
   known analytically:
       ∂/∂a_j max_k (a⋆a)_k = 2·a_{k*‑j}   (k* any index attaining the max).
   We repeatedly move opposite to this sub‑gradient, project back onto the
   ℓ¹‑simplex and gradually shrink the step size.  This converges quickly
   to a flatter convolution profile.

3. **Light‑weight simulated‑annealing polish** – a few random mass‑transfer
   moves are applied to escape tiny plateaus.

All phases respect the total time budget (default 1000 s).  The function
returns the best sequence found as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex Σx = s, x≥0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection onto {x_i ≥ 0, Σx_i = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    """a_i = (i + shift)^{-p}, projected onto Σa = √(2 n)."""
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    target = np.sqrt(2.0 * n)
    a = raw * (target / raw.sum())
    return np.maximum(a, 0.0)


def _symmetric_powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    """Symmetric power‑law: a_i = (i+shift)^{-p} + (n‑1‑i+shift)^{-p}."""
    idx = np.arange(n, dtype=float)
    left = (idx + shift) ** (-p)
    right = (n - 1 - idx + shift) ** (-p)
    raw = left + right
    target = np.sqrt(2.0 * n)
    a = raw * (target / raw.sum())
    return np.maximum(a, 0.0)


def _exponential_sequence(n: int, r: float) -> np.ndarray:
    """a_i = r^{i}, r∈(0,1), projected onto Σa = √(2 n)."""
    idx = np.arange(n, dtype=float)
    raw = r ** idx
    target = np.sqrt(2.0 * n)
    a = raw * (target / raw.sum())
    return np.maximum(a, 0.0)


def _gaussian_sequence(n: int, sigma: float, centre: float = None) -> np.ndarray:
    """Discrete Gaussian, projected onto Σa = √(2 n)."""
    if centre is None:
        centre = (n - 1) / 2.0
    idx = np.arange(n, dtype=float)
    raw = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    target = np.sqrt(2.0 * n)
    a = raw * (target / raw.sum())
    return np.maximum(a, 0.0)


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution objective
# ----------------------------------------------------------------------


def _subgradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected sub‑gradient descent for  max_k (a⋆a)_k.
    The algorithm runs until ``deadline`` is reached.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    # guarantee feasibility at the start
    a = _project_to_simplex(a, target_sum)

    rng = np.random.default_rng()
    it = 0
    # base step size – tuned empirically
    base_lr = 0.5 * target_sum

    while time.time() < deadline:
        # compute full convolution once per iteration
        b = _full_convolution_fft(a)
        M = b.max()
        active = np.where(b == M)[0]

        # sub‑gradient (sum of contributions of all active indices)
        g = np.zeros_like(a)
        for k in active:
            start = max(0, k - (n - 1))
            end = min(k, n - 1)
            idx = np.arange(start, end + 1, dtype=int)
            # contribution 2 * a_{k‑idx}
            g[idx] += 2.0 * a[k - idx]

        # diminishing step size
        lr = base_lr / np.sqrt(it + 1.0)
        # normalise step to keep updates moderate
        norm = np.linalg.norm(g)
        if norm > 0:
            step = lr / (norm + 1e-12)
            a -= step * g

        # projection back onto the simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape tiny plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1

    return a

# Very light simulated annealing polish
# ----------------------------------------------------------------------


def _light_sa(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Random mass‑transfer SA for a few seconds (only if time permits)."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    T = best_score * 0.02 + 1e-12
    cooling = 0.9997
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-15:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        frac = rng.random() * 0.12
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta

        cur = _max_convolution_fft(a)
        dE = cur - best_score
        accept = False
        if cur <= best_score:
            accept = True
        elif rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur < best_score:
                best_score = cur
                best = a.copy()
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # very occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 5e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur = _max_convolution_fft(a)
            if cur < best_score:
                best_score = cur
                best = a.copy()

    return best

            if score < best_obj:
                best_obj = score
                best_seq = a_rand.copy()
                print(f"[{time.time():.1f}] New global best RAND n={n}"
                      f" obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 2. Refinement of elite candidates (remaining time)
    # ------------------------------------------------------------------
    for rank, (init_score, seed_seq) in enumerate(elite):
        if time.time() >= deadline:
            break
        n = seed_seq.size
        # allocate a slice of the remaining budget (≈ 30 s per elite, but
        # respect the global deadline)
        sub_dead = min(deadline, time.time() + 30.0)

        print(f"[{time.time():.1f}] Refining elite #{rank+1} (n={n})")
        a_gd = _subgradient_descent_opt(seed_seq, sub_dead)

        # short SA polish (uses whatever time is left)
        a_refined = _light_sa(a_gd, deadline)

        obj = _max_convolution_fft(a_refined)
        if obj < best_obj:
            best_obj = obj
            best_seq = a_refined.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL BEST after refinement"
                  f" n={n} obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 3. Final tidy‑up
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)

    # Trim trailing near‑zeros for a cleaner output
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # sanity check with the provided evaluator
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence = {val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best obj ≈ {best_obj:.6f}"
          f" length = {best_seq.size}")

    return best_seq.tolist()
