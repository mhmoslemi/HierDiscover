"""
Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence``.

Algorithm
---------
* Phase 1 – fast random‑shape search.
  – Pick a problem size n (typically 8 k–20 k).
  – Build a candidate from a few parametric families
    (edge‑heavy power laws, reciprocal “Beta’’ shapes, mixtures,
    or a convex blend of the current best solutions).
  – Normalise to the required ℓ¹‑norm √(2·n) and evaluate the
    max‑convolution (the true objective because the evaluator
    returns exactly this value after normalisation).
  – Keep the best few candidates.

* Phase 2 – refined optimisation.
  – Starting from the current best sequence, run a smooth‑max
    Adam optimiser (log‑sum‑exp approximation of max) followed
    by a sub‑gradient phase that directly minimises the worst
    convolution entry.
  – The optimiser works on the simplex {x≥0, Σx=√(2·n)} by
    Euclidean projection after each update.

* The whole search is time‑aware (≈1000 s).  Intermediate
  scores are printed; the final   sequence is returned as a
  plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based convolution utilities (full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return a * a (full length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of (1/α)·log‑sum‑exp(α·(a*a))  w.r.t. a.
    Returns a vector of length n.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights (smooth max)
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Return the soft‑max weights w_i = exp(alpha*conv_i)/Σj exp(alpha*conv_j)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Upsampling / interpolation
# ----------------------------------------------------------------------
def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Resize an existing sequence to length n using linear interpolation."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Random / parametric initialisers
# ----------------------------------------------------------------------
def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n with ℓ¹‑norm √(2·n).
    A mixture of several families is used to diversify the start points.
    """
    S = np.sqrt(2.0 * n)

    typ = rng.integers(0, 5)

    if typ == 0 and base_seq is not None and base_seq.size > 10:
        a = _upsample_seq(base_seq, n)
        a += rng.normal(scale=a.mean() * 1e-3, size=n)
        a = np.clip(a, 0.0, None)

    elif typ == 1:                     # uniform Dirichlet(1)
        a = rng.random(n)

    elif typ == 2:                     # geometric decay from both ends
        rate = rng.uniform(0.9, 0.9995)
        left = rate ** np.arange(n)
        right = rate ** np.arange(n)[::-1]
        a = (left + right) / 2.0

    elif typ == 3:                     # edge‑heavy reciprocal shape
        idx = np.arange(n, dtype=float)
        a = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.8)
        a = a ** power

    else:                              # random convex combination of bases
        cand = []
        cand.append(rng.random(n))                     # uniform
        rate = rng.uniform(0.9, 0.9995)                 # geometric
        left = rate ** np.arange(n)
        right = rate ** np.arange(n)[::-1]
        cand.append((left + right) / 2.0)
        idx = np.arange(n, dtype=float)                # reciprocal
        rec = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.8)
        cand.append(rec ** power)
        w = rng.random(len(cand))
        w /= w.sum()
        a = sum(wi * ci for wi, ci in zip(w, cand))

    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0)
        a_sum = a.sum()
    a = a / a_sum * S
    return a


# ----------------------------------------------------------------------
# Parameterised families for the fast random‑shape phase
# ----------------------------------------------------------------------
def _edge_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Edge‑heavy power‑law shape."""
    r = rng.uniform(0.15, 2.5)            # exponent of the decay
    power = rng.uniform(0.8, 2.0)         # additional non‑linearity
    eps = rng.uniform(0.0, 2.0)           # tiny shift to avoid singularity
    idx = np.arange(n, dtype=float)
    a = ((idx + eps) ** -r + (n - idx - 1 + eps) ** -r) ** power
    return a


def _beta_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Reciprocal‑Beta shape (edge‑heavy)."""
    p = rng.uniform(-2.0, -0.2)           # negative power makes edges heavier
    idx = np.arange(n, dtype=float)
    a = ((idx + 1.0) * (n - idx)) ** p
    return a


def _mixed_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random convex blend of a uniform and an edge‑heavy shape."""
    w = rng.random()
    a1 = rng.random(n)                               # uniform
    a2 = _edge_shape(n, rng)                         # edge‑heavy
    a = w * a1 + (1 - w) * a2
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smooth log‑sum‑exp max‑approximation
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected Adam minimising the smoothed max‑norm."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    lr = 0.3                     # a little higher than before
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    alpha = 5.0                  # start with a relatively sharp soft‑max
    alpha_mul = 1.02

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = np.max(conv)
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a, S)

        # occasional random mass‑swap (helps escape plateaus)
        if rng.random() < 0.02:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.04 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < best_score or rng.random() < 0.1:
                    a = cand

        alpha = min(alpha * alpha_mul, 40.0)   # cap a bit earlier
        lr *= 0.9999

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient phase (focus on the worst entry of the convolution)
# ----------------------------------------------------------------------
def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the true max‑norm."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.12
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # build a sub‑gradient for the worst shift
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # occasional random perturbation
        if rng.random() < 0.04:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.2:
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimiser (Adam → sub‑gradient)
# ----------------------------------------------------------------------
def _optimize_combined(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Run Adam for ~60 % of the remaining budget, then a sub‑gradient phase."""
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    smooth_deadline = now + 0.6 * remaining
    a_s, score_s = _optimize_adam(a0, smooth_deadline, rng)

    if time.time() < deadline:
        a_f, score_f = _optimize_subgrad(a_s, deadline, rng)
        if score_f < score_s:
            return a_f, score_f
    return a_s, score_s
