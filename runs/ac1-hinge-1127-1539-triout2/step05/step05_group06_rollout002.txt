"""
Propose a non‑negative step sequence that minimizes the peak of its
self‑convolution (the value returned by ``evaluate_sequence``).

Algorithm
---------
1. Work on the L¹‑simplex  Σ a_i = √(2·n).  
   The objective is exactly the maximum entry of the full convolution a∗a,
   which is invariant under scaling, so we can keep the simplex constraint.

2. Three increasing problem sizes are used: 2000 → 4000 → 8000.
   For each size the current best vector is up‑sampled (linear interpolation)
   and refined.

3. Refinement consists of a long Adam optimisation on a smooth
   approximation of the max‑norm (soft‑max with a temperature that is
   increased gradually).  The gradient of the soft‑max is computed by a
   weighted convolution using FFTs.  After each step the vector is projected
   back onto the simplex.

4. When the Adam phase ends, a short projected sub‑gradient descent
   (using the true max entry) is run to polish the result.

5. Random “mass‑swap’’ and jitter moves are injected periodically to
   escape plateaus.

The whole procedure respects the global time budget (≈1000 s).  The best
vector found is returned as a plain Python list.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient term for the smooth max‑norm.
    g_j = 2 Σ_i w_{i+j} a_i .
    Both ``a`` (len n) and ``w`` (len 2·n‑1) are real vectors.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights w_i = exp(alpha·conv_i) / Σ_j exp(alpha·conv_j) (stable)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a to length new_n."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=float)
    return np.interp(x_new, x_old, a)


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def _optimize_adam_long(a0: np.ndarray,
                        deadline: float,
                        rng: np.random.Generator,
                        lr0: float = 0.25,
                        alpha_start: float = 0.4,
                        alpha_max: float = 450.0,
                        alpha_inc: float = 1.004) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max approximation of the max‑norm.
    Runs until *deadline* is reached.
    Returns the best vector (max‑conv minimal) and its associated max‑conv.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # keep best seen
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    alpha = alpha_start
    # simple learning‑rate schedule (decays roughly with √t)
    while time.time() < deadline:
        t += 1

        # --- compute current convolution and soft‑max weights ---
        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)

        # --- gradient w.r.t. a for the smooth objective ---
        grad = _conv_with_weights(a, w)

        # --- Adam update ---
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate decay (inverse‑sqrt)
        lr = lr0 / (np.sqrt(t) + 1.0)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # --- projection onto simplex ---
        a = _project_simplex(a, S)

        # --- occasional random mass‑swap (exploration) ---
        if rng.random() < 0.025:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.015
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        # --- record best ---
        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # occasional progress report
            if t % 2000 == 0:
                print(f"[Adam] iter={t:,} best_max={best_score:.6f}")

        # --- temperature increase for sharper approximation ---
        alpha = min(alpha * alpha_inc, alpha_max)

        # --- simple patience check (do not stop early) ---
        # (removed aggressive early‑stop used previously)

    return best_a, best_score


def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      eta0: float = 0.30) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent acting on the true max‑norm.
    Runs until *deadline*.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the max‑entry (see analysis)
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if it % 1000 == 0:
                print(f"[Subgrad] iter={it:,} best_max={best_score:.6f}")

        # random perturbation to help escape flat regions
        if rng.random() < 0.04:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.025
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = np.max(_full_convolution(cand))
                if cand_score < cur_score or rng.random() < 0.2:
                    a = cand
                    cur_score = cand_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()

        # No aggressive early‑stop – let it run until the slice deadline

    return best_a, best_score


def _optimize_stage(a0: np.ndarray,
                    slice_deadline: float,
                    rng: np.random.Generator) -> Tuple[np.ndarray, float]:
    """
    One optimisation stage: Adam → (if time permits) sub‑gradient.
    Returns the best vector found in this stage and its max‑conv.
    """
    # run Adam for ~60 % of the allocated time
    now = time.time()
    remaining = slice_deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    adam_deadline = now + 0.6 * remaining
    a_adam, score_adam = _optimize_adam_long(a0, adam_deadline, rng)

    # if time left, polish with sub‑gradient
    if time.time() < slice_deadline:
        a_sub, score_sub = _optimize_subgrad(a_adam, slice_deadline, rng)
        if score_sub < score_adam:
            return a_sub, score_sub
    return a_adam, score_adam

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs) -> List[float]:

    Returns the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # tiny safety margin for final check

    # ------------------------------------------------------------------
    # Initialise from a previously discovered construction if available
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:

        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Define the schedule of problem sizes (moderate → large)
    # ------------------------------------------------------------------
    size_schedule = [2000, 4000, 8000]      # can be tuned
    # allocate a fraction of total time to each stage (roughly equal)
    total_budget = deadline - time.time()
    stage_fractions = np.linspace(0.3, 0.4, len(size_schedule))
    stage_fractions = stage_fractions / stage_fractions.sum()   # normalise

    for idx, n in enumerate(size_schedule):
        # compute slice deadline for this stage

        slice_time = remaining_total * stage_fractions[idx]

    # ------------------------------------------------------------------

    final_seq = best_seq.tolist()

    print(f"[final] external_score = {external_score:.6f} "
          f"(internal max‑conv = {best_score:.6f}) length = {len(final_seq)}")
    return final_seq
