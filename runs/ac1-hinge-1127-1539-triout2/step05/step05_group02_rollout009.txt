def _resize_sequence(seq, new_len):
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0, 1, old_len)
    new_x = np.linspace(0, 1, new_len)
    new_seq = np.interp(new_x, old_x, seq)
    return new_seq

def _annealed_gradient_descent(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    a = _project_to_simplex(a, target_sum)
    best_a = a.copy()
    best_val = _score(a)
    beta = 1.0
    max_beta = 8192.0
    step_base = target_sum * 0.25
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.9
    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        # apply momentum
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        # project
        a = _project_to_simplex(a, target_sum)
        # occasional jitter
        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it} β={beta:.1f} val={best_val:.6f}")
        if it % 600 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)
        it += 1
    return best_a, best_val

def _simulated_annealing(a0, deadline, start_temp=1.0, end_temp=1e-4,
                         max_frac_start=0.3, max_frac_end=0.01,
                         max_window=50):
    a = a0.copy()
    n = a.shape[0]
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)
    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()
    conv = _conv_fft(a)  # current conv
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau
        # compute max conv idx
        k_max = int(np.argmax(conv))
        # compute sensitivities
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1)
        sens[idx] = a[k_max - idx]  # a reversed portion
        weight = a * sens
        move_type = None
        # mass transfer biased
        if rng.random() < 0.7:
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight / total_w)
            else:
                donor = rng.integers(n)
            # choose receiver among low-weight positions
            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            if low_idx.size:
                receiver = rng.choice(low_idx)
            else:
                receiver = rng.integers(n)
            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"
        elif rng.random() < 0.9:
            # local smoothing
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                old_seg = a[start:end].copy()
                avg = old_seg.mean()
                a[start:end] = avg
                move_type = "smooth"
                saved_seg = old_seg
                saved_start, saved_end = start, end
        else:
            # random swap
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"
        # compute new conv and score
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = _score(a)
        # acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it} new best {best_score:.6f}")
        else:
            # revert
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]
        # occasional jitter to escape
        if it % 8000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0)
            # renorm to same sum (scale)
            cur_sum = a.sum()
            if cur_sum > 0:
                a = a * (np.sqrt(2.0 * n) / cur_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)
        it += 1
    return best_a, best_score

def _run_one_instance(n, init_seq, deadline):
    target_sum = np.sqrt(2.0 * n)
    if init_seq is not None:
        a = init_seq.copy()
        if a.size != n:
            # resize via interpolation if sizes differ
            a = _resize_sequence(a, n)
    else:
        # plain random
        rng = np.random.default_rng()
        a = rng.random(n)
    # project onto simplex
    a = _project_to_simplex(a, target_sum)
    # gradient descent phase
    remaining = deadline - time.time()
    if remaining > 20.0:
        gd_deadline = time.time() + min(30.0, remaining * 0.3)
        a, gd_score = _annealed_gradient_descent(a, gd_deadline)
    else:
        gd_score = _score(a)
    # simulated annealing phase
    cur_time = time.time()
    if cur_time < deadline:
        a, sa_score = _simulated_annealing(a, deadline)
        final_score = min(gd_score, sa_score)
    else:
        final_score = gd_score
    return a, final_score

"""
Proposes a non‑negative step‑function (height sequence) that minimises the
benchmark defined by `evaluate_sequence`.  The optimiser proceeds in three
scales (≈1500 → 3000 → 6000 elements).  Each scale consists of

1. Projected gradient descent on a soft‑max approximation of the
   convolution maximum (β is slowly increased up to 8192).  A simple
   momentum term and occasional jitter help escaping shallow minima.

2. Simulated‑annealing refinement.  Moves are mass‑transfer from a
   “sensitive’’ index (large contribution to the current maximum
   convolution entry) to a low‑sensitivity index, short‑window smoothing
   and occasional swaps.  Temperature and the allowed transfer fraction
   decay exponentially.

The best sequence across all scales is returned.  All numbers are forced
to be non‑negative; the sequence is projected onto the ℓ¹‑simplex
Σa_i = √(2·n) after each update (scaling does not affect the objective
but improves numerical stability).
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(new_x, old_x, seq)
    return new_seq.astype(np.float64)


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient: 2 * (a reversed) ⋆ p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected gradient descent on the soft‑max approximation.
    β is increased up to 8192.  A simple momentum term and jitter are used.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 8192.0
    step_base = target_sum * 0.25
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.9
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 600 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return best_a, best_val


def _simulated_annealing(a0: np.ndarray, deadline: float,
                         start_temp: float = 1.0,
                         end_temp: float = 1e-4,
                         max_frac_start: float = 0.30,
                         max_frac_end: float = 0.01,
                         max_window: int = 50) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing refinement.
    Moves: mass transfer, short‑window smoothing, random swap.
    Temperature and allowed transfer fraction decay exponentially.
    """
    a = a0.copy()
    n = a.size
    cur_score = _score(a)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-9)
    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # index of the current maximum convolution entry
        k_max = int(np.argmax(conv))

        # sensitivities for a (derivative of the current max entry)
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]

        weight = a * sens
        move_type = None

        # ------------------- mass transfer -------------------
        if rng.random() < 0.70:
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight / total_w)
            else:
                donor = rng.integers(n)

            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            receiver = rng.choice(low_idx) if low_idx.size else rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        # ------------------- local smoothing -----------------
        elif rng.random() < 0.90:
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                old_seg = a[start:end].copy()
                avg = old_seg.mean()
                a[start:end] = avg
                move_type = "smooth"
                saved_seg = old_seg
                saved_start, saved_end = start, end

        # ------------------- swap ---------------------------
        else:
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # compute new convolution and score
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = _score(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it:5d} new best={best_score:.6f}")
        else:
            # revert the rejected move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter to escape stagnation
        if it % 8000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            cur_sum = a.sum()
            if cur_sum > 0:
                a = a * (np.sqrt(2.0 * n) / cur_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


def _run_one_instance(n: int, init_seq: np.ndarray, deadline: float):
    """
    Optimise a single length‑n instance.
    Performs a short gradient‑descent phase followed by simulated annealing.
    Returns (best_sequence, best_score).
    """
    target_sum = np.sqrt(2.0 * n)
    a = init_seq.astype(np.float64, copy=True)
    if a.size != n:
        a = _resize_sequence(a, n)
    a = _project_to_simplex(a, target_sum)

    # Gradient‑descent phase (≈30 s or the first 30 % of the remaining budget)
    remaining = deadline - time.time()
    if remaining > 15.0:
        gd_deadline = time.time() + min(30.0, remaining * 0.30)
        a, gd_score = _annealed_gradient_descent(a, gd_deadline)
    else:
        gd_score = _score(a)

    # SA phase for the rest of the allocated time
    if time.time() < deadline:
        a, sa_score = _simulated_annealing(a, deadline)
        final_score = min(gd_score, sa_score)
    else:
        final_score = gd_score

    return a, final_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a height sequence with low benchmark score.
    The routine works on three increasing lengths (≈1500 → 3000 → 6000)
    re‑using the best solution of the previous scale as a warm start.
    """
    start = time.time()
    deadline = start + budget_s - 1.0      # safety margin

    best_seq = None
    best_score = float("inf")
    cur_seq = None

    # Scales and fraction of the remaining time allotted to each scale
    scales = [(1500, 0.20), (3000, 0.30), (6000, 0.50)]

    for n, frac in scales:
        now = time.time()
        if now >= deadline - 5.0:
            break
        remaining = deadline - now
        stage_deadline = now + remaining * frac

        # Initialise: up‑scale the current best if we have one,
        # otherwise fall back to the globally supplied sequence or a power‑law.
        if cur_seq is not None:
            init = _resize_sequence(cur_seq, n)
        else:
            if "height_sequence_1" in globals():
                init = _resize_sequence(np.array(globals()["height_sequence_1"],
                                                dtype=np.float64), n)
            else:
                init = _power_law_sequence(n, 0.55, np.sqrt(2.0 * n))

        seq, score = _run_one_instance(n, init, stage_deadline)

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Scale n={n} improved best score={best_score:.6f}")

        cur_seq = seq  # warm start for the next (larger) scale

    # Trim trailing near‑zeros for neatness
    if best_seq is None:
        best_seq = []

    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # One final evaluation using the provided benchmark (optional)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
