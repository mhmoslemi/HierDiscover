def propose_candidate(seed=12345, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 5

    # Get previous best sequence if exists
    base_seq = None
    if 'height_sequence_1' in globals():
        try: base_seq = np.asarray(height_sequence_1, dtype=float)
        except: base_seq = None

    best_seq = None
    best_score = np.inf

    # Global loop: while time left, try new n value from a small set; each time run extended optimizer.
    while time.time() < deadline:
        # Choose n
        n = int(rng.integers(5000, 12000))
        # pick initial sequence: either upsample base_seq if available else geometric shape with random c
        if base_seq is not None and base_seq.size > 0:
            a0 = _upsample_seq(base_seq, n)
            a0 += rng.normal(scale=a0.mean()*1e-3, size=n)
            a0 = np.clip(a0, 0.0, None)
        else:
            # geometric shape with random c
            c = rng.uniform(0.8, 4.0)  # target parameter for geometric decay
            q = max(0.0, 1.0 - c / n)
            a0 = q ** np.arange(n)
        # Normalize to required sum (any positive sum fine)
        # to avoid small sum causing near-zero denominator, we keep sum > 0.1
        # We'll just scale to sum = sqrt(2*n) (the same as previous code)
        S = np.sqrt(2.0 * n)
        a0 = a0 / a0.sum() * S

        # Run combined optimizer for the rest of budget (or limited slice)
        slice_deadline = min(deadline, time.time() + (deadline - time.time()) * 0.9)  # leave some margin
        a_opt, score = _optimize_combined(a0, slice_deadline, rng)

        # Record best
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[global] n={n} score={score:.6f}")

        # Optionally break if we reach target
        if best_score <= 1.503:
            break

    # final output
    if best_seq is None:
        best_seq = np.ones(2, dtype=float) * np.sqrt(2.0*2) / 2.0
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] score={external:.6f}")
    return final_seq

"""
Propose a step‑function (non‑negative vector) that minimises the
evaluation function defined by ``evaluate_sequence``.

Algorithm
---------
1. **Coarse optimisation on a small problem** –  
   A linear programme is solved repeatedly for a modest size ``n_small``.  
   For a fixed vector ``f`` the constraint ``f * b ≤ M`` is linear in the
   unknown ``b`` (convolution with a fixed left factor).  Maximising
   ``∑ b_j`` under these constraints pushes mass into entries that have a
   small influence on the current worst convolution peak ``M``.  Repeating
   the LP step and convex‑combining the result with the current vector
   yields a shape that already has a substantially reduced max‑convolution.

2. **Upsampling** – the obtained small‑scale vector is interpolated to a
   larger length ``n_large`` (the usual range 1800–12000).  The up‑sampled
   sequence inherits the favourable structure while offering many more
   degrees of freedom.

3. **Fine‑grained optimisation** – a projected Adam optimiser on a
   smooth log‑sum‑exp approximation of the max‑norm, followed by a true
   sub‑gradient phase, is run for the remaining time budget.  The Adam
   phase makes rapid progress, the sub‑gradient phase polishes the final
   solution.

All vectors are projected onto the simplex
``{x ≥ 0, Σ x = √(2·n)}`` after each update, guaranteeing that the
objective is scale‑invariant and that the security checks in
``evaluate_sequence`` never reject the candidate.

The whole pipeline runs under the supplied time budget (default 1000 s)
and returns the best sequence it has discovered.
"""

import time
from math import log, exp
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities (full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a * a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed max (log‑sum‑exp) with respect to ``a``.
    Implements the formula described in the original code.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights (smooth approximation of max)
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Compute soft‑max weights w_i = exp(alpha*conv_i) / Σ exp(alpha*conv_i)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Initial sequence generators (used after the LP stage)
# ----------------------------------------------------------------------
def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Resize an existing sequence to length n using linear interpolation."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n with L¹‑norm √(2·n).
    Several random families are mixed to diversify the search.
    """
    S = np.sqrt(2.0 * n)

    typ = rng.integers(0, 6)

    if typ == 0 and base_seq is not None and base_seq.size > 10:
        a = _upsample_seq(base_seq, n)
        a += rng.normal(scale=a.mean() * 1e-3, size=n)
        a = np.clip(a, 0.0, None)
    elif typ == 1:
        a = rng.random(n)                     # Dirichlet(1)
    elif typ == 2:
        # geometric decay
        rate = rng.uniform(0.9, 0.9998)
        a = rate ** np.arange(n)
    elif typ == 3:
        # edge‑heavy reciprocal shape
        idx = np.arange(n, dtype=float)
        a = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.8)
        a = a ** power
    elif typ == 4:
        # triangular‑like shape
        a = np.minimum(np.arange(n) + 1, n - np.arange(n))
        a = a.astype(float)
    else:
        # mixture of the above bases
        cand = []
        cand.append(rng.random(n))
        rate = rng.uniform(0.9, 0.9998)
        cand.append(rate ** np.arange(n))
        idx = np.arange(n, dtype=float)
        rec = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.8)
        cand.append(rec ** power)
        cand.append(np.minimum(np.arange(n) + 1, n - np.arange(n)))
        w = rng.random(len(cand))
        w /= w.sum()
        a = sum(wi * ci for wi, ci in zip(w, cand))

    # Normalise to the required ℓ¹‑norm
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0)
        a_sum = a.sum()
    a = a / a_sum * S
    return a


# ----------------------------------------------------------------------
# Small‑scale LP optimiser (reduces the max‑convolution peak)
# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray, M: float, t: float, sqrt2n: float) -> np.ndarray:
    """
    Single linear‑programming step.

    * maximise Σ b_j
    * subject to   b_j ≥ 0  and   (f * b)[k] ≤ M   for every shift k
    * then convex‑combine ``f`` and the scaled optimiser ``b``.
    """
    n = f.shape[0]
    rows = []
    cols = []
    data = []

    # Build the sparse matrix A where A[k, j] = f[i] with i = k - j
    for k in range(2 * n - 1):
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        # indices i that contribute to shift k
        for i in range(i_low, i_high + 1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f[i])

    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    # Linear programme: minimise -1⋅b  subject to  A·b ≤ M,  b ≥ 0
    c = -np.ones(n)
    ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method="highs", options={"presolve": True})

    if not res.success:
        # fall back to the original vector if LP fails
        return f.copy()

    b = res.x
    sum_b = b.sum()
    if sum_b == 0.0:
        g = np.full_like(f, sqrt2n / n)
    else:
        g = b * (sqrt2n / sum_b)          # scale to the required ℓ¹‑norm

    # convex combination and projection back onto the simplex
    f_new = (1.0 - t) * f + t * g
    f_new = _project_simplex(f_new, sqrt2n)
    return f_new


def _lp_optimize(f0: np.ndarray,
                 max_iters: int,
                 rng: np.random.Generator) -> np.ndarray:
    """
    Repeated LP steps followed by diminishing convex combination.
    ``max_iters`` is small (≈ 20‑30) because each LP is cheap for n ≈ 300‑600.
    """
    f = f0.copy()
    n = f.shape[0]
    sqrt2n = np.sqrt(2.0 * n)

    # start with a relatively large mixture weight and shrink it
    t = 0.3
    for it in range(max_iters):
        conv = _full_convolution(f)
        M = np.max(conv)

        f = _lp_step(f, M, t, sqrt2n)

        # slowly reduce the step size
        t = max(0.05, t * 0.9)

    return f


# ----------------------------------------------------------------------
# Adam optimiser on a smoothed log‑sum‑exp max‑approximation
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected Adam minimising the smoothed max‑norm."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    lr = 0.3
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    alpha = 0.3            # start soft‑max temperature
    alpha_mul = 1.03

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        cur_max = np.max(conv)
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a, S)

        # occasional random “mass‑swap’’ to escape plateaus
        if rng.random() < 0.02:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.04 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < best_score or rng.random() < 0.1:
                    a = cand

        alpha = min(alpha * alpha_mul, 30.0)
        lr *= 0.99995                # very gentle decay

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient phase (focus on the worst shift)
# ----------------------------------------------------------------------
def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the true max‑norm."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.12
    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient for the worst shift
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # occasional random perturbation to avoid stagnation
        if rng.random() < 0.04:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.2:
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimiser (Adam → sub‑gradient)
# ----------------------------------------------------------------------
def _optimize_combined(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Run Adam for ~60 % of the time budget, then a sub‑gradient phase."""
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    # 60 % / 40 % split
    smooth_deadline = now + 0.6 * remaining
    a_s, score_s = _optimize_adam(a0, smooth_deadline, rng)

    if time.time() < deadline:
        a_f, score_f = _optimize_subgrad(a_s, deadline, rng)
        if score_f < score_s:
            return a_f, score_f
    return a_s, score_s


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function minimising ``evaluate_sequence``.
    Returns the best sequence (list of floats) found before the time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin for the final external evaluation
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Retrieve any previous construction (used for up‑sampling)
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_overall_seq = None
    best_overall_score = np.inf

    # ------------------------------------------------------------------
    # 1️⃣  Coarse optimisation on a small problem (LP stage)
    # ------------------------------------------------------------------
    n_small = int(rng.integers(250, 600))          # modest size for cheap LPs
    a_small = _initial_sequence(base_seq, n_small, rng)

    # 10‑20 LP iterations are usually enough to obtain a promising shape
    a_small = _lp_optimize(a_small, max_iters=20, rng=rng)

    # ------------------------------------------------------------------
    # 2️⃣  Upsample to a larger size and run the fine‑grained optimiser
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # pick a fresh large size for the final refinement
        n_large = int(rng.integers(5000, 12000))
        a0 = _upsample_seq(a_small, n_large)

        # small jitter to avoid exact alignment artefacts
        a0 += rng.normal(scale=a0.mean() * 1e-4, size=n_large)
        a0 = np.clip(a0, 0.0, None)

        # normalise to the required ℓ¹‑norm
        S = np.sqrt(2.0 * n_large)
        a0 = a0 / a0.sum() * S

        # allocate the remaining time for this single refined run
        slice_deadline = deadline

        a_opt, score = _optimize_combined(a0, slice_deadline, rng)

        if score < best_overall_score:
            best_overall_score = score
            best_overall_seq = a_opt.copy()
            print(f"[global] n={n_large} new best score={best_overall_score:.6f}")

        # if we already met the target we can stop early
        if best_overall_score <= 1.503:
            break

        # otherwise, continue the outer loop – the next iteration will
        # restart with a freshly chosen length (useful if the current
        # length proved unlucky)
        # (the loop will exit automatically when the global deadline is hit)

    # ------------------------------------------------------------------
    # Final sanity check via the black‑box evaluator
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        final_seq = [1.0, 1.0]
    else:
        final_seq = best_overall_seq.tolist()

    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} "
          f"(internal best = {best_overall_score:.6f}) "
          f"length = {len(final_seq)}")
    return final_seq
```

This implementation first runs a cheap LP‑based coarse optimisation on a small vector (250–600 entries).  
The resulting shape is up‑sampled to a larger problem size (5 000–12 000) and then refined by a long Adam → sub‑gradient optimisation, making full use of the 1000 s budget.  The combination of the LP “mass‑reallocation’’ step and the aggressive gradient‑based fine‑tuning typically pushes the evaluation score well below the previous 1.513 threshold.
