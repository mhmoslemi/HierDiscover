def step_lp(f):
    n = len(f)
    M = np.max(np.convolve(f,f))
    # Build constraint matrix A (2n-1 rows, n cols)
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        start = max(0, k - (n-1))
        end = min(k, n-1)
        for i in range(start, end+1):
            j = k - i
            # term f_i * g_j
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # inequality: A @ g <= M
    # plus g >= 0
    # objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if not res.success: return f (or break)
    g0 = res.x
    # Rescale sum to sqrt(2n)
    target = np.sqrt(2.0 * n)
    g = g0 * (target / np.sum(g0)) if np.sum(g0) > 0 else g0
    return g
```

"""
Hybrid LP + simulated‑annealing optimizer for the max‑convolution benchmark.

The routine performs a few linear‑programming refinements that spread the
mass of the sequence while keeping the convolution with the current iterate
below its present maximum.  After the LP phase a long simulated‑annealing
run (mass transfers, short‑window smoothing and random swaps) finishes the
search.  Multiple restarts and several initialisation strategies are used
within the 1000 s time budget.  The best sequence discovered is returned.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (real part)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Fast evaluation of the objective.
    For a normalised sum (Σa = sqrt(2·n)) this equals max(conv(a,a)).
    """
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) sequence of length n, normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------


def _lp_optimize(f: np.ndarray, M: float, target_sum: float) -> np.ndarray | None:
    """
    Solve  max Σ g_j  s.t. g≥0 and (f * g)[k] ≤ M  (for all k).
    Returns the maximiser g (scaled later to the target ℓ¹‑norm) or None
    if the LP fails.
    """
    n = f.size
    # Build the sparse Toeplitz‑like constraint matrix A (shape (2n‑1, n)):
    #   (f * g)[k] = Σ_j f_{k-j} * g_j   (indices outside [0,n‑1] give 0)
    rows = []
    cols = []
    data = []

    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx_i = np.arange(i_start, i_end + 1, dtype=np.int64)
        idx_j = k - idx_i
        rows.extend([k] * len(idx_i))
        cols.extend(idx_j.tolist())
        data.extend(f[idx_i].tolist())

    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # maximise Σg ⇒ minimise –Σg
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    except Exception:      # safety net – should never happen in the contest
        return None

    if not res.success:
        return None

    g0 = res.x
    sum_g = g0.sum()
    if sum_g <= 1e-12:
        return None
    # Scale to the required ℓ¹‑norm (does not keep the constraints,
    # but the subsequent blending step will keep the overall max low).
    g = g0 * (target_sum / sum_g)
    return g


# ----------------------------------------------------------------------
# Simulated annealing (largely the same as the previous baseline)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float,
                         start_temp: float = 0.5,
                         end_temp: float = 1e-5,
                         max_frac_start: float = 0.40,
                         max_frac_end: float = 0.005,
                         max_window: int = 30) -> tuple[np.ndarray, float]:
    """
    Refine a candidate with a temperature‑controlled random walk.
    Moves:
        * mass transfer donor→receiver,
        * short‑window smoothing,
        * random index swap.
    """
    a = a0.copy()
    n = a.size
    cur_score = _score(a)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)

    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()
    conv = _conv_fft(a)                     # current convolution

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # locate the (one of) the current maximum entry(s)
        max_b = conv.max()
        k_max = int(np.argmax(conv))

        # sensitivity of each coordinate w.r.t. the active entry
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]

        weight = a * sens

        # --------------------------------------------------------------
        # propose a move
        # --------------------------------------------------------------
        move_type = None
        # 70 % mass‑transfer
        if rng.random() < 0.70:
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight / total_w)
            else:
                donor = rng.integers(n)

            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            receiver = rng.choice(low_idx) if low_idx.size else rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        # 20 % local smoothing
        elif rng.random() < 0.90:
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                old_seg = a[start:end].copy()
                avg = old_seg.mean()
                a[start:end] = avg
                move_type = "smooth"
                saved_seg = old_seg
                saved_start, saved_end = start, end

        # 10 % random swap
        else:
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # --------------------------------------------------------------
        # evaluate the proposed state (single FFT)
        # --------------------------------------------------------------
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = _score(a)

        # --------------------------------------------------------------
        # Metropolis acceptance
        # --------------------------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA new best = {best_score:.6f}")
        else:
            # revert the move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional jitter to escape deep minima
        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = a * (a.sum() / a.sum())   # re‑normalise to keep the sum unchanged
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises the benchmark.
    The strategy:
        – several initialisations (global warm‑start, power‑law, random Dirichlet);
        – a short linear‑programming phase that spreads the mass while
          respecting the current convolution bound;
        – a long simulated‑annealing refinement.
    The best sequence found before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 1.0          # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper to produce an initial vector of length n
    # ------------------------------------------------------------------
    def _initial_vector(n: int, mode: str) -> np.ndarray:
        target = np.sqrt(2.0 * n)
        if mode == "global" and "height_sequence_1" in globals():
            seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
            if seq.size > n:
                seq = seq[:n]
            else:
                seq = np.pad(seq, (0, n - seq.size), "constant")
        elif mode == "power":
            exp = rng.uniform(0.30, 0.80)
            seq = (np.arange(1, n + 1, dtype=np.float64) ** (-exp))
        else:   # random Dirichlet‑like draw
            seq = rng.random(n)
        seq = seq.astype(np.float64)
        seq = _project_to_simplex(seq, target)
        return seq

    # ------------------------------------------------------------------
    # Candidate lengths – we give the longest runs higher priority.
    # ------------------------------------------------------------------
    lengths = [1200, 1500, 1800, 2000, 2500]

    for n in lengths:
        if time.time() >= deadline:
            break

        # allow a few restarts per length (usually one is enough)
        for run_idx in range(2):
            if time.time() >= deadline:
                break

            # ------------------------------------------------------------------
            # Initialise a candidate
            # ------------------------------------------------------------------
            mode = "global" if (run_idx == 0 and "height_sequence_1" in globals()) else \
                   ("power" if run_idx % 2 else "random")
            f = _initial_vector(n, mode)

            target_sum = np.sqrt(2.0 * n)

            # ------------------------------------------------------------------
            # Small LP refinement phase (max 4 inner iterations)
            # ------------------------------------------------------------------
            sub_deadline = min(deadline, time.time() + 150.0)   # at most 150 s for LP + early SA
            lp_iter = 0
            while time.time() < sub_deadline and lp_iter < 4:
                M = _max_convolution_fft(f)
                g = _lp_optimize(f, M, target_sum)
                if g is None:
                    break
                # blending factor (large at first, then decays)
                t = 0.5 if lp_iter == 0 else (0.3 if lp_iter == 1 else 0.1)
                f_new = (1.0 - t) * f + t * g
                f_new = _project_to_simplex(f_new, target_sum)

                # keep the move only if it improves the objective
                if _score(f_new) < _score(f):
                    f = f_new
                else:
                    break
                lp_iter += 1

            # ------------------------------------------------------------------
            # Simulated‑annealing refinement for the remaining time budget
            # ------------------------------------------------------------------
            if time.time() < deadline:
                f, final_score = _simulated_annealing(f, deadline)

            # ------------------------------------------------------------------
            # Record the best sequence seen so far (using the official scorer)
            # ------------------------------------------------------------------
            try:
                eval_score = evaluate_sequence(f.tolist())
            except Exception:
                eval_score = float("inf")
            if eval_score < best_score:
                best_score = eval_score
                best_seq = f.copy()
                print(f"[{time.time():.1f}] NEW GLOBAL BEST: n={n}, mode={mode}, "
                      f"score={best_score:.6f}")

    # ----------------------------------------------------------------------
    # Fallback – should never be needed, but guarantees a non‑empty return
    # ----------------------------------------------------------------------
    if best_seq is None:
        n = 1500
        best_seq = _power_law_sequence(n, 0.55, np.sqrt(2.0 * n))

    # Strip trailing zeros for a compact representation
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = np.array([], dtype=np.float64)

    print(f"[{time.time():.1f}] SEARCH FINISHED. Best score = {best_score:.6f}, "
          f"length = {len(best_seq)}")
    return best_seq.tolist()
