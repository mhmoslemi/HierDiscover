def local_improvement(a, max_iter, deadline):
    while time.time()<deadline and iter<max_iter:
        b = _max_convolution_fft(a)  # also get full conv for index min?
        # But we need both max and min indices.
        full = _full_convolution_fft(a)
        k_max = np.argmax(full)
        k_min = np.argmin(full)
        # find indices that contribute.
        # for k_max: possible i range max(0,k_max-(n-1))..min(n-1,k_max)
        lo_max = max(0, k_max - (n - 1))
        hi_max = min(n - 1, k_max)
        idx_max = np.arange(lo_max, hi_max+1)
        contrib_max = a[idx_max] * a[k_max - idx_max]
        i = idx_max[np.argmax(contrib_max)]

        lo_min = max(0, k_min - (n - 1))
        hi_min = min(n - 1, k_min)
        idx_min = np.arange(lo_min, hi_min+1)
        contrib_min = a[idx_min] * a[k_min - idx_min]
        # choose j with smallest product (maybe zero)
        j = idx_min[np.argmin(contrib_min)]

        if i == j: break

        # transfer delta = min(transfer_rate * a[i], a[i]) maybe limited
        delta = min(a[i] * 0.05, a[i])  # 5% of a_i
        if delta <= 1e-12: break

        a[i] -= delta
        a[j] += delta

        # compute new max, if not improved, revert some probability acceptance with temperature.
        # for simplicity accept only if max decreases.
        new_full = _full_convolution_fft(a)
        new_max = new_full.max()
        if new_max < full.max():
            # keep change
            best = a.copy()
            best_max = new_max
        else:
            # revert move
            a[j] -= delta
            a[i] += delta
        # maybe occasional random jitter

def _grad_max(a):
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_val = b.max()
    active = np.isclose(b, max_val, atol=1e-12)   # bool mask
    # compute p = active / active_count
    p = active.astype(float)

    a_rev = a[::-1]
    size = 1 << ((n + (2*n-1) - 1).bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size)
    P_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * P_fft, n=size)[: n + (2*n-1) - 1]
    g = 2 * conv_rev[n-1:n-1+n]
    return g

"""
Improved descent for the self‑convolution max‑objective.

The algorithm mixes three ideas:

1. **Analytic seed** – the sequence a_i = C(2i,i) / 4**i
   (the “inverse‑square‑root” kernel) gives a *perfect* constant
   self‑convolution on the first n entries, i.e. max(b)=1.
   After scaling to the ℓ¹‑simplex this provides a very good
   starting point close to the optimum.

2. **Projected max‑descent** – we minimise the exact ℓ∞‑norm of
   the convolution by a sub‑gradient method.  The sub‑gradient of
   max(b) is the average of the gradients of the active (maximal)
   convolution entries and can be computed by a single FFT‑based
   convolution with a tiny “active‑mask’’ vector.  After each step
   we project back onto the ℓ¹‑simplex (Euclidean projection).

3. **Refined targeted simulated annealing** – a cheap local move
   that shifts a fraction of mass from an index contributing heavily
   to the current maximal convolution entry to an index that
   contributes little to the minimal entry.  The move is accepted
   with a temperature schedule, allowing occasional uphill steps.

All three phases respect a global time budget (default 1000 s).  The
best sequence seen so far is returned as a plain list of non‑negative
floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities (full linear convolution)
# ----------------------------------------------------------------------


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a*a (length 2·n‑1) using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using the fast convolution above."""
    return float(_full_convolution_fft(a).max())


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex Σx_i = s , x_i ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = s} (Duchi et al. 2008)."""
    if s <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic “inverse‑square‑root’’ seed (a_i = C(2i,i)/4**i)
# ----------------------------------------------------------------------


def _inverse_sqrt_shape(n: int) -> np.ndarray:
    """
    Return a length‑n non‑negative sequence a_i = comb(2i,i) / 4**i.
    The recurrence a_{i+1} = a_i * (2i+1) / (2(i+1)) is used for stability.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return a


# ----------------------------------------------------------------------
# Sub‑gradient of max(b) where b = a*a (full linear convolution)
# ----------------------------------------------------------------------


def _grad_max(a: np.ndarray) -> np.ndarray:
    """
    Return a sub‑gradient of the function max_k (a*a)_k.
    The gradient is the average of the gradients of all active (maximal)
    convolution entries.  It is computed via a single FFT convolution.
    """
    n = a.shape[0]

    # full convolution and active mask
    b = _full_convolution_fft(a)
    max_val = b.max()
    # active entries: those within a tiny absolute tolerance of the max
    tol = 1e-12
    active = (b >= max_val - tol).astype(np.float64)
    cnt = active.sum()
    if cnt == 0:
        # numerical safety – fall back to a single max index
        active[:] = 0.0
        active[np.argmax(b)] = 1.0
        cnt = 1.0
    p = active / cnt           # probability weight on each k

    # gradient = 2 * (a_rev ⋆ p)  (central n entries)
    a_rev = a[::-1]

    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size = 1 << ((conv_len).bit_length())
    Afft = np.fft.rfft(a_rev, n=size)
    Pfft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Afft * Pfft, n=size)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g


# ----------------------------------------------------------------------
# Projected max‑descent optimiser (sub‑gradient method)
# ----------------------------------------------------------------------


def _max_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Sub‑gradient descent on the exact max‑convolution objective.
    The iterate stays on the ℓ¹‑simplex automatically via Euclidean
    projection after each step.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    # initialise on the simplex (already true for our seeds but keep safe)
    s = a.sum()
    if s > 0:
        a *= target_sum / s
    else:
        a = np.full_like(a, target_sum / n)

    step = 0
    best_max = _max_convolution_fft(a)

    while time.time() < deadline:
        g = _grad_max(a)

        # simple diminishing step size
        eta = 0.5 / np.sqrt(step + 1.0)

        # gradient step
        a_new = a - eta * g

        # enforce non‑negativity and simplex constraint
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_to_simplex(a_new, target_sum)

        new_max = _max_convolution_fft(a_new)

        # accept if we made progress – otherwise keep the old point
        if new_max < best_max:
            a = a_new
            best_max = new_max
        else:
            # occasional acceptance of uphill steps (simulated‑annealing flavour)
            # temperature decays with the iteration counter
            T = max(1e-3, 0.1 / np.sqrt(step + 1.0))
            if np.random.rand() < np.exp(-(new_max - best_max) / T):
                a = a_new
                best_max = new_max

        step += 1

    return a


# ----------------------------------------------------------------------
# Targeted simulated annealing (as in the previous baseline)
# ----------------------------------------------------------------------


def _targeted_simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Local move: transfer a fraction of mass from an index that
    contributes heavily to the current maximal convolution entry to a
    random destination.  Upside moves are always accepted, downhill
    moves with a temperature schedule.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # initial temperature proportional to the current objective
    T = best_score * 0.02 + 1e-8
    cooling = 0.9995
    rng = np.random.default_rng()
    max_transfer = 0.25  # up to 25 % of the selected mass

    while time.time() < deadline:
        b = _full_convolution_fft(a)
        k_max = int(np.argmax(b))

        # indices that affect b[k_max]
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idx_range = np.arange(lo, hi + 1)
        contrib = a[idx_range] * a[k_max - idx_range]
        weight_sum = contrib.sum()

        if weight_sum > 0:
            probs = contrib / weight_sum
            i = rng.choice(idx_range, p=probs)
        else:
            i = rng.integers(0, n)

        # random destination j ≠ i
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        delta = a[i] * rng.random() * max_transfer
        if delta <= 0.0:
            continue
        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score

        accept = False
        if cur_score <= best_score:
            accept = True
        elif rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # revert
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # occasional global jitter to escape deep minima
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, np.sqrt(2.0 * n))
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()

    return best


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` under the supplied time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_obj = np.inf

    # ------------------------------------------------------------------
    # Helper to compute the official objective (without the expensive
    # regularisation performed inside evaluate_sequence).
    # ------------------------------------------------------------------
    def _objective(a: np.ndarray) -> float:
        n = a.size
        max_b = _max_convolution_fft(a)
        sum_a = a.sum()
        return 2.0 * n * max_b / (sum_a ** 2)

    # ------------------------------------------------------------------
    # Phase 1 – try a few deterministic analytic seeds (different lengths)
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000, 10000]

    for n in lengths:
        if time.time() >= deadline:
            break

        a = _inverse_sqrt_shape(n)
        # project to the required ℓ¹‑simplex (sum = √(2 n))
        a = _project_to_simplex(a, np.sqrt(2.0 * n))

        obj = _objective(a)
        if obj < best_obj:
            best_obj, best_seq = obj, a.copy()
            print(f"[{time.time():.1f}] Analytic seed n={n}  obj={obj:.6f}")

        # Phase 2 – projected max‑descent on this seed
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        md_deadline = time.time() + remaining * 0.4
        a_md = _max_descent_opt(a, md_deadline)
        obj_md = _objective(a_md)
        if obj_md < best_obj:
            best_obj, best_seq = obj_md, a_md.copy()
            print(f"[{time.time():.1f}] max‑descent improvement n={n}  obj={obj_md:.6f}")

        # Phase 3 – targeted simulated annealing
        remaining = deadline - time.time()
        if remaining < 3.0:
            break
        a_sa = _targeted_simulated_annealing(best_seq, deadline)
        obj_sa = _objective(a_sa)
        if obj_sa < best_obj:
            best_obj, best_seq = obj_sa, a_sa.copy()
            print(f"[{time.time():.1f}] SA improvement n={n}  obj={obj_sa:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – random restarts (Dirichlet‑style) for the remaining time
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # pick a random length (log‑uniform between 1500 and 12000)
        n = int(np.exp(rng.uniform(np.log(1500), np.log(12000))))
        target_sum = np.sqrt(2.0 * n)

        # Dirichlet random point on the simplex
        a = rng.dirichlet(alpha=np.ones(n))
        a = a * target_sum

        obj = _objective(a)
        if obj < best_obj:
            best_obj, best_seq = obj, a.copy()
            print(f"[{time.time():.1f}] Random seed n={n}  obj={obj:.6f}")

        # short max‑descent (≈10 % of remaining time)
        remaining = deadline - time.time()
        if remaining < 2.0:
            break
        md_deadline = time.time() + remaining * 0.1
        a_md = _max_descent_opt(a, md_deadline)
        obj_md = _objective(a_md)
        if obj_md < best_obj:
            best_obj, best_seq = obj_md, a_md.copy()
            print(f"[{time.time():.1f}] max‑descent (random) n={n}  obj={obj_md:.6f}")

        # final SA on this candidate (use whatever time left)
        if time.time() < deadline:
            a_sa = _targeted_simulated_annealing(best_seq, deadline)
            obj_sa = _objective(a_sa)
            if obj_sa < best_obj:
                best_obj, best_seq = obj_sa, a_sa.copy()
                print(f"[{time.time():.1f}] SA (random) n={n}  obj={obj_sa:.6f}")

    # ------------------------------------------------------------------
    # Strip trailing near‑zero entries and return as a plain list
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform sequence of length 3000
        n = 3000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)

    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # final sanity check with the official evaluator
    try:
        ev = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence = {ev:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective = {best_obj:.6f}   length = {best_seq.size}")
    return best_seq.tolist()
