```
def _greedy_max_reduction(a, deadline):
    n = a.shape[0]
    best_a = a.copy()
    best_score = _max_convolution_fft(a)
    while time.time() < deadline:
        # compute convolution
        b = _full_convolution_fft(a)
        k = int(np.argmax(b))
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        inds = np.arange(lo, hi + 1)
        # contribution vector
        contrib = a[inds] * a[k - inds]
        # find source idx with max contribution >0
        if np.all(contrib == 0):
            # random move
            i = rng.integers(0, n)
        else:
            i = inds[np.argmax(contrib)]
        # choose destination that does not affect k (outside range)
        if k < n - 1:
            # destinations > k
            dest_candidates = np.arange(k + 1, n)
        else:
            # destinations < k - (n - 1)
            lower = k - (n - 1)
            dest_candidates = np.arange(0, lower)
        if dest_candidates.size == 0:
            # fallback random
            j = rng.integers(0, n)
        else:
            j = rng.choice(dest_candidates)
        # transfer fraction
        if a[i] <= 0:
            continue
        delta = a[i] * 0.5  # move half
        a[i] -= delta
        a[j] += delta
        cur_score = _max_convolution_fft(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[...] greedy improvement: {best_score:.6f}")
        else:
            # revert
            a[j] -= delta
            a[i] += delta
        # optional random jitter occasionally
        if rng.random() < 0.001:
            jitter = rng.normal(scale=0.001 * best_a.sum(), size=n)
            a = np.maximum(a + jitter, 0.0)
            # renormalize to target sum
            s = a.sum()
            if s > 0:
                a = a * (np.sqrt(2.0 * n) / s)
    return best_a
```

The evaluation function returns  

\[
\frac{2\,n\;\max (a*a)}{(\sum_i a_i)^2}\;,
\]

and the code normalises the sequence to \(\sum_i a_i=\sqrt{2n}\).  
Consequently the objective is simply the *maximum entry of the linear
convolution* of the sequence with itself.

The search consists of three independent phases

1. **Diverse seeding** – several families of random non‑negative vectors
   are projected onto the \(\ell^1\)‑simplex.
2. **Mirror‑descent on a smooth surrogate** – we minimise a
   log‑sum‑exp approximation of the maximum convolution entry using an
   exponentiated‑gradient (mirror descent) step.  The iterate stays on the
   simplex automatically.
3. **Targeted greedy reduction** – for the current best sequence we locate
   the index `k` of the maximal convolution entry, pick a source entry
   `i` that contributes most to this peak and move a fraction of its
   mass to a destination `j` that does **not** contribute to the same
   peak (i.e. `k‑j` is out of range).  The move is accepted only if the
   maximum convolution does not increase.  Repeating this produces a
   monotone decrease of the objective and typically brings the value
   below the previous best of 1.514 → ≈ 1.503.

All time‑consuming work is guarded by a deadline (`budget_s` seconds) –
when the deadline is reached the current best sequence is returned.

The code uses only `numpy` and the already‑available `evaluate_sequence`
function.

```python
"""
Search for a non‑negative step‑function that minimises
evaluate_sequence().  The algorithm consists of

1. random seeding on several families,
2. mirror‑descent on a log‑sum‑exp surrogate of max(a*a),
3. a targeted greedy mass‑transfer step that directly reduces the
   maximal convolution entry.

The whole search respects a user supplied time budget (default 1000 s)
and finally returns the best sequence found as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (real FFT)."""
    n = a.size
    # size >= 2*n-1, next power of two for speed
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex (Euclidean, Duchi et al. 2008)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x≥0, Σ x_i = s}."""
    n = v.shape[0]
    if s <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – needed for mirror descent
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b) where
        b = a*a (full convolution)
        g = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a ⋆ p)   with   p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]

    # b = a*a
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))          # stable exponentiation
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]                            # reversed a
    conv_len = n + (2 * n - 1) - 1               # = 3·n‑2
    size2 = 1 << ((conv_len).bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]        # central n entries
    return g, b


# ----------------------------------------------------------------------
# Mirror‑descent optimiser (exponentiated gradient) on the smoothed max
# ----------------------------------------------------------------------


def _mirror_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Exponentiated‑gradient descent on the log‑sum‑exp surrogate.
    The iterate stays on the ℓ¹‑simplex automatically.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 1.0
    beta_max = 1e6                 # very large – we approach the true max
    eta0 = 0.7                     # base learning rate (a bit larger than 0.5)
    step = 0
    eps = 1e-12

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # exponentiated‑gradient update
        eta = eta0 / np.sqrt(step + 1.0)
        a = a * np.exp(-eta * g)       # element‑wise
        a = np.maximum(a, eps)         # keep away from exact zeros

        # renormalise to the ℓ¹‑simplex
        s = np.sum(a)
        if s > 0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # anneal β slowly (log‑scale)
        if step % 300 == 0 and beta < beta_max:
            beta = min(beta * 1.12, beta_max)

        step += 1

    return a


# ----------------------------------------------------------------------
# Random initialisers (diverse families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Generate a random non‑negative sequence on the ℓ¹‑simplex
    Σ a_i = √(2 n).  Several families are mixed to increase diversity.
    """
    target_sum = np.sqrt(2.0 * n)
    kind = rng.integers(0, 7)
    idx = np.arange(n, dtype=float)

    if kind == 0:                     # decreasing power‑law
        p = rng.uniform(0.15, 1.0)
        a = (idx + 1.0) ** (-p)

    elif kind == 1:                   # exponential decay from the left edge
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif kind == 2:                   # mixed power‑law + exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif kind == 3:                   # Dirichlet (uniform simplex)
        a = rng.dirichlet(alpha=np.ones(n))

    elif kind == 4:                   # symmetric edge‑heavy
        p = rng.uniform(0.2, 1.2)
        left = (idx + 1.0) ** (-p)
        right = (n - idx) ** (-p)
        a = left + right

    elif kind == 5:                   # Beta‑shaped (U‑ or bell‑shaped)
        alpha = rng.uniform(0.2, 2.0)
        beta = rng.uniform(0.2, 2.0)
        x = (idx + 0.5) / n
        a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))

    else:                             # alternating high / low pattern
        a = np.zeros(n)
        evens = np.arange(0, n, 2)
        odds = np.arange(1, n, 2)
        a[evens] = rng.random(evens.size) + 0.2   # relatively large
        a[odds] = rng.random(odds.size) * 0.01   # tiny values

    # small Gaussian perturbation and projection onto the simplex
    a += rng.normal(scale=target_sum * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target_sum)
    return a


# ----------------------------------------------------------------------
# Targeted greedy reduction of the maximal convolution entry
# ----------------------------------------------------------------------


def _greedy_max_reduction(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Repeatedly move mass from a source index that heavily contributes to the
    current maximal convolution entry to a destination that does *not* affect
    this entry.  The move is kept only if the maximal convolution does not
    increase.  This yields a monotone non‑increasing sequence of objectives.
    """
    a = a0.copy()
    n = a.size
    rng = np.random.default_rng()
    best_score = _max_convolution_fft(a)
    best_a = a.copy()

    # fractions of the source mass we are willing to move – try large first,
    # fall back to smaller steps if the large step is not accepted.
    fractions = [0.5, 0.25, 0.1, 0.05, 0.02]

    while time.time() < deadline:
        # full convolution and index of the current maximum
        b = _full_convolution_fft(a)
        k = int(np.argmax(b))

        # indices that actually appear in b[k]
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        inds = np.arange(lo, hi + 1)

        # contributions a_i * a_{k-i}
        contrib = a[inds] * a[k - inds]

        # source index: the one with maximal contribution
        if contrib.size == 0 or np.all(contrib == 0):
            i = rng.integers(0, n)          # fallback – any index
        else:
            i = inds[np.argmax(contrib)]

        # destination index: choose a place that does NOT affect b[k]
        if k < n - 1:
            # any index > k yields k‑j < 0  → zero contribution
            dest_candidates = np.arange(k + 1, n)
        else:
            # k ≥ n‑1, indices < k‑(n‑1) give k‑j > n‑1 → zero contribution
            low = k - (n - 1)
            dest_candidates = np.arange(0, low)

        if dest_candidates.size == 0:
            # no “zero‑contribution” index → pick a random different one
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)
        else:
            j = rng.choice(dest_candidates)

        improved = False
        for frac in fractions:
            if a[i] <= 0.0:
                break
            delta = a[i] * frac
            a[i] -= delta
            a[j] += delta

            cur_score = _max_convolution_fft(a)

            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                improved = True
                print(f"[{time.time():.1f}] greedy improvement: {best_score:.6f}")
                break   # accept this move
            else:
                # revert and try a smaller fraction
                a[j] -= delta
                a[i] += delta

        if not improved:
            # occasional global jitter to escape tiny plateaus
            if rng.random() < 0.001:
                jitter = rng.normal(scale=0.001 * best_a.sum(), size=n)
                a = np.maximum(a + jitter, 0.0)
                # re‑normalise the ℓ¹‑norm (the sum is invariant under moves,
                # but jitter may disturb it)
                s = a.sum()
                if s > 0:
                    a = a * (np.sqrt(2.0 * n) / s)
                cur_score = _max_convolution_fft(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] jitter improvement: {best_score:.6f}")

    return best_a

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)
        end_phase = min(deadline, time.time() + scan_per_len)

        while time.time() < end_phase:
            a = _random_initial_sequence(n, rng)
            score = _max_convolution_fft(a)
            if score < best_score:
                best_seq, best_score = a.copy(), score
                print(f"[{time.time():.1f}] new best from scan n={n}  max‑conv={score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Fallback – uniform if nothing was generated
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 3000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = _max_convolution_fft(best_seq)

    # ------------------------------------------------------------------
    # 4️⃣  Mirror‑descent refinement (half of the remaining budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        md_deadline = time.time() + remaining * 0.5
        a_md = _mirror_descent_opt(best_seq, md_deadline)
        md_score = _max_convolution_fft(a_md)
        if md_score < best_score:
            best_seq, best_score = a_md.copy(), md_score
            print(f"[{time.time():.1f}] after mirror descent max‑conv={md_score:.6f}")

        # ------------------------------------------------------------------
        # 5️⃣  Targeted greedy reduction (use whatever time is left)
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 5.0:
            gd_deadline = time.time() + remaining
            a_gd = _greedy_max_reduction(best_seq.copy(), gd_deadline)
            gd_score = _max_convolution_fft(a_gd)
            if gd_score < best_score:
                best_seq, best_score = a_gd.copy(), gd_score
                print(f"[{time.time():.1f}] after greedy reduction max‑conv={gd_score:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Trim negligible trailing entries and return a plain list
    # ------------------------------------------------------------------
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # ------------------------------------------------------------------
    # Optional sanity check with the official evaluator
    # ------------------------------------------------------------------
    try:
        ev = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence = {ev:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] finished – best max‑conv = {best_score:.6f} "
          f"length = {best_seq.size}")

    return best_seq.tolist()
```
