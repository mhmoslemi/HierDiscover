"""
Optimized step‑function search with multi‑scale restarts and adaptive refinement.

The algorithm works as follows:

1. The total budget (≈1000 s) is divided among several sequence lengths.
   Larger lengths provide more degrees of freedom, while short lengths are
   cheap to explore.

2. For each length `n` we build a small pool of initial candidates:
   * an interpolated copy of the best sequence already found for a shorter
     length (if any);
   * a few random vectors drawn from a variety of distributions
     (power‑law, exponential, Dirichlet, piece‑wise, etc.).

3. Each candidate undergoes three refinement phases, each bounded by a
   sub‑deadline:
   * **smooth‑max descent** – projected gradient descent on a log‑sum‑exp
     surrogate of the ∞‑norm of the self‑convolution.
   * **exact max‑index sub‑gradient descent** – classic sub‑gradient on the
     true peak value.
   * **mass‑balance greedy move** – repeatedly transfers a modest fraction
     of the most offending coefficient to a light location, always staying
     on the ℓ¹‑simplex.

   After every phase we evaluate the current peak (`_score`) and keep the
   best local solution.

4. The globally best sequence across all lengths is stored.  When the main
   loop finishes, any remaining time is spent on a light simulated‑annealing
   polish.

All iterates are projected onto the simplex
    { a_i ≥ 0, Σ a_i = √(2 n) }
so that the official metric returned by `evaluate_sequence` is exactly the
maximum entry of the self‑convolution.  Progress is printed to stdout.

The routine respects the 1000 s wall‑clock limit and returns a plain
Python list of non‑negative heights.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1, dtype=float)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·n−1) using an FFT.
    Returned array is real‑valued.
    """
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Objective after ℓ¹‑normalisation Σa = √(2 n):
    simply the supremum norm of the self‑convolution.
    """
    return float(_max_convolution_fft(a).max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) gradient – Phase 1
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) where
        g = ∇_a  ( 1/β·log Σ exp(β·b) )
          = 2·(a ⋆ p)          with   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # ---- b = a*a -------------------------------------------------------
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ---- soft‑max probabilities p --------------------------------------
    max_b = float(np.max(b))
    exps = np.exp(beta * (b - max_b))          # shift for numerical stability
    p = exps / np.sum(exps)

    # ---- gradient g = 2·(a ⋆ p) ----------------------------------------
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1            # = 3 n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _smooth_max_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Annealed projected gradient descent on the smooth max.
    Uses momentum, an adaptive step and occasional Gaussian jitter.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 4096.0
    beta_factor = 1.4

    step_base = target_sum * 0.15          # fraction of the ℓ¹‑norm
    momentum = np.zeros_like(a)
    rho = 0.90                               # momentum decay

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        # simplex projection (keeps Σa = target_sum)
        a = _project_to_simplex(a, target_sum)

        # a little jitter every few hundred iterations
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal β
        if it % 350 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a


# ----------------------------------------------------------------------
# Exact max‑index sub‑gradient – Phase 2
# ----------------------------------------------------------------------


def _max_index_subgradient(a: np.ndarray):
    """
    Sub‑gradient of max(b) where b = a*a.
    Returns (g, k) where k is the index attaining the maximum.
    """
    n = a.shape[0]
    conv = _max_convolution_fft(a)
    k = int(np.argmax(conv))

    g = np.zeros_like(a)
    if k < n:                         # left half of the convolution
        g[:k + 1] = 2.0 * a[k::-1]
    else:                             # right half
        d = k - (n - 1)               # shift amount (0 … n‑1)
        g[d:] = 2.0 * a[: n - d]
    return g, k


def _subgradient_descent_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected sub‑gradient descent on the true max‑norm (no smoothing).
    A diminishing step‑size  η_t = α / √t is used.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        g, _ = _max_index_subgradient(a)

        # diminishing step size
        step = 0.8 * target_sum / (np.sqrt(it + 1) * np.linalg.norm(g) + 1e-12)
        a -= step * g

        # enforce feasibility
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        # occasional tiny jitter to avoid cycling
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1
    return a

# Greedy mass‑balance refinement – Phase 3
# ----------------------------------------------------------------------


def _mass_balance_refine(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Fast greedy move that explicitly reduces the contribution of the
    currently maximal convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sum(a)               # should already be √(2 n)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _max_convolution_fft(a)
        k = int(np.argmax(conv))

        # contribution of each index to that entry: a_i * a_{k‑i}
        idx = np.arange(n)
        other = k - idx
        mask = (other >= 0) & (other < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[other[mask]]

        i_high = int(np.argmax(contrib))

        # pick a relatively empty position – among the 20 % smallest masses
        low_candidates = np.where(a <= np.percentile(a, 20))[0]
        if low_candidates.size == 0:
            i_low = int(np.argmin(a))
        else:
            i_low = int(rng.choice(low_candidates))

        # move a modest fraction (3 %) of the heavy mass
        delta = a[i_high] * 0.03
        if delta < 1e-12:
            break

        a[i_high] -= delta
        a[i_low]  += delta

        # keep feasibility (non‑negativity + ℓ¹‑norm)
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        if it % 100 == 0:
            # occasional global jitter
            if rng.random() < 0.02:
                jitter = rng.normal(scale=target_sum * 1e-3, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                a = _project_to_simplex(a, target_sum)

        it += 1

    return a


# ----------------------------------------------------------------------
# Random initialisation helpers
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative point on the simplex Σa = √(2 n).
    Several families (power‑law, exponential, Dirichlet, piecewise) are
    sampled to increase diversity.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                                 # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                               # pure exponential
        lam = rng.exponential(scale=0.02)
        a = np.exp(-lam * idx)

    elif family == 2:                               # power × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.02)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                               # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n))

    elif family == 4:                               # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                             np.exp(-lam2 * np.arange(b, n))])

    else:                                           # almost uniform with jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _interpolate_sequence(seq: np.ndarray, n: int) -> np.ndarray:
    """
    Resample `seq` linearly to a new length `n` and project onto the simplex.
    """
    if seq.size == n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, seq.size)
    x_new = np.linspace(0.0, 1.0, n)
    a = np.interp(x_new, x_old, seq)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, np.sqrt(2.0 * n))
    return a


# ----------------------------------------------------------------------
# Light simulated‑annealing polishing (kept from reference)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Very light SA that moves a random fraction of mass between two sites."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    cur_score = _score(a)
    best_score = cur_score

    T = cur_score * 0.02 + 1e-9
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-16:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        frac = rng.random() * 0.15
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta

        new_score = _score(a)
        dE = new_score - cur_score
        accept = dE <= 0.0 or (T > 0.0 and rng.random() < np.exp(-dE / T))

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling

    return best


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` metric.  The routine respects the wall‑clock
    budget, explores a range of lengths, applies three refinement phases,
    and finally runs a short simulated‑annealing polish.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Global best tracking
    # ------------------------------------------------------------------
    global_best_seq = None
    global_best_score = np.inf

    # ------------------------------------------------------------------
    # Load a seed sequence if the environment already provides one
    # ------------------------------------------------------------------
    earlier = globals().get("height_sequence_1")
    if isinstance(earlier, (list, tuple, np.ndarray)):
        seq0 = np.asarray(earlier, dtype=float)
        n0 = seq0.size
        seq0 = _project_to_simplex(seq0, np.sqrt(2.0 * n0))
        sc0 = _score(seq0)
        global_best_seq = seq0.copy()
        global_best_score = sc0
        print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) score={sc0:.6f}")

    # ------------------------------------------------------------------
    # Lengths to investigate – larger n gives more freedom
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 4000, 5000, 6000, 8000, 10000]

    # Iterate over lengths, allocating a fair share of the remaining budget
    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        time_left = deadline - time.time()
        lengths_remaining = len(lengths) - idx
        # Give each upcoming length roughly equal remaining share
        per_length_budget = max(5.0, time_left / lengths_remaining)

        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Build a small pool of starting vectors for this length
        # --------------------------------------------------------------
        init_vectors = []

        # 1) interpolated version of the current global best (if any)
        if global_best_seq is not None:
            init_vectors.append(_interpolate_sequence(global_best_seq, n))

        # 2) a few random starts (we keep it small to stay within time)
        for _ in range(2):
            init_vectors.append(_random_initial_sequence(n, rng))

        best_local_seq = None
        best_local_score = np.inf

        # --------------------------------------------------------------
        # Process each starter
        # --------------------------------------------------------------
        for a_start in init_vectors:
            if time.time() >= deadline:
                break

            # ----- Phase 1 : smooth‑max descent -----
            # Give roughly half of the per‑length budget to the smooth phase
            phase1_dead = min(deadline, time.time() + 0.5 * per_length_budget)
            a = _smooth_max_descent(a_start, phase1_dead)

            # ----- Phase 2 : exact sub‑gradient descent -----
            if time.time() < deadline:
                phase2_dead = min(deadline, time.time() + 0.3 * per_length_budget)
                a = _subgradient_descent_max(a, phase2_dead)

            # ----- Phase 3 : greedy mass‑balance -----
            if time.time() < deadline:
                phase3_dead = min(deadline, time.time() + 0.2 * per_length_budget)
                a = _mass_balance_refine(a, phase3_dead)

            # ----- Evaluate -----
            sc = _score(a)
            if sc < best_local_score:
                best_local_score = sc
                best_local_seq = a.copy()
                print(f"[{time.time():.1f}] Length {n} new local best = {sc:.6f}")

        # --------------------------------------------------------------
        # Update global best if this length produced improvement
        # --------------------------------------------------------------
        if best_local_seq is not None and best_local_score < global_best_score:
            global_best_score = best_local_score
            global_best_seq = best_local_seq.copy()
            print(f"[{time.time():.1f}] *** Global best updated: n={n} score={global_best_score:.6f}")

        # Reduce the per‑length budget for the next iteration (already
        # accounted for by the elapsed time checks).

    # ------------------------------------------------------------------
    # Final simulated‑annealing polishing with whatever time is left
    # ------------------------------------------------------------------
    if global_best_seq is not None:
        time_left = deadline - time.time()
        if time_left > 5.0:
            a_final = _simulated_annealing(global_best_seq, deadline)
            sc = _score(a_final)
            if sc < global_best_score:
                global_best_score = sc
                global_best_seq = a_final.copy()
                print(f"[{time.time():.1f}] Final SA improvement: score={sc:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the unlikely case we never produced a sequence
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 2000
        global_best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)
        global_best_score = _score(global_best_seq)

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries and report the official value
    # ------------------------------------------------------------------
    nz = np.where(global_best_seq > 1e-12)[0]
    if nz.size:
        global_best_seq = global_best_seq[: nz[-1] + 1]

    try:
        val = evaluate_sequence(global_best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={global_best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return global_best_seq.tolist()
