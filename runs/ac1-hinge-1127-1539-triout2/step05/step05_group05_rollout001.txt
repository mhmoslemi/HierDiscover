"""
Hybrid exploratory + deterministic optimisation for the non‑negative step‑function
problem.

The algorithm runs under a fixed time budget (default 1000 s) and consists of
three phases:

1. **Broad random exploration** – For a set of candidate lengths we draw
   sequences from a rich collection of parametric families (power‑law,
   exponential, Dirichlet, beta, symmetric edge‑heavy, Gaussian, triangular,
   …).  Each sample is projected onto the ℓ¹‑simplex Σa = √(2 n) and its
   maximal self‑convolution entry is measured with an FFT.  The best sample
   seen so far is kept.

2. **Sub‑gradient descent** – Starting from the best sample we minimise the
   exact ℓ∞‑norm ‖a∗a‖∞ by a simple sub‑gradient method:
   g = 2·a reversed and shifted to the index where the convolution attains its
   maximum.  After each step we project back onto the simplex.

3. **Targeted simulated annealing** – A fine‑grained SA that repeatedly moves
   a random fraction of mass from an index contributing heavily to the current
   worst convolution entry to a random destination.  The temperature is
   annealed and occasional global jitter helps escape local minima.

All heavy work (FFT convolution, simplex projection, gradient computation)
is written with NumPy for speed.  The routine finally returns a plain Python
list of non‑negative floats that can be fed to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean, Duchi et al. 2008)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x_i≥0, Σx_i = s}."""
    n = v.shape[0]
    if s <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random families for the broad exploration phase
# ----------------------------------------------------------------------


def _random_family_sequence(n: int, kind: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative sequence of length n belonging to a chosen family.
    The result is projected onto the ℓ¹‑simplex Σa = √(2 n).
    """
    target_sum = np.sqrt(2.0 * n)
    idx = np.arange(n, dtype=float)

    if kind == 0:  # pure power‑law (decreasing)
        p = rng.uniform(0.05, 2.0)
        a = (idx + 1.0) ** (-p)

    elif kind == 1:  # exponential decay
        lam = rng.uniform(0.0, 0.04)
        a = np.exp(-lam * idx)

    elif kind == 2:  # power‑law * exponential
        p = rng.uniform(0.05, 2.0)
        lam = rng.uniform(0.0, 0.04)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif kind == 3:  # Dirichlet (uniform simplex)
        a = rng.dirichlet(alpha=np.ones(n))

    elif kind == 4:  # symmetric edge‑heavy
        p = rng.uniform(0.2, 1.5)
        left = (idx + 1.0) ** (-p)
        right = (n - idx) ** (-p)
        a = left + right

    elif kind == 5:  # Beta‑shaped
        alpha = rng.uniform(0.2, 3.0)
        beta = rng.uniform(0.2, 3.0)
        x = (idx + 0.5) / n
        a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))

    elif kind == 6:  # Gaussian bell
        sigma = rng.uniform(0.1, 0.5) * n
        centre = (n - 1) / 2.0
        a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)

    elif kind == 7:  # Triangular (optional exponent)
        exponent = rng.uniform(0.5, 3.0)
        centre = (n - 1) / 2.0
        a = 1.0 - np.abs(idx - centre) / centre
        a = np.maximum(a, 0.0) ** exponent

    else:  # fallback – small random perturbation of uniform
        a = np.full(n, 1.0)

    # small Gaussian jitter (helps break symmetry)
    a += rng.normal(scale=target_sum * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target_sum)
    return a


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact ℓ∞‑norm
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simple sub‑gradient descent for f(a)=‖a∗a‖∞.
    Each iteration moves opposite to the sub‑gradient at the current maximal
    convolution entry and then projects onto the ℓ¹‑simplex.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    step = 0
    while time.time() < deadline:
        # full convolution and index of max entry
        b = _full_convolution_fft(a)
        k_max = int(np.argmax(b))

        # sub‑gradient: g_i = 2 * a_{k_max-i} if the index exists, else 0
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idx = np.arange(lo, hi + 1, dtype=int)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        # step size schedule (decreasing)
        eta = 0.5 / np.sqrt(step + 1.0)
        a -= eta * g
        a = np.maximum(a, 0.0)

        # project back to simplex
        s = np.sum(a)
        if s > 0:
            a *= (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        step += 1
    return a


# ----------------------------------------------------------------------
# Targeted simulated annealing (refinement)
# ----------------------------------------------------------------------


def _targeted_simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Refine a candidate by moving mass from indices that contribute heavily to
    the current worst convolution entry to a random destination.  Temperature
    decays multiplicatively; occasional global jitter helps to escape local
    minima.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # initial temperature proportional to the current objective
    T = best_score * 0.03 + 1e-9
    cooling = 0.9997
    rng = np.random.default_rng()
    max_transfer = 0.30  # up to 30 % of the selected mass

    while time.time() < deadline:
        # current convolution and its maximal index
        b = _full_convolution_fft(a)
        k_max = int(np.argmax(b))

        # indices that affect this entry
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idx_range = np.arange(lo, hi + 1, dtype=int)
        contrib = a[idx_range] * a[k_max - idx_range]
        weight_sum = float(np.sum(contrib))

        if weight_sum > 0.0:
            probs = contrib / weight_sum
            i = rng.choice(idx_range, p=probs)
        else:
            i = rng.integers(0, n)

        # choose a destination different from i
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        # transfer a random fraction of mass at i to j
        delta = a[i] * rng.random() * max_transfer
        if delta <= 0.0:
            continue
        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score

        # Metropolis acceptance
        accept = False
        if cur_score <= best_score:
            accept = True
        elif rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA improvement: {best_score:.6f}")
        else:
            # revert move
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            s = np.sum(a)
            if s > 0:
                a *= (np.sqrt(2.0 * n) / s)
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA jitter improvement: {best_score:.6f}")

    return best

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The procedure respects the supplied time budget (default 1000 s) and
    returns the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # If a handcrafted construction is present, start from it.
    # ------------------------------------------------------------------
    global_seq = globals().get("height_sequence_1")
    if isinstance(global_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(global_seq, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        sc = _max_convolution_fft(a0)
        best_seq, best_score = a0.copy(), sc
        print(f"[{time.time():.1f}] Seeded from global height_sequence_1 (n={n0}) "
              f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – broad random exploration (≈30 % of the budget)
    # ------------------------------------------------------------------
    explore_budget = (budget_s * 0.30)
    explore_deadline = start_time + explore_budget

    # candidate lengths – we try a mixture of short and long sequences
    length_pool = [500, 800, 1000, 1200, 1500, 1800,
                   2000, 2500, 3000, 3500, 4000, 5000, 6000, 8000, 10000]

    n_families = 8  # we defined families 0..7
    while time.time() < explore_deadline:
        n = int(rng.choice(length_pool))
        kind = int(rng.integers(0, n_families))
        cand = _random_family_sequence(n, kind, rng)
        score = _max_convolution_fft(cand)
        if score < best_score:
            best_seq, best_score = cand.copy(), score
            print(f"[{time.time():.1f}] New best (explore) n={n} kind={kind} "
                  f"score={score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – sub‑gradient descent (≈25 % of the budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if best_seq is not None and remaining > 5.0:
        sub_deadline = time.time() + remaining * 0.45
        a_sub = _subgradient_descent(best_seq, sub_deadline)
        sub_score = _max_convolution_fft(a_sub)
        if sub_score < best_score:
            best_seq, best_score = a_sub.copy(), sub_score
            print(f"[{time.time():.1f}] New best after sub‑gradient: "
                  f"score={sub_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – targeted simulated annealing (use whatever time remains)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a_sa = _targeted_simulated_annealing(best_seq, deadline)
        sa_score = _max_convolution_fft(a_sa)
        if sa_score < best_score:
            best_seq, best_score = a_sa.copy(), sa_score
            print(f"[{time.time():.1f}] New best after SA: score={sa_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up tiny trailing entries
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback – uniform of modest length
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = _max_convolution_fft(best_seq)

    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # ------------------------------------------------------------------
    # Optional final sanity check via the official evaluator
    # ------------------------------------------------------------------
    try:
        ev = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence = {ev:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective (max‑conv) = "
          f"{best_score:.6f}   length = {best_seq.size}")

    return best_seq.tolist()
