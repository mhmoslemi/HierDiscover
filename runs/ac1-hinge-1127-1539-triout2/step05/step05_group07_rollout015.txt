def propose_candidate(seed=42, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0
    base_seq = None
    if 'height_sequence_1' in globals():
        base_seq = np.array(globals()['height_sequence_1'], float)

    best_seq = None
    best_score = np.inf
    attempt = 0
    while time.time() < deadline:
        remaining = deadline - time.time()
        # choose size
        if rng.random() < 0.5:
            n = rng.integers(2000, 8000)
        else:
            n = rng.integers(8000, 20000)
        # allocate slice
        slice_time = min(remaining * 0.9, 25.0)  # max 25 seconds per restart
        slice_deadline = time.time() + slice_time
        a0 = _make_initial_vector(n, rng, base_seq)
        a_opt, peak = _adam_optimize(a0, slice_deadline, rng)
        score = peak  # since sum = sqrt(2n)
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[restart] attempt={attempt} n={n} new best score={best_score:.6f}")
        attempt += 1
    if best_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()
        external = evaluate_sequence(final_seq)
        print(f"[final] external score = {external:.6f} internal best = {best_score:.6f} length = {len(final_seq)}")
        globals()['height_sequence_1'] = final_seq
    return final_seq

"""
adversarial step‑function optimizer – version 2

* Uses the same log‑softmax parametrisation as the reference
  implementation but with a more aggressive cooling schedule.
* Supports a richer set of random initial shapes (including an
  “inverse‑sqrt” shape that roughly mimics the continuous arcsine
  density, known to flatten self‑convolution).
* Allocates a larger time slice per restart (≈20 s) and performs a
  final refinement on the best candidate found.
* All helper functions are defined at module level (no closures).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution of two real vectors using a real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    n = a.shape[0]
    m = b.shape[0]
    total_len = n + m - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    B = np.fft.rfft(b, n=f_len)
    conv = np.fft.irfft(A * B, n=f_len)[:total_len]
    return conv


def _self_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for the self‑convolution a*a."""
    return _full_convolution_fft(a, a)


def _grad_wrt_a(a: np.ndarray, mask: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k mask_k·(a*a)_k  w.r.t. a.
    Implements   g = 2·conv(mask, a_rev)   and extracts the
    central n entries.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    conv_len = mask.shape[0] + n - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[n - 1 : n - 1 + n]
    return g


# ----------------------------------------------------------------------
# Soft‑max (smooth max) utilities
# ----------------------------------------------------------------------
def _softmax_weights(b: np.ndarray, tau: float) -> np.ndarray:
    """Return soft‑max weights over the entries of b (shape 2n‑1)."""
    if tau < 1e-12:
        # exact sub‑gradient: only the true maxima get weight 1
        mask = (b == b.max()).astype(float)
        return mask
    # classic log‑sum‑exp trick for numerical stability
    max_b = b.max()
    diff = (b - max_b) / tau
    exps = np.exp(diff)
    w = exps / exps.sum()
    return w


def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Returns (grad, max_b) where
        grad = ∂/∂a  τ‑soft‑max( a*a )
        max_b = hard maximum of the self‑convolution.
    """
    b = _self_convolution(a)
    max_b = b.max()
    w = _softmax_weights(b, tau)
    grad = _grad_wrt_a(a, w)
    return grad, max_b


# ----------------------------------------------------------------------
# Random initialisation helpers
# ----------------------------------------------------------------------
def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a 1‑D sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.ceil(pos).astype(int)
    hi = np.clip(hi, 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


def _make_initial_vector(n: int, rng: np.random.Generator,
                         base_seq: np.ndarray | None = None) -> np.ndarray:
    """
    Produce a feasible start vector of length n with ℓ₁‑norm sqrt(2·n).
    A few different shapes are tried; with probability 0.2 the vector is
    a scaled version of the current best (if supplied).
    """
    S = np.sqrt(2.0 * n)                     # target ℓ₁‑norm

    shape = rng.choice(['uniform', 'edge', 'inverse_sqrt', 'random'])
    if shape == 'uniform':
        a = np.full(n, S / n)
    elif shape == 'edge':
        # mass heavy near the boundaries, controlled by a random exponent
        p_exp = rng.uniform(0.5, 2.5)
        idx = np.arange(n)
        d = np.minimum(idx, n - 1 - idx) + 1.0
        a = d.astype(float) ** (-p_exp)
        a = a / a.sum() * S
    elif shape == 'inverse_sqrt':
        # discrete analogue of the arcsine density: 1/√(i·(n‑1‑i))
        i = np.arange(n, dtype=float)
        a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5) + 1e-12)
        a = a / a.sum() * S
    else:   # random Dirichlet‑like start
        a = rng.random(n)
        a = a / a.sum() * S

    # Occasionally blend with the current best known construction
    if base_seq is not None and rng.random() < 0.2:
        a = _scale_sequence(base_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S

    # Guard against tiny negative rounding artefacts
    a = np.clip(a, 0.0, None)
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Minimise the (soft) maximum of the self‑convolution.
    Returns the best vector encountered (with ℓ₁‑norm sqrt(2·n))
    and the corresponding hard peak value.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)           # fixed ℓ₁‑norm

    # initialise logits so that softmax(p)*S == a0
    eps = 1e-12
    init_soft = np.maximum(a0 / S, eps)
    init_soft = init_soft / init_soft.sum()
    p = np.log(init_soft)          # unrestricted logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.35                       # a bit more aggressive than the reference

    # smoothing schedule – start relatively large, cool quickly
    b0 = _self_convolution(a0)
    tau = max(b0.max(), 1.0)
    tau_decay = 0.995               # faster cooling
    min_tau = 1e-8

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ------- reconstruct a from logits -------
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)               # improve numerical stability
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ------- convolution & (soft) max -------
        grad_a, peak = _softmax_gradient(a, tau)

        # keep the best *hard* peak we have seen
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ------- gradient w.r.t. logits -------
        mu = np.dot(a, grad_a) / S               # scalar term from softmax Jacobian
        grad_p = a * (grad_a - mu)               # shape (n,)

        # optional clipping to keep steps stable for large n
        clip_thr = 5.0 * S
        gnorm = np.max(np.abs(grad_p))
        if gnorm > clip_thr:
            grad_p = grad_p * (clip_thr / gnorm)

        # ------- Adam update -------
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ------- schedules -------
        tau = max(tau * tau_decay, min_tau)
        lr *= 0.9995

        # occasional jitter to escape plateaus
        if it % 400 == 0:
            p += rng.normal(scale=0.015, size=n)

    return best_a, best_peak
