"""
Improved step‑function optimiser.

The goal is to minimise the maximum value of the self‑convolution
(a*a)[k] under the constraint Σ a_i = √(2·n) and a_i ≥ 0.
The external evaluator returns exactly this maximum.

The optimiser works in two phases.

1. **Exploratory shape search** – a fast grid search over several
   analytically‑motivated families (uniform, geometric decay,
   edge‑heavy reciprocal, mixtures and logistic shapes).  For each
   randomly chosen length n a candidate is generated, scaled onto the
   required ℓ¹‑simplex and scored via a single FFT‑based convolution.
   The best‑scoring raw shape is kept.

2. **Refinement** – starting from the best shape a longer Adam → sub‑gradient
   optimisation (the same building blocks as the reference code) is run.
   The optimiser repeatedly projects onto the simplex, sorts the vector
   (to keep a roughly symmetric “U‑shape”) and uses a smooth log‑sum‑exp
   surrogate for the max‑norm.  The remaining budget is split into short
   optimisation slices; after each slice the best solution is retained.
   Occasionally the current best is up‑sampled to a larger size to give
   the optimiser more degrees of freedom.

The routine respects the passed time budget (default 1000 s) and finally
returns the best sequence found as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean L2)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full convolution a*a (length 2·n‑1)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Gradient of the smooth max (log‑sum‑exp) w.r.t. a
# ----------------------------------------------------------------------
def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights (smooth approximation of max)
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Shape families for the fast exploratory phase
# ----------------------------------------------------------------------
def _shape_uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=float)


def _shape_edge_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) + (n-i)^(-p)  (symmetric, edge‑heavy)."""
    idx = np.arange(n, dtype=float)
    a = 1.0 / (idx + 1.0) ** p + 1.0 / (n - idx) ** p
    return a


def _shape_geometric(n: int, rate: float) -> np.ndarray:
    """Symmetric geometric decay a_i ∝ rate**i + rate**(n‑1‑i)."""
    left = rate ** np.arange(n, dtype=float)
    right = left[::-1]
    return left + right


def _shape_logistic(n: int, sigma: float) -> np.ndarray:
    """Monotone logistic decay, symmetric via mirroring."""
    x = np.arange(n, dtype=float) - (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(x / sigma))
    return a


def _shape_mix_edge_uniform(n: int, p: float, w: float) -> np.ndarray:
    """Convex blend of uniform and edge‑heavy reciprocal shape."""
    return w * _shape_uniform(n) + (1.0 - w) * _shape_edge_power(n, p)


# ----------------------------------------------------------------------
# Adam optimiser on the smooth log‑sum‑exp max‑approximation
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected Adam minimising the smoothed max‑norm."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    lr = 0.2
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    alpha = 0.5
    alpha_mul = 1.015

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        cur_max = np.max(conv)

        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Euclidean projection onto the simplex and enforce a symmetric decreasing order
        a = _project_simplex(a, S)
        a = np.sort(a)[::-1]                     # keep a roughly “U‑shaped”

        # occasional mass‑swap move
        if rng.random() < 0.02:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.04 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand = np.sort(cand)[::-1]
                cand_max = np.max(_full_convolution(cand))
                if cand_max < best_score or rng.random() < 0.1:
                    a = cand

        alpha = min(alpha * alpha_mul, 30.0)
        lr *= 0.9999

        # a modest stagnation guard (no premature stop)
        if it % 20000 == 0 and best_score == cur_max:
            # keep running – the outer time check will stop us.
            pass

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient phase (focus on the true max‑norm)
# ----------------------------------------------------------------------
def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the true max‑norm (only the worst shift)."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.12
    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)
        a = np.sort(a)[::-1]

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        if rng.random() < 0.04:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand = np.sort(cand)[::-1]
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.2:
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()

        if it % 10000 == 0 and best_score == cur_max:
            # keep iterating – outer loop decides termination.
            pass

    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimiser (Adam → sub‑gradient)
# ----------------------------------------------------------------------
def _optimize_combined(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Run a smooth Adam phase for ~60 % of the remaining budget,
    then a sub‑gradient refinement for the rest."""
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    smooth_deadline = now + 0.6 * remaining
    a_s, score_s = _optimize_adam(a0, smooth_deadline, rng)

    if time.time() < deadline:
        a_f, score_f = _optimize_subgrad(a_s, deadline, rng)
        if score_f < score_s:
            return a_f, score_f
    return a_s, score_s


# ----------------------------------------------------------------------
# Upsampling helper (linear interpolation)
# ----------------------------------------------------------------------
def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Fast evaluation of a candidate (max of self‑convolution)
# ----------------------------------------------------------------------
def _eval_max_conv(seq: np.ndarray) -> float:
    """Return the objective value (max convolution) for a given sequence."""
    return float(np.max(_full_convolution(seq)))
