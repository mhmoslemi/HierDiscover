"""
Propose a step‑function (non‑negative vector) whose self‑convolution peak
is as low as possible.  The search is performed in three phases:

1. **Randomised initialisation** – a rich family of analytic shapes
   (uniform, geometric decay, edge‑heavy reciprocal, triangular, and
   up‑sampled previous best) is sampled and projected onto the simplex
   Σa = √(2·n).

2. **Smooth Adam optimisation** – the peak of the convolution is
   approximated by a log‑sum‑exp (soft‑max).  An Adam optimiser with
   Euclidean simplex projection (and an optional symmetry projection)
   drives the sequence towards a lower peak.  The temperature of the
   soft‑max is increased gradually.

3. **Sub‑gradient refinement + local random mass‑swap search** – the true
   max‑norm is tackled directly by a sub‑gradient step, followed by a
   short simulated‑annealing‑style local search that repeatedly swaps
   small amounts of mass between indices.  The best candidate across all
   restarts is finally returned.

The whole routine is time‑aware: it repeatedly restarts with different
lengths (≈2 k–12 k) while keeping the overall best solution found within
the allocated budget (default 1000 s).  The implementation avoids any
file I/O, uses only NumPy and the provided ``evaluate_sequence`` function,
and prints progress information for debugging/analysis.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x>=0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex with total mass s."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2*n-1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of (1/α)·log‑sum‑exp(α·(a*a)) w.r.t. a.
    w must be a probability vector (soft‑max weights).
    """
    n = a.shape[0]
    # convolution of a with reversed weights
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    # the gradient extract slice (see original implementation)
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights (log‑sum‑exp)
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Compute soft‑max probabilities over convolution entries."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Helper to generate a symmetric version of a vector (optional)
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry (v[i] = v[-1-i])."""
    return 0.5 * (v + v[::-1])


# ----------------------------------------------------------------------
# Initial sequence generators
# ----------------------------------------------------------------------
def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to new length n."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


def _initial_sequence(base_seq: np.ndarray | None,
                     n: int,
                     rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n with ℓ¹‑norm √(2·n).
    Several families are mixed to provide diversity.
    """
    S = np.sqrt(2.0 * n)

    typ = rng.integers(0, 6)   # 0–5 inclusive

    if typ == 0 and base_seq is not None and base_seq.size > 10:
        a = _upsample_seq(base_seq, n)
        a += rng.normal(scale=a.mean() * 1e-3, size=n)
        a = np.clip(a, 0.0, None)

    elif typ == 1:
        # uniform Dirichlet(1)
        a = rng.random(n)

    elif typ == 2:
        # geometric decay from both ends
        rate = rng.uniform(0.9, 0.9995)
        left = rate ** np.arange(n)
        right = left[::-1]
        a = (left + right) / 2.0

    elif typ == 3:
        # edge‑heavy reciprocal shape
        idx = np.arange(n, dtype=float)
        a = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.8)
        a = a ** power

    elif typ == 4:
        # triangular shape (peak in centre)
        centre = (n - 1) / 2.0
        a = 1.0 - np.abs(np.arange(n) - centre) / centre
        a = np.clip(a, 0.0, None)

    else:
        # random convex combination of a few bases
        cand = []

        cand.append(rng.random(n))                     # uniform
        rate = rng.uniform(0.9, 0.9995)
        left = rate ** np.arange(n)
        cand.append((left + left[::-1]) / 2.0)          # geometric
        idx = np.arange(n, dtype=float)
        rec = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.8)
        cand.append(rec ** power)                       # reciprocal

        w = rng.random(len(cand))
        w /= w.sum()
        a = sum(wi * ci for wi, ci in zip(w, cand))

    # Normalisation to the required ℓ¹‑norm
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0)
        a_sum = a.sum()
    a = a / a_sum * S
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smooth log‑sum‑exp approximation
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Projected Adam minimising the soft‑max approximation of the convolution peak.
    Returns (best_vector, best_peak_value).
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.4
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule for the soft‑max
    alpha = 0.5
    alpha_mul = 1.02

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # soft‑max weights and gradient wrt smooth objective
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # optional symmetry enforcement
        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional random jitter (helps escape flat regions)
        if rng.random() < 0.018:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.04 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < best_score or rng.random() < 0.12:
                    a = cand

        # temperature and learning‑rate schedules
        alpha = min(alpha * alpha_mul, 45.0)
        lr *= 0.99995

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true maximum convolution entry
# ----------------------------------------------------------------------
def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent focusing on the current worst shift, then a short
    random‑swap local search.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.14
    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient for the max entry (see analysis)
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        if enforce_sym:
            a = _symmetrize(a)
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # occasional random mass‑swap (simulated‑annealing style)
        if rng.random() < 0.045:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                # accept if improvement or with small probability
                if cand_max < cur_max or rng.random() < 0.22:
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()

        # early exit if stagnating for many iterations
        if it % 8000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Short local random‑swap search (fine‑tuning)
# ----------------------------------------------------------------------
def _local_swap_search(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """Very cheap hill‑climbing that swaps tiny masses between random positions."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.02 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = _symmetrize(cand)
        cand_score = np.max(_full_convolution(cand))
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand
            a = cand
    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimisation pipeline (Adam → sub‑gradient → local search)
# ----------------------------------------------------------------------
def _optimize_combined(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Runs the three optimisation phases within the supplied time budget.
    """
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    # split 60 % / 30 % / 10 %
    t_adam = now + 0.60 * remaining
    a_adam, sc_adam = _optimize_adam(a0, t_adam, rng, enforce_sym)

    t_sub = min(time.time() + 0.30 * remaining, deadline)
    a_sub, sc_sub = _optimize_subgrad(a_adam, t_sub, rng, enforce_sym)

    # final local refinement if time remains
    if time.time() < deadline:
        a_loc, sc_loc = _local_swap_search(a_sub, deadline, rng, enforce_sym)
    else:
        a_loc, sc_loc = a_sub, sc_sub

    # choose the best among the three phases
    candidates = [(a_adam, sc_adam), (a_sub, sc_sub), (a_loc, sc_loc)]
    best_seq, best_score = min(candidates, key=lambda kv: kv[1])
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point for the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function with a minimal self‑convolution peak.
    Returns the best sequence (list of floats) found before the time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Keep a small safety margin for the final external evaluation.
    deadline = start_time + budget_s - 5.0

    # Try to reuse a previously discovered construction if available.
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_overall_seq = None
    best_overall_score = np.inf
    best_overall_n = None
    attempts = 0
    last_improve_time = start_time

    # ------------------------------------------------------------------
    # Main timed loop – repeatedly restart with a fresh length and
    # initialisation, keeping the global best.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        attempts += 1

        # Choose problem size.
        # Occasionally respect the size of the current best and perturb it.
        if best_overall_seq is not None and rng.random() < 0.55:
            # perturb length of the current best by ±10 %
            scale = rng.uniform(0.9, 1.1)
            n = max(800, int(best_overall_n * scale))
        else:
            n = int(rng.integers(1800, 12000))

        # Randomly decide whether to enforce symmetry for this attempt.
        enforce_sym = rng.random() < 0.48

        # Generate a fresh initial vector.
        a0 = _initial_sequence(base_seq, n, rng)

        # Occasionally force exact symmetry on the start point (helps convergence).
        if enforce_sym:
            a0 = _symmetrize(a0)

        # Allocate a time slice (≈ 12–22 s) for this attempt, respecting the global deadline.
        slice_deadline = min(time.time() + rng.uniform(12.0, 22.0), deadline)

        # Run the combined optimiser.
        a_opt, score = _optimize_combined(a0, slice_deadline, rng, enforce_sym)

        # Update global best.
        if score < best_overall_score:
            best_overall_score = score
            best_overall_seq = a_opt.copy()
            best_overall_n = n
            last_improve_time = time.time()
            print(f"[global] attempt={attempts} n={n} sym={enforce_sym} "
                  f"new best={best_overall_score:.6f}")

        # If we have already passed the target we may stop early.
        if best_overall_score <= 1.503:
            break

        # If no improvement for a while, discard the current base shape to regain diversity.
        if time.time() - last_improve_time > 45.0:
            base_seq = None

    # ------------------------------------------------------------------
    # Final sanity check via the black‑box evaluator.
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_overall_seq.tolist()

    external_score = evaluate_sequence(final_seq)
    print(f"[final] external={external_score:.6f} "
          f"internal_best={best_overall_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    # expose the best sequence as a global variable for possible up‑sampling later
    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)

    return final_seq
