"""
Propose a non‑negative step function (list of heights) that minimizes the
official ``evaluate_sequence`` metric.

The optimiser works in several stages:

1.  **Initial seeds** – for each candidate length we try a few deterministic
    shapes (sine, semi‑circle, triangle), an up‑sampled version of the best
    sequence found so far, and a random Dirichlet‑type start.

2.  **Smooth‑max Adam descent** – we minimise a smooth approximation of the
    infinity norm
        φβ(a)= (1/β)·log Σexp(β·(a*a)_k)
    using the Adam optimiser and a projection onto the ℓ¹‑simplex
    Σa_i = √(2·n).  β is annealed (β←β·1.5) until a large value, making the
    smooth objective approach the true max.

3.  **Exact max‑sub‑gradient descent** – a classic diminishing‑step‑size
    projected sub‑gradient method on the true max‑norm.

4.  **Targeted mass‑balance** – the current maximal convolution entry is
    identified; mass is transferred from an index that contributes most to it
    to an index that contributes least, using a simple line‑search that aims
    to bring the peak down to the second‑largest entry.

5.  **Final simulated annealing** – a light SA polishing pass that swaps
    small fractions of mass between random positions.

All stages respect the feasibility constraints (non‑negativity and fixed
ℓ¹‑norm).  The loop respects a 1000 s wall‑clock budget; progress and
intermediate scores are printed.  The best sequence found is returned as a
plain Python list.
"""

import time
import numpy as np
from math import pi, sqrt

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1, dtype=float)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution (after ℓ¹‑normalisation)."""
    return float(_max_convolution_fft(a).max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) utilities
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of φβ(a) = (1/β)·log Σ exp(β·b_k)  where b = a*a.
    Returns (grad, b) .
    The gradient is 2·(a ⋆ p)  with  p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = float(np.max(b))
    exps = np.exp(beta * (b - max_b))            # shift for stability
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1               # = 3 n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _adam_softmax_descent(a0: np.ndarray, deadline: float,
                          beta_init: float = 0.5,
                          beta_max: float = 4096.0,
                          beta_factor: float = 1.5) -> np.ndarray:
    """
    Adam optimiser on the smooth‑max objective with projection onto the ℓ¹‑simplex.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = beta_init
    # base learning rate – a fraction of the ℓ¹‑norm
    base_lr = target_sum * 0.1

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        # anneal β every few hundred iterations
        if t % 250 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        # occasional jitter to avoid stagnation
        if t % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

    return a

def _max_index_subgradient(a: np.ndarray):
    """
    Sub‑gradient of max_k (a*a)_k.
    Returns (g, k*) where k* is an index attaining the maximum.
    """
    n = a.shape[0]
    conv = _max_convolution_fft(a)
    k = int(np.argmax(conv))

    g = np.zeros_like(a)
    if k < n:                         # left half of the convolution
        g[:k + 1] = 2.0 * a[k::-1]
    else:                             # right half
        d = k - (n - 1)               # shift amount (0 … n‑1)
        g[d:] = 2.0 * a[: n - d]
    return g, k


def _subgradient_descent_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Projected sub‑gradient descent on the true max‑norm."""
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        g, _ = _max_index_subgradient(a)

        # diminishing step size η_t = α / √t
        step = 0.9 * target_sum / (np.sqrt(it + 1) *
                                   np.linalg.norm(g) + 1e-12)
        a -= step * g

        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1
    return a

def _targeted_mass_balance(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Greedy reduction of the current maximum convolution entry.
    At each iteration a fraction of mass is moved from the index that
    contributes most to the peak to an index that contributes least,
    trying to bring the peak down to the second‑largest entry.
    """
    a = a0.copy()
    n = a.size
    S = np.sum(a)                     # should already equal √(2 n)
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _max_convolution_fft(a)
        k = int(np.argmax(conv))
        M = conv[k]

        # second‑largest entry (ignore the current peak)
        # Using a masked copy is cheap for moderate n.
        conv_masked = conv.copy()
        conv_masked[k] = -np.inf
        second = float(np.max(conv_masked))

        # compute partner values a_{k-i}
        idx = np.arange(n)
        other = k - idx
        valid = (other >= 0) & (other < n)
        partner = np.zeros_like(a)
        partner[valid] = a[other[valid]]

        # indices that most / least influence the peak
        i_high = int(np.argmax(partner))
        i_low = int(np.argmin(partner))

        # required reduction to match the second‑largest entry
        diff = M - second
        if diff <= 1e-12:
            # already flat – perform a tiny random move to escape plateaus
            i_high = int(rng.integers(0, n))
            i_low = int(rng.integers(0, n))
            delta = a[i_high] * 0.03
        else:
            denom = 2.0 * (partner[i_high] - partner[i_low]) + 1e-15
            delta_req = diff / denom
            # Clamp delta to a reasonable fraction of the source mass
            delta = min(delta_req * 0.9, a[i_high] * 0.5)
            if delta <= 1e-12:                 # fallback
                delta = a[i_high] * 0.03

        a[i_high] -= delta
        a[i_low] += delta

        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, S)

    return a

def _shape_sine(n: int) -> np.ndarray:
    """Half‑sine shape a_i ∝ sin(π·(i+1)/(2·n+1))."""
    theta = np.pi * (np.arange(n, dtype=float) + 1) / (2.0 * n + 1.0)
    a = np.sin(theta)
    return a


def _shape_semicircle(n: int) -> np.ndarray:
    """Discrete semi‑circle (upper half of a unit circle)."""
    x = (np.arange(n, dtype=float) - (n - 1) / 2.0) / (n / 2.0)
    a = np.sqrt(np.maximum(0.0, 1.0 - x * x))
    return a


def _shape_triangle(n: int) -> np.ndarray:
    """Symmetric triangle: rises then falls linearly."""
    half = (n + 1) // 2
    left = np.linspace(1.0, half, half, endpoint=False)
    right = left[::-1] if n % 2 == 0 else np.concatenate([left, left[-2::-1]])
    a = np.concatenate([left, right])[:n]
    return a


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Diverse random point on the simplex Σa = √(2 n)."""
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                         # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                       # pure exponential
        lam = rng.exponential(scale=0.02)
        a = np.exp(-lam * idx)

    elif family == 2:                       # power × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.02)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                       # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n))

    elif family == 4:                       # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                             np.exp(-lam2 * np.arange(b, n))])

    else:                                   # almost uniform with jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(old: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a shorter sequence to a longer length."""
    old_n = old.size
    if old_n == new_n:
        return old.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    up = np.interp(new_x, old_x, old)
    # re‑project to the appropriate simplex for the new length
    target = np.sqrt(2.0 * new_n)
    up = _project_to_simplex(up, target)
    return up

def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Very light SA that moves a random fraction of mass between two sites."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    cur_score = _score(a)
    best_score = cur_score

    T = cur_score * 0.02 + 1e-9
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-16:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        frac = rng.random() * 0.15
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta

        new_score = _score(a)
        dE = new_score - cur_score
        accept = dE <= 0.0 or (T > 0.0 and rng.random() < np.exp(-dE / T))

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling

    return best

    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the given wall‑clock budget,
    explores several lengths, starts from a variety of deterministic and
    random seeds, applies Adam‑based smooth‑max descent, exact sub‑gradient
    descent, an aggressive mass‑balance refinement and finishes with a short
    simulated‑annealing polish.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Optional warm‑start from a previously found construction
    # ------------------------------------------------------------------
    earlier = globals().get("height_sequence_1")
    if isinstance(earlier, (list, tuple, np.ndarray)):
        a0 = np.asarray(earlier, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, sqrt(2.0 * n0))
        sc0 = _score(a0)
        best_seq = a0.copy()
        best_score = sc0
        print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) "
              f"score={sc0:.6f}")

    # ------------------------------------------------------------------
    # Candidate lengths – larger n gives more freedom but costs a bit more
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000, 10000, 12000, 15000]

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)
        print(f"[{time.time():.1f}] ── Exploring length n={n}")

        # ------------------------------------------------------------------
        # Build a small pool of seeds for this length
        # ------------------------------------------------------------------
        seeds = []

        # 1) Upsampled version of the current best (if any)
        if best_seq is not None:
            up = _upsample_sequence(best_seq, n)
            seeds.append(up)

        # 2) Deterministic shapes
        seeds.append(_project_to_simplex(_shape_sine(n), target_sum))
        seeds.append(_project_to_simplex(_shape_semicircle(n), target_sum))
        seeds.append(_project_to_simplex(_shape_triangle(n), target_sum))

        # 3) A fresh random start
        seeds.append(_random_initial_sequence(n, rng))

        # ------------------------------------------------------------------
        # Process each seed – allocate a modest slice of the remaining budget
        # ------------------------------------------------------------------
        for seed_idx, a_start in enumerate(seeds):
            if time.time() >= deadline:
                break

            # ----- Phase 1 : Adam smooth‑max descent -----
            phase_dead = min(deadline,
                             time.time() + min(12.0,
                                               (deadline - time.time()) * 0.15))
            a = _adam_softmax_descent(a_start, phase_dead)

            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[{time.time():.1f}] New best after Adam (seed {seed_idx}) "
                      f"n={n} score={sc:.6f}")

            # ----- Phase 2 : exact max‑sub‑gradient descent -----
            sub_dead = min(deadline,
                           time.time() + min(8.0,
                                             (deadline - time.time()) * 0.10))
            a = _subgradient_descent_max(a, sub_dead)

            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[{time.time():.1f}] New best after sub‑grad (seed {seed_idx}) "
                      f"n={n} score={sc:.6f}")

            # ----- Phase 3 : targeted mass‑balance refinement -----
            mb_dead = min(deadline,
                          time.time() + min(6.0,
                                            (deadline - time.time()) * 0.07))
            a = _targeted_mass_balance(a, mb_dead)

            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[{time.time():.1f}] New best after mass‑balance (seed {seed_idx}) "
                      f"n={n} score={sc:.6f}")

        # ------------------------------------------------------------------
        # If we spent very little time on this length, allow an extra random run
        # ------------------------------------------------------------------
        extra_time = deadline - time.time()
        if extra_time > 5.0:
            # one more random restart with a short Adam run
            a_rand = _random_initial_sequence(n, rng)
            phase_dead = min(deadline, time.time() + min(5.0, extra_time * 0.3))
            a = _adam_softmax_descent(a_rand, phase_dead)
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Extra random Adam improvement n={n} "
                      f"score={sc:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with simulated annealing (if time remains)
    # ------------------------------------------------------------------
    if best_seq is not None:
        left = deadline - time.time()
        if left > 8.0:
            a_final = _simulated_annealing(best_seq, deadline)
            sc = _score(a_final)
            if sc < best_score:
                best_score = sc
                best_seq = a_final.copy()
                print(f"[{time.time():.1f}] Final SA improvement: score={sc:.6f}")

    # ------------------------------------------------------------------
    # Clean up – trim trailing near‑zero entries and report the official score
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback: uniform distribution of length 1000 (should never happen)
        n = 1000
        best_seq = np.full(n, sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
