"""
Propose a step‑function (non‑negative sequence) that minimises the
``evaluate_sequence`` objective.

Approach
--------
1. **Parametric families** – generate candidate vectors from several simple
   families (power‑law, exponential, truncated linear).  For each family
   the L¹‑norm is projected onto the simplex Σa = √(2 n).

2. **Ordering heuristics** – for every candidate vector generate a handful
   of orderings that aim to reduce large‑by‑large overlaps:
      * ascending / descending
      * alternating small–large
      * arithmetic “step” permutations with coprime strides
      * a few random shuffles.

   The best ordering is kept.

3. **Continuous refinement** – the best few candidates are refined with a
   projected gradient descent on a smooth log‑sum‑exp surrogate of the
   maximal convolution entry (the code from the previous solution).

4. **Permutation‑annealing** – after the gradient step the values are kept,
   but we further improve the ordering by a simulated‑annealing process that
   swaps entries (preserving the L¹‑norm) and accepts worsening moves with a
   temperature schedule.

5. **Return the best vector found** – converted to a plain Python list.

All steps respect the global deadline (default 1000 s).  The algorithm
relies only on NumPy and the provided ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex Σx = s, x≥0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection onto {x_i ≥ 0, Σx_i = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) surrogate of max(b) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b) where
        b = a*a (full convolution)
        g = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a ⋆ p) ,  p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]

    # full convolution a*a
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Projected gradient descent on the smoothed maximum
# ----------------------------------------------------------------------


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Annealed projected gradient descent on the log‑sum‑exp smooth max."""
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 1.0
    max_beta = 1e5
    momentum = np.zeros_like(a)
    rho = 0.9                     # momentum decay
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum step
        momentum = rho * momentum + (1.0 - rho) * g
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            lr = 0.5 * target_sum / (norm * (beta + 1.0))
            a -= lr * momentum

        # simplex projection
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 8e-4 / (beta + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal beta
        if it % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.06, max_beta)

        it += 1
    return a


# ----------------------------------------------------------------------
# Permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------


def _permutation_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Improve the ordering of a fixed multiset by random swaps.
    The L¹‑norm stays unchanged, only the permutation varies.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    T = best_score * 0.02 + 1e-9
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # swap
        a[i], a[j] = a[j], a[i]

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        accept = False
        if cur_score <= best_score:
            accept = True
        elif T > 0.0 and rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # revert swap
            a[i], a[j] = a[j], a[i]

        T *= cooling

        # occasional global jitter (re‑shuffle a tiny fraction)
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.02
            if np.any(mask):
                shuffled = a[mask]
                rng.shuffle(shuffled)
                a[mask] = shuffled
                cur_score = _max_convolution_fft(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()

    return best


# ----------------------------------------------------------------------
# Parametric families (power‑law, exponential, truncated linear)
# ----------------------------------------------------------------------


def _normalize_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale to meet the ℓ¹‑sum target (preserving non‑negativity)."""
    scale = target / np.sum(v)
    a = v * scale
    # clamp tiny negatives that may appear due to numeric noise
    a = np.maximum(a, 0.0)
    return a


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    target = np.sqrt(2.0 * n)
    return _normalize_to_sum(raw, target)


def _exponential_sequence(n: int, rate: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.exp(-rate * (idx + shift))
    target = np.sqrt(2.0 * n)
    return _normalize_to_sum(raw, target)


def _linear_sequence(n: int, slope: float, intercept: float) -> np.ndarray:
    """
    a_i = max(0, intercept - slope * i)
    """
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, intercept - slope * idx)
    # Avoid the degenerate all‑zero case
    if raw.sum() == 0.0:
        raw[0] = 1.0
    target = np.sqrt(2.0 * n)
    return _normalize_to_sum(raw, target)


# ----------------------------------------------------------------------
# Ordering heuristics
# ----------------------------------------------------------------------


def _candidate_orderings(base: np.ndarray, rng: np.random.Generator) -> list[np.ndarray]:
    """
    Generate a small collection of orderings for the same multiset of values.
    The aim is to reduce large‑by‑large overlaps.
    """
    n = base.size
    asc = np.sort(base)          # smallest → largest
    desc = asc[::-1]             # largest → smallest
    out = [asc, desc]

    # alternating small–large (small first)
    alt = np.empty_like(base)
    lo, hi = 0, n - 1
    for i in range(n):
        if i % 2 == 0:
            alt[i] = asc[lo]
            lo += 1
        else:
            alt[i] = asc[hi]
            hi -= 1
    out.append(alt)

    # step‑permutations with coprime strides
    for step in (2, 3, 5, 7, 11, 13, 17, 19, 23):
        if np.gcd(step, n) != 1:
            continue
        perm = np.empty_like(base)
        pos = 0
        for i in range(n):
            perm[pos] = asc[i]      # assign i‑th smallest to position pos
            pos = (pos + step) % n
        out.append(perm)

    # a few random shuffles (preserve the multiset)
    for _ in range(7):
        shuffled = asc.copy()
        rng.shuffle(shuffled)
        out.append(shuffled)

    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the target
    ``evaluate_sequence`` objective.  The routine respects the allotted
    time budget, explores several parametric families, tries a handful of
    orderings, refines promising candidates with gradient descent and a
    permutation‑annealing phase, and finally returns the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_obj = np.inf

    # ------------------------------------------------------------------
    # 1. Parametric sweep (coarse but covering a few families)
    # ------------------------------------------------------------------
    # Candidate lengths – we go up to 8000 (still cheap with FFT)
    lengths = [2500, 3000, 3500, 4000, 5000, 6000, 8000]
    # power‑law grid
    p_grid = np.linspace(0.44, 0.62, 10)
    shift_grid = np.linspace(0.2, 3.0, 8)

    # exponential grid
    rate_grid = np.linspace(0.0005, 0.005, 10)
    exp_shift_grid = np.linspace(0.0, 2.0, 5)

    # linear grid
    slope_grid = np.linspace(0.0001, 0.005, 8)
    intercept_grid = np.linspace(0.5, 2.5, 6)

    # Keep a short elite list (by objective) for later refinement
    elite = []   # list of (score, array)

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # ----- power‑law -----
        for p in p_grid:
            for shift in shift_grid:
                base = _powerlaw_sequence(n, p, shift)
                # explore several orderings
                for cand in _candidate_orderings(base, rng):
                    score = _max_convolution_fft(cand)
                    if score < best_obj:
                        best_obj = score
                        best_seq = cand.copy()
                        print(f"[{time.time():.1f}] New global best (pow) n={n} p={p:.3f} shift={shift:.2f} obj={best_obj:.6f}")
                    # keep elite
                    if len(elite) < 7 or score < elite[-1][0]:
                        elite.append((score, cand))
                        elite.sort(key=lambda x: x[0])
                        elite = elite[:7]

        # ----- exponential -----
        for rate in rate_grid:
            for shift in exp_shift_grid:
                base = _exponential_sequence(n, rate, shift)
                for cand in _candidate_orderings(base, rng):
                    score = _max_convolution_fft(cand)
                    if score < best_obj:
                        best_obj = score
                        best_seq = cand.copy()
                        print(f"[{time.time():.1f}] New global best (exp) n={n} rate={rate:.5f} shift={shift:.2f} obj={best_obj:.6f}")
                    if len(elite) < 7 or score < elite[-1][0]:
                        elite.append((score, cand))
                        elite.sort(key=lambda x: x[0])
                        elite = elite[:7]

        # ----- linear (truncated) -----
        for slope in slope_grid:
            for intercept in intercept_grid:
                base = _linear_sequence(n, slope, intercept)
                for cand in _candidate_orderings(base, rng):
                    score = _max_convolution_fft(cand)
                    if score < best_obj:
                        best_obj = score
                        best_seq = cand.copy()
                        print(f"[{time.time():.1f}] New global best (lin) n={n} slope={slope:.5f} int={intercept:.2f} obj={best_obj:.6f}")
                    if len(elite) < 7 or score < elite[-1][0]:
                        elite.append((score, cand))
                        elite.sort(key=lambda x: x[0])
                        elite = elite[:7]

        # short random sampling for the current length (adds diversity)
        rnd_end = min(deadline, time.time() + 1.5)
        while time.time() < rnd_end:
            rand = rng.random(n)
            rand = _normalize_to_sum(rand, target_sum)
            score = _max_convolution_fft(rand)
            if score < best_obj:
                best_obj = score
                best_seq = rand.copy()
                print(f"[{time.time():.1f}] New global best (rand) n={n} obj={best_obj:.6f}")
            if len(elite) < 7 or score < elite[-1][0]:
                elite.append((score, rand))
                elite.sort(key=lambda x: x[0])
                elite = elite[:7]

    # ------------------------------------------------------------------
    # 2. Refinement of elite candidates (gradient + permutation SA)
    # ------------------------------------------------------------------
    for score0, seed_seq in elite:
        if time.time() >= deadline:
            break
        n = seed_seq.size
        sub_dead = min(deadline, time.time() + 25.0)   # up to 25 s per elite
        # gradient descent on the smooth surrogate
        refined = _gradient_descent_opt(seed_seq, sub_dead)

        # permutation‑annealing (uses the remaining global budget)
        refined = _permutation_annealing(refined, deadline)

        obj = _max_convolution_fft(refined)
        if obj < best_obj:
            best_obj = obj
            best_seq = refined.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL BEST (refined) n={n} obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 3. Final sanity check
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 3000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_obj = _max_convolution_fft(best_seq)

    # Trim trailing negligible entries (purely cosmetic)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Report the final evaluate_sequence value for verification
    try:
        eval_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {eval_val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective = {best_obj:.6f} length = {best_seq.size}")
    return best_seq.tolist()
