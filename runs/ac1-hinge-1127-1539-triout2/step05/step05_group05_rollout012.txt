def _adam_optimize(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2 * n)
    beta = 1.0
    beta_max = 5e5
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha0 = 0.2
    eps = 1e-8
    while time.time() < deadline:
        # increase beta gradually
        if t % 200 == 0 and beta < beta_max:
            beta = min(beta * 1.2, beta_max)
        sm, g = _smoothmax_and_gradient(a, beta)
        # Adam updates
        t += 1
        m = 0.9 * m + 0.1 * g
        v = 0.999 * v + 0.001 * (g**2)
        m_hat = m / (1 - 0.9**t)
        v_hat = v / (1 - 0.999**t)
        step = alpha0 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        # project to simplex
        a = _project_to_simplex(np.maximum(a, 0.0), S)
    return a
```

def _lp_update(f, M):
    n = len(f)
    g = cp.Variable(n)
    constraints = [g >= 0]
    # define convolution: for each k, sum_i f_i * g_{k-i} <= M
    # need to treat out-of-range as 0; we can build matrix.
    for k in range(2*n - 1):
        # compute term = sum_i f_i * g_{k-i}
        # only i where 0 <= k-i < n <=> max(0, k-(n-1)) <= i <= min(k, n-1)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        # sum over i of f[i] * g[k-i] -> linear combination of g entries
        coeffs = f[i_low:i_high+1][::-1]   # because g index = k - i
        # add constraint: coeffs @ g[i_low:(i_high+1)] <= M
        constraints.append(cp.sum(cp.multiply(coeffs, g[i_low:i_high+1])) <= M)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True)
    return g.value

def _lp_opt_step(f, M):
    n = len(f)
    c = -np.ones(n)  # maximize sum(g) => minimize -sum(g)
    # Build A_ub (2n-1 rows, n cols)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_low = max(0, k - (n-1))
        i_high = min(k, n-1)
        # g index ranges from i_low to i_high
        # coefficient for g_j where j = k - i: we need f_i
        # So we fill row k, col j with f_i where i = k - j.
        # Equivalent: For j in [i_low,...,i_high], i = k - j
        # So A[k, j] = f[k - j]
        A[k, i_low:i_high+1] = f[k - np.arange(i_low, i_high+1)]
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    # Use linprog
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    g = res.x
    return g

    step = 0
    while time.time() < deadline:

        max_b = np.max(b)

        ks = np.where(np.abs(b - max_b) <= 1e-12)[0]

"""
Propose a step‑function that minimises

    2 * n * max(convolve(a, a)) / (sum(a)**2)

under the ℓ¹‑simplex constraint Σ a_i = √(2 n) and a_i ≥ 0.
The search consists of three layers:

1. **Fast multi‑length scan** – several random families (including new
   “U‑shape” and “sinusoidal” families) are projected on the simplex.
   The best few candidates are kept.

2. **Refinement** – for each elite candidate we run a long
   exponentiated‑gradient (mirror‑descent) optimisation on the smooth
   log‑sum‑exp surrogate of the max‑convolution, followed by a
   deterministic sub‑gradient descent on the true ℓ∞‑norm.  The
   step‑size schedules are aggressive and the smoothing parameter β is
   annealed to ~1e7, giving a close approximation of the real max.

3. **Targeted simulated‑annealing** – a final short SA run (the same
   routine as in the reference implementation) removes remaining
   spikes.

All phases respect the global 1000 s budget.  The function returns a
plain Python list of non‑negative floats; the evaluator `evaluate_sequence`
is called only once at the end for a sanity check.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (linear convolution via zero‑padded FFT)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Return max entry of the linear convolution a*a."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean, Duchi et al. 2008)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x_i ≥ 0, Σ x_i = s}."""
    if s <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) of the convolution and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Return (g, b) where
        b = a*a (full convolution)
        g = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a ⋆ p)   with   p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]

    # ----- b = a*a -------------------------------------------------------
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ----- soft‑max probabilities ---------------------------------------
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))          # stable exp
    p = exps / np.sum(exps)

    # ----- gradient g = 2·(a ⋆ p) --------------------------------------
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1               # = 3·n‑2
    size2 = 1 << ((conv_len).bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]        # central n entries
    return g, b


# ----------------------------------------------------------------------
# Mirror‑descent optimiser (exponentiated gradient) on the smooth max
# ----------------------------------------------------------------------


def _mirror_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Exponentiated‑gradient descent on the log‑sum‑exp surrogate.
    Keeps the iterate on the ℓ¹‑simplex automatically.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 1.0
    beta_max = 1e7                     # stronger final smoothing
    eta0 = 0.9                         # bigger base learning rate
    step = 0
    eps = 1e-12

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # exponentiated‑gradient update
        eta = eta0 / np.sqrt(step + 1.0)
        a = a * np.exp(-eta * g)       # element‑wise
        a = np.maximum(a, eps)         # avoid exact zeros for the log

        # renormalise to the ℓ¹‑simplex
        s = np.sum(a)
        if s > 0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # anneal β (slow, to keep the surrogate smooth early on)
        if step % 150 == 0 and beta < beta_max:
            beta = min(beta * 1.12, beta_max)

        step += 1

    return a


# ----------------------------------------------------------------------
# Deterministic sub‑gradient descent on the true max‑norm
# ----------------------------------------------------------------------


def _subgradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected sub‑gradient descent on the true ℓ∞‑norm of the self‑convolution.
    The sub‑gradient at the current maximal index k is 2·a shifted.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    rng = np.random.default_rng()
    last_improvement = time.time()
    while time.time() < deadline:
        b = _full_convolution_fft(a)
        max_b = float(b.max())
        # all indices that achieve (or are very close to) the maximum
        ks = np.where(np.abs(b - max_b) <= 1e-12)[0]

        # accumulate the sub‑gradient over the active set
        g = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            j = k - idx
            g[idx] += 2.0 * a[j]
        g /= max(1, ks.size)                     # average if several active ks

        # simple back‑tracking line‑search
        lr = 0.2 * target_sum / n                 # initial step size
        improved = False
        for _ in range(12):
            a_candidate = a - lr * g
            a_candidate = np.maximum(a_candidate, 0.0)
            a_candidate = _project_to_simplex(a_candidate, target_sum)
            new_max = _max_convolution_fft(a_candidate)
            if new_max < max_b - 1e-12:
                a = a_candidate
                improved = True
                last_improvement = time.time()
                break
            lr *= 0.5
        if not improved:
            # occasional random jitter to escape shallow plateaus
            if rng.random() < 0.005:
                jitter = rng.normal(scale=target_sum * 0.001, size=n)
                a = a + jitter
                a = np.maximum(a, 0.0)
                a = _project_to_simplex(a, target_sum)
    return a


# ----------------------------------------------------------------------
# Targeted simulated‑annealing refinement (as in the reference code)
# ----------------------------------------------------------------------


def _targeted_simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    SA that repeatedly moves a fraction of mass from an index that
    heavily contributes to the current maximal convolution entry to a
    random destination.  Temperature decays multiplicatively.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # initial temperature proportional to current objective
    T = best_score * 0.02 + 1e-8
    cooling = 0.9993
    rng = np.random.default_rng()
    max_transfer = 0.25           # up to 25 % of the selected mass

    while time.time() < deadline:
        b = _full_convolution_fft(a)
        k_max = int(np.argmax(b))

        # indices that actually affect b[k_max]
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idx_range = np.arange(lo, hi + 1)                     # possible i
        contrib = a[idx_range] * a[k_max - idx_range]        # a_i * a_{k‑i}
        weight_sum = np.sum(contrib)

        if weight_sum > 0:
            probs = contrib / weight_sum
            i = rng.choice(idx_range, p=probs)
        else:
            i = rng.integers(0, n)

        # pick a destination j ≠ i uniformly
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        # transfer a random fraction of the mass at i to j
        delta = a[i] * rng.random() * max_transfer
        if delta <= 0.0:
            continue
        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score

        accept = False
        if cur_score <= best_score:
            accept = True
        elif rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # revert the move
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, np.sqrt(2.0 * n))
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()

    return best


# ----------------------------------------------------------------------
# Random initialisers (diverse families, plus two new shapes)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Generate a random non‑negative sequence on the ℓ¹‑simplex
    Σ a_i = √(2 n).  Several families are mixed to increase diversity,
    including two new families (U‑shape and sinusoid).
    """
    target_sum = np.sqrt(2.0 * n)
    kind = rng.integers(0, 8)                     # now 0…7
    idx = np.arange(n, dtype=float)

    if kind == 0:                     # pure power‑law (decreasing)
        p = rng.uniform(0.15, 1.0)
        a = (idx + 1.0) ** (-p)

    elif kind == 1:                 # exponential decay from the left edge
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif kind == 2:                 # mixed power‑law + exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif kind == 3:                 # Dirichlet (uniform simplex)
        a = rng.dirichlet(alpha=np.ones(n))

    elif kind == 4:                 # symmetric edge‑heavy
        p = rng.uniform(0.2, 1.2)
        left = (idx + 1.0) ** (-p)
        right = (n - idx) ** (-p)
        a = left + right

    elif kind == 5:                 # Beta‑shaped
        alpha = rng.uniform(0.2, 2.0)
        beta = rng.uniform(0.2, 2.0)
        x = (idx + 0.5) / n
        a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))

    elif kind == 6:                 # U‑shape (inverse distance to centre)
        centre = (n - 1) / 2.0
        dist = np.abs(idx - centre) + 0.5
        a = 1.0 / np.sqrt(dist)

    else:                           # Sinusoidal (edge heavy)
        a = np.sin(np.pi * (idx + 0.5) / n) + 0.1

    # small Gaussian perturbation & simplex projection
    a += rng.normal(scale=target_sum * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target_sum)
    return a

    print(f"[{time.time():.1f}] Finished. Best objective (max‑conv) = "
          f"{best_score:.6f}   length = {best_seq.size}")

    return best_seq.tolist()
