def build_conv_matrix(a):
    n = len(a)
    rows = []
    cols = []
    data = []
    for col in range(n):
        # variable g[col] influences rows from col to col+n-1
        row_start = col
        # for each offset d from 0..n-1
        # coefficient a[d]
        rows.extend(range(row_start, row_start+n))
        cols.extend([col]*n)
        data.extend(a.tolist())
    # Now create sparse matrix shape (2*n-1, n)

    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

"""
A fast, targeted hill‑climbing optimizer for the max‑convolution problem.

The algorithm works in three stages

1. Initialise a non‑negative sequence on the ℓ¹‑simplex
   Σ a_i = √(2 n).  If a previous construction is available it is reused,
   otherwise a simple Dirichlet draw is used.

2. A short projected‑gradient descent on a smooth log‑sum‑exp proxy of the
   max‑convolution quickly moves the vector into a good region.

3. An extensive simulated‑annealing hill‑climb.  In each move a random
   fraction δ of mass is transferred from a “high‑impact’’ source index
   to a “low‑impact’’ destination index.  The impact of an index is the
   total contribution of that entry to the current maximal convolution
   values and is recomputed every few hundred iterations.  The convolution
   update is performed analytically in O(n) time using the formula

        Δbₖ = 2·δ·(a_{k‑j} – a_{k‑i}) (+ δ² at 2i, 2j and –2·δ² at i+j),

   which allows millions of cheap moves inside the wall‑clock budget.

The routine respects a user‑supplied time budget (default 1000 s) and
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a (length 2n‑1) using FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_full_convolution(a).max())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0, Σ x_i = s}.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient (used in the short
# projected‑gradient descent phase)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b)  where b = a*a .
    Returns (g, b) with g = 2·(a ⋆ p) and pₖ ∝ exp(β·bₖ).
    """
    n = a.shape[0]

    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1  # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Short projected gradient descent on the smooth max.
    Uses momentum, an annealed inverse temperature β and occasional jitter.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 8192.0
    beta_factor = 1.4

    step_base = target_sum * 0.15
    momentum = np.zeros_like(a)
    rho = 0.90

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β annealing
        if it % 350 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a

# Random initialiser
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative sequence on the ℓ¹‑simplex Σ a = √(2 n).
    """
    target = np.sqrt(2.0 * n)
    # simple Dirichlet draw (uniform on the simplex)
    a = rng.dirichlet(alpha=np.ones(n)) * target
    return a

# Targeted mass‑transfer hill climbing
# ----------------------------------------------------------------------


def _compute_importance(a: np.ndarray, b: np.ndarray, max_val: float) -> np.ndarray:
    """
    For each index i compute its total contribution to all entries of the
    convolution that are (within numerical tolerance) equal to max_val.
    This gives a cheap estimate of the gradient of the max‑convolution.
    """
    n = a.size
    # indices where the convolution attains its maximum (allow tiny fuzz)
    tol = max_val * 1e-12 + 1e-14
    max_idx = np.where(b >= max_val - tol)[0]

    importance = np.zeros(n, dtype=float)

    for k in max_idx:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # vector of column indices
        idxs = np.arange(i_start, i_end + 1)
        # a_{k-i}
        importance[i_start : i_end + 1] += a[k - idxs]
    return importance


def _hill_climbing_targeted(a0: np.ndarray,
                            deadline: float,
                            log_prefix: str = ""):
    """
    Simulated‑annealing hill‑climb that moves mass from high‑impact indices
    to low‑impact ones using the O(n) convolution update.
    Returns the best (a, b) pair observed before the deadline.
    """
    a = a0.copy()
    n = a.size
    b = _full_convolution(a)
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()
    best_b = b.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.02 + 1e-9  # initial temperature
    cooling = 0.9998            # slower cooling than the baseline
    it = 0
    # recompute importance every few hundred iterations
    recompute_every = max(1, n // 8)

    # initialise importance‐based sampling probabilities
    importance = _compute_importance(a, b, cur_score)
    eps = 1e-20
    src_weights = a * (importance + eps)
    src_weights_sum = src_weights.sum()
    if src_weights_sum == 0:
        src_weights = np.ones_like(a)
        src_weights_sum = src_weights.sum()
    src_probs = src_weights / src_weights_sum

    # destination prefers low importance
    dst_weights = (importance.max() - importance + eps)
    dst_weights_sum = dst_weights.sum()
    if dst_weights_sum == 0:
        dst_weights = np.ones_like(a)
        dst_weights_sum = dst_weights.sum()
    dst_probs = dst_weights / dst_weights_sum

    while time.time() < deadline:
        # occasionally refresh the importance sampling distribution
        if it % recompute_every == 0:
            importance = _compute_importance(a, b, cur_score)

            src_weights = a * (importance + eps)
            src_weights_sum = src_weights.sum()
            if src_weights_sum == 0:
                src_weights = np.ones_like(a)
                src_weights_sum = src_weights.sum()
            src_probs = src_weights / src_weights_sum

            dst_weights = (importance.max() - importance + eps)
            dst_weights_sum = dst_weights.sum()
            if dst_weights_sum == 0:
                dst_weights = np.ones_like(a)
                dst_weights_sum = dst_weights.sum()
            dst_probs = dst_weights / dst_weights_sum

        # pick a source proportional to impact * mass
        i = rng.choice(n, p=src_probs)

        # pick a destination that has low impact
        j = rng.choice(n, p=dst_probs)

        if i == j or a[i] < 1e-14:
            it += 1
            T *= cooling
            continue

        # move a random fraction (up to 50 % of the source mass)
        delta = rng.random() * (0.5 * a[i])
        if delta <= 0.0:
            it += 1
            T *= cooling
            continue

        # ----- update convolution in O(n) -----
        b_new = b.copy()
        # linear part
        b_new[i : i + n] -= 2.0 * delta * a
        b_new[j : j + n] += 2.0 * delta * a
        # quadratic correction
        k2i = 2 * i
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        k2j = 2 * j
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        kij = i + j
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # ---- acceptance ----
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # Metropolis acceptance
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score

            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                best_b = b.copy()
                print(
                    f"[{time.time():.1f}] {log_prefix} improvement: {best_score:.6f}"
                )

        T *= cooling
        it += 1

    return best_a, best_b

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock budget (default 1000 s) and returns
    the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose problem size
    # ------------------------------------------------------------------
    # Prefer the length of a previously found construction; otherwise 5000.
    earlier = globals().get("height_sequence_1")
    if isinstance(earlier, (list, tuple, np.ndarray)):
        a0 = np.asarray(earlier, dtype=float)
        n = a0.size
        # if the earlier sequence is huge, down‑sample to a manageable size
        if n > 5000:
            # simple linear interpolation (will be re‑projected later)
            x_old = np.linspace(0, 1, n)
            x_new = np.linspace(0, 1, 5000)
            a0 = np.interp(x_new, x_old, a0)
            n = a0.size
    else:
        n = 5000
        a0 = None

    target_sum = np.sqrt(2.0 * n)

    if a0 is not None:
        a = _project_to_simplex(a0, target_sum)
        print(f"[{time.time():.1f}] Seeded from existing sequence (n={n})")
    else:
        a = _random_initial_sequence(n, rng)
        print(f"[{time.time():.1f}] Random initialise (n={n})")

    # ------------------------------------------------------------------
    # Phase A – a short projected‑gradient descent (quick warm‑start)
    # ------------------------------------------------------------------
    gd_deadline = min(deadline, time.time() + min(30.0, (deadline - time.time()) * 0.05))
    if time.time() < gd_deadline:
        a = _gradient_descent_opt(a, gd_deadline)
        gd_score = _max_convolution_fft(a)
        print(
            f"[{time.time():.1f}] GD warm‑start score: {gd_score:.6f}"
        )

    # ------------------------------------------------------------------
    # Phase B – long targeted hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_best, _ = _hill_climbing_targeted(a, deadline, log_prefix="HC")
    else:
        a_best = a

    # ------------------------------------------------------------------
    # Clean‑up: enforce feasibility and drop trailing near‑zeros
    # ------------------------------------------------------------------
    a_best = np.maximum(a_best, 0.0)
    a_best = _project_to_simplex(a_best, target_sum)

    nz = np.where(a_best > 1e-12)[0]
    if nz.size:
        a_best = a_best[: nz[-1] + 1]
    else:
        a_best = a_best[:1]

    # ------------------------------------------------------------------
    # Final evaluation (for logging)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(a_best.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={a_best.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return a_best.tolist()
