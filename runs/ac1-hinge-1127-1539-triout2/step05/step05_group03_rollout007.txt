Implementation details of active-set refinement:

def refine_active_set(a, deadline, rng):
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = a.copy()
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    while time.time() < deadline:
        conv = _full_convolution(a)
        M = conv.max()
        # record best
        if M < best_score:
            best_score = M
            best_a = a.copy()
        # active constraint indices: top K or all > M - eps
        K = 200
        # indices sorted descending
        idx_desc = np.argpartition(conv, -K)[-K:]  # unsorted set of top K
        # optionally also include indices within eps
        eps_rel = 0.01  # 1% tolerance
        threshold = M * (1 - eps_rel)
        extra_idx = np.where(conv >= threshold)[0]
        # combine unique
        active_idx = np.unique(np.concatenate((idx_desc, extra_idx)))
        # build A_ub
        rows = []
        cols = []
        data = []
        for row_idx, k in enumerate(active_idx):
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            # For i in [i_start, i_end], column j = k - i
            # coefficient = a[i] (since f = a)
            # Append vectors
            # vector length = i_end - i_start + 1
            i_vals = np.arange(i_start, i_end + 1)
            cols.extend(k - i_vals)
            rows.extend([row_idx] * len(i_vals))
            data.extend(a[i_vals])
        A = sparse.csr_matrix((data, (rows, cols)), shape=(len(active_idx), n))
        # linear program: maximize sum(g) => minimize -1^T g
        c = -np.ones(n)
        bounds = [(0, None)] * n
        # inequality A g <= M (same M)
        b_ub = np.full(len(active_idx), M, dtype=float)
        # solve
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})
        if not res.success:
            # if fails, break
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        # rescale to match sum S
        factor = S / sum_g
        g_scaled = g * factor
        # mixing
        t = 0.2
        a = (1 - t) * a + t * g_scaled
        # optional random mass swap jitter
        if rng.random() < 0.05:
            i, j = rng.integers(0, n, size=2)
            if a[i] > 1e-9:
                delta = rng.random() * 0.05 * a[i]
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)
        # early stopping if no improvement after some iterations
        # Could track if best_score hasn't improved for many loops.
        # But continue until time runs out.
    return best_a, best_score

Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm
---------
* Initialise a moderate‑size sequence (≈2‑4 k) with a simple decreasing
  shape (geometric or power‑law) and project it onto the ℓ¹‑simplex
  Σa = √(2 n).
* Repeatedly improve the sequence by solving a *linear programme* on the
  active constraints of the self‑convolution:
      maximise Σ g_j          (g ≥ 0)
      s.t.    (a * g)_k ≤ M   for the k with the largest convolution values,
  where M = max(a*a).  The LP has only a few hundred constraints
  (the current “critical” shifts) and therefore solves very quickly
  with ``scipy.optimize.linprog``.
* The LP solution g may have a larger ℓ¹‑norm than a; we scale it back
  to the required sum (the scaling factor is ≤ 1, so the convolution
  bounds stay valid).  If the scaled g already yields a smaller peak,
  we replace a by g, otherwise we move a a little towards g
  (mixing factor t≈0.15) and re‑project onto the simplex.
* Once the time budget for the moderate size is exhausted we up‑sample
  the best sequence to a larger length (≈8‑12 k) and repeat the same
  active‑set LP refinement.
* The whole procedure runs for the allotted time (≤ 1000 s).  The final
  sequence is returned as a plain list of floats.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via FFT (returns length 2·n‑1)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Upsample / linear interpolation
# ----------------------------------------------------------------------
def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Resize an existing sequence to length n using linear interpolation."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Initial sequence factories (non‑negative, ℓ¹‑norm √(2·n))
# ----------------------------------------------------------------------
def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """Create a random non‑negative vector of length n with Σa = √(2·n)."""
    S = np.sqrt(2.0 * n)

    # 0 – up‑sampled previous best (if supplied)
    # 1 – uniform Dirichlet
    # 2 – geometric decay (left‑hand side, symmetrised)
    # 3 – power‑law / reciprocal edge‑heavy
    # 4 – random convex combination of a few bases
    typ = rng.integers(0, 5)

    if typ == 0 and base_seq is not None and base_seq.size > 10:
        a = _upsample_seq(base_seq, n)
        a += rng.normal(scale=a.mean() * 1e-3, size=n)   # tiny jitter
        a = np.clip(a, 0.0, None)
    elif typ == 1:
        a = rng.random(n)               # Dirichlet(1)
    elif typ == 2:
        rate = rng.uniform(0.92, 0.998)
        left = rate ** np.arange(n)
        right = rate ** np.arange(n)[::-1]
        a = (left + right) / 2.0
    elif typ == 3:
        idx = np.arange(n, dtype=float)
        a = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.8)
        a = a ** power
    else:
        cand = []
        cand.append(rng.random(n))
        rate = rng.uniform(0.92, 0.998)
        left = rate ** np.arange(n)
        right = rate ** np.arange(n)[::-1]
        cand.append((left + right) / 2.0)
        idx = np.arange(n, dtype=float)
        rec = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.8)
        cand.append(rec ** power)
        w = rng.random(len(cand))
        w /= w.sum()
        a = sum(wi * ci for wi, ci in zip(w, cand))

    # Normalise to the required ℓ¹‑norm
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0)
        a_sum = a.sum()
    a = a / a_sum * S
    return a

# Active‑set linear‑programme refinement

        # Row k corresponds to (a * g)_{active_idx[k]} ≤ M.

            if a_new[i] > 1e-12:
                delta = rng.random() * 0.045 * a_new[i]
                a_new[i] -= delta
                a_new[j] += delta
                a_new = _project_simplex(a_new, S)

        a = a_new

        # Simple stagnation check – break if we have not improved for a while.
        if it % 500 == 0 and best_M == M:
            break

    return best_a, best_M


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin for the final external evaluation
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Retrieve a previously discovered best sequence (if available) for up‑sampling
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    # ------------------------------------------------------------------
    # Phase 1 – medium sized problem (≈2–4 k entries)
    # ------------------------------------------------------------------
    n1 = int(rng.integers(2000, 4000))
    if base_seq is not None:
        a = _upsample_seq(base_seq, n1)
    else:
        a = _initial_sequence(None, n1, rng)
    a = _project_simplex(a, np.sqrt(2.0 * n1))

    phase1_deadline = min(deadline, time.time() + 0.55 * (deadline - time.time()))
    a, best_M1 = _refine_active_set(a, phase1_deadline, rng)

    # ------------------------------------------------------------------
    # Phase 2 – larger problem (≈8–12 k entries) to exploit the larger
    # search space.  The shape found in phase 1 is up‑sampled and refined.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n2 = int(rng.integers(8000, 12000))
        a = _upsample_seq(a, n2)
        a = _project_simplex(a, np.sqrt(2.0 * n2))
        a, best_M2 = _refine_active_set(a, deadline, rng)
        final_M = best_M2
    else:
        final_M = best_M1

    # ------------------------------------------------------------------
    # Convert to plain list, compute the official score and report.
    # ------------------------------------------------------------------
    final_seq = a.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} "
          f"internal max‑conv = {final_M:.6f} "
          f"length = {len(final_seq)}")
    return final_seq
