"""
Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence``.

The algorithm consists of three stages:

1. **Parameter sweep** – generate a family of analytic shapes
   a_i = (i+1)^(-p) (monotonically decreasing) and the symmetric
   variant a_i = (i+1)^(-p) + (n-i)^(-p).  For each exponent p we
   mix the shape with a uniform vector, normalise the ℓ¹‑norm to
   √(2·n) and keep the best candidate (lowest peak of the self‑
   convolution).

2. **Smooth Adam optimisation** – starting from the best candidate,
   minimise a smooth log‑sum‑exp approximation of the peak
   convolution.  The gradient is obtained via FFT‑based convolutions
   and the iterate is projected onto the simplex after each step.

3. **Sub‑gradient refinement** – finally perform a projected
   sub‑gradient descent that directly reduces the worst convolution
   entry.

The whole routine is time‑aware: it stops well before the external
budget so that the final evaluation cannot be cut off.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full convolution a * a (linear, length 2·n‑1)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Gradient of the smooth log‑sum‑exp max‑approximation
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights w_i = exp(alpha·conv_i) / Σ_j exp(alpha·conv_j)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  (1/α)·log‑sum‑exp(α·(a*a)) w.r.t. a.
    Returns a vector of length n.
    """
    n = a.shape[0]
    # reverse w because convolution formula uses reversed index order
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed max‑norm
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   alpha_start: float = 0.5,
                   alpha_max: float = 200.0) -> tuple[np.ndarray, float]:
    """Projected Adam minimising the smoothed max‑norm."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    lr = 0.3
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    alpha = alpha_start
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        cur_max = np.max(conv)
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # soft‑max on the convolution values
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto the simplex
        a = _project_simplex(a, S)

        # very occasional random mass‑swap – helps escape plateaus
        if rng.random() < 0.02:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.04 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < best_score or rng.random() < 0.1:
                    a = cand

        # raise temperature (sharpen the soft‑max)
        if alpha < alpha_max:
            alpha *= 1.011

        # decay learning rate gently
        lr *= 0.99995

        # early stop if no improvement for a while
        if it % 4000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient phase (focus on the current worst shift)
# ----------------------------------------------------------------------
def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      eta0: float = 0.12) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑norm."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the maximum entry
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # occasional jitter to help escape tiny plateaus
        if rng.random() < 0.04:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.2:
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()

        if it % 5000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimiser (Adam → sub‑gradient)
# ----------------------------------------------------------------------
def _optimize_combined(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run a smooth Adam phase for ~60 % of the remaining time,
    then a sub‑gradient refinement for the rest.
    """
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    smooth_deadline = now + 0.6 * remaining
    a_s, score_s = _optimize_adam(a0, smooth_deadline, rng)

    if time.time() < deadline:
        a_f, score_f = _optimize_subgrad(a_s, deadline, rng)
        if score_f < score_s:
            return a_f, score_f
    return a_s, score_s


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine first scans a small parametric family (power‑law
    decays possibly mixed with a uniform component) to obtain a good
    starting point, then refines it with Adam + sub‑gradient descent.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin for the final external evaluation
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1)  Parameter sweep for analytic shapes
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # choose a fairly large dimension – the bound improves with n
    n = int(rng.integers(14000, 19001))          # 14k … 19k
    S = np.sqrt(2.0 * n)

    # exponent grid (power‑law decay) – we search p ∈ [0.4, 0.85]
    p_grid = np.linspace(0.40, 0.85, 23)          # step ≈ 0.02
    lam_grid = np.linspace(0.0, 1.0, 7)          # mixing weight with uniform

    # uniform baseline (kept for completeness)
    uniform = np.ones(n)

    # helper to evaluate a raw (non‑normalised) vector
    def _evaluate_raw(vec_raw: np.ndarray) -> float:
        nonlocal best_seq, best_score
        if vec_raw.sum() == 0.0:
            return np.inf
        a = vec_raw / vec_raw.sum() * S               # simplex projection
        cur = np.max(_full_convolution(a))
        if cur < best_score:
            best_score = cur
            best_seq = a.copy()
        return cur

    # ---- monotone power‑law a_i = (i+1)^(-p) -------------------------
    idx = np.arange(n, dtype=float) + 1.0          # i+1
    for p in p_grid:
        base = idx ** (-p)                         # decreasing shape
        for lam in lam_grid:
            mix = lam * uniform + (1.0 - lam) * base
            _evaluate_raw(mix)

    # ---- symmetric edge‑heavy shape
    rev_idx = n - np.arange(n, dtype=float)       # n-i
    for p in p_grid:
        base = idx ** (-p) + rev_idx ** (-p)       # symmetric power‑law
        for lam in lam_grid:
            mix = lam * uniform + (1.0 - lam) * base
            _evaluate_raw(mix)

    # ---- a few random seeds (tiny jitter around the best found) ----
    if best_seq is not None:
        for _ in range(4):
            cand = best_seq.copy()
            jitter = rng.normal(scale=cand.mean() * 1e-3, size=n)
            cand += jitter
            cand = np.clip(cand, 0.0, None)
            cand = _project_simplex(cand, S)
            _evaluate_raw(cand)

    # ------------------------------------------------------------------
    # 2)  Refine the best candidate with the combined optimiser
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        refined_seq, refined_score = _optimize_combined(best_seq, deadline, rng)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()

    # ------------------------------------------------------------------
    # 3)  Fallback – if nothing succeeded, use a simple uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = uniform.copy()
        best_seq = _project_simplex(best_seq, S)

    final_seq = best_seq.tolist()

    # ------------------------------------------------------------------
    # 4)  Final sanity check using the black‑box evaluator
    # ------------------------------------------------------------------
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} "
          f"(internal best = {best_score:.6f}) length = {len(final_seq)}")
    return final_seq
