def _random_local_search(a_init, deadline, rng, S):
    a = a_init.copy()
    n = len(a)
    best_a = a.copy()
    conv = _full_convolution(a)

    while time.time() < deadline:

        # select i with a[i] > 0
        i = rng.integers(0, n)
        # ensure a_i > epsilon else pick again
        if a[i] <= 0:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = a[i] * rng.random() * 0.1  # move up to 10% of mass at i

            best_a = cand.copy()
            a = cand
        else:
            # maybe accept with small probability

                a = cand

def _full_convolution(a):
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv

def _optimize_adam(a0, deadline, rng, S):
    a = a0.copy()
    n = len(a)
    lr = 0.5
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    alpha = 0.5  # temp
    while time.time() < deadline:
        t += 1
        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w) # shape n, includes factor 2
        # Adam update
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # project onto simplex (L1 norm)
        a = _project_simplex(a, S)

        # occasionally random perturb
        if rng.random() < 0.02:
            # random mass transfer
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = a[i] * rng.random() * 0.02
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # increase alpha periodically
        alpha = min(alpha * 1.03, 500.0)
        # early break if no improvement for many steps
        if t % 3000 == 0 and best_score == cur_score:
            break

    return best_a, best_score

"""
Propose a non‑negative step sequence that minimises the self‑convolution peak.

Algorithm
---------
* Work on the L¹‑simplex Σ a_i = √(2·n) – the evaluation metric equals the
  maximal entry of the full convolution a*a when the sum is fixed.
* Use a multiscale scheme: start with a modest length, optimise, then up‑sample
  the best shape to a larger length and continue.  Larger n gives more freedom.
* At each scale run three phases
    1. Adam on a smooth approximation of the max‑norm (log‑sum‑exp).  The
       gradient is obtained by a weighted self‑convolution (FFT) and the iterate
       is projected back onto the simplex.
    2. Projected sub‑gradient descent that directly attacks the current worst
       convolution entry.
    3. Simple random “mass‑swap’’ hill‑climbing (simulated‑annealing style) to
       escape possible plateaus.
* The whole budget (≈1000 s) is split over a few increasing problem sizes.
* The best sequence found is returned; its external score is printed for
  reference.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean L1‑ball)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient term for the smooth max‑norm:
        g_j = 2 Σ_i w_{i+j} a_i .
    Both ``a`` (len n) and ``w`` (len 2·n‑1) are real vectors.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Smooth approximation of the max (log‑sum‑exp) and its weights
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights: w_i = exp(alpha·conv_i) / Σ_j exp(alpha·conv_j) (stable)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Linear up‑sampling (interpolation) – used when moving to a larger n
# ----------------------------------------------------------------------


def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a to length new_n."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=float)
    return np.interp(x_new, x_old, a)


# ----------------------------------------------------------------------
# Adam optimisation on the smooth max‑norm
# ----------------------------------------------------------------------


def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   S: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth max‑norm.
    Returns the best vector (projected onto the simplex) and its max‑conv value.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters (tuned heuristically)
    lr = 0.5
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    alpha = 0.5                     # temperature for the soft‑max
    while time.time() < deadline:
        t += 1

        # ---- gradient of soft‑max ----
        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- projection onto the simplex Σa = S ----
        a = _project_simplex(a, S)

        # ---- occasional random mass‑swap (exploration) ----
        if rng.random() < 0.02:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.02
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ---- temperature schedule (sharpen soft‑max) ----
        alpha = min(alpha * 1.03, 500.0)

        # cheap early‑stop if stagnating
        if t % 3000 == 0 and best_score == cur_score:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient optimiser (operates on the true max‑norm)
# ----------------------------------------------------------------------


def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that directly reduces the worst convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.35                   # larger initial step size
    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient for the entry at k_max
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional random perturbation to escape plateaus
        if rng.random() < 0.05:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.02
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = np.max(_full_convolution(cand))
                if cand_score < cur_score or rng.random() < 0.2:
                    a = cand
                    cur_score = cand_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()

        # cheap early‑stop test
        if it % 8000 == 0 and best_score == cur_score:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple random local search (mass‑swap hill climbing)
# ----------------------------------------------------------------------


def _random_local_search(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         S: float) -> tuple[np.ndarray, float]:
    """
    Randomly move a fraction of mass between two positions.
    Accept moves that improve the maximal convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    while time.time() < deadline:
        # pick source with non‑zero mass
        i = rng.integers(0, n)
        if a[i] <= 0.0:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # move at most 8 % of the source mass
        delta = a[i] * rng.random() * 0.08
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        # sum unchanged, non‑negativity guaranteed by construction
        conv = _full_convolution(cand)
        cand_score = np.max(conv)

        # accept if better, otherwise accept with a tiny probability
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
        elif rng.random() < 0.01:
            a = cand

        # occasional extra jitter to keep the search alive
        if rng.random() < 0.002:
            a = _project_simplex(a + rng.normal(scale=1e-4, size=n) * S, S)

    return best_a, best_score

                     budget_s: int = 1000,

    projected sub‑gradient descent and random local search.
    Returns the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final external evaluation
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_score = np.inf
    attempts = 0

    # A schedule of increasing problem sizes (adjusted on the fly)
    size_schedule = [600, 1500, 3500, 6200]

    for n in size_schedule:
        if time.time() >= deadline:
            break

        # allocate a time slice for this scale (≈30 % of remaining budget)
        now = time.time()
        remaining = deadline - now
        slice_time = min(remaining * 0.30, 180.0)   # cap to keep enough time later
        slice_deadline = now + slice_time

        # initialise a
        if best_seq is None:
            a0 = rng.random(n)
        else:
            a0 = _upsample(best_seq, n)

        # enforce the L¹‑simplex Σa = √(2·n)
        S = np.sqrt(2.0 * n)
        a0 = a0 / a0.sum() * S

        # --------------------------------------------------------------
        # Phase 1 – Adam on smooth max‑norm
        # --------------------------------------------------------------
        adam_deadline = min(slice_deadline, time.time() + slice_time * 0.55)

        # --------------------------------------------------------------
        # Phase 2 – Projected sub‑gradient descent
        # --------------------------------------------------------------
        subgrad_deadline = slice_deadline

        # --------------------------------------------------------------

        # keep global best

            best_seq = a_refined.copy()
            best_score = score_refined

    # ------------------------------------------------------------------
    # Return the result
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
        best_score = np.inf

    # External validation (the harness will also call evaluate_sequence)
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} "
          f"(internal best = {best_score:.6f}) length = {len(final_seq)}")
    return final_seq
