"""
Heuristic optimizer for the non‑negative step‑function heights.

The algorithm proceeds in three stages:

1. **Initialisation** – generate a few candidates for each length:
   * an up‑sampled version of the current best,
   * a handful of simple parametric shapes (power‑law, exponential,
     or mixed) and a few random draws.
   All candidates are projected onto the ℓ¹‑simplex
   Σ a_i = sqrt(2 n).

2. **Smoothed gradient descent** – minimise a smooth log‑sum‑exp
   approximation of the maximum of the self‑convolution using projected
   gradient descent with momentum and an annealing inverse temperature β.
   The routine is stopped when the time‑budget for the current phase is
   exhausted.

3. **Targeted mass‑transfer hill climbing** – starting from the best
   vector obtained above, repeatedly move a small amount of mass from a
   “source’’ index to a “destination’’ index.  The source is chosen among
   those positions that contribute most to the current peak(s) of the
   convolution (computed cheaply via one FFT correlation); the destination
   is drawn among a few random candidates and the best improvement is kept.
   Exact updates of the convolution are performed in O(n) time using the
   analytical formula
        Δbₖ = 2·δ·(a_{k‑j} – a_{k‑i}) (+δ² at 2i,2j and –2·δ² at i+j).
   An annealed temperature allows occasional uphill moves.

The whole search respects a wall‑clock budget (default 1000 s) and returns
the best sequence found as a plain Python list.  All numbers are kept
non‑negative and projected onto the required simplex, so the official
``evaluate_sequence`` sees a valid input.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Small linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a as a NumPy array."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0, Σ x_i = s}.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed maximum (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of   (1/β)·log Σ exp(β·b)   where   b = a*a .
    Returns (g, b) with g = 2·(a ⋆ p) and
    pₖ ∝ exp(β·bₖ).  Correlation is performed via FFT.
    """
    n = a.shape[0]

    # b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]                          # reversed a
    conv_len = n + (2 * n - 1) - 1           # = 3n‑2
    size2 = _next_pow2(conv_len)
    ar_fft = np.fft.rfft(a_rev, n=size2)
    pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(ar_fft * pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smoothed max.
    Uses momentum, an annealed inverse temperature β and occasional jitter.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 8192.0
    beta_factor = 1.4

    step_base = target_sum * 0.15
    momentum = np.zeros_like(a)
    rho = 0.90

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β annealing
        if it % 350 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative sequence on the ℓ¹‑simplex Σa = sqrt(2 n).
    Several families (power‑law, exponential, Dirichlet, piecewise) are
    sampled uniformly.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # pure power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length.
    The result is projected back onto the appropriate simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Targeted hill‑climbing (exact convolution updates)
# ----------------------------------------------------------------------


def _targeted_hill_climb(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Hill‑climbing that focuses mass moves on positions that influence the
    current maximum of the self‑convolution.  The move set is exact:
    b = a*a is updated in O(n) time using the analytic formula.
    Temperature anneals slowly; uphill steps are occasionally accepted.
    Returns the best pair (a, b) observed before the deadline.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()
    best_b = b.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.02 + 1e-9      # initial temperature
    cooling = 0.9995
    iter_counter = 0
    recompute_interval = 5          # recompute influence every few steps

    while time.time() < deadline:
        # ---- recompute influence of each position on the current peaks ----
        if iter_counter % recompute_interval == 0:
            max_val = cur_score
            # indices where the convolution attains (or almost attains) the max
            K = np.where(b >= max_val - 1e-12)[0]
            h = np.zeros(b.shape, dtype=float)
            h[K] = 1.0

            size = _next_pow2(n + (2 * n - 1) - 1)         # 3n‑2 (next pow2)
            a_fft = np.fft.rfft(a, n=size)
            h_fft = np.fft.rfft(h, n=size)
            # correlation a ⋆ h (linear, not circular)
            corr = np.fft.irfft(a_fft * np.conj(h_fft), n=size)[:n]
            influence = corr                               # ∂max/∂a_i up to a factor 2
        # ---------------------------------------------------------------
        # pick a source index – we want a large *influence* entry
        src_score = a * influence
        # avoid zero‑mass positions
        src_score[a <= 1e-15] = -np.inf
        i = int(np.argmax(src_score))
        if a[i] <= 0.0:                     # fallback (should be rare)
            candidates = np.where(a > 0)[0]
            if candidates.size == 0:
                break
            i = int(rng.choice(candidates))

        # amount of mass to move (up to 50 % of source)
        max_delta = a[i] * 0.5
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # pre‑compute the linear shift due to removing δ from i
        shift = 2.0 * delta * a
        b_base = b.copy()
        b_base[i:i + n] -= shift

        # try a few random destinations, keep the best improvement
        best_j = None
        best_new_score = cur_score
        n_trials = 5
        for _ in range(n_trials):
            j = rng.integers(0, n)
            if j == i:
                continue
            b_new = b_base.copy()
            b_new[j:j + n] += shift

            k2i = 2 * i
            k2j = 2 * j
            kij = i + j
            if k2i < b_new.size:
                b_new[k2i] += delta * delta
            if k2j < b_new.size:
                b_new[k2j] += delta * delta
            if kij < b_new.size:
                b_new[kij] -= 2.0 * delta * delta

            new_score = b_new.max()
            if new_score < best_new_score:
                best_new_score = new_score
                best_j = j

        # decide whether to accept the move
        accept = False
        if best_new_score < cur_score:
            accept = True
            j = best_j
        else:
            # uphill acceptance
            if T > 0.0 and rng.random() < np.exp(-(best_new_score - cur_score) / T):
                accept = True
                j = rng.integers(0, n)   # any destination is fine for uphill

        if accept:
            # apply the chosen move to a and b
            a[i] -= delta
            a[j] += delta
            # keep a numerically non‑negative
            a[i] = max(a[i], 0.0)
            a[j] = max(a[j], 0.0)

            # rebuild b from the base version (which already had the source removal)
            b = b_base.copy()
            b[j:j + n] += shift
            # quadratic corrections for the chosen destination
            k2i = 2 * i
            k2j = 2 * j
            kij = i + j
            if k2i < b.size:
                b[k2i] += delta * delta
            if k2j < b.size:
                b[k2j] += delta * delta
            if kij < b.size:
                b[kij] -= 2.0 * delta * delta

            cur_score = best_new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                best_b = b.copy()
                print(f"[{time.time():.1f}] targeted HC improvement → {best_score:.6f}")

        T *= cooling
        iter_counter += 1

    return best_a, best_b

            (1.1, 0.020),
        ]
        for p, lam in param_grid:
            a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)
            a = _project_to_simplex(a, target_sum)
            candidates.append(a)

        # a couple of purely random starts
        for _ in range(2):
            candidates.append(_random_initial_sequence(n, rng))

        # --------------------------------------------------------------
        # Evaluate each candidate: gradient descent + targeted hill‑climb
        # --------------------------------------------------------------
        for cid, a_start in enumerate(candidates):
            if time.time() >= deadline:
                break

            # ------- projected gradient descent (smooth max) -------
            # allocate at most 30 s or 30 % of the remaining budget
            remain = deadline - time.time()
            gd_time = min(30.0, remain * 0.3)
            gd_end = time.time() + gd_time
            a_gd = _gradient_descent_opt(a_start, gd_end)

            # evaluate
            b_full = _full_convolution(a_gd)
            cur_score = b_full.max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a_gd.copy()
                print(f"[{time.time():.1f}] n={n} cand={cid} GD improvement: {cur_score:.6f}")

            # ------- targeted hill climbing (exact max) -------
            if time.time() < deadline:
                remain = deadline - time.time()
                hc_time = min(25.0, remain * 0.2)
                hc_end = time.time() + hc_time
                a_hc, b_hc = _targeted_hill_climb(a_gd, b_full, hc_end)
                cur_score = b_hc.max()
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a_hc.copy()
                    print(f"[{time.time():.1f}] n={n} cand={cid} HC improvement: {cur_score:.6f}")

        # --------------------------------------------------------------
        # One more short polishing on the current best for this length
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size == n and time.time() < deadline - 5.0:
            a_cur = best_seq.copy()
            b_cur = _full_convolution(a_cur)
            remain = deadline - time.time()
            extra_hc = min(20.0, remain * 0.15)
            a_pol, b_pol = _targeted_hill_climb(a_cur, b_cur, time.time() + extra_hc)
            cur_score = b_pol.max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a_pol.copy()
                print(f"[{time.time():.1f}] n={n} final polishing: {cur_score:.6f}")

    # ------------------------------------------------------------------
    # Final clean‑up: remove trailing near‑zeros, enforce simplex one more time
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform distribution (should never happen)
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    best_seq[best_seq < 0] = 0.0
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # final projection onto the correct ℓ¹‑simplex
    best_seq = _project_to_simplex(best_seq, np.sqrt(2.0 * best_seq.size))

    # ------------------------------------------------------------------
    # Report the official evaluation value
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
