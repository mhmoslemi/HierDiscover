def _random_initial_sequence(n, rng):
    target_sum = np.sqrt(2 * n)
    # choose family uniformly
    family = rng.integers(0, 7)
    if family == 0:   # dense Dirichlet
        a = rng.dirichlet(alpha=np.ones(n))
    elif family == 1: # power-law decreasing
        p = rng.uniform(0.15, 1.3)
        idx = np.arange(n) + 1
        a = idx ** (-p)
    elif family == 2: # exponentials
        lam = rng.exponential(scale=0.02)
        idx = np.arange(n)
        a = np.exp(-lam * idx)
    elif family == 3: # mixed
        p = rng.uniform(0.15, 1.3)
        lam = rng.exponential(scale=0.02)
        idx = np.arange(n) + 1
        a = idx ** (-p) * np.exp(-lam * idx)
    elif family == 4: # Beta
        alpha = rng.uniform(0.2, 2.0)
        beta = rng.uniform(0.2, 2.0)
        x = (np.arange(n) + 0.5) / n
        a = x ** (alpha - 1) * (1 - x) ** (beta - 1)
    elif family == 5: # Sparse random
        # select k ~ sqrt(2n) * uniform(0.5, 2)

        a = np.zeros(n)

        a = np.full(n, 1.0 / n)

    # ensure non-negatives
    a = np.maximum(a, 0.0)

    # project to simplex
    a = _project_to_simplex(a, target_sum)

"""
Propose a step‑function sequence that minimizes the maximal self‑convolution.
The search consists of:

1.  Diverse random initialisations on the ℓ¹‑simplex (Σa_i = √(2 n)).
    Families include Dirichlet, power‑law, exponential, Beta‑shaped and a
    sparse “few‑spikes” construction.

2.  Adam‑based optimisation of a smooth surrogate of the max‑convolution:
    L_β(a)= (1/β)·log Σ_k exp(β·(a*a)_k).  The gradient is
        ∇_a L_β = 2·(a ⋆ p)  with  p_k ∝ exp(β·(a*a)_k).
    Variables are kept in log‑space to enforce non‑negativity, and after
    every update the vector is projected back onto the ℓ¹‑simplex.

3.  The best candidate seen (according to the exact max‑convolution) is
    returned.  The routine respects the supplied time budget (default 1000 s).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex Σ x_i = s,  x_i ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x_i≥0, Σx_i = s}."""
    if s <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp surrogate of the max‑convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b) where
        b = a*a (full convolution, length 2·n‑1)
        g = ∇_a  ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a ⋆ p)   with   p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]

    # ----- b = a*a -------------------------------------------------------
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ----- soft‑max probabilities ---------------------------------------
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))          # stability trick
    p = exps / np.sum(exps)

    # ----- gradient g = 2·(a ⋆ p) --------------------------------------
    a_rev = a[::-1]                            # reversed a
    conv_len = n + (2 * n - 1) - 1              # = 3·n‑2
    size2 = 1 << ((conv_len).bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]        # central n entries
    return g, b


# ----------------------------------------------------------------------
# Adam optimiser for the smoothed max‑convolution objective
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray, deadline: float, rng: np.random.Generator) -> np.ndarray:
    """
    Adam optimisation of the log‑sum‑exp surrogate.
    The iterate stays on the ℓ¹‑simplex (sum = √(2 n)) and on R⁺ⁿ.
    Returns the best (lowest true max‑convolution) sequence seen before the deadline.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    eps = 1e-12
    x = np.log(np.maximum(a, eps))            # log‑space variables

    beta = 1.0
    beta_max = 1e5

    # Adam hyper‑parameters
    lr = 0.2
    beta1 = 0.9
    beta2 = 0.999
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    step = 0
    best_a = a.copy()
    best_score = _max_convolution_fft(a)

    while time.time() < deadline:
        # ---------- reconstruct a from log‑variables ----------
        a = np.exp(x)
        s = np.sum(a)
        if s > 0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # ---------- gradient of smooth max ----------
        g_a, _ = _gradient_softmax(a, beta)
        grad_x = a * g_a                     # chain rule (∂L/∂x_i = a_i * ∂L/∂a_i)

        step += 1

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        x = x - lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # ---------- anneal smoothing parameter ----------
        if step % 300 == 0 and beta < beta_max:
            beta = min(beta * 1.3, beta_max)

        # ---------- occasional exact evaluation ----------
        if step % 200 == 0:
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] Adam improvement (n={n}) step {step} score {cur_score:.6f}")
                if best_score <= 1.503:      # target reached – stop early
                    deadline = time.time()

        # ---------- occasional jitter to escape plateaus ----------
        if step % 1500 == 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = a * (target_sum / np.sum(a))
            x = np.log(np.maximum(a, eps))

    return best_a


# ----------------------------------------------------------------------
# Random initialiser – diverse families, including a sparse one
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random non‑negative sequence on the ℓ¹‑simplex."""
    target_sum = np.sqrt(2.0 * n)
    family = rng.integers(0, 7)
    idx = np.arange(n, dtype=float)

    if family == 0:                                  # Dirichlet (dense uniform)
        a = rng.dirichlet(alpha=np.ones(n))

    elif family == 1:                                # power‑law decreasing
        p = rng.uniform(0.15, 1.2)
        a = (idx + 1.0) ** (-p)

    elif family == 2:                                # exponential decay
        lam = rng.exponential(scale=0.02)
        a = np.exp(-lam * idx)

    elif family == 3:                                # mixed power‑law + exponential
        p = rng.uniform(0.15, 1.2)
        lam = rng.exponential(scale=0.02)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 4:                                # Beta‑shaped (U or bell)
        alpha = rng.uniform(0.2, 2.0)
        beta = rng.uniform(0.2, 2.0)
        x = (idx + 0.5) / n
        a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))

    elif family == 5:                                # symmetric edge‑heavy
        p = rng.uniform(0.2, 1.2)
        left = (idx + 1.0) ** (-p)
        right = (n - idx) ** (-p)
        a = left + right

    else:                                            # sparse “few spikes” construction
        # target number of non‑zero entries ≈ √(2 n) scaled by a factor in [0.5, 2]
        k = max(2, int(rng.uniform(0.5, 2.0) * np.sqrt(2.0 * n)))
        positions = rng.choice(n, size=k, replace=False)
        weights = rng.dirichlet(alpha=np.ones(k))
        a = np.zeros(n)
        a[positions] = weights

    # small Gaussian perturbation & projection to the simplex
    a += rng.normal(scale=target_sum * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target_sum)
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Start from the globally stored construction if it exists
    # --------------------------------------------------------------
    global_seq = globals().get("height_sequence_1")
    if isinstance(global_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(global_seq, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        sc = _max_convolution_fft(a0)
        best_seq, best_score = a0, sc
        print(f"[{time.time():.1f}] start from height_sequence_1 (n={n0}) score={sc:.6f}")

    # --------------------------------------------------------------
    # Define a set of problem sizes to explore
    # --------------------------------------------------------------
    lengths = [800, 1200, 2000, 4000, 8000, 12000]
    # make sure we stay within the clock budget
    remaining_len = len(lengths)
    while lengths and time.time() < deadline:
        n = lengths.pop(0)
        # allocate an (approx.) equal share of the remaining budget
        time_left = deadline - time.time()
        if time_left < 10.0:
            break
        per_len_deadline = time.time() + time_left / max(1, remaining_len)
        remaining_len -= 1

        # ----------------------------------------------------------
        # Run several independent Adam‑optimisations for this length
        # ----------------------------------------------------------
        attempts = 0
        while time.time() < per_len_deadline and attempts < 30:
            a_init = _random_initial_sequence(n, rng)
            # give each Adam run a modest chunk of the remaining time
            sub_deadline = min(per_len_deadline, time.time() + 7.0)
            a_opt = _adam_optimize(a_init, sub_deadline, rng)
            cur_score = _max_convolution_fft(a_opt)
            if cur_score < best_score:
                best_seq, best_score = a_opt.copy(), cur_score
                print(f"[{time.time():.1f}] NEW GLOBAL BEST (n={n}) score={cur_score:.6f}")
                if best_score <= 1.503:      # early exit if target reached
                    deadline = time.time()
            attempts += 1

    # --------------------------------------------------------------
    # Fallback: uniform distribution if nothing was found
    # --------------------------------------------------------------
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = _max_convolution_fft(best_seq)

    # --------------------------------------------------------------
    # Trim trailing near‑zero entries and return as plain list
    # --------------------------------------------------------------
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # sanity check with the official evaluator (optional)
    try:
        ev = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence = {ev:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best max‑conv = {best_score:.6f}  length = {best_seq.size}")
    return best_seq.tolist()
