"""
Step‑function optimizer – enhanced 2‑phase descent + aggressive
targeted mass‑balancing + greedy hill‑climbing.

* Phase 1 – smooth‑max (log‑sum‑exp) projected gradient descent.
  Same as the reference solution but with a slightly larger step
  base and a faster β‑annealing schedule.

* Phase 2 – exact sub‑gradient descent on the true max‑norm.

* Phase 3 – targeted mass‑balance moves:
  the current maximal convolution entry k is located.
  The index i that contributes most to that entry (max a_i·a_{k‑i})
  transfers a modest fraction of its mass to the index j that
  minimises a_{k‑j}.  This explicitly reduces the peak while keeping
  the ℓ¹–norm unchanged.

* Phase 4 – greedy hill‑climbing:
  random pairwise mass transfers are accepted only if they lower the
  maximal convolution value.  This quickly exploits any remaining
  slack.

* Phase 5 – a short simulated‑annealing polish.

All phases respect non‑negativity and the ℓ¹‑simplex
{a_i≥0, Σa_i=√(2 n)}; the objective after this normalisation is simply
the maximal convolution entry.  The routine explores several lengths
(2000 … 10000) with a few random restarts, keeps the best candidate and
returns it as a plain list before the 1000 s wall‑clock budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1, dtype=float)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·n−1) using an FFT.
    Returned array is real‑valued.
    """
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Objective after ℓ¹‑normalisation Σa = √(2 n):
    simply the supremum norm of the self‑convolution.
    """
    return float(_max_convolution_fft(a).max())


# ----------------------------------------------------------------------
# Soft‑max gradient (log‑sum‑exp) – Phase 1
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) where
        g = ∇_a  ( 1/β·log Σ exp(β·b) )
          = 2·(a ⋆ p)          with   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # ---- b = a*a -------------------------------------------------------
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ---- soft‑max probabilities p --------------------------------------
    max_b = float(np.max(b))
    exps = np.exp(beta * (b - max_b))          # shift for numerical stability
    p = exps / np.sum(exps)

    # ---- gradient g = 2·(a ⋆ p) ----------------------------------------
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1            # = 3 n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _smooth_max_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Annealed projected gradient descent on the smooth max.
    Uses momentum, an adaptive step and occasional Gaussian jitter.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 8192.0
    beta_factor = 1.5

    step_base = target_sum * 0.20          # a slightly larger step fraction
    momentum = np.zeros_like(a)
    rho = 0.90                               # momentum decay

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        # simplex projection (keeps Σa = target_sum)
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal β
        if it % 400 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a


# ----------------------------------------------------------------------
# Exact max‑index sub‑gradient – Phase 2
# ----------------------------------------------------------------------


def _max_index_subgradient(a: np.ndarray):
    """
    Gradient of max(b) where b = a*a.
    Returns (g, k) where k = argmax b and
        g_i = 2·a_{k‑i}  (zero if index out of range).
    """
    n = a.shape[0]
    conv = _max_convolution_fft(a)
    k = int(np.argmax(conv))

    g = np.zeros_like(a)
    # fill gradient according to the reflected a
    if k < n:                         # left half of the convolution
        g[:k + 1] = 2.0 * a[k::-1]
    else:                             # right half
        d = k - (n - 1)               # shift amount (0 … n‑1)
        g[d:] = 2.0 * a[: n - d]
    return g, k


def _subgradient_descent_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected sub‑gradient descent on the true max‑norm (no smoothing).
    A diminishing step‑size  η_t = α / √t is used.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        g, _ = _max_index_subgradient(a)

        # diminishing step size
        step = 0.7 * target_sum / (np.sqrt(it + 1) * np.linalg.norm(g) + 1e-12)
        a -= step * g

        # enforce feasibility
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        # tiny jitter to avoid cycling
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 4e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1
    return a


# ----------------------------------------------------------------------
# Random initialisation helpers
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative point on the simplex Σa = √(2 n).
    Several families (power‑law, exponential, Dirichlet, piecewise) are
    sampled to increase diversity.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                                 # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                               # pure exponential
        lam = rng.exponential(scale=0.02)
        a = np.exp(-lam * idx)

    elif family == 2:                               # power × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.02)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                               # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n))

    elif family == 4:                               # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                             np.exp(-lam2 * np.arange(b, n))])

    else:                                           # almost uniform with jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


# ----------------------------------------------------------------------
# Simple simulated‑annealing (kept from the reference solution)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Very light SA that moves a random fraction of mass between two sites."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    cur_score = _score(a)
    best_score = cur_score

    T = cur_score * 0.02 + 1e-9
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-16:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        frac = rng.random() * 0.15
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta

        new_score = _score(a)
        dE = new_score - cur_score
        accept = dE <= 0.0 or (T > 0.0 and rng.random() < np.exp(-dE / T))

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling

    return best


# ----------------------------------------------------------------------
# New Phase 3 – Targeted mass‑balance
# ----------------------------------------------------------------------


def _targeted_mass_balance(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Aggressive greedy mass transfer that directly attacks the currently
    maximal convolution entry.  In each iteration the index i that
    contributes most to that entry loses a fraction of its mass, which
    is given to the index j that minimises a_{k‑j}.  The move reduces
    the peak by ≈2·Δ·(a_{k‑i}−a_{k‑j}).
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    rng = np.random.default_rng()

    # start with a relatively large transfer fraction and decay it
    frac = 0.08
    no_improve = 0
    prev_peak = np.inf

    while time.time() < deadline:
        conv = _max_convolution_fft(a)
        k = int(np.argmax(conv))

        # contributions to the peak entry
        idx = np.arange(n)
        other = k - idx
        mask = (other >= 0) & (other < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[other[mask]]

        i_high = int(np.argmax(contrib))

        # choose destination that minimises a_{k‑j}
        a_kj = np.full(n, np.inf)
        a_kj[mask] = a[other[mask]]
        j_low = int(np.argmin(a_kj))

        # transfer a modest fraction of the heavy mass
        delta = a[i_high] * frac
        if delta < 1e-12:
            break
        a[i_high] -= delta
        a[j_low] += delta

        # re‑project to eliminate tiny numerical drift
        a = _project_to_simplex(a, target_sum)

        # monitor improvement
        new_peak = conv.max()
        if new_peak >= prev_peak - 1e-9:
            no_improve += 1
        else:
            no_improve = 0
            prev_peak = new_peak

        # decay the transfer fraction a little over time
        if no_improve % 30 == 0:
            frac = max(frac * 0.9, 0.01)

        if no_improve > 120:        # stagnation → stop
            break

    return a


# ----------------------------------------------------------------------
# New Phase 4 – Greedy hill‑climbing
# ----------------------------------------------------------------------


def _greedy_hill_climb(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Random pairwise mass moves that are kept only when they lower the
    maximal convolution entry.  The step size is a random fraction
    (up to 20 %) of the source mass.
    """
    a = a0.copy()
    n = a.shape[0]
    best_val = _score(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        # pick a source with non‑trivial mass
        src = rng.integers(0, n)
        if a[src] <= 1e-14:
            continue
        dst = rng.integers(0, n)
        if src == dst:
            continue

        frac = rng.random() * 0.20
        delta = a[src] * frac
        if delta < 1e-12:
            continue

        a[src] -= delta
        a[dst] += delta

        new_val = _score(a)
        if new_val < best_val:
            best_val = new_val
            # keep the move
        else:
            # revert
            a[dst] -= delta
            a[src] += delta

        it += 1
        # occasional random jitter to shake off flat regions
        if it % 500 == 0:
            jitter = rng.normal(scale=best_val * 1e-3, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, np.sqrt(2.0 * n))

    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` metric.  The routine respects the wall‑clock
    budget, explores several lengths, applies the multi‑phase descent
    described above and finally hands the best sequence back as a plain
    Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    def maybe_update(seq: np.ndarray):
        nonlocal best_seq, best_score
        sc = _score(seq)
        if sc < best_score - 1e-9:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] New global best score {sc:.6f} len={seq.size}")

    # ------------------------------------------------------------------
    # Seed from a previously discovered construction if available
    # ------------------------------------------------------------------
    earlier = globals().get("height_sequence_1")
    if isinstance(earlier, (list, tuple, np.ndarray)):
        a0 = np.asarray(earlier, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        maybe_update(a0)

    # ------------------------------------------------------------------
    # Lengths to investigate – larger n gives more flexibility
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000, 10000]

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)

        # a few random restarts per length
        restarts = 3
        for _ in range(restarts):
            if time.time() >= deadline:
                break

            a = _random_initial_sequence(n, rng)

            # Phase 1 – short smooth‑max GD
            sub_dead = min(deadline, time.time() + 2.5)
            a = _smooth_max_descent(a, sub_dead)
            maybe_update(a)

            # Phase 2 – sub‑gradient descent
            sub_dead = min(deadline, time.time() + 5.0)
            a = _subgradient_descent_max(a, sub_dead)
            maybe_update(a)

            # Phase 3 – targeted mass‑balance
            sub_dead = min(deadline, time.time() + 5.0)
            a = _targeted_mass_balance(a, sub_dead)
            maybe_update(a)

            # Phase 4 – greedy hill‑climbing
            sub_dead = min(deadline, time.time() + 5.0)
            a = _greedy_hill_climb(a, sub_dead)
            maybe_update(a)

    # ------------------------------------------------------------------
    # Final simulated‑annealing polishing with whatever time is left
    # ------------------------------------------------------------------
    if best_seq is not None:
        left = deadline - time.time()
        if left > 8.0:
            a_final = _simulated_annealing(best_seq, deadline)
            maybe_update(a_final)

    # ------------------------------------------------------------------
    # Clean up tiny trailing zeros and report the official evaluation
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution of length 2000
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
