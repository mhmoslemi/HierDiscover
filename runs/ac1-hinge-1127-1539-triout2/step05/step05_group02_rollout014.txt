def _adam_subgradient_descent(a0, deadline):
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)
    # parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1   # base learning rate
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = _score(a)
    while time.time() < deadline:
        # compute conv, max, gradient
        conv = _conv_fft(a) # returns full length
        max_val = conv.max()
        k = int(np.argmax(conv))
        # compute subgradient
        g = np.zeros_like(a)
        # for i where j=k-i in range
        # compute j = k - i
        # Python loop vectorized: indices_i = np.arange(n); j = k - indices_i; mask = (j>=0) & (j<n); g[mask] = 2*a[j[mask]]
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g[mask] = 2.0 * a[j[mask]]

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # project back onto simplex (ensuring nonnegativity)
        a = _project_to_simplex(a, target)

        # evaluate score
        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] Adam iter={t} n={n} best={best_score:.6f}")

        # Possibly adjust learning rate if stagnating
        # maybe decay lr as time passes
        if t % 5000 == 0:
            lr *= 0.9

    return best_a, best_score

def _adam_subgradient(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    target = np.sqrt(2.0 * n)

    beta1=0.9; beta2=0.999; eps=1e-8; lr=0.05

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _score(a)

    it = 0
    while time.time() < deadline:
        # compute conv via FFT
        conv = _conv_fft(a)   # vector length 2n-1
        max_val = conv.max()
        k = int(np.argmax(conv))

        # compute subgradient g = 2 * a shifted by k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # Adam update
        it += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_to_simplex(a, target)

        # evaluate score
        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] Adam n={n} iter={it} score={best_score:.6f}")

        # occasional jitter:
        if it % 5000 == 0:
            jitter_std = target * 0.001 / (it**0.5 + 1)
            a += rng.normal(scale=jitter_std, size=n)
            a = np.maximum(a, 0)
            a = _project_to_simplex(a, target)

        # optionally adjust lr (decay)
        if it % 2000 == 0:
            lr *= 0.9

    return best_a, best_score

```python
"""
Optimisation of a non‑negative step‑function to minimise the benchmark
``evaluate_sequence``.  The search builds several candidates of different
lengths, initialises them with simple power‑law or random patterns, and
optimises each with a projected Adam sub‑gradient descent on the exact
maximum of the self‑convolution.  The iterate is always projected onto the
ℓ¹‑simplex (∑aᵢ=√(2·n)) so the scaling does not affect the objective.
The best sequence found before the time‑budget expires is returned.
"""

import time
import numpy as np

def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())

def _score(a: np.ndarray) -> float:
    """Exact objective used by ``evaluate_sequence`` (fast version)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))

def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _adam_subgradient(a0: np.ndarray, deadline: float, rng: np.random.Generator):
    """
    Adam optimiser applied to the exact sub‑gradient of
    ``max_k (a*a)[k]``.  After each update the vector is projected onto the
    ℓ¹‑simplex to keep the scaling invariant.
    """
    a = a0.copy()
    n = a.shape[0]
    target = np.sqrt(2.0 * n)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_val = _score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # compute the convolution and locate the current worst index
        conv = _conv_fft(a)
        k = int(np.argmax(conv))

        # exact sub‑gradient of max(conv) w.r.t a:
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_to_simplex(a, target)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] Adam n={n} iter={it} score={best_val:.6f}")

        # occasional jitter to escape plateaus

"""
Optimisation of a non‑negative step‑function to minimise the benchmark
``evaluate_sequence``.  The search builds several candidates of different
lengths, initialises them with simple power‑law or random patterns, and
optimises each with a projected Adam sub‑gradient descent on the exact
maximum of the self‑convolution.  The iterate is always projected onto the
ℓ¹‑simplex (∑aᵢ = √(2·n)) so the scaling does not affect the objective.
The best sequence found before the time‑budget expires is returned.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT (real‑valued input)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())       # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Fast replica of the benchmark objective (no safety checks)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σ x_i = s}.  Implements Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _adam_subgradient(a0: np.ndarray, deadline: float,
                      rng: np.random.Generator):
    """
    Adam optimiser applied to the exact sub‑gradient of
    ``max_k (a*a)[k]``.  After each update the vector is projected onto the
    ℓ¹‑simplex to keep the scaling invariant.
    """
    a = a0.copy()
    n = a.shape[0]
    target = np.sqrt(2.0 * n)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_val = _score(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # convolution and index of the current maximum entry
        conv = _conv_fft(a)
        k = int(np.argmax(conv))

        # exact sub‑gradient of max(conv) w.r.t. a:
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_to_simplex(a, target)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] Adam n={n} iter={it} score={best_val:.6f}")

        # occasional jitter to escape plateaus
        if it % 5000 == 0:
            jitter = rng.normal(scale=target * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

        # simple learning‑rate decay
        if it % 2000 == 0:
            lr *= 0.9

    return best_a, best_val

# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """

    """
