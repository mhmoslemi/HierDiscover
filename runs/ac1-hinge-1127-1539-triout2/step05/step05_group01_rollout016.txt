"""
search for a non‑negative step‑function that minimises
``evaluate_sequence``.  The routine works in three phases:

1. **Broad parametric sweep** – power‑law, symmetric power‑law,
   exponential and symmetric exponential families are sampled on a
   geometric grid of lengths (up to 32 k).  The 10 best candidates
   (lowest evaluate_sequence value) are retained.

2. **Projected gradient descent** – each elite seed is refined by
   minimising a smooth log‑sum‑exp surrogate of the maximal convolution
   entry.  An annealed momentum scheme keeps the iterates on the
   ℓ¹‑simplex Σa_i=√(2 n).

3. **Simulated‑annealing fine‑tuning** – the refined vectors are further
   improved by a random mass‑transfer SA with multiplicative cooling.
   Finally a short hill‑climb finishes the optimisation.

All phases respect a user‑supplied time budget (default 1000 s) and
never perform I/O.  The returned list is already trimmed of
negligible trailing entries.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the full linear convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex Σx = s, x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection onto {x_i ≥ 0, Σx_i = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    if s <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) surrogate of max(b) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (grad, b) where
        b = a*a (full convolution)
        grad = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
              = 2·(a ⋆ p) ,  p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]

    # ---- convolution a*a (full) ----
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # ---- soft‑max probabilities ----
    max_b = np.max(b)
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)

    # ---- gradient g = 2·(a ⋆ p_rev) ----
    # we need convolution of a with reversed p
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    ar_fft = np.fft.rfft(a_rev, n=size2)
    pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(ar_fft * pr_fft, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]   # keep middle n entries
    return grad, b


# ----------------------------------------------------------------------
# Projected gradient descent on the smoothed max
# ----------------------------------------------------------------------


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Annealed projected gradient descent on log‑sum‑exp smooth max."""
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    beta = 1.0
    max_beta = 1e7
    momentum = np.zeros_like(a)
    rho = 0.9                     # momentum decay
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum step
        momentum = rho * momentum + (1.0 - rho) * g
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            # step size inversely proportional to the current smoothness
            lr = 0.5 * target_sum / (norm * (beta + 1.0))
            a -= lr * momentum

        # simplex projection
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 6e-4 / (beta + 1.0),
                               size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # exponential annealing of β
        if it % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.07, max_beta)

        it += 1
    return a


# ----------------------------------------------------------------------
# Simple simulated‑annealing refinement
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Random mass‑transfer SA with multiplicative cooling."""
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # initialise temperature from current objective
    T = best_score * 0.02 + 1e-9
    cooling = 0.9994
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # move a random fraction (up to 12 %) of mass from i to j
        frac = rng.random() * 0.12
        delta = a[i] * frac
        if delta <= 0.0:
            continue
        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        accept = False
        if cur_score <= best_score:
            accept = True
        elif T > 0.0 and rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
    return best


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    """a_i = (i+shift)^(-p)  , projected onto Σa = √(2 n)."""
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    target = np.sqrt(2.0 * n)
    a = raw * (target / raw.sum())
    return np.maximum(a, 0.0)


def _symmetric_powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    """a_i = (i+shift)^(-p) + (n-1-i+shift)^(-p), then projected."""
    idx = np.arange(n, dtype=float)
    left = (idx + shift) ** (-p)
    right = (n - 1 - idx + shift) ** (-p)
    raw = left + right
    target = np.sqrt(2.0 * n)
    a = raw * (target / raw.sum())
    return np.maximum(a, 0.0)


def _exponential_sequence(n: int, lam: float, shift: float) -> np.ndarray:
    """a_i = exp(-lam·(i+shift)), projected onto Σa = √(2 n)."""
    idx = np.arange(n, dtype=float)
    raw = np.exp(-lam * (idx + shift))
    target = np.sqrt(2.0 * n)
    a = raw * (target / raw.sum())
    return np.maximum(a, 0.0)


def _symmetric_exponential_sequence(n: int, lam: float, shift: float) -> np.ndarray:
    """a_i = exp(-lam·i) + exp(-lam·(n‑1‑i)), projected."""
    idx = np.arange(n, dtype=float)
    left = np.exp(-lam * (idx + shift))
    right = np.exp(-lam * (n - 1 - idx + shift))
    raw = left + right
    target = np.sqrt(2.0 * n)
    a = raw * (target / raw.sum())
    return np.maximum(a, 0.0)

            best_eval = eval_val
            best_seq = a_sa.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL BEST (refined)  n={n}  eval={best_eval:.6f}")

    # ------------------------------------------------------------------
    # 3. Final local hill‑climbing (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a = best_seq.copy()
        n = a.size
        cur_max = _max_convolution_fft(a)
        cur_eval = 2.0 * n * cur_max / (np.sum(a) ** 2)
        while time.time() < deadline:
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] <= 1e-16:
                continue
            frac = rng.random() * 0.08
            delta = a[i] * frac
            a[i] -= delta
            a[j] += delta
            new_max = _max_convolution_fft(a)
            new_eval = 2.0 * n * new_max / (np.sum(a) ** 2)
            if new_eval < cur_eval:
                cur_eval = new_eval
                cur_max = new_max
                best_seq = a.copy()
            else:
                # revert
                a[j] -= delta
                a[i] += delta

    # ------------------------------------------------------------------
    # 4. Trim negligible tail and final sanity check
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
    else:
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[: nz[-1] + 1]

    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.1f}] Evaluation error: {exc}")

    print(f"[{time.time():.1f}] Finished – best eval = {best_eval:.6f}  length = {best_seq.size}")
    return best_seq.tolist()
