def _subgradient_descent(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_score = _score(a)
    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(conv.argmax())
        # compute sensitivities
        sens = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]
        grad = 2.0 * sens
        # step size
        # maybe base step = target_sum * 0.2 / (np.linalg.norm(grad) + 1e-12)
        step = (target_sum * 0.3) / (np.linalg.norm(grad) + 1e-12)
        a = a - step * grad
        a = _project_to_simplex(a, target_sum)
        # evaluate
        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # jitter occasionally
        if it % 200 == 0:
            # add small noise and renormalize to keep sum
            noise = np.random.default_rng().normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + noise, 0)
            a = a * (target_sum / a.sum())
        it += 1
    return best_a, best_score
```

def _softmax_gradient_descent(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_val = _score(a)
    beta = 1.0
    max_beta = 4096.0
    rng = np.random.default_rng()
    step_base = target_sum * 0.15  # smaller step for stability
    it = 0
    while time.time() < deadline:
        # compute gradient using softmax approx
        # compute convolution via FFT
        size = 1 << ((2*n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
        max_b = np.max(conv)
        exps = np.exp(beta * (conv - max_b))
        p = exps / np.sum(exps)
        # compute gradient of weighted sum w.r.t. a
        # as in earlier code: gradient = 2 * conv(p) with reversed a
        a_rev = a[::-1]
        Pr_fft = np.fft.rfft(p, n=size)
        Ar_fft = np.fft.rfft(a_rev, n=size)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[:2*n-1]
        grad = 2.0 * conv_rev[n-1 : n-1 + n]
        # step
        normg = np.linalg.norm(grad)
        if normg > 0:
            lr = step_base / (normg * (beta + 1.0))
            a = a - lr * grad
        # projection
        a = _project_to_simplex(a, target_sum)
        # jitter occasionally
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)
        # evaluate
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] GD improve beta={beta:.1f} score={best_val:.6f}")
        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)
        it += 1
    return best_a, best_val

def _simulated_annealing(a0, deadline,
                         start_temp=1.0,
                         end_temp=1e-7,
                         max_frac_start=0.3,
                         max_frac_end=0.001,
                         max_window=40):
    a = a0.copy()
    n = a.shape[0]
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)
    size_fft = 1 << ((2*n - 1).bit_length())
    rng = np.random.default_rng()
    conv = _conv_fft(a)
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # locate max convolution index
        k_max = int(conv.argmax())
        # compute sensitivity vector for current max entry
        sens = np.zeros(n, dtype=np.float64)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]   # product contributions
        
        # weight for selecting donor is a * sens (higher mass & high contribution)
        donor_weight = a * sens
        move_type = None
        # Choose move type
        r = rng.random()
        if r < 0.6:  # mass transfer
            total_w = donor_weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=donor_weight / total_w)
            else:
                donor = rng.integers(n)
            # choose receiver among low-weight indices
            low_weight = donor_weight.copy()
            # maybe select from indices with small weight
            low_thr = np.percentile(donor_weight, 20)
            low_candidates = np.where(donor_weight <= low_thr)[0]
            if low_candidates.size > 0:
                receiver = rng.choice(low_candidates)
            else:
                receiver = rng.integers(n)
            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"
        elif r < 0.85:  # local smoothing
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                seg = a[start:end].copy()
                avg = seg.mean()
                a[start:end] = avg
                move_type = "smooth"
                saved_seg = seg
                saved_start, saved_end = start, end
        elif r < 0.95:  # symmetric transfer
            i = rng.integers(0, n // 2)
            j = n - 1 - i
            # choose direction based on which has larger contribution
            if a[i] > a[j]:
                donor, receiver = i, j
            else:
                donor, receiver = j, i
            delta = a[donor] * rng.random() * max_frac
            if delta > 0:
                a[donor] -= delta
                a[receiver] += delta
                move_type = "sym"
        else:  # random swap
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # evaluate new state
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[:2*n-1]
        new_score = _score(a)

        # accept?
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if temp > 0:
                prob = np.exp(-(new_score - cur_score) / temp)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA new best={best_score:.6f}")
        else:
            # revert move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "sym":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]
        # optional jitter
        if it % 4000 == 0 and it > 0:
            jitter = rng.normal(scale=cur_score * 0.0005, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            # renormalize sum to target
            target = np.sqrt(2.0 * n)
            a = a * (target / a.sum())
            conv = _conv_fft(a)
            cur_score = _score(a)
        it += 1
    return best_a, best_score

"""
Propose a non‑negative step‑function (height sequence) with a low
benchmark score defined by ``evaluate_sequence``.

The optimiser works in three stages:

1.  **Diversified initialisation** – several candidates are drawn
    (warm start, random Dirichlet, power‑law).  All candidates are
    projected onto the ℓ¹‑simplex Σa = √(2·n).

2.  **Soft‑max projected gradient descent** – a smooth approximation of
    the max‑convolution is minimised.  The temperature β is increased up
    to 4096, a simple back‑tracking step size is used and the iterates are
    kept feasible by Euclidean projection onto the simplex.

3.  **Simulated‑annealing refinement** – moves consist of
    *mass‑transfer* (guided by the sensitivity of the current maximal
    convolution entry), *local smoothing*, *symmetric transfer* (shifting
    mass between symmetric positions) and occasional *random swaps*.
    An exponential temperature schedule together with a decreasing
    maximal transfer fraction allows a thorough exploration.

The whole budget (default 1000 s) is split over several problem sizes
(1024, 1500, 2048, 3000, 4096).  The best sequence found over all runs is
returned (trailing near‑zero entries are stripped).  No external files,
network or multiprocessing are used – only NumPy and the provided
``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (real valued)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    return np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Fast objective for a normalised sum.
    When Σa = √(2·n) the score coincides with max convolution.
    """
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = s}.
    Implements Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _random_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform Dirichlet start, normalised to √(2·n)."""
    target = np.sqrt(2.0 * n)
    x = rng.random(n)
    return x / x.sum() * target


def _power_law_init(n: int, exponent: float, rng: np.random.Generator) -> np.ndarray:
    """Power‑law start a_i ∝ (i+1)^{‑exponent}, normalised."""
    target = np.sqrt(2.0 * n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-exponent)
    return a * target / a.sum()


def _warm_start_init(n: int) -> np.ndarray:
    """Use the globally supplied sequence if available."""
    if "height_sequence_1" in globals():
        seq = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if seq.size > n:
            seq = seq[:n]
        else:
            seq = np.pad(seq, (0, n - seq.size), "constant")
        target = np.sqrt(2.0 * n)
        return _project_to_simplex(seq, target)
    # fallback – random
    rng = np.random.default_rng()
    return _random_init(n, rng)


# ----------------------------------------------------------------------
# Soft‑max projected gradient descent (stage 2)
# ----------------------------------------------------------------------


def _softmax_gradient_descent(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on a soft‑max approximation of max(convolution).
    The temperature β is increased up to 4096.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 4096.0
    rng = np.random.default_rng()
    step_base = target_sum * 0.15
    it = 0

    while time.time() < deadline:
        size = 1 << ((2 * n - 1).bit_length())
        af = np.fft.rfft(a, n=size)

        conv = np.fft.irfft(af * af, n=size)[: 2 * n - 1]
        max_b = conv.max()

        # soft‑max weights
        exps = np.exp(beta * (conv - max_b))
        p = exps / exps.sum()

        # gradient w.r.t. a (see derivation in the analysis)
        a_rev = a[::-1]
        pr = np.fft.rfft(p, n=size)
        ar = np.fft.rfft(a_rev, n=size)
        conv_rev = np.fft.irfft(ar * pr, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # step
        norm_g = np.linalg.norm(grad)
        if norm_g > 0:
            lr = step_base / (norm_g * (beta + 1.0))
            a -= lr * grad

        # projection onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] GD improve β={beta:.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (stage 3)
# ----------------------------------------------------------------------


def _simulated_annealing(
    a0: np.ndarray,
    deadline: float,
    start_temp: float = 1.0,
    end_temp: float = 1e-7,
    max_frac_start: float = 0.30,
    max_frac_end: float = 0.001,
    max_window: int = 40,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing search with several move types.
    The move set is deliberately rich to escape local minima.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)
    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # --------------------------------------------------------------
        # identify the current max‑convolution entry and its sensitivities
        # --------------------------------------------------------------
        k_max = int(conv.argmax())
        sens = np.zeros(n, dtype=np.float64)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]   # contribution of a_i to b_kmax

        weight = a * sens   # a donor that heavily influences the peak

        # --------------------------------------------------------------
        # propose a move
        # --------------------------------------------------------------
        move_type = None
        r = rng.random()
        if r < 0.60:   # mass transfer (donor → receiver)
            total_w = weight.sum()
            donor = rng.choice(n, p=weight / total_w) if total_w > 0 else rng.integers(n)

            low_thr = np.percentile(weight, 20)
            low_cand = np.where(weight <= low_thr)[0]
            receiver = rng.choice(low_cand) if low_cand.size else rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"
        elif r < 0.80:   # local smoothing of a short window
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                seg = a[start:end].copy()
                avg = seg.mean()
                a[start:end] = avg
                move_type = "smooth"
                saved_seg = seg
                saved_start, saved_end = start, end
        elif r < 0.95:   # symmetric transfer (i ↔ n‑1‑i)
            i = rng.integers(0, n // 2)
            j = n - 1 - i
            donor, receiver = (i, j) if a[i] >= a[j] else (j, i)
            delta = a[donor] * rng.random() * max_frac
            if delta > 0:
                a[donor] -= delta
                a[receiver] += delta
                move_type = "sym"
        else:            # random swap of two positions
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # --------------------------------------------------------------
        # evaluate the new state (single FFT)
        # --------------------------------------------------------------
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = _score(a)

        # --------------------------------------------------------------
        # Metropolis acceptance
        # --------------------------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA new best = {best_score:.6f}")
        else:
            # revert the move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "sym":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter to overcome deep traps
        if it % 4000 == 0 and it > 0:
            jitter = rng.normal(scale=cur_score * 5e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            target = np.sqrt(2.0 * n)
            a = a * (target / a.sum())
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Single‑instance optimiser (one problem size)
# ----------------------------------------------------------------------


def _run_instance(n: int, mode: str, deadline: float) -> tuple[np.ndarray, float]:
    """
    Optimise a step‑function of length ``n``.
    ``mode`` can be 'warm', 'power' or 'random'.
    Returns the best sequence discovered and its corresponding score.
    """
    rng = np.random.default_rng()
    target = np.sqrt(2.0 * n)

    # ----- initial candidate -------------------------------------------------
    if mode == "warm":
        seq = _warm_start_init(n)
    elif mode == "power":
        seq = _power_law_init(n, exponent=0.55, rng=rng)
    else:  # random
        seq = _random_init(n, rng)

    seq = _project_to_simplex(seq, target)

    best_seq = seq.copy()
    best_val = _score(seq)

    # ----- gradient descent (soft‑max) --------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        gd_deadline = time.time() + min(20.0, remaining * 0.20)
        seq_gd, val_gd = _softmax_gradient_descent(seq, gd_deadline)
        if val_gd < best_val:
            best_val = val_gd
            best_seq = seq_gd.copy()

    # ----- simulated annealing ---------------------------------------------
    if time.time() < deadline:
        seq_sa, val_sa = _simulated_annealing(best_seq, deadline)
        if val_sa < best_val:
            best_val = val_sa
            best_seq = seq_sa.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the benchmark score.
    The routine explores several problem sizes, applies a short projected‑gradient
    phase followed by a long simulated‑annealing refinement and returns the
    best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 1.0   # safety margin

    best_seq = None
    best_score = np.inf

    # problem sizes to explore (powers of two are FFT‑friendly)
    sizes = [1024, 1500, 2048, 3000, 4096]

    # split the total budget roughly equally among the sizes
    per_size_budget = (deadline - start) / len(sizes)

    for n in sizes:
        if time.time() >= deadline:
            break

        # decide initialisation mode
        if "height_sequence_1" in globals() and n == len(height_sequence_1):
            mode = "warm"
        else:
            mode = "power" if rng.random() < 0.5 else "random"

        # optimise this instance
        size_deadline = min(deadline, time.time() + per_size_budget)
        seq, score = _run_instance(n, mode, size_deadline)

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] New global best – length {n} → score {best_score:.6f}")

    # final polishing of the overall best sequence with the remaining time
    remaining = deadline - time.time()
    if remaining > 10.0 and best_seq is not None:
        seq, score = _simulated_annealing(best_seq, time.time() + remaining)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] Polished final best → score {best_score:.6f}")

    # strip trailing near‑zero entries
    if best_seq is not None:
        arr = np.array(best_seq, dtype=np.float64)
        nz = np.where(arr > 1e-12)[0]
        if nz.size:
            arr = arr[: nz[-1] + 1]
        best_seq = arr.tolist()
    else:
        best_seq = []

    print(f"[{time.time():.1f}] Final best score = {best_score:.6f}, length = {len(best_seq)}")
    return best_seq
