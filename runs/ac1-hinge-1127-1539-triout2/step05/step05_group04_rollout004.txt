"""
Step‑function optimizer – aggressive large‑scale sub‑gradient + mass‑balance.

* The objective after ℓ¹‑normalisation Σa = √(2 n) equals the maximal entry
  of the self‑convolution (the “peak”).  The algorithm therefore works
  directly on max (a*a).

* Strategy
  – Try a few short random restarts on several modest lengths (2k‑8k) just
    to obtain a decent starting point.
  – Switch to a single large length (≈16 k).  On this length we spend the
    bulk of the wall‑clock budget on a pure projected sub‑gradient descent
    on the true ℓ∞‑norm (no smoothing) with a generous step‑size factor.
  – After the sub‑gradient phase we run a greedy “mass‑balance” routine
    that repeatedly moves a small fraction of the most offending coefficient
    to a light location, further flattening the convolution.
  – If any time remains we polish the best candidate with a light simulated‑
    annealing pass.

All steps respect the ℓ¹‑simplex (non‑negativity + Σa = √(2 n)) by Euclidean
projection (Duchi et al., 2008).  FFTs are used for all convolutions.
The code is deliberately lightweight and stays well within the 1000 s
budget on a single CPU.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1, dtype=float)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·n−1) using an FFT.
    Returned array is real‑valued.
    """
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Objective after ℓ¹‑normalisation Σa = √(2 n):
    simply the supremum norm of the self‑convolution.
    """
    return float(_max_convolution_fft(a).max())


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑norm
# ----------------------------------------------------------------------


def _subgradient_descent_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected sub‑gradient descent on   max_k (a*a)[k] .
    A simple diminishing step size η_t = C / √t·‖g‖ is used, where C is a
    generous constant (here 1.5·Σa) to make progress fast.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        # -------- compute the active index and sub‑gradient -----------------
        conv = _max_convolution_fft(a)
        k = int(np.argmax(conv))

        # gradient g_i = 2·a_{k‑i} where defined
        g = np.zeros_like(a)
        if k < n:                         # left half
            g[: k + 1] = 2.0 * a[k :: -1]   # a[k], a[k-1], ...
        else:                             # right half
            d = k - (n - 1)               # shift into the right side
            g[d:] = 2.0 * a[: n - d]

        # -------- step size ------------------------------------------------
        norm_g = np.linalg.norm(g) + 1e-12
        # constant 1.5 is larger than the 0.8 used in the reference code
        step = 1.5 * target_sum / (np.sqrt(it + 1) * norm_g)

        a -= step * g

        # -------- feasibility projection -----------------------------------
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        # occasional tiny jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1
    return a


# ----------------------------------------------------------------------
# Greedy mass‑balance refinement
# ----------------------------------------------------------------------


def _mass_balance_refine(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Repeatedly moves a modest fraction (≈5 %) of the coefficient that
    contributes most to the current maximal convolution entry towards a
    light location.  The move is always followed by a simplex projection.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sum(a)                     # should already be √(2 n)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _max_convolution_fft(a)
        k = int(np.argmax(conv))

        # contribution of each index i to entry k:
        idx = np.arange(n)
        other = k - idx
        mask = (other >= 0) & (other < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[other[mask]]

        i_high = int(np.argmax(contrib))

        # choose a light target – among the 15 % smallest masses
        low_candidates = np.where(a <= np.percentile(a, 15))[0]
        if low_candidates.size == 0:
            i_low = int(np.argmin(a))
        else:
            i_low = int(rng.choice(low_candidates))

        # move a fraction of the heavy mass (5 %)
        delta = a[i_high] * 0.05
        if delta < 1e-14:
            break

        a[i_high] -= delta
        a[i_low] += delta

        # enforce feasibility
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            # occasional global jitter to avoid stagnation
            if rng.random() < 0.01:
                jitter = rng.normal(scale=target_sum * 5e-4, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                a = _project_to_simplex(a, target_sum)

        it += 1
    return a


# ----------------------------------------------------------------------
# Light simulated‑annealing polishing (kept from reference)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Very light SA that moves a random fraction of mass between two sites."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    cur_score = _score(a)
    best_score = cur_score

    T = cur_score * 0.02 + 1e-9
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-16:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        frac = rng.random() * 0.15
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta

        new_score = _score(a)
        dE = new_score - cur_score
        accept = dE <= 0.0 or (T > 0.0 and rng.random() < np.exp(-dE / T))

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling
    return best

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative point on the simplex Σa = √(2 n).
    Several families are sampled to increase diversity.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 7)
    idx = np.arange(n, dtype=float)

    if family == 0:                       # power‑law
        p = rng.uniform(0.2, 1.2)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                     # pure exponential
        lam = rng.exponential(scale=0.02)
        a = np.exp(-lam * idx)

    elif family == 2:                     # power × exponential
        p = rng.uniform(0.2, 1.0)
        lam = rng.exponential(scale=0.02)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                     # Dirichlet with large concentration (near‑uniform)
        a = rng.dirichlet(alpha=np.full(n, 10.0))

    elif family == 4:                     # Dirichlet with small concentration (spiky)
        a = rng.dirichlet(alpha=np.full(n, 0.5))

    elif family == 5:                     # piecewise two‑exponential
        split = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(split)),
                             np.exp(-lam2 * np.arange(split, n))])

    else:                                 # almost uniform with jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` metric.  The routine follows the plan described
    at the top of the file and respects the wall‑clock budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Small lengths – quick random restarts (helps to escape pathological
    # initial points).  We keep at most one candidate per length.
    # ------------------------------------------------------------------
    quick_lengths = [2000, 4000, 8000]
    for n in quick_lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # a few cheap restarts
        restarts = 6
        for _ in range(restarts):
            if time.time() >= deadline:
                break
            a = _random_initial_sequence(n, rng)
            # a very brief sub‑gradient run just to get a decent basin
            short_dead = min(deadline, time.time() + 2.5)
            a = _subgradient_descent_max(a, short_dead)

            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Quick candidate n={n}  score={sc:.6f}")

    # ------------------------------------------------------------------
    # Main phase – large length with the bulk of the budget
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n_large = 16000                     # chosen to give a good trade‑off
        target_sum = np.sqrt(2.0 * n_large)

        # initialise – keep best of the quick runs if it already has this length,
        # otherwise start from a fresh random point
        if best_seq is not None and best_seq.size == n_large:
            a = best_seq.copy()
        else:
            a = _random_initial_sequence(n_large, rng)

        # --------- projected sub‑gradient descent (major part) ------------
        sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.65)
        a = _subgradient_descent_max(a, sub_dead)
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time():.1f}] After sub‑gradient n={n_large}  score={sc:.6f}")

        # --------- mass‑balance fine‑tuning --------------------------------
        if time.time() < deadline:
            mass_dead = deadline - 5.0  # reserve a few seconds for final SA
            a = _mass_balance_refine(a, mass_dead)
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[{time.time():.1f}] After mass‑balance n={n_large}  score={sc:.6f}")

    # ------------------------------------------------------------------
    # Final simulated‑annealing polishing (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 4.0:
        a_final = _simulated_annealing(best_seq, deadline)
        sc = _score(a_final)
        if sc < best_score:
            best_score = sc
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final SA improvement: score={sc:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: drop trailing zeros and report the official evaluation
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback: uniform sequence of modest size
        n_fallback = 2000
        best_seq = np.full(n_fallback, np.sqrt(2.0 * n_fallback) / n_fallback, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
