"""
adversarial step‑function optimizer – enhanced version

This optimizer searches for a non‑negative vector `a` of length `n`
with ℓ₁‑norm `√(2·n)` that minimises

    2·n·max(a*a) / (∑a)²

i.e. the peak of its self‑convolution under a fixed total mass.
The method consists of three stages

1. **Exploration** – many short Adam runs with a variety of
   random initial shapes (uniform, edge‑heavy power laws,
   random Dirichlet, scaled previous best).  The ADAM optimiser
   works on the log‑softmax parametrisation of the simplex,
   uses a τ‑soft‑max approximation of the ∞‑norm and a geometric
   cooling of τ.

2. **Intensification** – the currently best sequence is further
   optimised for a longer time with a slower τ‑schedule and
   more aggressive jitter.

3. **Local refinement** – a cheap hill‑climbing step moves a
   small amount of mass between two random positions and keeps
   the change if it reduces the convolution peak.

All convolutions are performed with a real FFT (O(n log n)) and
the code respects the 1000 s budget, keeping the best sequence in
the global variable ``height_sequence_1``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Return the maximum entry of a*a."""
    return float(_full_convolution_fft(a).max())


# ----------------------------------------------------------------------
# Soft‑max gradient of the convolution peak
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Compute (g, max_b) where g = ∂/∂a τ‑soft‑max(b)
    for b = a*a and max_b = max(b).
    The gradient is 2 * (mask * a) convolved with reversed a.
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        w = exps / exps.sum()
        mask = w

    # gradient wrt a: 2 * conv(mask * a, rev(a))
    a_rev = a[::-1]
    conv_len = len(mask) + n - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   long_run: bool = False):
    """
    Adam optimisation of the τ‑soft‑max approximation.
    Returns (best_a, best_peak).  If ``long_run`` is True a slower
    cooling schedule and more jitter are used.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    # initialise logits so that softmax(logits) * S == a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                     # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.25 if not long_run else 0.15

    # smoothing schedule
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 1.0)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    # schedule parameters
    tau_decay = 0.999 if not long_run else 0.9998
    lr_decay = 0.9999 if not long_run else 0.99995
    jitter_interval = 300 if not long_run else 150
    jitter_scale = 0.025 if not long_run else 0.015

    while time.time() < deadline:
        it += 1

        # ---------- a from logits ----------
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ---------- convolution and gradient ----------
        g, peak = _softmax_gradient(a, tau)

        # update best (hard) peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ---------- gradient wrt logits ----------
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # simple clipping for stability
        maxi = np.max(np.abs(grad_p))
        clip_thr = 5.0 * S
        if maxi > clip_thr:
            grad_p = grad_p * (clip_thr / maxi)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---------- schedules ----------
        tau *= tau_decay
        lr *= lr_decay

        # occasional jitter in log‑space
        if it % jitter_interval == 0:
            p += rng.normal(scale=jitter_scale, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Generation of initial vectors
# ----------------------------------------------------------------------
def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.ceil(pos).astype(int)
    hi = np.clip(hi, 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


def _initial_vector(n: int, rng: np.random.Generator, base_seq=None) -> np.ndarray:
    """
    Produce a feasible start vector for length n.
    Several families are sampled; occasionally the previous best
    sequence (scaled) is mixed in.
    """
    S = np.sqrt(2.0 * n)

    shape = rng.choice(['uniform', 'edge', 'power', 'dirichlet'])
    if shape == 'uniform':
        a = np.full(n, S / n)
    elif shape == 'edge':
        # a_i ∝ (dist_to_edge+1)^(-p)
        p_exp = rng.uniform(0.8, 2.5)
        idx = np.arange(n)
        d = np.minimum(idx, n - 1 - idx) + 1
        a = (d.astype(float) ** (-p_exp))
        a = a / a.sum() * S
    elif shape == 'power':
        # a_i ∝ (i+1)^(-p) + (n-i)^(-p)
        p_exp = rng.uniform(0.5, 2.0)
        idx = np.arange(n) + 1
        left = idx ** (-p_exp)
        right = (n - idx + 1) ** (-p_exp)
        a = left + right
        a = a / a.sum() * S
    else:   # random Dirichlet
        alpha = rng.uniform(0.3, 3.0)
        a = rng.gamma(alpha, 1.0, size=n)
        a = a / a.sum() * S

    # mix with the previous best with modest probability
    if base_seq is not None and rng.random() < 0.25:
        a = _scale_sequence(base_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S

    return a


# ----------------------------------------------------------------------
# Local stochastic refinement (greedy hill‑climbing)
# ----------------------------------------------------------------------
def _local_refine(seq: np.ndarray,
                  time_limit: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple greedy perturbation: move a small fraction of mass from
    a random source index to a random target index if the move
    decreases the convolution peak.
    """
    a = seq.copy()
    n = len(a)
    S = a.sum()
    best_peak = _max_convolution(a)

    end = time.time() + time_limit
    while time.time() < end:
        i, j = rng.choice(n, size=2, replace=False)
        # pick a step that does not make a[i] negative
        max_move = a[i]
        if max_move <= 1e-12:
            continue
        step = rng.uniform(0.01, 0.2) * max_move   # up to 20 % of source mass
        a[i] -= step
        a[j] += step

        cur_peak = _max_convolution(a)
        if cur_peak < best_peak:
            best_peak = cur_peak
            # keep the move
        else:
            # revert
            a[i] += step
            a[j] -= step

    # renormalise to avoid tiny drift
    a = a * (S / a.sum())
    return a, best_peak

# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Return the maximum entry of a*a."""
    return float(_full_convolution_fft(a).max())


# ----------------------------------------------------------------------
# Soft‑max gradient of the convolution peak
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Compute (g, max_b) where g = ∂/∂a τ‑soft‑max(b)
    for b = a*a and max_b = max(b).
    The gradient is 2 * (mask * a) convolved with reversed a.
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        w = exps / exps.sum()
        mask = w

    # gradient wrt a: 2 * conv(mask * a, rev(a))
    a_rev = a[::-1]
    conv_len = len(mask) + n - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   long_run: bool = False):
    """
    Adam optimisation of the τ‑soft‑max approximation.
    Returns (best_a, best_peak).  If ``long_run`` is True a slower
    cooling schedule and more jitter are used.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    # initialise logits so that softmax(logits) * S == a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                     # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.25 if not long_run else 0.15

    # smoothing schedule
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 1.0)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    # schedule parameters
    tau_decay = 0.999 if not long_run else 0.9998
    lr_decay = 0.9999 if not long_run else 0.99995
    jitter_interval = 300 if not long_run else 150
    jitter_scale = 0.025 if not long_run else 0.015

    while time.time() < deadline:
        it += 1

        # ---------- a from logits ----------
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ---------- convolution and gradient ----------
        g, peak = _softmax_gradient(a, tau)

        # update best (hard) peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ---------- gradient wrt logits ----------
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # simple clipping for stability
        maxi = np.max(np.abs(grad_p))
        clip_thr = 5.0 * S
        if maxi > clip_thr:
            grad_p = grad_p * (clip_thr / maxi)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---------- schedules ----------
        tau *= tau_decay
        lr *= lr_decay

        # occasional jitter in log‑space
        if it % jitter_interval == 0:
            p += rng.normal(scale=jitter_scale, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Generation of initial vectors
# ----------------------------------------------------------------------
def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.ceil(pos).astype(int)
    hi = np.clip(hi, 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


def _initial_vector(n: int, rng: np.random.Generator, base_seq=None) -> np.ndarray:
    """
    Produce a feasible start vector for length n.
    Several families are sampled; occasionally the previous best
    sequence (scaled) is mixed in.
    """
    S = np.sqrt(2.0 * n)

    shape = rng.choice(['uniform', 'edge', 'power', 'dirichlet'])
    if shape == 'uniform':
        a = np.full(n, S / n)
    elif shape == 'edge':
        # a_i ∝ (dist_to_edge+1)^(-p)
        p_exp = rng.uniform(0.8, 2.5)
        idx = np.arange(n)
        d = np.minimum(idx, n - 1 - idx) + 1
        a = (d.astype(float) ** (-p_exp))
        a = a / a.sum() * S
    elif shape == 'power':
        # a_i ∝ (i+1)^(-p) + (n-i)^(-p)
        p_exp = rng.uniform(0.5, 2.0)
        idx = np.arange(n) + 1
        left = idx ** (-p_exp)
        right = (n - idx + 1) ** (-p_exp)
        a = left + right
        a = a / a.sum() * S
    else:   # random Dirichlet
        alpha = rng.uniform(0.3, 3.0)
        a = rng.gamma(alpha, 1.0, size=n)
        a = a / a.sum() * S

    # occasional mix‑in of the best known construction
    if base_seq is not None and rng.random() < 0.25:
        a = _scale_sequence(base_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S

    return a


# ----------------------------------------------------------------------
# Local stochastic refinement (greedy hill‑climbing)
# ----------------------------------------------------------------------
def _local_refine(seq: np.ndarray,
                  time_limit: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple greedy perturbation: move a small fraction of mass from
    a random source index to a random target index if the move
    decreases the convolution peak.
    """
    a = seq.copy()
    n = len(a)
    total = a.sum()
    best_peak = _max_convolution(a)

    end = time.time() + time_limit
    while time.time() < end:
        i, j = rng.choice(n, size=2, replace=False)
        if a[i] <= 1e-12:
            continue
        step = rng.uniform(0.01, 0.2) * a[i]   # up to 20 % of source mass
        a[i] -= step
        a[j] += step

        cur_peak = _max_convolution(a)
        if cur_peak < best_peak:
            best_peak = cur_peak
            # keep the change
        else:
            # revert
            a[i] += step
            a[j] -= step

    # renormalise to compensate for tiny numerical drift
    a = a * (total / a.sum())
    return a, best_peak


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine runs a series of Adam restarts, a longer
    optimisation of the current best, and a short greedy refinement.
    The best sequence found is stored in the global variable
    ``height_sequence_1`` and returned as a plain list.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Load previous construction if it exists
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_val = np.inf          # stores the hard peak max_b
    attempts = 0

    # ------------------------------------------------------------------
    # Phase 1 – many short restarts (exploration)
    # ------------------------------------------------------------------
    while time.time() < end_time - 150.0:   # reserve ~150 s for later phases
        attempts += 1
        # pick a length – a mix of moderate and larger sizes
        if rng.random() < 0.7:
            n = int(rng.integers(1500, 8000))
        else:
            n = int(rng.integers(8000, 20000))

        a_start = _initial_vector(n, rng, base_seq)

        # allocate about 10 s for this run
        slice_deadline = min(time.time() + 10.0, end_time - 150.0)

        a_opt, peak = _adam_optimize(a_start, slice_deadline, rng,
                                     long_run=False)

        if peak < best_val:
            best_val = peak
            best_seq = a_opt.copy()
            print(f"[restart] attempt={attempts} n={n}  new best peak={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – intensive optimisation of the current best
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = end_time - time.time() - 30.0   # keep 30 s for final refinement
        if remaining > 0:
            slice_deadline = time.time() + remaining
            a_opt, peak = _adam_optimize(best_seq, slice_deadline, rng,
                                         long_run=True)
            if peak < best_val:
                best_val = peak
                best_seq = a_opt.copy()
                print(f"[intensify] n={len(best_seq)} improved peak={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – cheap greedy refinement
    # ------------------------------------------------------------------
    if best_seq is not None:
        time_left = end_time - time.time()
        if time_left > 5.0:
            a_ref, peak = _local_refine(best_seq, time_left - 5.0, rng)
            if peak < best_val:
                best_val = peak
                best_seq = a_ref.copy()
                print(f"[refine] final peak={best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the supplied black‑box evaluator
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
        external = evaluate_sequence(final_seq)
        print(f"[final] no candidate built, external score={external:.6f}")
    else:
        final_seq = best_seq.tolist()
        external = evaluate_sequence(final_seq)
        print(f"[final] external score={external:.6f}  internal peak={best_val:.6f}  length={len(final_seq)}")
        globals()['height_sequence_1'] = final_seq

    return final_seq
