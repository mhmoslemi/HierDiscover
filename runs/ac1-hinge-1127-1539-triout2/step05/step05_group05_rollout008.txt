def _random_sequence(n, rng):
    kind = rng.integers(0,7)
    idx = np.arange(n)
    target_sum = np.sqrt(2.0 * n)

    if kind == 0:
        # Dirichlet with random alpha
        alpha = rng.uniform(0.1, 5.0)
        a = rng.dirichlet(np.full(n, alpha))
    elif kind == 1:
        # Sparse Dirichlet low alpha
        alpha = rng.uniform(0.01, 0.2)
        a = rng.dirichlet(np.full(n, alpha))
    elif kind == 2:
        # Exponential decay
        lam = rng.exponential(scale=0.01)
        a = np.exp(-lam * idx)
    elif kind == 3:
        # Power law decreasing
        p = rng.uniform(0.1, 1.5)
        a = (idx + 1) ** (-p)
    elif kind == 4:
        # Symmetric shape using half-sample
        half_n = (n + 1) // 2
        # generate half via Dirichlet
        alpha = rng.uniform(0.1, 5.0)
        half = rng.dirichlet(np.full(half_n, alpha))
        if n % 2 == 0:
            a = np.concatenate([half, half[::-1]])
        else:
            a = np.concatenate([half, half[:-1][::-1]])
    elif kind == 5:
        # Beta shape (U/bell)
        alpha = rng.uniform(0.2, 2.5)
        beta = rng.uniform(0.2, 2.5)
        x = (idx + 0.5) / n
        a = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1))
    else:
        # Uniform (Dirichlet alpha=1)
        a = rng.dirichlet(np.ones(n))

    # ensure positivity
    a = np.maximum(a, 0.0)
    # project onto simplex (if not already)
    a = _project_to_simplex(a, target_sum)
    return a

def _gradient_lpnorm(a, p):
    n = a.shape[0]
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[:2 * n - 1]

    # p>=2, b>=0.

"""
Propose a step‑function (non‑negative sequence) with a small maximal
self‑convolution value.  The algorithm works in three phases under the
provided time budget:

1. **Diverse random seeding** – for several lengths (up to ~10 000) a
   large number of random points on the ℓ¹‑simplex Σa = √(2 n) are
   generated using Dirichlet, power‑law, exponential and Beta shapes.
   The best candidate of each length (lowest max‑convolution) is kept.

2. **Lp‑norm descent** – each stored candidate is refined by gradient
   descent on the L_p‑norm (p=8) of its full convolution
   ``b = a*a``.  The gradient of ``∑ b_k^p`` w.r.t. ``a`` can be written
   as a convolution and is evaluated by FFTs.  After each gradient step
   the vector is projected back onto the ℓ¹‑simplex.

3. **Targeted simulated annealing** – the best sequence obtained so far
   is further polished by a classic SA that repeatedly moves a small
   fraction of mass from an index that heavily contributes to the
   current maximal convolution entry to a random destination.  A
   slowly‑cooling temperature schedule and occasional global jitter
   help escape local minima.

The final sequence is truncated at trailing values ≈0 and returned as
a plain ``list[float]``.  All numbers are non‑negative and the routine
never exceeds the supplied time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean, Duchi et al. 2008)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector `v` onto the simplex {x ≥ 0, Σx = s}."""
    if s <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------

    """Maximum entry of the linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Random initialisers – many families for diversity
# ----------------------------------------------------------------------


def _random_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random non‑negative sequence on the ℓ¹‑simplex."""
    target = np.sqrt(2.0 * n)
    kind = rng.integers(0, 7)
    idx = np.arange(n, dtype=float)

    if kind == 0:               # Dirichlet with random concentration
        alpha = rng.uniform(0.1, 5.0)
        a = rng.dirichlet(np.full(n, alpha))

    elif kind == 1:             # Very sparse Dirichlet (tiny α)
        alpha = rng.uniform(0.01, 0.2)
        a = rng.dirichlet(np.full(n, alpha))

    elif kind == 2:             # Exponential decay from the left edge
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif kind == 3:             # Power‑law decreasing
        p = rng.uniform(0.1, 1.5)
        a = (idx + 1.0) ** (-p)

    elif kind == 4:             # Symmetric half‑sample (mirrored)
        half = (n + 1) // 2
        alpha = rng.uniform(0.1, 5.0)
        h = rng.dirichlet(np.full(half, alpha))
        if n % 2 == 0:
            a = np.concatenate([h, h[::-1]])
        else:
            a = np.concatenate([h, h[:-1][::-1]])

    elif kind == 5:             # Beta‑shaped (U‑ or bell‑shaped)
        a_val = rng.uniform(0.2, 2.5)
        b_val = rng.uniform(0.2, 2.5)
        x = (idx + 0.5) / n
        a = (x ** (a_val - 1.0)) * ((1.0 - x) ** (b_val - 1.0))

    else:                       # Uniform Dirichlet (α = 1)
        a = rng.dirichlet(np.ones(n))

    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


# ----------------------------------------------------------------------
# Gradient of the L_p‑norm of the convolution (p ≥ 2, integer)
# ----------------------------------------------------------------------


def _gradient_lpnorm(a: np.ndarray, p: int):
    """
    Returns (g, b) where
        b = a*a (full convolution, length 2·n‑1)
        g = ∇_a  ( Σ_k b_k**p )
          = 2·p·(a_rev ⋆ w)[central n entries],
      with w_k = b_k**(p‑1).
    """
    n = a.shape[0]

    # b = a * a
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # weights w = b**(p-1)
    w = np.power(b, p - 1, dtype=np.float64)

    # convolution a_rev * w
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << ((conv_len).bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Wr_fft = np.fft.rfft(w, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Wr_fft, n=size2)[:conv_len]

    # central n entries correspond to gradient components
    g = 2.0 * p * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Lp‑norm descent (projected gradient)
# ----------------------------------------------------------------------


def _lp_descent_opt(a0: np.ndarray, p: int, deadline: float) -> np.ndarray:
    """
    Gradient descent on Σ b_k**p (b = a*a) with projection onto the ℓ¹‑simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    step = 0
    eta0 = 0.5          # base step size
    eps = 1e-12

    while time.time() < deadline:
        g, _ = _gradient_lpnorm(a, p)

        # diminishing step size
        eta = eta0 / np.sqrt(step + 1.0)
        a -= eta * g

        # enforce non‑negativity (tiny epsilon avoids exact zeros)
        a = np.maximum(a, eps)

        # project back onto the ℓ¹‑simplex
        s = np.sum(a)
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        step += 1

    return a


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated gradient) – kept for occasional use
# ----------------------------------------------------------------------


def _mirror_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Exponentiated‑gradient descent on the log‑sum‑exp surrogate of max(b).
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 1.0
    beta_max = 1e5
    eta0 = 0.5
    step = 0
    eps = 1e-12

    while time.time() < deadline:
        # gradient of (1/β)·log Σ exp(β·b)
        # reuse the soft‑max gradient from the original code
        size1 = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size1)
        b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

        max_b = np.max(b)
        exps = np.exp(beta * (b - max_b))
        p_vec = exps / np.sum(exps)

        a_rev = a[::-1]
        conv_len = n + (2 * n - 1) - 1
        size2 = 1 << ((conv_len).bit_length())
        Ar_fft = np.fft.rfft(a_rev, n=size2)
        Pr_fft = np.fft.rfft(p_vec, n=size2)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
        g = 2.0 * conv_rev[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(step + 1.0)
        a = a * np.exp(-eta * g)
        a = np.maximum(a, eps)

        s = np.sum(a)
        if s > 0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        if step % 200 == 0 and beta < beta_max:
            beta = min(beta * 1.07, beta_max)

        step += 1

    return a


# ----------------------------------------------------------------------
# Targeted simulated‑annealing refinement
# ----------------------------------------------------------------------


def _targeted_simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    A simple SA that moves a random fraction of mass from a contributor to the
    current maximal convolution entry to a random destination.  The temperature
    decays slowly; occasional global jitter helps escape deep traps.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_fft(best)

    T = best_score * 0.02 + 1e-8
    cooling = 0.9995
    rng = np.random.default_rng()
    max_transfer = 0.25          # up to 25 % of the chosen mass per move

    while time.time() < deadline:
        b = _full_convolution_fft(a)
        k_max = int(np.argmax(b))

        # indices that actually appear in b[k_max]
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idx_range = np.arange(lo, hi + 1, dtype=int)
        contrib = a[idx_range] * a[k_max - idx_range]
        weight_sum = np.sum(contrib)

        if weight_sum > 0.0:
            probs = contrib / weight_sum
            i = rng.choice(idx_range, p=probs)
        else:
            i = rng.integers(0, n)

        # pick a destination j ≠ i uniformly
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        # move a random fraction of the mass at i to j
        delta = a[i] * rng.random() * max_transfer
        if delta <= 0.0:
            continue
        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score

        accept = False
        if cur_score <= best_score:
            accept = True
        elif T > 0.0 and rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA improvement: {best_score:.6f}")
        else:
            # revert the move
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # occasional global jitter to shake the system
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            # re‑project onto the simplex (preserve total mass)
            s = np.sum(a)
            if s > 0:
                a = a * (np.sqrt(2.0 * n) / s)
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA jitter improvement: {best_score:.6f}")

    return best


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the given time budget (seconds) and returns a plain
    Python list of floats.
    """
    rng = np.random.default_rng(seed)
    safety = 0.5                       # give the evaluator a tiny safety margin
    deadline = time.time() + budget_s - safety

    # ------------------------------------------------------------------
    # Global best holder
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Phase 1 – diverse random seeding over several lengths
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 4000, 5000, 6000, 8000, 10000]
    scan_time_per_len = 12.0          # seconds for each length (adjustable)

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)
        end_phase = min(deadline, time.time() + scan_time_per_len)

        local_best_seq = None
        local_best_score = np.inf

        while time.time() < end_phase:
            a = _random_sequence(n, rng)
            score = _max_convolution_fft(a)
            if score < local_best_score:
                local_best_seq = a.copy()
                local_best_score = score
                # also update global best immediately
                if score < best_score:
                    best_seq, best_score = a.copy(), score
                    print(f"[{time.time():.1f}] New global best (scan) n={n}  score={score:.6f}")

        # keep the best of this length for later refinement
        if local_best_seq is not None:
            # store as a candidate for the next phase
            pass  # (we will just keep the global best – the list is small)

    # ------------------------------------------------------------------
    # Fallback – if nothing was generated (unlikely)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 3000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = _max_convolution_fft(best_seq)

    # ------------------------------------------------------------------
    # Phase 2 – Lp‑norm descent on the current best (and a few extra restarts)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        # allocate half of the remaining time to gradient refinement
        grad_deadline = time.time() + remaining * 0.5

        # run a few restarts with different random perturbations
        restarts = 3
        for r in range(restarts):
            # perturb the current best slightly before descent
            a0 = best_seq.copy()
            jitter = rng.normal(scale=a0.sum() * 0.005, size=a0.size)
            a0 += jitter
            a0 = np.maximum(a0, 0.0)
            # re‑project onto the simplex for the appropriate length
            a0 = _project_to_simplex(a0, np.sqrt(2.0 * a0.size))

            a_opt = _lp_descent_opt(a0, p=8, deadline=grad_deadline)
            score = _max_convolution_fft(a_opt)

            if score < best_score:
                best_seq, best_score = a_opt.copy(), score
                print(f"[{time.time():.1f}] New best after Lp‑descent (restart {r}) score={score:.6f}")

        # optionally try a single mirror‑descent refinement as well
        md_deadline = time.time() + remaining * 0.2
        a_md = _mirror_descent_opt(best_seq, md_deadline)
        md_score = _max_convolution_fft(a_md)
        if md_score < best_score:
            best_seq, best_score = a_md.copy(), md_score
            print(f"[{time.time():.1f}] New best after mirror descent score={md_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – targeted simulated annealing on the final best sequence
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_sa = _targeted_simulated_annealing(best_seq, deadline)
        final_score = _max_convolution_fft(a_sa)
        if final_score < best_score:
            best_seq, best_score = a_sa.copy(), final_score
            print(f"[{time.time():.1f}] New best after SA score={final_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing zeros (tiny values) – improves readability
    # ------------------------------------------------------------------
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # ------------------------------------------------------------------
    # Final sanity check using the official evaluator (optional)
    # ------------------------------------------------------------------
    try:
        ev = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence = {ev:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best max‑convolution = "
          f"{best_score:.6f}   length = {best_seq.size}")

    return best_seq.tolist()
