def projected_subgradient(a, target_sum, max_iters, deadline):
    a = a.copy()
    n = a.shape[0]
    v = np.zeros_like(a)
    mu = 0.9
    lr_base = target_sum * 0.5  # scaling
    for t in range(max_iters):
        if time.time() >= deadline:
            break
        # compute convolution and max index
        conv = _conv_fft(a)
        k_max = int(np.argmax(conv))
        g = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        # indices i in [lo, hi], g[i] = 2 * a[k_max - i]
        idx = np.arange(lo, hi+1)
        g[idx] = 2 * a[k_max - idx]
        # step size
        lr = lr_base / np.sqrt(t+1)
        v = mu * v - lr * g
        a_candidate = a + v
        # project onto simplex & non-negativity
        a = _project_to_simplex(a_candidate, target_sum)
    return a, _score(a)

def enlarge_sequence(seq, new_len):
    old_len = len(seq)
    if new_len == old_len:
        return np.array(seq, copy=True)
    x_old = np.linspace(0, 1, old_len)
    x_new = np.linspace(0, 1, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    return new_seq

def propose_candidate(seed=42, budget_s=1000):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 1.0

    # initial best_seq: from global if exists else random 1500 power law
    if 'height_sequence_1' in globals():
        base_seq = np.array(height_sequence_1, dtype=np.float64)
    else:
        base_seq = _power_law_sequence(1500, 0.55, np.sqrt(2*1500))

    # ensure normalized sum sqrt(2 * len(base_seq))
    base_n = base_seq.size
    base_seq = _project_to_simplex(base_seq, np.sqrt(2.0*base_n))

    best_seq = base_seq
    best_score = _score(best_seq)

    # list target lengths
    target_lengths = [1500, 3000, 5000, 8000]
    # allocate time for each in proportion
    remaining = deadline - time.time()
    per_len_time = remaining / len(target_lengths)  # approx equal slices

    for N in target_lengths:
        if time.time() >= deadline:
            break
        # enlarge previous best pattern
        seq = enlarge_sequence(best_seq, N)
        # project
        seq = _project_to_simplex(seq, np.sqrt(2.0*N))
        # run gradient descent for part of per_len_time
        gd_time = per_len_time * 0.3
        gd_deadline = time.time() + gd_time
        seq, gd_score = _annealed_gradient_descent(seq, gd_deadline)
        # run SA for rest of per_len_time
        sa_deadline = time.time() + per_len_time - gd_time
        if sa_deadline > deadline:
            sa_deadline = deadline
        seq, sa_score = _simulated_annealing(seq, sa_deadline)
        cur_score = min(gd_score, sa_score)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] New global best for N={N}: score={best_score:.6f}")

        # after each length, use this refined best as base for next length
        # also keep a copy truncated to N (like best_seq) for next iteration.

    # final trim trailing zeros
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    print(f"[{time.time():.1f}] final best score={best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()

def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]

    else:

        theta = cssv[rho] / (rho + 1)

    return w

    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    step_base = target_sum * 0.20
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            lr = step_base / (norm_g * (beta + 1.0))
            a = a - lr * g

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaux
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] GD improvement: β={beta:.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing refinement
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float,
                         start_temp: float = 0.5,
                         end_temp: float = 1e-5,
                         max_frac_start: float = 0.40,
                         max_frac_end: float = 0.005,
                         max_window: int = 30) -> tuple[np.ndarray, float]:
    """Long simulated‑annealing refinement."""
    a = a0.copy()
    n = a.size
    cur_score = _score(a)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)

    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # location of the current maximal convolution entry
        max_b = conv.max()
        k_max = int(np.argmax(conv))

        # sensitivity of each coordinate for that entry
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]

        weight = a * sens
        move_type = None

        # --------------------------------------------------------------
        # 70 % mass‑transfer moves
        # --------------------------------------------------------------
        if rng.random() < 0.70:
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight / total_w)
            else:
                donor = rng.integers(n)

            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            receiver = rng.choice(low_idx) if low_idx.size else rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        # --------------------------------------------------------------
        # 20 % local smoothing moves
        # --------------------------------------------------------------
        elif rng.random() < 0.90:
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                old_seg = a[start:end].copy()
                avg = old_seg.mean()
                a[start:end] = avg
                move_type = "smooth"
                saved_seg = old_seg
                saved_start, saved_end = start, end

        # --------------------------------------------------------------
        # 10 % random swaps
        # --------------------------------------------------------------
        else:
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # --------------------------------------------------------------
        # Evaluate move (single FFT)
        # --------------------------------------------------------------
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = _score(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA new best = {best_score:.6f}")
        else:
            # revert the move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter to shake out deep minima
        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=cur_score * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = a * (np.sqrt(2.0 * n) / a.sum())
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper for resizing a sequence
# ----------------------------------------------------------------------


def enlarge_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Resize a non‑negative sequence by linear interpolation."""
    old_len = len(seq)
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    return np.interp(x_new, x_old, seq)

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine builds increasingly larger candidates by interpolation,
    runs a short projected‑gradient descent (soft‑max of max‑convolution),
    then refines the result with simulated annealing.  The best sequence
    discovered before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # 1) Initialise a base sequence.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        base_seq = np.array(height_sequence_1, dtype=np.float64)
        base_n = base_seq.size
        base_seq = _project_to_simplex(base_seq, np.sqrt(2.0 * base_n))
    else:
        # fallback: simple decreasing power‑law
        base_n = 1500
        base_seq = (np.arange(1, base_n + 1, dtype=np.float64) ** (-0.55))
        base_seq = base_seq / base_seq.sum() * np.sqrt(2.0 * base_n)

    best_seq = base_seq.copy()
    best_score = _score(best_seq)

    # ------------------------------------------------------------------
    # 2) Hierarchical enlargement (1500 → 3000 → 5000 → 8000)
    # ------------------------------------------------------------------
    target_lengths = [1500, 3000, 5000, 8000]
    # Roughly even time split among the different lengths
    remaining = deadline - time.time()
    per_stage = max(remaining / len(target_lengths), 0.0)

    for N in target_lengths:
        if time.time() >= deadline:
            break

        # Enlarge the current best pattern to length N and re‑normalise
        seq = enlarge_sequence(best_seq, N)
        seq = _project_to_simplex(seq, np.sqrt(2.0 * N))

        # -- Gradient descent (≈30 % of the stage time) --
        gd_slice = per_stage * 0.30
        gd_deadline = time.time() + min(gd_slice, deadline - time.time())
        seq, gd_score = _annealed_gradient_descent(seq, gd_deadline)

        # -- Simulated annealing for the remaining time of the stage --
        sa_deadline = deadline if (deadline - time.time()) < per_stage * 0.70 \
                     else time.time() + per_stage * 0.70
        seq, sa_score = _simulated_annealing(seq, sa_deadline)

        cur_score = min(gd_score, sa_score)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] New global best at length {N}: "
                  f"score={best_score:.6f}")

        # The refined sequence becomes the seed for the next (larger) length
        # (no extra processing needed)

    # ------------------------------------------------------------------
    # 3) Final clean‑up and return
    # ------------------------------------------------------------------
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    print(f"[{time.time():.1f}] final best score={best_score:.6f}, "
          f"length={len(best_seq)}")
    return best_seq.tolist()
