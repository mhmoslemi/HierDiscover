def solve_lp(f, M, S, solver='ECOS', max_iters=2000): 
    n = len(f) 
    g = cp.Variable(n, nonneg=True) 
    f_const = cp.Constant(f) 
    constraints = [cp.conv(f_const, g) <= M] 
    # do not constrain sum(g) <= S; we maximize sum(g) 
    objective = cp.Maximize(cp.sum(g)) 
    prob = cp.Problem(objective, constraints) 
    prob.solve(solver=solver, warm_start=True)  # warm_start if repeated 
    # Check status 
    if prob.status not in ["optimal", "optimal_inaccurate"]: 
        # fallback maybe to naive uniform 
        return None 
    g_val = g.value 
    # scale 
    s = g_val.sum() 
    if s <= 0: 
        g_scaled = np.full_like(g_val, S / n)  # uniform 
    else: 
        g_scaled = g_val * (S / s) 
    return g_scaled

"""
search for a non‑negative step sequence minimising the self‑convolution peak.

The algorithm is a hybrid of three ideas:

1.  A smooth‑max Adam optimiser (gradient of a soft‑max of the
    convolution) together with projected sub‑gradient steps – this
    quickly finds a decent low‑peak sequence.

2.  A linear‑programming refinement that implements the “alternating
    LP” idea from the paper: for the current sequence f we solve

           maximise   Σ g_i
           subject to g_i ≥ 0 ,   (f * g)_k ≤ max(f * f)  ∀k

    The solution g is scaled to the L¹‑simplex and blended with f.
    Repeating this step (with a small blending factor) provably never
    increases the peak and often reduces it.

3.  A final optional up‑sampling to a larger length and another round
    of the LP refinement.

All steps respect the constraints required by ``evaluate_sequence``:
non‑negative entries and Σ a_i = √(2·n).  The routine stops after the
allocated budget (default 1000 s) and returns the best sequence found.
"""

import time
import numpy as np
import cvxpy as cp


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolutions
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient term for the smooth max‑norm:
        g_j = 2 Σ_i w_{i+j} a_i .
    Both ``a`` (len n) and ``w`` (len 2·n‑1) are real vectors.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights (smooth approximation of max)
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """w_i = exp(alpha·conv_i) / Σ_j exp(alpha·conv_j) (stable)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Linear‑programming step (alternating LP)
# ----------------------------------------------------------------------
def _solve_lp_for_g(f: np.ndarray, M: float, target_sum: float) -> np.ndarray | None:
    """
    Solve  maximise Σ g_i
    subject to   g_i ≥ 0,   (f * g)_k ≤ M   (full convolution)
    and return g scaled so that Σ g_i = target_sum.
    Returns None if the solver fails.
    """
    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)
    f_const = cp.Constant(f)

    # linear constraints: full convolution of f and g ≤ M
    conv_fg = cp.conv(f_const, g)          # length 2n‑1
    constraints = [conv_fg <= M]

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:               # pragma: no cover
        return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None

    g_val = np.asarray(g.value).flatten()
    s = g_val.sum()
    if s <= 0:
        # fallback: uniform vector (should never happen)
        return np.full_like(g_val, target_sum / n)
    # scale to the required L1‑norm
    return g_val * (target_sum / s)


def _refine_by_lp(a0: np.ndarray,
                  deadline: float,
                  rng: np.random.Generator,
                  max_iters: int = 12) -> tuple[np.ndarray, float]:
    """
    Alternating‑LP refinement: repeatedly solve the LP for the current
    sequence, blend the result with a small step size and project onto
    the simplex.  Returns the best sequence seen during the refinement.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    best = a.copy()
    best_score = np.max(_full_convolution(a))
    t = 0.25                     # initial blending factor

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        M = np.max(_full_convolution(a))

        g = _solve_lp_for_g(a, M, S)
        if g is None:
            # simple random jitter as a safe fallback
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.02
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)
            continue

        # convex combination
        a_new = (1.0 - t) * a + t * g
        a_new = _project_simplex(a_new, S)

        cur_score = np.max(_full_convolution(a_new))
        if cur_score < best_score:
            best_score = cur_score
            best = a_new.copy()
            # if we improved, we can be a little bolder
            t = min(0.6, t * 1.08)
        else:
            # otherwise be more cautious
            t = max(0.05, t * 0.85)

        a = a_new

    return best, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max‑norm
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Adam on a soft‑max approximation of max(conv). Returns best vector."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    lr = 0.25
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    alpha = 0.6               # start temperature
    while time.time() < deadline:
        t += 1
        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # simplex projection
        a = _project_simplex(a, S)

        # occasional small mass‑swap
        if rng.random() < 0.05:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.015
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # tighten the soft‑max
        alpha = min(alpha * 1.025, 300.0)

        # cheap early stop
        if t % 6000 == 0 and best_score == cur_score:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient optimiser (acts on the true max‑norm)
# ----------------------------------------------------------------------
def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that directly reduces the worst entry."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.30
    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of max entry
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1, dtype=int)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # random jitter to escape plateaus
        if rng.random() < 0.06:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.02
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = np.max(_full_convolution(cand))
                if cand_score < cur_score or rng.random() < 0.15:
                    a = cand
                    cur_score = cand_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()

        if it % 9000 == 0 and best_score == cur_score:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimiser (Adam → sub‑gradient)
# ----------------------------------------------------------------------
def _optimize_combined(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Run Adam for ~60 % of the allocated time, then finish with sub‑gradient."""
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    inner_deadline = now + 0.6 * remaining
    a_mid, sc_mid = _optimize_adam(a0, inner_deadline, rng)

    if time.time() < deadline:
        a_fin, sc_fin = _optimize_subgrad(a_mid, deadline, rng)
        if sc_fin < sc_mid:
            return a_fin, sc_fin
    return a_mid, sc_mid


# ----------------------------------------------------------------------
# Upsampling helper (linear interpolation)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate ``a`` to length ``new_n``."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=float)
    return np.interp(x_new, x_old, a)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------

        refined_seq, refined_score = _refine_by_lp(best_seq, lp_deadline, rng,
                                                   max_iters=20)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[lp‑refine] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check & return
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()

    # external evaluation (only for logging – the harness does not require it)
    try:
        external_score = evaluate_sequence(final_seq)
    except Exception:                 # pragma: no cover
        external_score = float('nan')
    print(f"[final] external score = {external_score:.6f} "
          f"(internal best = {best_score:.6f}) length = {len(final_seq)}")
    return final_seq
