"""
Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence``.
The algorithm works on the L¹‑simplex Σ a_i = √(2·n) (so the external score
equals the maximum entry of the full convolution).  It consists of three
phases:

1. **Random seeding / warm‑start** – either from the global
   ``height_sequence_1`` (if available) or from a fresh random draw,
   projected onto the simplex.

2. **Smooth optimisation** – Adam on a soft‑max (log‑sum‑exp) approximation
   of the max‑convolution.  Gradients are obtained via FFT convolution.
   After each Adam step the vector is projected onto the simplex and
   symmetrised (a = (a + a[::-1]) / 2) to reduce the search space.

3. **Sub‑gradient refinement** – projected sub‑gradient descent on the
   true max‑norm, followed by a short stochastic local‑search (mass swaps).

The optimisation is performed iteratively on a moderate size (≈3000) and
periodically up‑sampled (linear interpolation) to a larger size (up to
MAX_N = 12 000) to give the method more degrees of freedom.  The entire
budget (default 1000 s) is used; the best sequence encountered is returned
as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient term for the smooth max‑norm:
        g_j = 2 Σ_i w_{i+j} a_i .
    Both ``a`` (len n) and ``w`` (len 2·n‑1) are real vectors.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights (smooth approximation of max)
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """w_i = exp(alpha·conv_i) / Σ_j exp(alpha·conv_j)  (stable)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Upsampling helper (linear interpolation)
# ----------------------------------------------------------------------


def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a to length new_n."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=float)
    return np.interp(x_new, x_old, a)


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max‑norm
# ----------------------------------------------------------------------


def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Adam on a soft‑max approximation of max(conv). Returns best vector."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.5          # a bit larger than the previous version
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    alpha = 0.5          # start with a smooth objective
    while time.time() < deadline:
        t += 1
        conv = _full_convolution(a)
        # soft‑max weights
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # Project back onto the simplex
        a = _project_simplex(a, S)

        # Enforce symmetry (helps the search)
        a = (a + a[::-1]) * 0.5
        a = _project_simplex(a, S)   # re‑project to correct the sum

        # Random tiny mass‑swap for exploration
        if rng.random() < 0.06:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.02
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)
                a = (a + a[::-1]) * 0.5
                a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # Temperature schedule – sharpen the soft‑max
        alpha = min(alpha * 1.025, 300.0)

        # Very cheap early‑stop check
        if t % 6000 == 0 and best_score == cur_score:
            # If no progress for a while, quit early
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient optimiser (acts on the true max‑norm)
# ----------------------------------------------------------------------


def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that directly attacks the worst entry."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.3
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # Sub‑gradient of the max entry (factor 2 already incorporated)
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j] * 2.0   # factor 2 from derivative of a_i a_j

        eta = eta0 / np.sqrt(it)
        a -= eta * g
        a = _project_simplex(a, S)
        a = (a + a[::-1]) * 0.5
        a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # Random perturbation (mass swap) with a small probability
        if rng.random() < 0.07:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.03
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand = (cand + cand[::-1]) * 0.5
                cand = _project_simplex(cand, S)
                cand_score = np.max(_full_convolution(cand))
                # Accept if it improves or by a small chance (simulated annealing)
                if cand_score < cur_score or rng.random() < 0.15:
                    a = cand
                    cur_score = cand_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()

        if it % 8000 == 0 and best_score == cur_score:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimiser (Adam → sub‑gradient)
# ----------------------------------------------------------------------


def _optimize_combined(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Run Adam for ≈65 % of the allocated time, then finish with sub‑gradient."""
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    smooth_deadline = now + 0.65 * remaining
    a_s, sc_s = _optimize_adam(a0, smooth_deadline, rng)

    if time.time() < deadline:
        a_f, sc_f = _optimize_subgrad(a_s, deadline, rng)
        if sc_f < sc_s:
            return a_f, sc_f
    return a_s, sc_s
