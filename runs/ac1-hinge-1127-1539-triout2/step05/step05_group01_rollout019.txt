    def init_powerlaw(n):

        p = np.linspace(0.45, 0.80, rng.integers(5, 15))
        shift = rng.uniform(0.5,5.0)
        # compute raw = (i+shift)^{-p}
        i = np.arange(n, dtype=float)
        raw = (i + shift) ** (-p)
        raw = raw / raw.sum()
        return raw
    def init_exponential(n):
        lam = rng.uniform(0.001,0.01)  # small lam
        i = np.arange(n, dtype=float)
        raw = np.exp(-lam * i)
        raw = raw / raw.sum()
        return raw

        p = np.zeros(n)
        p[idx] = 1.0 / m
        return p

def _adam_optimize(a0, deadline, target_sum):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9; beta2 = 0.999
    eps=1e-8
    t=0
    beta = 1.0
    max_beta = 1e6
    while time.time() < deadline:
        g, conv = _gradient_softmax(a, beta)
        t += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = 0.1 * m_hat / (np.sqrt(v_hat) + eps)  # learning_rate 0.1
        a = a - step
        # project
        a = _project_to_simplex(a, target_sum)
        # jitter occasionally
        if t % 200 == 0:
            jitter = np.random.randn(a.shape[0]) * target_sum * 0.001
            a = a + jitter
            a = _project_to_simplex(np.maximum(a, 0), target_sum)
        # increase beta
        if t % 100 == 0 and beta < max_beta:
            beta = min(beta * 1.1, max_beta)
    return a

"""
Propose a step‑function (non‑negative sequence) that minimises the
``evaluate_sequence`` objective.

Main ideas
----------
* Work in the probability simplex:  p_i ≥ 0, Σ p_i = 1.
  The heights are a_i = √(2·n)·p_i, therefore the objective is
      O(p) = 2·n·max_k (p*p)[k] .
* Minimise a smooth surrogate of the max – the log‑sum‑exp (soft‑max)
  of the self‑convolution – using Adam (adaptive moments) and a
  simplex projection after each step.
* Use an aggressive β‑annealing schedule (β starts at 1 and is
  multiplied by 1.1 every few hundred iterations) so that the
  surrogate becomes tighter while optimisation proceeds.
* Several random initialisations (uniform, power‑law, exponential,
  sparse) are tried; each gets a fixed share of the total time budget.
* The best candidate is finally polished with a short simulated‑annealing
  (random mass‑transfer moves) that accepts uphill moves with a
  multiplicative cooling schedule.
* Trailing entries that are ≈0 are trimmed – this does not affect the
  objective but makes the output shorter.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, linear convolution)
# ----------------------------------------------------------------------


def _max_convolution_fft(x: np.ndarray) -> float:
    """Maximum entry of the full linear convolution x*x (length 2·n‑1)."""
    n = x.size
    size = 1 << ((2 * n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    conv = np.fft.irfft(fx * fx, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(x: np.ndarray) -> np.ndarray:
    """Full linear convolution x*x (length 2·n‑1)."""
    n = x.size
    size = 1 << ((2 * n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    conv = np.fft.irfft(fx * fx, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex Σx = s, x≥0
# (Duchi et al. 2008)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) surrogate of max(b) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(p: np.ndarray, beta: float):
    """
    Returns (grad, conv) for the soft‑max surrogate

        L(p) = (1/β)·log Σ_k exp(β·conv_k) ,   conv = p * p

    grad = ∇_p L(p) = 2·(p ⋆ q)      where   q_k ∝ exp(β·conv_k).
    """
    n = p.shape[0]

    # ----- convolution p*p -----
    size = 1 << ((2 * n - 1).bit_length())
    p_fft = np.fft.rfft(p, n=size)
    conv = np.fft.irfft(p_fft * p_fft, n=size)[: 2 * n - 1]

    # ----- soft‑max probabilities -----
    max_c = np.max(conv)
    exp_shifted = np.exp(beta * (conv - max_c))
    q = exp_shifted / np.sum(exp_shifted)          # length 2·n‑1

    # ----- gradient g = 2·(p ⋆ q) -----
    # We need the convolution of reversed(p) with q and then take the middle n values.
    p_rev = p[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Pr_fft = np.fft.rfft(p_rev, n=size2)
    Qr_fft = np.fft.rfft(q, n=size2)
    tmp = np.fft.irfft(Pr_fft * Qr_fft, n=size2)[:conv_len]
    grad = 2.0 * tmp[n - 1 : n - 1 + n]       # size n
    return grad, conv


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max surrogate (probability vector)
# ----------------------------------------------------------------------


def _adam_optimize(p0: np.ndarray, sub_deadline: float) -> np.ndarray:
    """
    Adam optimiser to minimise the log‑sum‑exp surrogate of max(p*p).
    The vector stays in the probability simplex (sum = 1, p ≥ 0).
    """
    p = np.copy(p0)                     # probabilities, Σp = 1
    n = p.shape[0]
    m = np.zeros_like(p)                # first moment
    v = np.zeros_like(p)                # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0
    beta = 1.0
    max_beta = 1e6
    rng = np.random.default_rng()

    # learning‑rate schedule – a modest base value multiplied by a factor
    base_lr = 0.2

    while time.time() < sub_deadline:
        t += 1
        g, conv = _gradient_softmax(p, beta)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        # gradient step (ascent on negative loss)
        p = p - step

        # projection back onto simplex Σp = 1
        p = _project_to_simplex(p, 1.0)

        # occasional jitter to escape tiny plateaus
        if t % 500 == 0:
            jitter = rng.normal(scale=1e-3, size=n)
            p = p + jitter
            p = _project_to_simplex(np.maximum(p, 0.0), 1.0)

        # β annealing
        if t % 300 == 0 and beta < max_beta:
            beta = min(beta * 1.1, max_beta)

    return p


# ----------------------------------------------------------------------
# Simple simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Random mass‑transfer SA to polish a candidate."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    T = best_score * 0.02 + 1e-9
    cooling = 0.9993
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        # move a fraction of i's mass to j
        frac = rng.random() * 0.12
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta

        cur = _max_convolution_fft(a)
        dE = cur - best_score
        if cur < best_score or (T > 0 and rng.random() < np.exp(-dE / T)):
            if cur < best_score:
                best_score = cur
                best = a.copy()
        else:
            # revert
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = a / a.sum() * np.sqrt(2.0 * n)
            cur = _max_convolution_fft(a)
            if cur < best_score:
                best_score = cur
                best = a.copy()

    return best


# ----------------------------------------------------------------------
# Families for random initialisations
# ----------------------------------------------------------------------


def _init_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(0.45, 0.80)
    shift = rng.uniform(0.5, 5.0)
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    return raw / raw.sum()


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    lam = rng.uniform(5e-4, 5e-3)          # small decay → slowly decreasing tail
    idx = np.arange(n, dtype=float)
    raw = np.exp(-lam * idx)
    return raw / raw.sum()


def _init_sparse(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sparse support of size ≈√(2·n) with equal masses."""
    m = max(2, int(np.sqrt(2.0 * n)))
    idx = rng.choice(n, size=m, replace=False)
    p = np.zeros(n, dtype=float)
    p[idx] = 1.0 / m
    return p


_INITIALISERS = [
    _init_uniform,
    _init_powerlaw,
    _init_exponential,
    _init_sparse,
]

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:

    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5          # small safety margin

    best_seq = None
    best_obj = np.inf

    # ------------------------------------------------------------------
    # 1. Run several Adam‑optimised candidates (different random starts)
    # ------------------------------------------------------------------
    lengths = [3000, 3500, 4000, 4500]               # diverse sizes
    # allocate roughly equal time to each length

            p_opt = _adam_optimize(p0, sub_dead)

    print(f"[{time.time():.1f}] Finished. Best objective = {best_obj:.6f} "
