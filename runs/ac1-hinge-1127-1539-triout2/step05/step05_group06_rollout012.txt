"""
Proposes a step‑function (non‑negative vector) that minimises the
self‑convolution peak evaluated by ``evaluate_sequence``.

Approach
--------
* Work on the ℓ₁‑simplex Σ a_i = √(2·n).  The objective becomes simply
  max_i (a∗a)[i] (the maximum entry of the full convolution).
* Optimise a smooth approximation of the max by minimising
  f_α(p) = (1/α)·log Σ exp(α·(p∗p))[i]  with p_i ≥ 0, Σ p_i = 1.
  The gradient is obtained analytically using FFTs.
* Parameterise p_i = exp(z_i) / Σ exp(z_j) (no constraints on z).
  L‑BFGS‑B is used to optimise the smooth objective for a sequence of
  increasing “temperature’’ parameters α (5, 20, 80, 250) – a continuation
  scheme that gradually sharpens the approximation of the true max.
* Perform many short random restarts (different lengths n ∈ [800,4000]).
  Keep the best sequence found.
* When time permits, up‑sample the current best solution to a larger
  length (up to 8000) and run the same optimisation again – more degrees
  of freedom usually allow a flatter convolution and a lower peak.

The code is fully self‑contained, uses only NumPy, SciPy and the provided
``evaluate_sequence`` function, and respects the 1000 s time budget.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Global variable used by the L‑BFGS wrapper (avoids closures)
# ----------------------------------------------------------------------
_CURRENT_ALPHA = None  # will be set before each optimisation phase


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient term for the smooth max‑norm.
    Implements g_j = 2 Σ_i w_{i+j} a_i .
    Both ``a`` (len n) and ``w`` (len 2·n‑1) are real vectors.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Smooth max objective (log‑sum‑exp) and its gradient w.r.t. z
# ----------------------------------------------------------------------
def _smooth_max_obj(z: np.ndarray, alpha: float):
    """
    Returns (f, g) where
        f = (1/α)·log Σ_i exp(α·conv_i)   (smooth approximation of max)
        g = ∇_z f
    ``z`` are unconstrained log‑probabilities (p_i = exp(z_i)/Σexp(z)).
    """
    # --- p = softmax(z) -------------------------------------------------
    max_z = np.max(z)
    exp_z = np.exp(z - max_z)
    p = exp_z / exp_z.sum()

    # --- conv = p * p ---------------------------------------------------
    conv = _full_convolution(p)

    # --- softmax weights w_i = exp(α·conv_i) / Σ exp(α·conv) ------------
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    sum_exp = exp_shifted.sum()
    w = exp_shifted / sum_exp

    # smooth objective value
    f = (max_scaled + np.log(sum_exp)) / alpha

    # --- gradient wrt p -------------------------------------------------
    grad_p = _conv_with_weights(p, w)   # shape (n,)

    # --- gradient wr.t. z (chain rule for softmax) ----------------------
    avg = np.dot(p, grad_p)
    g = p * (grad_p - avg)
    return f, g


def _smooth_max_obj_wrapper(z: np.ndarray):
    """
    Wrapper required by ``scipy.optimize.minimize``.
    Uses the global ``_CURRENT_ALPHA``.
    """
    return _smooth_max_obj(z, _CURRENT_ALPHA)


# ----------------------------------------------------------------------
# Linear interpolation for up‑sampling
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a to length new_n."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=float)
    return np.interp(x_new, x_old, a)


# ----------------------------------------------------------------------
# Core optimisation routine for a single length n
# ----------------------------------------------------------------------
def _optimise_length(n: int, rng: np.random.Generator, deadline: float):
    """
    Runs a continuation L‑BFGS optimisation for a vector of length n.
    Returns the best vector a (sum = sqrt(2·n)) found within the time slice.
    """
    # initialise a random probability vector p (sum = 1)
    p0 = rng.random(n)
    eps = 1e-12
    p0 = np.maximum(p0, eps)
    p0 /= p0.sum()
    z = np.log(p0)                     # unconstrained parameters

    # continuation schedule for the temperature α
    alphas = [5.0, 20.0, 80.0, 250.0]
    for alpha in alphas:
        global _CURRENT_ALPHA
        _CURRENT_ALPHA = alpha

        # decide how many L‑BFGS iterations we can afford this phase
        # (a rough estimate: ~300 iterations is well under a second for n≈2000)
        max_iter = 300
        # run L‑BFGS‑B
        res = _opt.minimize(_smooth_max_obj_wrapper, z,
                            method='L-BFGS-B', jac=True,
                            options={'maxiter': max_iter, 'disp': False})
        z = res.x
        # stop early if we run out of time
        if time.time() > deadline - 5.0:
            break

    # convert back to a vector with the required ℓ₁ norm √(2·n)
    max_z = np.max(z)
    p = np.exp(z - max_z)
    p /= p.sum()
    S = np.sqrt(2.0 * n)
    a = p * S
    return a

    finally attempts a refinement on a larger up‑sampled version of the

    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin for the final external evaluation
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_score = np.inf
    attempts = 0

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    while time.time() < deadline - 20.0:          # keep ~20 s for final step
        attempts += 1

            p /= p.sum()
            a_big = p * S_big
