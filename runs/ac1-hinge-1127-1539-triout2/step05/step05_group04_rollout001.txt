```python
def upsample_sequence(a, new_n):
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    # Interpolate linearly over indices
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_new = np.interp(new_x, old_x, a)
    # project to simplex to get correct sum
    target_sum = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target_sum)
    return a_new
```

"""
Propose a non‑negative step function (a list of heights) that minimizes
the official ``evaluate_sequence`` score.

The objective, after normalising the ℓ¹‑norm Σa = √(2 n), is simply
the maximum entry of the self‑convolution a*a.  The algorithm works
in three stages:

1. **Exploration** – for several lengths (2000 … 20000) many random
   start points are generated and a short projected gradient descent on a
   smooth approximation of the max‑norm (log‑sum‑exp) is performed.
   The best candidate is kept.

2. **Refinement** – the current best sequence for a given length is
   refined with a longer smooth‑max descent, an exact sub‑gradient descent
   on the true max‑norm and a cheap greedy “mass‑balance’’ move that
   transfers mass from the heaviest contributor to the lightest one.
   After each projection the vector is forced to be symmetric
   (a[i] = a[n‑1‑i]) because symmetry dramatically reduces the peak of
   the convolution.

3. **Polishing** – a light simulated‑annealing pass is run on whatever
   time remains.

All steps respect the simplex constraint (non‑negative, prescribed
ℓ¹‑norm) using the Euclidean projection of Duchi et al.  The code
periodically checks the wall‑clock deadline (default 1000 s) and stops
as soon as the budget is exhausted, returning the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1, dtype=float)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Utilities for FFT based convolutions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Under the ℓ¹‑normalisation Σa = √(2 n) the official score equals max(a*a)."""
    return float(_max_convolution_fft(a).max())


# ----------------------------------------------------------------------
# Symmetrisation helper
# ----------------------------------------------------------------------
def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Force a = reverse(a) by averaging; retains ℓ¹‑norm."""
    return (a + a[::-1]) * 0.5


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) gradient utilities – Phase 1
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) where g = 2·(a ⋆ p) and p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = float(np.max(b))
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3 n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _smooth_max_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Annealed projected gradient descent on the smooth max.
    Momentum, adaptive step‑size and occasional jitter help escape plateaus.
    Symmetry is enforced after each projection.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 8192.0
    beta_factor = 1.5

    # step size is a fraction of the ℓ¹‑norm
    step_base = target_sum * 0.18
    momentum = np.zeros_like(a)
    rho = 0.90

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        # enforce simplex, then symmetrise and re‑project (tiny numerical drift)
        a = _project_to_simplex(a, target_sum)
        a = _symmetrize(a)
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)

        # anneal β
        if it % 400 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑norm – Phase 2
# ----------------------------------------------------------------------
def _max_index_subgradient(a: np.ndarray):
    """
    Sub‑gradient of max_k (a*a)_k.
    Returns (g, k*) where k* is an argmax index.
    """
    n = a.shape[0]
    conv = _max_convolution_fft(a)
    k = int(np.argmax(conv))

    g = np.zeros_like(a)
    if k < n:                         # left half
        g[:k + 1] = 2.0 * a[k::-1]
    else:                             # right half
        d = k - (n - 1)               # shift amount (0 … n‑1)
        g[d:] = 2.0 * a[: n - d]
    return g, k


def _subgradient_descent_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected sub‑gradient descent on max_k (a*a)_k.
    Diminishing step size η_t = α/√t, with symmetry enforcement.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        g, _ = _max_index_subgradient(a)

        step = 0.85 * target_sum / (np.sqrt(it + 1) * np.linalg.norm(g) + 1e-12)
        a -= step * g

        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)
        a = _symmetrize(a)
        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            a = _symmetrize(a)
            a = _project_to_simplex(a, target_sum)

        it += 1
    return a


# ----------------------------------------------------------------------
# Cheap mass‑balance refinement – Phase 2 (b) 
# ----------------------------------------------------------------------
def _mass_balance_refine(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Greedy move that reduces the contribution of the currently maximal
    convolution entry.  At each iteration a small fraction of the most
    offending coefficient is transferred to a light coefficient.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sum(a)   # should already equal √(2 n)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _max_convolution_fft(a)
        k = int(np.argmax(conv))

        # contributions of each index to entry k
        idx = np.arange(n)
        other = k - idx
        mask = (other >= 0) & (other < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[other[mask]]

        i_high = int(np.argmax(contrib))

        # pick a very light position (below 20‑th percentile)
        low_candidates = np.where(a <= np.percentile(a, 20))[0]
        if low_candidates.size == 0:
            i_low = int(np.argmin(a))
        else:
            i_low = int(rng.choice(low_candidates))

        delta = a[i_high] * 0.028   # 2.8 % of the heavy mass
        if delta < 1e-14:
            break

        a[i_high] -= delta
        a[i_low]  += delta

        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)
        a = _symmetrize(a)
        a = _project_to_simplex(a, target_sum)

        if it % 120 == 0:
            if rng.random() < 0.02:
                jitter = rng.normal(scale=target_sum * 9e-4, size=n)
                a += jitter
                a = np.maximum(a, 0.0)
                a = _project_to_simplex(a, target_sum)
                a = _symmetrize(a)
                a = _project_to_simplex(a, target_sum)

        it += 1
    return a


# ----------------------------------------------------------------------
# Light simulated‑annealing polish – Phase 3
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Very light SA that moves a random fraction of mass between two sites,
    accepting uphill moves with a slowly decreasing temperature.
    Symmetry is restored after each accepted move.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    cur_score = _score(a)
    best_score = cur_score

    T = cur_score * 0.018 + 1e-9
    cooling = 0.9993
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-16:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        frac = rng.random() * 0.14
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta

        # symmetrise to keep the search in the symmetric subspace
        a = _symmetrize(a)
        a = _project_to_simplex(a, np.sqrt(2.0 * n))

        new_score = _score(a)
        dE = new_score - cur_score
        accept = dE <= 0.0 or (T > 0.0 and rng.random() < np.exp(-dE / T))

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
        else:
            # revert
            a[j] -= delta
            a[i] += delta
            a = _symmetrize(a)
            a = _project_to_simplex(a, np.sqrt(2.0 * n))

        T *= cooling

    return best


# ----------------------------------------------------------------------
# Random initialisation helpers
# ----------------------------------------------------------------------
def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Return a random point on the simplex Σa = √(2 n).
    Several families are mixed to increase diversity.
    The result is symmetrised.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                      # power‑law
        p = rng.uniform(0.12, 1.15)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                    # exponential
        lam = rng.exponential(scale=0.018)
        a = np.exp(-lam * idx)

    elif family == 2:                    # power × exponential
        p = rng.uniform(0.12, 1.0)
        lam = rng.exponential(scale=0.018)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                    # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n))

    elif family == 4:                    # piecewise exponential
        split = rng.integers(int(0.25 * n), int(0.75 * n))
        lam1 = rng.exponential(scale=0.018)
        lam2 = rng.exponential(scale=0.018)
        a = np.concatenate([np.exp(-lam1 * np.arange(split)),
                             np.exp(-lam2 * np.arange(split, n))])

    else:                                # almost uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.0012, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    a = _symmetrize(a)
    a = _project_to_simplex(a, target)
    return a


# ----------------------------------------------------------------------
# Upsampling helper – enlarges a solution while preserving shape
# ----------------------------------------------------------------------
def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, followed by projection."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_x, old_x, a)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    a_new = _symmetrize(a_new)
    a_new = _project_to_simplex(a_new, target)
    return a_new

    if best_seq is not None:
        left = deadline - time.time()
        if left > 5.0:
            a_final = _simulated_annealing(best_seq, deadline)
            sc = _score(a_final)
            if sc < best_score:
                best_score = sc
                best_seq = a_final.copy()
                print(f"[{time.time():.1f}] Final SA improvement: score={sc:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zeros and validate via the official function
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform distribution of length 1000 (should never happen)
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
