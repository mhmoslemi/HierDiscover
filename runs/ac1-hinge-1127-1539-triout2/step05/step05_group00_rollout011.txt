def propose_candidate(seed=0, budget_s=1000):
    start = time.time()
    deadline = start + budget_s - 0.5  # margin
    rng = np.random.default_rng(seed)

    # Determine final length based on time remaining, maybe up to 25000
    # choose n_final = 20000
    n_final = 20000

    # Phase 1: search using smaller length for many restarts
    n_small = 4096
    best_a = None
    best_obj = np.inf

    # allocate time for Phase 1: 0.5 * (deadline-start)
    phase1_deadline = start + (deadline - start) * 0.5

    while time.time() < phase1_deadline:
        # generate random initial a of length n_small
        a0 = _random_initial_sequence(n_small, rng)
        # run gradient descent for short period, maybe 5 seconds each?
        gd_dead = min(phase1_deadline, time.time() + 5)
        a_opt = _adam_optimize(a0, gd_dead)
        # evaluate
        obj = _max_convolution_fft(a_opt)  # equal to max_b

            best_obj = obj
            best_a = a_opt.copy()
            print(f"[{time.time():.1f}] Phase1 new best obj={obj:.6f}")

    # Phase 2: upsample best a to final length
    if best_a is None:
        # fallback uniform
        best_a = np.full(n_small, np.sqrt(2 * n_small) / n_small)
    a_big = _upsample_sequence(best_a, n_final)

    # Phase 2 gradient descent longer
    gd_dead2 = min(deadline, time.time() + (deadline - time.time()) * 0.4)
    a_big_opt = _adam_optimize(a_big, gd_dead2)
    obj_big = _max_convolution_fft(a_big_opt)
    if obj_big < best_obj:
        best_obj = obj_big
        best_seq = a_big_opt.copy()
        print(f"[{time.time():.1f}] Phase2 GD best obj={obj_big:.6f}")
    else:
        best_seq = best_a.copy() if best_a.size == n_final else _upsample_sequence(best_a, n_final)

    # Phase 3: hill climbing for remaining time
    if time.time() < deadline - 2.0:
        b_full = _full_convolution(best_seq)
        final_dead = deadline
        a_final, b_final = _hill_climbing(best_seq, b_full, final_dead)
        final_obj = b_final.max()
        if final_obj < best_obj:
            best_obj = final_obj
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final hill climbing improvement {final_obj:.6f}")

def _adam_optimize(a0: np.ndarray, deadline: float) -> np.ndarray:
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)
    
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    # Temperature schedule
    beta = 0.5
    beta_max = 16384.0
    beta_factor = 1.5
    step_scale = target * 0.1   # base step

    rng = np.random.default_rng()
    while time.time() < deadline:
        t += 1
        # compute gradient
        g, _ = _gradient_softmax(a, beta)
        # update moments
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        # bias correction
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        # step size maybe adapt with beta
        lr = step_scale / (np.sqrt(v_hat) + eps)
        a -= lr * m_hat
        # projection onto simplex
        a = _project_to_simplex(a, target)

        # jitter occasionally
        if t % 200 == 0:
            jitter = rng.normal(scale=target * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

        # anneal beta
        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a

"""
Step‑function optimizer (new version).

The search proceeds in three phases:

1. **Exploration on a modest size (≈4 k).**
   Many random starts are generated (uniform, power‑law, exponential,
   Dirichlet, and a deterministic arcsine shape).  Each start is refined
   by a projected Adam optimiser that minimises a smooth log‑sum‑exp
   approximation of the max‑convolution.  The best vector found is kept.

2. **Upsampling + long Adam run.**
   The best vector from phase‑1 is linearly interpolated to a larger
   length (≈20 k) and projected onto the ℓ¹‑simplex
   Σa = √(2 n).  A longer Adam optimisation (higher temperature, more
   iterations) is performed on this enlarged vector.

3. **Exact hill‑climbing (mass‑transfer).**
   Starting from the Adam result we keep the full convolution a*a and
   repeatedly move a random fraction of mass from a “large’’ entry to a
   “small’’ one.  The convolution update is performed in O(n) using the
   analytical Δb formula, and Metropolis‑style acceptance with an
   exponential cooling schedule allows occasional uphill moves.

All vectors are kept non‑negative and projected onto the simplex after
each modification, so the official ``evaluate_sequence`` sees a valid
step‑function.  The routine respects a 1000 s wall‑clock budget and
returns the best sequence discovered as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    { x | x_i ≥ 0, Σ x_i = s } (Duchi et al. 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families for random initialisation (including arcsine)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative vector on the ℓ¹‑simplex Σa = √(2 n).
    Several families are sampled uniformly.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 7)          # 0‑5 were used before, 6 = arcsine
    idx = np.arange(n, dtype=float)

    if family == 0:                      # pure power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                    # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                    # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                    # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                    # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )

    elif family == 5:                    # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    else:                                # deterministic arcsine shape
        eps = 1e-6
        i = idx + 0.5
        j = (n - 0.5) - idx
        a = 1.0 / np.sqrt((i + eps) * (j + eps))

    # Add a little noise, enforce non‑negativity and project onto simplex
    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a onto a new length and projection onto simplex."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with g = 2·(a ⋆ p) and pₖ ∝ exp(β·bₖ).
    """
    n = a.shape[0]

    # b = a*a (full linear convolution)
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities over the convolution entries
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]                      # reversed a
    conv_len = n + (2 * n - 1) - 1       # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _adam_optimize(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected Adam optimiser on the smooth log‑sum‑exp of the convolution.
    The temperature β is annealed up to a large value to approach the true
    max‑norm.
    """
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # temperature schedule (soft‑max)
    beta = 0.5
    beta_max = 16384.0
    beta_factor = 1.5

    # base learning‑rate (scalar)
    base_lr = target * 0.12

    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        # gradient of the smoothed max
        g, _ = _gradient_softmax(a, beta)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # parameter update
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # projection onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target)

        # occasional jitter to escape tiny plateaus
        if t % 250 == 0:
            jitter = rng.normal(scale=target * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

        # anneal temperature
        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Mass‑transfer hill climbing (exact convolution updates)
# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Simulated‑annealing style hill climbing using exact convolution updates.
    A random source index (weighted by its mass) donates a random fraction of
    its mass to a uniformly chosen destination index.  The move is accepted
    if it lowers the peak of the convolution, or with a Metropolis probability
    that decays exponentially with temperature T.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()
    best_b = b.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.02 + 1e-9   # initial temperature
    cooling = 0.9997

    while time.time() < deadline:
        # source index proportional to its current mass
        if a.sum() == 0.0:
            break
        probs = a / a.sum()
        i = rng.choice(n, p=probs)

        if a[i] < 1e-14:
            continue

        # destination (uniform)
        j = rng.integers(0, n)
        if i == j:
            continue

        # amount to move (up to 50 % of source)
        max_delta = a[i] * 0.5
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ----- update convolution analytically -----
        b_new = b.copy()
        # linear part – shifting the whole vector a by i and j
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a

        # quadratic corrections
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # ----- Metropolis acceptance -----
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-(new_score - cur_score) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                best_b = b.copy()
                print(f"[{time.time():.1f}] hill‑climb improvement: {best_score:.6f}")

        T *= cooling

    return best_a, best_b

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects a 1000 s wall‑clock
    budget and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Phase 1 – many short Adam runs on a modest size (≈4096)
    # ------------------------------------------------------------------
    n_small = 4096
    target_small = np.sqrt(2.0 * n_small)
    phase1_deadline = min(deadline, start_time + (deadline - start_time) * 0.45)

    while time.time() < phase1_deadline:
        a0 = _random_initial_sequence(n_small, rng)
        # give each Adam run at most 5 s (or whatever remains)
        run_dead = min(phase1_deadline, time.time() + 5.0)
        a_opt = _adam_optimize(a0, run_dead)
        score = _max_convolution_fft(a_opt)   # equal to evaluate_sequence value
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] Phase 1 new best  n={n_small}  score={score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – upsample to a larger size and perform a long Adam run
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution
        best_seq = np.full(n_small, target_small / n_small)

    n_large = 20000                     # final ambition (fits easily in 1000 s)
    a_large = _upsample_sequence(best_seq, n_large)

    # allocate the bulk of the remaining time to the final Adam optimisation
    remaining = deadline - time.time()
    phase2_deadline = time.time() + remaining * 0.45
    a_large_opt = _adam_optimize(a_large, phase2_deadline)

    score_large = _max_convolution_fft(a_large_opt)
    if score_large < best_score:
        best_score = score_large
        best_seq = a_large_opt.copy()
        print(f"[{time.time():.1f}] Phase 2 Adam improvement  n={n_large}  score={score_large:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exact hill climbing on the current best vector
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        b_full = _full_convolution(best_seq)
        a_final, b_final = _hill_climbing(best_seq, b_full, deadline)
        final_score = b_final.max()
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Phase 3 hill‑climb improvement  score={final_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up – drop trailing near‑zeros for nicer output
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Final official evaluation (for logging)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:   # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
