def _subgradient_opt(a0, deadline):
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()
    step_base = target_sum * 0.05
    while time.time() < deadline:
        # compute convolution
        b = _full_convolution(a)
        k_max = int(np.argmax(b))
        # compute subgradient
        g = np.zeros_like(a)
        if k_max < n:
            idx = np.arange(k_max+1)
            g[idx] = 2 * a[k_max - idx]
        else:
            start = k_max - n + 1
            idx = np.arange(start, n)
            g[idx] = 2 * a[k_max - idx]

        # step size inversely proportional to norm
        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            step = step_base / (norm_g + 1e-12)
            a -= step * g

        # project onto simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if rng.random() < 0.01:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

    return a

"""
Improved step‑function optimizer.

The strategy has three main phases (all bound by the wall‑clock deadline):

1. **Multiple restarts + Adam on a smoothed max** –  a soft‑max
   approximation of the infinity‑norm of the self‑convolution is minimised.
   Adam gives fast, well‑scaled updates.  The temperature β is annealed up to
   1e5 so that the smooth objective becomes almost the true max.

2. **Sub‑gradient refinement** – after the smooth phase we switch to a true
   sub‑gradient of the max (gradient of the active shift).  This sharpens the
   solution very quickly.

3. **Exact hill‑climbing** – a cheap O(n) mass‑transfer move (the analytical
   update of the full convolution) is performed with a simple annealing
   schedule to escape shallow local minima.

The algorithm explores several lengths (up to ≈15000) and keeps the best
sequence found.  All vectors are projected onto the ℓ¹‑simplex
 Σa = √(2 n) after each modification, guaranteeing that the official
`evaluate_sequence` sees a valid step‑function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returned as a NumPy array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed maximum (log‑sum‑exp) and its gradients
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of   (1/β)·log Σ exp(β·b)   where   b = a*a .
    Returns (g, b) with g = 2·(a ⋆ p) and pₖ ∝ exp(β·bₖ).
    """
    n = a.shape[0]

    # b = a*a (full convolution)
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _adam_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Adam optimisation of the smoothed max.
    The inverse temperature β is annealed up to 1e5.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # start with a modest temperature, increase exponentially
    beta = 0.5
    beta_max = 1e5
    beta_factor = 1.5

    lr_base = target_sum * 0.15
    eps = 1e-8

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        t += 1
        m = 0.9 * m + 0.1 * g
        v = 0.999 * v + 0.001 * (g * g)

        m_hat = m / (1.0 - 0.9 ** t)
        v_hat = v / (1.0 - 0.999 ** t)

        step = lr_base / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid plateaus
        if t % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β‑annealing
        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


def _subgradient_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Sub‑gradient descent on the true max ‖a*a‖∞.
    At each step the active shift k* is identified and the gradient
    of that shift (2·a shifted) is used.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    rng = np.random.default_rng()
    step_coeff = target_sum * 0.08

    while time.time() < deadline:
        b = _full_convolution(a)
        k = int(np.argmax(b))

        # sub‑gradient of the active shift
        g = np.zeros_like(a)
        if k < n:
            idx = np.arange(k + 1)
            g[idx] = 2.0 * a[k - idx]
        else:
            start = k - n + 1
            idx = np.arange(start, n)
            g[idx] = 2.0 * a[k - idx]

        norm = np.linalg.norm(g)
        if norm > 0.0:
            step = step_coeff / (norm + 1e-12)
            a -= step * g

        a = _project_to_simplex(a, target_sum)

        # tiny random jitter occasionally
        if rng.random() < 0.01:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

    return a


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative sequence on the ℓ¹‑simplex Σa = √(2 n).
    Several families are sampled uniformly.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # pure power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation onto a new length and projection onto the
    appropriate simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Exact hill‑climbing (uses analytical convolution updates)
# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Mass‑transfer hill‑climbing with analytic updates of the full
    convolution.  The routine returns the best pair (a, b) seen before
    the deadline.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()
    best_b = b.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.02 + 1e-9
    cooling = 0.9997

    while time.time() < deadline:
        # source proportional to its current mass
        if a.sum() == 0.0:
            break
        i = rng.choice(n, p=a / a.sum())
        if a[i] < 1e-14:
            continue

        # destination uniformly
        j = rng.integers(0, n)
        if i == j:
            continue

        # transfer a random fraction (up to 50 % of the source)
        delta = rng.random() * 0.5 * a[i]
        if delta <= 0.0:
            continue

        # ---------- update convolution analytically ----------
        b_new = b.copy()
        # -2·δ·a contribution from the source
        b_new[i:i + n] -= 2.0 * delta * a
        # +2·δ·a from the destination
        b_new[j:j + n] += 2.0 * delta * a

        # quadratic terms for the three affected indices
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # ---------- Metropolis acceptance ----------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if rng.random() < np.exp(-(new_score - cur_score) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                best_b = b.copy()
                print(f"[{time.time():.1f}] hill‑climb improvement: {best_score:.6f}")

        T *= cooling

    return best_a, best_b


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock budget (default 1000 s) and returns
    the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Optional seed from a previously stored solution
    # ------------------------------------------------------------------
    earlier = globals().get("height_sequence_1")
    if isinstance(earlier, (list, tuple, np.ndarray)):
        a0 = np.asarray(earlier, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        best_seq = a0.copy()
        best_score = _max_convolution_fft(a0)
        print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Length schedule – we start modest and increase while time permits
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000, 15000]

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Decide how many random restarts we can afford for this length
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        # split remaining time roughly evenly among the lengths we have left
        lengths_left = len(lengths) - lengths.index(n)
        per_len = remaining / max(1, lengths_left)

        restarts = max(1, int(per_len // 30))   # ~30 s per restart
        for r in range(restarts):
            if time.time() >= deadline:
                break

            # ----------------------------------------------------------
            # Initialisation
            # ----------------------------------------------------------
            if best_seq is not None and best_seq.size != n:
                a = _upsample_sequence(best_seq, n)
            else:
                a = _random_initial_sequence(n, rng)

            a = _project_to_simplex(a, target_sum)

            # ----------------------------------------------------------
            # Phase 1 – Adam on smoothed max
            # ----------------------------------------------------------
            phase1_dead = min(deadline,
                               time.time() + min(20.0, (deadline - time.time()) * 0.3))
            a = _adam_opt(a, phase1_dead)
            score1 = _max_convolution_fft(a)

            # ----------------------------------------------------------
            # Phase 2 – sub‑gradient refinement
            # ----------------------------------------------------------
            if time.time() < deadline:
                phase2_dead = min(deadline,
                                  time.time() + min(15.0, (deadline - time.time()) * 0.2))
                a = _subgradient_opt(a, phase2_dead)
                score2 = _max_convolution_fft(a)
            else:
                score2 = score1

            # ----------------------------------------------------------
            # Phase 3 – exact hill‑climbing
            # ----------------------------------------------------------
            if time.time() < deadline:
                b_full = _full_convolution(a)
                phase3_dead = min(deadline,
                                  time.time() + min(20.0, (deadline - time.time()) * 0.3))
                a, b_full = _hill_climbing(a, b_full, phase3_dead)
                score3 = b_full.max()
            else:
                score3 = score2

            # ----------------------------------------------------------
            # Keep the best candidate seen so far
            # ----------------------------------------------------------
            cur_best = min(score1, score2, score3)
            if cur_best < best_score:
                best_score = cur_best
                best_seq = a.copy()
                print(f"[{time.time():.1f}] New global best (n={n}) score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with any remaining budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a = best_seq.copy()
        b = _full_convolution(a)
        a, b = _hill_climbing(a, b, deadline)
        final_score = b.max()
        if final_score < best_score:
            best_score = final_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Final polishing improvement: {final_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zeros for neatness
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Report the official evaluation (objective = max of the self‑convolution)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
