"""
Purpose
-------
Search for a non‑negative step‑function (a list of heights) that minimises the
benchmark score defined by ``evaluate_sequence``.  The score, when the
sequence is normalised to Σa = sqrt(2 n) (the usual normalisation), is exactly
the maximum entry of the auto‑convolution a*a.  Therefore the optimisation
problem reduces to:

    minimise   max_k (a * a)[k]    subject to   a_i ≥ 0 ,  Σ a_i = sqrt(2 n)

The programme proceeds in three stages:

1.  **Fast parameter sweep** – generate many simple families of sequences
    (power‑law, symmetric power‑law, symmetric exponential) with a single
    tunable parameter; keep the best few.

2.  **Projected gradient descent on a soft‑max surrogate** – starting from the
    survivors of the sweep, run a few iterations of gradient descent on a
    smooth approximation of the max‑operator.  The temperature β is increased
    up to 1024, giving a very sharp surrogate.

3.  **Linear‑programming based “max‑reduction” step** – for the current
    iterate f, solve the LP

        maximise   Σ b_j
        subject to   (f * b)_k ≤ M   for all k   and   b_j ≥ 0 ,

    where M = max_k (f * f)[k].  The optimal b is rescaled to the required
    ℓ¹‑norm and mixed with f (convex combination).  Repeating this step
    monotonically reduces the maximum convolution value.

The three stages are interleaved while time permits.  The best sequence found
before the time budget expires is returned as a plain ``list[float]``.
"""

import time
import numpy as np
import scipy.optimize  # for the linear programming step


# ----------------------------------------------------------------------
#  Basic convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """
    Return the maximum entry of the linear convolution a*a using FFT.
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using FFT (returns the whole 2n‑1 vector).
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score_from_seq(a: np.ndarray) -> float:
    """
    The benchmark score for a normalised sequence.
    If the sum deviates from sqrt(2 n) the score is still 2·n·max/Σ²,
    but the optimisation loops keep the sum fixed, so we simply return the
    maximum convolution entry.
    """
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
#  Simple feasibility helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Implements the algorithm of Duchi, Shalev‑Shwartz, Singer & Chandra (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _normalise(a: np.ndarray) -> np.ndarray:
    """
    Scale a to satisfy Σa = sqrt(2 n) and enforce non‑negativity.
    """
    n = a.size
    target = np.sqrt(2.0 * n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


# ----------------------------------------------------------------------
#  Families of analytically parametrised sequences (used in the fast sweep)
# ----------------------------------------------------------------------


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i = (i+1)^(-p)  (unnormalised)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    return i ** (-p)


def _sym_power_law(n: int, p: float) -> np.ndarray:
    """
    Symmetric version: (i+1)^(-p) + (n-i)^(-p).  The shape is high at the
    two ends and low in the middle.
    """
    i = np.arange(n, dtype=np.float64)
    left = (i + 1) ** (-p)
    right = (n - i) ** (-p)
    return left + right


def _sym_exponential(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential: exp(lam * (i - (n‑1)/2))."""
    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    return np.exp(lam * i)


def _random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex (Dirichlet(1,…,1))."""
    return rng.dirichlet(np.ones(n))


# ----------------------------------------------------------------------
#  Gradient descent on a soft‑max approximation of the max‑convolution
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of   S_beta(a) = log Σ_k exp(beta * (a*a)[k]) / beta
    which smoothly approximates max_k (a*a)[k].
    Returns the gradient vector and the exact convolution vector.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_c = conv.max()
    # soft‑max probabilities
    exps = np.exp(beta * (conv - max_c))
    p = exps / exps.sum()

    # gradient = 2 * (reverse(a) * p) convolved
    a_rev = a[::-1]
    rev_fft = np.fft.rfft(a_rev, n=size)
    p_fft = np.fft.rfft(p, n=size)
    grad_full = np.fft.irfft(rev_fft * p_fft, n=size)[: 2 * n - 1]
    # the relevant part is indices n‑1 … 2n‑2 (i.e. the forward direction)
    grad = 2.0 * grad_full[n - 1: n - 1 + n]
    return grad, conv


def _gradient_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the soft‑max surrogate.
    The temperature β is increased geometrically up to 1024.
    """
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)

    beta = 1.0
    max_beta = 1024.0
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        grad, _ = _softmax_gradient(a, beta)

        # simple diminishing‑step rule
        step = 0.5 / (beta + 1.0) / (np.linalg.norm(grad) + 1e-12)
        a -= step * grad
        a = _project_to_simplex(a, target)

        # occasional jitter to avoid stagnation
        if it % 200 == 0:
            jitter = rng.normal(scale=target * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = _project_to_simplex(a, target)

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return a


# ----------------------------------------------------------------------
#  Linear‑programming based “max‑reduction” step
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Construct the matrix A of the linear map b ↦ f*b (full linear convolution).
    Row k contains f_{k‑j} for all valid j.  Shape: (2n‑1) × n.
    """
    n = f.size
    rows = 2 * n - 1
    A = np.empty((rows, n), dtype=np.float64)
    for k in range(rows):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        idx = np.arange(j_start, j_end + 1, dtype=np.int64)
        A[k, idx] = f[k - idx]
    return A


def _lp_improvement(f: np.ndarray, deadline: float) -> np.ndarray:
    """
    Repeatedly solve the LP
        maximise Σ b_j   subject to   (f*b)_k ≤ M   and   b ≥ 0,
    where M = max_k (f*f)_k.
    The optimiser mixes the current iterate with the scaled LP solution.
    The process stops when time runs out or when no improvement is observed.
    """
    n = f.size
    target = np.sqrt(2.0 * n)
    best = f.copy()
    best_val = _score_from_seq(best)

    # Parameters controlling the convex‑combination step
    t_start = 0.6
    t_end = 0.05

    it = 0
    while time.time() < deadline:
        # current maximum convolution value
        conv_f = _conv_full_fft(f)
        M = conv_f.max()

        # Build the LP matrix (depends on f)
        A = _build_conv_matrix(f)
        ub = np.full(A.shape[0], M, dtype=np.float64)

        # maximise Σ b  → minimise -Σ b
        c = -np.ones(n, dtype=np.float64)

        # Solve LP with HiGHS (fast interior‑point implementation)
        res = scipy.optimize.linprog(
            c, A_ub=A, b_ub=ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": max(0.1, deadline - time.time())}
        )
        if not res.success:
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 1e-12:
            break

        # Rescale to the required ℓ¹‑norm
        scale = target / sum_b
        g = b * scale

        # Convex combination – step size shrinks with the iteration count
        t = t_start * ((t_end / t_start) ** (it / 20.0))  # decay over ~20 steps
        f = (1.0 - t) * f + t * g
        f = _project_to_simplex(f, target)

        cur_val = _score_from_seq(f)
        if cur_val < best_val:
            best_val = cur_val
            best = f.copy()
            print(f"[{time.time():.1f}] LP iteration {it}: score={best_val:.6f}")

        # if improvement becomes negligible we may stop early
        if it > 5 and best_val / cur_val - 1.0 < 1e-4:
            break

        it += 1

    return best


# ----------------------------------------------------------------------
#  Simulated annealing (light‑weight refinement)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Very simple temperature‑controlled random walk.  Moves consist of
    transferring a small amount of mass between two coordinates.
    """
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)
    best = a.copy()
    best_val = _score_from_seq(best)

    start = time.time()
    total = max(deadline - start, 1e-6)
    rng = np.random.default_rng()

    # pre‑allocate FFT size for fast evaluation
    size_fft = 1 << ((2 * n - 1).bit_length())

    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / total
        temp = 0.3 * (0.001 / 0.3) ** tau   # exponential decay 0.3 → 0.001

        # pick two different indices
        i, j = rng.choice(n, size=2, replace=False)
        if a[i] == 0:
            continue
        # propose moving a fraction of mass from i to j
        frac = rng.random() * 0.02 * (1.0 - tau)   # smaller moves later
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta

        # re‑normalise (tiny numerical drift may appear)
        a = _project_to_simplex(a, target)

        # evaluate
        new_val = _score_from_seq(a)

        # Metropolis acceptance
        accept = False
        if new_val < best_val:
            accept = True
        else:
            prob = np.exp(-(new_val - best_val) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            if new_val < best_val:
                best_val = new_val
                best = a.copy()
                print(f"[{time.time():.1f}] SA new best {best_val:.6f}")
        else:
            # revert move
            a[i] += delta
            a[j] -= delta

    return best


# ----------------------------------------------------------------------
#  Main optimisation driver for a single length “n”
# ----------------------------------------------------------------------


def _optimise_length(n: int, deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run the three‑stage optimisation (parameter sweep → gradient descent →
    LP reduction → simulated annealing) for a specific length ``n``.
    Returns the best sequence and its score.
    """
    target = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # 1.  Fast sweep over a few parametric families
    # ------------------------------------------------------------------
    candidates = []

    # a) pure power‑law
    for p in np.linspace(0.25, 0.90, 7):
        v = _power_law(n, p)
        candidates.append(_normalise(v))

    # b) symmetric power‑law
    for p in np.linspace(0.25, 1.20, 9):
        v = _sym_power_law(n, p)
        candidates.append(_normalise(v))

    # c) symmetric exponential
    for lam in np.linspace(0.0005, 0.015, 9):
        v = _sym_exponential(n, lam)
        candidates.append(_normalise(v))

    # d) a few random Dirichlet draws
    for _ in range(5):
        v = _random_dirichlet(n, rng)
        candidates.append(_normalise(v))

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 2.  Refine each candidate (time‑sharing)
    # ------------------------------------------------------------------
    for base in candidates:
        # allocate a small slice of the remaining budget to each base
        now = time.time()
        if now >= deadline:
            break
        # We give at most 1 % of the total remaining time to a single base,
        # but at least 2 seconds.
        remaining = deadline - now
        slice_deadline = now + max(2.0, 0.01 * remaining)

        # Gradient descent on soft‑max
        a = _gradient_descent(base, slice_deadline)

        # LP‑based reduction (if time permits)
        if time.time() < slice_deadline:
            a = _lp_improvement(a, slice_deadline)

        # Final simulated‑annealing polishing
        if time.time() < slice_deadline:
            a = _simulated_annealing(a, slice_deadline)

        score = _score_from_seq(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Length {n} improved to {best_score:.6f}")

    return best_seq, best_score

#  Public entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function with a low benchmark score.
    The routine tries a few different sequence lengths and several
    initialisations, applying gradient descent, an LP‑based max‑reduction
    step and a brief simulated‑annealing refinement.
    The best sequence discovered before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # small safety margin

    best_overall = None
    best_overall_score = float("inf")

    # Try a handful of lengths – 1200, 1500 and 2000 give a good trade‑off
    # between expressive power and computational cost.
    for n in (1200, 1500, 2000):
        cur_deadline = min(deadline, time.time() + (deadline - time.time()) * 0.33)
        seq, sc = _optimise_length(n, cur_deadline, rng)
        if seq is not None and sc < best_overall_score:
            best_overall_score = sc
            best_overall = seq.copy()
            print(f"[{time.time():.1f}] New global best (len={n}) → score={best_overall_score:.6f}")

        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    #   If a global variable ``height_sequence_1`` is present, try a warm‑start.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        warm = np.array(globals()["height_sequence_1"], dtype=np.float64)
        # clip / pad to a reasonable size (use the best length seen so far)
        desired_len = best_overall.shape[0] if best_overall is not None else 1500
        if warm.size > desired_len:
            warm = warm[:desired_len]
        elif warm.size < desired_len:
            warm = np.pad(warm, (0, desired_len - warm.size), "constant")
        warm = _normalise(warm)

        # a short refinement (gradient+SA) limited to the remaining time
        if time.time() < deadline:
            warm = _gradient_descent(warm, deadline)
        if time.time() < deadline:
            warm = _simulated_annealing(warm, deadline)

        warm_score = _score_from_seq(warm)
        if warm_score < best_overall_score:
            best_overall_score = warm_score
            best_overall = warm.copy()
            print(f"[{time.time():.1f}] Warm‑start improved global best → score={best_overall_score:.6f}")

    # ------------------------------------------------------------------
    #  Final housekeeping – strip trailing zeros and return
    # ------------------------------------------------------------------
    if best_overall is None:
        best_overall = np.zeros(1, dtype=float)

    eps = 1e-12
    nz = np.where(best_overall > eps)[0]
    if nz.size:
        best_overall = best_overall[: nz[-1] + 1]

    print(f"[{time.time():.1f}] Final best score={best_overall_score:.6f}, length={len(best_overall)}")
    return best_overall.tolist()
