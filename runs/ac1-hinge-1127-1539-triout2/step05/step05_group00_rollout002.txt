"""
Propose a candidate step‑function that minimizes ``evaluate_sequence``.
The algorithm mixes three ideas:

1. **Random families + simplex projection** – generate many
   non‑negative vectors on the ℓ¹‑simplex Σa = √(2 n) using a variety of
   parametric families (exponential, power‑law, mixed, Dirichlet, etc.).

2. **Adam‑based smoothed‑max gradient descent** – minimise a
   log‑sum‑exp approximation of the convolution maximum.  The gradient
   g = 2·(a ⋆ p) is obtained by FFT.  Adam (first‑ and second‑moment
   adaptation) plus a simple annealing schedule for the smoothing
   parameter β gives fast convergence.

3. **Targeted mass‑transfer hill‑climbing** – after gradient descent we
   keep the exact convolution ``b = a*a`` and repeatedly move a small
   fraction of mass from a source index that contributes heavily to the
   current maximum entry of ``b`` to a destination index that contributes
   little.  The move is accepted if it lowers the maximum or with a
   Metropolis probability (simulated‑annealing) to escape plateaus.
   The update of ``b`` is performed in O(n) using an analytical formula.

The procedure runs for a user‑provided wall‑clock budget (default 1000 s)
and returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra and utility helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a as a NumPy array."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed maximum (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of   (1/β)·log Σ exp(β·b)   where   b = a*a .
    Returns (g, b) with g = 2·(a ⋆ p) and
    pₖ ∝ exp(β·bₖ).  Correlation is performed via FFT.
    """
    n = a.shape[0]

    # full convolution b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]                          # reversed a
    conv_len = n + (2 * n - 1) - 1           # = 3n‑2
    size2 = _next_pow2(conv_len)
    ar_fft = np.fft.rfft(a_rev, n=size2)
    pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(ar_fft * pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _adam_optimize(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Adam‐based projected gradient descent on the smooth max.
    Uses an annealed inverse temperature β and occasional jitter.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # schedule for the smoothing parameter
    beta = 0.5
    beta_max = 8192.0
    beta_factor = 1.4

    # step‑size scaling
    step_base = target_sum * 0.15

    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = step_base / (np.sqrt(v_hat) + eps)
        a -= lr * m_hat

        # projection onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # anneal β
        if t % 350 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        # occasional jitter to avoid stagnation
        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

    return a

# Random initialisers (families on the simplex)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random non‑negative sequence on the ℓ¹‑simplex Σa = √(2 n).
    Several families (power‑law, exponential, mixed, Dirichlet,
    piecewise‑exponential, near‑uniform) are sampled uniformly.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # pure power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length.
    The result is projected back onto the appropriate simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

def _objective_from_vector(a: np.ndarray) -> float:
    """Exact objective = 2·n·max(b)/(sum(a)²)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    sum_a = np.sum(a)
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _objective_from_conv(b: np.ndarray, n: int, sum_a: float) -> float:
    """Objective given already‑computed convolution ``b``."""
    max_b = float(b.max())
    return float(2.0 * n * max_b / (sum_a * sum_a))

# Targeted mass‑transfer hill climbing (exact convolution updates)
# ----------------------------------------------------------------------


def _targeted_hill_climbing(a0: np.ndarray,
                            b0: np.ndarray,
                            deadline: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Hill‑climbing that always moves mass from an index that contributes
    heavily to the current maximum entry of the convolution to an index
    that contributes little.  The move is accepted if it improves the
    maximum or with a Metropolis probability (temperature ``T``).
    Returns the best (a, b) pair observed before the deadline.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = float(b.max())
    best_score = cur_score
    best_a = a.copy()
    best_b = b.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.02 + 1e-9
    cooling = 0.9995

    while time.time() < deadline:
        # index where the convolution attains its maximum
        k = int(np.argmax(b))

        # --- select source i (large contribution to b[k]) ---
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]          # contribution of each pair
        total = float(contrib.sum())
        if total <= 0.0:
            # fallback to uniform random source
            i = rng.integers(0, n)
        else:
            probs = contrib / total
            i = rng.choice(i_range, p=probs)

        # --- select destination j (small contribution to b[k]) ---
        partner = np.zeros(n)
        j_start = i_start
        j_end = i_end
        jj = np.arange(j_start, j_end + 1)
        partner[jj] = a[k - jj]                     # a[k‑j] where defined
        # prefer indices with minimal partner value
        min_val = float(partner.min())
        cand_js = np.flatnonzero(partner == min_val)
        # ensure we do not stay in place
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            j = (i + rng.integers(1, n)) % n
        else:
            j = rng.choice(cand_js)

        # --- amount of mass to move ---
        max_delta = a[i] * 0.5          # never move more than half the source
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # --- exact update of the full convolution (O(n)) ---
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = float(b_new.max())

        # --- Metropolis acceptance ---
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-(new_score - cur_score) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                best_b = b.copy()
                print(f"[{time.time():.1f}] hill‑climb new best {best_score:.6f}")

        T *= cooling

    return best_a, best_b

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock budget (default 1000 s) and returns
    the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Optional seed from a previously stored construction
    # ------------------------------------------------------------------
    earlier = globals().get("height_sequence_1")
    if isinstance(earlier, (list, tuple, np.ndarray)):
        a0 = np.asarray(earlier, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        best_seq = a0.copy()
        best_score = _objective_from_vector(a0)
        print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Length schedule – we gradually increase the dimension
    # ------------------------------------------------------------------
    lengths = [4000, 8000, 12000, 16000, 20000]

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Initialise a candidate for this length
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)

        a_init = _project_to_simplex(a_init, target_sum)

        # --------------------------------------------------------------
        # Phase A – several short Adam runs (exploration + refinement)
        # --------------------------------------------------------------
        phase_end = min(deadline,
                        time.time() + max(15.0, (deadline - time.time()) * 0.25))
        a_current = a_init.copy()
        while time.time() < phase_end:
            # jitter a little to diversify restarts
            jitter = rng.normal(scale=target_sum * 0.02, size=n)
            a_start = a_current + jitter
            a_start = np.maximum(a_start, 0.0)
            a_start = _project_to_simplex(a_start, target_sum)

            a_opt = _adam_optimize(a_start, phase_end)
            score_opt = _objective_from_vector(a_opt)
            if score_opt < best_score:
                best_score = score_opt
                best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] Adam improvement (n={n}) score={score_opt:.6f}")

            a_current = a_opt   # seed next short run from the best we just found

        # --------------------------------------------------------------
        # Phase B – targeted hill climbing (exact convolution updates)
        # --------------------------------------------------------------
        if time.time() < deadline:
            b_full = _full_convolution(a_current)
            hill_dead = min(deadline,
                            time.time() + max(30.0, (deadline - time.time()) * 0.35))
            a_hc, b_hc = _targeted_hill_climbing(a_current, b_full, hill_dead)
            score_hc = _objective_from_conv(b_hc, n, np.sum(a_hc))
            if score_hc < best_score:
                best_score = score_hc
                best_seq = a_hc.copy()
                print(f"[{time.time():.1f}] Hill‑climb improvement (n={n}) score={score_hc:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with any remaining budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        b_full = _full_convolution(best_seq)
        a_final, b_final = _targeted_hill_climbing(best_seq, b_full, deadline)
        final_score = _objective_from_conv(b_final, n, np.sum(a_final))
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final polishing improvement: score={final_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros for neatness
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback – uniform distribution (should never happen)
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log the official evaluation (objective = max of the self‑convolution)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
