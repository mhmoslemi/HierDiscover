from scipy.optimize import linprog

def solve_lp_max_sum(f, M):
    n = len(f)
    # Build constraint matrix A_ub where rows represent (f*g)_k <= M.
    # Using dense array, shape (2n-1, n)
    rows = np.arange(2*n-1)[:, None]  # shape (2n-1,1)
    cols = np.arange(n)  # shape (n,)
    mask = (rows - cols >= 0) & (rows - cols < n)
    A = np.where(mask, f[cols], 0.0)   # broadcasting: use f[cols] to fill each column
    # Now we have A shape (2n-1,n)
    b = np.full(2*n-1, M)
    # objective c: minimize -sum g => c = -1's.
    c = -np.ones(n)
    # bounds (nonneg) will be (0, None)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs', options={'presolve': True, 'time_limit': 30})
    if res.success:
        return res.x
    else:
        # fallback zero
        return np.zeros(n)

def _solve_lp_max_sum(f,M):
    n = len(f)
    # Build constraint matrix A as dense numpy array of shape (2n-1, n)
    rows = np.arange(2*n-1)[:, None]
    cols = np.arange(n)
    mask = (rows - cols >= 0) & (rows - cols < n)
    A = np.where(mask, f[cols], 0.0)

    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M]
        obj = cp.Maximize(cp.sum(g))
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000)
        if g.value is not None:
            return np.array(g.value).ravel()
    except Exception as e:
        pass
    # fallback to scipy
    try:
        from scipy.optimize import linprog
        c = -np.ones(n)
        res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    # fallback zero
    return np.zeros(n)

    Σaᵢ = √(2 n).

              gᵢ ≥ 0

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray | None = None) -> np.ndarray:
    """Full linear convolution a*b (or a*a if b is None) using FFT."""
    if b is None:
        b = a
    n, m = a.size, b.size
    size = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: n + m - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    return float(_full_convolution_fft(a).max())


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex Σxᵢ = s, xᵢ ≥ 0 (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s."""
    if s <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear programme: maximise Σ gᵢ   s.t.   (f * g)ₖ ≤ M , g ≥ 0
# ----------------------------------------------------------------------
def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """Solve the LP for the current sequence f and bound M."""
    n = f.size
    # Build constraint matrix A where (A @ g)_k = (f * g)_k
    rows = np.arange(2 * n - 1)[:, None]          # shape (2n‑1,1)
    cols = np.arange(n)                           # shape (n,)
    mask = (rows - cols >= 0) & (rows - cols < n)  # True where f_i contributes
    A = np.where(mask, f[cols], 0.0)               # (2n‑1, n) dense matrix

        constraints = [A @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000)

    except Exception:

    return np.zeros(n)
