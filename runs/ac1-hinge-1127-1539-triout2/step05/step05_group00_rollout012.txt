def _biased_hill_climbing(a0, b0, deadline):
    a = a0.copy()
    b = b0.copy()
    n = a.shape[0]
    cur_max = b.max()
    best_max = cur_max
    best_a = a.copy()
    rng = np.random.default_rng()
    T = cur_max * 0.1 + 1e-9
    cooling = 0.9995
    it = 0
    while time.time() < deadline:
        # compute source index
        k_max = int(np.argmax(b))
        # compute contributions
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idxs = np.arange(i_start, i_end + 1)
        partners = k_max - idxs
        contrib = a[idxs] * a[partners]
        total_contrib = contrib.sum()
        if total_contrib > 0 and rng.random() < 0.9:
            # weighted selection
            probs = contrib / total_contrib
            i = rng.choice(idxs, p=probs)
        else:
            # fallback uniform
            i = rng.integers(0, n)

        # pick destination j, avoid i
        j = rng.integers(0, n-1)
        while j == i:
            j = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # choose delta random fraction up to 0.5 of source
        max_delta = a[i] * 0.5
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        # update b using copy method
        b_new = b.copy()
        # slice update
        b_new[i:i+n] -= 2.0 * delta * a
        b_new[j:j+n] += 2.0 * delta * a

        # quadratic corrections
        k2i = 2*i
        k2j = 2*j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                # print improvement
                print(f"[{time.time():.1f}] bias hill climb improvement: {best_max:.6f}")

        T *= cooling
        it += 1
        # periodic full recompute to avoid drift errors
        if it % 2000 == 0:
            b = _full_convolution(a)
            cur_max = b.max()

    return best_a, b

"""
Propose a step‑function that (slightly) improves on the previous best
objective 1.513264.  The algorithm consists of three stages:

1. Initialise (seed from the historic ``height_sequence_1`` if available,
   otherwise random) and normalise to the ℓ¹‑simplex Σa = √(2 n).

2. Projected gradient descent on a smooth log‑sum‑exp approximation of the
   max‑convolution (the same routine as the reference implementation).

3. A *biased* hill‑climbing phase that preferentially moves mass from the
   indices contributing most to the current maximal convolution entry.
   Moves are evaluated with an O(n) exact update of the full self‑convolution
   and accepted with a simple simulated‑annealing rule.

Finally a short un‑biased hill‑climb polish is performed on the remaining
budget.  All helper functions are top‑level, no closures or lambdas are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a as a NumPy array."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of   (1/β)·log Σ exp(β·b)   where   b = a*a .
    Returns (g, b) with g = 2·(a ⋆ p) and
    pₖ ∝ exp(β·bₖ).  Correlation is performed via FFT.
    """
    n = a.shape[0]

    # b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]  # reversed a
    conv_len = n + (2 * n - 1) - 1  # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smooth max.
    Uses momentum, an annealed inverse temperature β and occasional jitter.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 8192.0
    beta_factor = 1.4

    step_base = target_sum * 0.15
    momentum = np.zeros_like(a)
    rho = 0.90

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β annealing
        if it % 350 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative sequence on the ℓ¹‑simplex Σa = sqrt(2 n).
    Several families (power‑law, exponential, Dirichlet, piecewise) are
    sampled uniformly.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # pure power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


# ----------------------------------------------------------------------
# Biased hill‑climbing (focus on the current maximal convolution entry)
# ----------------------------------------------------------------------


def _biased_hill_climbing(a0: np.ndarray,
                          b0: np.ndarray,
                          deadline: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Hill‑climbing that preferentially moves mass from indices that
    contribute most to the current maximal convolution entry.
    Uses a simple simulated‑annealing acceptance rule.
    Returns the best (a, b) observed before the deadline.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_max = b.max()
    best_max = cur_max
    best_a = a.copy()
    best_b = b.copy()

    rng = np.random.default_rng()
    # temperature proportional to the current max (allows early uphill moves)
    T = cur_max * 0.05 + 1e-9
    cooling = 0.9997

    it = 0
    while time.time() < deadline:
        # locate the current maximal shift
        k_max = int(np.argmax(b))

        # ----- choose source index -----
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idxs = np.arange(i_start, i_end + 1)
        partners = k_max - idxs
        contrib = a[idxs] * a[partners]
        total = contrib.sum()

        # with high probability sample proportionally to the contribution,
        # otherwise fall back to uniform sampling (helps escape traps)
        if total > 0.0 and rng.random() < 0.9:
            probs = contrib / total
            i = rng.choice(idxs, p=probs)
        else:
            i = rng.integers(0, n)

        # ----- choose destination -----
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # amount of mass to move (up to 40 % of source)
        max_delta = a[i] * 0.4
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ---- exact update of the convolution (O(n)) ----
        b_new = b.copy()
        # linear part
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        # quadratic correction
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        # ---- acceptance test ----
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                best_b = b.copy()
                print(f"[{time.time():.1f}] bias‑climb improvement: {best_max:.6f}")

        # annealing schedule
        T *= cooling
        it += 1

        # mitigate numerical drift – recompute full convolution every few thousand steps
        if it % 2500 == 0:
            b = _full_convolution(a)
            cur_max = b.max()

    return best_a, best_b


# ----------------------------------------------------------------------
# Unbiased hill‑climbing (identical to the reference implementation)
# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray,
                   b0: np.ndarray,
                   deadline: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Traditional mass‑transfer hill‑climbing (used for a final polish).
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()
    best_b = b.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.02 + 1e-9
    cooling = 0.9997

    while time.time() < deadline:
        if a.sum() == 0:
            break
        probs = a / a.sum()
        i = rng.choice(n, p=probs)
        if a[i] < 1e-14:
            continue

        j = rng.integers(0, n)
        if i == j:
            continue

        max_delta = a[i] * 0.5
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a

        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-(new_score - cur_score) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                best_b = b.copy()
                print(f"[{time.time():.1f}] hill‑climb improvement: {best_score:.6f}")

        T *= cooling

    return best_a, best_b


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock budget (default 1000 s) and returns
    the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # 1.  Initialise – try to reuse the historic construction
    # ------------------------------------------------------------------
    earlier = globals().get("height_sequence_1")
    if isinstance(earlier, (list, tuple, np.ndarray)):
        a0 = np.asarray(earlier, dtype=float)
        n = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n))
        best_seq = a0.copy()
        best_score = _max_convolution_fft(best_seq)
        print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n}) "
              f"score={best_score:.6f}")
    else:
        # fall back to a medium‑size random start
        n = 5000
        a0 = _random_initial_sequence(n, rng)
        best_seq = a0.copy()
        best_score = _max_convolution_fft(best_seq)
        print(f"[{time.time():.1f}] Random start (n={n}) score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Projected gradient descent (quick phase)
    # ------------------------------------------------------------------
    phase_end = min(deadline,
                    time.time() + min(80.0, (deadline - time.time()) * 0.3))
    a_gd = _gradient_descent_opt(best_seq, phase_end)
    score_gd = _max_convolution_fft(a_gd)
    if score_gd < best_score:
        best_score = score_gd
        best_seq = a_gd.copy()
        print(f"[{time.time():.1f}] GD improvement score={score_gd:.6f}")

    # ------------------------------------------------------------------
    # 3.  Biased hill‑climbing (the core improvement phase)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        b_full = _full_convolution(best_seq)
        # allocate roughly 60 % of the remaining budget to the biased search
        remaining = deadline - time.time()
        biased_deadline = time.time() + min(remaining * 0.6, 300.0)
        a_bias, b_bias = _biased_hill_climbing(best_seq, b_full, biased_deadline)
        bias_score = b_bias.max()
        if bias_score < best_score:
            best_score = bias_score
            best_seq = a_bias.copy()
            print(f"[{time.time():.1f}] Biased hill‑climb score={bias_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Final unguided hill‑climb polish (use whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        a_final = best_seq.copy()
        b_final = _full_convolution(a_final)
        final_dead = deadline
        a_final, b_final = _hill_climbing(a_final, b_final, final_dead)
        final_score = b_final.max()
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final polishing score={final_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros for neatness
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Report the official evaluation (for debugging)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
