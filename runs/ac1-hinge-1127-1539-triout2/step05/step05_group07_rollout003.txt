The goal is to find a non‑negative vector `a` that minimises  

\[
\frac{2\,n\;\max (a*a)}{(\sum a)^2},
\]

i.e. minimise the peak of the self‑convolution under a fixed ℓ₁‑norm.  
The implementation below combines three ideas that together improve the
previous best bound:

1. **Entropy‑projected Adam** – the vector is parametrised by logits,
   the soft‑max guarantees non‑negativity and exact ℓ₁‑normalisation.
   A τ‑soft‑max approximation of the ∞‑norm provides a smooth gradient.
2. **Active‑set sub‑gradient** – once the temperature has become small,
   the gradient is computed only from the (almost) active constraints,
   which is the exact sub‑gradient of the true max‑function.
3. **Linear‑program refinement** – after a short Adam run we solve a tiny
   linear programme  

   \[
   \max\; \mathbf{1}^\top g\quad
   \text{s.t.}\quad (a*g)_i\le M,\;g\ge0,
   \]

   where \(M=\max(a*a)\).  The LP increases the ℓ₁‑mass without raising
   the convolution peak; the resulting vector is mixed back into the
   current candidate.  The LP is built with a sparse Toeplitz matrix,
   allowing dimensions up to a few thousand comfortably.

The search runs many independent restarts with different lengths and
initial shapes, keeps the best candidate seen, and finally performs a
long refinement on that best vector.  All operations respect the 1000 s
time budget.

```python
"""
adversarial step‑function optimiser (v2)

* entropy‑projected Adam on a τ‑soft‑max of the convolution peak
* active‑set sub‑gradient once the temperature is low
* cheap linear‑program refinement for moderate lengths (≤ 2500)
* many independent restarts, final long refinement on the best candidate
* all sequences are non‑negative and automatically ℓ₁‑normalised to √(2 n)
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse


# ----------------------------------------------------------------------
# Basic FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT (full length 2·n‑1)."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


# ----------------------------------------------------------------------
# Linear‑program (LP) utilities
# ----------------------------------------------------------------------
def _conv_constraint_matrix(f: np.ndarray) -> scipy.sparse.csr_matrix:
    """
    Build a sparse matrix A (shape (2n‑1, n)) such that
    (f * g)_i = (A @ g)_i  for 0 ≤ i < 2n‑1.
    """
    n = f.shape[0]
    # each column j contains the vector f placed starting at row j
    data = np.tile(f, n)                     # length n*n
    rows = np.arange(n)[:, None] + np.arange(n)   # shape (n,n)
    rows = rows.ravel()                      # 0 … 2n‑2 repeated n times
    cols = np.repeat(np.arange(n), n)
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_refine(f: np.ndarray, M: float, S: float,
               rng: np.random.Generator, max_iters: int = 3):
    """
    Perform a few LP refinements:
        maximise 1ᵀg    subject to (f*g)_i ≤ M , g ≥ 0
    The optimal g is scaled to ℓ₁‑norm S and mixed back into f.
    Returns the refined vector and its new peak.
    """
    n = f.shape[0]
    A = _conv_constraint_matrix(f)
    b_ub = np.full(2 * n - 1, M, dtype=float)

    for _ in range(max_iters):
        # maximise sum(g)  → minimise  -sum(g)
        res = scipy.optimize.linprog(
            c=-np.ones(n, dtype=float),
            A_ub=A,
            b_ub=b_ub,
            bounds=[(0.0, None)] * n,
            method='highs',
            options={'presolve': True}
        )
        if not res.success:
            break
        g = res.x
        if g.sum() == 0.0:
            break
        # bring g back to the required ℓ₁‑norm
        g = g * (S / g.sum())
        # convex combination (t = 0.5 works well)
        t = 0.5
        f = (1.0 - t) * f + t * g
        # recompute peak to tighten the bound for the next LP
        M = _full_convolution_fft(f).max()
        b_ub = np.full(2 * n - 1, M, dtype=float)

    return f, M


# ----------------------------------------------------------------------
# Gradient of the τ‑soft‑max approximation of max(b), b = a*a
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Returns (grad, max_b) where grad = ∂/∂a τ‑soft‑max(b)
    and max_b = max_i (a*a)_i.
    For very small tau the exact sub‑gradient (active set) is used.
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = b.max()

    # decide whether to use the full soft‑max or the active set
    if tau > 1e-4:
        diff = b - max_b                     # ≤ 0
        exps = np.exp(diff / tau)
        w = exps / exps.sum()                # soft‑max weights, shape (2n‑1,)
    else:
        # exact sub‑gradient: all entries attaining the maximum receive equal weight
        mask = (b == max_b).astype(float)
        w = mask / mask.sum()                # normalise

    # gradient  = 2 * (w * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    f_len = _next_pow2(conv_len)
    Fw = np.fft.rfft(w, n=f_len)
    Fa = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fw * Fa, n=f_len)[:conv_len]
    grad = 2.0 * conv[n - 1:n - 1 + n]      # extract the n‑length central part
    return grad, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Minimise the τ‑soft‑max of the convolution peak.
    Returns the best vector encountered and the corresponding *hard* peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    # initialise logits from a0 (softmax(p) * S == a0)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                     # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.25

    # initialise temperature
    tau = max(_full_convolution_fft(a0).max(), 1.0)

    best_a = a0.copy()
    best_peak = _full_convolution_fft(a0).max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ------- a from logits (exact ℓ₁‑norm) -------
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)          # improve numeric stability
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ------- convolution and gradient -------
        g, peak = _softmax_gradient(a, tau)

        # keep the best *hard* peak seen so far
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ------- gradient w.r.t. logits (mirror descent) -------
        mu = np.dot(a, g) / S               # scalar
        grad_p = a * (g - mu)               # shape (n,)

        # optional clipping for very large updates
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 4.0 * S:
            grad_p = grad_p * (4.0 * S / clip_val)

        # ------- Adam update -------
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ------- schedules -------
        tau *= 0.985   # faster cooling than the previous version
        lr *= 0.9995   # gentle decay of step size

        # jitter every few hundred iterations to shake out of plateaus
        if it % 400 == 0:
            p += rng.normal(scale=0.015, size=n)

    return best_a, best_peak

# Simple linear interpolation for resizing a sequence

def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.ceil(pos).astype(int)
    hi = np.clip(hi, 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]

        # ---- cheap LP refinement for small dimensions -------------------------
        if n <= 2500:
            a_opt, peak = _lp_refine(a_opt, peak, S, rng, max_iters=4)

        sc, pk = _internal_score(a_opt)

        if sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"[restart] attempt={attempt:3d} n={n:5d}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final long refinement on the best candidate (if any)
    # ------------------------------------------------------------------
    if best_seq is not None:
        n_best = len(best_seq)
        S_best = np.sqrt(2.0 * n_best)
        deadline = time.time() + min(30.0, end_time - time.time())
        a_final, _ = _adam_optimize(best_seq, deadline, rng)
        if n_best <= 2500:
            a_final, _ = _lp_refine(a_final,
                                    _full_convolution_fft(a_final).max(),
                                    S_best, rng, max_iters=6)
        best_seq = a_final
        best_score, _ = _internal_score(best_seq)
        print(f"[final] refined best, internal score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Sanity‑check via the black‑box evaluator and store the result
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()
        ext = evaluate_sequence(final_seq)
        print(f"[external] external score = {ext:.6f}   internal = {best_score:.6f}"
              f"   length = {len(final_seq)}")
        globals()['height_sequence_1'] = final_seq

    return final_seq
```
