"""
Step‑function optimizer – enhanced multi‑phase search.

The goal is to minimise the maximum entry of the discrete self‑convolution
`a*a` under the ℓ¹–normalisation  Σa = √(2 n)  (the official score equals that
maximum).  The algorithm explores several lengths, draws diverse initial
points, applies a smooth‑max (log‑sum‑exp) projected gradient descent,
refines with a true sub‑gradient descent, then performs an *aggressive*
mass‑transfer step that directly attacks the currently maximal convolution
entry.  Finally a short simulated‑annealing polishing pass is run.

All primitives (simplex projection, FFT‑based convolution, etc.) are kept
as top‑level functions – no closures or lambdas – and the routine respects
the wall‑clock budget supplied via ``budget_s`` (default 1000 s).  The best
sequence found is returned as a plain Python list.
"""

import time
import numpy as np

# Utility functions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1, dtype=float)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using an FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Official objective – maximal entry of the self‑convolution."""
    return float(_max_convolution_fft(a).max())


# ----------------------------------------------------------------------
# Phase 1 – smooth‑max (log‑sum‑exp) projected gradient descent
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b)  where  b = a*a.
    Returns (g, b) where
        g = ∇_a ( 1/β·log Σ exp(β·b) ) = 2·(a ⋆ p)
    and p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # b = a * a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max weights
    max_b = float(np.max(b))
    exps = np.exp(beta * (b - max_b))          # shift for stability
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1            # = 3 n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _smooth_max_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Annealed projected gradient descent on the smooth‑max.
    Uses momentum, an adaptive step size and occasional jitter.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 4096.0
    beta_factor = 1.4

    step_base = target_sum * 0.20          # a slightly larger base step
    momentum = np.zeros_like(a)
    rho = 0.90

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        # simplex projection
        a = _project_to_simplex(a, target_sum)

        # jitter every few hundred iterations
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β‑annealing
        if it % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a


# ----------------------------------------------------------------------
# Phase 2 – exact sub‑gradient descent on the true max‑norm
# ----------------------------------------------------------------------


def _max_index_subgradient(a: np.ndarray):
    """Sub‑gradient of max(b) where b = a*a (active index k*)."""
    n = a.shape[0]
    conv = _max_convolution_fft(a)
    k = int(np.argmax(conv))

    g = np.zeros_like(a)
    if k < n:                         # left half of the convolution
        g[:k + 1] = 2.0 * a[k::-1]
    else:                             # right half
        d = k - (n - 1)               # shift amount (0 … n‑1)
        g[d:] = 2.0 * a[: n - d]
    return g, k


def _subgradient_descent_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected sub‑gradient descent on the ∞‑norm of the self‑convolution.
    Uses a classic η_t = α / √t schedule.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        g, _ = _max_index_subgradient(a)

        step = 0.9 * target_sum / (np.sqrt(it + 1) * np.linalg.norm(g) + 1e-12)
        a -= step * g

        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        if it % 150 == 0:
            jitter = rng.normal(scale=target_sum * 4e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1
    return a

# Phase 2‑b – aggressive mass‑transfer targeted at the current peak

def _aggressive_reduce(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Repeatedly move a modest fraction of the “heaviest” coefficient
    (the one that contributes most to the current maximal convolution entry)
    to a “light” location.  The transferred amount is chosen adaptively
    so that the peak is driven down as fast as possible while keeping the
    feasibility constraints satisfied.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sum(a)               # already √(2 n)
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _max_convolution_fft(a)
        k = int(np.argmax(conv))
        max_val = conv[k]

        # contributions to the offending entry
        idx = np.arange(n)
        other = k - idx
        mask = (other >= 0) & (other < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[other[mask]]

        i_high = int(np.argmax(contrib))

        # cheap choice for a donor – among the 15 % smallest masses
        low_candidates = np.where(a <= np.percentile(a, 15))[0]
        if low_candidates.size == 0:
            i_low = int(np.argmin(a))
        else:
            i_low = int(rng.choice(low_candidates))

        # compute the partner’s coefficients in the offending convolution entry
        a_other_high = a[k - i_high] if 0 <= k - i_high < n else 0.0
        a_other_low = a[k - i_low] if 0 <= k - i_low < n else 0.0

        # we wish to reduce max_val; the reduction obtained by moving δ from
        # i_high to i_low equals δ·(a_other_high - a_other_low).
        gain_per_unit = a_other_high - a_other_low
        if gain_per_unit <= 0:
            # nothing useful can be done – pick a different low index
            i_low = int(np.argmin(a))
            a_other_low = a[k - i_low] if 0 <= k - i_low < n else 0.0
            gain_per_unit = a_other_high - a_other_low

        # desired reduction (a safety factor keeps us away from the boundary)
        desired_reduction = max_val - 1.0       # we aim at ~1.0 but never go below it
        delta = min(a[i_high] * 0.25,
                    desired_reduction / (gain_per_unit + 1e-12) * 0.5)

        if delta < 1e-12:
            break

        a[i_high] -= delta
        a[i_low] += delta

        # re‑project onto the simplex – the sum is unchanged but numerical
        # drift may appear
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid cycling
        if rng.random() < 0.01:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

    return a

def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Very light SA that moves a random fraction of mass between two sites."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    cur_score = _score(a)
    best_score = cur_score

    T = cur_score * 0.02 + 1e-9
    cooling = 0.9996
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-16:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        frac = rng.random() * 0.18
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta

        new_score = _score(a)
        dE = new_score - cur_score
        accept = dE <= 0.0 or (rng.random() < np.exp(-dE / T))

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling

    return best

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative point on the simplex Σa = √(2 n).
    Several families (power‑law, exponential, Dirichlet, piecewise, V‑shape)
    are sampled to increase diversity.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 7)
    idx = np.arange(n, dtype=float)

    if family == 0:                                 # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                               # pure exponential
        lam = rng.exponential(scale=0.02)
        a = np.exp(-lam * idx)

    elif family == 2:                               # power × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.02)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                               # Dirichlet (uniform simplex)
        a = rng.dirichlet(alpha=np.ones(n))

    elif family == 4:                               # piecewise exponential
        b = rng.integers(int(0.25 * n), int(0.75 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                             np.exp(-lam2 * np.arange(b, n))])

    elif family == 5:                               # V‑shape (larger at the ends)
        dist = np.abs(idx - (n - 1) / 2.0) + 1.0
        a = dist

    else:                                           # almost uniform w/ jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a

                print(f"[{time.time():.1f}]  Sub‑gradient improvement n={n}  score={sc:.6f}")

            # 3) aggressive mass‑balance reduction
            sub_dead = min(deadline, time.time() + min(40.0, remaining * 0.15))
            a_ref3 = _aggressive_reduce(best_seq, sub_dead)
            sc = _score(a_ref3)
            if sc < best_score:
                best_score = sc
                best_seq = a_ref3.copy()
                print(f"[{time.time():.1f}]  Aggressive reduction n={n}  score={sc:.6f}")

    # ------------------------------------------------------------------
    # Final simulated‑annealing polishing with whatever time is left
    # ------------------------------------------------------------------
    if best_seq is not None:
        left = deadline - time.time()
        if left > 4.0:
            a_final = _simulated_annealing(best_seq, deadline)
            sc = _score(a_final)
            if sc < best_score:
                best_score = sc
                best_seq = a_final.copy()
                print(f"[{time.time():.1f}]  Final SA improvement: score={sc:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up tiny trailing zeros and report the official evaluation
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution of modest length
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
