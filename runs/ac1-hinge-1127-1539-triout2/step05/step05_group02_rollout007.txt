Multi‑scale projected gradient descent + Simulated Annealing.

The optimizer works on several problem sizes (by default 800, 1500, 3000,
5000).  For each size a candidate sequence is initialised (global warm‑start,
power‑law or random Dirichlet), projected onto the ℓ¹‑simplex
Σa = √(2·n) and then refined in two phases:

1.  Gradient descent on a soft‑max approximation of max (a*a).  The temperature
    β is slowly increased up to 4096, while the step size is adapted to the
    gradient norm.  Jitter is added periodically to escape flat regions.

2.  A long simulated‑annealing phase that moves mass between indices that
    contribute most to the current worst convolution entry, performs short‑window
    smoothing and occasional swaps.  The temperature decays exponentially.

All intermediate iterates are projected onto the simplex, guaranteeing that
the evaluation function `evaluate_sequence` (which is scale‑invariant) sees a
properly normalised candidate.  The best sequence found over all runs is
returned (trailing zeros are stripped).

The method stays within the 1000 s budget, uses only NumPy and the provided
`evaluate_sequence` function, and prints progress information.
"""

    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Fast evaluation of the objective (max convolution) for a normalised sum."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Implements Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (gradient, raw convolution array) where gradient has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    # full convolution a*a
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # soft‑max weights
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient w.r.t. a (see analysis in the description)
    a_rev = a[::-1]
    Pr_fft = np.fft.rfft(p, n=size)
    Ar_fft = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    # the gradient lives on the original index range
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _annealed_gradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected gradient descent on the soft‑max approximation of the max.
    The temperature β is increased up to 4096.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 4096.0
    # base step size – a fraction of the simplex radius
    step_base = target_sum * 0.12
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            # learning rate shrinks with β to keep updates stable
            lr = step_base / (norm_g * (beta + 1.0))
            a = a - lr * g

        # feasibility projection
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to help escape plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] GD improvement β={beta:.1f} score={best_val:.6f}")

        # increase temperature slowly
        if it % 600 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return best_a, best_val


def _simulated_annealing(
    a0: np.ndarray,
    deadline: float,
    start_temp: float = 0.5,
    end_temp: float = 1e-5,
    max_frac_start: float = 0.40,
    max_frac_end: float = 0.005,
    max_window: int = 30,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing refinement.
    Moves:
      * donor→receiver mass transfer (≈70 %),
      * local smoothing of a short window,
      * random swap (≈10 %).
    The temperature decays exponentially, as does the allowed move size.
    """
    a = a0.copy()
    n = a.size
    cur_score = _score(a)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)
    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()
    conv = _conv_fft(a)  # current convolution

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # current worst convolution entry
        max_b = conv.max()
        k_max = int(np.argmax(conv))

        # sensitivity (contribution) of each coordinate to the worst entry
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]                     # a_j where i+j=k_max
        weight = a * sens                               # actual contribution a_i * a_{k_max-i}

        # ------------------------------------------------------------------
        # Propose a move
        # ------------------------------------------------------------------
        move_type = None
        if rng.random() < 0.70:  # mass transfer
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight / total_w)
            else:
                donor = rng.integers(n)

            # pick a receiver that contributes little to the worst entry
            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            receiver = rng.choice(low_idx) if low_idx.size else rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        elif rng.random() < 0.90:  # local smoothing
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                saved_seg = a[start:end].copy()
                avg = saved_seg.mean()
                a[start:end] = avg
                move_type = "smooth"
                saved_start, saved_end = start, end

        else:  # random swap
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # ------------------------------------------------------------------
        # Evaluate the new state (single FFT)
        # ------------------------------------------------------------------
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = _score(a)

        # ------------------------------------------------------------------
        # Metropolis acceptance
        # ------------------------------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA new best = {best_score:.6f}")
        else:
            # revert the move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter to shake out deep minima
        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = a * (a.sum() / a.sum())  # re‑normalise (no‑op, keeps sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


def _run_one_instance(n: int, init_mode: str, deadline: float) -> tuple[np.ndarray, float]:
    """
    Optimise a single problem size.
    init_mode ∈ {'global','power','random'}.
    Returns the best sequence and its score.
    """
    target = np.sqrt(2.0 * n)

    # ---- initialisation -------------------------------------------------
    if init_mode == "global" and "height_sequence_1" in globals():
        seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if seq.size > n:
            seq = seq[:n]
        else:
            seq = np.pad(seq, (0, n - seq.size), "constant")
    elif init_mode == "power":
        seq = _power_law_sequence(n, 0.55, target)
    elif init_mode == "random":
        rng = np.random.default_rng()
        seq = rng.random(n)
        seq = seq / seq.sum() * target
    else:
        seq = _power_law_sequence(n, 0.55, target)

    seq = _project_to_simplex(seq, target)

    # ---- gradient descent (if we have enough time) ----------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        gd_deadline = time.time() + min(30.0, remaining * 0.30)
        seq, gd_score = _annealed_gradient_descent(seq, gd_deadline)
    else:
        gd_score = _score(seq)

    # ---- simulated annealing --------------------------------------------
    cur_time = time.time()
    if cur_time < deadline:
        seq, sa_score = _simulated_annealing(seq, deadline)
        final_score = min(gd_score, sa_score)
    else:
        final_score = gd_score

    return seq, final_score


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Main entry point used by the evaluation harness.
    Runs several independent optimisation instances of different lengths,
    each consisting of a short projected‑gradient phase followed by a
    simulated‑annealing phase.  The best sequence found before the time
    budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = float("inf")

    # Problem sizes we will try – larger sizes give more degrees of freedom
    sizes = [800, 1500, 3000, 5000]

    # Cycle through the sizes, allocating a slice of the remaining time to each.
    for n in sizes:
        now = time.time()
        if now >= deadline - 5.0:
            break
        remaining = deadline - now
        # Keep at least 30 s for the final refinement.
        time_for_this = min(remaining * 0.45, 250.0)

        # Choose an initialisation mode (rotate for diversity)
        init_modes = ["global", "power", "random"]
        init_mode = init_modes[rng.integers(len(init_modes))]

        run_deadline = time.time() + time_for_this
        seq, score = _run_one_instance(n, init_mode, run_deadline)

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] size={n} init={init_mode} → new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final short gradient refinement on the best candidate (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        extra_deadline = time.time() + (deadline - time.time()) * 0.25
        best_seq, _ = _annealed_gradient_descent(best_seq, extra_deadline)

    # Strip trailing zeros (they do not affect the score)
    if best_seq is not None:
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[: nz[-1] + 1]
        else:
            best_seq = np.array([], dtype=float)

    print(f"[{time.time():.1f}] final best score≈{best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
