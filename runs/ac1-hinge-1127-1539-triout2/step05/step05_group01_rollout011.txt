def _adam_optimize(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0*n)

    # hyperparameters
    alpha = 0.05  # base learning rate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    beta_sched = 1.0
    max_beta = 1e6

    while time.time() < deadline:
        t += 1
        # compute gradient of smoothed max with current beta
        g, _ = _gradient_softmax(a, beta_sched)
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # project
        a = _project_to_simplex(a, target_sum)

        # update beta schedule occasionally
        if t % 50 == 0 and beta_sched < max_beta:
            beta_sched = min(beta_sched * 1.2, max_beta)

        # optional jitter to escape stagnation every few hundred iterations
        if t % 200 == 0:
            jitter = np.random.default_rng().normal(scale=target_sum * 1e-4, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

    return a

def _local_search(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_fft(best)
    T = best_score * 0.05 + 1e-8
    cooling = 0.9997
    rng = np.random.default_rng()
    # We'll do moves of removing fraction of mass from i to j; but accept only if reduces max.
    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        max_transfer = a[i] * 0.2  # up to 20% of mass
        delta = rng.random() * max_transfer
        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        if cur_score < best_score or (T > 0 and rng.random() < np.exp((best_score - cur_score)/T)):
            # accept
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # revert
            a[j] -= delta
            a[i] += delta

        T *= cooling

def _adam_descent(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    # hyperparameters
    base_lr = 0.2   # step size
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    beta = 1.0
    max_beta = 1e5

    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)

        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        a = _project_to_simplex(a, target_sum)

        # update beta schedule
        if t % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.2, max_beta)

        # occasional random jitter
        if t % 500 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

    return a

def _local_search(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_fft(best)
    T = best_score * 0.02 + 1e-8
    cooling = 0.9996
    rng = np.random.default_rng()
    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = a[i] * rng.random() * 0.3  # up to 30% move
        a[i] -= delta
        a[j] += delta
        cur_score = _max_convolution_fft(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
        else:
            # acceptance with annealing
            if T > 0 and rng.random() < np.exp((best_score - cur_score) / T):
                # accept worse move (but not update best)
                pass
            else:
                # revert
                a[j] -= delta
                a[i] += delta
        T *= cooling
        # occasional jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
    return best

"""
Propose a step‑function (non‑negative sequence) that minimises
``evaluate_sequence``.

The search proceeds in three phases:

1. **Deterministic parametric sweep** – generate power‑law and symmetric
   power‑law vectors for several lengths, exponents and shifts.  Keep the
   best few (elite) candidates.

2. **Projected Adam optimisation** – each elite candidate is refined by
   minimising a smooth log‑sum‑exp approximation of the maximal convolution
   entry.  Adam’s adaptive step size together with a slowly increasing
   temperature (β) yields fast convergence.  After every update the vector
   is projected onto the ℓ¹‑simplex Σa = √(2 n).

3. **Hill‑climbing local search** – a short random‑mass‑transfer phase
   (simulated annealing style) tweaks the refined vectors with decreasing
   temperature to escape possible local minima.

All phases respect a user supplied time budget (default 1000 s).  The routine
never performs I/O besides ``print`` statements and returns the best
sequence found as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex Σx = s, x≥0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection onto {x_i ≥ 0, Σx_i = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) surrogate of max(b) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b) where
        b = a*a (full convolution)
        g = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a ⋆ p) ,  p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]

    # ----- convolution a*a -----
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # ----- soft‑max probabilities -----
    max_b = np.max(b)
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)

    # ----- gradient g = 2·(a ⋆ p) -----
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Helper for sorting elite list (avoid lambda)
# ----------------------------------------------------------------------


def _score_key(item):
    """Return the score part of an (score, vector) pair."""
    return item[0]


# ----------------------------------------------------------------------
# Projected Adam optimisation on the smoothed maximum
# ----------------------------------------------------------------------


def _adam_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Adam optimisation of the log‑sum‑exp smooth max, with simplex projection."""
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    base_lr = 0.2
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    beta = 1.0
    max_beta = 1e5
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_to_simplex(a, target_sum)

        # increase smoothing parameter β periodically
        if t % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.2, max_beta)

        # occasional jitter to destabilise plateaus
        if t % 500 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

    return a


# ----------------------------------------------------------------------
# Simple hill‑climbing / simulated‑annealing refinement
# ----------------------------------------------------------------------


def _local_search(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Random mass‑transfer hill‑climbing with a decreasing temperature."""
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_fft(best)

    T = best_score * 0.02 + 1e-8
    cooling = 0.9996
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = a[i] * rng.random() * 0.30  # move up to 30 % of mass
        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
        else:
            # annealed acceptance of worse moves
            if T > 0.0 and rng.random() < np.exp((best_score - cur_score) / T):
                # keep the worse state (but best unchanged)
                pass
            else:
                # revert the move
                a[j] -= delta
                a[i] += delta

        T *= cooling

        # occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()

    return best


# ----------------------------------------------------------------------
# Parametric families (power‑law / symmetric power‑law)
# ----------------------------------------------------------------------


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    """
    a_i = (i + shift)^{-p},   i = 0,…,n‑1 .
    Returned vector is projected onto the ℓ¹‑simplex Σa = √(2 n).
    """
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    target_sum = np.sqrt(2.0 * n)
    scale = target_sum / raw.sum()
    a = raw * scale
    a = np.maximum(a, 0.0)
    return a


def _symmetric_powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    """
    Symmetric version: a_i = (i+shift)^{-p} + (n-1-i+shift)^{-p}.
    Afterwards projected onto the ℓ¹‑simplex.
    """
    idx = np.arange(n, dtype=float)
    left = (idx + shift) ** (-p)
    right = (n - 1 - idx + shift) ** (-p)
    raw = left + right
    target_sum = np.sqrt(2.0 * n)
    scale = target_sum / raw.sum()
    a = raw * scale
    a = np.maximum(a, 0.0)
    return a
