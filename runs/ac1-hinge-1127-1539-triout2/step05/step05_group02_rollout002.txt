def subgradient_step(a, beta):
    conv = _conv_fft(a) # compute conv
    max_val = conv.max()
    # find indices within relative tolerance
    eps = 1e-12
    ties = np.where(conv >= max_val - eps)[0]
    n = a.size
    g = np.zeros_like(a)
    for k in ties:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi+1)
        g[idx] += 2 * a[k - idx]
    g /= max(len(ties), 1)
    # step
    step = beta / (np.linalg.norm(g) + 1e-12)
    a = a - step * g

"""
Step‑function optimiser for the ``evaluate_sequence`` benchmark.

The optimiser works in three stages:

1. **Parametric sweep** – For a set of candidate lengths we generate simple
   families of non‑negative sequences (power‑law and exponential).  For each
   family we evaluate the objective and also try a few random permutations,
   because the convolution depends on the ordering of the coefficients.
   The best candidate for the length is retained.

2. **Projected sub‑gradient refinement** – Starting from the best candidate,
   we run a projected sub‑gradient descent on the true ``max`` of the
   auto‑convolution (the evaluation metric once the ℓ¹‑norm is fixed to
   ``sqrt(2·n)``).  The gradient is obtained analytically from the active
   convolution indices; we use a diminishing step‑size together with a simple
   momentum term and re‑project onto the simplex after each step.

3. **Optional simulated‑annealing** – If time remains, we perform a brief
   annealing phase consisting of mass‑transfer moves, local smoothing and
   occasional swaps.  This helps to escape shallow local minima.

The routine stops when the global time budget (default 1000 s) is exhausted
and returns the best sequence discovered.  All numbers are non‑negative,
projected onto the ℓ¹‑simplex ``Σ a_i = sqrt(2·n)``; therefore the
objective value equals the maximum entry of the convolution.
"""

import time
import numpy as np

# Helper utilities

def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s <= 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact objective: 2·n·max(conv)/(sum(a)²) (identical to evaluate_sequence)."""
    s = a.sum()
    if s < 0.01:
        return float("inf")
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    return float(2.0 * n * max_b / (s * s))


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Generate (i+1)^(-p) sequence of length n and scale to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """Generate exp(-lam·i) sequence of length n and scale to target_sum."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a

def _param_search(n: int, target_sum: float, deadline: float) -> tuple[np.ndarray, float]:
    """Search a few hand‑crafted families for a good initial sequence."""
    rng = np.random.default_rng()
    best_seq = None
    best_val = float("inf")

    # power‑law family
    for p in np.linspace(0.25, 2.0, 35):
        if time.time() >= deadline:
            break
        a = _power_law_sequence(n, p, target_sum)
        val = _score(a)
        if val < best_val:
            best_val, best_seq = val, a.copy()
        # try a few random permutations of the same multiset
        for _ in range(2):
            perm = rng.permutation(n)
            a_perm = a[perm]
            val = _score(a_perm)
            if val < best_val:
                best_val, best_seq = val, a_perm.copy()

    # exponential family
    for lam in np.linspace(0.0005, 0.03, 20):
        if time.time() >= deadline:
            break
        a = _exponential_sequence(n, lam, target_sum)
        val = _score(a)
        if val < best_val:
            best_val, best_seq = val, a.copy()
        for _ in range(1):
            perm = rng.permutation(n)
            a_perm = a[perm]
            val = _score(a_perm)
            if val < best_val:
                best_val, best_seq = val, a_perm.copy()

    # a very simple linear decreasing shape
    for slope in np.linspace(0.0, target_sum / n * 2.5, 10):
        if time.time() >= deadline:
            break
        a = np.maximum(target_sum / n * 2.0 - slope * np.arange(n), 0.0)
        if a.sum() == 0:
            continue
        a *= target_sum / a.sum()
        val = _score(a)
        if val < best_val:
            best_val, best_seq = val, a.copy()

    if best_seq is None:
        # fallback: uniform distribution (guaranteed to exist)
        a = np.full(n, target_sum / n, dtype=np.float64)
        best_seq, best_val = a, _score(a)

    return best_seq, best_val

# Projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _subgradient_step(a: np.ndarray) -> np.ndarray:
    """Return a new sequence after one sub‑gradient step (no projection)."""
    n = a.shape[0]
    conv = _conv_fft(a)
    max_val = conv.max()
    # indices where the maximum is attained (within a tiny tolerance)
    tol = 1e-12
    max_idxs = np.where(conv >= max_val - tol)[0]

    # accumulate gradient contributions
    g = np.zeros_like(a)
    for k in max_idxs:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        # contribution 2*a[k-idx] for each i = idx
        g[idx] += 2.0 * a[k - idx]
    if max_idxs.size > 1:
        g /= float(max_idxs.size)

    # simple normalised step (the step length will be set outside)
    return g


def _refine_subgradient(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with diminishing step size and momentum."""
    a = a0.copy()
    n = a.shape[0]
    target = np.sqrt(2.0 * n)

    # make sure we start on the simplex
    a = _project_to_simplex(a, target)

    best = a.copy()
    best_val = _score(a)

    # momentum term
    momentum = np.zeros_like(a)
    beta1 = 0.85

    it = 0
    while time.time() < deadline:
        # sub‑gradient of the current max‑convolution
        g = _subgradient_step(a)

        # update momentum
        momentum = beta1 * momentum + (1.0 - beta1) * g

        # step size schedule: η₀ / sqrt(t+1)
        eta0 = target * 0.12          # heuristic initial magnitude
        step = eta0 / np.sqrt(it + 1)

        a_new = a - step * momentum
        a_new = _project_to_simplex(a_new, target)

        cur_val = _score(a_new)
        if cur_val < best_val:
            best_val, best = cur_val, a_new.copy()
            print(f"[{time.time():.1f}] sub‑grad new best = {best_val:.6f}")

        # accept if improvement, otherwise keep the old point (no simulated annealing here)
        if cur_val <= _score(a) or np.random.rand() < 0.01:
            a = a_new

        # occasional jitter to avoid stagnation
        if it % 400 == 0 and it > 0:
            jitter = np.random.default_rng().normal(scale=target * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

        it += 1

    return best, best_val

def _anneal_refinement(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Very brief annealing phase: mass transfer / smoothing / swaps."""
    a = a0.copy()
    n = a.shape[0]
    target = np.sqrt(2.0 * n)

    best = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total = max(deadline - start, 1e-6)

    # pre‑compute convolution size for reuse
    size_fft = 1 << ((2 * n - 1).bit_length())

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / total
        # exponential cooling of temperature and move size
        temp = 0.6 * (0.001 / 0.6) ** tau   # from 0.6 → 0.001
        max_frac = 0.30 * (0.002 / 0.30) ** tau  # from 30% → 0.2%

        # evaluate current convolution
        conv = _conv_fft(a)
        cur_max = conv.max()
        k_max = int(conv.argmax())

        # sensitivity vector (how much each a_i contributes to the current max)
        sens = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]

        # -------------------------------------------------------------
        # propose a move
        # -------------------------------------------------------------
        move_type = None
        if rng.random() < 0.68:            # mass transfer (dominant)
            weights = a * sens
            total_w = weights.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weights / total_w)
            else:
                donor = rng.integers(n)

            # pick a receiver among indices with relatively low weight
            low_thr = np.percentile(weights, 20)
            low_idx = np.where(weights <= low_thr)[0]
            if low_idx.size:
                receiver = rng.choice(low_idx)
            else:
                receiver = rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        elif rng.random() < 0.90:          # short smoothing
            start_win = rng.integers(0, n)
            length = rng.integers(2, min(40, n - start_win) + 1)
            end_win = start_win + length
            seg_old = a[start_win:end_win].copy()
            a[start_win:end_win] = seg_old.mean()
            move_type = "smooth"
            saved = (start_win, seg_old)

        else:                               # random swap
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # project back onto simplex (mass is preserved up to numeric error)
        a = _project_to_simplex(a, target)

        # evaluate new state
        new_val = _score(a)

        # Metropolis acceptance
        accept = False
        if new_val < best_val:
            accept = True
        else:
            prob = np.exp(-(new_val - best_val) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            if new_val < best_val:
                best_val, best = new_val, a.copy()
                print(f"[{time.time():.1f}] SA new best = {best_val:.6f}")
        else:
            # revert move
            if move_type == "transfer":
                a[receiver] -= delta
                a[donor] += delta
            elif move_type == "smooth":
                start_win, seg_old = saved
                a[start_win:start_win + seg_old.size] = seg_old
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        it += 1

    return best, best_val

    score before the allocated time expires.
    """

    # degrees of freedom but each iteration is more expensive.

        # 1) fast parametric sweep
        # ---------------------------------------------------------
        sweep_deadline = time.time() + alloc * 0.40
        cand_seq, cand_score = _param_search(n, target, sweep_deadline)

        # ---------------------------------------------------------
        # 2) sub‑gradient refinement
        # ---------------------------------------------------------
        refine_deadline = time.time() + alloc * 0.45
        cand_seq, cand_score = _refine_subgradient(cand_seq, refine_deadline)

        # ---------------------------------------------------------
        # 3) short annealing phase (if time permits)
        # ---------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = deadline
            cand_seq, cand_score = _anneal_refinement(cand_seq, anneal_deadline)

        # keep global best
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time():.1f}] Global best updated n={n} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final post‑processing – strip trailing zeros for neatness
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.array([], dtype=float)
    else:
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[: nz[-1] + 1]

    print(f"[{time.time():.1f}] Finished. Best score = {best_score:.6f}, length = {len(best_seq)}")
    return best_seq.tolist()
