"""
Proposes a non‑negative step‑function (list of heights) that minimises
```
evaluate_sequence
```
by directly minimising the peak of the self‑convolution.
The algorithm works on the ℓ¹‑simplex (Σa = √(2·n)) and combines:

* a smoothed‑max (soft‑max) Adam phase – fast descent while the objective
  is a smooth approximation of the ∞‑norm.  The temperature α is slowly
  increased up to 200 so that the smooth loss becomes the true max.
* a pure sub‑gradient phase that follows the worst shift (the exact
  max‑norm).  Step‑size η_t = η₀ / √t with η₀ = 0.2.
* occasional “mass‑swap’’ moves that transfer a small amount of mass
  between two coordinates, helping the search escape plateaus.
* a final short random‑swap refinement.

All updates are projected onto the simplex using the Euclidean projection.
The whole search runs for the allotted budget (≈1000 s) and keeps the
best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending order
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()   # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv

    Gradient of   f(a) = Σ_i w_i·(a*a)_i
    wrt a.  Returns a vector of length n.
    """
    n = a.shape[0]
    # convolution of a with reversed weights
    w_rev = w[::-1]
    conv_len = n + w_rev.shape[0] - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    # the gradient corresponds to the slice that aligns with a
    # (see derivation in the analysis)
    start = w_rev.shape[0] - 1
    grad = 2.0 * full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Soft‑max (smooth max) utilities
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Weights w_i = exp(alpha·conv_i) / Σ_j exp(alpha·conv_j) (stable)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Simple sequence generators (all respect the ℓ¹‑norm √(2·n))
# ----------------------------------------------------------------------
def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a shorter base sequence to length n."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


def _initial_sequence(base_seq: np.ndarray | None,
                     n: int,
                     rng: np.random.Generator) -> np.ndarray:
    """Return a non‑negative vector of length n with ℓ¹‑norm √(2·n)."""
    S = np.sqrt(2.0 * n)

    # 0 – up‑sampled previous best (if available)
    # 1 – pure Dirichlet (uniform random)
    # 2 – geometric decay (symmetrised)
    # 3 – edge‑heavy reciprocal shape
    # 4 – random convex combination of the three bases
    typ = rng.integers(0, 5)

    if typ == 0 and base_seq is not None and base_seq.size > 10:
        a = _upsample_seq(base_seq, n)
        a += rng.normal(scale=a.mean() * 1e-3, size=n)   # tiny jitter
        a = np.clip(a, 0.0, None)
    elif typ == 1:
        a = rng.random(n)               # Dirichlet(1)
    elif typ == 2:
        rate = rng.uniform(0.9, 0.9995)
        left = rate ** np.arange(n)
        right = left[::-1]
        a = (left + right) / 2.0
    elif typ == 3:
        idx = np.arange(n, dtype=float)
        a = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.8)
        a = a ** power
    else:
        # random convex combo
        cand = []

        cand.append(rng.random(n))                     # uniform
        rate = rng.uniform(0.9, 0.9995)
        left = rate ** np.arange(n)
        right = left[::-1]
        cand.append((left + right) / 2.0)               # geometric
        idx = np.arange(n, dtype=float)
        rec = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.8)
        cand.append(rec ** power)                       # reciprocal

        w = rng.random(len(cand))
        w /= w.sum()
        a = sum(wi * ci for wi, ci in zip(w, cand))

    # Normalise to the required ℓ¹‑norm
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0)
        a_sum = a.sum()
    a = a / a_sum * S
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed max‑norm
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected Adam minimising the smoothed max‑norm."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.5
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # soft‑max temperature schedule
    alpha = 0.5
    alpha_mul = 1.02           # slowly tighten
    alpha_max = 200.0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        cur_max = np.max(conv)
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # smooth‑max weights and gradient
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection back onto the simplex
        a = _project_simplex(a, S)

        # occasional random mass swap (helps escape plateaus)
        if rng.random() < 0.02:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.04 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < best_score or rng.random() < 0.1:
                    a = cand

        # temperature & learning‑rate schedules
        alpha = min(alpha * alpha_mul, alpha_max)
        lr *= 0.9999

        # simple stagnation test
        if it % 3000 == 0 and best_score == cur_max:
            # no improvement in the last 3000 steps → break early
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑norm (worst shift only)
# ----------------------------------------------------------------------
def _subgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.2
    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient (2 * reversed a on the active anti‑diagonal)
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]            # factor 2 omitted (absorbed in step‑size)

        eta = eta0 / np.sqrt(it)
        a -= eta * g
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # occasional random perturbation
        if rng.random() < 0.04:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.2:
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()

        if it % 5000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimiser (smooth Adam → sub‑gradient)
# ----------------------------------------------------------------------
def _combined_optimize(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run a smoothed‑max Adam phase for roughly 60 % of the remaining time,
    then a sub‑gradient refinement for the rest.
    """
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    # 60 % for Adam, 40 % for sub‑gradient
    adam_deadline = now + 0.6 * remaining
    a_mid, score_mid = _adam_optimize(a0, adam_deadline, rng)

    if time.time() < deadline:
        a_fin, score_fin = _subgrad_optimize(a_mid, deadline, rng)
        if score_fin < score_mid:
            return a_fin, score_fin
    return a_mid, score_mid


# ----------------------------------------------------------------------
# Final short random‑swap refinement (purely local)
# ----------------------------------------------------------------------
def _local_refine(a: np.ndarray,
                  time_budget: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Perform cheap random mass‑swap moves while time permits."""
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))
    deadline = time.time() + time_budget
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if a[i] <= 0:
            continue
        delta = 0.02 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        cand_max = np.max(_full_convolution(cand))
        if cand_max < best_score:
            best_score = cand_max
            best_a = cand.copy()
            a = cand
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function minimising ``evaluate_sequence``.
    Returns the best sequence (list of floats) found before the time limit.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # Keep a small safety margin for the external call
    deadline = start + budget_s - 5.0

    # ------------------------------------------------------------------
    # Grab previously discovered best sequence if available
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    attempts = 0
    last_improve = start

    # ------------------------------------------------------------------
    # Main optimisation loop – we repeatedly start from a fresh random
    # initialise and run the combined optimiser.  The length n is drawn
    # from a fairly large interval (15k–25k) – larger n gives more
    # freedom to flatten the convolution.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        attempts += 1
        # pick a size; we avoid very small n because the theoretical
        # lower bound 1 + 1/(2n-1) gets tighter for large n
        n = int(rng.integers(15000, 25001))

        a0 = _initial_sequence(base_seq, n, rng)

        # give each attempt roughly 30 s (or whatever is left)
        attempt_deadline = min(time.time() + 30.0, deadline)

        a_opt, score = _combined_optimize(a0, attempt_deadline, rng)

        # small local refinement (5 s max) to shave off a little more
        if time.time() < deadline:
            budget_refine = min(5.0, deadline - time.time())
            a_opt, score = _local_refine(a_opt, budget_refine, rng)

        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            last_improve = time.time()
            print(f"[global] attempt={attempts} n={n} "
                  f"new best score={best_score:.6f}")

        # early exit if we have already crossed a comfortable safety
        if best_score <= 1.48:
            break

        # if we have been idle for a while, discard the old shape to
        # encourage diversity
        if time.time() - last_improve > 60.0:
            base_seq = None

    # ------------------------------------------------------------------
    # Fallback – should never happen but guards against an empty result
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(1, np.sqrt(2.0))

    final_seq = best_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} "
          f"(internal best = {best_score:.6f}) "
          f"length = {len(final_seq)}")
    return final_seq
