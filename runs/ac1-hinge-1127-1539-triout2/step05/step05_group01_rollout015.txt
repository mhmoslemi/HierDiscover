"""
Propose a non‑negative step function that minimizes ``evaluate_sequence``.
The algorithm works in four stages:

1. **Fast parametric sweep** – a few power‑law (and symmetric) candidates
   of different lengths (4000 – 8000) are created and the best few are kept.
2. **Greedy mass‑transfer descent** – for the current best vector we repeatedly
   move a fraction of mass from the index that contributes most to the
   maximal convolution entry to an index that contributes (almost) nothing.
   The move is kept only if the maximal entry does not increase.
   This quickly flattens the convolution profile.
3. **Projected gradient descent on a smooth max‑approximation** – the
   log‑sum‑exp surrogate is minimized while keeping the vector on the ℓ¹‑simplex.
4. **Simulated annealing refinement** – a classic SA routine with
   mass‑transfer moves finishes the search using the remaining time budget.

All stages respect a global time budget (default 1000 s) by checking the
deadline after each iteration.  The returned sequence is already normalised
so that Σa = sqrt(2·n); consequently ``evaluate_sequence`` simply returns the
maximal convolution entry.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex Σx = s, x ≥ 0 (Duchi et al. 2008)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection onto {x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed surrogate of max(b) and its gradient (log‑sum‑exp)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b) where
        b = a*a (full convolution)
        g = ∇ₐ ((1/β)·log Σₖ exp(β·bₖ)) = 2·(a ⋆ p),   pₖ ∝ exp(β·bₖ).
    """
    n = a.shape[0]

    # full convolution a*a
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # soft‑max probabilities for the entries of b
    max_b = np.max(b)
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)

    # gradient = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Projected gradient descent on the smoothed max
# ----------------------------------------------------------------------


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Annealed projected gradient descent on the log‑sum‑exp smooth max."""
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 1.0
    max_beta = 1e7
    momentum = np.zeros_like(a)
    rho = 0.9                     # momentum decay
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum step
        momentum = rho * momentum + (1.0 - rho) * g
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            lr = 0.6 * target_sum / (norm * (beta + 1.0))
            a -= lr * momentum

        # simplex projection (keeps Σa = √(2n))
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # exponential annealing of β
        if it % 150 == 0 and beta < max_beta:
            beta = min(beta * 1.07, max_beta)

        it += 1

    return a


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Random mass‑transfer SA with multiplicative cooling."""
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # initialise temperature from the current objective
    T = best_score * 0.02 + 1e-12
    cooling = 0.9994
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        # move a fraction of mass from i to j
        frac = rng.random() * 0.15
        delta = a[i] * frac
        if delta <= 0.0:
            continue
        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        accept = False
        if cur_score <= best_score:
            accept = True
        elif T > 0.0 and rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()

    return best


# ----------------------------------------------------------------------
# Greedy mass‑transfer descent (targeting the current maximal entry)
# ----------------------------------------------------------------------


def _greedy_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Repeatedly move a fraction of mass from the index that contributes most
    to the current maximal convolution entry to an index with (almost) no
    contribution.  A move is kept only if the maximal entry does not increase.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        # full convolution and location of the max entry
        b = _full_convolution_fft(a)
        max_b = b.max()
        k = int(b.argmax())

        # indices that actually contribute to b_k
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idxs = np.arange(i_start, i_end + 1)
        # contribution of a_i to b_k is a_{k-i}
        contrib = a[k - idxs]

        # index with largest contribution (the culprit)
        i_max = idxs[np.argmax(contrib)]

        # choose a target index where the contribution is zero (or tiny)
        # For k < n‑1 any index > k gives zero; otherwise any index < k‑n+1 gives zero.
        if k < n - 1:
            j_candidates = [n - 1]               # far right side
        else:
            j_candidates = [0]                    # far left side

        # add a couple of random alternatives to escape plateaux
        for _ in range(2):
            j_candidates.append(rng.integers(0, n))

        # discard the source index itself
        j_candidates = [j for j in j_candidates if j != i_max]

        improved = False
        # try a few candidate targets, keep the best non‑worsening move
        for j in j_candidates:
            if a[i_max] <= 1e-15:
                break
            # try a batch of fractions; start with 20 % and fall back to 5 %
            for frac in (0.2, 0.05):
                delta = a[i_max] * frac
                if delta <= 0.0:
                    continue
                a_new = a.copy()
                a_new[i_max] -= delta
                a_new[j] += delta
                new_max = _max_convolution_fft(a_new)
                if new_max <= max_b + 1e-12:       # accept non‑increase
                    a = a_new
                    improved = True
                    break
            if improved:
                break

        # If no improving move was found, shrink the step size globally
        if not improved:
            # tiny jitter to shake the system
            jitter = rng.normal(scale=target_sum * 1e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    # final projection to be safe
    a = _project_to_simplex(a, target_sum)
    return a


# ----------------------------------------------------------------------
# Parametric families (power‑law / symmetric power‑law)
# ----------------------------------------------------------------------


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    """
    a_i = (i + shift)^{-p}, i = 0,…,n‑1.
    The vector is projected onto the ℓ¹‑simplex Σa = √(2 n).
    """
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    target_sum = np.sqrt(2.0 * n)
    scale = target_sum / raw.sum()
    a = raw * scale
    a = np.maximum(a, 0.0)
    return a


def _symmetric_powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    """
    Symmetric version: a_i = (i+shift)^{-p} + (n‑1‑i+shift)^{-p}.
    Afterwards projected onto the ℓ¹‑simplex.
    """
    idx = np.arange(n, dtype=float)
    left = (idx + shift) ** (-p)
    right = (n - 1 - idx + shift) ** (-p)
    raw = left + right
    target_sum = np.sqrt(2.0 * n)
    scale = target_sum / raw.sum()
    a = raw * scale
    a = np.maximum(a, 0.0)
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the time budget, builds a few parametric seeds,
    applies a greedy descent, refines with projected gradient descent and
    finishes with simulated annealing.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    deadline = time.time() + budget_s - 1.0   # safety margin

    best_seq = None
    best_obj = np.inf

    # --------------------------------------------------------------
    # 1. Light parametric sweep (few candidates, generous time budget)
    # --------------------------------------------------------------
    lengths = [4000, 6000, 8000]
    p_grid = [0.48, 0.5, 0.52, 0.55]
    shift_grid = [0.5, 1.0, 2.0, 4.0]

    # keep a small elite list (size ≤ 8)
    elite = []

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Exploring length {n}")

        for p in p_grid:
            for shift in shift_grid:
                # power‑law candidate
                a = _powerlaw_sequence(n, p, shift)
                obj = _max_convolution_fft(a)
                if obj < best_obj:
                    best_obj = obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] New global best (grid) n={n} p={p:.3f} "
                          f"shift={shift:.2f}  obj={best_obj:.6f}")

                # symmetric version
                a_sym = _symmetric_powerlaw_sequence(n, p, shift)
                obj_sym = _max_convolution_fft(a_sym)
                if obj_sym < best_obj:
                    best_obj = obj_sym
                    best_seq = a_sym.copy()
                    print(f"[{time.time():.1f}] New global best (sym) n={n} p={p:.3f} "
                          f"shift={shift:.2f}  obj={best_obj:.6f}")

                # maintain elite list (top few)
                if len(elite) < 8 or obj < elite[-1][0]:
                    elite.append((obj, a))
                    elite.sort(key=lambda x: x[0])
                    elite = elite[:8]
                if len(elite) < 8 or obj_sym < elite[-1][0]:
                    elite.append((obj_sym, a_sym))
                    elite.sort(key=lambda x: x[0])
                    elite = elite[:8]

        # short random sampling for diversity (2 s per length)
        rnd_end = min(deadline, time.time() + 2.0)
        while time.time() < rnd_end:
            a_rand = rng.random(n)
            a_rand = a_rand / a_rand.sum() * np.sqrt(2.0 * n)
            obj = _max_convolution_fft(a_rand)
            if obj < best_obj:
                best_obj = obj
                best_seq = a_rand.copy()
                print(f"[{time.time():.1f}] New global best (rand) n={n}  obj={best_obj:.6f}")

    # --------------------------------------------------------------
    # 2. Optionally incorporate a previously discovered construction
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            h = np.asarray(height_sequence_1, dtype=float)
            n_h = h.size
            target = np.sqrt(2.0 * n_h)
            h = h * (target / h.sum())
            obj_h = _max_convolution_fft(h)
            if obj_h < best_obj:
                best_obj = obj_h
                best_seq = h.copy()
                print(f"[{time.time():.1f}] Adopted height_sequence_1 with obj={best_obj:.6f}")
                elite.append((obj_h, h))
                elite.sort(key=lambda x: x[0])
                elite = elite[:8]
        except Exception:
            pass

    # --------------------------------------------------------------
    # 3. Greedy descent on the current best (or on each elite candidate)
    # --------------------------------------------------------------
    # Run greedy descent on a handful of elite vectors; keep the best afterwards.
    for _, seed_vec in elite:
        if time.time() >= deadline:
            break
        n = seed_vec.size
        sub_dead = min(deadline, time.time() + 20.0)   # 20 s per elite vector
        a_greedy = _greedy_descent_opt(seed_vec, sub_dead)
        obj = _max_convolution_fft(a_greedy)
        if obj < best_obj:
            best_obj = obj
            best_seq = a_greedy.copy()
            print(f"[{time.time():.1f}] New best after greedy n={n} obj={best_obj:.6f}")

    # If we somehow have no candidate yet (unlikely), fall back to uniform.
    if best_seq is None:
        n = 4000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_obj = _max_convolution_fft(best_seq)

    # --------------------------------------------------------------
    # 4. Refine with projected gradient descent (smooth max)
    # --------------------------------------------------------------
    n = best_seq.size
    sub_dead = min(deadline, time.time() + 120.0)   # up to 2 min for GD
    best_seq = _gradient_descent_opt(best_seq, sub_dead)
    best_obj = _max_convolution_fft(best_seq)
    print(f"[{time.time():.1f}] After gradient descent obj={best_obj:.6f}")

    # --------------------------------------------------------------
    # 5. Final simulated‑annealing refinement (uses all remaining time)
    # --------------------------------------------------------------
    best_seq = _simulated_annealing(best_seq, deadline)
    best_obj = _max_convolution_fft(best_seq)
    print(f"[{time.time():.1f}] Final SA objective = {best_obj:.6f}")

    # Final safety: enforce non‑negativity and project onto simplex
    best_seq = np.maximum(best_seq, 0.0)
    target_sum = np.sqrt(2.0 * best_seq.size)
    best_seq = _project_to_simplex(best_seq, target_sum)

    # Trim trailing near‑zero entries for neatness
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Optional sanity check – just to log the evaluate_sequence value
    try:
        eval_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence = {eval_val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective = {best_obj:.6f} "
          f"length = {best_seq.size}")

    return best_seq.tolist()
