def _proj_gradient_descent(a0, deadline, max_iter=100000):

    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0*n)
    step_alpha = 0.5
    step = 0
    best_a = a.copy()
    best_score = _max_convolution_fft(a)

    while time.time() < deadline and step < max_iter:
        step += 1
        # compute gradient at max entry
        b = _full_convolution_fft(a)
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i = np.arange(lo, hi+1)
        j = k_max - i
        g = np.zeros_like(a)
        g[i] = 2.0 * a[j]

        # step size schedule
        eta = step_alpha / np.sqrt(step)

        # gradient descent step: a - eta * g
        a_new = a - eta * g

        # projection onto simplex of target sum (nonnegative)
        # ensure no negative numbers messing with projection; but projection handles nonnegative.
        a_new = _project_to_simplex(a_new, target_sum)

        # compute objective
        cur_score = _max_convolution_fft(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            print(f"[{time.time():.1f}] GD improvement: {best_score:.6f}")

        # optionally early stop if no improvement after many steps. But we can continue.

        a = a_new

"""
propose_candidate – heuristic optimizer for the self‑convolution ℓ∞ problem.

The algorithm consists of four stages:

1. **Deterministic family sweep** – for several lengths we generate
   sequences from simple parametric families (edge‑heavy power law,
   pure power law, exponential, mixed power‑law + exponential and
   Beta‑shaped).  Every shape is projected onto the ℓ¹‑simplex
   Σa_i = √(2 n) and its maximal convolution entry is evaluated.
   The best shape becomes the current incumbent.

2. **Random seeding** – a short Monte‑Carlo phase creates additional
   candidates (Dirichlet, power‑law, etc.) to possibly improve the
   incumbent.

3. **Projected sub‑gradient descent** – we directly minimise the true
   ℓ∞‑norm of the convolution.  At each iteration we locate the
   index k where (a*a)_k is maximal, compute the exact sub‑gradient
   (which is simply 2·a_{k‑i}), take a step of size η proportional to
   1/‖g‖ and project back onto the ℓ¹‑simplex.  The step size shrinks
   as 1/√t.  This often reduces the worst‑case convolution entry
   faster than smooth‑max approximations.

4. **Targeted simulated annealing** – a final refinement that moves a
   random fraction of mass from an index contributing heavily to the
   current worst entry towards a random destination.  Uphill moves are
   accepted with a slowly cooling temperature.

All stages respect the supplied time budget (default 1000 s).  The
returned value is a plain Python list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities
# ----------------------------------------------------------------------


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_full_convolution_fft(a).max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean, Duchi et al. 2008)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x_i≥0, Σx_i = s}."""
    n = v.shape[0]
    if s <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for deterministic sweep
# ----------------------------------------------------------------------


def _make_edge_heavy(n: int, p: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    return (idx + 1.0) ** (-p) + (n - idx) ** (-p)


def _make_power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    return (idx + 1.0) ** (-p)


def _make_exponential(n: int, lam: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    return np.exp(-lam * idx)


def _make_mixed(n: int, p: float, lam: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    return (idx + 1.0) ** (-p) * np.exp(-lam * idx)


def _make_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=float) + 0.5) / n
    return (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))


# ----------------------------------------------------------------------
# Random initialisers (diverse families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Generate a random non‑negative sequence on the ℓ¹‑simplex."""
    target_sum = np.sqrt(2.0 * n)
    kind = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if kind == 0:  # pure power‑law (decreasing)
        p = rng.uniform(0.15, 1.0)
        a = (idx + 1.0) ** (-p)

    elif kind == 1:  # exponential decay from the left edge
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif kind == 2:  # mixed power‑law + exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif kind == 3:  # Dirichlet (uniform simplex)
        a = rng.dirichlet(alpha=np.ones(n))

    elif kind == 4:  # symmetric edge‑heavy
        p = rng.uniform(0.2, 1.2)
        left = (idx + 1.0) ** (-p)
        right = (n - idx) ** (-p)
        a = left + right

    else:  # Beta‑shaped
        alpha = rng.uniform(0.2, 2.0)
        beta = rng.uniform(0.2, 2.0)
        x = (idx + 0.5) / n
        a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))

    a += rng.normal(scale=target_sum * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target_sum)
    return a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true ℓ∞‑norm
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Iteratively reduce max(a*a) using the exact sub‑gradient."""
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = _max_convolution_fft(a)

    step = 0
    base_eta = 1.0  # controls initial step magnitude

    while time.time() < deadline:
        step += 1

        # compute full convolution and locate the worst index
        conv = _full_convolution_fft(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the worst entry
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k_max - i
        g = np.zeros_like(a)
        g[i] = 2.0 * a[j]

        grad_norm = np.linalg.norm(g)
        if grad_norm == 0:
            break

        # diminishing step size (η ∝ 1/(‖g‖·√t))
        eta = base_eta / (grad_norm * np.sqrt(step))

        # gradient step + projection onto the ℓ¹‑simplex
        a_new = a - eta * g
        a_new = _project_to_simplex(a_new, target_sum)

        cur_score = _max_convolution_fft(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            print(f"[{time.time():.1f}] GD improvement: {best_score:.6f}")

        a = a_new

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted simulated‑annealing refinement
# ----------------------------------------------------------------------


def _targeted_simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """SA that moves mass from indices heavily involved in the worst entry."""
    a = a0.copy()
    n = a.shape[0]

    best = a.copy()
    best_score = _max_convolution_fft(best)

    # temperature proportional to current objective
    T = best_score * 0.02 + 1e-8
    cooling = 0.9995
    rng = np.random.default_rng()

    max_transfer = 0.25  # up to 25 % of the selected mass

    while time.time() < deadline:
        conv = _full_convolution_fft(a)
        k_max = int(np.argmax(conv))

        # indices contributing to the worst entry
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idx_range = np.arange(lo, hi + 1)
        contrib = a[idx_range] * a[k_max - idx_range]
        weight_sum = contrib.sum()

        if weight_sum > 0:
            probs = contrib / weight_sum
            i = rng.choice(idx_range, p=probs)
        else:
            i = rng.integers(0, n)

        # random destination ≠ i
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        # transfer a fraction of mass from i to j
        delta = a[i] * rng.random() * max_transfer
        if delta <= 0.0:
            continue
        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score

        accept = False
        if cur_score <= best_score:
            accept = True
        elif rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA improvement: {best_score:.6f}")
        else:
            # revert move
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, np.sqrt(2.0 * n))
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA jitter improvement: {best_score:.6f}")

    return best

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative sequence that minimises evaluate_sequence.
    The routine combines a deterministic family sweep, random seeding,
    projected sub‑gradient descent and a final targeted SA.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # If a previously discovered sequence exists, use it as a starting point.
    # ------------------------------------------------------------------
    global_seq = globals().get("height_sequence_1")
    if isinstance(global_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(global_seq, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        score0 = _max_convolution_fft(a0)
        best_seq, best_score = a0.copy(), score0
        print(f"[{time.time():.1f}] Start from global height_sequence_1 "
              f"(n={n0})  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – deterministic family sweep (multiple lengths)
    # ------------------------------------------------------------------
    lengths = [800, 1200, 2000, 3000, 4000]   # moderate sizes that fit the budget
    p_vals = np.linspace(0.2, 2.0, 10)
    lam_vals = np.logspace(-4, -0.5, 12)
    alpha_vals = np.linspace(0.3, 3.0, 9)
    beta_vals = np.linspace(0.3, 3.0, 9)

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # ---- edge‑heavy power law ----
        for p in p_vals:
            a_raw = _make_edge_heavy(n, p)
            if a_raw.sum() == 0:
                continue
            a = a_raw * (target_sum / a_raw.sum())
            score = _max_convolution_fft(a)
            if score < best_score:
                best_seq, best_score = a.copy(), score
                print(f"[{time.time():.1f}] New best (edge‑heavy) n={n}, p={p:.3f}  score={score:.6f}")

        # ---- pure power law ----
        for p in p_vals:
            a_raw = _make_power_law(n, p)
            a = a_raw * (target_sum / a_raw.sum())
            score = _max_convolution_fft(a)
            if score < best_score:
                best_seq, best_score = a.copy(), score
                print(f"[{time.time():.1f}] New best (power‑law) n={n}, p={p:.3f}  score={score:.6f}")

        # ---- exponential decay ----
        for lam in lam_vals:
            a_raw = _make_exponential(n, lam)
            a = a_raw * (target_sum / a_raw.sum())
            score = _max_convolution_fft(a)
            if score < best_score:
                best_seq, best_score = a.copy(), score
                print(f"[{time.time():.1f}] New best (exp) n={n}, lam={lam:.5f}  score={score:.6f}")

        # ---- mixed power‑law + exponential ----
        for p in p_vals[::2]:        # sparse subset to keep time reasonable
            for lam in lam_vals[::2]:
                a_raw = _make_mixed(n, p, lam)
                a = a_raw * (target_sum / a_raw.sum())
                score = _max_convolution_fft(a)
                if score < best_score:
                    best_seq, best_score = a.copy(), score
                    print(f"[{time.time():.1f}] New best (mixed) n={n}, p={p:.3f}, lam={lam:.5f} "
                          f"score={score:.6f}")

        # ---- Beta‑shaped ----
        for alpha in alpha_vals:
            for beta in beta_vals:
                a_raw = _make_beta_shape(n, alpha, beta)
                a = a_raw * (target_sum / a_raw.sum())
                score = _max_convolution_fft(a)
                if score < best_score:
                    best_seq, best_score = a.copy(), score
                    print(f"[{time.time():.1f}] New best (beta) n={n}, α={alpha:.2f}, β={beta:.2f} "
                          f"score={score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – random seeding (short burst) to diversify the pool
    # ------------------------------------------------------------------
    random_lengths = [1200, 2000, 3000]
    random_time_per_len = 6.0
    for n in random_lengths:
        if time.time() >= deadline:
            break
        end_phase = min(deadline, time.time() + random_time_per_len)
        while time.time() < end_phase:
            a = _random_initial_sequence(n, rng)
            score = _max_convolution_fft(a)
            if score < best_score:
                best_seq, best_score = a.copy(), score
                print(f"[{time.time():.1f}] New best (random) n={n}  score={score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform if nothing was generated yet
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1200
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = _max_convolution_fft(best_seq)
        print(f"[{time.time():.1f}] Fallback to uniform: n={n}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent (use half of remaining time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 8.0:                           # ensure we have a tiny margin
        gd_deadline = time.time() + remaining * 0.5
        a_gd, gd_score = _subgradient_descent(best_seq, gd_deadline)
        if gd_score < best_score:
            best_seq, best_score = a_gd.copy(), gd_score
            print(f"[{time.time():.1f}] New best after GD: score={gd_score:.6f}")

        # ------------------------------------------------------------------
        # Phase 4 – targeted simulated annealing (remaining time)
        # ------------------------------------------------------------------
        a_sa = _targeted_simulated_annealing(best_seq, deadline)
        final_score = _max_convolution_fft(a_sa)
        if final_score < best_score:
            best_seq, best_score = a_sa.copy(), final_score
            print(f"[{time.time():.1f}] New best after SA: score={final_score:.6f}")

    # ------------------------------------------------------------------
    # Strip trailing near‑zero entries, return as a plain list
    # ------------------------------------------------------------------
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Optional sanity check with official evaluator
    try:
        ev = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence = {ev:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective (max‑conv) = "
          f"{best_score:.6f}   length = {best_seq.size}")

    return best_seq.tolist()
