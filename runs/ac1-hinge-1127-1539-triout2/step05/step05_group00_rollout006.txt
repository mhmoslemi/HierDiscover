"""
Propose a step‑function that minimizes the objective used in
``evaluate_sequence``.

The algorithm consists of three parts:

1. **Exploration with sparse random masks** – for a set of lengths we
   repeatedly draw a random support set (10‑30 % of the entries) and
   assign the total ℓ¹‑mass sqrt(2 n) either uniformly or with a simple
   power‑law / exponential decay on the support.  The convolution maximum
   is obtained with an FFT and the best mask seen so far is kept.
   Because the support is sparse the aperiodic autocorrelation becomes
   much flatter than for a dense vector, often giving a value < 1.5.

2. **Mass‑transfer hill climbing** – starting from the best mask we run
   a simulated‑annealing hill‑climb that repeatedly moves a random
   fraction of mass from one index to another while updating the full
   convolution analytically (O(n) per move).  This refines the mask and
   usually lowers the maximum further.

3. **Final polishing** – any remaining budget is spent on a final
   hill‑climb of the current best sequence.

All numbers are kept non‑negative and the ℓ¹‑norm is exactly
``sqrt(2·n)`` throughout, so ``evaluate_sequence`` simply returns the
maximum entry of the self‑convolution.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Return max entry of the linear convolution a * a using FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a as a NumPy array (length 2n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the simplex (not used in the main loop)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    { x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Hill climbing with exact convolution updates
# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Simulated‑annealing hill‑climb that preserves the ℓ¹‑norm.
    a0 – initial sequence (already normalised)
    b0 – its full convolution a0*a0
    The loop runs until ``deadline`` and returns the best pair (a, b)
    observed.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()
    best_b = b.copy()

    rng = np.random.default_rng()
    # initial temperature proportional to the current max
    T = cur_score * 0.02 + 1e-9
    cooling = 0.9997

    while time.time() < deadline:
        # source proportional to its mass
        total_mass = a.sum()
        if total_mass == 0.0:
            break
        probs = a / total_mass
        i = rng.choice(n, p=probs)

        if a[i] < 1e-14:
            continue

        # destination uniformly (allow zero‑entries)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move a random fraction (up to 50 %) of the source mass
        max_delta = a[i] * 0.5
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ---- compute new convolution (O(n)) ----
        b_new = b.copy()
        # contribution of –δ at i
        b_new[i:i + n] -= 2.0 * delta * a
        # contribution of +δ at j
        b_new[j:j + n] += 2.0 * delta * a

        # quadratic correction terms (self‑products)
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # ---- acceptance test ----
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-(new_score - cur_score) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                best_b = b.copy()
                # occasional progress report
                print(f"[{time.time():.1f}] hill‑climb improvement: {best_score:.6f}")

        T *= cooling

    return best_a, best_b


# ----------------------------------------------------------------------
# Random sparse mask generator
# ----------------------------------------------------------------------


def _random_mask_candidate(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative vector on the ℓ¹‑simplex Σa = sqrt(2·n)
    whose support is a random subset (≈10‑30 % of the coordinates).
    Three families for the values on the support are used:
    uniform, power‑law, exponential.
    """
    target = np.sqrt(2.0 * n)

    # support size (p fraction of n)
    p = rng.uniform(0.15, 0.30)          # 15‑30 % of the entries
    m = max(2, int(p * n))
    if m > n:
        m = n

    # choose indices without replacement
    support = rng.choice(n, size=m, replace=False)

    # decide the shape of the weights on the support
    family = rng.integers(0, 3)

    if family == 0:                     # uniform on the support
        w = target / m
        a = np.zeros(n, dtype=float)
        a[support] = w

    elif family == 1:                   # descending power‑law
        power = rng.uniform(0.15, 1.2)
        raw = (np.arange(1, m + 1, dtype=float) ** (-power))
        raw *= target / raw.sum()
        a = np.zeros(n, dtype=float)
        a[support] = raw

    else:                               # exponential decay on the support
        lam = rng.exponential(scale=0.02)
        raw = np.exp(-lam * np.arange(m, dtype=float))
        raw *= target / raw.sum()
        a = np.zeros(n, dtype=float)
        a[support] = raw

    # tiny jitter to break exact ties (helps the optimiser later)
    jitter_scale = target * 0.0005
    a[support] += rng.normal(scale=jitter_scale, size=m)

    # keep non‑negative and renormalise to the exact simplex
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0.0:
        a *= (target / s)
    else:
        a = np.full(n, target / n)

    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects the wall‑clock budget
    (default 1000 s) and returns the best sequence found as a plain Python
    list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Seed from a previously stored construction, if present
    # --------------------------------------------------------------
    earlier = globals().get("height_sequence_1")
    if isinstance(earlier, (list, tuple, np.ndarray)):
        a0 = np.asarray(earlier, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        score0 = _max_convolution_fft(a0)
        best_seq = a0.copy()
        best_score = score0
        print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) score={score0:.6f}")

    # --------------------------------------------------------------
    # Length schedule – we try a few different dimensions
    # --------------------------------------------------------------
    lengths = [2000, 3500, 5000, 8000, 12000]

    # -----------------------------------------------------------------
    # Phase A – fast random‑mask exploration (≈80 % of the budget)
    # -----------------------------------------------------------------
    explore_deadline = start + budget_s * 0.80
    while time.time() < explore_deadline:
        # pick a length (cycle through the list)
        n = lengths[int(rng.random() * len(lengths))]
        a = _random_mask_candidate(n, rng)
        score = _max_convolution_fft(a)

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] New best (n={n}) score={score:.6f}")

        # occasionally up‑sample the current best to a larger length
        # (helps us reuse a good pattern when the budget permits)
        if best_seq is not None and best_seq.size < n and rng.random() < 0.05:
            # simple linear interpolation + simplex projection
            old_n = best_seq.size
            x_old = np.linspace(0, 1, old_n)
            x_new = np.linspace(0, 1, n)
            a_up = np.interp(x_new, x_old, best_seq)
            a_up = np.maximum(a_up, 0.0)
            a_up = _project_to_simplex(a_up, np.sqrt(2.0 * n))
            up_score = _max_convolution_fft(a_up)
            if up_score < best_score:
                best_score = up_score
                best_seq = a_up.copy()
                print(f"[{time.time():.1f}] Upsampled improvement (n={n}) score={up_score:.6f}")

    # -----------------------------------------------------------------
    # Phase B – targeted hill‑climbing on the best sequence found
    # -----------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        b_full = _full_convolution(best_seq)
        hill_deadline = deadline  # use whatever time is left
        a_hc, b_hc = _hill_climbing(best_seq, b_full, hill_deadline)
        final_score = b_hc.max()
        if final_score < best_score:
            best_score = final_score
            best_seq = a_hc.copy()
            print(f"[{time.time():.1f}] Final hill‑climb improvement: score={final_score:.6f}")

    # -----------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros for neatness
    # -----------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # -----------------------------------------------------------------
    # Log the official evaluation (objective = max of the self‑convolution)
    # -----------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
