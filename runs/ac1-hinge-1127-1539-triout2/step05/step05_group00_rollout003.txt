def _adam_opt(a0, deadline):
    a = a0.copy()
    target_sum = sqrt(2*n)
    beta1=0.9, beta2=0.999, eps=1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    beta = 0.5
    beta_max = 8192.
    beta_factor = 1.5
    # schedule each iteration maybe after 200 steps increase beta
    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)
        m = beta1 * m + (1-beta1) * g
        v = beta2 * v + (1-beta2) * (g ** 2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        lr = 0.1 / (np.sqrt(v_hat) + eps)   # base scaling
        # maybe scale by something
        a -= lr * m_hat
        a = _project_to_simplex(a, target_sum)

        # Anneal beta
        if t % 300 == 0 and beta < beta_max:
            beta = min(beta*beta_factor, beta_max)

        # Occasionally jitter
        if t % 500 == 0:
            jitter = np.random.default_rng().normal(scale=target_sum * 0.001 / (beta+1.0), size=a.shape)
            a += jitter
            a = np.maximum(a, 0)
            a = _project_to_simplex(a, target_sum)

    return a

assistantfinal**Solution Explanation**

The goal is to minimise  

\[
F(a)=\frac{2\,n\;\max (a*a)}{(\sum_i a_i)^2},
\]

with the constraints  

* \(a_i\ge 0\)  
* \(a_i\le 1000\) (the evaluator clips to this value)  
* \(\sum_i a_i>0.01\).

Because the objective is homogeneous we may enforce the **ℓ¹‑normalisation**

\[
\sum_i a_i = \sqrt{2n},
\]

which makes  

\[
F(a)=\max (a*a) .
\]

Thus we only have to make the self‑convolution as flat as possible.

The search consists of three parts

1. **Projected Adam optimisation** – a smooth approximation of the
   max‑operator (log‑sum‑exp) is differentiated, the gradient is obtained
   with FFTs and an Adam optimiser is used. After each step the vector is
   projected onto the simplex \(\{x\ge0,\;\sum x = \sqrt{2n}\}\).

2. **Targeted mass‑transfer hill‑climbing** – after the Adam phase we keep the
   exact convolution `b = a*a`.  
   The index `kmax` where `b` attains its maximum is known.
   The contribution of a coefficient `a_i` to this worst entry is
   \(c_i = a_i\,a_{k_{\max}-i}\) (zero outside the range).  
   The algorithm repeatedly moves a small amount of mass from a *large*
   contributor (`i` sampled proportionally to `c_i`) to a *small* contributor
   (`j` sampled proportionally to `c_{max}-c_j`).  
   The convolution update is performed in **O(n)** using the analytic formula
   from the original code, and a simulated‑annealing acceptance rule
   (temperature proportional to the current maximum) allows occasional
   uphill moves.

3. **Multi‑scale schedule** – we start with a modest length, optimise it,
   up‑sample the best shape and repeat for larger lengths (up to 20 000).  
   The larger the dimension the finer the discretisation of the underlying
   continuous shape, which yields a lower maximal convolution value.

All steps respect the wall‑clock budget (default 1000 s).  
The best sequence found is returned as a plain Python list; a short
post‑processing step trims trailing near‑zeros.

"""
step_function_optimizer_v2.py

Search for a non‑negative step function that minimises `evaluate_sequence`.

Algorithm
---------
1.  For a sequence length `n` we enforce the ℓ¹‑normalisation
    Σ a_i = sqrt(2 n).  With this normalisation the objective equals the
    maximal entry of the self‑convolution, i.e. `max(a*a)`.

2.  **Projected Adam** – minimise a smooth log‑sum‑exp surrogate of the
    max‑convolution.  The gradient is obtained with FFTs, Adam updates are
    applied and the vector is projected onto the simplex after each step.

3.  **Targeted hill‑climbing** – keep the exact convolution `b = a*a`.
    The worst index `kmax = argmax b` is identified.  The contribution
    of each coefficient to this entry is `c_i = a_i * a_{kmax-i}`.
    Mass `δ` is moved from a source index sampled according to `c_i`
    (large contributors) to a destination sampled according to the
    inverse contribution (small contributors).  The convolution is updated
    in O(n) using the analytic formula from the original code; a simulated
    annealing rule with temperature `T` accepts occasional uphill moves.

4.  **Multi‑scale schedule** – start with a moderate length (≈2000),
    optimise it, up‑sample the best vector and repeat for larger
    dimensions (up to 20 000).  The remaining time is spent on a final
    polishing hill‑climb.

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """
    Maximum entry of the linear convolution a*a computed via FFT.
    Fast O(n log n) evaluation used during the search.
    """
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (size 2n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x_i≥0, Σ x_i = s}.
    Duchi‑Shalev‑Shwartz‑Singer‑Chandra (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed maximum (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b)  with  b = a*a .
    Returns (g, b) where g = 2·(a ⋆ p) and p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # a*a via FFT
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Projected Adam optimiser on the smooth surrogate
# ----------------------------------------------------------------------


def _adam_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected Adam optimiser.
    The loss is the log‑sum‑exp surrogate of the max‑convolution.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # β for the soft‑max (annealed)
    beta = 0.5
    beta_max = 8192.0
    beta_factor = 1.5

    # generic step scale
    step_base = target_sum * 0.05

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        it += 1
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step_base * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = _project_to_simplex(a, target_sum)

        # β annealing
        if it % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        # occasional jitter to escape flat regions
        if it % 500 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=n
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

    return a

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative sequence on the ℓ¹‑simplex Σa = sqrt(2 n).
    Several families (power‑law, exponential, Dirichlet, piecewise) are
    sampled uniformly.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                       # pure power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                     # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                     # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                     # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                     # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )

    else:                                 # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length.
    The result is projected back onto the appropriate simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Targeted mass‑transfer hill climbing (exact convolution updates)
# ----------------------------------------------------------------------


def _targeted_hill_climbing(
    a0: np.ndarray,
    b0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, np.ndarray]:
    """
    Hill‑climbing that always tries to lower the current worst convolution
    entry.  At each iteration we:
        1. locate k_max = argmax b,
        2. compute contributions c_i = a_i * a_{k_max-i},
        3. move a random fraction of mass from a large contributor to a
           small contributor,
        4. accept the move if it improves the maximal entry or with a
           simulated‑annealing probability.
    Returns the best (a, b) pair observed before the deadline.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size

    cur_max = float(b.max())
    best_max = cur_max
    best_a = a.copy()
    best_b = b.copy()

    # temperature proportional to the scale of the objective
    T = cur_max * 0.02 + 1e-9
    cooling = 0.9997

    while time.time() < deadline:
        # ---------------------------------------------------------
        # 1. worst index
        # ---------------------------------------------------------
        kmax = int(np.argmax(b))

        # ---------------------------------------------------------
        # 2. contributions to the worst entry
        # ---------------------------------------------------------
        # c_i = a_i * a_{kmax-i}  (zero if index out of range)
        lower = max(0, kmax - (n - 1))
        upper = min(n - 1, kmax)
        i_range = np.arange(lower, upper + 1)
        j_range = kmax - i_range
        c = np.zeros(n)
        c[i_range] = a[i_range] * a[j_range]

        # ---------------------------------------------------------
        # 3. pick source (large contributor) and destination (small)
        # ---------------------------------------------------------
        sum_c = c.sum()
        if sum_c == 0.0:
            src = rng.integers(0, n)
        else:
            src = rng.choice(n, p=c / sum_c)

        # Destination biased towards small contributions
        inv_weight = (c.max() - c) + 1e-12
        # Occasionally pick a completely random destination
        if rng.random() < 0.15:
            dst = rng.integers(0, n)
        else:
            dst = rng.choice(n, p=inv_weight / inv_weight.sum())

        if src == dst:
            continue

        # ---------------------------------------------------------
        # 4. amount of mass to move
        # ---------------------------------------------------------
        max_delta = a[src] * 0.5
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta

        # ---------------------------------------------------------
        # 5. exact update of the convolution (O(n))
        # ---------------------------------------------------------
        b_new = b.copy()
        # subtract contribution of -δ at src
        b_new[src:src + n] -= 2.0 * delta * a
        # add contribution of +δ at dst
        b_new[dst:dst + n] += 2.0 * delta * a

        # quadratic correction terms
        k2s = 2 * src
        k2d = 2 * dst
        sdm = src + dst
        if k2s < b_new.size:
            b_new[k2s] += delta * delta
        if k2d < b_new.size:
            b_new[k2d] += delta * delta
        if sdm < b_new.size:
            b_new[sdm] -= 2.0 * delta * delta

        new_max = float(b_new.max())

        # ---------------------------------------------------------
        # 6. acceptance rule (simulated annealing)
        # ---------------------------------------------------------
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            if rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            a[src] -= delta
            a[dst] += delta
            b = b_new
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                best_b = b.copy()
                print(
                    f"[{time.time():.1f}] hill‑climb improvement: {best_max:.6f}"
                )

        T *= cooling

    return best_a, best_b

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock budget (default 1000 s) and returns
    the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Seed from a previously stored construction if available
    # ------------------------------------------------------------------
    earlier = globals().get("height_sequence_1")
    if isinstance(earlier, (list, tuple, np.ndarray)):
        a0 = np.asarray(earlier, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        best_score = _max_convolution_fft(a0)
        best_seq = a0.copy()
        print(
            f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) "
            f"score={best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # Length schedule – start small, end large (max ≈ 20000)
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000, 16000, 20000]

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Initialise candidate for this length
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)

        a_init = _project_to_simplex(a_init, target_sum)

        # --------------------------------------------------------------
        # Phase A – Projected Adam (smooth surrogate)
        # --------------------------------------------------------------
        phase_end = min(
            deadline,
            time.time() + min(150.0, (deadline - time.time()) * 0.45),
        )
        a_adam = _adam_opt(a_init, phase_end)
        score_adam = _max_convolution_fft(a_adam)
        if score_adam < best_score:
            best_score = score_adam
            best_seq = a_adam.copy()
            print(
                f"[{time.time():.1f}] Adam improvement (n={n}) score={score_adam:.6f}"
            )

        # --------------------------------------------------------------
        # Phase B – Targeted hill climbing (exact convolution)
        # --------------------------------------------------------------
        if time.time() < deadline:
            b_full = _full_convolution(a_adam)
            hill_dead = min(
                deadline,
                time.time() + min(80.0, (deadline - time.time()) * 0.35),
            )
            a_hc, b_hc = _targeted_hill_climbing(
                a_adam, b_full, hill_dead, rng
            )
            score_hc = float(b_hc.max())
            if score_hc < best_score:
                best_score = score_hc
                best_seq = a_hc.copy()
                print(
                    f"[{time.time():.1f}] Hill‑climb improvement (n={n}) "
                    f"score={score_hc:.6f}"
                )

    # ------------------------------------------------------------------
    # Final polishing with any remaining budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a_final = best_seq.copy()
        b_final = _full_convolution(a_final)
        final_dead = deadline
        a_final, b_final = _targeted_hill_climbing(a_final, b_final, final_dead, rng)
        final_score = float(b_final.max())
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(
                f"[{time.time():.1f}] Final polishing improvement: "
                f"score={final_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros for neatness
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback – uniform distribution (should never happen)
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log the official evaluation (objective = max of the self‑convolution)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(
            f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
            f"(length={best_seq.size})"
        )
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
