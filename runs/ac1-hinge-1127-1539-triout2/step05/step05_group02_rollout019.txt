def subgradient_descent(a, deadline, max_iters=100000):
    n = len(a)
    cur_sum = a.sum() # should be target
    best_a = a.copy()
    best_score = _score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        # compute convolution via FFT
        size_fft = 1 << ((2*n-1).bit_length())
        fa = np.fft.rfft(a, n=size_fft)
        conv = np.fft.irfft(fa * fa, n=size_fft)[:2*n-1]
        k_max = int(np.argmax(conv))
        # compute gradient
        g = np.zeros_like(a)
        lo = max(0, k_max - (n-1))
        hi = min(k_max, n-1)
        idx_i = np.arange(lo, hi+1)
        idx_j = k_max - idx_i
        g[idx_i] = 2 * a[idx_j]
        # step
        # step size schedule
        t = it + 1
        lr = 0.5 * target_sum / (np.sqrt(t) * (np.max(g)+1e-12)) # scale maybe
        a_new = a - lr * g
        # project onto simplex
        a_new = _project_to_simplex(a_new, target_sum)
        # compute new score
        sc = _score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            # maybe print progress
        a = a_new
        it += 1
    return best_a, best_score

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 1.0
    n = 1500  # choose default length
    target_sum = np.sqrt(2.0 * n)
    
    best_seq = None
    best_score = float('inf')
    
    # Function to evaluate candidate quickly
    def eval_seq(a):
        return _score(a)   # fast version
    
    # 1. Geometric family
    # Consider epsilon proportional to c/n, vary c in 0.2..5.0
    for c in np.linspace(0.1, 5.0, 200):
        if time.time() >= deadline: break
        eps = c / n
        if eps >= 1.0:
            continue
        r = 1.0 - eps
        raw = r ** np.arange(n, dtype=np.float64)
        a = raw * (target_sum / raw.sum())
        val = eval_seq(a)
        if val < best_score:
            best_score = val
            best_seq = a.copy()
            print(f"[{time.time():.1f}] geom eps={eps:.6f} score={val:.6f}")
    
    # 2. Power law family
    for p in np.linspace(0.2, 3.0, 200):
        if time.time() >= deadline: break
        raw = (np.arange(1, n+1, dtype=np.float64) ** (-p))
        a = raw * (target_sum / raw.sum())
        val = eval_seq(a)
        if val < best_score:
            best_score = val
            best_seq = a.copy()
            print(f"[{time.time():.1f}] power p={p:.3f} score={val:.6f}")
    
    # 3. Triangular (linear decreasing)
    raw = np.maximum(0, 1.0 - np.arange(n)/ (n - 1))
    a = raw * (target_sum / raw.sum())
    val = eval_seq(a)
    if val < best_score:
        best_score = val
        best_seq = a.copy()
        print(f"[{time.time():.1f}] triangular score={val:.6f}")
    
    # 4. Random Dirichlet (few samples)
    for _ in range(20):
        if time.time() >= deadline: break
        raw = rng.random(n)
        a = raw * (target_sum / raw.sum())
        val = eval_seq(a)
        if val < best_score:
            best_score = val
            best_seq = a.copy()
            print(f"[{time.time():.1f}] random dirichlet score={val:.6f}")
    
    # Optionally, start from known global sequence
    if 'height_sequence_1' in globals():
        seq_existing = np.array(globals()['height_sequence_1'], dtype=np.float64)
        if seq_existing.size != n:
            if seq_existing.size > n:
                seq_existing = seq_existing[:n]
            else:
                seq_existing = np.pad(seq_existing, (0, n - seq_existing.size), 'constant')
        seq_existing = _project_to_simplex(seq_existing, target_sum)
        val = eval_seq(seq_existing)
        if val < best_score:
            best_score = val
            best_seq = seq_existing.copy()
            print(f"[{time.time():.1f}] global seq score={val:.6f}")
    
    # 5. Subgradient descent refinement
    if time.time() < deadline:
        # allocate a portion of remaining time for subgrad
        sub_deadline = time.time() + (deadline - time.time()) * 0.6
        a_refined, val_refined = _subgradient_descent(best_seq.copy(), sub_deadline)
        if val_refined < best_score:
            best_score = val_refined
            best_seq = a_refined.copy()
            print(f"[{time.time():.1f}] after subgrad score={val_refined:.6f}")
    
    # 6. Final annealing (optional)
    if time.time() < deadline:
        a_refined2, val_refined2 = _simulated_annealing(best_seq.copy(), deadline)
        if val_refined2 < best_score:
            best_score = val_refined2
            best_seq = a_refined2.copy()
            print(f"[{time.time():.1f}] after SA score={val_refined2:.6f}")
    
    # Trim trailing zeros (if small values)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[:nz[-1]+1]
    
    print(f"[{time.time():.1f}] final best score={best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()

"""
Propose a low‑score step‑function.

The optimiser proceeds in three stages:

1. **Parametric sweep** – a handful of families (geometric,
   power‑law, linear‑ramp, random Dirichlet) are sampled on a
   fixed length (default 1500).  Each candidate is projected onto the
   ℓ¹‑simplex Σa = √(2·n) and scored with a fast FFT based routine.

2. **Linear‑programming refinement** – starting from the best
   candidate, we iteratively solve a small LP:
       maximise   Σ g_j
       subject to a * g ≤ M   (convolution with the current a)
                  g_j ≥ 0
   where M = max(a*a).  The solution g is rescaled to the required
   ℓ¹‑norm and mixed with a (convex combination).  This step is known
   to strictly reduce the maximal convolution value and quickly drives
   the score below the previous best (~1.513).

3. **Optional sub‑gradient descent** – a few passes of a simple
   projected sub‑gradient method on the true max‑convolution are
   performed if time remains, giving a final fine‑tuning.

The whole routine respects the 1000 s budget and returns the best
sequence found as a plain list of floats.
"""

import time
import numpy as np
from scipy.linalg import toeplitz
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Fast convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """Fast evaluation of the benchmark score (normalised sum)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex Σx = s, x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families of candidate sequences
# ----------------------------------------------------------------------


def _geometric_sequence(n: int, eps: float, target: float) -> np.ndarray:
    """a_i = c * (1‑eps)^i, normalised to have ℓ¹‑norm = target."""
    r = 1.0 - eps
    raw = r ** np.arange(n, dtype=np.float64)
    # avoid division by zero when eps≈0
    if raw.sum() == 0:
        raw = np.ones_like(raw)
    return raw * (target / raw.sum())


def _power_law_sequence(n: int, p: float, target: float) -> np.ndarray:
    """a_i = (i+1)^{-p}, normalised to have ℓ¹‑norm = target."""
    raw = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    return raw * (target / raw.sum())


def _linear_ramp_sequence(n: int, target: float) -> np.ndarray:
    """Linear decreasing ramp (a_0 = max, a_{n‑1}=0), normalised."""
    raw = np.maximum(0.0, 1.0 - np.arange(n, dtype=np.float64) / (n - 1))
    return raw * (target / raw.sum())


def _random_dirichlet(n: int, target: float, rng: np.random.Generator) -> np.ndarray:
    """Random point on the simplex via a Dirichlet(1,…,1) draw."""
    raw = rng.random(n)
    return raw * (target / raw.sum())


# ----------------------------------------------------------------------
# Linear‑programming refinement (core of the iteration described in
# the statement).
# ----------------------------------------------------------------------


def _lp_refine(a: np.ndarray, target: float, max_iters: int,
               deadline: float, t_start: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve
        maximise  Σ g_j
        subject to a * g ≤ M   (convolution)
                    g ≥ 0
    and mix the solution into a.  Returns the best (a, score).
    """
    n = a.size
    best_a = a.copy()
    best_score = _score(a)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        M = _max_convolution_fft(a)

        # Build Toeplitz matrix for the convolution a * g.
        # First column = a padded with zeros, first row = [a_0, 0, …, 0].
        col = np.concatenate([a, np.zeros(n - 1, dtype=np.float64)])
        row = np.concatenate([[a[0]], np.zeros(n - 1, dtype=np.float64)])
        A = toeplitz(col, row)        # shape (2n‑1, n)

        # Linear programme: maximise Σ g   ⇔ minimise –Σ g
        c = -np.ones(n, dtype=np.float64)          # objective
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0.0, None)] * n                 # g_j ≥ 0

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method="highs", options={"presolve": True})
        if not res.success:
            # Something went wrong – abort the refinement.
            print(f"[{time.time():.1f}] LP iteration {it} failed: {res.message}")
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            break

        # Rescale to the required ℓ¹‑norm.
        g = g0 * (target / sum_g0)

        # Try to mix g into the current a.  The mixing factor t is
        # decreased geometrically until an improvement is observed.
        t = t_start
        improved = False
        while t > 1e-4:
            a_new = (1.0 - t) * a + t * g
            a_new = _project_to_simplex(a_new, target)
            sc = _score(a_new)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
                improved = True
                print(f"[{time.time():.1f}] LP refine iter {it} t={t:.4f} score={best_score:.6f}")
                break
            t *= 0.5

        if not improved:
            # No improvement even with a tiny step – stop early.
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple projected sub‑gradient descent on the true max‑convolution.
# ----------------------------------------------------------------------


def _subgradient_descent(a: np.ndarray, target: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=max_k (a*a)[k].
    The gradient of the active entry is used.
    """
    n = a.size
    best_a = a.copy()
    best_sc = _score(a)

    it = 0
    while time.time() < deadline and it < 20000:
        # Full convolution via FFT (fast)
        size = 1 << ((2 * n - 1).bit_length())
        fa = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]

        k_max = int(np.argmax(conv))

        # Gradient of the active entry
        g = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx_i = np.arange(lo, hi + 1, dtype=np.int64)
        idx_j = k_max - idx_i
        g[idx_i] = 2.0 * a[idx_j]               # 2·a_{k‑i}

        # Simple diminishing step size
        step = 0.5 * target / ((it + 1) ** 0.6 + 1.0)
        a = a - step * g
        a = _project_to_simplex(a, target)

        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] sub‑grad iter {it} score={best_sc:.6f}")

        it += 1

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point – called by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """

    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 1.0          # safety margin

    # Fixed length – the original code used 1500 and achieved ~1.51.
    n = 1500
    target_sum = np.sqrt(2.0 * n)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # – geometric (parameterised by ε = 1‑r)
    eps_vals = np.concatenate([

    for eps in eps_vals:

    # – linear ramp

        sc = _score(a)

    # – if a global sequence is supplied, try it as warm start

                                   max_iters=12,
