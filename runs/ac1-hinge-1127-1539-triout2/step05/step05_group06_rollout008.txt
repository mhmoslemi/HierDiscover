"""
Propose a non‑negative step sequence that minimises
evaluate_sequence(...).  The objective is the peak value of the
full (linear) self‑convolution of the sequence with the ℓ1‑norm
normalised to √(2·n).  The optimiser mixes three ideas:

* **Adam** on a smooth soft‑max approximation of the max‑entry.
  The gradient is obtained efficiently with FFT‑based convolutions.
* **Projected sub‑gradient** that directly attacks the current worst
  convolution entry (the “active’’ shift).
* **Combinatorial re‑ordering** – after continuous optimisation the
  values are reshuffled (reverse, “wave’’ interleaving) and the best
  ordering is kept.  Large‑scale up‑sampling gives the algorithm more
  degrees of freedom as time permits.

The routine runs for at most ``budget_s`` seconds (default 1000) and
returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – keeps Σ a_i = S and a_i ≥ 0.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σ x = s}."""
    if s <= 0:
        raise ValueError("simplex sum must be positive")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth max‑norm:
        g_j = 2 Σ_i w_{i+j} a_i .
    Both ``a`` (len n) and ``w`` (len 2·n‑1) are real vectors.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    # pick the slice that gives the required gradient (see derivation above)
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights (smooth approximation of max)
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """w_i = exp(alpha·conv_i) / Σ_j exp(alpha·conv_j)  (numerically stable)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Linear up‑sampling (used when we grow the problem size)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a to length new_n."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=float)
    return np.interp(x_new, x_old, a)


# ----------------------------------------------------------------------
# Heuristic re‑ordering helpers
# ----------------------------------------------------------------------
def _wave_reorder(a: np.ndarray) -> np.ndarray:
    """
    Put the largest values at the two ends and then walk
    inwards alternating left/right (large‑small interleaving).
    """
    vals = np.sort(a)[::-1]          # descending
    n = a.shape[0]
    out = np.empty_like(a)
    left, right = 0, n - 1
    toggle = True
    for v in vals:
        if toggle:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        toggle = not toggle
    return out


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam on a smooth soft‑max approximation of max(conv).
    Returns the best vector seen before *deadline*.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.5
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    alpha = 0.5                     # temperature – starts smooth
    while time.time() < deadline:
        t += 1

        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex
        a = _project_simplex(a, S)

        # occasional tiny mass‑swap for exploration
        if rng.random() < 0.03:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.01
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        cur = np.max(_full_convolution(a))
        if cur < best_score:
            best_score = cur
            best_a = a.copy()

        # temperature schedule – make the soft‑max sharper
        alpha = min(alpha * 1.02, 350.0)

        # cheap early‑stop
        if t % 6000 == 0 and cur == best_score:
            break

    return best_a, best_score


def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent that attacks the currently worst
    convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.30                # base step size
    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the max entry (factor 2 omitted – absorbed in η)
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur = np.max(_full_convolution(a))
        if cur < best_score:
            best_score = cur
            best_a = a.copy()

        # random jitter to escape plateaus
        if rng.random() < 0.04:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.02
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = np.max(_full_convolution(cand))
                if cand_score < cur or rng.random() < 0.2:
                    a = cand
                    cur = cand_score
                    if cur < best_score:
                        best_score = cur
                        best_a = a.copy()

        # cheap early‑stop
        if it % 8000 == 0 and cur == best_score:
            break

    return best_a, best_score


def _optimize_combined_adam(a0: np.ndarray,
                            deadline: float,
                            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run Adam for ≈55 % of the remaining time, then finish with sub‑gradient.
    """
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    # Adam phase
    split = now + 0.55 * remaining
    a_mid, sc_mid = _optimize_adam(a0, split, rng)

    # Sub‑gradient phase (if time left)
    if time.time() < deadline:
        a_end, sc_end = _optimize_subgrad(a_mid, deadline, rng)
        if sc_end < sc_mid:
            return a_end, sc_end
    return a_mid, sc_mid


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function (non‑negative vector) that minimises
    ``evaluate_sequence``.  The routine mixes Adam, projected
    sub‑gradient, up‑sampling, and simple combinatorial re‑ordering.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # safety margin for the final external call
    deadline = start + budget_s - 5.0

    best_seq = None
    best_score = np.inf
    attempts = 0
    last_improve = start

    # optionally warm‑start from a previously discovered construction
    if 'height_sequence_1' in globals():
        try:
            h = np.asarray(height_sequence_1, dtype=float)
            if h.ndim == 1 and h.size > 1:
                # project onto the required simplex (any length is fine)
                best_seq = _project_simplex(h, np.sqrt(2.0 * h.size))
                best_score = np.max(_full_convolution(best_seq))
                print("[warm‑start] initial external score =", best_score)
        except Exception:
            best_seq = None

    # ------------------------------------------------------------------
    # Main optimisation loop (many medium‑length trials)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        attempts += 1

        # decide problem size -------------------------------------------------
        if best_seq is None or (time.time() - last_improve) > 45:
            # fresh random start, size in a moderate range
            n = int(rng.integers(2000, 6000))
            a0 = rng.random(n)
        else:
            # occasionally grow the current best
            if rng.random() < 0.22 and len(best_seq) < 12000:
                n = min(int(len(best_seq) * 1.45), 12000)
                a0 = _upsample(best_seq, n)
            else:
                n = len(best_seq)
                a0 = best_seq.copy()
                # tiny jitter to keep the search alive
                jitter = rng.normal(scale=a0.mean() * 8e-5, size=n)
                a0 = np.clip(a0 + jitter, 0.0, None)

        # project onto the ℓ1‑simplex Σ a_i = √(2·n)
        S = np.sqrt(2.0 * n)
        a0_sum = a0.sum()
        if a0_sum == 0.0:
            a0 = np.full(n, 1.0)
            a0_sum = a0.sum()
        a0 = a0 / a0_sum * S

        # allocate a time slice for this trial (≈30 s or the remaining budget)
        slice_deadline = min(time.time() + 30.0, deadline)

        # run the combined optimiser
        a_opt, score_opt = _optimize_combined_adam(a0, slice_deadline, rng)

        # ------------------------------------------------------------------
        # Post‑optimisation re‑ordering heuristics (they keep the same multiset)
        # ------------------------------------------------------------------
        for reorder in (lambda x: x,                    # keep as is
                       lambda x: x[::-1],               # reverse
                       _wave_reorder):                  # wave interleaving
            cand = reorder(a_opt)
            # re‑project just in case numerical drift happened
            cand = _project_simplex(cand, np.sqrt(2.0 * cand.size))
            cand_score = np.max(_full_convolution(cand))
            if cand_score < score_opt:
                a_opt, score_opt = cand, cand_score

        # keep global best
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            last_improve = time.time()
            print(f"[global] attempt={attempts} n={n} new best max={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check & return
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()

    # One last external evaluation (the harness will also do this)
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} "
          f"(internal best = {best_score:.6f}) length = {len(final_seq)}")
    return final_seq
