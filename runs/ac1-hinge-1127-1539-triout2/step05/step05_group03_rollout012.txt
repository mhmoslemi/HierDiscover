def _sample_family(n, rng):
    typ = rng.integers(0, 10)
    S = np.sqrt(2.0 * n)
    a = None
    if typ == 0:
        # Uniform Dirichlet
        a = rng.random(n)
    elif typ == 1:
        # Dirichlet with alpha <1 (more sparse)
        alpha = rng.uniform(0.3, 0.7)
        a = rng.gamma(alpha, 1.0, size=n)
    elif typ == 2:
        # Dirichlet with alpha >1 (more even)
        alpha = rng.uniform(1.5, 3.0)
        a = rng.gamma(alpha, 1.0, size=n)
    elif typ == 3:
        # Exponential decay geometric
        r = rng.uniform(0.99, 0.999999)  # r close to 1
        a = r ** np.arange(n)
    elif typ == 4:
        # Power law decay
        p = rng.uniform(0.2, 2.0)
        a = (1.0 / (np.arange(1, n+1) ** p))
    elif typ == 5:
        # Sine window
        power = rng.uniform(0.5, 2.0)
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** power
    elif typ == 6:
        # Hann window
        a = 0.5 * (1 - np.cos(2 * np.pi * np.arange(n) / (n-1)))

        a = (0.42 - 0.5 * np.cos(2*np.pi*np.arange(n)/(n-1)) + 0.08*np.cos(4*np.pi*np.arange(n)/(n-1)))
    elif typ == 8:
        # Kaiser window with beta random

        alpha = (n - 1) / 2.0
        t = (i - alpha) / alpha
        a = i0(beta * np.sqrt(1 - t**2)) / i0(beta)
    else: # typ==9
        # Random convex combination of a few base shapes
        bases = []
        bases.append(rng.random(n))
        bases.append(r ** np.arange(n))
        sin_shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
        bases.append(sin_shape)
        weights = rng.random(len(bases))
        weights /= weights.sum()
        a = sum(w*b for w,b in zip(weights, bases))

    # Normalize to sum S
    if a is None or a.sum() == 0:

"""
Propose a step‑function (non‑negative vector) that lowers the
self‑convolution peak used by ``evaluate_sequence``.

Algorithm
---------
* **Exploratory stage** – quickly generate thousands of candidate
  vectors from a diverse collection of analytic windows
  (geometric, power‑law, sine, Hann, Blackman, Kaiser, …) and
  evaluate them with the black‑box evaluator.  The best candidate
  (lowest external score) is kept.

* **Refinement stage** – starting from the best candidate,
  run a two‑phase projected Adam optimiser on a smooth log‑sum‑exp
  surrogate of the max‑norm, followed by a sub‑gradient phase that
  directly reduces the worst convolution entry.  The optimiser
  respects the ℓ¹‑simplex (∑a = √(2·n)) and the non‑negativity bound.

* The whole search is time‑aware (≤ budget – 5 s) and prints
  progress information.  The final returned sequence is the best
  found before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of (1/α)·log‑sum‑exp(α·(a*a)) w.r.t. a.
    Returns a vector of length n.
    """
    n = a.shape[0]
    # reverse w for the convolution that appears in the gradient
    w_rev = w[::-1]
    conv_len = 3 * n - 2          # length of a * w_rev * a
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max (smooth max) utilities
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """w_i = exp(alpha*conv_i) / Σ_j exp(alpha*conv_j)  (stable)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Upsampling helpers
# ----------------------------------------------------------------------
def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Resize an existing sequence to length n using linear interpolation."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Diverse analytic families of candidate vectors
# ----------------------------------------------------------------------
def _sample_family(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n with ℓ¹‑norm √(2·n).
    A random family is selected from a set of analytic windows.
    """
    S = np.sqrt(2.0 * n)
    typ = rng.integers(0, 10)

    if typ == 0:                     # uniform Dirichlet(1)
        a = rng.random(n)

    elif typ == 1:                   # Dirichlet with α < 1 (sparser)
        alpha = rng.uniform(0.3, 0.7)
        a = rng.gamma(alpha, 1.0, size=n)

    elif typ == 2:                   # Dirichlet with α > 1 (more even)
        alpha = rng.uniform(1.5, 3.0)
        a = rng.gamma(alpha, 1.0, size=n)

    elif typ == 3:                   # geometric decay
        r = rng.uniform(0.97, 0.9999)
        a = r ** np.arange(n)

    elif typ == 4:                   # power‑law decay
        p = rng.uniform(0.2, 2.5)
        a = 1.0 / (np.arange(1, n + 1) ** p)

    elif typ == 5:                   # sine‑shaped window
        power = rng.uniform(0.5, 2.5)
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** power

    elif typ == 6:                   # Hann window
        a = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))

    elif typ == 7:                   # Blackman window
        a = (0.42
             - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
             + 0.08 * np.cos(4.0 * np.pi * np.arange(n) / (n - 1)))

    elif typ == 8:                   # Kaiser window (β random)
        try:
            from scipy.special import i0
            beta = rng.uniform(2.0, 10.0)
            i = np.arange(n, dtype=float)
            alpha = (n - 1) / 2.0
            t = (i - alpha) / alpha
            a = i0(beta * np.sqrt(1.0 - t ** 2)) / i0(beta)
        except Exception:
            # fallback to exponential if scipy is unavailable
            r = rng.uniform(0.97, 0.9999)
            a = r ** np.arange(n)

    else:                            # random convex combination of a few bases
        bases = [
            rng.random(n),
            (0.97 ** np.arange(n)),
            np.sin(np.pi * (np.arange(n) + 0.5) / n)
        ]
        w = rng.random(len(bases))
        w /= w.sum()
        a = sum(wi * bi for wi, bi in zip(w, bases))

    # Normalise to required ℓ¹‑norm
    if a.sum() == 0.0:
        a = np.full(n, 1.0)
    a = a / a.sum() * S
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smooth log‑sum‑exp max‑approximation
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected Adam minimising the smoothed max‑norm."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.25
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # start with a modest temperature and increase quickly
    alpha = 1.0
    alpha_mul = 1.02

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = np.max(conv)
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # smooth‑max weighting
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update with projection onto simplex
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a, S)

        # occasional random mass‑swap to escape plateaus
        if rng.random() < 0.025:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.05 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < best_score or rng.random() < 0.1:
                    a = cand

        # temperature schedule (sharper approximation)
        alpha = min(alpha * alpha_mul, 40.0)
        # gently decay learning rate
        lr *= 0.99995

        # early stop if we are stuck for a while
        if it % 4000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient phase (focus on the worst shift)
# ----------------------------------------------------------------------
def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the true max‑norm (only the worst entry)."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.15
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient for the worst shift
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # occasional random perturbation
        if rng.random() < 0.05:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.04 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.2:
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()

        if it % 5000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimiser (Adam → sub‑gradient)
# ----------------------------------------------------------------------
def _optimize_combined(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run a smooth Adam phase for ~70 % of the remaining budget,
    then a sub‑gradient refinement.
    """
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    # split 0.7 / 0.3 between smoothing and sub‑gradient
    smooth_deadline = now + 0.7 * remaining
    a_s, score_s = _optimize_adam(a0, smooth_deadline, rng)

    if time.time() < deadline:
        a_f, score_f = _optimize_subgrad(a_s, deadline, rng)
        if score_f < score_s:
            return a_f, score_f
    return a_s, score_s


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function minimising ``evaluate_sequence``.
    Returns the best sequence (list of floats) found before the time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin for the final external evaluation
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Use a previously discovered construction if available (allows up‑sampling)
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_overall_seq = None
    best_overall_score = np.inf
    attempts = 0
    last_improve_time = start_time

    # -------------------- exploratory random families --------------------
    explore_deadline = start_time + 0.25 * (deadline - start_time)
    while time.time() < explore_deadline:
        attempts += 1
        n = int(rng.integers(2000, 12000))
        seq = _sample_family(n, rng)
        score = evaluate_sequence(seq.tolist())
        if score < best_overall_score:
            best_overall_score = score
            best_overall_seq = seq.copy()
            last_improve_time = time.time()
            print(f"[explore] attempt={attempts} n={n} score={best_overall_score:.6f}")

    # -------------------- refinement phase --------------------
    while time.time() < deadline:
        attempts += 1

        # decide on a starting point for this optimisation run
        if best_overall_seq is None:
            # no good candidate yet – generate a fresh one
            n = int(rng.integers(2500, 15000))
            a0 = _sample_family(n, rng)
        else:
            # occasionally jitter or up‑sample the current best
            if rng.random() < 0.4:
                # up‑sample to a new length (adds extra degrees of freedom)
                n = int(rng.integers(2500, 15000))
                a0 = _upsample_seq(best_overall_seq, n)
                # tiny random perturbation to avoid exact symmetry
                a0 += rng.normal(scale=a0.mean() * 5e-4, size=n)
                a0 = np.clip(a0, 0.0, None)
                a0 = a0 / a0.sum() * np.sqrt(2.0 * n)
            else:
                # keep the same size but add some noise
                a0 = best_overall_seq.copy()
                a0 += rng.normal(scale=a0.mean() * 1e-3, size=a0.size)
                a0 = np.clip(a0, 0.0, None)
                a0 = a0 / a0.sum() * np.sqrt(2.0 * a0.size)

        slice_deadline = min(time.time() + 12.0, deadline)

        a_opt, score = _optimize_combined(a0, slice_deadline, rng)

        if score < best_overall_score:
            best_overall_score = score
            best_overall_seq = a_opt.copy()
            last_improve_time = time.time()
            print(f"[global] attempt={attempts} n={a_opt.size} "
                  f"new best score={best_overall_score:.6f}")

        # early stop if we already beat the target bound
        if best_overall_score <= 1.503:
            break

        # if we have been idle for a while, discard the old shape to diversify
        if time.time() - last_improve_time > 45.0:
            base_seq = None

    # ------------------------------------------------------------------
    # Final sanity check via the black‑box evaluator
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_overall_seq.tolist()

    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} "
          f"(internal best = {best_overall_score:.6f}) "
          f"length = {len(final_seq)}")
    return final_seq
