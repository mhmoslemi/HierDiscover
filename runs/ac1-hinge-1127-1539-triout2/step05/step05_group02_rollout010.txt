import time
import numpy as np

def _max_convolution_fft(a):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    return float(conv.max())

def _conv_fft(a):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    return conv

def _score(a):
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))

def _project_to_simplex(v, s):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / np.arange(1, u.shape[0] + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _power_law_sequence(n, p, target_sum):
    a = np.arange(1, n+1, dtype=np.float64) ** (-p)
    a *= target_sum / a.sum()
    return a

def _exponential_sequence(n, r, target_sum):
    a = r ** np.arange(n, dtype=np.float64)
    a *= target_sum / a.sum()
    return a

def _gradient_softmax(a, beta):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
    max_b = b.max()
    # compute softmax weights
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()
    # gradient: 2 * (a * p) convolved with reversed a
    a_rev = a[::-1]
    size2 = size
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:2*n-1]
    g = 2.0 * conv_rev[n-1:n-1+n]
    return g, b

def _annealed_gradient_descent(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_val = _score(a)
    beta = 1.0
    max_beta = 4096.0
    step_base = target_sum * 0.20
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        g_norm = np.linalg.norm(g)
        if g_norm > 0:
            lr = step_base / (g_norm * (beta + 1.0))
            a = a - lr * g
        # projection
        a = _project_to_simplex(a, target_sum)
        # occasional jitter to escape flat regions
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] GD improvement β={beta:.1f} score={best_val:.6f}")
        # increase β gradually
        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 1.3, max_beta)
        it += 1
    return best_a, best_val

def _subgradient_refine(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    size_fft = 1 << ((2*n - 1).bit_length())
    # simple linear cooling schedule for occasional uphill jumps
    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)
    it = 0
    while time.time() < deadline:
        # compute full convolution
        fa = np.fft.rfft(a, n=size_fft)
        conv = np.fft.irfft(fa * fa, n=size_fft)[:2*n-1]
        max_b = conv.max()
        k = int(np.argmax(conv))
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        # contribution weights
        weight = a[idx] * a[k - idx]
        donor = idx[np.argmax(weight)]
        receiver = idx[np.argmin(weight)]
        if donor == receiver:
            donor = rng.integers(n)
            receiver = rng.integers(n)
        max_delta = a[donor] * 0.15  # up to 15% of donor's mass
        delta = max_delta * rng.random()
        if delta <= 0:
            it += 1
            continue
        a[donor] -= delta
        a[receiver] += delta
        new_score = _score(a)
        # acceptance
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] subgrad improvement {best_score:.6f}")
        else:
            # tiny temperature for occasional uphill moves
            elapsed = time.time() - start_time
            tau = elapsed / total_time
            temp = 0.001 * (1 - tau) + 1e-6
            prob = np.exp(-(new_score - cur_score) / temp)
            if rng.random() < prob:
                cur_score = new_score
            else:
                # revert
                a[donor] += delta
                a[receiver] -= delta
        it += 1
    return best_a, best_score

def _simulated_annealing(a0, deadline,
                         start_temp=0.5, end_temp=1e-6,
                         max_frac_start=0.30, max_frac_end=0.002,
                         max_window=40):
    a = a0.copy()
    n = a.shape[0]
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    size_fft = 1 << ((2*n - 1).bit_length())
    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)
    # keep current convolution for faster max retrieval
    conv = _conv_fft(a)
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau
        # current maximum position
        k_max = int(np.argmax(conv))
        # sensitivity weights for active entry (as in original code)
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]
        weight = a * sens
        move_type = None
        # --- propose a move ---
        if rng.random() < 0.70:  # mass transfer
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight / total_w)
            else:
                donor = rng.integers(n)
            # pick a low‑weight receiver
            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            receiver = rng.choice(low_idx) if low_idx.size else rng.integers(n)
            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"
        elif rng.random() < 0.90:  # local smoothing
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                saved_seg = a[start:end].copy()
                avg = saved_seg.mean()
                a[start:end] = avg
                saved_start, saved_end = start, end
                move_type = "smooth"
        else:  # swap
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"
        # evaluate new state
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[:2*n-1]
        new_score = _score(a)
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA new best = {best_score:.6f}")
        else:
            # revert move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]
        # occasional global jitter to jump out of deep traps
        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = a * (a.sum() / a.sum())  # just to keep sum exact (tiny drift)
            # re‑compute convolution
            conv = _conv_fft(a)
            cur_score = _score(a)
        it += 1
    return best_a, best_score

def _run_one_instance(n, init_mode, param, deadline):
    target = np.sqrt(2.0 * n)
    if init_mode == "global" and "height_sequence_1" in globals():
        seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if seq.size > n:
            seq = seq[:n]
        else:
            seq = np.pad(seq, (0, n - seq.size), "constant")
    elif init_mode == "power":
        exp = param if param is not None else 0.55
        seq = _power_law_sequence(n, exp, target)
    elif init_mode == "exp":
        r = param if param is not None else 0.99
        seq = _exponential_sequence(n, r, target)
    elif init_mode == "random":
        rng = np.random.default_rng()
        seq = rng.random(n)
        seq = seq / seq.sum() * target
    else:
        seq = _power_law_sequence(n, 0.55, target)
    seq = _project_to_simplex(seq, target)
    # 1) gradient descent
    remaining = deadline - time.time()
    if remaining > 15.0:
        gd_deadline = time.time() + min(remaining * 0.30, 45.0)
        seq, gd_score = _annealed_gradient_descent(seq, gd_deadline)
    else:
        gd_score = _score(seq)
    # 2) sub‑gradient refinement
    cur_time = time.time()

"""
Optimiser for the step‑function benchmark.

The algorithm works in three stages:

1. Initialise a candidate (global warm‑start, power‑law, exponential or random)
   and project it onto the ℓ¹‑simplex Σa = √(2 n).

2. Projected gradient descent on a soft‑max approximation of the convolution
   maximum.  The temperature β is increased up to 4096, giving a very sharp
   approximation of the true max‑operator.

3. Two‑phase refinement:
   – a deterministic sub‑gradient “mass‑transfer’’ step that directly reduces
     the active convolution entry,
   – a long simulated‑annealing phase with sophisticated moves
     (mass transfer, short‑window smoothing, swaps).  All steps keep the
     sequence non‑negative and on the simplex.

Several runs with different lengths and initialisations are performed until
the time budget (default 1000 s) expires.  The best sequence found is returned
as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution / scoring utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (returns a NumPy array)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Fast evaluation of the objective (max convolution) for a normalised sum."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Implements Duchi et al. (2008).
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / np.arange(1, u.shape[0] + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families of initial sequences
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, r: float, target_sum: float) -> np.ndarray:
    """Create a geometric sequence r**i of length n and normalise to target_sum."""
    a = r ** np.arange(n, dtype=np.float64)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Gradient descent on the soft‑max approximation of max(conv)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # gradient: 2 * (a_rev * p) convolution, then slice back to length n
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _annealed_gradient_descent(a0: np.ndarray, deadline: float):
    """Projected gradient descent on the soft‑max approximation."""
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 4096.0          # sharper than the original implementation
    step_base = target_sum * 0.20
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        g_norm = np.linalg.norm(g)
        if g_norm > 0:
            lr = step_base / (g_norm * (beta + 1.0))
            a -= lr * g

        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:                     # tiny stochastic jitter
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] GD improvement β={beta:.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 1.3, max_beta)

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic sub‑gradient refinement (mass transfer)
# ----------------------------------------------------------------------


def _subgradient_refine(a0: np.ndarray, deadline: float):
    """Move a small amount of mass from the most “responsible’’ index to the
    least responsible one.  Accept only improvements (tiny uphill moves are
    occasionally allowed)."""
    a = a0.copy()
    n = a.shape[0]
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    size_fft = 1 << ((2 * n - 1).bit_length())
    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)

    it = 0
    while time.time() < deadline:
        # full convolution for the current state
        fa = np.fft.rfft(a, n=size_fft)
        conv = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        max_b = conv.max()
        k = int(np.argmax(conv))

        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each coordinate to the active entry
        weight = a[idx] * a[k - idx]
        donor = idx[np.argmax(weight)]
        receiver = idx[np.argmin(weight)]

        # fallback to random indices if everything is equal
        if donor == receiver:
            donor = rng.integers(n)
            receiver = rng.integers(n)

        max_delta = a[donor] * 0.15          # up to 15 % of donor's mass
        delta = max_delta * rng.random()
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        new_score = _score(a)

        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑gradient improvement {best_score:.6f}")
        else:
            # tiny temperature to occasionally accept uphill moves
            elapsed = time.time() - start_time
            tau = elapsed / total_time
            temp = 0.001 * (1.0 - tau) + 1e-12
            prob = np.exp(-(new_score - cur_score) / temp)
            if rng.random() < prob:
                cur_score = new_score
            else:
                a[donor] += delta
                a[receiver] -= delta
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing (same structure as the original code, but with
# a few tuned parameters and a final jitter that re‑projects onto the simplex)
# ----------------------------------------------------------------------


def _simulated_annealing(
    a0: np.ndarray,
    deadline: float,
    start_temp: float = 0.5,
    end_temp: float = 1e-6,
    max_frac_start: float = 0.30,
    max_frac_end: float = 0.002,
    max_window: int = 40,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    size_fft = 1 << ((2 * n - 1).bit_length())
    conv = _conv_fft(a)                       # current convolution

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # active convolution entry and its sensitivities
        k_max = int(np.argmax(conv))
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]
        weight = a * sens

        # ------------------------------------------------------------------
        # Propose a move
        # ------------------------------------------------------------------
        move_type = None
        if rng.random() < 0.70:                     # mass transfer
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight / total_w)
            else:
                donor = rng.integers(n)

            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            receiver = rng.choice(low_idx) if low_idx.size else rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        elif rng.random() < 0.90:                 # local smoothing
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                saved_seg = a[start:end].copy()
                avg = saved_seg.mean()
                a[start:end] = avg
                saved_start, saved_end = start, end
                move_type = "smooth"

        else:                                     # random swap
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # ------------------------------------------------------------------
        # Evaluate the new state (single FFT)
        # ------------------------------------------------------------------
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = _score(a)

        # ------------------------------------------------------------------
        # Metropolis acceptance
        # ------------------------------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA new best = {best_score:.6f}")
        else:
            # revert the move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter to shake the system
        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# One optimisation run for a given length / initialisation
# ----------------------------------------------------------------------


def _run_one_instance(
    n: int, init_mode: str, param, deadline: float
) -> tuple[np.ndarray, float]:
    """
    Optimise a single instance of length ``n``.
    ``init_mode`` can be:
        * "global" – use the supplied ``height_sequence_1`` if present,
        * "power"  – power‑law with exponent ``param`` (default 0.55),
        * "exp"    – exponential decay with ratio ``param`` (default 0.99),
        * "random" – uniform random start.
    Returns the best sequence and its score.
    """
    target = np.sqrt(2.0 * n)

    if init_mode == "global" and "height_sequence_1" in globals():
        seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if seq.size > n:
            seq = seq[:n]
        else:
            seq = np.pad(seq, (0, n - seq.size), "constant")
    elif init_mode == "power":
        exp = param if param is not None else 0.55
        seq = _power_law_sequence(n, exp, target)
    elif init_mode == "exp":
        r = param if param is not None else 0.99
        seq = _exponential_sequence(n, r, target)
    elif init_mode == "random":
        rng = np.random.default_rng()
        seq = rng.random(n)
        seq = seq / seq.sum() * target
    else:
        seq = _power_law_sequence(n, 0.55, target)

    seq = _project_to_simplex(seq, target)

    # -------- 1) gradient descent (soft‑max) --------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        gd_deadline = time.time() + min(remaining * 0.30, 45.0)
        seq, gd_score = _annealed_gradient_descent(seq, gd_deadline)
    else:
        gd_score = _score(seq)

    # -------- 2) deterministic sub‑gradient refinement -----------------------
    cur_time = time.time()
    if cur_time < deadline:
        sub_deadline = time.time() + min((deadline - cur_time) * 0.15, 30.0)
        seq, sub_score = _subgradient_refine(seq, sub_deadline)
    else:
        sub_score = gd_score

    # -------- 3) simulated annealing -----------------------------------------
    cur_time = time.time()
    if cur_time < deadline:
        seq, sa_score = _simulated_annealing(seq, deadline)
    else:
        sa_score = sub_score

    final_score = min(gd_score, sub_score, sa_score)
    return seq, final_score
