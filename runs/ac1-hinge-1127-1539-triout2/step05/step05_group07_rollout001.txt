def propose_candidate(seed=42, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5

    best_seq = None
    best_score = np.inf

    # Helper functions: convolution, eval_score
    # Already defined _full_convolution_fft

    def eval_metric(a):
        n = len(a)
        b = _full_convolution_fft(a)
        max_b = b.max()
        sum_a = a.sum()
        if sum_a < 0.01:
            return np.inf
        return 2.0 * n * max_b / (sum_a ** 2)

    # 1. Random parametric exploration
    # Determine time budget portion for exploration ~0.3
    explore_deadline = deadline - 300  # keep 300 seconds for refinement
    while time.time() < explore_deadline:
        # choose n
        n = int(rng.integers(1500, 20000))
        # choose shape type
        shape_type = rng.choice(['edge_exp','edge_product','cosine','uniform'])
        if shape_type == 'uniform':
            a = np.full(n, 1.0)
        elif shape_type == 'edge_exp':
            p = rng.uniform(0.3, 1.2)
            dist = np.arange(n)
            d = np.minimum(dist, n-1-dist) + 1
            w = d.astype(float) ** (-p)
            a = w
        elif shape_type == 'edge_product':
            p = rng.uniform(0.3, 1.2)
            idx = np.arange(n)
            d1 = idx + 1
            d2 = n - idx
            w = (d1 * d2) ** (-p)

        score = 2.0 * n * peak / (np.sum(a_opt) ** 2)
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[refine] n={n} new best score={score:.6f}")

    # final output
    if best_seq is None:
        final_seq = [np.sqrt(2.0)]
    else:

        external = evaluate_sequence(final_seq)
        print(f"[final] external score = {external:.6f}, internal best = {best_score:.6f}, length = {len(final_seq)}")
        globals()['height_sequence_1'] = final_seq

"""
adversarial step‑function optimizer (enhanced)

The search consists of three stages:

1. **Broad parametric sampling** – generate a large number of candidate
   sequences from a few simple analytic families (edge‑heavy power laws,
   product‑style, cosine bump, uniform).  Each candidate is normalized so
   that its ℓ₁‑norm equals √(2·n).  The best of these fast‑evaluated
   candidates is kept.

2. **Refinement with Adam on the simplex** – starting from the current
   best candidate (or a scaled version of it) we run a soft‑max Adam
   optimiser that directly minimizes the smooth‑max approximation of the
   self‑convolution peak.  The optimiser runs for a configurable slice of
   the remaining budget; after each run the best sequence seen is kept.

3. **Optional local perturbations** – a few final random mass‑shifts are
   tried around the best solution to escape tiny plateaus.

The objective for a normalized sequence (∑a = √(2n)) is simply
`max(a*a)`, because the evaluation function reduces to
`2·n·max_b / (∑a)² = max_b`.  All steps respect the non‑negativity and
clipping performed by ``evaluate_sequence`` (the search never exceeds
the 1000‑value cap).  The routine stops early if it reaches the target
score 1.503.

The implementation is completely self‑contained and uses only NumPy
(and the provided ``evaluate_sequence``) – no external solvers are
required.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, scaling, etc.)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Compute the full linear self‑convolution a*a via a real FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Resize a sequence by linear interpolation.
    """
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.ceil(pos).astype(int)
    hi = np.clip(hi, 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Soft‑max approximation of max(b) and its gradient
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Returns (g, max_b) where
        g = ∂/∂a  τ‑soft‑max(b)
        max_b = max(b)   with b = a*a (full convolution)
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        # Exact sub‑gradient: equally split weight among all active entries
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)      # soft‑max weights, shape (2n‑1,)

    # Gradient w.r.t. a: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + n - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimisation on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth‑max approximation of the self‑convolution peak.
    Parameters
    ----------
    a0 : np.ndarray
        Feasible start (non‑negative, sum = √(2·n)).
    deadline : float
        Absolute time after which the optimiser must stop.
    rng : np.random.Generator
        Random generator for occasional jitter.
    Returns
    -------
    best_a : np.ndarray
        The best vector encountered during the run.
    best_peak : float
        Corresponding hard (non‑smoothed) convolution peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                # target ℓ₁‑norm

    # Initialise logits (log‑softmax) so that softmax(p)*S == a0
    eps = 1e-12
    prob = np.maximum(a0 / S, eps)
    prob = prob / prob.sum()
    p = np.log(prob)                     # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.25

    # Initialise smoothing parameter τ
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 1.0)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ---- a from logits (softmax on simplex) ----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ---- gradient of smooth max ----
        g, peak = _softmax_gradient(a, tau)

        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ---- gradient w.r.t. logits (chain rule) ----
        mu = np.dot(a, g) / S          # scalar = ⟨a,∇⟩ / S
        grad_p = a * (g - mu)          # shape (n,)

        # Optional gradient clipping for stability
        clip_norm = np.linalg.norm(grad_p)
        if clip_norm > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_norm)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- schedules ----
        tau *= 0.9993                 # tighten soft‑max
        lr  *= 0.9997                 # gentle decay

        # occasional jitter in log‑space (helps escape plateaus)
        if it % 400 == 0:
            p += rng.normal(scale=0.015, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Metric computation (identical to evaluate_sequence for unclipped values)
# ----------------------------------------------------------------------
def _metric(a: np.ndarray) -> float:
    """
    Compute 2·n·max(a*a) / (sum(a)²) for a non‑negative array a.
    The function mirrors the behaviour of evaluate_sequence for values
    that stay below the clipping threshold (≤1000).
    """
    n = len(a)
    if n == 0:
        return np.inf
    b = _full_convolution_fft(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine mixes fast parametric sampling with a few long Adam
    refinements, keeping the best candidate seen.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # safety margin for the harness

    # --- keep the best candidate globally (allows warm‑starts) ---
    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 1. Broad parametric exploration (≈30 % of the budget)
    # ------------------------------------------------------------------
    explore_deadline = end_time - 300.0   # reserve ~5 min for refinement
    while time.time() < explore_deadline:
        # pick a length (medium‑sized to keep evaluations cheap)
        n = int(rng.integers(1500, 22000))

        # pick a shape family
        shape = rng.choice(['edge_pow', 'edge_prod', 'cosine', 'uniform'])
        if shape == 'uniform':
            w = np.ones(n)
        elif shape == 'edge_pow':
            p = rng.uniform(0.35, 1.25)
            idx = np.arange(n)
            d = np.minimum(idx, n - 1 - idx) + 1  # distance to nearest edge (1‑based)
            w = d.astype(float) ** (-p)
        elif shape == 'edge_prod':
            p = rng.uniform(0.3, 1.2)
            idx = np.arange(n)
            d1 = idx + 1
            d2 = n - idx
            w = (d1 * d2) ** (-p)
        else:  # cosine bump (positive everywhere)
            i = np.arange(n)
            w = 1.0 + np.cos(np.pi * (2 * i - n + 1) / n)

        # add a small random perturbation in ~10 % of cases
        if rng.random() < 0.25:
            w += rng.normal(scale=0.05 * w.mean(), size=n)
            w = np.clip(w, 0.0, None)

        # normalise to the required ℓ₁‑norm √(2·n)
        S = np.sqrt(2.0 * n)
        a = w / w.sum() * S

        score = _metric(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[explore] n={n:5d} shape={shape:10s} "
                  f"{'p=' + f'{p:.3f}' if shape!='uniform' else ''} "
                  f"score={score:.6f}")

        # early stop if we already beat the target
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 2. Refinement with Adam (use remaining time)
    # ------------------------------------------------------------------
    while time.time() < end_time:
        # choose a starting point
        if best_seq is not None and rng.random() < 0.75:
            # sometimes scale the best known sequence to a nearby length
            if rng.random() < 0.5:
                scale = rng.uniform(0.8, 1.2)
                n_new = max(500, int(len(best_seq) * scale))
                a0 = _scale_sequence(best_seq, n_new)
            else:
                a0 = best_seq.copy()
            # jitter a little to avoid exact stagnation
            jitter = rng.normal(scale=0.01 * a0.mean(), size=a0.shape)
            a0 = np.clip(a0 + jitter, 0.0, None)
            # renormalise to the exact ℓ₁‑norm for its length
            S = np.sqrt(2.0 * a0.shape[0])
            a0 = a0 / a0.sum() * S
        else:
            # fresh random start
            n_new = int(rng.integers(1500, 25000))
            a0 = rng.random(n_new)
            S = np.sqrt(2.0 * n_new)
            a0 = a0 / a0.sum() * S

        # allocate a slice of the remaining budget (≈25 s per run, but never
        # exceed the global deadline)
        slice_deadline = min(time.time() + 25.0, end_time)

        a_opt, peak = _adam_optimize(a0, slice_deadline, rng)
        score = 2.0 * a_opt.shape[0] * peak / (np.sum(a_opt) ** 2)

        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[refine] n={a_opt.shape[0]:5d}  new best score={score:.6f}")

        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 3. Final tiny local search (optional, only if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < end_time - 2.0:
        n = best_seq.shape[0]
        S = np.sqrt(2.0 * n)
        for _ in range(200):
            # propose a small random mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = rng.uniform(0.0, 0.02) * best_seq[i]
            if delta <= 0:
                continue
            a_new = best_seq.copy()
            a_new[i] -= delta
            a_new[j] += delta
            # enforce non‑negativity (already guaranteed) and renorm
            a_new = a_new / a_new.sum() * S
            score = _metric(a_new)
            if score < best_score:
                best_score = score
                best_seq = a_new
                print(f"[local] improved to {score:.6f}")

                if best_score <= 1.503:
                    break
        # end of tiny local search

    # ------------------------------------------------------------------
    # Return the best sequence found (as plain Python list)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – a trivial normalized constant vector
        n = 1
        best_seq = np.array([np.sqrt(2.0 * n)])
    final_seq = best_seq.tolist()
    # sanity check via the black‑box evaluator (may be slightly different
    # if any entry hit the 1000‑clip, but that is harmless)
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  internal best = {best_score:.6f} "
          f"length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
