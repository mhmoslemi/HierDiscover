def _targeted_hill_climbing(a, b, deadline):
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()
    best_b = b.copy()
    rng = np.random.default_rng()
    T = cur_score * 0.02 + 1e-9
    cooling = 0.9995
    while time.time() < deadline:
        max_idx = int(np.argmax(b))
        # contributions array:
        # for i in [0,n):
        #   partner = max_idx - i
        #   if 0 <= partner < n:
        #       contrib[i] = a[i] * a[partner]
        #   else:
        #       contrib[i] = 0
        # Vectorized:
        idx = np.arange(n)
        partner = max_idx - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]
        # source i: choose from mask with probability prop to contrib (or weighted)
        total_contrib = contrib.sum()
        if total_contrib == 0:
            # fallback random
            i = rng.integers(0,n)
        else:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        # ensure a[i] > 0
        if a[i] < 1e-12:
            continue
        # pick destination j among indices not contributing heavily
        # e.g., choose j uniformly among those with a[j] low.
        # to bias, compute small values:
        low_mask = a < (a.mean())
        if low_mask.any():
            low_idx = np.where(low_mask)[0]
            j = rng.choice(low_idx)
        else:
            j = rng.integers(0,n)
        if i == j:
            continue
        # amount delta
        max_delta = a[i] * 0.5
        delta = rng.random() * max_delta
        if delta <= 0:
            continue
        # update b formula same as before:
        b_new = b.copy()
        # subtract contributions from source i: -2*delta*a (full a vector)
        b_new[i:i+n] -= 2.0 * delta * a
        # add contributions from destination j
        b_new[j:j+n] += 2.0 * delta * a
        # quadratic corrections
        k2i = 2*i
        k2j = 2*j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta*delta
        if k2j < b_new.size:
            b_new[k2j] += delta*delta
        if kij < b_new.size:
            b_new[kij] -= 2.0*delta*delta
        new_score = b_new.max()
        # acceptance: same as before
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-(new_score - cur_score)/T):
                accept = True
        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                best_b = b.copy()
                print(f"[{time.time():.1f}] target hill improvement {best_score:.6f}")
        T *= cooling
    return best_a, best_b

"""
Propose a step‑function that minimises the evaluation metric.

Algorithm (≈ 1 000 s budget)

1. **Initialisation** – start from a stored construction (if available) or
   generate a random non‑negative vector on the ℓ¹‑simplex Σa = √(2 n).
   Several parametric families (power‑law, exponential, Dirichlet, etc.) are
   sampled.

2. **Projected gradient descent** – minimise a smooth log‑sum‑exp
   approximation of the infinity‑norm of the self‑convolution.
   The inverse temperature β is annealed, momentum is used and a small
   jitter is added every few hundred iterations.  After each projection the
   vector is forced to be symmetric (a ← (a+a[::-1])/2) – symmetry empirically
   flattens the convolution.

3. **Mass‑transfer hill climbing** – a cheap O(n) update of the full
   convolution is used while moving a random fraction of mass from a large
   entry to a small entry.  The usual simulated‑annealing acceptance rule
   permits occasional uphill steps.

4. **Targeted hill climbing** – the index `k` where the current convolution
   attains its maximum is identified.  The contribution of each coordinate to
   this peak is `c_i = a_i·a_{k‑i}` (zero when the partner is out of range).
   A source coordinate is drawn with probability proportional to `c_i`,
   while the destination is chosen among the currently small entries.
   The exact convolution update described in the comment above is applied,
   and the move is accepted with a Metropolis rule.

5. **Length schedule** – the above three phases are performed for a schedule
   of increasing lengths (up to ≈20 000).  After each length the best vector
   is up‑sampled (linear interpolation) and used as the seed for the next
   length.

6. **Final polishing** – a last targeted hill‑climb runs for any remaining
   time.  Trailing near‑zero entries are stripped before returning the
   sequence.

All steps respect the global wall‑clock deadline; the best sequence found
so far is always kept.  The function returns a plain Python list ready for
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed maximum (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of   (1/β)·log Σ exp(β·b)   where   b = a*a .
    Returns (g, b) with g = 2·(a ⋆ p) and
    pₖ ∝ exp(β·bₖ).  Correlation is performed via FFT.
    """
    n = a.shape[0]

    # b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]                          # reversed a
    conv_len = n + (2 * n - 1) - 1           # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smooth max.
    Momentum, annealed β and occasional jitter are used.
    After each projection the vector is symmetrised.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 8192.0
    beta_factor = 1.4

    step_base = target_sum * 0.15
    momentum = np.zeros_like(a)
    rho = 0.90

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        # projection + symmetrisation
        a = _project_to_simplex(a, target_sum)
        a = (a + a[::-1]) * 0.5

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            a = (a + a[::-1]) * 0.5

        # β annealing
        if it % 350 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative sequence on the ℓ¹‑simplex Σa = √(2 n).
    Several families (power‑law, exponential, Dirichlet, piecewise) are
    sampled uniformly.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # pure power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n) + 1)
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    a = (a + a[::-1]) * 0.5
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length.
    The result is projected back onto the appropriate simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    a_new = (a_new + a_new[::-1]) * 0.5
    return a_new

# Mass‑transfer hill climbing (exact convolution updates)
# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Classical hill‑climbing with random mass transfers.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()
    best_b = b.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.02 + 1e-9
    cooling = 0.9997

    while time.time() < deadline:
        # source chosen proportionally to its mass
        if a.sum() == 0:
            break
        probs = a / a.sum()
        i = rng.choice(n, p=probs)

        if a[i] < 1e-14:
            continue

        # uniform destination
        j = rng.integers(0, n)
        if i == j:
            continue

        # move at most 50 % of source
        max_delta = a[i] * 0.5
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ----- exact convolution update -----
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a

        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-(new_score - cur_score) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                best_b = b.copy()
                print(f"[{time.time():.1f}] hill_climb improvement: {best_score:.6f}")

        T *= cooling

    return best_a, best_b


def _targeted_hill_climbing(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Hill‑climbing that always attacks the current maximal convolution entry.
    The source index is chosen proportionally to its contribution to that
    peak, the destination is taken among the currently small entries.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()
    best_b = b.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.02 + 1e-9
    cooling = 0.9995

    while time.time() < deadline:
        # index of the current maximum
        k_max = int(np.argmax(b))

        # contribution of each position to this maximum
        idx = np.arange(n)
        partner = k_max - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # destination: pick from the “light” entries
        low_mask = a < (a.mean())
        if low_mask.any():
            dest_candidates = np.where(low_mask)[0]
            j = rng.choice(dest_candidates)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        # amount of mass to move
        max_delta = a[i] * 0.5
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ----- exact convolution update (same formula as in _hill_climbing) -----
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a

        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # acceptance (Metropolis)
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-(new_score - cur_score) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                best_b = b.copy()
                print(f"[{time.time():.1f}] targeted hill improvement: {best_score:.6f}")

        T *= cooling

    return best_a, best_b

        # --------------------------------------------------------------
        # Phase B – generic hill climbing (≈25 % of per‑length budget)
        # --------------------------------------------------------------
        if time.time() < deadline:
            b_full = _full_convolution(a)
            hill_deadline = time.time() + per_len_budget * 0.25
            a_hc, b_hc = _hill_climbing(a, b_full, hill_deadline)
            score_hc = b_hc.max()
            if score_hc < best_score:
                best_score = score_hc
                best_seq = a_hc.copy()
                print(f"[{time.time():.1f}] Hill‑climb improvement (n={n}) score={score_hc:.6f}")

        # --------------------------------------------------------------
        # Phase C – targeted hill climbing (remaining budget for this length)
        # --------------------------------------------------------------
        if time.time() < deadline:
            # ensure we work with the latest best of the current length
            if best_seq.size == n:
                a_cur = best_seq.copy()
            else:
                a_cur = a.copy()
            b_cur = _full_convolution(a_cur)
            targ_deadline = time.time() + per_len_budget * 0.25
            a_tg, b_tg = _targeted_hill_climbing(a_cur, b_cur, targ_deadline)
            score_tg = b_tg.max()
            if score_tg < best_score:
                best_score = score_tg
                best_seq = a_tg.copy()
                print(f"[{time.time():.1f}] Targeted hill improvement (n={n}) score={score_tg:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with any remaining budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a_final = best_seq.copy()
        b_final = _full_convolution(a_final)
        a_final, b_final = _targeted_hill_climbing(a_final, b_final, deadline)
        final_score = b_final.max()
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final polishing improvement: score={final_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: strip trailing near‑zeros for neatness
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Official evaluation (objective = max of self‑convolution)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
