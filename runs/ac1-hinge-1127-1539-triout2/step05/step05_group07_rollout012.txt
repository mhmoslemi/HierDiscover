"""
adversarial step‑function optimizer – LP/alternating‑projection variant

The algorithm searches for a non‑negative vector `a` (length n) that
minimises the evaluation function

    score = 2·n·max(a*a) / (sum(a)**2)

which is exactly the peak of the (full) self‑convolution because the
objective is scale‑invariant.

Procedure
---------

* Choose a length n (randomised between a few thousand and ten‑thousand).
* Initialise a feasible vector `a` (uniform, edge‑heavy, Dirichlet or a
  scaled copy of the best sequence seen so far).
* Repeat a few alternating steps:

    1. Compute the current peak `M = max(conv(a,a))`.
    2. Build the (2n‑1) × n convolution matrix `T(a)` such that
       `T(a) @ b = conv(a,b)`.  `T(a)` is very sparse – each column i
       consists of the constant value `a[i]` on a contiguous diagonal.
    3. Solve the linear program

           maximise   Σ b_j
           subject to T(a)·b ≤ M      (all convolution entries ≤ M)
                      b ≥ 0           (non‑negativity)

       The LP has a trivial feasible point (`b = a`) and its optimum
       pushes mass to positions that are “safe’’ w.r.t. the current bound.
    4. Scale the LP solution to the required ℓ₁‑norm
       `S = sqrt(2·n)` (scaling never violates the bound because it only
       shrinks the vector).
    5. Update the iterate by a convex combination

           a ← (1‑t)·a + t·b_scaled

       (`t` starts at 0.5 and is reduced whenever the step does not
        improve the peak).
    6. Occasionally add a tiny Gaussian jitter in the log‑space to
       escape plateaus.

* Keep the best vector (lowest peak) over all restarts.
* Stop when the time budget (default 1000 s) is exhausted and return the
  best sequence found.

If SciPy is not available the code falls back to the previous
gradient‑descent optimiser.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (sparse matrices + LP solver)
# ----------------------------------------------------------------------
try:
    import scipy.sparse as _sp
    import scipy.optimize as _opt
    _SCIPY = True
except Exception:                     # pragma: no cover
    _SCIPY = False

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT (fast, O(n log n))."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation (preserves mass)."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.ceil(pos).astype(int)
    hi = np.clip(hi, 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


def _build_conv_matrix(a: np.ndarray):
    """
    Return a sparse CSR matrix T with shape (2n‑1, n) such that
    T @ b == np.convolve(a, b)   (full linear convolution).
    Column i of T is the constant vector a[i] placed on the diagonal
    with offset i.
    """
    n = a.shape[0]
    m = 2 * n - 1
    # each diagonal i (offset i) has length m‑i, all entries = a[i]
    offsets = np.arange(n, dtype=int)
    diags = [np.full(m - d, a[d], dtype=np.float64) for d in range(n)]
    return _sp.diags(diags, offsets, shape=(m, n), format='csr')


# ----------------------------------------------------------------------
# LP‑based alternating projection optimizer
# ----------------------------------------------------------------------
def _lp_alternating_optimize(a0: np.ndarray,
                             budget_deadline: float,
                             rng: np.random.Generator):
    """
    Perform the alternating LP / convex‑combination steps described in the
    module docstring.  Returns the best vector seen and its peak value.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    a = a0.copy()
    best_a = a.copy()
    best_peak = _full_convolution_fft(a).max()
    t = 0.5                         # initial mixing weight

    it = 0
    while time.time() < budget_deadline:
        it += 1
        conv = _full_convolution_fft(a)
        M = conv.max()

        # ----- build sparse convolution matrix -----
        T = _build_conv_matrix(a)            # (2n‑1, n)

        # ----- linear programme: maximise Σ b_j  -----
        c = -np.ones(n, dtype=np.float64)    # minimise -sum → maximise sum
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        # bounds = (0, ∞) for each variable
        bounds = [(0, None)] * n

        # Give the LP a few seconds at most (safety against pathological runs)
        time_left = max(0.0, budget_deadline - time.time())
        lp_options = {'presolve': True, 'time_limit': time_left}

        res = _opt.linprog(c, A_ub=T, b_ub=b_ub,
                           bounds=bounds, method='highs',
                           options=lp_options)

        if not res.success:                     # pragma: no cover
            break

        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b == 0.0:
            break

        # ----- scale to the required ℓ₁‑norm -----
        b_scaled = b_opt * (S / sum_b)

        # ----- convex combination -----
        a_new = (1.0 - t) * a + t * b_scaled

        # ----- evaluate new peak -----
        new_peak = _full_convolution_fft(a_new).max()

        if new_peak < best_peak - 1e-9:
            best_peak = new_peak
            best_a = a_new.copy()
            a = a_new
            # keep mixing weight relatively large when we improve
            t = min(0.9, t * 1.05)
        else:
            # no improvement → shrink the step size
            t *= 0.5
            if t < 1e-4:
                break

        # occasional tiny jitter (helps escape tiny plateaus)
        if it % 3 == 0:
            jitter = rng.normal(scale=0.001 * S, size=n)
            a = np.clip(a + jitter, 0.0, None)
            a = a / a.sum() * S

    return best_a, best_peak

# Gradient‑descent fallback (used when SciPy is unavailable)
# ----------------------------------------------------------------------
def _grad_descent_fallback(a0: np.ndarray,
                           deadline: float,
                           rng: np.random.Generator):
    # A very compact replica of the previous gradient based optimiser.
    # Kept for environments lacking SciPy.
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)

    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.2
    tau = max(_full_convolution_fft(a0).max(), 1.0)

    best_a = a0.copy()
    best_peak = _full_convolution_fft(a0).max()
    it = 0

    while time.time() < deadline:
        it += 1
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # gradient of τ‑soft‑max approximation of max(b)
        b = _full_convolution_fft(a)
        max_b = b.max()
        if tau < 1e-12:
            mask = (b == max_b).astype(float)
        else:
            diff = b - max_b
            exps = np.exp(diff / tau)
            mask = exps / exps.sum()
        a_rev = a[::-1]
        conv_len = len(mask) + n - 1
        f_len = _next_pow2(conv_len)
        Fmask = np.fft.rfft(mask, n=f_len)
        Fa_rev = np.fft.rfft(a_rev, n=f_len)
        conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
        g = 2.0 * conv[n - 1:n - 1 + n]

        if max_b < best_peak:
            best_peak = max_b
            best_a = a.copy()

        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        tau *= 0.9995
        lr *= 0.9999
        if it % 500 == 0:
            p += rng.normal(scale=0.02, size=n)

    return best_a, best_peak

    # ------------------------------------------------------------------
    def _initial_vector(n):
        S = np.sqrt(2.0 * n)
        shape = rng.choice(['uniform', 'edge', 'dirichlet', 'mix'])
        if shape == 'uniform':
            a = np.full(n, S / n)
        elif shape == 'edge':
            p_exp = rng.uniform(0.5, 2.5)
            idx = np.arange(n)
            d = np.minimum(idx, n - 1 - idx) + 1    # distance to nearest edge (1‑based)
            a = (d.astype(float) ** (-p_exp))
            a = a / a.sum() * S
        elif shape == 'dirichlet':
            # Dirichlet with small concentration → spiky but still non‑negative
            a = rng.gamma(0.5, 1.0, size=n)
            a = a / a.sum() * S
        else:   # mix – start from the globally‑best sequence if it exists
            if base_seq is not None:
                a = _scale_sequence(base_seq, n)
                a = np.clip(a, 0.0, None)
                a = a / a.sum() * S
            else:
                a = np.full(n, S / n)
        return a

    # ------------------------------------------------------------------
    # Main optimisation loop – keep restarting until time runs out
    # ------------------------------------------------------------------
    while time.time() < end_time:
        attempt += 1
        # choose a length – a mix of moderate and larger sizes
        if rng.random() < 0.6:
            n = rng.integers(1800, 5000)   # moderate
        else:
            n = rng.integers(5000, 10000)  # larger

        a_start = _initial_vector(n)

        # devote a slice of the remaining budget to this restart
        slice_deadline = min(time.time() + 14.0, end_time)

        if _SCIPY:
            a_opt, peak = _lp_alternating_optimize(a_start, slice_deadline, rng)
        else:                                   # pragma: no cover
            a_opt, peak = _grad_descent_fallback(a_start, slice_deadline, rng)

        # the score reported by evaluate_sequence is exactly the peak
        score = peak

        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[restart] attempt={attempt} n={n}  new best score={best_score:.6f}")

        # early exit if we have already crossed the target bound
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Final sanity check via the provided black‑box evaluator
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
        external = evaluate_sequence(final_seq)
        print(f"[final] external score={external:.6f} (fallback)")
    else:
        final_seq = best_seq.tolist()
        external = evaluate_sequence(final_seq)
        print(f"[final] external score={external:.6f}  best internal max={best_score:.6f}  length={len(final_seq)}")
        globals()['height_sequence_1'] = final_seq

    return final_seq
