assistantfinal```python
"""
Search for a non‑negative step function that minimizes the maximal entry of its
self‑convolution under the ℓ¹‑normalisation required by ``evaluate_sequence``.
The algorithm proceeds in three stages, respecting a total time budget (default
1000 s):

1. **Random parametric exploration** – many families (edge‑heavy power law,
   edge‑heavy exponential, mixed power‑exponential, beta‑shaped, Gaussian,
   Dirichlet) are sampled.  Each candidate is projected onto the simplex
   Σa_i = √(2 n) and evaluated with a fast FFT‑based max‑convolution.

2. **Adam‑style projected gradient descent** on the smooth log‑sum‑exp
   surrogate of the max‑convolution.  The smoothing parameter β is annealed
   gradually.  After each gradient step the iterate is projected back onto the
   simplex.

3. **Targeted simulated annealing** – mass is repeatedly moved from an index
   that heavily contributes to the current maximal convolution entry to a
   random destination.  Uphill moves are accepted with a slowly decaying
   temperature.  Occasionally a global jitter is applied to escape deep
   minima.

All phases stop when the overall deadline is reached.  The best sequence
found is returned as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x_i ≥ 0, Σx_i = s}."""
    if s <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) of the convolution and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (grad, b) where
        b = a * a   (full convolution)
        grad = ∇_a  ( (1/β)·log Σ_k exp(β·b_k) )
              = 2·(a ⋆ p)   with   p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]

    # --- convolution a*a -------------------------------------------------
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # --- soft‑max probabilities -----------------------------------------
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))          # for numerical stability
    p = exps / np.sum(exps)

    # --- gradient g = 2·(a ⋆ p) -----------------------------------------
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1               # = 3·n‑2
    size2 = 1 << ((conv_len).bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]        # central n entries
    return g, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on the log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def _adam_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Adam optimisation with Euclidean projection onto the ℓ¹‑simplex.
    The smoothing parameter β (log‑sum‑exp) is annealed during the run.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    step = 0
    beta = 1.0               # smoothing parameter
    beta_max = 1e5

    while time.time() < deadline:
        step += 1
        g, _ = _gradient_softmax(a, beta)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)

        # learning rate schedule (per‑coordinate)
        lr = 0.05 / (np.sqrt(v_hat) + eps)

        a = a - lr * m_hat
        # projection onto simplex
        a = _project_to_simplex(a, target_sum)

        # anneal β slowly
        if step % 200 == 0 and beta < beta_max:
            beta = min(beta * 1.07, beta_max)

    return a


# ----------------------------------------------------------------------
# Targeted simulated annealing refinement
# ----------------------------------------------------------------------


def _targeted_simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Move a random fraction of mass from an index that heavily contributes to
    the current maximal convolution entry to a random destination.  Uphill
    moves are accepted with a temperature schedule; occasional jitter helps
    escape local minima.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # initial temperature proportional to the current objective
    T = best_score * 0.02 + 1e-9
    cooling = 0.9997                # slower cooling than the original code
    rng = np.random.default_rng()
    max_transfer = 0.30            # up to 30 % of the selected mass

    while time.time() < deadline:
        # full convolution and index of the current maximum
        b = _full_convolution_fft(a)
        k_max = int(np.argmax(b))

        # indices that actually affect b[k_max]
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idx_range = np.arange(lo, hi + 1)
        contrib = a[idx_range] * a[k_max - idx_range]
        weight_sum = float(contrib.sum())

        if weight_sum > 0:
            probs = contrib / weight_sum
            i = rng.choice(idx_range, p=probs)
        else:
            i = rng.integers(0, n)

        # destination j ≠ i uniformly
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        # transfer a random fraction of the mass at i to j
        delta = a[i] * rng.random() * max_transfer
        if delta <= 0.0:
            continue
        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score

        accept = False
        if cur_score <= best_score:
            accept = True
        elif T > 0.0 and rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA improvement: {best_score:.6f}")
        else:
            # revert move
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, np.sqrt(2.0 * n))
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA jitter improvement: {best_score:.6f}")

    return best


# ----------------------------------------------------------------------
# Random shape families (projected onto the ℓ¹‑simplex)
# ----------------------------------------------------------------------


def _random_shape_sequence(n: int, rng) -> np.ndarray:
    """
    Generate a non‑negative sequence of length n from several parametric
    families and project it onto Σa_i = √(2 n).
    """
    idx = np.arange(n, dtype=float)
    kind = rng.integers(0, 7)

    if kind == 0:                     # edge‑heavy power‑law (symmetric)
        p = rng.uniform(0.05, 2.0)
        a = (idx + 1.0) ** (-p) + (n - idx) ** (-p)

    elif kind == 1:                   # edge‑heavy exponential (symmetric)
        lam = rng.exponential(scale=0.02) + 1e-8
        a = np.exp(-lam * idx) + np.exp(-lam * (n - 1 - idx))

    elif kind == 2:                   # mixed power‑law on each side
        p1 = rng.uniform(0.05, 2.0)
        p2 = rng.uniform(0.05, 2.0)
        w = rng.random()
        a = w * (idx + 1.0) ** (-p1) + (1.0 - w) * (n - idx) ** (-p2)

    elif kind == 3:                   # beta‑shaped (U‑ or bell‑shaped)
        alpha = rng.uniform(0.2, 3.0)
        beta = rng.uniform(0.2, 3.0)
        x = (idx + 0.5) / n
        a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))

    elif kind == 4:                   # Gaussian centred (may be asymmetric)
        mu = rng.integers(0, n)
        sigma = rng.uniform(0.05 * n, 0.30 * n)
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)

    elif kind == 5:                   # Dirichlet (uniformly random simplex)
        a = rng.dirichlet(alpha=np.ones(n))

    else:                             # uniform with tiny jitter
        a = np.full(n, 1.0)
        a += rng.normal(scale=0.01, size=n)

    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, np.sqrt(2.0 * n))
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative sequence) that minimises
    ``evaluate_sequence`` under the supplied time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 0. Initialise from the global construction if it exists
    # ------------------------------------------------------------------
    global_seq = globals().get("height_sequence_1")
    if isinstance(global_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(global_seq, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        sc = _max_convolution_fft(a0)
        best_seq, best_score = a0.copy(), sc
        print(f"[{time.time():.1f}] Start from global height_sequence_1 "
              f"(n={n0})  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Random shape exploration (≈30 % of the budget)
    # ------------------------------------------------------------------
    lengths = [800, 1000, 1200, 1500, 2000, 2500, 3000, 4000, 5000]
    scan_deadline = min(deadline, time.time() + (budget_s * 0.30))

    while time.time() < scan_deadline:
        n = int(rng.choice(lengths))
        a = _random_shape_sequence(n, rng)
        score = _max_convolution_fft(a)

        if score < best_score:
            best_seq, best_score = a.copy(), score
            print(f"[{time.time():.1f}] New best (random scan) n={n}  score={score:.6f}")

        # early exit if we already beat the target
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback to a uniform candidate if nothing was generated
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1500
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = _max_convolution_fft(best_seq)

    # ------------------------------------------------------------------
    # 2. Adam‑based projected gradient descent (≈40 % of remaining budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        md_deadline = time.time() + remaining * 0.45
        a_md = _adam_opt(best_seq, md_deadline)
        md_score = _max_convolution_fft(a_md)
        if md_score < best_score:
            best_seq, best_score = a_md.copy(), md_score
            print(f"[{time.time():.1f}] New best after Adam: score={md_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Targeted simulated annealing (use whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        a_sa = _targeted_simulated_annealing(best_seq, deadline)
        sa_score = _max_convolution_fft(a_sa)
        if sa_score < best_score:
            best_seq, best_score = a_sa.copy(), sa_score
            print(f"[{time.time():.1f}] New best after SA: score={sa_score:.6f}")

    # ------------------------------------------------------------------
    # Clean up tiny trailing entries
    # ------------------------------------------------------------------
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator
    # ------------------------------------------------------------------
    try:
        ev = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence = {ev:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective (max‑conv) = "
          f"{best_score:.6f}   length = {best_seq.size}")

    return best_seq.tolist()
