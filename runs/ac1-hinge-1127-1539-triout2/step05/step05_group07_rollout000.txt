"""
adversarial step‑function optimizer (improved)

This version builds upon the previous gradient‑based search but adds several
enhancements:

* Two‑phase Adam optimisation:
    - Phase 1 uses a τ‑soft‑max (smooth approximation of max) to shape the
      distribution.
    - Phase 2 switches to the exact max‑subgradient (mask = 1 on the active
      convolution peaks) for a high‑precision refinement.
* Longer optimisation windows per restart and a more aggressive cooling
  schedule.
* A simple stochastic hill‑climbing stage after the Adam run that perturbs
  the log‑softmax parameters and accepts any improvement.
* Automatic up‑sampling of the current best sequence to larger lengths
  (interpolation) and re‑optimisation – larger n often yields a lower bound.
* Careful time‑budget handling (≈ 1000 s total) and extensive progress
  logging.

The optimiser respects the constraints of ``evaluate_sequence`` (non‑negative
entries, scaling invariance) by working on the log‑softmax parametrisation,
which guarantees a non‑negative vector whose ℓ₁‑norm is fixed to
√(2 n).  The returned list can be fed directly to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a with itself using real FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation to length new_n."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Gradient helpers
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max of the self‑convolution.
    Returns (g, max_b) where g = ∂/∂a τ‑soft‑max(b) and max_b = max(b).
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        # soft‑max weights (more stable formulation)
        diff = b - max_b               # <= 0
        exps = np.exp(diff / tau)      # in (0,1]
        mask = exps / exps.sum()        # shape (2n‑1,)

    # gradient w.r.t a: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = mask.shape[0] + n - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]
    return g, max_b


def _max_gradient(a: np.ndarray):
    """
    Exact sub‑gradient of max(b) where b = a*a.
    Returns (g, max_b) where g = ∂ max(b) / ∂a.
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = b.max()
    mask = (b == max_b).astype(float)
    # average over ties so that step size does not explode
    nz = mask.sum()
    if nz > 1:
        mask /= nz

    a_rev = a[::-1]
    conv_len = mask.shape[0] + n - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   tau_decay: float = 0.998,
                   lr_decay: float = 0.9995,
                   switch_tau: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that starts with a soft‑max (τ) gradient and switches
    to the exact max‑gradient once τ falls below *switch_tau*.
    Returns the best vector encountered and its hard‑max convolution value.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)               # target ℓ₁‑norm

    # initialise logits from a0 (softmax(p) * S == a0)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft /= soft.sum()
    p = np.log(soft)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.25

    # initial smoothing parameter
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 1.0)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # --------- compute a from logits (log‑softmax) ----------
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # --------- compute gradient ----------
        if tau > switch_tau:
            g, peak = _softmax_gradient(a, tau)
        else:
            g, peak = _max_gradient(a)

        # keep best (hard) peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # --------- gradient w.r.t logits ----------
        mu = np.dot(a, g) / S                # scalar used for simplex projection
        grad_p = a * (g - mu)

        # optional clipping for stability
        clip = np.max(np.abs(grad_p))
        if clip > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip)

        # --------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # --------- schedules ----------
        tau *= tau_decay
        lr *= lr_decay

        # occasional jitter to escape plateaus
        if it % 400 == 0:
            p += rng.normal(scale=0.015, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing in log‑space
# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray,
                deadline: float,
                rng: np.random.Generator,
                init_sigma: float = 0.03,
                sigma_decay: float = 0.999) -> np.ndarray:
    """
    Randomly perturb the logits of *a* and keep any improvement.
    The perturbation size decays over time.
    Returns the improved vector (or the original if no improvement).
    """
    n = a.shape[0]
    S = a.sum()
    # initialise logits from a
    eps = 1e-12
    soft = np.maximum(a / S, eps)
    soft /= soft.sum()
    p = np.log(soft)

    # current best
    best_a = a.copy()
    best_peak = _full_convolution_fft(best_a).max()
    sigma = init_sigma

    while time.time() < deadline:
        # propose new logits
        p_new = p + rng.normal(scale=sigma, size=n)
        exp_p = np.exp(p_new - np.max(p_new))
        soft_new = exp_p / exp_p.sum()
        a_new = soft_new * S

        peak_new = _full_convolution_fft(a_new).max()
        if peak_new < best_peak:
            best_peak = peak_new
            best_a = a_new.copy()
            p = p_new                     # keep the better centre
        sigma *= sigma_decay               # slowly shrink steps

    return best_a

    """

    refinement) and also tries up‑sampling the current best solution
    to larger lengths.

    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    end_time = start_time + budget_s - 5.0   # keep a safety margin

    # ------------------------------------------------------------------
    # Retrieve a previously found construction if available
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    attempt = 0

    # ------------------------------------------------------------------
    # Helper: generate an initial feasible vector for length n
    # ------------------------------------------------------------------
    def _initial_vector(n):
        """Return a feasible a (ℓ₁‑norm = sqrt(2n)) for the given length."""
        S = np.sqrt(2.0 * n)
        shape = rng.choice(['uniform', 'edge', 'random'])
        if shape == 'uniform':
            a = np.full(n, S / n)
        elif shape == 'edge':
            # edge‑heavy: power‑law decay from the edges
            p_exp = rng.uniform(0.8, 2.2)
            idx = np.arange(n)
            d = np.minimum(idx, n - 1 - idx) + 1
            a = d.astype(float) ** (-p_exp)
            a = a / a.sum() * S
        else:   # random Dirichlet‑like
            a = rng.random(n)
            a = a / a.sum() * S

        # occasional warm‑start from the global best (interpolated)
        if base_seq is not None and rng.random() < 0.25:
            a = _scale_sequence(base_seq, n)
            a = np.clip(a, 0.0, None)
            a = a / a.sum() * S
        return a

    # ------------------------------------------------------------------
    # Main optimisation loop (restarts)
    # ------------------------------------------------------------------
    while time.time() < end_time:
        attempt += 1

        # Choose a length: we allocate more attempts to moderate sizes,
        # but every few attempts we try a larger problem.
        if rng.random() < 0.7:
            n = int(rng.integers(1500, 8000))
        else:
            n = int(rng.integers(8000, 18000))

        a_start = _initial_vector(n)

        # allocate a slice of the remaining budget for this restart
        slice_deadline = min(time.time() + 18.0, end_time)

        # ---- Adam optimisation (two‑phase) ----
        a_opt, peak = _adam_optimize(a_start, slice_deadline, rng)

        # ---- Local stochastic refinement (if time permits) ----
        if time.time() < slice_deadline:
            a_opt = _hill_climb(a_opt, slice_deadline, rng)

        # compute internal score (same formula as evaluate_sequence)
        score = 2.0 * n * peak / (np.sum(a_opt) ** 2)

        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[restart] attempt={attempt} n={n}  new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Upsample phase: try to improve the best solution by increasing its
    # length (interpolation) and re‑optimising.
    # ------------------------------------------------------------------
    if best_seq is not None:
        # try at most three up‑sampling iterations
        for up in range(3):
            remaining = end_time - time.time()
            if remaining < 5.0:
                break
            # enlarge length by a factor between 1.3 and 2.0
            factor = rng.uniform(1.3, 2.0)
            new_n = int(min(len(best_seq) * factor, 25000))
            a_up = _scale_sequence(best_seq, new_n)

            # re‑optimise with a fresh time window (≈10 s)
            slice_deadline = min(time.time() + 10.0, end_time)
            a_opt, peak = _adam_optimize(a_up, slice_deadline, rng)
            if time.time() < slice_deadline:

            score = 2.0 * new_n * peak / (np.sum(a_opt) ** 2)
            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"[upsample] round={up+1} n={new_n}  improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the black‑box evaluator
    # ------------------------------------------------------------------
    if best_seq is None:
