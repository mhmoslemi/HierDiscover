   best_score = cur_score

   while time.time() < deadline:
       # compute gradient mask
       # find mask of indices achieving near max
       mask = np.zeros_like(b, dtype=float)
       mask[b >= cur_score - eps_score] = 1.0
       # compute gradient
       g = _gradient_mask(a, mask)
       # source probabilities proportional to g
       g_sum = g.sum()
       if g_sum <= 0:
           # fallback: random source
           i = rng.integers(0, n)
       else:
           probs_src = g / g_sum
           i = rng.choice(n, p=probs_src)
       
       # destination probabilities inversely proportional to g (favor low)
       inv = g.max() - g + 1e-12
       inv_sum = inv.sum()
       probs_dst = inv / inv_sum
       j = rng.choice(n, p=probs_dst)
       
       if i == j:
           continue
       
       # amount delta
       max_delta = a[i] * 0.5
       # For small a[i] skip
       if max_delta <= 0:
           continue
       delta = rng.random() * max_delta
       
       # update convolution quickly
       b_new = b.copy()
       b_new[i:i + n] -= 2.0 * delta * a
       b_new[j:j + n] += 2.0 * delta * a
       k2i = 2*i; k2j = 2*j; kij = i + j
       if k2i < b_new.size:

       if k2j < b_new.size:

"""
Improved step‑function optimizer.

The routine works in stages:

1️⃣  **Initialisation** – take the best known construction (if any) and
    up‑sample it to the current length, otherwise draw a random vector
    on the ℓ¹‑simplex Σa = √(2 n) (power‑law, exponential, Dirichlet …).

2️⃣  **Smooth max‑gradient descent** – minimise a log‑sum‑exp
    approximation of max(a∗a) with projected gradient descent.
    The inverse temperature β is annealed from 0.5 up to 16384,
    momentum and occasional jitter help escaping plateaus.

3️⃣  **Sub‑gradient hill‑climbing** – use the exact convolution
    (O(n) update) and a sub‑gradient derived from the active
    indices of the current maximum.  Mass is transferred from a
    source whose contribution to the max is large to a destination
    whose contribution is small.  A simulated‑annealing schedule
    (temperature proportional to the current max) accepts occasional
    uphill moves.

4️⃣  **Polishing** – another hill‑climb on the best vector found so far.

The algorithm respects a wall‑clock budget (default 1000 s) and returns
the best sequence as a plain Python list.  Larger problem sizes (up to
≈ 20 000 entries) are explored; the extra degrees of freedom usually
lower the maximal self‑convolution and therefore the value returned by
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Draw a random non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # pure power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, then re‑project to simplex."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient (used in GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns g = 2·(a ⋆ p)   and the full convolution b.
    """
    n = a.shape[0]

    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smoothed max.
    Momentum, β‑annealing and occasional jitter are employed.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 16384.0          # a bit higher than in the reference code
    beta_factor = 1.4

    step_base = target_sum * 0.15
    momentum = np.zeros_like(a)
    rho = 0.90

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # jitter every few hundred iterations
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β annealing
        if it % 350 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a


# ----------------------------------------------------------------------
# Sub‑gradient on the active set (used in hill‑climbing)
# ----------------------------------------------------------------------


def _gradient_mask(a: np.ndarray, mask: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of Σ_{k∈K} (a*a)_k where mask[k]=1 for k∈K.
    Returns g = 2·(a ⋆ mask) of length n.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    conv_len = n + mask.size - 1          # = 3n‑2
    size = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Mr_fft = np.fft.rfft(mask, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Mr_fft, n=size)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g


# ----------------------------------------------------------------------
# Mass‑transfer hill climbing – improved version
# ----------------------------------------------------------------------


def _hill_climbing_improved(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Hill‑climbing that moves mass from indices contributing heavily to the
    current maximum of the convolution to indices contributing little.
    Acceptance follows a simulated‑annealing rule.
    Returns the best (a, b) pair observed before the deadline.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    # start with a relatively high temperature
    T = cur_score * 0.25 + 1e-12
    cooling = 0.9995
    eps = 1e-12

    while time.time() < deadline:
        # mask of active positions (within eps of the max)
        mask = np.zeros_like(b, dtype=float)
        mask[b >= cur_score - eps] = 1.0

        # sub‑gradient of the active set
        g = _gradient_mask(a, mask)

        # source ∝ gradient, destination ∝ (max‑gradient)
        g_sum = g.sum()
        if g_sum <= 0:
            i = rng.integers(0, n)
        else:
            probs_src = g / g_sum
            i = rng.choice(n, p=probs_src)

        inv = g.max() - g + 1e-12
        probs_dst = inv / inv.sum()
        j = rng.choice(n, p=probs_dst)

        if i == j:
            continue

        max_delta = a[i] * 0.5
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta

        # O(n) exact update of the convolution
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-(new_score - cur_score) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill‑climb improvement: {best_score:.6f}")

        T *= cooling

    # recompute the final convolution for consistency
    b_final = _full_convolution(best_a)
    return best_a, b_final


# ----------------------------------------------------------------------
# Main search entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the given wall‑clock budget (default 1000 s) and
    returns the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0      # safety margin

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Seed from a previously stored construction if available
    # --------------------------------------------------------------
    earlier = globals().get("height_sequence_1")
    if isinstance(earlier, (list, tuple, np.ndarray)):
        a0 = np.asarray(earlier, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        best_score = _max_convolution_fft(a0)
        best_seq = a0.copy()
        print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) score={best_score:.6f}")

    # --------------------------------------------------------------
    # Length schedule – we explore increasingly larger dimensions
    # --------------------------------------------------------------
    lengths = [2000, 5000, 10000, 20000]

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)

        # ----- initialise -------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)

        a_init = _project_to_simplex(a_init, target_sum)

        # ----- allocate a slice of the remaining budget ------------------
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # allocate at most 180 s to GD and 120 s to hill‑climbing, but
        # respect the proportion of the remaining time.
        gd_time = min(remaining * 0.35, 180.0)
        hc_time = min(remaining * 0.25, 120.0)

        # ----- gradient descent (smooth max) -----------------------------
        a_gd = _gradient_descent_opt(a_init, time.time() + gd_time)
        score_gd = _max_convolution_fft(a_gd)
        if score_gd < best_score:
            best_score = score_gd
            best_seq = a_gd.copy()
            print(f"[{time.time():.1f}] GD improvement (n={n}) score={score_gd:.6f}")

        # ----- hill‑climbing (exact convolution updates) ---------------
        if time.time() + hc_time < deadline:
            b_full = _full_convolution(a_gd)
            a_hc, b_hc = _hill_climbing_improved(a_gd, b_full, time.time() + hc_time)
            score_hc = b_hc.max()
            if score_hc < best_score:
                best_score = score_hc
                best_seq = a_hc.copy()
                print(f"[{time.time():.1f}] HC improvement (n={n}) score={score_hc:.6f}")

    # --------------------------------------------------------------
    # Final polishing with any remaining budget
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        a_final = best_seq.copy()
        b_final = _full_convolution(a_final)
        a_final, b_final = _hill_climbing_improved(a_final, b_final, deadline)
        final_score = b_final.max()
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final polish score={final_score:.6f}")

    # --------------------------------------------------------------
    # Trim trailing near‑zeros for a neat output
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # --------------------------------------------------------------
    # Report the official evaluation (objective = max self‑convolution)
    # --------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
