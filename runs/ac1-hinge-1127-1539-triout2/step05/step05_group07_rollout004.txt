def propose_candidate(...):
    rng = np.random.default_rng(seed)
    end = time.time() + budget_s - 5 ...
    best_seq = None; best_score = np.inf

    # start with initial length
    length_seq = [2000, 5000, 10000, 20000]   # maybe limited to max available
    for n in length_seq:
        # allocate slice of remaining time; allocate proportionally to n  
        remaining = end - time.time()
        if remaining < 10:
            break
        slice_time = min(remaining * 0.2, remaining - 5)  # 20% per n?
        # initial a: if we have previous best, upscale; else generate random
        if best_seq is not None:
            a0 = _scale_sequence(np.asarray(best_seq), n)
            a0 = np.clip(a0, 0, None)
            # re-normalize to sum sqrt(2n)
            a0 = a0 / a0.sum() * np.sqrt(2.0 * n)
        else:
            # generate using edge heavy
            a0 = _initial_vector(n) # define function similar but without base_seq
        # run optimizer
        a_opt, peak = _adam_optimize(a0, time.time() + slice_time, rng)
        score = 2.0 * n * peak / (np.sum(a_opt) ** 2)  # =peak
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"New best after n={n}: score {best_score}")
    # final local refinement on final best
    final_n = len(best_seq)
    # apply random hill climbs for remaining time
    while time.time() < end:
        # propose random move: take two random indices i,j, move delta
        delta = rng.uniform(0, 0.01) * np.sqrt(2.0 * final_n)  # small fraction of total?
        i = rng.integers(final_n)
        j = rng.integers(final_n)
        if best_seq[i] < delta:
            delta = best_seq[i]
        if delta <= 0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # ensure nonneg and sum constant
        # compute score using evaluate_sequence
        ext = evaluate_sequence(new_seq.tolist())
        if ext < best_score:
            best_score = ext
            best_seq = new_seq
            print(f"Improved by hill climb: {best_score}")
    # final
    globals()['height_sequence_1'] = best_seq.tolist()
    return best_seq.tolist()

"""
adversarial step‑function optimizer – multi‑scale Adam + cheap
parameter sweep

The optimizer works in three phases:

1. **Coarse search (small n, many restarts).**  
   For a modest length (≈2500) we generate many random
   initial vectors (edge‑heavy, Dirichlet, uniform) and run a very
   short Adam optimisation (≈5 s each).  The best vector from this
   phase is kept.

2. **Scale‑up & fine‑tune (medium → large n).**  
   The best vector is interpolated to a larger length (≈8000) and
   optimised for a longer period.  The result is again interpolated
   to a still larger length (≈20000) and refined for the remaining
   budget.

3. **Local stochastic refinement.**  
   With the remaining time we perform inexpensive random mass‑transfer
   moves (pick two indices i, j and move a small amount of mass from
   i to j).  If the peak of the self‑convolution improves we keep the
   change.  This often escapes the last plateau of the gradient based
   optimiser.

All optimisation respects the simplex constraint
∑aᵢ = √(2 n) and non‑negativity, because the vector is generated from
log‑softmax logits (the Adam step works on the logits).  The
evaluation function is exactly the peak of the self‑convolution, so we
store the best peak seen and finally return the corresponding sequence.
The best sequence is also saved in the global ``height_sequence_1`` for
future restarts.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution a*a using a real FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.ceil(pos).astype(int)
    hi = np.clip(hi, 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Soft‑max approximation of the ∞‑norm of the convolution
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Returns (grad, max_b) where grad = ∂/∂a  τ‑soft‑max(b)
    and max_b is the hard maximum of b = a*a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        # Exact sub‑gradient: give weight 1 to every active entry
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        w = exps / np.sum(exps)          # soft‑max weights
        mask = w

    # gradient = 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + len(a) - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    grad = 2.0 * conv[len(a) - 1:len(a) - 1 + len(a)]
    return grad, max_b


# ----------------------------------------------------------------------
# Adam optimisation on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Minimise the τ‑soft‑max approximation of the convolution peak.
    Returns the best vector encountered and the corresponding (hard) peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)               # required ℓ₁‑norm

    # initialise logits so that softmax(p)*S == a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                    # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.25

    # initial smoothing parameter
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 1.0)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ---------- map logits → feasible a ----------
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S                     # a lives on the simplex

        # ---------- convolution & gradient ----------
        g, peak = _softmax_gradient(a, tau)

        # keep best (hard) peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ---------- gradient wrt logits ----------
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)            # shape (n,)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---------- schedules ----------
        tau *= 0.9995            # tighten the soft‑max
        lr *= 0.9999             # slowly decay step size

        # occasional Gaussian jitter in log‑space
        if it % 400 == 0:
            p += rng.normal(scale=0.02, size=n)

    return best_a, best_peak

# Simple families for initial vectors – edge‑heavy power law scan
# ----------------------------------------------------------------------
def _edge_power_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a power‑law edge‑heavy vector (exponent tuned by a quick scan)."""
    S = np.sqrt(2.0 * n)
    idx = np.arange(n)
    dist = np.minimum(idx, n - 1 - idx) + 1.0          # distance to nearest edge (1‑based)

    best_score = np.inf
    best_a = None
    # scan a modest grid of exponents
    for p in np.linspace(0.6, 3.0, 48):
        a = dist ** (-p)
        a = a / a.sum() * S
        peak = _full_convolution_fft(a).max()
        if peak < best_score:
            best_score = peak
            best_a = a

    # add a little random jitter to avoid deterministic behaviour
    jitter = rng.normal(scale=0.01 * S, size=n)
    a_jit = best_a + jitter
    a_jit = np.clip(a_jit, 0, None)
    a_jit = a_jit / a_jit.sum() * S

    return a_jit if rng.random() < 0.5 else best_a


def _random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random Dirichlet‑like start."""
    S = np.sqrt(2.0 * n)
    a = rng.random(n)
    a = a / a.sum() * S
    return a


def _initial_vector(n: int, rng: np.random.Generator, base_seq: np.ndarray | None) -> np.ndarray:
    """Mix a fresh random start with the current best (if any)."""
    shape = rng.choice(['edge', 'random'])
    if shape == 'edge':
        a = _edge_power_initial(n, rng)
    else:
        a = _random_initial(n, rng)

    # optionally blend with a scaled copy of the current best sequence
    if base_seq is not None and rng.random() < 0.25:
        a_scaled = _scale_sequence(base_seq, n)
        a_scaled = np.clip(a_scaled, 0, None)
        a = 0.6 * a + 0.4 * a_scaled
        a = a / a.sum() * np.sqrt(2.0 * n)

    return a

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine runs a time‑limited multi‑scale optimisation:
    * many cheap restarts on a small size (≈2500)
    * progressive up‑scaling to medium (≈8000) and large (≈20000) sizes
    * a final stochastic refinement phase.
    The best candidate found is stored in ``height_sequence_1`` and
    returned as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a few seconds for the harness’s safety margin
    end_time = start_time + budget_s - 5.0

    # keep best sequence (as a numpy array) and its peak value
    best_seq: np.ndarray | None = None
    best_peak = np.inf

    # ------------------------------------------------------------------
    # Phase 1 – many quick restarts on a modest size
    # ------------------------------------------------------------------
    n_small = 2500
    slice_len = 5.0                     # seconds per tiny optimisation
    while time.time() + slice_len < end_time:
        a0 = _initial_vector(n_small, rng,
                             base_seq=None if best_seq is None else best_seq)
        a_opt, peak = _adam_optimize(a0,
                                     time.time() + slice_len,
                                     rng)
        if peak < best_peak:
            best_peak = peak
            best_seq = a_opt.copy()
            print(f"[phase‑1] n={n_small} new best peak={best_peak:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – progressive upscale & longer Adam runs
    # ------------------------------------------------------------------
    upscale_stages = [8000, 20000]   # medium → large
    for n_target in upscale_stages:
        if best_seq is None:
            break
        # make a feasible start for the new length
        a_start = _scale_sequence(best_seq, n_target)
        a_start = np.clip(a_start, 0, None)
        a_start = a_start / a_start.sum() * np.sqrt(2.0 * n_target)

        # allocate a generous slice (≈20‑30 % of remaining time)
        remaining = end_time - time.time()
        if remaining < 20.0:
            break
        slice_time = max(30.0, remaining * 0.25)
        slice_time = min(slice_time, remaining - 5.0)

        a_opt, peak = _adam_optimize(a_start,
                                     time.time() + slice_time,
                                     rng)

        if peak < best_peak:
            best_peak = peak
            best_seq = a_opt.copy()
            print(f"[phase‑2] n={n_target} new best peak={best_peak:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – cheap random mass‑transfer refinement (hill climbing)
    # ------------------------------------------------------------------
    if best_seq is not None:
        n = len(best_seq)
        # Pre‑compute the current peak once
        cur_peak = best_peak
        while time.time() < end_time:
            # propose a tiny move: transfer a random fraction of mass
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # limit transfer to at most 5 % of the donor’s mass
            max_transfer = 0.05 * best_seq[i]
            if max_transfer <= 0.0:
                continue
            delta = rng.random() * max_transfer
            new_seq = best_seq.copy()
            new_seq[i] -= delta
            new_seq[j] += delta

            # recompute the peak (fast FFT)
            new_peak = _full_convolution_fft(new_seq).max()
            if new_peak < cur_peak:
                cur_peak = new_peak
                best_seq = new_seq
                best_peak = new_peak
                print(f"[refine] improved peak={best_peak:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check using the black‑box evaluator
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
        external = evaluate_sequence(final_seq)
        print(f"[final] fallback external score={external:.6f}")
    else:
        final_seq = best_seq.tolist()
        external = evaluate_sequence(final_seq)
        print(f"[final] external score={external:.6f}  internal peak={best_peak:.6f}  length={len(final_seq)}")

    # expose the result for possible future restarts
    globals()['height_sequence_1'] = final_seq
    return final_seq
