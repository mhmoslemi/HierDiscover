def propose_candidate(seed=42, budget_s=1000):
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.

    best_seq = None
    best_score = np.inf

    # define candidate lengths to search from existing best if any
    lengths = list(range(1000, 25001, 2000))   # 1000,3000,...? maybe variable
    # Add length from existing best if any
    if base_seq exists: lengths.append(len(base_seq))

    # Parameter sweep for edge shape
    for n in lengths:
        if time.time() >= end_time: break
        S = np.sqrt(2*n)
        # precompute distances
        idx = np.arange(n)
        d = np.minimum(idx, n-1-idx) + 1  # integer distances
        # sample p values
        # coarse grid
        for p in np.linspace(0.1, 5.0, 30):
            a = d.astype(float) ** (-p)
            a_sum = a.sum()
            a = a * (S / a_sum)
            # compute convolution using FFT
            conv = np.fft.irfft(np.fft.rfft(a, n=_next_pow2(2*n-1)) * np.fft.rfft(a, n=_next_pow2(2*n-1)))[:2*n-1]
            peak = conv.max()
            score = 2.0 * n * peak / (S**2)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[edge sweep] n={n} p={p:.3f} score={score:.6f}")
        # exponential shape
        for beta in np.linspace(0.001, 2.0, 30):
            a = np.exp(-beta * (np.minimum(idx, n-1-idx)))
            a = a * (S / a.sum())
            conv = np.fft.irfft(np.fft.rfft(a, n=_next_pow2(2*n-1)) * np.fft.rfft(a, n=_next_pow2(2*n-1)))[:2*n-1]
            peak = conv.max()
            score = 2.0 * n * peak / (S**2)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[exp sweep] n={n} beta={beta:.3f} score={score:.6f}")

    # After coarse sweep, we could refine near best parameters using random perturbations. Or directly apply gradient descent.

    # Gradient refine using Adam optimizer from earlier code but with tuned hyperparameters
    if best_seq is not None:
        # restructure to run a few rounds of Adam starting from best_seq but allow larger n if we want.
        # We'll run maybe 5 restarts each with time slice ~ remaining time / 5
        remaining = end_time - time.time()
        if remaining > 10:
            n_best = len(best_seq)
            slice_time = remaining / 5.0
            for i in range(5):
                # a copy of best_seq maybe scaled slightly with jitter
                a_start = best_seq.copy()
                # add small random jitter (in log space) to avoid being stuck
                jitter = rng.normal(scale=0.02, size=n_best)
                # Apply jitter to logits:
                # convert to logits via softmax: p = log(a_i) - log(sum) difference
                # Use direct: a_j = a_start * exp(jitter) and re-normalize sum to S.
                a_start = a_start * np.exp(jitter)
                a_start = a_start * (np.sqrt(2.0 * n_best) / a_start.sum())
                # run adam
                a_opt, peak = _adam_optimize(a_start, time.time() + slice_time, rng)
                score = 2.0 * n_best * peak / (np.sum(a_opt) ** 2)
                if score < best_score:
                    best_score = score
                    best_seq = a_opt
                    print(f"[adam refine] iter={i} score={score:.6f}")

    # final check using evaluate_sequence
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external {external:.6f} internal best {best_score:.6f} length {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq

"""
Adversarial step‑function optimizer (improved).

The strategy consists of two stages:

1. **Global parametric sweep** – try a few families of analytic shapes
   (edge‑heavy power law, edge‑heavy exponential and simple sin‑shaped
   bells) for many lengths `n`.  Each shape is scaled to the required
   ℓ₁‑norm `sqrt(2n)` and its self‑convolution peak is evaluated via
   an FFT.  The best candidate found by this cheap sweep is kept.

2. **Local refinement** – starting from the best candidate, run the
   Adam‑based soft‑max‑gradient descent (the method used previously)
   for the remaining time budget.  The Adam optimizer works on the
   simplex via a log‑softmax parametrisation and gradually tightens
   the soft‑max smoothing parameter `tau`.

Both stages respect the 1000 s time limit; progress is printed to
standard output.  The final sequence is stored in the global variable
`height_sequence_1` for reuse by subsequent runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of a τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (g, max_b) where g = ∂/∂a τ‑soft‑max(b) and max_b is the
    (hard) maximum of b.
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)      # soft‑max weights, length 2n‑1

    # gradient w.r.t. a: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + n - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]
    return g, max_b


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the soft‑max approximation of the convolution peak.
    Returns (best_a, best_peak) where ``best_peak`` is the *hard* maximum
    of the self‑convolution of ``best_a``.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    # initialise logits so that softmax(p) * S == a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.2

    # initial smoothing parameter
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 1.0)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # --- a from logits (simple soft‑max) ---
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # --- convolution and gradient ---
        g, peak = _softmax_gradient(a, tau)

        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # --- gradient w.r.t. logits (mirror of softmax Jacobian) ---
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # optional gradient clipping (helps stability for large n)
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # --- Adam update ---
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # --- schedules ---
        tau *= 0.9995               # tighten soft‑max slowly
        lr *= 0.9999                # decay step size

        # occasional jitter in log‑space to escape plateaus
        if it % 500 == 0:
            p += rng.normal(scale=0.02, size=n)

    return best_a, best_peak


def _score_of(a: np.ndarray) -> float:
    """
    Compute the evaluation score (the same quantity that
    ``evaluate_sequence`` returns after proper scaling).
    The vector ``a`` is assumed to be already scaled to the
    required ℓ₁‑norm sqrt(2·len(a)).
    """
    n = a.shape[0]
    S2 = (np.sqrt(2.0 * n)) ** 2          # = 2·n
    max_b = _full_convolution_fft(a).max()
    return 2.0 * n * max_b / S2           # simplifies to max_b


def _make_shape_edge(n: int, p: float, S: float) -> np.ndarray:
    """Edge‑heavy power‑law shape."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1
    raw = d.astype(float) ** (-p)
    return raw * (S / raw.sum())


def _make_shape_exp(n: int, beta: float, S: float) -> np.ndarray:
    """Edge‑heavy exponential shape."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx)
    raw = np.exp(-beta * d)
    return raw * (S / raw.sum())


def _make_shape_sin(n: int, S: float) -> np.ndarray:
    """Half‑sine bell (zero at the ends, max in the middle)."""
    i = np.arange(n) + 1
    raw = np.sin(np.pi * i / (n + 1.0))
    return raw * (S / raw.sum())


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine performs a fast parametric sweep followed by a longer
    Adam‑based local optimisation.  The best sequence found is stored
    globally as ``height_sequence_1`` and also returned.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # Retrieve a previously obtained construction if present
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Helper: evaluate a candidate and possibly update global best
    # ------------------------------------------------------------------
    def _consider(candidate: np.ndarray, source_desc: str):
        nonlocal best_seq, best_score
        # candidate is already scaled to the correct ℓ₁‑norm
        score = _score_of(candidate)
        if score < best_score:
            best_score = score
            best_seq = candidate.copy()
            print(f"[improve] {source_desc} score={score:.6f}  n={len(candidate)}")

    # ------------------------------------------------------------------
    # Add existing sequence (if any) as a starting point
    # ------------------------------------------------------------------
    if base_seq is not None:
        n0 = len(base_seq)
        # re‑scale to the required norm (just in case)
        S0 = np.sqrt(2.0 * n0)
        a0 = base_seq * (S0 / base_seq.sum())
        _consider(a0, "previous best")

    # ------------------------------------------------------------------
    # Global parametric sweep
    # ------------------------------------------------------------------
    # candidate lengths – a modest set covering small to fairly large n
    candidate_ns = [1500, 2500, 3500, 5000, 8000, 12000, 18000, 25000]
    if base_seq is not None:
        candidate_ns.append(len(base_seq))

    # parameter grids
    p_vals = np.linspace(0.2, 5.0, 28)          # power‑law exponent
    beta_vals = np.linspace(0.001, 3.0, 28)     # exponential decay
    alpha_vals = [0.0, 0.25, 0.5, 0.75]          # mixing with uniform

    for n in candidate_ns:
        if time.time() >= end_time:
            break
        S = np.sqrt(2.0 * n)          # required ℓ₁‑norm
        # uniform vector for mixing
        uniform_raw = np.ones(n)

        # ----- edge‑power shape -----
        for p in p_vals:
            raw = np.minimum(np.arange(n), n - 1 - np.arange(n)) + 1
            raw = raw.astype(float) ** (-p)
            # try a few mixes with uniform
            for alpha in alpha_vals:
                mixed_raw = alpha * uniform_raw + (1.0 - alpha) * raw
                a = mixed_raw * (S / mixed_raw.sum())
                _consider(a, f"edge p={p:.3f} α={alpha:.2f}")

        # ----- edge‑exponential shape -----
        for beta in beta_vals:
            d = np.minimum(np.arange(n), n - 1 - np.arange(n))
            raw = np.exp(-beta * d)
            for alpha in alpha_vals:
                mixed_raw = alpha * uniform_raw + (1.0 - alpha) * raw
                a = mixed_raw * (S / mixed_raw.sum())
                _consider(a, f"exp β={beta:.3f} α={alpha:.2f}")

        # ----- sin‑bell shape (no mixing needed) -----
        sine_raw = np.sin(np.pi * (np.arange(n) + 1) / (n + 1.0))
        a = sine_raw * (S / sine_raw.sum())
        _consider(a, "sin bell")

        # ----- random Dirichlet seeds (few) -----
        for _ in range(3):
            raw = rng.random(n)
            a = raw * (S / raw.sum())
            _consider(a, "random dirichlet")

    # ------------------------------------------------------------------
    # Local refinement with Adam optimiser (remaining time budget)
    # ------------------------------------------------------------------
    remaining = end_time - time.time()
    if best_seq is not None and remaining > 15.0:
        n_best = len(best_seq)
        # split the remaining time into a few refinement runs
        runs = max(1, int(remaining // 30.0))
        slice_time = remaining / runs
        for run in range(runs):
            if time.time() >= end_time:
                break
            # start from the current best, add a little jitter in log‑space
            a_start = best_seq.copy()
            jitter = rng.normal(scale=0.03, size=n_best)
            # apply jitter to the logits (log‑softmax parametrisation)
            # convert a_start to logits:
            maxp = np.max(a_start)
            logit = np.log(np.maximum(a_start, 1e-12))
            logit += jitter
            # bring back to feasible simplex
            exp_logit = np.exp(logit - np.max(logit))
            a_start = exp_logit / exp_logit.sum() * np.sqrt(2.0 * n_best)

            a_opt, peak = _adam_optimize(a_start,
                                         time.time() + slice_time,
                                         rng)
            score = 2.0 * n_best * peak / ((np.sqrt(2.0 * n_best)) ** 2)
            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"[adam refine] run={run+1}/{runs}  new best score={score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the external black‑box evaluator
    # ------------------------------------------------------------------
    if best_seq is None:                     # should never happen
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()
        external = evaluate_sequence(final_seq)
        print(f"[final] external score={external:.6f}  internal best={best_score:.6f}  length={len(final_seq)}")
        globals()['height_sequence_1'] = final_seq

    return final_seq
