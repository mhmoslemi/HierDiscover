def _max_convolution_subgradient(a):
    n = len(a)
    b = _full_convolution_fft(a)  # returns array length 2n-1
    M = b.max()
    # mask of indices where near M (tolerance maybe 1e-12 relative)
    tol = 1e-12
    mask = (b >= M - tol).astype(float)
    # Normalize mask to sum=1 for convex combination
    mask_sum = mask.sum()
    if mask_sum == 0:
        mask = np.ones_like(b) / len(b)
    else:
        mask /= mask_sum
    # Compute gradient via convolution: g_i = 2 * sum_{k} mask_k * a_{k-i}
    # This equals 2 * conv_rev(a, mask)
    # Where conv_rev: convolve a_rev (reverse) with mask.
    a_rev = a[::-1]
    conv_len = n + len(mask) - 1 # = n + (2n-1) - 1 = 3n-2
    size = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size)
    M_fft = np.fft.rfft(mask, n=size)
    conv = np.fft.irfft(Ar_fft * M_fft, n=size)[:conv_len]
    g = 2.0 * conv[n-1:n-1+n]
    return g, M

"""
Propose a non‑negative step function that (approximately) minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works in three stages:

1. **Broad parametric sweep** – generate many deterministic families
   (power‑law, exponential, combined) for several lengths, keep the
   best few candidates (elite).

2. **Adam‑based smooth‑max descent** – each elite candidate is refined
   by minimizing a log‑sum‑exp surrogate of the maximal convolution
   entry.  The surrogate temperature β is annealed up to 1e9, and the
   iterate is repeatedly projected onto the ℓ¹‑simplex
   Σa = √(2·n).

3. **Long simulated‑annealing (SA) phase** – the refined vector is
   further tweaked by random mass‑transfer moves with a slow cooling
   schedule.  The best vector found during SA (and the whole run) is
   returned.

All steps respect a user‑supplied time budget (default 1000 s).  No
file or network I/O is performed; only progress is printed via `print`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """
    Return the maximal entry of the linear convolution a*a (full length 2·n‑1).
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Return the full linear convolution a*a (length 2·n‑1).
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex Σx = s , x ≥ 0  (Duchi et al. 2008)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Project vector v onto the simplex {x ≥ 0, Σx = s}.
    """
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) surrogate of max(b) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    For a given vector a compute
        b = a*a                (full convolution)
        g = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
    Returns tuple (g, b).  g = 2·(a ⋆ p) where p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # ---- b = a*a ----------------------------------------------------
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # ---- soft‑max probabilities p ------------------------------------
    max_b = np.max(b)
    # shift to avoid overflow
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)

    # ---- gradient g = 2·(a ⋆ p) ------------------------------------
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    ar_fft = np.fft.rfft(a_rev, n=size2)
    pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(ar_fft * pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on the smooth max
# ----------------------------------------------------------------------


def _adam_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Minimise the log‑sum‑exp surrogate of max(conv(a,a)) using Adam.
    The iterate is kept on the ℓ¹‑simplex Σa = √(2·n).
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta = 1.0                 # temperature for the surrogate
    max_beta = 1e9
    t = 1

    # a small RNG for occasional jitter
    rng = np.random.default_rng()

    while time.time() < deadline:
        # gradient of the smooth max
        g, _ = _gradient_softmax(a, beta)

        # Adam update (we move opposite to the gradient)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)

        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = 0.3 / (np.sqrt(v_hat) + eps)   # global scaling factor 0.3 works well
        a = a - step * m_hat

        # projection onto the simplex
        a = _project_to_simplex(a, target_sum)

        # anneal the temperature
        if beta < max_beta:
            beta = min(beta * 1.02, max_beta)

        t += 1

        # occasional tiny random jitter to avoid stagnation
        if t % 500 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a = a + jitter
            a = _project_to_simplex(a, target_sum)

    return a


# ----------------------------------------------------------------------
# Simple simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Random mass‑transfer SA with multiplicative cooling.
    Keeps the best vector seen during the run.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # initialise temperature from current objective
    T = best_score * 0.02 + 1e-8
    cooling = 0.9999    # slower cooling for a longer search
    rng = np.random.default_rng()

    while time.time() < deadline:
        # pick two distinct positions
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # transfer a random fraction (up to 15 %) of mass from i to j
        frac = rng.random() * 0.15
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        accept = False
        if cur_score <= best_score:
            accept = True
        elif T > 0.0 and rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # revert move
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # very occasional global jitter (helps escape deep local minima)
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()

    return best


# ----------------------------------------------------------------------
# Helper to evaluate a candidate (max‑convolution, because sum is normalised)
# ----------------------------------------------------------------------


def _objective_maxconv(a: np.ndarray) -> float:
    """
    Return max(conv(a,a)).  The input is assumed to be already projected onto
    the ℓ¹‑simplex Σa = √(2·n) – under this normalisation the true
    evaluate_sequence objective equals this value.
    """
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Candidate generators (parametric families)
# ----------------------------------------------------------------------


def _powerlaw_family(n: int, p: float, shift: float) -> np.ndarray:
    raw = (np.arange(n, dtype=float) + shift) ** (-p)
    target = np.sqrt(2.0 * n)
    a = raw * target / raw.sum()
    return np.maximum(a, 0.0)


def _symmetric_powerlaw_family(n: int, p: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    left = (idx + shift) ** (-p)
    right = (n - 1 - idx + shift) ** (-p)
    raw = left + right
    target = np.sqrt(2.0 * n)
    a = raw * target / raw.sum()
    return np.maximum(a, 0.0)


def _exponential_family(n: int, lam: float) -> np.ndarray:
    raw = np.exp(-lam * np.arange(n, dtype=float))
    target = np.sqrt(2.0 * n)
    a = raw * target / raw.sum()
    return np.maximum(a, 0.0)


def _combined_family(n: int, p: float, lam: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = ((idx + shift) ** (-p)) * np.exp(-lam * idx)
    target = np.sqrt(2.0 * n)
    a = raw * target / raw.sum()
    return np.maximum(a, 0.0)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects the supplied time budget,
    explores several parametric families, refines the elite candidates
    with Adam‑based smooth‑max descent and a final simulated‑annealing
    phase, and returns the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_obj = np.inf
    elite = []        # list of (obj, vector)

    # ------------------------------------------------------------------
    # 1. Broad deterministic sweep
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 4000, 5000, 6000, 8000]
    # grids (coarse – we want many different shapes quickly)
    p_grid = np.linspace(0.45, 0.78, 9)            # power‑law exponent
    shift_grid = np.linspace(0.5, 5.0, 9)          # shift for power‑law
    lam_grid = np.geomspace(5e-4, 5e-2, 9)        # exponential rate

    def consider(vec: np.ndarray):
        nonlocal best_seq, best_obj, elite
        n = vec.size
        target = np.sqrt(2.0 * n)
        vec = _project_to_simplex(vec, target)
        obj = _objective_maxconv(vec)
        if obj < best_obj:
            best_obj = obj
            best_seq = vec.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL BEST length={n} obj={obj:.6f}")
        # keep top‑10 elite
        if len(elite) < 10 or obj < elite[-1][0]:
            elite.append((obj, vec))
            elite.sort(key=lambda x: x[0])
            elite = elite[:10]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Exploring length {n}")

        # power‑law + symmetric
        for p in p_grid:
            for shift in shift_grid:
                a = _powerlaw_family(n, p, shift)
                consider(a)
                a_sym = _symmetric_powerlaw_family(n, p, shift)
                consider(a_sym)

        # exponential family
        for lam in lam_grid:
            a = _exponential_family(n, lam)
            consider(a)

        # combined power‑law * exponential
        for p in p_grid:
            for lam in lam_grid:
                a = _combined_family(n, p, lam, shift=0.5)
                consider(a)

        # a few random Dirichlet draws for diversity
        for _ in range(5):
            a = rng.random(n)
            a = a * np.sqrt(2.0 * n) / a.sum()
            consider(a)

    # ------------------------------------------------------------------
    # 2. Refinement of elite candidates (Adam + SA)
    # ------------------------------------------------------------------
    for _, seed_vec in elite:
        if time.time() >= deadline:
            break
        n = seed_vec.size
        sub_dead = min(deadline, time.time() + 30.0)   # ≤30 s per candidate
        # Adam‑style smooth‑max descent
        refined = _adam_descent(seed_vec, sub_dead)
        # SA phase (uses the global deadline)
        refined = _simulated_annealing(refined, deadline)
        obj = _objective_maxconv(refined)
        if obj < best_obj:
            best_obj = obj
            best_seq = refined.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL BEST (refined) n={n} obj={obj:.6f}")

    # ------------------------------------------------------------------
    # 3. Final SA on the current best (use any remaining time)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        best_seq = _simulated_annealing(best_seq, deadline)
        best_obj = _objective_maxconv(best_seq)

    # ------------------------------------------------------------------
    # 4. Cleanup and final evaluation
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)

    # trim trailing near‑zero entries for neatness
    best_seq = np.array(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # final evaluation (the harness will also call evaluate_sequence)
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {val:.6f}")
    except Exception as exc:               # noqa: BLE001
        print(f"Evaluation error: {exc}")

    print(f"[{time.time():.1f}] Finished. Best objective = {best_obj:.6f} "
          f"length = {best_seq.size}")

    return best_seq.tolist()
