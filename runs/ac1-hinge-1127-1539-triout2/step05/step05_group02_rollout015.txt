        max_frac_end = 0.005

            w = a[supp] * a[k_max - supp]

            if weight is not None:
                donor = rng.choice(supp, p=weight)
            else:
                donor = rng.integers(n)
            # choose receiver outside support with low a (or random)
            non_sup = np.setdiff1d(np.arange(n), supp, assume_unique=True)
            if non_sup.size>0:
                receiver = rng.choice(non_sup)
            else:
                receiver = rng.integers(n)
            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                a[donor] -= delta
                a[receiver] += delta
                move_type = "transfer"
        elif rng.random() < 0.9:
            # smoothing
            start_idx = rng.integers(0, n)
            length = rng.integers(2, 30+1)
            end_idx = min(start_idx+length, n)

                a[start_idx:end_idx] = old.mean()
                move_type = "smooth"

        else:

            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # evaluate
        new_score = _score(a)
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob
        if accept:

"""
Propose a low‑score step‑function sequence.

The routine is a multi‑scale optimiser that combines three
ingredients:

1. **Coarse Adam optimisation** on a smooth soft‑max
   approximation of the ‖f*f‖∞ term.  The temperature β is
   increased gradually from 1 to 1024, giving increasingly
   sharper gradients.

2. **Sub‑gradient descent on the true L∞ objective**.
   The gradient of the maximal convolution entry is
   2·a_{k‑i} for all indices i that participate in the current
   maximum k.  A simple diminishing step‑size is used.

3. **Targeted simulated‑annealing refinement**.
   Moves transfer mass from indices that contribute heavily to the
   current maximum toward low‑contributing indices, perform short‑
   window smoothing, and occasional random swaps.  Temperature and
   move‑size decay exponentially.

The optimiser is run on several problem sizes (1500, 3000, 5000)
with independent random restarts.  The best sequence found
within the allotted time budget is returned (trailing zeros are
removed).  All sequences are projected onto the ℓ¹‑simplex
Σa = √(2·n) and kept non‑negative.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s} (Duchi et al., 2008)."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of power‑of‑two length."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a (full convolution) computed via FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Exact benchmark score after normalising the sum of a."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:                 # safety (should never happen)
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Soft‑max gradient of the convolution maximum (used by Adam)
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(a*a).

    Returns (gradient, full_convolution) where the convolution
    is the vector b = a * a (linear convolution).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    ar_fft = np.fft.rfft(a_rev, n=size)
    pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(ar_fft * pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam optimiser on the smooth surrogate
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray, deadline: float, target_sum: float):
    """Adam‑style optimisation of the soft‑max surrogate."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    beta = 1.0
    beta_max = 1024.0
    best_a = a.copy()
    best_val = _score(a)

    # tuning constants
    lr0 = 0.05          # base learning rate
    eps1 = 0.9
    eps2 = 0.999

    while time.time() < deadline:
        grad, _ = _softmax_gradient(a, beta)
        t += 1
        m = eps1 * m + (1 - eps1) * grad
        v = eps2 * v + (1 - eps2) * (grad ** 2)

        # bias‑corrected step‑size
        lr = lr0 * np.sqrt(1 - eps2 ** t) / (np.sqrt(1 - eps1 ** t) + 1e-12)

        a -= lr * m / (np.sqrt(v) + 1e-8)
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaux
        if t % 500 == 0:
            jitter = np.random.default_rng().normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] Adam improve β={beta:.1f} score={best_val:.6f}")

        # raise temperature slowly
        if t % 1000 == 0 and beta < beta_max:
            beta = min(beta * 1.5, beta_max)

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true L∞ objective
# ----------------------------------------------------------------------


def _max_subgradient_descent(a0: np.ndarray,
                             deadline: float,
                             target_sum: float):
    """
    Sub‑gradient descent on the exact max‑convolution term.
    Gradient for the maximal entry k is 2·a_{k‑i} for each i that
    participates in the maximum.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _score(a)

    iter_cnt = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(np.argmax(conv))

        n = a.size
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # sub‑gradient (one of the possible sub‑gradients)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        # diminishing step size proportional to the target sum
        step = 0.05 * target_sum / (np.linalg.norm(g) + 1e-12)
        a -= step * g
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] Sub‑grad improve score={best_val:.6f}")

        iter_cnt += 1
        # small jitter every few hundred iterations
        if iter_cnt % 500 == 0:
            jitter = np.random.default_rng().normal(
                scale=target_sum * 0.0005, size=a.shape
            )
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (targeted moves)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         start_temp: float = 0.5,
                         end_temp: float = 1e-5,
                         max_frac_start: float = 0.40,
                         max_frac_end: float = 0.005,
                         max_window: int = 30) -> tuple[np.ndarray, float]:
    """
    Refinement via simulated annealing.
    Moves:
      • mass transfer from highly contributing indices to low‑contributing ones,
      • short‑window averaging,
      • random pair swap.
    """
    a = a0.copy()
    n = a.size
    cur_score = _score(a)

    best_a = a.copy()
    best_score = cur_score

    start = time.time()
    total = max(deadline - start, 1e-6)
    rng = np.random.default_rng()
    size_fft = 1 << ((2 * n - 1).bit_length())

    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / total
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # --------------------------------------------------------------
        # Choose a move
        # --------------------------------------------------------------
        move_type = None
        conv = _conv_fft(a)                     # current convolution
        k_max = int(np.argmax(conv))

        # compute contribution weights of each index to the current maximum
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        supp = np.arange(lo, hi + 1, dtype=np.int64)
        contrib = a[supp] * a[k_max - supp]     # product a_i * a_{k‑i}
        total_contrib = contrib.sum()

        if rng.random() < 0.70:                      # mass transfer
            if total_contrib > 0:
                donor = rng.choice(supp, p=contrib / total_contrib)
            else:
                donor = rng.integers(n)

            # choose receiver from indices with small contribution (or random)
            low_thr = np.percentile(contrib, 10) if contrib.size else 0.0
            low_idx = supp[contrib <= low_thr]
            if low_idx.size:
                receiver = rng.choice(low_idx)
            else:
                receiver = rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                a[donor] -= delta
                a[receiver] += delta
                move_type = "transfer"

        elif rng.random() < 0.90:                    # short‑window smoothing
            start_idx = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end_idx = min(start_idx + length, n)
            if end_idx - start_idx >= 2:
                saved = a[start_idx:end_idx].copy()
                a[start_idx:end_idx] = saved.mean()
                move_type = "smooth"

        else:                                         # random swap
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # --------------------------------------------------------------
        # Evaluate move
        # --------------------------------------------------------------
        new_score = _score(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA new best={best_score:.6f}")
        else:
            # revert
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[start_idx:end_idx] = saved
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter to escape deep valleys
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Full optimisation for a single problem size
# ----------------------------------------------------------------------


def _run_one_instance(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Optimises a sequence of length n.
    Returns the best sequence and its exact score.
    """
    target = np.sqrt(2.0 * n)

    # --- initial guess -------------------------------------------------
    rng = np.random.default_rng()
    if "height_sequence_1" in globals():
        # use the supplied global as warm‑start
        base = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if base.size > n:
            a0 = base[:n]
        else:
            a0 = np.pad(base, (0, n - base.size), mode="constant")
        a0 = _project_to_simplex(a0, target)
    else:
        a0 = rng.random(n)
        a0 = _project_to_simplex(a0, target)

    best_seq = a0.copy()
    best_val = _score(best_seq)

    # --- stage 1: Adam on soft‑max ------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        sub_dead = time.time() + min(30.0, remaining * 0.4)
        seq, val = _adam_optimize(best_seq, sub_dead, target)
        if val < best_val:
            best_val, best_seq = val, seq.copy()

    # --- stage 2: sub‑gradient descent ---------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        sub_dead = time.time() + min(20.0, remaining * 0.3)
        seq, val = _max_subgradient_descent(best_seq, sub_dead, target)
        if val < best_val:
            best_val, best_seq = val, seq.copy()

    # --- stage 3: simulated annealing ---------------------------------
    if time.time() < deadline:
        seq, val = _simulated_annealing(best_seq, deadline, target)
        if val < best_val:
            best_val, best_seq = val, seq.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine runs several independent optimisations on different
    lengths (1500, 3000, 5000) within the allotted time budget and
    returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 1.0          # a safety margin

    best_seq = None
    best_score = float("inf")

    # problem sizes to explore (increase flexibility vs. time)
    lengths = [1500, 3000, 5000]

    for n in lengths:
        if time.time() >= deadline:
            break
        # allocate a fair share of the remaining time for this size
        now = time.time()
        remaining = deadline - now
        # give a bit more time to the larger instances
        sub_deadline = now + remaining * (1.0 / (len(lengths) - lengths.index(n)))
        print(f"[{time.time():.1f}] Optimising length {n} (time budget {remaining:.1f}s)")
        seq, score = _run_one_instance(n, sub_deadline)
        if seq is not None and score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] New global best: length={n} score={best_score:.6f}")

    # safety fallback
    if best_seq is None:
        best_seq = np.array([np.sqrt(2.0)], dtype=float)

    # strip trailing near‑zeros for neatness
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    print(f"[{time.time():.1f}] FINAL best score={best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
