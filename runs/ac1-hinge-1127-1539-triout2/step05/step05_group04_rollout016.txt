def _project_weighted_simplex(v, w, s):
    # v and w arrays
    # Ensure w positive
    # Compute initial bounds
    # lower bound can be -np.max(v)/np.min(w) - 1.0 (ensuring f lower bound > s)
    low = -np.max(v)/np.min(w) - 1.0
    high = np.max(v / w)  # maybe >0

    # Ensure f(low) > s and f(high) < s
    # define f
    def f(tau):
        # returns sum_i w_i * max(0, v_i - tau * w_i)
        temp = v - tau * w
        temp = np.where(temp > 0, temp, 0.0)
        return np.dot(w, temp)  # sum w_i * temp_i

    # Expand low if needed
    if f(low) < s:
        # need lower bound where sum bigger; but low is too high; we can set low = low - (abs(low)+1)*2 until f(low) >= s.
        while f(low) < s:
            low = low - (abs(low) + 1) * 2
    # Expand high if needed
    if f(high) > s:
        while f(high) > s:
            high = high + (abs(high) + 1) * 2

    for _ in range(60):
        mid = (low + high) / 2.0
        fm = f(mid)
        if fm > s:
            low = mid
        else:
            high = mid
    tau = (low + high) / 2.0
    y = v - tau * w
    y = np.maximum(y, 0.0)
    return y

"""
Propose a step‑function with a low evaluation score.

Algorithm
---------
1.  Keep the ℓ¹‑norm fixed to √(2 n) (the official normalisation).  The score
    after normalisation equals the maximal entry of the self‑convolution.
2.  Enforce a *monotonically decreasing* shape `a₀ ≥ a₁ ≥ … ≥ a_{n‑1}`.
    This structure pairs large coefficients with small ones in the
    reversed sequence and is known to give a smaller convolution peak.
    The monotonicity constraint is handled by optimisation on the
    *difference* vector `x_i = a_i – a_{i+1}` (with `x_i ≥ 0`).
    The linear constraint `∑ (i+1)·x_i = √(2 n)` defines a weighted simplex.
3.  Perform an annealed projected gradient descent on a smooth approximation
    of the ℓ∞‑norm (log‑sum‑exp of the convolution).  The gradient of the
    smooth max is computed analytically via FFTs.  After each gradient step the
    vector `x` is projected onto the weighted simplex, guaranteeing a
    decreasing `a`.
4.  Refine the result with the original (non‑monotone) exact sub‑gradient
    descent on the true max‑norm, a cheap greedy *mass‑balance* move and a
    short simulated‑annealing polishing pass.
5.  Test several lengths (2000…12000) and a few random restarts; keep the
    best sequence found within the 1000 s wall‑clock budget.

The routine returns the best sequence as a plain Python list.  All numbers are
non‑negative, and the final call to ``evaluate_sequence`` verifies the score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the (unweighted) simplex
    {x | x_i ≥ 0, Σ x_i = s}.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1, dtype=float)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _project_weighted_simplex(v: np.ndarray, w: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto {x | x_i ≥ 0, Σ w_i·x_i = s},
    where w_i > 0 (weighted simplex).  Solution has the form
        x_i = max(0, v_i – τ·w_i)
    for a suitable scalar τ, found by binary search.
    """
    v = np.asarray(v, dtype=float)
    w = np.asarray(w, dtype=float)

    def weighted_sum(tau):
        tmp = v - tau * w
        np.maximum(tmp, 0.0, out=tmp)
        return np.dot(w, tmp)

    # Upper bound: τ such that all entries become non‑positive → sum = 0
    tau_high = np.max(v / w)
    # Lower bound: make the sum large enough (τ negative)
    tau_low = -np.max(v) / np.min(w) - 1.0
    # Ensure the bracketing interval contains the root
    while weighted_sum(tau_low) < s:
        tau_low = tau_low * 2.0 - 1.0
    while weighted_sum(tau_high) > s:
        tau_high = tau_high + np.abs(tau_high) + 1.0

    for _ in range(60):
        tau_mid = (tau_low + tau_high) / 2.0
        val = weighted_sum(tau_mid)
        if val > s:
            tau_low = tau_mid
        else:
            tau_high = tau_mid
    tau = (tau_low + tau_high) / 2.0
    x = v - tau * w
    np.maximum(x, 0.0, out=x)
    return x


def _max_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n−1) using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective after ℓ¹‑normalisation Σa = √(2 n): the peak of a*a."""
    return float(_max_convolution_fft(a).max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) gradient – Phase 1
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) where
        g = 2·(a ⋆ p)      with   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities p
    max_b = float(np.max(b))
    exps = np.exp(beta * (b - max_b))          # numerical stability
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1            # = 3 n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _smooth_max_descent_monotone(a0: np.ndarray,
                                 deadline: float,
                                 rng: np.random.Generator,
                                 w: np.ndarray,
                                 target_sum: float) -> np.ndarray:
    """
    Annealed projected gradient descent on the smooth‑max objective,
    keeping the sequence monotonically decreasing via a weighted‑simplex
    projection on the difference vector x.
    """
    n = a0.size
    # Convert a0 → x (differences, non‑negative)
    x = np.empty(n, dtype=float)
    x[:-1] = a0[:-1] - a0[1:]
    x[-1] = a0[-1]

    beta = 0.5
    beta_max = 4096.0
    beta_factor = 1.4

    step_base = target_sum * 0.15
    momentum = np.zeros_like(x)
    rho = 0.90

    it = 0
    while time.time() < deadline:
        # reconstruct monotone a from x
        a = np.cumsum(x[::-1])[::-1]

        g_a, _ = _gradient_softmax(a, beta)
        # gradient w.r.t. x is cumulative sum of g_a
        g_x = np.cumsum(g_a)

        # momentum update (on x‑space)
        momentum = rho * momentum + (1.0 - rho) * g_x
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            x -= lr * momentum

        # projection onto the weighted simplex (keeps monotonicity)
        x = _project_weighted_simplex(x, w, target_sum)

        # occasional jitter to escape plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                               size=n)
            x += jitter
            x = _project_weighted_simplex(x, w, target_sum)

        # anneal β
        if it % 350 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1

    # final a (monotone, ℓ¹‑normalised)
    a = np.cumsum(x[::-1])[::-1]
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target_sum)   # tiny safety projection
    return a

def _max_index_subgradient(a: np.ndarray):
    """
    Sub‑gradient of max(b) where b = a*a.
    Returns (g, k) with
        g_i = 2·a_{k-i} for indices i that participate in the maximal entry.
    """
    n = a.shape[0]
    conv = _max_convolution_fft(a)
    k = int(np.argmax(conv))

    g = np.zeros_like(a)
    if k < n:                         # left half of the convolution
        g[:k + 1] = 2.0 * a[k::-1]
    else:                             # right half
        d = k - (n - 1)               # shift amount (0 … n‑1)
        g[d:] = 2.0 * a[: n - d]
    return g, k


def _subgradient_descent_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Projected sub‑gradient descent on the true max‑norm."""
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        g, _ = _max_index_subgradient(a)

        # diminishing step size η_t = α / √t
        step = 0.8 * target_sum / (np.sqrt(it + 1) *
                                   np.linalg.norm(g) + 1e-12)
        a -= step * g

        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1
    return a


def _mass_balance_refine(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Greedy transfer that reduces the contribution of the current maximal
    convolution entry by moving a small fraction of the heaviest point to a
    relatively empty location.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sum(a)               # should be √(2 n)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _max_convolution_fft(a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        other = k - idx
        mask = (other >= 0) & (other < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[other[mask]]

        i_high = int(np.argmax(contrib))                # most offending
        low_candidates = np.where(a <= np.percentile(a, 20))[0]
        if low_candidates.size == 0:
            i_low = int(np.argmin(a))
        else:
            i_low = int(rng.choice(low_candidates))

        delta = a[i_high] * 0.03
        if delta < 1e-12:
            break

        a[i_high] -= delta
        a[i_low]  += delta

        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        if it % 100 == 0 and rng.random() < 0.02:
            jitter = rng.normal(scale=target_sum * 1e-3, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return a

def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Very light SA that moves a random fraction of mass between two sites."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    cur_score = _score(a)
    best_score = cur_score

    T = cur_score * 0.02 + 1e-9
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-16:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        frac = rng.random() * 0.15
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta

        new_score = _score(a)
        dE = new_score - cur_score
        accept = dE <= 0.0 or (T > 0.0 and rng.random() < np.exp(-dE / T))

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling

    return best


# ----------------------------------------------------------------------
# Random monotone initialiser
# ----------------------------------------------------------------------


def _random_initial_monotone(n: int, rng: np.random.Generator,
                            target_sum: float) -> np.ndarray:
    """Generate a random decreasing sequence of length n with ℓ¹‑norm √(2 n)."""
    w = np.arange(1, n + 1, dtype=float)
    # start from an exponential sample for the difference vector x
    y = rng.exponential(scale=1.0, size=n)
    x = _project_weighted_simplex(y, w, target_sum)
    a = np.cumsum(x[::-1])[::-1]
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` metric.  The routine respects the wall‑clock budget,
    explores several lengths, applies a monotone projected gradient descent,
    refines the result with exact sub‑gradient descent, a cheap mass‑balance
    move and a short simulated‑annealing polishing, and finally returns the
    best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Seed from a previously discovered construction if available
    # ------------------------------------------------------------------
    earlier = globals().get("height_sequence_1")
    if isinstance(earlier, (list, tuple, np.ndarray)):
        seq = np.asarray(earlier, dtype=float)
        try:
            val = evaluate_sequence(seq.tolist())
            best_seq = seq.copy()
            best_score = val
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 "
                  f"score={val:.6f}")
        except Exception as exc:  # pragma: no cover
            print(f"Error evaluating seed sequence: {exc}")

    # ------------------------------------------------------------------
    # Lengths to investigate – larger n gives more freedom
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000]

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)
        w = np.arange(1, n + 1, dtype=float)

        # --------------------------------------------------------------
        # Phase A – cheap random restarts with monotone smooth‑max GD
        # --------------------------------------------------------------
        phase_a_dead = min(deadline,
                           time.time() + min(40.0,
                                            (deadline - time.time()) * 0.25))
        while time.time() < phase_a_dead:
            a_start = _random_initial_monotone(n, rng, target_sum)

            sub_dead = min(deadline, time.time() + 4.0)
            a_gd = _smooth_max_descent_monotone(a_start, sub_dead, rng,
                                                w, target_sum)
            sc = _score(a_gd)
            if sc < best_score:
                best_score = sc
                best_seq = a_gd.copy()
                print(f"[{time.time():.1f}] New global best (rand) n={n} "
                      f"score={sc:.6f}")

        # --------------------------------------------------------------
        # Phase B – deeper refinement if we already have a candidate of
        #          this length
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size == n:
            # 1) Longer monotone smooth‑max descent
            phase_b_dead = min(deadline,
                               time.time() + min(80.0, deadline - time.time()))
            a_ref = _smooth_max_descent_monotone(best_seq,
                                                 phase_b_dead, rng, w,
                                                 target_sum)
            sc = _score(a_ref)
            if sc < best_score:
                best_score = sc
                best_seq = a_ref.copy()
                print(f"[{time.time():.1f}] Refined (smooth) n={n} "
                      f"score={sc:.6f}")

            # 2) Exact max‑index sub‑gradient descent (no monotone constraint)
            sub_dead = min(deadline, time.time() + 30.0)
            a_ref2 = _subgradient_descent_max(best_seq, sub_dead)
            sc = _score(a_ref2)
            if sc < best_score:
                best_score = sc
                best_seq = a_ref2.copy()
                print(f"[{time.time():.1f}] Sub‑gradient improvement n={n} "
                      f"score={sc:.6f}")

            # 3) Cheap mass‑balance fine‑tuning
            sub_dead = min(deadline, time.time() + 20.0)
            a_ref3 = _mass_balance_refine(best_seq, sub_dead)
            sc = _score(a_ref3)
            if sc < best_score:
                best_score = sc
                best_seq = a_ref3.copy()
                print(f"[{time.time():.1f}] Mass‑balance refinement n={n} "
                      f"score={sc:.6f}")

    # ------------------------------------------------------------------
    # Final simulated‑annealing polishing with whatever time is left
    # ------------------------------------------------------------------
    if best_seq is not None:
        left = deadline - time.time()
        if left > 5.0:
            a_final = _simulated_annealing(best_seq, deadline)
            sc = _score(a_final)
            if sc < best_score:
                best_score = sc
                best_seq = a_final.copy()
                print(f"[{time.time():.1f}] Final SA improvement: "
                      f"score={sc:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up tiny trailing zeros, evaluate, and return
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution of length 1000
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
