"""
Improved step‑function optimizer.

The search proceeds in three stages for a single, relatively large length
(e.g. 15000).  Starting from an up‑sampled previous construction (if any)
or a random point on the ℓ¹‑simplex Σa = √(2 n) we run:

1. **Smoothed‑max projected gradient descent** – the objective
   ϕβ(a) = (1/β)·log∑ exp(β·(a∗a)) approximates the max‑convolution.
   Its gradient g = 2·(a⋆p) (p a soft‑max over the convolution) is
   obtained by two FFTs.  We use momentum, an annealed inverse temperature
   β and project after each step onto the simplex.

2. **Exact mass‑transfer hill climbing** – with the full convolution
   b = a∗a kept, random moves transfer a fraction of mass from a large
   entry to a small one.  The convolution update is performed analytically
   in O(n) using the formula from the original code.  A simple simulated‑
   annealing schedule accepts occasional uphill steps.

3. **Sub‑gradient polishing** – after the hill climb we run a few
   sub‑gradient steps on the true max‑convolution (no smoothing).  The
   sub‑gradient is built from the rows of the Jacobian belonging to the
   maximal convolution indices; a diminishing step size and projection
   complete the iteration.

The whole routine respects a wall‑clock budget (default 1000 s) and
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a using FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (size 2·n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed maximum (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of   (1/β)·log Σ exp(β·b)   where   b = a*a .
    Returns (g, b) with g = 2·(a ⋆ p) and
    pₖ ∝ exp(β·bₖ).  Correlation is performed via FFT.
    """
    n = a.shape[0]

    # b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]                          # reversed a
    conv_len = n + (2 * n - 1) - 1           # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smoothed max.
    Uses momentum, an annealed inverse temperature β and occasional jitter.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 8192.0
    beta_factor = 1.4

    step_base = target_sum * 0.15
    momentum = np.zeros_like(a)
    rho = 0.90

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β annealing
        if it % 350 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative sequence on the ℓ¹‑simplex Σa = √(2 n).
    Several families (power‑law, exponential, Dirichlet, piecewise) are
    sampled uniformly.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # pure power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                  # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                  # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                  # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                  # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length.
    The result is projected back onto the appropriate simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Mass‑transfer hill climbing (exact convolution updates)
# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Hill‑climbing with exact convolution updates.
    a0 – initial sequence (ℓ¹‑normalised)
    b0 – its full convolution a0*a0
    Returns the best pair (a, b) observed before ``deadline``.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()
    best_b = b.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.03 + 1e-9            # a bit hotter than before
    cooling = 0.9998

    while time.time() < deadline:
        # pick a source index weighted by its current mass
        if a.sum() == 0.0:
            break
        probs = a / a.sum()
        i = rng.choice(n, p=probs)

        # source too small – skip
        if a[i] < 1e-14:
            continue

        # choose a destination from the low‑mass tail
        # 20% chance to pick uniformly, 80% from the smallest entries
        if rng.random() < 0.2:
            j = rng.integers(0, n)
        else:
            # pick an index among the 5 % smallest entries
            k = max(1, n // 20)
            small_idx = np.argpartition(a, k)[:k]
            j = rng.choice(small_idx)

        if i == j:
            continue

        # amount of mass to move (up to 40 % of source)
        max_delta = a[i] * 0.4
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ---------- compute new convolution ----------
        b_new = b.copy()
        # contribution from the –δ at i
        b_new[i:i + n] -= 2.0 * delta * a
        # contribution from the +δ at j
        b_new[j:j + n] += 2.0 * delta * a

        # quadratic correction terms
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # ---------- acceptance ----------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-(new_score - cur_score) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                best_b = b.copy()
                print(f"[{time.time():.1f}] hill‑climb improvement: {best_score:.6f}")

        T *= cooling

    return best_a, best_b


# ----------------------------------------------------------------------
# Sub‑gradient polish (direct max‑convolution)
# ----------------------------------------------------------------------


def _subgradient_polish(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    A few sub‑gradient steps on the true max‑convolution.
    Each step moves opposite to the sub‑gradient of max(b) and projects onto
    the ℓ¹‑simplex.
    """
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)
    t = 1
    rng = np.random.default_rng()

    while time.time() < deadline:
        b = _full_convolution(a)
        M = b.max()
        ks = np.flatnonzero(b == M)          # indices attaining the max
        g = np.zeros_like(a)

        for k in ks:
            # build the vector s with s[i] = a[k-i] where defined
            s = np.zeros_like(a)
            if k < n:
                # i = 0..k
                s[:k + 1] = a[k::-1]
            else:
                L = 2 * n - 1 - k
                start = k - n + 1
                s[start:] = a[n - L:n][::-1]
            g += 2.0 * s
            if k % 2 == 0:
                i0 = k // 2
                if 0 <= i0 < n:
                    g[i0] -= a[i0]          # correct the double count on the diagonal

        # diminishing step size proportional to 1/√t
        step = target * 0.12 / (np.linalg.norm(g) + 1e-12) / np.sqrt(t)
        # occasional random jitter to avoid stagnation
        if t % 250 == 0:
            step *= 0.5
            jitter = rng.normal(scale=target * 0.0005, size=n)
            a += jitter
            a = np.maximum(a, 0.0)

        a -= step * g
        a = _project_to_simplex(a, target)
        t += 1

    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock budget (default 1000 s) and returns
    the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # safety margin

    # --------------------------------------------------------------
    # Determine target length – we aim straight for a relatively large n
    # --------------------------------------------------------------
    # The larger n, the more degrees of freedom; 15000 is a good compromise
    # between quality and per‑iteration cost.
    n_target = 15000

    # --------------------------------------------------------------
    # Initialise starting point (upsample previous solution if present)
    # --------------------------------------------------------------
    earlier = globals().get("height_sequence_1")
    if isinstance(earlier, (list, tuple, np.ndarray)):
        a_start = np.asarray(earlier, dtype=float)
        a_start = _upsample_sequence(a_start, n_target)
        print(f"[{time.time():.1f}] Seeded from height_sequence_1 (upsampled to n={n_target})")
    else:
        a_start = _random_initial_sequence(n_target, rng)
        print(f"[{time.time():.1f}] Random start, n={n_target}")

    # --------------------------------------------------------------
    # Phase A – long projected gradient descent on smooth max
    # --------------------------------------------------------------
    phase_a_deadline = min(deadline, time.time() + (deadline - start_time) * 0.55)
    a_gd = _gradient_descent_opt(a_start, phase_a_deadline)
    best_score = _max_convolution_fft(a_gd)
    best_seq = a_gd.copy()
    print(f"[{time.time():.1f}] Phase A (GD) best score: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – exact hill‑climbing with analytical convolution updates
    # --------------------------------------------------------------
    if time.time() < deadline:
        b_full = _full_convolution(best_seq)
        phase_b_deadline = min(deadline, time.time() + (deadline - start_time) * 0.35)
        a_hc, b_hc = _hill_climbing(best_seq, b_full, phase_b_deadline)
        score_hc = b_hc.max()
        if score_hc < best_score:
            best_score = score_hc
            best_seq = a_hc.copy()
            print(f"[{time.time():.1f}] Phase B (hill‑climb) improvement: {best_score:.6f}")
        else:
            print(f"[{time.time():.1f}] Phase B completed, no improvement (score={score_hc:.6f})")

    # --------------------------------------------------------------
    # Phase C – sub‑gradient polishing (direct max)
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase_c_deadline = deadline
        a_final = _subgradient_polish(best_seq, phase_c_deadline)
        final_score = _max_convolution_fft(a_final)
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Phase C (sub‑gradient) improvement: {best_score:.6f}")
        else:
            print(f"[{time.time():.1f}] Phase C completed, no improvement (score={final_score:.6f})")

    # --------------------------------------------------------------
    # Trim trailing near‑zeros for neatness
    # --------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # --------------------------------------------------------------
    # Final official evaluation
    # --------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:      # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
