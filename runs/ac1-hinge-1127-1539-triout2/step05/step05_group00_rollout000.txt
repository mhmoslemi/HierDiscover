def _hill_climbing_improved(a0, b0, deadline):
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()
    best_b = b.copy()
    rng = np.random.default_rng()
    T = cur_score * 0.05  # higher temp
    cooling = 0.9995

    while time.time() < deadline:
        # compute max index
        m = int(np.argmax(b))
        # compute contributions
        contrib = np.zeros(n)
        low = max(0, m - (n-1))
        high = min(n-1, m)
        # i in [low, high]; for each i, j = m-i 
        # j is within [0,n-1]
        # Compute product:
        # but we use vectorized:
        i_range = np.arange(low, high+1)
        j_range = m - i_range
        contrib[i_range] = a[i_range] * a[j_range]

        # choose source index
        if contrib.sum() > 0 and rng.random() < 0.7:
            probs = contrib / contrib.sum()
            i = rng.choice(n, p=probs)
        else:
            # fallback: choose proportionally to a
            probs = a / a.sum() if a.sum() > 0 else np.full(n, 1.0/n)
            i = rng.choice(n, p=probs)

        if a[i] < 1e-14:
            continue

        # choose destination
        j = rng.integers(0, n)
        if i == j: 
            continue

        # fraction of source mass to move
        max_delta = a[i] * 0.5
        delta = rng.random() * max_delta

        # compute new b using formula
        b_new = b.copy()
        b_new[i:i+n] -= 2.0 * delta * a
        b_new[j:j+n] += 2.0 * delta * a
        k2i = 2*i; k2j = 2*j; kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta*delta
        if k2j < b_new.size:
            b_new[k2j] += delta*delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta*delta

        new_score = b_new.max()
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if T > 1e-12 and rng.random() < np.exp(-(new_score - cur_score) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                best_b = b.copy()
                print(f"[{time.time():.1f}] improved hill: {best_score:.6f}")
        T *= cooling
    return best_a, best_b

"""
Propose candidate – adaptive Adam + directed hill‑climbing.

The routine works in three stages for a series of increasing lengths:

1. **Warm‑start** – take the best found sequence (or a random simplex
   point) and up‑sample it to the new length.

2. **Adam optimisation** – minimise a smooth log‑sum‑exp
   approximation of the max‑convolution.  The Adam optimiser together
   with an annealing inverse‑temperature β provides fast descent
   while keeping the ℓ¹‑norm fixed via Euclidean projection onto the
   simplex Σa = √(2 n).

3. **Directed hill‑climbing** – repeatedly move a fraction of mass from
   indices that contribute most to the current maximal convolution entry
   to indices that contribute little.  Changes to the full convolution
   are updated in O(n) using analytical formulas, and a simulated‑annealing
   schedule allows occasional uphill moves.

The algorithm respects a wall‑clock budget (default 1000 s) and returns
the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a as a NumPy array."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed maximum (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of   (1/β)·log Σ exp(β·b)   where   b = a*a .
    Returns (g, b) with g = 2·(a ⋆ p) and
    pₖ ∝ exp(β·bₖ).  Correlation is performed via FFT.
    """
    n = a.shape[0]

    # b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]                          # reversed a
    conv_len = n + (2 * n - 1) - 1           # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max‑convolution objective
# ----------------------------------------------------------------------


def _adam_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected Adam optimisation of the smooth max‑convolution.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.25 * target_sum          # base learning rate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature for log‑sum‑exp
    beta = 0.5
    beta_max = 8192.0
    beta_factor = 1.3

    rng = np.random.default_rng()
    t = 0
    while time.time() < deadline:
        t += 1

        # gradient of the smoothed max
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        # occasional small jitter to escape plateaux
        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β annealing (smooth → true max)
        if t % 350 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a

# Random initialisers

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative sequence on the ℓ¹‑simplex Σa = √(2 n).
    Several families are sampled uniformly.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )

    else:                               # near‑uniform with jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length.
    The result is projected back onto the appropriate simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Directed hill‑climbing with exact convolution updates
# ----------------------------------------------------------------------


def _hill_climbing_improved(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Hill‑climbing that preferentially removes mass from indices that
    contribute most to the current maximal convolution entry.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()
    best_b = b.copy()

    rng = np.random.default_rng()
    # Start with a relatively high temperature to allow uphill moves.
    T = cur_score * 0.05 + 1e-12
    cooling = 0.9996

    while time.time() < deadline:
        # locate the current maximal convolution index
        m = int(np.argmax(b))

        # compute contributions a_i * a_{m-i} for all i that can pair to m
        contrib = np.zeros(n)
        lo = max(0, m - (n - 1))
        hi = min(n - 1, m)
        i_range = np.arange(lo, hi + 1)
        j_range = m - i_range
        contrib[i_range] = a[i_range] * a[j_range]

        # choose source index i
        if contrib.sum() > 0 and rng.random() < 0.7:
            probs = contrib / contrib.sum()
            i = rng.choice(n, p=probs)
        else:
            # fallback: proportional to current mass
            probs = a / a.sum() if a.sum() > 0 else np.full(n, 1.0 / n)
            i = rng.choice(n, p=probs)

        if a[i] < 1e-14:
            continue

        # choose destination index j (uniform)
        j = rng.integers(0, n)
        if i == j:
            continue

        # amount of mass to move (up to 30 % of source)
        max_delta = a[i] * 0.3
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ---- exact update of the convolution ----
        b_new = b.copy()
        # linear part
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a

        # quadratic correction terms
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # ---- acceptance ----
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # simulated‑annealing acceptance
            if T > 1e-12 and rng.random() < np.exp(-(new_score - cur_score) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                best_b = b.copy()
                print(f"[{time.time():.1f}] hill‑climb improvement: {best_score:.6f}")

        T *= cooling

    return best_a, best_b


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock budget (default 1000 s) and returns
    the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Initialise from a previously stored construction if available
    # ------------------------------------------------------------------
    earlier = globals().get("height_sequence_1")
    if isinstance(earlier, (list, tuple, np.ndarray)):
        a0 = np.asarray(earlier, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        best_seq = a0.copy()
        best_score = _max_convolution_fft(a0)
        print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) "
              f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Length schedule – we slowly increase the dimension
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 16000, 25000]

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Initialise a candidate for this length
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)
        a_init = _project_to_simplex(a_init, target_sum)

        # --------------------------------------------------------------
        # Phase A – Adam optimisation (smooth max)
        # --------------------------------------------------------------
        phase_end = min(deadline,
                        time.time() + min(150.0, (deadline - time.time()) * 0.45))
        a_opt = _adam_opt(a_init, phase_end)
        score_opt = _max_convolution_fft(a_opt)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] Adam improvement (n={n}) score={score_opt:.6f}")

        # --------------------------------------------------------------
        # Phase B – directed hill‑climbing with exact convolution updates
        # --------------------------------------------------------------
        if time.time() < deadline:
            b_full = _full_convolution(a_opt)
            hill_dead = min(deadline,
                            time.time() + min(80.0, (deadline - time.time()) * 0.35))
            a_hc, b_hc = _hill_climbing_improved(a_opt, b_full, hill_dead)
            score_hc = b_hc.max()
            if score_hc < best_score:
                best_score = score_hc
                best_seq = a_hc.copy()
                print(f"[{time.time():.1f}] Hill‑climb improvement (n={n}) "
                      f"score={score_hc:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with any remaining budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a_final = best_seq.copy()
        b_final = _full_convolution(a_final)
        final_dead = deadline
        a_final, b_final = _hill_climbing_improved(a_final, b_final, final_dead)
        final_score = b_final.max()
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final polishing improvement: "
                  f"score={final_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros for neatness
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log the official evaluation (objective = max of the self‑convolution)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
