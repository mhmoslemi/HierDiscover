def solve_lp_max_sum_under_M(a, M):
    n = a.shape[0]
    # Build A matrix as described.
    # Use scipy.sparse to build efficiently
    import scipy.sparse as sp
    # Create data for each column i: a[i] repeated n entries.
    rows = np.concatenate([np.arange(i, i+n) for i in range(n)])
    cols = np.repeat(np.arange(n), n)
    data = np.repeat(a, n) # shape n*n.
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # Solve LP: maximize sum(b) s.t. A b <= M, b >= 0.
    c = -np.ones(n)  # minimization of -sum(b)
    bounds = [(0, None)] * n
    # b_ub = M * np.ones(2*n-1)
    # Solve using scipy.optimize.linprog
    result = scipy.optimize.linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if not result.success:
        return None
    return result.x
```

"""
Hybrid optimiser for the step‑function benchmark.

The optimiser works in three stages:

1. **Parameter sweep** – evaluates a family
   a_i ∝ (i+1)^{-p}·exp(-λ·i) for a grid of (p,λ).  The best
   few candidates are kept.

2. **Adam projected gradient descent** – starting from each retained
   candidate, the soft‑max approximation of the max‑convolution is
   minimised with an Adam optimiser.  The iterate is always projected
   onto the ℓ¹‑simplex Σa = √(2·n).

3. **Deterministic hill‑climbing** – a fast greedy mass‑transfer
   routine reduces the current maximum convolution entry.  Moves are
   kept only if they strictly improve the objective.

The whole routine is time‑limited (default 1000 s).  Several
independent random restarts are performed in the remaining budget,
and the best sequence found is returned.
"""

import time
import numpy as np
import scipy.optimize  # only for the optional LP step (not used here)

# ----------------------------------------------------------------------
# Fast convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Score = max_{k} (a*a)_k  (sum is normalised to √(2n) elsewhere)."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution (used by Adam)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(conv).
    Returns (g, conv) where g is the gradient w.r.t. a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)          # soft‑max weights

    # gradient = 2 * (reverse(a) * p) convolution, extracted centre part
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    g = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Adam projected gradient descent on the soft‑max objective
# ----------------------------------------------------------------------


def _adam_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Adam optimiser on the soft‑max surrogate of max(conv).
    Keeps the iterate on the ℓ¹‑simplex Σa = target_sum.
    """
    a = a0.astype(np.float64).copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    beta = 1.0
    max_beta = 1024.0

    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)

        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # step size – tuned for the simplex geometry
        step = 0.2 * target_sum / (np.sqrt(v_hat).mean() + 1e-12)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # projection back onto the feasible simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if t % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.size)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] Adam improve → score={best_val:.6f}")

        # temperature schedule (increase β → sharper approximation)
        if t % 250 == 0 and beta < max_beta:
            beta = min(beta * 1.4, max_beta)

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic hill‑climbing focused on the current maximal entry
# ----------------------------------------------------------------------


def _hill_climb(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Greedy mass‑transfer moves that only accept strict improvements.
    The move always keeps the ℓ¹‑sum unchanged.
    """
    a = a0.astype(np.float64).copy()
    n = a.size
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline:
        # compute convolution once
        conv = _conv_fft(a)
        k_max = int(np.argmax(conv))

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each participating index to the maximal entry
        contrib = a[idx] * a[k_max - idx]

        # donor: index with large contribution (probability weighted)
        total = contrib.sum()
        if total > 0:
            donor = rng.choice(idx, p=contrib / total)
        else:
            donor = rng.integers(n)

        # receiver: any index *outside* the active window (zero contribution)
        outside = np.setdiff1d(np.arange(n), idx, assume_unique=True)
        if outside.size:
            receiver = rng.choice(outside)
        else:
            # fallback – pick a low‑contribution index inside the window
            receiver = idx[np.argmin(contrib)]

        if donor == receiver or a[donor] <= 0:
            continue

        # propose a transfer (up to 40 % of donor mass)
        delta = a[donor] * rng.random() * 0.40
        a[donor] -= delta
        a[receiver] += delta

        new_score = _score(a)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] Hill‑climb new best = {best_score:.6f}")
        else:
            # reject – revert
            a[donor] += delta
            a[receiver] -= delta

        # occasional tiny jitter to escape flat regions
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = a * (target_sum / a.sum())
            cur_score = _score(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Parameter grid search (power‑law × exponential)
# ----------------------------------------------------------------------


def _grid_search(n: int, target_sum: float, deadline: float,
                 best_seq: np.ndarray, best_score: float):
    """
    Scan a family a_i ∝ (i+1)^{-p}·exp(-λ·i) and keep the
    top few candidates for further refinement.
    """
    p_vals = np.linspace(0.10, 2.00, 38)      # step ≈0.05
    lam_vals = np.linspace(0.0, 0.020, 21)    # step ≈0.001
    candidates = []

    for p in p_vals:
        for lam in lam_vals:
            idx = np.arange(n, dtype=np.float64)
            a = (idx + 1) ** (-p) * np.exp(-lam * idx)
            a = a / a.sum() * target_sum
            sc = _score(a)
            candidates.append((sc, a))
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] Grid best p={p:.3f} λ={lam:.4f} → {sc:.6f}")

            # stop early if we are out of time
            if time.time() > deadline - 5.0:
                break
        if time.time() > deadline - 5.0:
            break

    # keep the 6 best candidates for a short Adam refinement
    candidates.sort(key=lambda x: x[0])
    top_candidates = [c[1] for c in candidates[:6]]
    return best_seq, best_score, top_candidates


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function minimising `evaluate_sequence`.
    The routine combines a parameter sweep, Adam gradient descent and a
    deterministic hill‑climb.  Several random restarts are performed
    within the allotted time budget.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # small safety margin

    # problem size – 1500 gives a good trade‑off between quality and speed
    n = 1500
    target_sum = np.sqrt(2.0 * n)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0)  Warm‑start from the globally supplied sequence if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if seq.size > n:
            seq = seq[:n]
        else:
            seq = np.pad(seq, (0, n - seq.size), "constant")
        seq = _project_to_simplex(seq, target_sum)
        sc = _score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] global warm‑start score={sc:.6f}")

    # ------------------------------------------------------------------
    # 1)  Parameter sweep (cheap)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score, top_candidates = _grid_search(
            n, target_sum, deadline, best_seq, best_score
        )
    else:
        top_candidates = []

    # ------------------------------------------------------------------
    # 2)  Refine a few top candidates with Adam + hill‑climb
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    for a0 in top_candidates:
        if time.time() + 10.0 >= deadline:
            break
        # Adam phase (≈5 s)
        adam_deadline = time.time() + 5.0
        a_adam, sc_adam = _adam_descent(a0, target_sum, adam_deadline)

        # Hill‑climb phase (use remaining time of this slot)
        hc_deadline = deadline if deadline - time.time() < 8.0 else time.time() + 8.0
        a_hc, sc_hc = _hill_climb(a_adam, target_sum, hc_deadline)

        if sc_hc < best_score:
            best_score, best_seq = sc_hc, a_hc.copy()
            print(f"[{time.time():.1f}] refined candidate → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3)  Random restarts: Dirichlet init → Adam → hill‑climb
    # ------------------------------------------------------------------
    while time.time() + 15.0 < deadline:
        # random Dirichlet‑type start
        a0 = rng.exponential(scale=1.0, size=n)
        a0 = a0 / a0.sum() * target_sum

        # short Adam optimisation
        adam_deadline = time.time() + 5.0
        a_adam, _ = _adam_descent(a0, target_sum, adam_deadline)

        # hill‑climb using the rest of the time budget
        a_final, sc_final = _hill_climb(a_adam, target_sum, deadline)

        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final.copy()
            print(f"[{time.time():.1f}] random restart improved best → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final tidy‑up – drop trailing near‑zero entries
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.zeros(n, dtype=float)

    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    print(f"[{time.time():.1f}] final best score={best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
