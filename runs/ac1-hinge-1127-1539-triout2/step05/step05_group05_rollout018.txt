"""
Propose a step‑function that minimizes the maximal entry of its self‑convolution.

Algorithm
---------
1. **Broad random scan** – generate many candidate sequences for several
   lengths (200 … 4000).  Families include Dirichlet, Beta‑shaped,
   Gaussian, two‑Gaussian mixtures and exponentials.  Each candidate is
   projected onto the ℓ¹‑simplex Σaᵢ = √(2 n) and its max‑convolution is
   evaluated with an FFT; the best candidate is kept.

2. **LP refinement** – for the current best sequence `a` (length ≤2000) we
   solve a linear program

        max   Σ gᵢ               (equivalently min –Σ gᵢ)
        s.t.  (a * g)ₖ ≤ M  for all k,
              gᵢ ≥ 0,

   where `M` is the present maximal convolution value.
   The LP pushes mass into indices that do not increase any convolution
   entry beyond `M`.  After solving we renormalise `g` to the required
   ℓ¹‑norm and mix it with `a` (a ← (1‑t) a + t g).  A few mix‑steps
   quickly lower the peak.

3. **Projected sub‑gradient descent** – repeatedly move a small step in
   the negative sub‑gradient of the max‑convolution (the gradient of the
   active convolution entry) and re‑project onto the simplex.  A
   geometrically decaying step‑size yields a fast local improvement.

4. **Targeted simulated annealing** – fine‑tune the result by repeatedly
   moving a random fraction of mass from an index that heavily contributes
   to the current maximal convolution entry to a random destination.
   Uphill moves are accepted with a temperature schedule; occasional
   global jitter helps escape shallow minima.

All phases respect the global time budget (default 1000 s).  The routine
returns a plain Python list; `evaluate_sequence` can be called afterwards
to obtain the official score.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Fast convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean, Duchi et al. 2008)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x_i≥0, Σx_i = s}."""
    if s <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random families
# ----------------------------------------------------------------------


def _rand_beta(n: int, rng) -> np.ndarray:
    """Beta‑shaped density (U‑shaped or bell‑shaped)."""
    alpha = rng.uniform(0.2, 5.0)
    beta = rng.uniform(0.2, 5.0)
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    return a


def _rand_gaussian(n: int, rng) -> np.ndarray:
    """Symmetric discrete Gaussian."""
    sigma = rng.uniform(0.05 * n, 0.20 * n)
    centre = (n - 1) / 2.0
    i = np.arange(n)
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a


def _rand_two_humps(n: int, rng) -> np.ndarray:
    """Mixture of two Gaussians placed near the edges."""
    sigma = rng.uniform(0.03 * n, 0.10 * n)
    left = rng.uniform(0.0, 0.2 * n)
    right = n - 1 - left
    i = np.arange(n)
    a = np.exp(-0.5 * ((i - left) / sigma) ** 2) + np.exp(
        -0.5 * ((i - right) / sigma) ** 2
    )
    return a


def _rand_exponential(n: int, rng) -> np.ndarray:
    """Exponential decay from the left edge."""
    lam = rng.exponential(scale=0.02)
    a = np.exp(-lam * np.arange(n))
    return a


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Generate a random non‑negative sequence on the ℓ¹‑simplex."""
    target_sum = np.sqrt(2.0 * n)
    kind = rng.integers(0, 5)
    if kind == 0:
        a = _rand_beta(n, rng)
    elif kind == 1:
        a = _rand_gaussian(n, rng)
    elif kind == 2:
        a = _rand_two_humps(n, rng)
    elif kind == 3:
        a = _rand_exponential(n, rng)
    else:
        a = rng.dirichlet(alpha=np.ones(n))
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target_sum)
    return a


# ----------------------------------------------------------------------
# Linear‑programming refinement (active‑set step)
# ----------------------------------------------------------------------


def _lp_step(a: np.ndarray) -> np.ndarray | None:
    """
    One LP step:
        maximise Σ g_i  subject to  (a * g)_k ≤ M  for all k,  g_i ≥ 0.
    Returns the optimal g (already normalised to Σg = √(2 n)),
    or None if the LP fails.
    """
    n = a.size
    S = np.sqrt(2.0 * n)
    M = _max_convolution_fft(a)

    rows = 2 * n - 1

    # Build the constraint matrix A_ub such that A_ub @ g ≤ M
    # For each column j (g_j) the row indices are k = i + j, i = 0..n-1
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j : j + n, j] = a  # rows i + j = j … j+n-1

    # Linear programme: minimise - Σ g_i  (i.e. maximise Σ g_i)
    c = -np.ones(n, dtype=np.float64)

    res = linprog(
        c,
        A_ub=A,
        b_ub=np.full(rows, M, dtype=np.float64),
        bounds=[(0.0, None)] * n,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        return None

    g_raw = res.x
    total = g_raw.sum()
    if total <= 0:
        return None
    # Renormalise to the required ℓ¹‑norm
    g = g_raw * (S / total)
    return g


def _lp_refine(a0: np.ndarray, deadline: float, max_iter: int = 6, t: float = 0.3) -> np.ndarray:
    """
    Repeated LP‑mixing.  `t` is the mixing weight (0 < t ≤ 1).
    """
    a = a0.copy()
    best_score = _max_convolution_fft(a)
    S = np.sqrt(2.0 * a.size)

    for it in range(max_iter):
        if time.time() >= deadline:
            break
        g = _lp_step(a)
        if g is None:
            break
        # convex combination and re‑project (numerical drift may appear)
        a = (1.0 - t) * a + t * g
        a = _project_to_simplex(a, S)
        cur = _max_convolution_fft(a)
        if cur < best_score:
            best_score = cur
            print(f"[{time.time():.1f}] LP refinement iter {it}: {cur:.6f}")
        else:
            # if no progress, reduce the mixing factor
            t *= 0.7
            if t < 1e-3:
                break
    return a

def _subgradient_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected sub‑gradient descent on max(a*a).  Step size decays geometrically.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    step = 0.5
    decay = 0.9995
    best = a.copy()
    best_score = _max_convolution_fft(a)

    while time.time() < deadline:
        # current full convolution and maximal index
        conv = _full_convolution_fft(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i = np.arange(lo, hi + 1)
        j = k_max - i
        g[i] = 2.0 * a[j]

        # gradient step and projection
        a = a - step * g
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        cur = _max_convolution_fft(a)
        if cur < best_score:
            best_score = cur
            best = a.copy()
            print(f"[{time.time():.1f}] Sub‑gradient improvement: {cur:.6f}")

        step *= decay
        if step < 1e-6:
            break

    return best


# ----------------------------------------------------------------------
# Targeted simulated annealing (fine tuning)
# ----------------------------------------------------------------------


def _targeted_simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Classic SA that moves mass from a “heavy” index to a random destination.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # initial temperature proportional to the current objective
    T = best_score * 0.02 + 1e-8
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        # compute convolution once per iteration
        conv = _full_convolution_fft(a)
        k_max = int(np.argmax(conv))

        # indices that actually influence the current maximal entry
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idx_range = np.arange(lo, hi + 1)
        contrib = a[idx_range] * a[k_max - idx_range]
        weight_sum = contrib.sum()

        if weight_sum > 0:
            probs = contrib / weight_sum
            i = rng.choice(idx_range, p=probs)
        else:
            i = rng.integers(0, n)

        # choose destination
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        # transfer a random fraction of the source mass
        delta = a[i] * rng.random() * 0.25
        if delta <= 0.0:
            continue
        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)

        accept = False
        if cur_score <= best_score:
            accept = True
        elif rng.random() < np.exp(-(cur_score - best_score) / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA improvement: {best_score:.6f}")
        else:
            # revert
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, np.sqrt(2.0 * n))
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA jitter improvement: {best_score:.6f}")

    return best


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative sequence) minimising
    ``evaluate_sequence`` under the supplied time budget.
    """
    rng = np.random.default_rng(seed)
    finish = time.time() + budget_s - 0.3  # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 0.  Start from a global construction if it exists
    # ------------------------------------------------------------------
    global_seq = globals().get("height_sequence_1")
    if isinstance(global_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(global_seq, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        sc = _max_convolution_fft(a0)
        best_seq, best_score = a0.copy(), sc
        print(
            f"[{time.time():.1f}] Starting from global height_sequence_1 (n={n0})  score={sc:.6f}"
        )

    # ------------------------------------------------------------------
    # 1. Random scan over several lengths
    # ------------------------------------------------------------------
    lengths = [800, 1200, 1600, 2000, 2400, 2800, 3200, 3600, 4000]
    scan_time_per_len = 6.0

    for n in lengths:
        if time.time() >= finish:
            break
        target_sum = np.sqrt(2.0 * n)
        end_len = min(finish, time.time() + scan_time_per_len)

        while time.time() < end_len:
            a = _random_initial_sequence(n, rng)
            score = _max_convolution_fft(a)
            if score < best_score:
                best_seq, best_score = a.copy(), score
                print(f"[{time.time():.1f}] New global best (scan) n={n}  score={score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform if nothing was generated
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1200
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = _max_convolution_fft(best_seq)

    # ------------------------------------------------------------------
    # 2. LP‑based refinement (only for moderate lengths to keep memory low)
    # ------------------------------------------------------------------
    remaining = finish - time.time()
    if remaining > 30.0 and best_seq.size <= 2000:
        lp_deadline = time.time() + remaining * 0.25
        best_seq = _lp_refine(best_seq, lp_deadline, max_iter=6, t=0.3)
        best_score = _max_convolution_fft(best_seq)
        print(f"[{time.time():.1f}] After LP refinement score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Projected sub‑gradient descent
    # ------------------------------------------------------------------
    remaining = finish - time.time()
    if remaining > 30.0:
        gd_deadline = time.time() + remaining * 0.30
        best_seq = _subgradient_descent(best_seq, gd_deadline)
        best_score = _max_convolution_fft(best_seq)
        print(f"[{time.time():.1f}] After sub‑gradient descent score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Targeted simulated annealing (final polishing)
    # ------------------------------------------------------------------
    remaining = finish - time.time()
    if remaining > 10.0:
        sa_deadline = time.time() + remaining
        best_seq = _targeted_simulated_annealing(best_seq, sa_deadline)
        best_score = _max_convolution_fft(best_seq)
        print(f"[{time.time():.1f}] After SA polishing score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zeros, sanity‑check and return
    # ------------------------------------------------------------------
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Optional validation with the official evaluator
    try:
        ev = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence = {ev:.6f}")
    except Exception:
        pass

    print(
        f"[{time.time():.1f}] Finished. Best objective (max‑conv) = {best_score:.6f}   length = {best_seq.size}"
    )
    return best_seq.tolist()
