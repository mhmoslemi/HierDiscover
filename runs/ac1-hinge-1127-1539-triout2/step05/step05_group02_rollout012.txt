"""
Algorithm: multiple restarts + projected subgradient descent (exact max gradient) + refined simulated annealing.
We start from several initial shapes (global sequence, power law, random). Each candidate is projected onto
L1‑simplex of sum sqrt(2·n). Subgradient descent directly reduces the current maximal convolution entry,
using adaptive step sizes and Euclidean projection onto the simplex.
After that a simulated‑annealing phase explores neighbourhoods using mass‑transfer, short‑window smoothing
and random swaps. The temperature and allowed move size are cooled exponentially.
The best sequence found within the allotted budget is returned (trailing near‑zero entries removed).
"""

import time
import numpy as np

# Helper functions: FFT convolution, score, simplex projection, subgradient, etc.

def _conv_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))

def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x>=0, sum=x = s}."""
    if s < 0:
        raise ValueError("target sum must be non‑negative")
    n = v.shape[0]
    # Duchi et al. algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a

def _subgradient_descent(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact maximum of the convolution.
    The gradient of the active convolution entry is
        g_i = 2*a_{k-i}   if i != k-i,
        g_i = a_i         if i == k-i,
    where k is the index of the current maximum.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_val = _score(a)
    it = 0
    rng = np.random.default_rng()
    # base step proportional to target_sum; we will scale by gradient norm.
    step_base = target_sum * 0.2

    while time.time() < deadline:
        # compute convolution and locate the active index
        conv = _conv_fft(a)
        max_b = conv.max()
        k = int(conv.argmax())

        # build sub‑gradient of b_k
        grad = np.zeros_like(a)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        i_valid = idx[mask]
        j_valid = j[mask]
        # equality case (i == j)
        eq = i_valid == j_valid
        grad[i_valid[eq]] = a[i_valid[eq]]
        # inequality case
        neq = ~eq
        if np.any(neq):
            grad[i_valid[neq]] = 2.0 * a[j_valid[neq]]

        # adaptive step size
        norm_g = np.linalg.norm(grad) + 1e-12
        lr = step_base / (norm_g * np.sqrt(it + 1))
        a = a - lr * grad

        # Euclidean projection back to simplex
        a = _project_to_simplex(a, target_sum)

        # small random jitter every few iterations to avoid stagnation
        if it % 500 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] subgrad it {it} → score={best_val:.6f}")

        it += 1

    return best_a, best_val

def _simulated_annealing(a0: np.ndarray, deadline: float,

      * mass transfer from a “high‑sensitivity” index to a low‑sensitivity one,

      * random pairwise swap.
    The temperature and allowed move size decay exponentially.
    """

    n = a.shape[0]
    cur_score = _score(a)
    best_a = a.copy()

    # keep current convolution for faster max location

        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        sens = np.zeros(n)

        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]       # exactly the gradient w.r.t. a_i for the active entry

        weight = a * sens

        # -------------------------------------------------
        # 1) mass transfer (dominant)
        # -------------------------------------------------
        if rng.random() < 0.70:

            if total_w > 0:

            else:
                donor = rng.integers(n)

"""
Optimisation of a non‑negative step‑function for the benchmark
``evaluate_sequence``.

The optimiser consists of several independent restarts (global, power‑law,
random).  Each restart follows a two‑stage pipeline:

1. **Projected sub‑gradient descent** on the exact maximum of the
   self‑convolution.  The active convolution index k is located,
   its sub‑gradient is
        g_i = 2·a_{k‑i}  for i≠k‑i,
        g_i = a_i        for i = k‑i,
   (zero outside the support).  After a gradient step we project back onto
   the ℓ¹‑simplex {a_i≥0, Σa_i = √(2·n)} to keep the normalisation used by
   ``evaluate_sequence``.

2. **Simulated‑annealing refinement**.  Moves are
      * mass transfer from a high‑sensitivity index (large contribution to
        the active convolution entry) to a low‑sensitivity one,
      * short‑window smoothing (replace a segment by its average), and
      * random pairwise swaps.
   Temperature and maximal transferred fraction decay exponentially.

The best sequence discovered before the time budget expires is returned.
Trailing near‑zero entries are stripped for compactness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (real‑to‑real)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective value used by the optimiser (max‑convolution with normalisation)."""
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x_i≥0, Σx_i = s} (Duchi et al.)."""
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# 1. Projected sub‑gradient descent (exact max)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Reduce the maximal entry of the convolution by a projected sub‑gradient scheme.
    The gradient is taken w.r.t. the active convolution index.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    it = 0
    rng = np.random.default_rng()
    # step magnitude (will be scaled by gradient norm and iteration count)
    step_base = target_sum * 0.2

    while time.time() < deadline:
        conv = _conv_fft(a)
        k = int(conv.argmax())

        # ----------------- sub‑gradient of b_k -----------------
        grad = np.zeros_like(a)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        i_valid = idx[mask]
        j_valid = j[mask]
        eq = i_valid == j_valid
        grad[i_valid[eq]] = a[i_valid[eq]]
        neq = ~eq
        if np.any(neq):
            grad[i_valid[neq]] = 2.0 * a[j_valid[neq]]

        # ----------------- adaptive step -----------------
        norm_g = np.linalg.norm(grad) + 1e-12
        lr = step_base / (norm_g * np.sqrt(it + 1))
        a = a - lr * grad

        # projection onto the simplex (keeps Σa_i = target_sum)
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 500 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] subgrad it {it:5d} → score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# 2. Simulated‑annealing refinement
# ----------------------------------------------------------------------


def _simulated_annealing(
    a0: np.ndarray,
    deadline: float,
    start_temp: float = 0.5,
    end_temp: float = 1e-6,
    max_frac_start: float = 0.30,
    max_frac_end: float = 0.005,
    max_window: int = 40,
) -> tuple[np.ndarray, float]:
    """
    Refine a sequence using simulated annealing.
    Three move types are used:
        • donor→receiver mass transfer (≈70 % of moves),
        • local window smoothing,
        • random swap (≈10 %).
    Temperature and maximal transferred fraction decay exponentially.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _score(a)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)

    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # ----- active convolution index & its sensitivity profile -----
        k_max = int(conv.argmax())
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]                # exactly the gradient of b_k

        weight = a * sens                          # larger → more likely to donate

        move_type = None
        # -------------------------------------------------
        # 1) mass transfer (dominant move)
        # -------------------------------------------------
        if rng.random() < 0.70:
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight / total_w)
            else:
                donor = rng.integers(n)

            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            receiver = rng.choice(low_idx) if low_idx.size else rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        # -------------------------------------------------
        # 2) short‑window smoothing
        # -------------------------------------------------
        elif rng.random() < 0.90:
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                saved_seg = a[start:end].copy()
                avg = saved_seg.mean()
                a[start:end] = avg
                move_type = "smooth"
                saved_start, saved_end = start, end

        # -------------------------------------------------
        # 3) random swap
        # -------------------------------------------------
        else:
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # ----- evaluate new state (single FFT) -----
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = _score(a)

        # ----- Metropolis acceptance test -----
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA new best = {best_score:.6f}")
        else:
            # revert move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter
        if it % 4000 == 0 and it > 0:
            cur_sum = a.sum()
            jitter = rng.normal(scale=cur_sum * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, cur_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

    # 1) Warm‑start from the supplied global sequence (length 1500)
    # ------------------------------------------------------------------
    try_deadline = min(deadline, time.time() + 300.0)
    seq, score = _run_one_instance(1500, "global", try_deadline)
    if score < best_score:
        best_score = score
        best_seq = seq.copy()
        print(f"[{time.time():.1f}] global start → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Large random run (more degrees of freedom)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq, score = _run_one_instance(3000, "random", deadline)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] random 3000 → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Medium‑sized power‑law run (fallback)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq, score = _run_one_instance(2000, "power", deadline)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] power‑law 2000 → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Additional large random run if time permits
    # ------------------------------------------------------------------
    if time.time() + 10.0 < deadline:
        seq, score = _run_one_instance(4000, "random", deadline)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] random 4000 → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final cleanup – strip trailing ~zero entries
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.array([], dtype=float)

    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    print(f"[{time.time():.1f}] final best score={best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
