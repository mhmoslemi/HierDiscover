"""
Step‑function optimizer – enhanced Adam‑based smooth‑max descent +
aggressive max‑convolution balancing.

The evaluation metric is
    score = 2·n·max(conv(a,a)) / (sum(a)^2)
which is invariant under scaling of a.  After normalising sum(a)=√(2·n) the
score equals the peak of the self‑convolution, so we minimise that peak.

Key ingredients
----------------
* **Adam optimiser** on the smooth‑max (log‑sum‑exp) of the convolution.
  The softmax temperature β is annealed from 0.5 up to 4096, turning the
  smooth objective gradually into the true max.
* **Exact sub‑gradient descent** (max‑index) also equipped with Adam‑style
  momentums.
* **Aggressive mass‑balancing**: repeatedly move a modest fraction of the
  most offending coefficient (largest contribution to the current maximal
  convolution entry) to the coefficient with the smallest contribution,
  accepting the move only when the peak decreases.
* **Random restarts** with a rich family of initialisations (power‑law,
  exponential, Dirichlet, piece‑wise) and a final light simulated‑annealing
  polish.
* The whole search respects a user supplied wall‑clock budget (default
  1000 s).  Progress and intermediate scores are printed.

The routine returns a plain Python list of non‑negative numbers.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x|x_i≥0, Σx_i=s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1, dtype=float)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n−1) using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Peak of the self‑convolution (objective after ℓ¹‑normalisation)."""
    return float(_max_convolution_fft(a).max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) gradient – Phase 1
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) where
        g = 2·(a ⋆ p)   with   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = float(np.max(b))
    exps = np.exp(beta * (b - max_b))          # shift for stability
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # 3 n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _adam_smooth_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Adam‑style projected gradient descent on the smooth‑max (log‑sum‑exp)
    objective.  The temperature β is annealed to approach the true max.
    """
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule
    beta = 0.5
    beta_max = 4096.0
    beta_factor = 1.4

    step0 = target * 0.1      # base step magnitude
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        it += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # scale step by the (inverse) norm of the first‑moment estimate
        step = step0 / (np.linalg.norm(m_hat) + eps)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target)

        # occasional jitter to escape flat regions
        if it % 300 == 0:
            jitter = rng.normal(scale=target * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

        # anneal temperature
        if it % 250 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Exact max‑index sub‑gradient + Adam – Phase 2
# ----------------------------------------------------------------------


def _max_index_subgradient(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a,a)).  Returns (g, k) where
        g_i = 2·a_{k‑i}  if the index is inside the support, else 0,
    and k = argmax conv(a,a).
    """
    n = a.shape[0]
    conv = _max_convolution_fft(a)
    k = int(np.argmax(conv))

    g = np.zeros_like(a)
    if k < n:                         # left half of the convolution
        g[:k + 1] = 2.0 * a[k::-1]
    else:                             # right half
        d = k - (n - 1)               # shift amount (0 … n‑1)
        g[d:] = 2.0 * a[: n - d]
    return g, k


def _adam_subgradient_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Adam‑style projected sub‑gradient descent on the true max‑norm.
    """
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)

    beta1, beta2, eps = 0.8, 0.99, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _max_index_subgradient(a)

        it += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # diminishing step size (∝ 1/√t) multiplied by Adam normalisation
        step = 0.6 * target / (np.sqrt(it) * (np.linalg.norm(m_hat) + eps))
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target)

        # tiny jitter to avoid cycling
        if it % 200 == 0:
            jitter = rng.normal(scale=target * 5e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

    return a


# ----------------------------------------------------------------------
# Aggressive max‑convolution mass‑balancing
# ----------------------------------------------------------------------


def _balance_refine(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Greedy re‑allocation of mass from the index that contributes most to
    the current maximal convolution entry to the index that contributes
    the least.  The move is accepted only if the peak decreases.
    """
    a = a0.copy()
    n = a.size
    target = np.sum(a)                     # should already be √(2·n)
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _max_convolution_fft(a)
        k = int(np.argmax(conv))

        # contribution of each index to entry k
        idx = np.arange(n)
        other = k - idx
        mask = (other >= 0) & (other < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[other[mask]]

        i_high = int(np.argmax(contrib))
        i_low = int(np.argmin(contrib))

        if i_high == i_low:
            break

        # Move a modest fraction of the heavy mass
        delta = a[i_high] * 0.07                     # 7 % – more aggressive
        if delta < 1e-12:
            break

        a_candidate = a.copy()
        a_candidate[i_high] -= delta
        a_candidate[i_low] += delta
        a_candidate = np.maximum(a_candidate, 0.0)
        a_candidate = _project_to_simplex(a_candidate, target)

        if _score(a_candidate) < _score(a):
            a = a_candidate
        # else keep the current a

        # occasional global jitter
        if rng.random() < 0.01:
            jitter = rng.normal(scale=target * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

    return a

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative point on the simplex Σa = √(2·n).
    Several families are sampled to increase diversity.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                                 # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                               # pure exponential
        lam = rng.exponential(scale=0.02)
        a = np.exp(-lam * idx)

    elif family == 2:                               # power × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.02)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                               # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n))

    elif family == 4:                               # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                             np.exp(-lam2 * np.arange(b, n))])

    else:                                            # almost uniform with jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a

def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Very light SA that moves a random fraction of mass between two sites."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    cur_score = _score(a)
    best_score = cur_score

    T = cur_score * 0.02 + 1e-9
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-16:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        frac = rng.random() * 0.15
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta

        new_score = _score(a)
        dE = new_score - cur_score
        accept = dE <= 0.0 or (T > 0.0 and rng.random() < np.exp(-dE / T))

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
        else:
            a[j] -= delta
            a[i] += delta

        T *= cooling

    return best

                best_score = sc
                best_seq = a_ref3.copy()
                print(f"[{time.time():.1f}] mass‑balance refinement n={n} score={sc:.6f}")

    # ------------------------------------------------------------------
    # Final simulated‑annealing polishing (use whatever time is left)
    # ------------------------------------------------------------------
    if best_seq is not None:
        left = deadline - time.time()
        if left > 5.0:
            a_final = _simulated_annealing(best_seq, deadline)
            sc = _score(a_final)
            if sc < best_score:
                best_score = sc
                best_seq = a_final.copy()
                print(f"[{time.time():.1f}] final SA improvement: score={sc:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up trailing near‑zeros and report the official metric
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution of length 1000 (should never happen)
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
