```
def normalize_to_sum(v, s):
    sum_raw = v.sum()
    if sum_raw == 0:
        # fallback uniform
        v = np.full_like(v, s / v.size)
    else:
        v = v * (s / sum_raw)
    v = np.maximum(v, 0)
    return v
```

```
    # but need to ensure sum(exp(y)) = sum(a0) = s. Actually if we compute exp(log(a0)) = a0, sum = s. So yes. So a = s * a0 / s = a0. So y initialization good.
    # We'll store m, v for Adam.
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1=0.9
    beta2=0.999
    eps_adam=1e-8
    t = 0
    beta = 1.0

        # compute inner = sum_i g_a_i * a_i
        inner = np.dot(g_a, a)
        # gradient wrt y
        g_y = a * (g_a - inner / s)
        # Adam update
        t += 1
        m = beta1*m + (1 - beta1) * g_y
        v = beta2*v + (1 - beta2) * (g_y**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        # learning rate schedule maybe: lr = 0.1 * s / (np.sqrt(v_hat) + eps)
        lr = 0.1 * s / (np.sqrt(v_hat) + eps_adam)
        y = y - lr * m_hat
        # increase beta occasionally
        if t % 100 == 0 and beta < beta_max:
            beta = min(beta * 1.05, beta_max)
    # final a
    mx = np.max(y)
    exp_y = np.exp(y - mx)
    a = s * exp_y / exp_y.sum()
    return a
```

```
    s = np.sqrt(2.0 * n)
    # init y
    eps = 1e-12
    y = np.log(np.maximum(a0, eps))

    m = np.zeros_like(y)
    v = np.zeros_like(y)

    beta2 = 0.999

    beta = 1.0

        y_max = np.max(y)
        exp_y = np.exp(y - y_max)

        # gradient w.r.t y
        inner = np.dot(g_a, a)
```

```
def _simulated_annealing(a0, deadline):
    a = a0.copy()
    n = a.size
    s = a.sum()
    best = a.copy()
    best_score = _max_convolution_fft(best)
    # initial temperature
    T = best_score * 0.02 + 1e-8
    cooling = 0.9995
    rng = np.random.default_rng()
    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        # move fraction of mass from i to j
        frac = rng.random() * 0.15
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta
        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        accept = False
        if cur_score <= best_score:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-dE / T):
                accept = True
        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # revert
            a[j] -= delta
            a[i] += delta
        T *= cooling

        # occasionally global jitter to escape
        if rng.random() < 0.001:
            jitter = rng.normal(scale=s * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            # re‑normalize to sum s
            a_sum = a.sum()
            if a_sum > 0:
                a = a * (s / a_sum)
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
    return best
```

"""
Improved search for a non‑negative step‑function that minimises
``evaluate_sequence`` (i.e. the maximal entry of the self‑convolution).

The algorithm consists of three phases:

1. **Diverse random sampling** – generate many candidate shapes
   (power‑law, exponential, Gaussian‑type, Dirichlet) for several
   lengths.  Keep the best few (the “elite”) according to the true
   objective max(conv(a,a)).

2. **Adam‑optimisation** – each elite candidate is refined by minimising
   a smooth log‑sum‑exp surrogate of the maximum convolution entry.
   The variables are the logits of a soft‑max parametrisation, which
   automatically enforces the ℓ¹‑simplex constraint Σa = √(2 n).

3. **Simulated‑annealing fine‑tuning** – a short SA phase further reduces
   the true maximum by random mass‑transfer moves.

All phases respect the user‑supplied time budget (default 1000 s) and
never perform any I/O beyond ``print`` logging.  The final best sequence
is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) surrogate of max(b) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b) where
        b = a*a (full convolution)
        g = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a ⋆ p) ,  p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]

    # ----- convolution a*a -----
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # ----- soft‑max probabilities -----
    max_b = np.max(b)
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)

    # ----- gradient g = 2·(a ⋆ p) -----
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Simple helper: normalise a non‑negative vector to a given ℓ¹‑sum
# ----------------------------------------------------------------------


def _normalise_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    total = v.sum()
    if total <= 0.0:
        # fallback to uniform
        return np.full_like(v, target / v.size)
    w = v * (target / total)
    w = np.maximum(w, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimisation on the log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray, deadline: float, rng: np.random.RandomState) -> np.ndarray:
    """
    Adam optimisation on the smooth max of the convolution.
    Parameters are the logits of a soft‑max, which guarantees Σa = √(2 n)
    and a_i ≥ 0 at every step.
    """
    n = a0.size
    s = np.sqrt(2.0 * n)               # exact ℓ¹‑norm we enforce
    eps = 1e-12

    # initialise logits y such that softmax(y) reproduces a0
    y = np.log(np.maximum(a0, eps))

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    t = 0

    beta = 1.0
    beta_max = 1e5

    while time.time() < deadline:
        # ---- a = s·softmax(y) ----
        y_max = np.max(y)
        exp_y = np.exp(y - y_max)
        a = s * exp_y / np.sum(exp_y)

        # ---- gradient of the smooth max (w.r.t. a) ----
        g_a, _ = _gradient_softmax(a, beta)

        # ---- chain rule: ∂f/∂y = diag(a)·(g_a – (a·g_a)/s) ----
        inner = np.dot(g_a, a)                # scalar = a·g_a
        g_y = a * (g_a - inner / s)

        # ---- Adam update ----
        t += 1
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = 0.1 * s / (np.sqrt(v_hat) + eps_adam)
        y -= lr * m_hat

        # ---- anneal smoothing parameter β ----
        if t % 50 == 0 and beta < beta_max:
            beta = min(beta * 1.04, beta_max)

    # final a
    y_max = np.max(y)
    exp_y = np.exp(y - y_max)
    a = s * exp_y / np.sum(exp_y)
    return a


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Random mass‑transfer SA with multiplicative cooling."""
    a = a0.copy()
    n = a.size
    s = a.sum()
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # temperature initialised from current objective
    T = best_score * 0.02 + 1e-8
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        # move up to 15 % of mass from i to j
        frac = rng.random() * 0.15
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        accept = False
        if cur_score <= best_score:
            accept = True
        elif rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # revert the move
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # occasional global jitter to escape stagnation
        if rng.random() < 0.001:
            jitter = rng.normal(scale=s * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 0.0:
                a = a * (s / a_sum)
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
    return best


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the target
    ``evaluate_sequence`` objective.  The routine respects the allotted
    time budget, explores a wide variety of parametric families, refines
    elite candidates with Adam optimisation, and finally applies a short
    simulated‑annealing phase.
    """
    # reproducible RNG
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq = None
    best_obj = np.inf
    elite = []                 # (objective, array)
    elite_max = 20

    # ------------------------------------------------------------------
    # 1. Diverse random sampling (≈ 50 % of the budget)
    # ------------------------------------------------------------------
    lengths = [1200, 1500, 1800, 2100, 2400, 3000, 3500, 4000, 4500, 5000]
    phase1_dead = start_time + 0.5 * budget_s - 0.5

    while time.time() < phase1_dead:
        n = int(rng.choice(lengths))
        target_sum = np.sqrt(2.0 * n)

        # pick a random family
        fam = rng.integers(0, 4)   # 0 power‑law, 1 exponential, 2 Gaussian‑type, 3 Dirichlet
        if fam == 0:
            # power‑law a_i = (i+shift)^{-p}
            p = rng.uniform(0.3, 1.2)
            shift = rng.uniform(0.1, 10.0)
            raw = (np.arange(n, dtype=float) + shift) ** (-p)
            a = _normalise_to_sum(raw, target_sum)
        elif fam == 1:
            # exponential decay
            lam = 10 ** rng.uniform(-5, 0)          # λ ∈ [1e‑5, 1]
            raw = np.exp(-lam * np.arange(n, dtype=float))
            a = _normalise_to_sum(raw, target_sum)
        elif fam == 2:
            # Gaussian‑like (centered at 0, decreasing)
            sigma = rng.uniform(0.1 * n, 0.5 * n)
            idx = np.arange(n, dtype=float)
            raw = np.exp(-0.5 * (idx / sigma) ** 2)
            a = _normalise_to_sum(raw, target_sum)
        else:
            # random Dirichlet / gamma draw
            raw = rng.gamma(shape=1.0, scale=1.0, size=n)
            a = _normalise_to_sum(raw, target_sum)

        obj = _max_convolution_fft(a)

        if obj < best_obj:
            best_obj = obj
            best_seq = a.copy()
            print(f"[{time.time():.1f}] New global best (sampling) n={n} obj={best_obj:.6f}")

        # maintain elite list
        if len(elite) < elite_max or obj < elite[-1][0]:
            elite.append((obj, a.copy()))
            elite.sort(key=lambda x: x[0])
            elite = elite[:elite_max]

    # also seed elite with the known construction if available
    try:
        known = np.array(height_sequence_1, dtype=float)
        if known.size > 0:
            n_known = known.size
            known = _normalise_to_sum(known, np.sqrt(2.0 * n_known))
            obj_known = _max_convolution_fft(known)
            if obj_known < best_obj:
                best_obj = obj_known
                best_seq = known.copy()
                print(f"[{time.time():.1f}] New global best (known) n={n_known} obj={best_obj:.6f}")
            elite.append((obj_known, known.copy()))
            elite.sort(key=lambda x: x[0])
            elite = elite[:elite_max]
    except NameError:
        pass  # ``height_sequence_1`` not defined – ignore

    # ------------------------------------------------------------------
    # 2. Refine elite candidates with Adam + SA
    # ------------------------------------------------------------------
    for idx, (obj_start, a0) in enumerate(elite):
        if time.time() >= deadline:
            break

        n = a0.size
        # allocate up to 30 s for Adam; the remaining time will be used by SA
        sub_dead = min(deadline, time.time() + 30.0)
        print(f"[{time.time():.1f}] Refinement {idx + 1}/{len(elite)} – n={n}")

        a_refined = _adam_optimize(a0, sub_dead, rng)

        # SA fine‑tuning on whatever time is left
        a_final = _simulated_annealing(a_refined, deadline)

        obj_final = _max_convolution_fft(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL BEST after refinement obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 3. Final sanity check & output
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback to a uniform vector (should never happen)
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_obj = _max_convolution_fft(best_seq)

    # Trim negligible trailing entries for neatness
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    try:
        eval_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {eval_val:.6f}")
    except Exception as e:
        print(f"[{time.time():.1f}] evaluate_sequence raised: {e}")

    print(f"[{time.time():.1f}] Finished. Best objective = {best_obj:.6f}  length = {best_seq.size}")
    return best_seq.tolist()
