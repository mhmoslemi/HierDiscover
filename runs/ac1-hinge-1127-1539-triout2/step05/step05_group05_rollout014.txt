"""
Improved search for a low‑maximum self‑convolution step function.

The key idea is to start from the *arcsine* (Beta(½,½)) distribution,
which is the unique continuous distribution whose sum of two independent
samples is exactly uniform on a bounded interval.  Discretising this
distribution on a sufficiently long grid yields a probability vector
p whose discrete self‑convolution is almost flat, i.e. the maximum
coefficient is close to the average (the theoretical lower bound 1).

The algorithm proceeds as follows:

1.  Choose a length `n` (default 4000 – large enough to give a fine
    discretisation while staying cheap to FFT‑convolve).

2.  Build a raw arcsine shape `a_i ∝ 1/√(x_i(1‑x_i))` with
    `x_i = (i+0.5)/n`.  Small epsilons avoid the singularities at the
    ends.  The vector is scaled to the conventional simplex
    `Σa_i = √(2 n)`.  This already yields an objective close to 1.

3.  Refine the candidate with a few mirror‑descent steps on the
    log‑sum‑exp smooth approximation of `max(conv)`.  The learning rate
    decays with the iteration counter and the temperature parameter β is
    multiplied by 1.2 every few iterations until a large cap (1e8).

4.  Perform a brief targeted simulated‑annealing (SA) phase that moves
    a fraction of mass from indices that contribute heavily to the
    current maximal convolution entry to random destinations.  The SA
    temperature decays exponentially.

All heavy work uses FFT‑based convolution (`O(n log n)`) and the routine
obeys the global 1000 s budget.  The final sequence is trimmed of trailing
zeros and returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Return the maximum entry of the full linear convolution a*a."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used only for initialisation
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) of the convolution and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns gradient of the smoothed max (log‑sum‑exp) of a*a.

    Output:
        g  – gradient vector (size n)
        b  – full convolution a*a (size 2·n‑1)
    """
    n = a.shape[0]

    # convolution b = a * a
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities over the entries of b
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))          # stable exponentiation
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)   (⋆ = convolution, a is reversed for correct indexing)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1               # = 3·n‑2
    size2 = 1 << ((conv_len).bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]         # central n entries
    return g, b


# ----------------------------------------------------------------------
# Mirror‑descent optimisation (exponentiated gradient) on the smoothed max
# ----------------------------------------------------------------------


def _mirror_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Exponentiated‑gradient descent on the log‑sum‑exp surrogate.
    The iterate stays on the ℓ¹‑simplex automatically.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5            # start temperature
    beta_max = 1e8
    eta0 = 0.8            # base step size
    step = 0
    eps = 1e-12

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # element‑wise multiplicative update (mirror descent)
        eta = eta0 / np.sqrt(step + 1.0)
        a = a * np.exp(-eta * g)
        a = np.maximum(a, eps)               # keep positivity

        # renormalise to the ℓ¹‑simplex
        s = a.sum()
        if s > 0:
            a *= (target_sum / s)
        else:
            a.fill(target_sum / n)

        # anneal β relatively fast – we want a tight approximation of max
        if step % 100 == 0 and beta < beta_max:
            beta = min(beta * 1.2, beta_max)

        step += 1
    return a


# ----------------------------------------------------------------------
# Targeted simulated‑annealing refinement
# ----------------------------------------------------------------------


def _targeted_simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simple SA that moves a fraction of mass from an index contributing
    heavily to the current maximal convolution entry toward a random
    destination.  Temperature decays exponentially.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_fft(best)

    T = best_score * 0.02 + 1e-9       # initial temperature
    cooling = 0.9994
    rng = np.random.default_rng()
    max_transfer = 0.25               # up to 25 % of the selected mass

    while time.time() < deadline:
        b = _full_convolution_fft(a)
        k_max = int(np.argmax(b))

        # indices i that actually appear in the term that yields b[k_max]
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idx_range = np.arange(lo, hi + 1)
        contrib = a[idx_range] * a[k_max - idx_range]
        weight_sum = contrib.sum()

        if weight_sum > 0:
            probs = contrib / weight_sum
            i = rng.choice(idx_range, p=probs)
        else:
            i = rng.integers(0, n)

        # pick a destination j ≠ i uniformly
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        delta = a[i] * rng.random() * max_transfer
        if delta <= 0.0:
            continue
        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score

        accept = False
        if cur_score <= best_score:
            accept = True
        elif T > 0.0 and rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # revert
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # occasional global jitter to escape plateaus
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0:
                a *= (np.sqrt(2.0 * n) / s)
    return best


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function (list of floats) that minimises
    ``evaluate_sequence`` under the supplied time budget.

    Strategy:
        * Build an arcsine‑shaped initial candidate (Beta(½,½)).
        * Optionally add a few diverse random seeds.
        * Refine the best candidate with mirror descent (log‑sum‑exp surrogate).
        * Finish with a short targeted simulated‑annealing pass.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose a length.  Larger n gives a finer discretisation of the arcsine
    # density and hence a flatter discrete sum distribution.
    # ------------------------------------------------------------------
    n = 4000
    if deadline - time.time() < 5.0:
        # Extremely low remaining budget – fall back to a small n
        n = 800

    # ------------------------------------------------------------------
    # 1. Initialise a few candidates; the arcsine (Beta(0.5,0.5)) shape is
    #    the principled starting point.
    # ------------------------------------------------------------------
    candidates = []

    # (a) Arcsine (Beta(0.5,0.5))
    idx = np.arange(n, dtype=float)
    eps = 1e-12
    x = (idx + 0.5) / n               # points in (0,1)
    raw = 1.0 / np.sqrt(np.clip(x * (1.0 - x), eps, None))
    a_arc = raw / raw.sum() * np.sqrt(2.0 * n)
    candidates.append(a_arc)

    # (b) Dirichlet uniform simplex (for diversity)
    a_dir = rng.dirichlet(alpha=np.ones(n)) * np.sqrt(2.0 * n)
    candidates.append(a_dir)

    # (c) Slightly perturbed Beta(0.5,0.5) – random shift to break symmetry
    alpha = 0.5 + 0.1 * rng.random()
    beta = 0.5 + 0.1 * rng.random()
    raw_beta = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    a_beta = raw_beta / raw_beta.sum() * np.sqrt(2.0 * n)
    candidates.append(a_beta)

    # (d) Edge‑heavy power‑law (another useful family)
    p = rng.uniform(0.15, 1.0)
    a_edge = ((idx + 1.0) ** (-p) + (n - idx) ** (-p))
    a_edge = a_edge / a_edge.sum() * np.sqrt(2.0 * n)
    candidates.append(a_edge)

    best_seq = None
    best_score = np.inf

    for a in candidates:
        score = _max_convolution_fft(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] initial candidate score {score:.6f}")

    # ------------------------------------------------------------------
    # 2. Mirror‑descent refinement (use roughly half of the remaining budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        md_deadline = time.time() + remaining * 0.45
        a_md = _mirror_descent_opt(best_seq, md_deadline)
        md_score = _max_convolution_fft(a_md)
        if md_score < best_score:
            best_seq, best_score = a_md, md_score
            print(f"[{time.time():.1f}] after mirror descent score {md_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Targeted simulated‑annealing refinement (use whatever time left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_sa = _targeted_simulated_annealing(best_seq, deadline)
        sa_score = _max_convolution_fft(a_sa)
        if sa_score < best_score:
            best_seq, best_score = a_sa, sa_score
            print(f"[{time.time():.1f}] after SA score {sa_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries (optional cosmetic step)
    # ------------------------------------------------------------------
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator (prints the true
    # objective that will be used for scoring)
    # ------------------------------------------------------------------
    try:
        ev = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence = {ev:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective (max‑conv) ≈ {best_score:.6f}   length = {best_seq.size}")

    return best_seq.tolist()
