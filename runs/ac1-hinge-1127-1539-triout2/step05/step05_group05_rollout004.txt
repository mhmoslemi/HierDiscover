"""
Propose a low‑max‑convolution step‑function.

Algorithm
---------
1. **Initial pool** – if a global `height_sequence_1` exists it is used as a
   starting point.  Otherwise a variety of random families (power‑law,
   exponential, Dirichlet, symmetric edge‑heavy, Beta‑shaped and a sine‑wave
   shape) are projected onto the ℓ¹‑simplex Σ a_i = √(2·n).

2. **Many short mirror‑descent runs** – the smoothed‑max objective
   (log‑sum‑exp of the self‑convolution) is minimised with exponentiated
   gradient updates (mirror descent).  A large number of independent runs,
   each lasting only a few seconds, are performed for several sequence
   lengths (800 … 8000).  The best sequence seen so far is retained.

3. **Targeted simulated annealing** – the current best sequence is refined
   by repeatedly moving a fraction of mass from an index that contributes
   heavily to the current maximal convolution entry to a random destination.
   The temperature decays multiplicatively; occasional global jitters help
   escape deep minima.

4. **Final polish** – a short mirror‑descent run (≈2 s) on the best
   candidate after SA is performed.

All heavy work uses FFT‑based convolutions (O(n log n)) and the routine
respects a user‑supplied time budget (default 1000 s).  The returned
sequence is a plain list of non‑negative floats; `evaluate_sequence` can be
called afterwards to obtain the official score.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex Σ x_i = s, x_i ≥ 0."""
    if s <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) of the convolution and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Return (g, b) where
        b = a * a (full convolution, length 2·n‑1)
        g = ∇_a  ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a ⋆ p)   with   p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]

    # ----- full convolution a*a ----------------------------------------
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ----- soft‑max probabilities ---------------------------------------
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))          # stability
    p = exps / np.sum(exps)

    # ----- gradient g = 2·(a ⋆ p) --------------------------------------
    # reversing a to obtain convolution a_rev * p
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1               # = 3·n‑2
    size2 = 1 << ((conv_len).bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]        # central n entries
    return g, b


# ----------------------------------------------------------------------
# Mirror‑descent optimiser (exponentiated gradient) on the smoothed max
# ----------------------------------------------------------------------


def _mirror_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Exponentiated‑gradient descent on the log‑sum‑exp surrogate.
    Keeps the iterate on the ℓ¹‑simplex automatically.
    Returns the *final* iterate (not the best seen during the run).
    """
    a = a0.copy()
    n = a.size
    target_sum = math.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 1e5
    eta0 = 0.5
    step = 0
    eps = 1e-12

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # exponentiated‑gradient update
        eta = eta0 / math.sqrt(step + 1.0)
        a = a * np.exp(-eta * g)
        a = np.maximum(a, eps)                     # avoid exact zeros

        # renormalise to the ℓ¹‑simplex
        s = np.sum(a)
        if s > 0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # anneal β slowly
        if step % 100 == 0 and beta < beta_max:
            beta = min(beta * 1.05, beta_max)

        step += 1

    return a


# ----------------------------------------------------------------------
# Targeted simulated‑annealing refinement
# ----------------------------------------------------------------------


def _targeted_simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    SA that repeatedly moves a fraction of mass from an index that
    heavily contributes to the current maximal convolution entry to a
    random destination.  Temperature decays multiplicatively.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # initial temperature proportional to current objective
    T = best_score * 0.02 + 1e-8
    cooling = 0.9995
    rng = np.random.default_rng()
    max_transfer = 0.25           # up‑to 25 % of selected mass

    while time.time() < deadline:
        # current convolution and its maximal index
        b = _full_convolution_fft(a)
        k_max = int(np.argmax(b))

        # indices that affect b[k_max]
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idx_range = np.arange(lo, hi + 1)
        contrib = a[idx_range] * a[k_max - idx_range]
        weight_sum = np.sum(contrib)

        if weight_sum > 0:
            probs = contrib / weight_sum
            i = rng.choice(idx_range, p=probs)
        else:
            i = rng.integers(0, n)

        # pick a destination j ≠ i uniformly
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        # transfer a random fraction of the mass at i to j
        delta = a[i] * rng.random() * max_transfer
        if delta <= 0.0:
            continue
        a[i] -= delta
        a[j] += delta

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score

        accept = False
        if cur_score <= best_score:
            accept = True
        elif T > 0.0 and rng.random() < math.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA improvement: {best_score:.6f}")
        else:
            # revert the move
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # occasional global jitter to help escape deep minima
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a_sum = np.sum(a)
            if a_sum > 0:
                a = a * (math.sqrt(2.0 * n) / a_sum)
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA jitter improvement: {best_score:.6f}")

    return best


# ----------------------------------------------------------------------
# Random initialisers (diverse families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative sequence on the ℓ¹‑simplex
    Σ a_i = √(2 n).  Several families are mixed to increase diversity,
    including a sine‑wave shape that often yields smooth convolutions.
    """
    target_sum = math.sqrt(2.0 * n)
    kind = rng.integers(0, 7)
    idx = np.arange(n, dtype=float)

    if kind == 0:                     # pure power‑law (decreasing)
        p = rng.uniform(0.15, 1.0)
        a = (idx + 1.0) ** (-p)

    elif kind == 1:                 # exponential decay from the left edge
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif kind == 2:                 # mixed power‑law + exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif kind == 3:                 # Dirichlet (uniform simplex)
        a = rng.dirichlet(alpha=np.ones(n))

    elif kind == 4:                 # symmetric edge‑heavy
        p = rng.uniform(0.2, 1.2)
        left = (idx + 1.0) ** (-p)
        right = (n - idx) ** (-p)
        a = left + right

    elif kind == 5:                 # Beta‑shaped
        alpha = rng.uniform(0.2, 2.0)
        beta = rng.uniform(0.2, 2.0)
        x = (idx + 0.5) / n
        a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))

    else:                           # sine‑wave shape (single bump)
        a = np.sin(np.pi * (idx + 0.5) / n)
        a = np.maximum(a, 0.0)

    # small Gaussian perturbation and projection onto the simplex
    a += rng.normal(scale=target_sum * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target_sum)
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative sequence) that minimises
    `evaluate_sequence` under the supplied time budget.  The routine
    performs many short mirror‑descent runs on a variety of lengths,
    followed by a targeted simulated‑annealing refinement and a final
    polishing step.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.3   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 0.  Use a global construction if available (good warm start)
    # ------------------------------------------------------------------
    global_seq = globals().get("height_sequence_1")
    if isinstance(global_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(global_seq, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, math.sqrt(2.0 * n0))
        sc = _max_convolution_fft(a0)
        best_seq, best_score = a0.copy(), sc
        print(f"[{time.time():.1f}] Warm‑start from height_sequence_1 "
              f"(n={n0})  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1.  Many short mirror‑descent runs with diverse lengths
    # ------------------------------------------------------------------
    lengths = [800, 1200, 2000, 3000, 4000, 5000, 6000, 8000]

    # We keep launching short runs until only a few seconds are left.
    while time.time() < deadline - 6.0:
        n = int(rng.choice(lengths))
        a_start = _random_initial_sequence(n, rng)

        # allow each run a small time slice (≈4 s or the remaining fraction)
        slice_deadline = time.time() + min(4.0, (deadline - time.time()) * 0.25)
        a_opt = _mirror_descent_opt(a_start, slice_deadline)
        score_opt = _max_convolution_fft(a_opt)

        if score_opt < best_score:
            best_seq, best_score = a_opt.copy(), score_opt
            print(f"[{time.time():.1f}] New best after MD (n={n})  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Targeted simulated annealing on the current best
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        a_sa = _targeted_simulated_annealing(best_seq, deadline)
        score_sa = _max_convolution_fft(a_sa)
        if score_sa < best_score:
            best_seq, best_score = a_sa.copy(), score_sa
            print(f"[{time.time():.1f}] New best after SA  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Final quick polish (mirror descent ≈2 s)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 0.2:
        polish_deadline = time.time() + 2.0
        a_polished = _mirror_descent_opt(best_seq, polish_deadline)
        score_polish = _max_convolution_fft(a_polished)
        if score_polish < best_score:
            best_seq, best_score = a_polished.copy(), score_polish
            print(f"[{time.time():.1f}] Final polish improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Trim trailing near‑zeros and return the plain list
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform distribution of length 2000
        n = 2000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n)
        best_score = _max_convolution_fft(best_seq)

    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Optional sanity check with the official evaluator
    try:
        ev = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence = {ev:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best max‑conv = {best_score:.6f} "
          f"length = {len(best_seq)}")
    return best_seq.tolist()
