def subgradient_step(a, lr):
    c = convolution_fft(a)
    max_val = c.max()
    k = int(np.argmax(c))
    grad = np.zeros_like(a)
    lo = max(0, k - (len(a) - 1))
    hi = min(k, len(a)-1)
    # indices i from lo to hi inclusive
    idx = np.arange(lo, hi+1)
    grad[idx] = 2.0 * a[k - idx]
    # take step
    a_new = a - lr * grad
    # clip to non-negative and project to simplex
    a_new = np.maximum(a_new, 0.0)
    a_new = project_to_simplex(a_new, np.sqrt(2*len(a)))
    return a_new, max_val

def subgradient_descent(a0, deadline):
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2*n)
    a = project_to_simplex(a, target_sum)
    best_a = a.copy()
    best_val = _score(a)

    it = 0
    # schedule: step_base = target_sum * 0.1 maybe
    step0 = target_sum * 0.1
    # loop
    while time.time() < deadline:
        # compute conv
        conv = _conv_fft(a)
        max_val = conv.max()
        # Determine gradient
        k_max = np.argmax(conv)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        # compute gradient
        # vectorized:
        idx = np.arange(lo, hi+1)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k_max - idx]
        # step size adapt: lr = step0 / (np.linalg.norm(grad) + 1e-12) * (1.0 / (1 + it*0.001))
        norm = np.linalg.norm(grad)
        if norm > 0:
            lr = step0 / (norm * (1.0 + it * 1e-4))
            a = a - lr * grad
        else:
            # maybe jitter small random to avoid stagnation
            a += np.random.default_rng().normal(scale=target_sum*0.001, size=n)
        # ensure non-neg and project
        a = np.maximum(a, 0.0)
        a = project_to_simplex(a, target_sum)
        # evaluate
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it} score={best_val:.6f}")
        it += 1

    return best_a, best_val

def propose_candidate(budget_s=1000):
    start = time.time()
    deadline = start + budget_s - 1

    best_seq = None
    best_val = float('inf')
    cur_seq = None

    scales = [(1500, 0.15), (3000, 0.25), (6000, 0.40), (12000, 0.20)]

    for n, fraction in scales:
        now = time.time()
        if now >= deadline - 5:
            break
        remaining = deadline - now
        stage_deadline = now + remaining * fraction

        # initial sequence
        if cur_seq is not None:
            init = _resize_sequence(cur_seq, n)
        else:
            # generate best from candidate constructors
            init_candidates = []
            # power laws with exponent list
            for p in [0.4,0.55,0.7,1.0]:
                init_candidates.append(_power_law_sequence(n, p, np.sqrt(2*n)))
            # sin shape
            sin_seq = np.sin(np.pi * (np.arange(1, n+1) / (n+1)))
            sin_seq = sin_seq / sin_seq.sum() * np.sqrt(2*n)
            init_candidates.append(sin_seq)
            # exponential
            for rate in [0.001,0.005,0.01,0.02]:
                exp_seq = np.exp(-rate * np.arange(n))
                exp_seq = exp_seq / exp_seq.sum() * np.sqrt(2*n)
                init_candidates.append(exp_seq)

            # pick best based on quick score
            scores = [_score(seq) for seq in init_candidates]
            init = init_candidates[int(np.argmin(scores))]

        # run subgradient descent for a portion of stage time
        time_remaining = stage_deadline - time.time()
        if time_remaining <= 0:
            break
        # allocate 40% of this stage time to subgradient, remainder to SA
        sub_deadline = time.time() + max(5.0, time_remaining * 0.4)
        seq_sg, val_sg = _subgradient_descent(init, sub_deadline)
        # allocate rest to SA

        else:
            seq_sa, val_sa = seq_sg, val_sg

        # decide best for this stage
        if val_sa < best_val:
            best_val = val_sa
            best_seq = seq_sa.copy()
            print(f"[{time.time():.2f}] Scale n={n} best score={best_val:.6f}")

        cur_seq = seq_sa

    # trim zeros
    if best_seq is None:
        best_seq = np.zeros(1)
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]

The benchmark reduces to minimizing the peak of the auto‑convolution
`c = a * a` under the constraint  

```
a_i ≥ 0 ,   Σ a_i = √(2·n)                (simplex)
```

because the objective returned by `evaluate_sequence` is exactly the
maximum entry of `c` when the ℓ¹‑norm is fixed.
The algorithm therefore works directly on the simplex and tries to
flatten `c`.

* **Initialisation** – several simple families (power‑law, sine,
  exponential) are generated, projected onto the simplex and the one
  with the smallest peak is taken as start.
* **Sub‑gradient descent** – at each iteration the convolution is
  computed (FFT), all entries that are within a tiny fraction of the
  current maximum are selected, and the (sub‑)gradient of the maximum
  w.r.t. `a` is obtained by a second convolution.  A step opposite to
  the gradient is taken, followed by projection onto the simplex.
  The step size is scaled by the gradient norm and slowly decays.
* **Simulated annealing refinement** – after the deterministic phase a
  classic SA routine (mass transfer, local smoothing, swaps) is run
  using the same FFT‑based convolutions.  Temperature and allowed
  transfer fraction decay exponentially.
* **Multi‑scale schedule** – the procedure is executed on increasing
  lengths (≈1500 → 3000 → 6000 → 12000).  The best solution of a scale is
  interpolated and used to warm‑start the next one.
* **Time management** – the total budget (default 1000 s) is split
  among the scales; each scale spends ~40 % of its time on sub‑gradient
  descent and the remainder on SA.

The whole optimizer runs in under the allotted time and returns the
best step‑function found.

```python
"""
Propose a non‑negative step‑function (height sequence) with a low
auto‑convolution peak.  The algorithm works on several scales.
For each length n it

1. builds several simple initial profiles (power‑law, sine, exponential)
   and keeps the one with the smallest convolution peak;
2. performs a sub‑gradient descent on the true max‑convolution (all
   near‑maximum entries are used, gradient obtained via a second FFT);
   after each step the vector is projected onto the ℓ¹‑simplex
   Σa_i = √(2·n);
3. refines the result with a simulated‑annealing stage (mass transfer,
   smoothing, swaps);
4. the best solution of the current scale seeds the next, larger scale.

The routine respects the 1000 s time budget and finally returns the
best sequence as a list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto {x | x_i ≥ 0, Σ x_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (FFT based)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective: 2·n·max(a*a) / (sum(a)²)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Sequence generators (used for initialisation)
# ----------------------------------------------------------------------
def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    a = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1)))
    a *= target_sum / a.sum()
    return a


def _exp_sequence(n: int, rate: float, target_sum: float) -> np.ndarray:
    a = np.exp(-rate * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(new_x, old_x, seq)
    return new_seq.astype(np.float64)


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, deadline: float,
                         eps: float = 1e-4) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent.
    eps defines the relative tolerance for the “active set’’:
    all convolution entries ≥ (1‑eps)·max are treated as active.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    step0 = target_sum * 0.2   # base step magnitude
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        # active mask
        mask = (conv >= max_val * (1.0 - eps)).astype(np.float64)

        # gradient = 2 * (mask * a_rev) convolved, slice to length n
        a_rev = a[::-1]
        size = 1 << ((len(mask) + len(a_rev) - 1).bit_length())
        mask_fft = np.fft.rfft(mask, n=size)
        a_rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(mask_fft * a_rev_fft, n=size)
        start = len(mask) - 1
        grad = 2.0 * grad_full[start:start + n]

        norm = np.linalg.norm(grad)
        if norm > 0:
            # diminishing step size
            lr = step0 / (norm * (1.0 + it * 1e-4))
            a -= lr * grad
        else:
            # pure jitter when gradient vanishes
            a += rng.normal(scale=target_sum * 0.001, size=n)

        # keep feasible
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGD it={it:5d} score={best_val:.6f}")

        # occasional random jitter to escape plateaus
        if it % 500 == 0 and it > 0:
            a += rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing (largely the same as before, with minor tweaks)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray, deadline: float,
                         start_temp: float = 1.0,
                         end_temp: float = 1e-4,
                         max_frac_start: float = 0.30,
                         max_frac_end: float = 0.01,
                         max_window: int = 50) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.size
    cur_score = _score(a)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-9)
    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # index of the current maximum convolution entry
        k_max = int(np.argmax(conv))

        # sensitivities for a (derivative of the current max entry)
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]

        weight = a * sens
        move_type = None

        # ------------------- mass transfer -------------------
        if rng.random() < 0.70:
            total_w = weight.sum()
            donor = rng.choice(n, p=weight / total_w) if total_w > 0 else rng.integers(n)

            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            receiver = rng.choice(low_idx) if low_idx.size else rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        # ------------------- local smoothing -----------------
        elif rng.random() < 0.90:
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                old_seg = a[start:end].copy()
                avg = old_seg.mean()
                a[start:end] = avg
                move_type = "smooth"
                saved_seg = old_seg
                saved_start, saved_end = start, end

        # ------------------- swap ---------------------------
        else:
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # new convolution & score
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = _score(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it:5d} new best={best_score:.6f}")
        else:
            # revert rejected move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter
        if it % 8000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            cur_sum = a.sum()
            if cur_sum > 0:
                a = a * (np.sqrt(2.0 * n) / cur_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a height sequence with a low auto‑convolution peak.
    The routine runs on several scales (≈1500 → 3000 → 6000 → 12000),
    each scale consisting of a sub‑gradient descent phase followed by
    simulated‑annealing refinement.  The best sequence found is returned.
    """
    start = time.time()
    deadline = start + budget_s - 1.0      # safety margin

    best_seq = None
    best_score = float("inf")
    cur_seq = None

    # (length, fraction of remaining time)
    scales = [(1500, 0.20), (3000, 0.30), (6000, 0.30), (12000, 0.20)]

    for n, frac in scales:
        now = time.time()
        if now >= deadline - 5.0:
            break
        remaining = deadline - now
        stage_deadline = now + remaining * frac

        # ----- initialise -------------------------------------------------
        if cur_seq is not None:
            init = _resize_sequence(cur_seq, n)
        else:
            # generate a few candidates and pick the best quickly
            target = np.sqrt(2.0 * n)
            candidates = [
                _power_law_sequence(n, p, target) for p in (0.4, 0.55, 0.7, 1.0)
            ]
            candidates.append(_sin_sequence(n, target))
            for rate in (0.001, 0.005, 0.01, 0.02):
                candidates.append(_exp_sequence(n, rate, target))
            scores = [_score(c) for c in candidates]
            init = candidates[int(np.argmin(scores))]

        # ----- sub‑gradient descent (≈40 % of stage time) ----------------
        time_left = stage_deadline - time.time()
        if time_left <= 0:
            break
        sg_deadline = time.time() + max(5.0, time_left * 0.40)
        seq_sg, val_sg = _subgradient_descent(init, sg_deadline)

        # ----- simulated annealing for the rest -------------------------
        if time.time() < stage_deadline:
            seq_sa, val_sa = _simulated_annealing(seq_sg, stage_deadline)
        else:
            seq_sa, val_sa = seq_sg, val_sg

        # ----- keep global best -----------------------------------------
        if val_sa < best_score:
            best_score = val_sa
            best_seq = seq_sa.copy()
            print(f"[{time.time():.2f}] Scale n={n} improved best score={best_score:.6f}")

        cur_seq = seq_sa  # warm‑start next scale

    # ----- tidy up -------------------------------------------------------
    if best_seq is None:
        best_seq = np.zeros(1)
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final evaluation (allows the platform to see the exact score)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Final evaluation failed:", exc)

    return final_seq
```
