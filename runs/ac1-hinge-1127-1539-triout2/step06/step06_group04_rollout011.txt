    j = rng.choice(n, p=prob_dst)

    if max_delta < 1e-16:

    b_new = b.copy()

    # subtract 2*delta*a shifted by i
    start_i = i
    end_i = i + n

"""
Hybrid projected‐gradient descent + multi‑peak hill‑climbing.

*  Gradient descent minimises a log‑sum‑exp smooth approximation of
   max(a∗a).  The same annealing and momentum scheme as before is used,
   but the initialisation pool is enlarged (Beta, Gaussian, mixture,
   cosine families) and a tiny symmetrisation step is added after each
   projection.

*  Hill climbing moves a small amount of mass from indices that
   contribute most to the *set* of current maximal convolution entries
   (all peaks within the top‑few values) to indices that contribute
   least.  The update of the full convolution is performed analytically,
   so each move is O(n).

*  A length schedule (2000 → 5000 → 8000 → 12000 → 18000 → 25000 → 32000)
   gradually enlarges the search space; each new length is seeded by a
   linear up‑sampling of the best solution so far.
*  The remaining budget after the schedule is spent on a final polishing
   hill‑climb.

The routine respects a wall‑clock budget (default 1000 s) and returns the
best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a∗a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a∗a (length 2n−1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Duchi‑et‑al. (2008) algorithm.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce a[i] = a[-1-i] (keeps the ℓ¹‑norm unchanged)."""
    return (a + a[::-1]) * 0.5


# ----------------------------------------------------------------------
# Random initialisers (extended family set)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 10)        # 0‑9 (new families added)

    idx = np.arange(n, dtype=float)

    if family == 0:                     # pure power‑law
        p = rng.uniform(0.10, 1.2)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.018)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.12, 1.0)
        lam = rng.exponential(scale=0.018)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        cut = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(cut)),
             np.exp(-lam2 * np.arange(cut, n))]
        )

    elif family == 5:                   # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    elif family == 6:                   # Beta‑distribution shape
        alpha = rng.uniform(0.2, 5.0)
        beta = rng.uniform(0.2, 5.0)
        a = rng.beta(alpha, beta, size=n)

    elif family == 7:                   # single Gaussian centred in the middle
        centre = (n - 1) / 2.0
        sigma = rng.exponential(scale=0.2 * n) + 0.05 * n
        a = np.exp(-((idx - centre) ** 2) / (2.0 * sigma ** 2))

    elif family == 8:                   # mixture of two Gaussians (ends)
        centre1 = rng.uniform(0.0, 0.3) * n
        centre2 = rng.uniform(0.7, 1.0) * n
        sigma1 = rng.exponential(scale=0.1 * n) + 0.02 * n
        sigma2 = rng.exponential(scale=0.1 * n) + 0.02 * n
        w = rng.random()
        a = w * np.exp(-((idx - centre1) ** 2) / (2.0 * sigma1 ** 2)) \
            + (1.0 - w) * np.exp(-((idx - centre2) ** 2) / (2.0 * sigma2 ** 2))

    elif family == 9:                   # raised cosine (smooth bump)
        theta = np.pi * (idx / (n - 1) - 1.0)   # runs from –π to 0
        a = (1.0 + np.cos(theta)) ** rng.uniform(1.0, 3.0)

    else:                               # random Gaussian‑like bump (fallback)
        centre = rng.uniform(0, 1)
        width = rng.exponential(scale=0.15) + 0.05
        a = np.exp(-((idx / n - centre) ** 2) / (2 * width ** 2))

    # small Gaussian noise and projection onto the simplex
    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient (identical to previous)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of   (1/β)·log Σ exp(β·b)   where   b = a*a .
    Returns (g, b) with   g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smoothed maximum.
    Momentum, β‑annealing and occasional jitter are used.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 16384.0
    beta_factor = 1.6

    step_base = target_sum * 0.11
    momentum = np.zeros_like(a)
    rho = 0.92

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        # projection + tiny symmetrisation
        a = _project_to_simplex(a, target_sum)
        a = _symmetrize(a)               # optional – helps flatten the shape
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β‑annealing
        if it % 350 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a


# ----------------------------------------------------------------------
# Multi‑peak hill‑climbing
# ----------------------------------------------------------------------


def _hill_climb_multi_peak(a0: np.ndarray, b0: np.ndarray, deadline: float,
                           top_k: int = 8,
                           init_temp_factor: float = 0.025,
                           cooling: float = 0.9995) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing that looks at the *set* of current maximal convolution entries.
    Moves a small amount of mass from a source that contributes heavily to the
    peak set toward a destination that contributes little.
    Returns the best sequence and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size

    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    T = cur_max * init_temp_factor
    while time.time() < deadline:
        # ------------------------------------------------------------------
        # 1) pick the set of relevant peaks
        # ------------------------------------------------------------------
        if top_k >= len(b):
            peak_idxs = np.arange(len(b))
        else:
            # indices of the top‑k values (unsorted)
            peak_idxs = np.argpartition(b, -top_k)[-top_k:]

        # ------------------------------------------------------------------
        # 2) compute contributions of each index to these peaks
        # ------------------------------------------------------------------
        contrib = np.zeros(n, dtype=float)
        for k in peak_idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            # vector of i values
            i_vals = np.arange(i_start, i_end + 1, dtype=int)
            j_vals = k - i_vals
            contrib[i_vals] += a[i_vals] * a[j_vals]

        # ------------------------------------------------------------------
        # 3) sample source (high contribution) and destination (low contribution)
        # ------------------------------------------------------------------
        total = contrib.sum()
        if total > 0.0:
            prob_src = contrib / total
            i = rng.choice(n, p=prob_src)
        else:
            # fallback: sample proportional to current mass
            prob_src = a / a.sum()
            i = rng.choice(n, p=prob_src)

        inv = contrib.max() - contrib + 1e-12
        prob_dst = inv / inv.sum()
        j = rng.choice(n, p=prob_dst)
        if i == j:
            continue

        # ------------------------------------------------------------------
        # 4) amount of mass to move
        # ------------------------------------------------------------------
        max_delta = a[i] * 0.5
        if max_delta < 1e-16:
            continue
        delta = rng.random() * max_delta

        # ------------------------------------------------------------------
        # 5) analytical update of the convolution (exact)
        # ------------------------------------------------------------------
        b_new = b.copy()
        # linear part
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a

        # quadratic correction terms
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        # ------------------------------------------------------------------
        # 6) acceptance test (simulated annealing style)
        # ------------------------------------------------------------------
        accept = False
        if new_max < cur_max - 1e-14:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            # guard against tiny negatives caused by rounding
            a[i] = max(a[i], 0.0)
            a[j] = max(a[j], 0.0)

            b = b_new
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] multi‑peak improvement: {best_max:.6f}")

        T *= cooling
    return best_a, best_max

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the supplied
    wall‑clock budget (default 1000 s) and returns the best sequence found
    as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    best_seq = None
    best_score = np.inf  # we store max(b) – lower is better

    # ------------------------------------------------------------------
    # Optional seed from a previously discovered construction
    # ------------------------------------------------------------------
    earlier = globals().get("height_sequence_1")
    if isinstance(earlier, (list, tuple, np.ndarray)):
        a0 = np.asarray(earlier, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        best_score = _max_convolution_fft(a0)
        best_seq = a0.copy()
        print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) "
              f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Length schedule – start small, then enlarge gradually.
    # ------------------------------------------------------------------
    lengths = [2000, 5000, 8000, 12000, 18000, 25000, 32000]

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Build a pool of initial candidates for this length
        # --------------------------------------------------------------
        candidates = []

        # up‑sample the current best (if any) – this is usually a good start
        if best_seq is not None and best_seq.size != n:
            cand = _upsample_sequence(best_seq, n)
            candidates.append(cand)

        # a couple of fresh random starts to diversify search
        for _ in range(2):
            candidates.append(_random_initial_sequence(n, rng))

        # --------------------------------------------------------------
        # Allocate time for this size (roughly proportional to n)
        # --------------------------------------------------------------
        time_left = deadline - time.time()
        # we reserve a small fraction for later sizes
        allocation = min(0.6 * time_left, max(30.0, 0.025 * n))
        phaseA_dead = time.time() + 0.55 * allocation
        phaseB_dead = time.time() + allocation

        # iterate over candidates
        for a_init in candidates:
            if time.time() >= deadline:
                break

            a_proj = _project_to_simplex(a_init, target_sum)

            # -------------------- Phase A – gradient descent --------------------
            a_gd = _gradient_descent_opt(a_proj, phaseA_dead)

            score_gd = _max_convolution_fft(a_gd)
            if score_gd < best_score:
                best_score = score_gd
                best_seq = a_gd.copy()
                print(f"[{time.time():.1f}] GD improvement (n={n}) score={score_gd:.6f}")

            # -------------------- Phase B – multi‑peak hill climbing --------------
            if time.time() < deadline:
                b_full = _full_convolution(a_gd)
                a_hc, max_hc = _hill_climb_multi_peak(a_gd, b_full, phaseB_dead)
                if max_hc < best_score:
                    best_score = max_hc
                    best_seq = a_hc.copy()
                    print(f"[{time.time():.1f}] Hill‑climb improvement (n={n}) "
                          f"score={max_hc:.6f}")

        # after finishing this length, shrink the remaining deadline a bit
        # (to guarantee a final polishing window)
        # (no explicit action needed – loop uses the updated deadline)

    # ------------------------------------------------------------------
    # Final polishing with any remaining budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a_final = best_seq.copy()
        b_final = _full_convolution(a_final)
        a_final, max_final = _hill_climb_multi_peak(
            a_final, b_final, deadline)
        if max_final < best_score:
            best_score = max_final
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final polishing improvement: "
                  f"score={max_final:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries for neatness
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform vector (should never happen)
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log the official evaluation value
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:   # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
