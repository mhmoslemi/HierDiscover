def compute_gradient(a, mask):
    a_rev = a[::-1]
    conv_len = len(mask) + len(a) - 1
    f_len = next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2 * conv[len(a)-1:len(a)-1+len(a)]
    return g

def project_simplex(v, S):
    # returns projection onto nonnegative simplex with sum S.
    # algorithm based on Duchi et al.
    # If all entries are zero, return uniform S/n.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (S - cssv) / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - S) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
adversarial step‑function optimizer – enhanced version

The strategy consists of three phases:

1. **Fast random sampling** – generate many edge‑heavy candidate
   sequences of random lengths (≈1 k … 25 k).  Each candidate is a
   simple parametric shape
          a_i ∝ (d_i+offset)^{-p}  mixed with a uniform background.
   The best sampled vector is kept.

2. **Exact sub‑gradient descent** – starting from the best sampled
   vector we minimise the true peak of the self‑convolution
   (the evaluation score) using the exact sub‑gradient of the
   max‑operator.  After each step the iterate is projected onto the
   ℓ₁‑simplex  {x≥0, Σx=√(2·n)}.  A decreasing step‑size together
   with occasional random jitter provides robustness.

3. **Length‑scaling refinement** – the refined vector is linearly
   interpolated to a few nearby lengths (±25 %).  After re‑normalising
   the ℓ₁‑norm we keep the scaling that gives the smallest peak.

The whole routine runs for the allotted time (default 1000 s) and
stores the final candidate in the global variable ``height_sequence_1``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.ceil(pos).astype(int)
    hi = np.clip(hi, 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


def _project_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex
    {x | x_i ≥ 0, Σ x_i = S}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (S - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - S) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of the (soft) max of the convolution peak
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Return (grad, max_b) where grad = ∂ τ‑soft‑max(b) / ∂a,
    with b = a*a (full linear self‑convolution).
    For tau≈0 the exact sub‑gradient of max(b) is returned.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        w = exps / np.sum(exps)
        mask = w

    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    grad = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return grad, max_b


# ----------------------------------------------------------------------
# Random parametric edge‑heavy initial shape
# ----------------------------------------------------------------------


def _random_edge_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Generate a feasible candidate of length n.
    The shape follows (d+offset)^(-p) mixed with a uniform background.
    """
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0  # distance to nearest edge (1‑based)
    p = rng.uniform(0.1, 2.0)                # power‑law exponent
    offset = rng.uniform(0.0, 10.0)           # denominator shift
    mix = rng.uniform(0.0, 1.0)               # mixing proportion with uniform

    edge_part = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge_part + (1.0 - mix) * base
    raw = np.clip(raw, 0.0, None)
    S = np.sqrt(2.0 * n)
    raw = raw / raw.sum() * S
    return raw


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the simplex
# ----------------------------------------------------------------------


def _subgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Minimise the true peak of a*a starting from a0.
    Uses the exact sub‑gradient of max(b) and projection onto the simplex.
    Returns (best_vector, best_peak).
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    a = a0.copy()
    best_a = a.copy()
    best_peak = _full_convolution_fft(a).max()

    # initial step size – a small fraction of the typical coefficient size
    step = max(S * 0.05, 1e-4)
    it = 0

    while time.time() < deadline:
        it += 1

        # exact sub‑gradient (tau = 0)
        grad, peak = _softmax_gradient(a, 0.0)

        # gradient descent step
        a_new = a - step * grad
        # projection back onto the simplex {x≥0, Σx=S}
        a_new = _project_simplex(a_new, S)

        # evaluate new peak
        new_peak = _full_convolution_fft(a_new).max()

        # keep best ever seen
        if new_peak < best_peak:
            best_peak = new_peak
            best_a = a_new.copy()
            # modestly increase step when we make progress
            step *= 1.05

        else:
            # shrink step on non‑improvement
            step *= 0.75

        # occasional random jitter to escape flat plateaus
        if it % 500 == 0:
            jitter = rng.normal(scale=step * 0.2, size=n)
            a_new = _project_simplex(a_new + jitter, S)

        a = a_new

        # avoid step becoming too tiny (which would stall the loop)
        if step < 1e-12:
            step = 1e-12

    return best_a, best_peak

# Main entry point required by the harness

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine runs a quick random sampling phase, a long exact
    sub‑gradient refinement, and finally a small length‑scaling sweep.
    The best sequence found is stored in the global variable
    ``height_sequence_1`` and also returned as a Python list.
    """
    rng = np.random.default_rng(seed)
    # safety margin of a few seconds for the harness
    end_time = time.time() + budget_s - 5.0

    best_seq = None
    best_score = np.inf
    best_n = None

    # --------------------------------------------------------------
    # Phase 1 – fast random sampling of edge‑heavy shapes
    # --------------------------------------------------------------
    sampling_deadline = time.time() + min(20.0, budget_s * 0.025)  # ≈5 % of the budget
    while time.time() < sampling_deadline:
        n = rng.integers(1500, 25000)        # moderate to large lengths
        cand = _random_edge_sequence(n, rng)
        peak = _full_convolution_fft(cand).max()
        if peak < best_score:
            best_score = peak
            best_seq = cand.copy()
            best_n = n
            print(f"[sample] n={n}  peak={best_score:.6f}")

    # Include any previously stored construction as a warm start
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(globals()['height_sequence_1'], dtype=float)
            # rescale to an appropriate length (keep original length if reasonable)
            if prev.size >= 500 and prev.size <= 30000:
                prev_scaled = _scale_sequence(prev, best_n or prev.size)
                prev_scaled = np.maximum(prev_scaled, 0.0)
                prev_scaled = prev_scaled / prev_scaled.sum() * np.sqrt(2.0 * prev_scaled.size)
                peak = _full_convolution_fft(prev_scaled).max()
                if peak < best_score:
                    best_score = peak
                    best_seq = prev_scaled.copy()
                    best_n = prev_scaled.size
                    print(f"[warm‑start] imported sequence improved to {best_score:.6f}")
        except Exception:
            pass

    # --------------------------------------------------------------
    # Phase 2 – exact sub‑gradient descent (majority of the budget)
    # --------------------------------------------------------------
    if best_seq is not None:
        remaining_time = end_time - time.time()
        if remaining_time > 15.0:
            # Ensure exact ℓ₁‑norm before optimisation
            S = np.sqrt(2.0 * best_seq.size)
            best_seq = np.maximum(best_seq, 0.0)
            best_seq = best_seq / best_seq.sum() * S

            print(f"[refine] starting sub‑gradient descent (len={best_seq.size})")
            refined_seq, refined_peak = _subgrad_optimize(best_seq,
                                                          end_time,
                                                          rng)
            if refined_peak < best_score:
                best_score = refined_peak
                best_seq = refined_seq.copy()
                print(f"[refine] sub‑gradient improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – small length‑scaling sweep (±25 %)
    # --------------------------------------------------------------
    if best_seq is not None:
        scales = [0.75, 0.85, 0.95, 1.0, 1.05, 1.15, 1.25]
        for sf in scales:
            new_n = max(3, int(round(best_seq.size * sf)))
            # skip if time is already exhausted
            if time.time() >= end_time - 2.0:
                break
            scaled = _scale_sequence(best_seq, new_n)
            scaled = np.maximum(scaled, 0.0)
            S = np.sqrt(2.0 * new_n)
            scaled = scaled / scaled.sum() * S
            peak = _full_convolution_fft(scaled).max()
            if peak < best_score:
                best_score = peak
                best_seq = scaled.copy()
                best_n = new_n
                print(f"[scale] n={new_n}  improved peak={best_score:.6f}")

    # --------------------------------------------------------------
    # Final sanity check via the black‑box evaluator
    # --------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
        external = evaluate_sequence(final_seq)
        print(f"[final] fallback sequence, external score={external:.6f}")
    else:
        final_seq = best_seq.tolist()
        external = evaluate_sequence(final_seq)
        print(f"[final] external score={external:.6f}  internal peak={best_score:.6f}  length={len(final_seq)}")

    # expose the result for possible warm‑starts in later runs
    globals()['height_sequence_1'] = final_seq
    return final_seq
