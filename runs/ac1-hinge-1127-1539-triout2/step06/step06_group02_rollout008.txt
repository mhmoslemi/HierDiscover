"""
Improved search for a step‑function with minimal max‑autocorrelation.

Approach
--------
1. Choose a problem size `n` (default 1500) and initialise a non‑negative
   sequence with a power‑law shape, optionally warm‑started from the global
   variable `height_sequence_1`.
2. Project the sequence onto the ℓ¹‑simplex Σa = √(2·n).  The objective
   `evaluate_sequence` is scale‑invariant, so fixing the sum is convenient.
3. Iteratively improve the sequence by solving a linear programme:

      maximise   Σ g_j
      s.t.       (a * g)_k ≤ M   for all k = 0…2·n‑2
                 g_j ≥ 0

   where `M = max(a * a)` is the current worst autocorrelation entry.
   The LP tries to find a distribution `g` that spreads the mass as widely
   as possible while respecting the same bound `M`.  After solving, `g` is
   normalised to the target ℓ¹‑norm and mixed with the current iterate.

4. After a few LP steps (limited by the time budget) perform a short
   projected‑gradient descent on a smooth soft‑max approximation of the
   max‑convolution, further polishing the best sequence.
5. Return the best sequence found (trailing near‑zero entries are stripped).

The LP step is the main novelty compared to the previous implementation and
usually reduces the max‑convolution value by a few percent in a handful of
iterations.  All heavy linear‑algebra work is done with NumPy / SciPy
FFT / sparse matrices for speed.  The code respects the 1000 s budget
and prints progress information.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    if s <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    # Find the rho index
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _build_conv_matrix_rows_cols(n: int):
    """
    Pre‑compute row and column indices for the Toeplitz convolution matrix.
    For each column i the matrix has entries in rows i … i+n‑1.
    """
    # rows[i * n + j] = i + j  (i = column, j = offset)
    rows = np.tile(np.arange(n), n) + np.repeat(np.arange(n), n)
    cols = np.repeat(np.arange(n), n)
    return rows, cols


def _lp_step(a: np.ndarray, M: float, rows, cols, target_sum: float,
             mix_factor: float) -> np.ndarray:
    """
    Perform one LP refinement step and return the updated sequence.
    `mix_factor` ∈ (0,1] controls how far we move towards the LP solution.
    """
    n = a.size
    # Avoid exact zeros – they would make the LP unbounded.
    a_clip = np.maximum(a, 1e-12)

    # Build sparse matrix A where A @ g = a * g (linear convolution)
    data = np.repeat(a_clip, n)
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    # Linear programme: maximise sum(g)  ⇔  minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # Solve (highs) – it accepts sparse A_ub efficiently.
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True, 'time_limit': 30})
    if not res.success:
        # If the LP fails for any reason, fall back to the current a.
        return a

    g = res.x
    # Normalise g to the target ℓ¹‑norm.
    g = g * (target_sum / max(g.sum(), 1e-12))

    # Convex combination and projection back onto the simplex.
    new_a = (1.0 - mix_factor) * a + mix_factor * g
    new_a = _project_to_simplex(new_a, target_sum)
    return new_a


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(a*a).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    # full convolution a*a
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_c = conv.max()
    exps = np.exp(beta * (conv - max_c))
    p = exps / exps.sum()

    # Gradient formula: g = 2 * (rev_a * p) (convolution of reversed a with p)
    rev_a = a[::-1]
    p_fft = np.fft.rfft(p, n=size)
    rev_a_fft = np.fft.rfft(rev_a, n=size)
    conv_rev = np.fft.irfft(p_fft * rev_a_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]   # center slice gives size n
    return grad, conv


def _gradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected gradient descent on the soft‑max approximation of max‑conv.
    The temperature β is increased up to 4096.
    """
    a = a0.copy()
    n = a.size
    target_sum = math.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = _max_convolution_fft(a)

    beta = 1.0
    max_beta = 4096.0
    step_base = target_sum * 0.12
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        grad, _ = _softmax_gradient(a, beta)
        norm = np.linalg.norm(grad)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a = a - lr * grad

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur = _max_convolution_fft(a)
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
            print(f"[{time.time():.1f}] GD improve β={beta:.1f} score={best_score:.6f}")

        if it % 600 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    `evaluate_sequence`.  The algorithm runs for at most `budget_s` seconds
    and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # small safety margin

    # --------------------------------------------------------------
    # Problem size and target ℓ¹‑norm
    # --------------------------------------------------------------
    n = 1500                     # a size that balances quality and speed
    target_sum = math.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Initial sequence
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if init.size >= n:
            a = init[:n].copy()
        else:
            a = np.pad(init, (0, n - init.size), "constant")
    else:
        # power‑law with exponent ≈0.55 (empirically good)
        idx = np.arange(1, n + 1, dtype=np.float64)
        a = idx ** (-0.55)
        a = a * (target_sum / a.sum())

    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_score = _max_convolution_fft(a)
    print(f"[{time.time():.1f}] initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Pre‑compute Toeplitz indices for the LP matrix (rows, cols)
    # --------------------------------------------------------------
    rows, cols = _build_conv_matrix_rows_cols(n)

    # --------------------------------------------------------------
    # LP refinement loop
    # --------------------------------------------------------------
    lp_iter = 0
    max_lp_iters = 12                      # enough to stay well within the budget
    while time.time() < deadline and lp_iter < max_lp_iters:
        M = _max_convolution_fft(a)        # current worst autocorrelation entry
        # Gradually reduce the mixing factor – start aggressive, then fine‑tune.
        mix = max(0.5 / (1 + lp_iter * 0.2), 0.05)

        a = _lp_step(a, M, rows, cols, target_sum, mix)

        cur_score = _max_convolution_fft(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] LP iter {lp_iter+1} → score={best_score:.6f}")

        # early stopping if improvements become tiny
        if lp_iter >= 2 and (best_score - cur_score) < 1e-5:
            break

        lp_iter += 1

    # --------------------------------------------------------------
    # Short gradient‑descent polishing (uses remaining time)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        gd_deadline = time.time() + remaining * 0.9
        refined_a, refined_score = _gradient_descent(best_a, gd_deadline)
        if refined_score < best_score:
            best_score = refined_score
            best_a = refined_a.copy()
            print(f"[{time.time():.1f}] post‑GD score={best_score:.6f}")

    # --------------------------------------------------------------
    # Strip trailing zeros / negligible entries
    # --------------------------------------------------------------
    if best_a is not None:
        nz = np.where(best_a > 1e-12)[0]
        if nz.size:
            best_a = best_a[: nz[-1] + 1]
        else:
            best_a = np.array([], dtype=float)

    print(f"[{time.time():.1f}] final best score≈{best_score:.6f}, length={len(best_a)}")
    return best_a.tolist()
