def _subgradient_descent(a, deadline):
    n = len(a)
    target_sum = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)  # value of max
    while time.time() < deadline:
        # compute conv
        conv_full = _conv_fft(a)
        max_b = conv_full.max()
        # compute mask
        idx_max = np.where(conv_full >= max_b - 1e-12)[0]
        if len(idx_max) == 0:
            break
        p = np.zeros_like(conv_full)
        p[idx_max] = 1.0 / len(idx_max)
        # gradient
        size = 1 << ((2*n - 1).bit_length())
        a_rev = a[::-1]
        p_fft = np.fft.rfft(p, n=size)
        a_rev_fft = np.fft.rfft(a_rev, n=size)
        conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[:2*n-1]
        g = 2.0 * conv_rev[n-1:n-1+n]
        # zero-sum projection
        g_proj = g - np.mean(g)
        norm = np.linalg.norm(g_proj)
        if norm < 1e-12:
            break
        # step
        step_coeff = 0.5  # maybe adjustable
        alpha = step_coeff * target_sum / (norm + 1e-12)
        # try with backtracking
        success = False
        for _ in range(5):
            a_new = a - alpha * g_proj
            a_new = _project_to_simplex(a_new, target_sum)
            new_max = _max_convolution_fft(a_new)
            if new_max < max_b - 1e-12:
                success = True
                a = a_new
                max_b = new_max
                break
            else:
                alpha *= 0.5
        if not success:
            # maybe jitter random noise to escape
            rng = np.random.default_rng()
            a = a + rng.normal(scale=target_sum*0.001, size=n)
            a = _project_to_simplex(a, target_sum)
            continue
        if max_b < best_val:
            best_val = max_b
            best_a = a.copy()
    return best_a, best_val

"""
Multi‑scale sub‑gradient descent + refined simulated annealing.

The optimizer works on a few problem sizes (default 2000, 5000, 8000).  For each
size a candidate sequence is initialised (global warm‑start, power‑law or
random Dirichlet), projected onto the ℓ¹‑simplex Σa = √(2·n) and then refined
in two phases:

1.  Sub‑gradient descent on the exact max (a*a).  The active set of the worst
    convolution entries is used to build the sub‑gradient (average of the
    gradients of all active constraints).  After each step the iterate is
    projected back onto the simplex; a simple back‑tracking line‑search keeps
    the objective monotone.

2.  A long simulated‑annealing phase that moves mass from “bad’’ indices to
    “good’’ ones (based on their contribution to the current worst entry),
    performs short‑window smoothing and occasional swaps.  The temperature
    decays exponentially.

All intermediate iterates respect the simplex constraint, therefore the
evaluation function `evaluate_sequence` (which is scale‑invariant) only sees
valid candidates.  The best sequence found over all runs is returned (trailing
zeros are stripped).  The whole routine respects a 1000 s budget.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft
from typing import Tuple

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σ_i x_i = s}.  Implements Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (real input)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = rfft(a, n=size)
    conv = irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = rfft(a, n=size)
    conv = irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _subgradient_descent(
    a0: np.ndarray, deadline: float, verbose: bool = False
) -> Tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on max(a*a) with exact active‑set gradient.
    Returns the best sequence found and its max‑convolution value.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    it = 0
    # a modest tolerance for detecting the active set
    eps_active = 1e-12
    # step‑size scaling factor (tuned empirically)
    step_coeff = 0.4

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_b = float(conv.max())
        # active set (indices where the max is attained)
        idx_max = np.where(conv >= max_b - eps_active)[0]
        if idx_max.size == 0:
            break

        # mask p = average over the active constraints
        p = np.zeros_like(conv)
        p[idx_max] = 1.0 / idx_max.size

        # gradient of Σ_{k∈K} b_k  w.r.t a (exact, using FFT)
        size = 1 << ((2 * n - 1).bit_length())
        a_rev = a[::-1]
        p_fft = rfft(p, n=size)
        a_rev_fft = rfft(a_rev, n=size)
        conv_rev = irfft(a_rev_fft * p_fft, n=size)[: 2 * n - 1]
        g = 2.0 * conv_rev[n - 1 : n - 1 + n]   # gradient of Σ b_k

        # make the direction feasible for the sum‑constraint (zero‑mean)
        g -= np.mean(g)

        norm = np.linalg.norm(g)
        if norm < 1e-14:
            break

        # initial step length (inverse‑proportional to the gradient norm)
        alpha = step_coeff * target_sum / (norm + 1e-12)

        # back‑tracking line search (max 6 reductions)
        improved = False
        for _ in range(6):
            a_candidate = a - alpha * g
            a_candidate = _project_to_simplex(a_candidate, target_sum)
            new_max = _max_convolution_fft(a_candidate)
            if new_max < max_b - 1e-12:
                a = a_candidate
                max_b = new_max
                improved = True
                break
            else:
                alpha *= 0.5

        if not improved:
            # occasional jitter to escape plateaus
            rng = np.random.default_rng()
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = _project_to_simplex(a + jitter, target_sum)
            continue

        if max_b < best_val:
            best_val = max_b
            best_a = a.copy()
            if verbose:
                print(
                    f"[{time.time():.1f}] subgrad it={it:5d}  max={best_val:.6f}"
                )
        it += 1

    return best_a, best_val


def _simulated_annealing(
    a0: np.ndarray,
    deadline: float,
    start_temp: float = 0.8,
    end_temp: float = 1e-6,
    max_frac_start: float = 0.30,
    max_frac_end: float = 0.002,
    max_window: int = 40,
    verbose: bool = False,
) -> Tuple[np.ndarray, float]:
    """
    Refines a sequence with a classic simulated‑annealing scheme.
    Moves consist of:
      * mass transfer from a “bad’’ donor to a “good’’ receiver,
      * local smoothing of a short window,
      * random swap of two entries.
    The temperature and maximal transferred fraction decay exponentially.
    """
    a = a0.copy()
    n = a.size
    cur_max = _max_convolution_fft(a)

    best_a = a.copy()
    best_max = cur_max

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-9)
    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # current worst convolution entry
        k_max = int(np.argmax(conv))
        max_b = conv[k_max]

        # contribution of each a_i to that entry
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        # a_i * a_{k_max-i}
        contrib = a[idx] * a[k_max - idx]

        # ------------------------------------------------------------------
        # Propose a move
        # ------------------------------------------------------------------
        move_type = None
        if rng.random() < 0.70:  # mass transfer
            total_w = contrib.sum()
            if total_w > 0:
                donor = rng.choice(idx, p=contrib / total_w)
            else:
                donor = rng.integers(n)

            # pick a receiver that contributes little to the worst entry
            low_thr = np.percentile(contrib, 10)
            low_candidates = idx[contrib <= low_thr]
            if low_candidates.size:
                receiver = rng.choice(low_candidates)
            else:
                receiver = rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        elif rng.random() < 0.90:  # short smoothing
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                seg = a[start:end].copy()
                avg = seg.mean()
                a[start:end] = avg
                move_type = "smooth"
                saved_seg = seg
                saved_range = (start, end)

        else:  # random swap
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # ------------------------------------------------------------------
        # Evaluate the new state (single FFT)
        # ------------------------------------------------------------------
        fa = rfft(a, n=size_fft)
        conv_new = irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_max = float(conv_new.max())

        # ------------------------------------------------------------------
        # Metropolis acceptance
        # ------------------------------------------------------------------
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob = np.exp(-(new_max - cur_max) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a  # keep the move
            conv = conv_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
                if verbose:
                    print(
                        f"[{time.time():.1f}] SA new best max={best_max:.6f}  it={it}"
                    )
        else:
            # revert
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                s, e = saved_range
                a[s:e] = saved_seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter
        if it % 6000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.0008, size=n)
            a = _project_to_simplex(a + jitter, np.sqrt(2.0 * n))
            conv = _conv_fft(a)
            cur_max = float(conv.max())

        it += 1

    return best_a, best_max


def _run_one_instance(
    n: int,
    init_mode: str,
    deadline: float,
    verbose: bool = False,
) -> Tuple[np.ndarray, float]:
    """
    Optimises a single problem size.
    Returns the best sequence and its max‑convolution value.
    """
    target_sum = np.sqrt(2.0 * n)

    # ---------- initialisation ----------
    if init_mode == "global" and "height_sequence_1" in globals():
        seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if seq.size > n:
            seq = seq[:n]
        else:
            seq = np.pad(seq, (0, n - seq.size), "constant")
    elif init_mode == "power":
        # power‑law with exponent 0.55 (works well as a starting guess)
        ex = np.arange(1, n + 1, dtype=np.float64) ** (-0.55)
        seq = ex * (target_sum / ex.sum())
    else:  # random Dirichlet
        rng = np.random.default_rng()
        seq = rng.random(n)
        seq = seq / seq.sum() * target_sum

    seq = _project_to_simplex(seq, target_sum)

    # ---------- sub‑gradient phase ----------
    remaining = deadline - time.time()
    if remaining > 15.0:
        sub_deadline = time.time() + min(30.0, remaining * 0.35)
        seq, sg_max = _subgradient_descent(seq, sub_deadline, verbose=verbose)
    else:
        sg_max = _max_convolution_fft(seq)

    # ---------- simulated‑annealing phase ----------
    cur_time = time.time()
    if cur_time < deadline:
        seq, sa_max = _simulated_annealing(seq, deadline, verbose=verbose)
        final_max = min(sg_max, sa_max)
    else:
        final_max = sg_max

    return seq, final_max


def propose_candidate(
    seed: int = 42, budget_s: int = 1000, **kwargs
) -> list[float]:
    """
    Main entry point used by the evaluation harness.
    Runs a few problem sizes, each consisting of a short sub‑gradient
    phase followed by a simulated‑annealing phase.  The best sequence found
    before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # safety margin

    best_seq = None
    best_score = float("inf")

    # problem sizes – larger n gives more freedom but costs more per iteration
    sizes = [2000, 5000, 8000]

    # iterate over sizes, allocating a slice of the remaining budget to each
    for n in sizes:
        now = time.time()
        if now >= deadline - 5.0:
            break
        remaining = deadline - now
        # keep at least 20 s for the final refinement
        time_for_this = min(remaining * 0.42, 300.0)

        # rotate initialisation modes for diversity
        modes = ["global", "power", "random"]
        init_mode = modes[rng.integers(len(modes))]

        run_deadline = time.time() + time_for_this
        seq, max_val = _run_one_instance(
            n,
            init_mode,
            run_deadline,
            verbose=False,
        )

        # the evaluate_sequence value equals max_val when the sum is √(2n)
        score = max_val
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(
                f"[{time.time():.1f}] size={n} init={init_mode} → new best score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Final short sub‑gradient refinement on the best candidate (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        extra_deadline = time.time() + (deadline - time.time()) * 0.25
        best_seq, _ = _subgradient_descent(best_seq, extra_deadline, verbose=False)

    # Strip trailing zeros (they do not affect the score)
    if best_seq is not None:
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[: nz[-1] + 1]
        else:
            best_seq = np.array([], dtype=float)

    # final reporting
    if best_seq is not None:
        final_score = evaluate_sequence(best_seq.tolist())
        print(
            f"[{time.time():.1f}] final best score≈{final_score:.6f}, length={len(best_seq)}"
        )
    else:
        print(f"[{time.time():.1f}] no sequence found")
    return best_seq.tolist()
