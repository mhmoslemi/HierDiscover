"""
Propose a step‑function (non‑negative vector) whose self‑convolution peak
is as low as possible.

Algorithm (≈ 1000 s budget)

1. **Base generation** – use the previously found best sequence (`height_sequence_1`)
   when available, otherwise draw from a rich family of analytic shapes
   (uniform, geometric, reciprocal, triangular, beta‑like mixtures).  The
   vector is projected onto the ℓ¹‑simplex Σa = √(2·n).

2. **Active‑set sub‑gradient refinement** – at each iteration we compute the
   full self‑convolution, identify all entries that are within 1 % of the
   current maximum (the *active set*) and sum their sub‑gradients.
   A decreasing step‑size (∝ 1/√t) is taken, the candidate is projected back
   onto the simplex and optionally symmetrised.  Simple line‑search (few
   scaling factors) and occasional random mass‑swaps help escape plateaus.

3. **Local swap polishing** – a very cheap hill‑climbing that repeatedly
   moves a tiny amount of mass between two random positions while staying
   on the simplex.

The loop repeats the whole pipeline for many different lengths `n`
(≈ 2 k–20 k).  The best sequence found before the time‑budget expires is
returned and stored in the global variable ``height_sequence_1`` for
future up‑sampling.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x≥0, Σx = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex with total mass s."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·n‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Symmetrisation helpers
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry v[i] = v[-1‑i]."""
    return 0.5 * (v + v[::-1])


# ----------------------------------------------------------------------
# Upsampling (linear interpolation)
# ----------------------------------------------------------------------
def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to a new length n."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Rich family of random initialisers
# ----------------------------------------------------------------------
def _initial_sequence(base_seq: np.ndarray | None,
                     n: int,
                     rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n with ℓ¹‑norm √(2·n).
    Several families are mixed to provide diversity.
    """
    S = np.sqrt(2.0 * n)

    typ = rng.integers(0, 7)   # 0–6 inclusive

    if typ == 0 and base_seq is not None and base_seq.size > 10:
        a = _upsample_seq(base_seq, n)
        a += rng.normal(scale=a.mean() * 1e-3, size=n)
        a = np.clip(a, 0.0, None)

    elif typ == 1:
        # uniform Dirichlet(1)
        a = rng.random(n)

    elif typ == 2:
        # geometric decay from both ends
        rate = rng.uniform(0.9, 0.9995)
        left = rate ** np.arange(n)
        right = left[::-1]
        a = (left + right) / 2.0

    elif typ == 3:
        # edge‑heavy reciprocal shape
        idx = np.arange(n, dtype=float)
        a = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.8)
        a = a ** power

    elif typ == 4:
        # symmetric beta‑like shape (flexible shape)
        a_ = np.linspace(0.0, 1.0, n)
        alpha = rng.uniform(0.4, 5.0)
        beta = rng.uniform(0.4, 5.0)
        a = (a_ ** (alpha - 1.0)) * ((1.0 - a_) ** (beta - 1.0))
        a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)

    elif typ == 5:
        # triangular (peak centre)
        centre = (n - 1) / 2.0
        a = 1.0 - np.abs(np.arange(n) - centre) / centre
        a = np.clip(a, 0.0, None)

    else:
        # random convex combination of a few bases
        candidates = []

        candidates.append(rng.random(n))                     # uniform
        rate = rng.uniform(0.9, 0.9995)
        left = rate ** np.arange(n)
        candidates.append((left + left[::-1]) / 2.0)          # geometric
        idx = np.arange(n, dtype=float)
        rec = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.8)
        candidates.append(rec ** power)                       # reciprocal

        w = rng.random(len(candidates))
        w /= w.sum()
        a = sum(wi * ci for wi, ci in zip(w, candidates))

    # Normalisation to the required ℓ¹‑norm
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0)
        a_sum = a.sum()
    a = a / a_sum * S
    return a


# ----------------------------------------------------------------------
# Active‑set sub‑gradient optimisation (core refinement)
# ----------------------------------------------------------------------
def _refine_active_subgrad(a0: np.ndarray,
                          deadline: float,
                          rng: np.random.Generator,
                          enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent that uses the *active set*
    (all convolution entries within 1 % of the current maximum) to compute the descent direction.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        max_b = conv.max()

        # --- active set -------------------------------------------------
        tol = max_b * 0.01                     # 1 % relative tolerance
        active = np.where(conv >= max_b - tol)[0]

        # gradient = sum_{k in active} subgrad_{k}
        g = np.zeros_like(a)
        for k in active:
            i_low = max(0, k - (n - 1))
            i_high = min(n - 1, k)
            i = np.arange(i_low, i_high + 1)
            j = k - i
            g[i] += a[j]          # sub‑gradient of entry k

        # --- step size -------------------------------------------------
        eta = 0.5 / np.sqrt(it)    # decreasing step

        # try a few scaling factors; accept the first that improves the max
        improved = False
        for scale in (1.0, 0.5, 0.2, 0.1):
            cand = a - scale * eta * g
            if enforce_sym:
                cand = _symmetrize(cand)
            cand = _project_simplex(cand, S)
            cand_score = np.max(_full_convolution(cand))
            if cand_score < max_b:
                a = cand
                max_b = cand_score
                improved = True
                break

        # occasional random mass‑swap (helps escape shallow plateaus)
        if not improved and rng.random() < 0.03:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = 0.04 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                if enforce_sym:
                    cand = _symmetrize(cand)
                cand_score = np.max(_full_convolution(cand))
                if cand_score < max_b or rng.random() < 0.2:
                    a = cand
                    max_b = cand_score

        # record best ever seen
        if max_b < best_score:
            best_score = max_b
            best_a = a.copy()

        # keep exact symmetry if requested
        if enforce_sym:
            a = _symmetrize(a)

        # early exit when we are close to the deadline
        if it % 4000 == 0 and time.time() > deadline - 0.5:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Very cheap local swap hill‑climbing (fine‑tuning)
# ----------------------------------------------------------------------
def _local_swap(a0: np.ndarray,
                deadline: float,
                rng: np.random.Generator,
                enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """Swap tiny amounts of mass between random indices while staying on the simplex."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.02 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = _symmetrize(cand)
        cand_score = np.max(_full_convolution(cand))
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand
            a = cand
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function with a minimal self‑convolution peak.
    Returns the best sequence (list of floats) found before the time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin for the final external evaluation
    deadline = start_time + budget_s - 5.0

    # try to reuse a previously discovered construction
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_overall_seq = None
    best_overall_score = np.inf
    best_overall_n = None
    attempts = 0
    last_improve_time = start_time

    # ------------------------------------------------------------------
    # Main timed loop – repeatedly restart with a fresh length and initialisation,
    # keeping the global best.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        attempts += 1

        # Choose a problem size.
        if best_overall_seq is not None and rng.random() < 0.55:
            # perturb the length of the current best by ±10 %
            scale = rng.uniform(0.9, 1.1)
            n = max(800, int(best_overall_n * scale))
        else:
            n = int(rng.integers(1800, 18000))

        enforce_sym = rng.random() < 0.5

        # --------------------------------------------------------------
        # Generate an initial vector.
        # --------------------------------------------------------------
        if base_seq is not None and rng.random() < 0.45:
            a0 = _upsample_seq(base_seq, n)
            a0 += rng.normal(scale=a0.mean() * 1e-3, size=n)
            a0 = np.clip(a0, 0.0, None)
        else:
            a0 = _initial_sequence(base_seq, n, rng)

        if enforce_sym:
            a0 = _symmetrize(a0)

        S = np.sqrt(2.0 * n)
        a0 = _project_simplex(a0, S)

        # --------------------------------------------------------------
        # Allocate a time slice for this attempt.
        # --------------------------------------------------------------
        slice_deadline = min(time.time() + rng.uniform(10.0, 18.0), deadline)

        # 1️⃣ Refine with active‑set sub‑gradient method
        a_opt, score = _refine_active_subgrad(a0, slice_deadline, rng, enforce_sym)

        # 2️⃣ Very cheap local swap polishing (if time permits)
        if time.time() < slice_deadline:
            a_opt, score = _local_swap(a_opt, slice_deadline, rng, enforce_sym)

        # --------------------------------------------------------------
        # Update global best.
        # --------------------------------------------------------------
        if score < best_overall_score:
            best_overall_score = score
            best_overall_seq = a_opt.copy()
            best_overall_n = n
            last_improve_time = time.time()
            print(f"[global] attempt={attempts} n={n} sym={enforce_sym} "
                  f"new best={best_overall_score:.6f}")

        # Early stop if we have already reached the target bound
        if best_overall_score <= 1.503:
            break

        # Reset base_seq if we haven't improved for a while – encourages diversity
        if time.time() - last_improve_time > 45.0:
            base_seq = None

    # ------------------------------------------------------------------
    # Final sanity check via the black‑box evaluator.
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_overall_seq.tolist()

    external_score = evaluate_sequence(final_seq)
    print(f"[final] external={external_score:.6f} "
          f"internal_best={best_overall_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    # expose the best sequence as a global variable for possible up‑sampling later
    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)

    return final_seq
