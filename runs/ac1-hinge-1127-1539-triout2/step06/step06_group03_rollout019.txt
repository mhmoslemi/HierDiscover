def gradient_descent_long(a0, deadline):
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    max_beta = 1e6
    step_factor = 0.9
    lr_base = 1.0

    momentum = np.zeros_like(a)
    rho = 0.9

    while time.time() < deadline:
        g, b = _gradient_softmax(a, beta)
        momentum = rho * momentum + (1 - rho) * g

        norm = np.linalg.norm(momentum)
        if norm > 0:
            # step size inversely proportional to norm & beta
            lr = lr_base * step_factor / (norm + 1e-12)
            a -= lr * momentum

        # projection
        a = _project_to_simplex(a, target_sum)

        # maybe jitter occasionally
        if np.random.rand() < 0.01:
            a += np.random.normal(scale=target_sum * 1e-3, size=n)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # increase beta gradually
        if beta < max_beta:
            beta *= 1.03

    return a

def _reorder_by_critical(a):
    n = a.size
    # compute convolution
    b = np.convolve(a, a)
    max_b = b.max()
    # define critical threshold relative tolerance
    thr = max_b * 0.98  # top 2%
    crit = (b >= thr).astype(float)  # indicator vector length 2n-1
    # count per index: sliding sum of length n over crit
    # efficient via FFT
    size = 1 << ((2 * n - 1 + n - 1).bit_length())  # for conv length 3n - 2
    # compute convolution of crit with ones_n (valid) via FFT or simpler using np.convolve in 'valid' mode.

    counts = np.convolve(crit, np.ones(n), mode='valid')

def _lp_step(f, deadline):
    n = len(f)
    target_sum = np.sqrt(2.0*n)
    M = _max_convolution_fft(f)  # current maximum

    # Build coefficient matrix
    # Use toeplitz
    from scipy.linalg import toeplitz
    c = np.concatenate([f, np.zeros(n-1)])
    r = np.concatenate([f[:1], np.zeros(n-1)])
    A = toeplitz(c, r)  # shape (2n-1, n)

    # b_ub = M * np.ones(2*n-1)
    b_ub = np.full(2*n-1, M)

    # objective minimize -sum(g) -> c_obj = -np.ones(n)
    from scipy.optimize import linprog
    res = linprog(-np.ones(n), A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs', options={'time_limit': max(1.0, deadline - time.time())})
    if not res.success:
        # return original
        return f
    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 <= 0:
        return f
    g = g0 * target_sum / sum_g0

    # convex combine
    t = 0.3
    new_f = (1 - t) * f + t * g
    return new_f

"""
A fast, long‑run optimizer for the “step‑function” problem.

Strategy
--------
1.  Initialise from the best known construction (`height_sequence_1`) or from a
    simple power‑law family if that variable does not exist.
2.  Run a *long* projected gradient descent on the smooth log‑sum‑exp
    approximation of the max‑convolution (the routine from the previous
    solution).  The descent runs for roughly half of the time budget and
    therefore can perform many thousands of FFT‑based steps.
3.  Re‑order the obtained values so that the largest coefficients are placed
    on indices that contribute the least to the currently critical (near‑max)
    convolution diagonals.  This cheap greedy “critical‑diagonal” heuristic
    often lowers the peak further.
4.  Finish with a simulated‑annealing permutation search (swap moves only)
    that works directly on the max‑convolution value.  The annealer runs for
    the remaining time.
5.  The best sequence seen during the whole run is returned.

All steps respect the global deadline (default 1000 s).  NumPy is used for
fast FFT‑convolutions; SciPy is not required.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex Σx = s, x≥0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection onto {x_i ≥ 0, Σx_i = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) surrogate of max(b) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b) where
        b = a*a (full convolution)
        g = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a ⋆ p) ,  p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]

    # full convolution a*a
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Projected gradient descent on the smoothed maximum
# ----------------------------------------------------------------------


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the log‑sum‑exp smooth max.
    Runs until the given deadline.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5            # start with a small smoothness
    max_beta = 1e6
    momentum = np.zeros_like(a)
    rho = 0.9             # momentum decay
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * g
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            # step size inversely proportional to the gradient norm
            lr = 0.8 * target_sum / (norm * (beta + 1.0))
            a -= lr * momentum

        # projection onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 300 == 0:
            jitter = rng.normal(
                scale=target_sum * 6e-4 / (beta + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # increase smoothing parameter
        if it % 250 == 0 and beta < max_beta:
            beta = min(beta * 1.07, max_beta)

        it += 1
    return a


# ----------------------------------------------------------------------
# Critical‑diagonal greedy reordering
# ----------------------------------------------------------------------


def _reorder_by_critical(a: np.ndarray) -> np.ndarray:
    """
    Re‑order the entries of ``a`` so that the largest coefficients are placed
    on indices that appear the least often in the currently critical
    (near‑maximum) convolution diagonals.
    """
    n = a.size
    # full convolution
    b = _full_convolution_fft(a)
    max_b = b.max()
    # treat those diagonals whose value is within 2 % of the maximum as critical
    crit_mask = (b >= max_b * 0.98).astype(float)      # length 2·n‑1

    # sliding‑window sum of the critical mask of length n gives,
    # for each position i, the number of critical diagonals that involve i
    cumsum = np.concatenate(([0.0], np.cumsum(crit_mask)))
    counts = cumsum[n:] - cumsum[:n]                 # shape (n,)

    # positions with smaller counts are “safer” for large values
    idx_by_safety = np.argsort(counts)               # ascending
    vals_desc = np.sort(a)[::-1]                     # descending
    new_a = np.empty_like(a)
    new_a[idx_by_safety] = vals_desc
    return new_a


# ----------------------------------------------------------------------
# Permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------


def _permutation_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Improve the ordering of a fixed multiset by random swaps.
    The L¹‑norm stays unchanged, only the permutation varies.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # initial temperature – a few per‑cent of the current best
    T = best_score * 0.03 + 1e-9
    cooling = 0.9993
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose swap
        a[i], a[j] = a[j], a[i]

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        if cur_score <= best_score or rng.random() < np.exp(-dE / T):
            # accept
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # reject – revert
            a[i], a[j] = a[j], a[i]

        T *= cooling

        # occasional tiny reshuffle to inject diversity
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.025
            if np.any(mask):
                shuffled = a[mask]
                rng.shuffle(shuffled)
                a[mask] = shuffled
                cur_score = _max_convolution_fft(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()

    return best


# ----------------------------------------------------------------------
# Parametric families – used only if no historic solution is available
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Scale to meet the ℓ¹‑sum target (preserving non‑negativity)."""
    scale = target / np.sum(v)
    a = v * scale
    a = np.maximum(a, 0.0)
    return a


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    target = np.sqrt(2.0 * n)
    return _project_to_simplex(raw, target)
