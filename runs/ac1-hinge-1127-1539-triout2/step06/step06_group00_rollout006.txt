"""
Hybrid multi‑restart Adam optimizer with multi‑scale refinement.

The algorithm repeatedly creates diverse feasible step‑functions,
optimises them on the simplex with a τ‑soft‑max approximation of the
convolution peak, and whenever a new best sequence is found it is
scaled to larger lengths and refined again.  A final long Adam run
uses a slowly decreasing temperature and learning‑rate schedule to
push the peak as low as possible before returning the best candidate.
"""

import time
import numpy as np
from math import sqrt

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.ceil(pos).astype(int)
    hi = np.clip(hi, 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Gradient of the τ‑soft‑max approximation of max(b) where b = a*a
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Return (grad, max_b) where grad = ∂ τ‑soft‑max(b) / ∂ a.
    b = a*a (full convolution).  If tau is tiny we fall back to an
    exact sub‑gradient (uniform weight on all active entries).
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        # exact sub‑gradient: equally weight all indices attaining max_b
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)          # soft‑max weights

    # gradient w.r.t. a: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   lr0: float = 0.5,
                   tau_factor: float = 0.995) -> tuple[np.ndarray, float]:
    """
    Minimise a τ‑soft‑max approximation of the convolution peak.
    Returns the best vector encountered and the corresponding *hard*
    peak (i.e. max of the true convolution).
    """
    n = a0.shape[0]
    S = sqrt(2.0 * n)          # required ℓ₁–norm

    # initialise logits so that softmax(p)*S == a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)               # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr0

    # initialise temperature
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 1.0)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ------- recover a from logits (log‑softmax) -------
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ------- convolution & gradient (soft‑max) -------
        g, peak = _softmax_gradient(a, tau)

        # keep best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ------- gradient w.r.t. logits (chain rule) -------
        mu = np.dot(a, g) / S          # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)          # shape (n,)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        clip_thr = 5.0 * S
        if clip_val > clip_thr:
            grad_p = grad_p * (clip_thr / clip_val)

        # ------- Adam update -------
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ------- schedules -------
        tau = max(tau * tau_factor, 1e-12)   # tighten temperature
        lr *= 0.9995                         # very slow decay

        # occasional jitter to escape plateaus
        if it % 800 == 0:
            p += rng.normal(scale=0.015, size=n)

    return best_a, best_peak

# Parametrised families for diverse initial vectors
# ----------------------------------------------------------------------
def _edge_symmetric(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Symmetric edge‑heavy shape: (distance_to_nearest_edge + offset)^{-p},
    optionally mixed with a uniform background.
    """
    S = sqrt(2.0 * n)
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.05, 3.0)
    offset = rng.uniform(0.0, 10.0)
    mix = rng.uniform(0.0, 1.0)               # weight of the edge part
    edge = (d + offset) ** (-p)
    uniform = np.ones(n)
    raw = mix * edge + (1.0 - mix) * uniform
    raw = raw / raw.sum() * S
    return raw


def _edge_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Symmetric edge‑heavy shape with exponential decay:
    exp(-p·distance) mixed with uniform.
    """
    S = sqrt(2.0 * n)
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.1, 4.0)
    mix = rng.uniform(0.0, 1.0)
    edge = np.exp(-p * (d - 1.0))
    uniform = np.ones(n)
    raw = mix * edge + (1.0 - mix) * uniform
    raw = raw / raw.sum() * S
    return raw


def _edge_asymmetric(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Edge‑heavy shape that decays only from the left edge (asymmetric).
    """
    S = sqrt(2.0 * n)
    d = np.arange(n) + 1.0                    # distance from the left edge
    p = rng.uniform(0.05, 3.0)
    offset = rng.uniform(0.0, 10.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    uniform = np.ones(n)
    raw = mix * edge + (1.0 - mix) * uniform
    raw = raw / raw.sum() * S
    return raw


def _dirichlet_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet (Gamma) random draw, then normalised."""
    S = sqrt(2.0 * n)
    alpha = rng.uniform(0.1, 2.0)
    a = rng.gamma(alpha, 1.0, size=n)
    a = a / a.sum() * S
    return a


def _uniform_initial(n: int) -> np.ndarray:
    """Simple uniform vector satisfying the ℓ₁–norm constraint."""
    S = sqrt(2.0 * n)
    return np.full(n, S / n)


def _initial_vector(n: int, rng: np.random.Generator, base_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Occasionally warm‑starts from the globally best sequence.
    """
    if base_seq is not None and rng.random() < 0.15:
        a = _scale_sequence(base_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * sqrt(2.0 * n)
        return a

    style = rng.choice(['uniform', 'dirichlet', 'edge_sym', 'edge_exp', 'edge_asym'])
    if style == 'uniform':
        a = _uniform_initial(n)
    elif style == 'dirichlet':
        a = _dirichlet_initial(n, rng)
    elif style == 'edge_sym':
        a = _edge_symmetric(n, rng)
    elif style == 'edge_exp':
        a = _edge_exponential(n, rng)
    else:  # edge_asym
        a = _edge_asymmetric(n, rng)
    return a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.
    The routine runs many short Adam restarts with diverse initial
    vectors, expands promising candidates to larger lengths, and
    finally performs a long refinement run.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    end_time = start_time + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # Retrieve a previously found construction if it exists
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    best_n = None

    def _update_best(seq: np.ndarray, score: float, n: int):
        nonlocal best_seq, best_score, best_n
        if score < best_score:
            best_seq = seq.copy()
            best_score = float(score)
            best_n = n
            print(f"[new best] n={n}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Exploration with many short Adam restarts
    # ------------------------------------------------------------------
    short_budget = 9.0   # seconds per short run
    while time.time() + short_budget + 15.0 < end_time:
        # choose a length (mix of moderate and larger sizes)
        if rng.random() < 0.6:
            n = rng.integers(1500, 8000)
        else:
            n = rng.integers(8000, 20000)

        a_start = _initial_vector(n, rng, base_seq)

        deadline = min(time.time() + short_budget, end_time - 15.0)
        a_opt, peak = _adam_optimize(a_start, deadline, rng,
                                     lr0=0.6, tau_factor=0.995)
        _update_best(a_opt, peak, n)

        # ------------------------------------------------------------------
        # If we have just improved, try scaling the new best to larger n
        # ------------------------------------------------------------------
        if best_seq is not None and best_seq is a_opt:
            for factor in (1.5, 2.0):
                new_n = int(best_n * factor)
                if new_n > 20000:
                    continue
                a_scaled = _scale_sequence(best_seq, new_n)
                a_scaled = np.clip(a_scaled, 0.0, None)
                a_scaled = a_scaled / a_scaled.sum() * sqrt(2.0 * new_n)

                deadline2 = min(time.time() + 6.0, end_time - 10.0)
                a_ref, peak_ref = _adam_optimize(a_scaled, deadline2, rng,
                                                 lr0=0.5, tau_factor=0.996)
                _update_best(a_ref, peak_ref, new_n)

    # ------------------------------------------------------------------
    # Phase 2 – Final long refinement on the current best candidate
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = end_time - time.time()
        if remaining > 8.0:
            # ensure exact ℓ₁‑norm before the final run
            best_seq = np.maximum(best_seq, 0.0)
            best_seq = best_seq / best_seq.sum() * sqrt(2.0 * best_seq.shape[0])

            a_final, peak_final = _adam_optimize(best_seq, end_time, rng,
                                                 lr0=0.4, tau_factor=0.997)
            if peak_final < best_score:
                best_seq = a_final
                best_score = float(peak_final)
                best_n = best_seq.shape[0]
                print(f"[refine] final refined score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation through the black‑box evaluator
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()
        external = evaluate_sequence(final_seq)
        print(f"[final] external score = {external:.6f}  internal max = {best_score:.6f}  length = {len(final_seq)}")
        globals()['height_sequence_1'] = final_seq

    return final_seq
