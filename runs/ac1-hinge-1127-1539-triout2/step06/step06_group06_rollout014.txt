def propose_candidate(budget_s: int = 1000, **kwargs):
    import time, numpy as np
    try:
        from scipy.optimize import linprog
        import scipy.linalg
        SCIPY = True
    except Exception:
        SCIPY = False

    def max_conv(a):
        n = a.shape[0]
        size = 1 << ((2*n-1).bit_length())
        fa = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
        return float(conv.max()), conv

    def project_simplex(v, s):
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - s
        rho = np.nonzero(u - cssv / np.arange(1, len(v)+1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = cssv[rho] / (rho + 1)
        w = np.maximum(v - theta, 0.0)
        return w

    # parameters
    start = time.time()
    deadline = start + budget_s - 1.0

    # choose length
    if 'height_sequence_1' in globals():
        init_global = np.array(globals()['height_sequence_1'], dtype=np.float64)
        n = len(init_global)
    else:
        n = 1500
    # ensure at least 1500 to have decent degrees
    if n < 1500:
        n = 1500

    target_sum = np.sqrt(2.0 * n)

    if 'height_sequence_1' in globals():
        seq = _resize_sequence(np.array(globals()['height_sequence_1'], dtype=np.float64), n)
    else:
        seq = _power_law_sequence(n, 0.55, target_sum)

    seq = project_simplex(seq, target_sum)

    best_seq = seq.copy()
    best_score, _ = max_conv(seq)

    # Phase 1: subgradient descent on max
    phase1_deadline = start + (deadline - start) * 0.4
    rng = np.random.default_rng()
    momentum = np.zeros_like(seq)
    gamma = 0.9
    step_base = target_sum * 0.2  # step scaling

    while time.time() < phase1_deadline:
        mval, conv_full = max_conv(seq)
        # locate a max index (choose the first)
        k_max = int(np.argmax(conv_full))
        # compute gradient of this max entry (subgradient)
        # Use FFT based shift to compute quickly
        size = 1 << ((2 * n - 1).bit_length())
        # p is delta at k_max
        p = np.zeros(2 * n - 1, dtype=np.float64)
        p[k_max] = 1.0
        a_rev_fft = np.fft.rfft(seq[::-1], n=size)
        p_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]  # shape n

        # momentum update
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (1.0 + (k_max / (2 * n))))  # some heuristic
            seq -= lr * momentum

        # project onto simplex and nonnegative
        seq = project_simplex(seq, target_sum)

        # occasional jitter
        if rng.random() < 0.05:
            seq += rng.normal(scale=target_sum * 0.001, size=n)
            seq = np.maximum(seq, 0.0)
            seq = project_simplex(seq, target_sum)

        cur_score = max_conv(seq)[0]
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] GD subgrad it score={best_score:.6f}")

    # Phase 2: Linear programming refinement (if available)
    if SCIPY:
        phase2_deadline = deadline
        # maximum number of LP iterations to avoid long loops
        max_lp_iters = 30
        lp_iter = 0
        while time.time() < phase2_deadline and lp_iter < max_lp_iters:
            lp_iter += 1
            # current max
            M, _ = max_conv(seq)
            # build Toeplitz matrix for constraints
            col = np.concatenate([seq, np.zeros(n - 1, dtype=np.float64)])
            row = np.concatenate([np.array([seq[0]], dtype=np.float64), np.zeros(n - 1, dtype=np.float64)])
            A_ub = scipy.linalg.toeplitz(col, row)  # shape (2n-1, n)
            b_ub = np.full(2 * n - 1, M, dtype=np.float64)
            c_obj = -np.ones(n, dtype=np.float64)  # maximize sum -> minimize -sum
            # solve LP
            res = linprog(c=c_obj, A_ub=A_ub, b_ub=b_ub,
                          bounds=[(0, None)] * n, method='highs', options={'presolve': True})
            if not res.success:
                print(f"[{time.time():.2f}] LP failed: {res.message}")
                break
            b_sol = res.x
            sum_b = b_sol.sum()
            if sum_b <= 0:
                break
            # scale to target sum
            g = b_sol * (target_sum / sum_b)
            M_g = max_conv(g)[0]
            # accept if improvement
            improved = False
            if M_g < M:
                seq = g
                improved = True
            else:
                # try convex combination with t=0.5
                t = 0.5
                cand = (1 - t) * seq + t * g
                M_cand = max_conv(cand)[0]
                if M_cand < M:
                    seq = cand
                    improved = True
            cur_score = max_conv(seq)[0]
            if cur_score < best_score:
                best_score = cur_score
                best_seq = seq.copy()
                print(f"[{time.time():.2f}] LP it={lp_iter} best score={best_score:.6f}")
            # if no improvement, add small jitter to escape
            if not improved:
                seq += rng.normal(scale=target_sum * 0.0005, size=n)
                seq = np.maximum(seq, 0.0)
                seq = project_simplex(seq, target_sum)

    arr = np.array(best_seq, dtype=float)

"""
Optimisation of a non‑negative step function (height sequence) to
minimise the benchmark `evaluate_sequence`.

The approach consists of two phases:

1. **Sub‑gradient descent on the true max‑convolution**  
   The objective after normalisation is simply the maximum entry of the
   self‑convolution `a*a`.  At each iteration the index of this maximum
   is identified; a sub‑gradient of that entry is computed (which is just
   a shifted reverse of the current vector) and a momentum step is taken.
   After each step the vector is projected onto the ℓ¹‑simplex
   `Σa_i = √(2·n)` to keep the normalisation exact.

2. **Linear‑programming refinement** (if SciPy is available)  
   For a fixed vector `f` with current maximum `M = max(f*f)`,
   we solve the LP

        maximise   Σ b_j
        subject to  f * b ≤ M      (element‑wise)
                    b_j ≥ 0

   The LP pushes mass to positions that least increase the convolution.
   The solution `b` is rescaled to the required ℓ¹‑norm and, if it yields a
   smaller convolution maximum, replaces the current vector.  A few LP
   iterations (≈10‑20) are performed within the remaining time budget.

Both phases respect the non‑negativity and sum constraints, and the best
sequence seen is returned.  The algorithm works for a single length
(`n≈1500`); larger lengths could be tried with more time but are omitted
to stay well within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray):
    """
    Returns (max_entry, full_convolution) of the linear convolution a*a.
    The convolution is computed by FFT for speed.
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max()), conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """
    Create a (i+1)^(-p) sequence of length n and normalise it to `target_sum`.
    """
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """
    Linear interpolation of `seq` to a new length `new_len`.
    """
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(new_x, old_x, seq)
    return new_seq.astype(np.float64)


def _constraint_matrix(f: np.ndarray):
    """
    Build the matrix A (shape (2n‑1, n)) such that (f * b)_k = (A b)_k.
    The matrix is a (lower) Toeplitz matrix with first column [f, 0,…,0]
    and first row [f[0], 0,…,0].
    """
    n = f.shape[0]
    # first column: f followed by (n‑1) zeros
    col = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    # first row: f[0] followed by zeros
    row = np.concatenate([np.array([f[0]], dtype=f.dtype), np.zeros(n - 1, dtype=f.dtype)])
    # dense Toeplitz matrix (2n‑1) × n
    from scipy.linalg import toeplitz
    return toeplitz(col, row)


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Produce a height sequence with a low benchmark score.

    Parameters
    ----------
    budget_s : int
        Time budget in seconds (default 1000).

    Returns
    -------
    list[float]
        The best sequence found (truncated after the last non‑zero entry).
    """
    # --------------------------------------------------------------
    # Imports that may be optional
    # --------------------------------------------------------------
    try:
        from scipy.optimize import linprog
        SCIPY_AVAILABLE = True
    except Exception:                     # pragma: no cover
        SCIPY_AVAILABLE = False

    start = time.time()
    deadline = start + budget_s - 1.0    # keep a safety margin

    # --------------------------------------------------------------
    # Initialise a sequence of length n (≈1500)
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=np.float64)
        n = max(1500, init_seq.size)
        seq = _resize_sequence(init_seq, n)
    else:
        n = 1500
        seq = _power_law_sequence(n, 0.55, np.sqrt(2.0 * n))

    target_sum = np.sqrt(2.0 * n)
    seq = _project_to_simplex(seq, target_sum)

    best_seq = seq.copy()
    best_score, _ = _max_convolution_fft(seq)

    # --------------------------------------------------------------
    # Phase 1 – sub‑gradient descent on the true max entry
    # --------------------------------------------------------------
    phase1_deadline = start + (deadline - start) * 0.40
    rng = np.random.default_rng()
    momentum = np.zeros_like(seq)
    gamma = 0.9
    step_base = target_sum * 0.20   # a modest step size

    while time.time() < phase1_deadline:
        max_val, full_conv = _max_convolution_fft(seq)
        k_max = int(np.argmax(full_conv))

        # Sub‑gradient of the max entry (a shifted reverse of `seq`)
        size_fft = 1 << ((2 * n - 1).bit_length())
        p = np.zeros(2 * n - 1, dtype=np.float64)
        p[k_max] = 1.0
        a_rev_fft = np.fft.rfft(seq[::-1], n=size_fft)
        p_fft = np.fft.rfft(p, n=size_fft)
        conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size_fft)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        # Momentum step
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm + 1e-12)
            seq -= lr * momentum

        seq = _project_to_simplex(seq, target_sum)

        # Occasional jitter to escape shallow plateaux
        if rng.random() < 0.04:
            seq += rng.normal(scale=target_sum * 0.001, size=n)
            seq = np.maximum(seq, 0.0)
            seq = _project_to_simplex(seq, target_sum)

        cur_score = _max_convolution_fft(seq)[0]
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] GD sub‑grad score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Linear‑programming refinement (if SciPy is present)
    # --------------------------------------------------------------
    if SCIPY_AVAILABLE:
        max_lp_iters = 20
        lp_iter = 0
        while time.time() < deadline and lp_iter < max_lp_iters:
            lp_iter += 1
            M, _ = _max_convolution_fft(seq)

            # Build constraint matrix A ( (f * b)_k ≤ M )
            A_ub = _constraint_matrix(seq)
            b_ub = np.full(2 * n - 1, M, dtype=np.float64)

            # Linear programme: maximise Σ b  ↔ minimise −Σ b
            res = linprog(c=-np.ones(n, dtype=np.float64),
                          A_ub=A_ub,
                          b_ub=b_ub,
                          bounds=[(0, None)] * n,
                          method='highs',
                          options={'presolve': True})

            if not res.success:                      # pragma: no cover
                print(f"[{time.time():.2f}] LP failed: {res.message}")
                break

            b_sol = np.maximum(res.x, 0.0)           # safeguard tiny negatives
            sum_b = b_sol.sum()
            if sum_b <= 0:
                break

            # Rescale to the required ℓ¹‑norm
            g = b_sol * (target_sum / sum_b)

            M_g = _max_convolution_fft(g)[0]

            # Accept the improvement (full replacement or ½‑mix)
            improved = False
            if M_g < M:
                seq = g
                improved = True
            else:
                cand = 0.5 * seq + 0.5 * g
                M_cand = _max_convolution_fft(cand)[0]
                if M_cand < M:
                    seq = cand
                    improved = True

            cur_score = _max_convolution_fft(seq)[0]
            if cur_score < best_score:
                best_score = cur_score
                best_seq = seq.copy()
                print(f"[{time.time():.2f}] LP it={lp_iter} best score={best_score:.6f}")

            if not improved:
                # small random perturbation to try a different direction
                seq += rng.normal(scale=target_sum * 0.0005, size=n)
                seq = np.maximum(seq, 0.0)
                seq = _project_to_simplex(seq, target_sum)

    # --------------------------------------------------------------
    # Final clean‑up and return
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # One final evaluation (optional, for logging)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:                         # pragma: no cover
        pass

    return final_seq
