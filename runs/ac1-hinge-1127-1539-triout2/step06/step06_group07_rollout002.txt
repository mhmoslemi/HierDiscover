propose_candidate

A three‑phase optimiser that builds on the previous Adam / sub‑gradient /

*Stage 1* – quick Adam + sub‑gradient on a modest size (n≈2000) to obtain a

*Stage 2* – up‑sample to a larger problem (default n=8000) and run a short
Adam refinement.  
*Stage 3* – greedy / simulated‑annealing hill‑climbing on the full
convolution.  Each move transfers a tiny amount of mass from one index to
another; the convolution is updated in O(n) by slice operations, so many
thousands of moves fit easily into the remaining budget.  

The whole routine respects the 1000 s time limit, prints a few progress
messages and finally returns the best sequence found (as a plain list of
floats).  The returned sequence satisfies the non‑negativity constraints
required by ``evaluate_sequence``.

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities (full linear convolution)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Soft‑max weights (smooth approximation of max)
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Stable soft‑max weights: w_i = exp(alpha·conv_i) / Σ_j exp(alpha·conv_j)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Gradient of the soft‑max approximation (used by Adam)
# ----------------------------------------------------------------------


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g_j = 2 Σ_i w_{i+j} a_i .
    ``a`` has length n, ``w`` has length 2·n‑1.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on a soft‑max approximation of the max‑norm
# ----------------------------------------------------------------------


def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth max‑norm."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    lr = 0.4
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    alpha = 0.5          # temperature – starts smooth
    alpha_max = 1500.0   # final sharpness

    while time.time() < deadline:
        t += 1
        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_simplex(a, S)

        # occasional random mass swap (exploration)
        if rng.random() < 0.03:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.02
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        alpha = min(alpha * 1.05, alpha_max)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient optimiser (acts on the true max‑norm)
# ----------------------------------------------------------------------


def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑norm."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.5
    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient for the worst entry
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional random perturbation
        if rng.random() < 0.07:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.03
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = np.max(_full_convolution(cand))
                if cand_score < cur_score or rng.random() < 0.2:
                    a = cand
                    cur_score = cand_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimiser (Adam → sub‑gradient) with configurable split
# ----------------------------------------------------------------------


def _optimize_combined_adam(a0: np.ndarray,
                            deadline: float,
                            rng: np.random.Generator,
                            fraction_adam: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Run Adam for a fraction of the allocated time, then finish with
    sub‑gradient descent.
    """
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    adam_deadline = now + fraction_adam * remaining
    a_mid, sc_mid = _optimize_adam(a0, adam_deadline, rng)

    if time.time() < deadline:
        a_fin, sc_fin = _optimize_subgrad(a_mid, deadline, rng)
        if sc_fin < sc_mid:
            return a_fin, sc_fin
    return a_mid, sc_mid


# ----------------------------------------------------------------------
# Upsampling helper (linear interpolation)
# ----------------------------------------------------------------------


def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a to length new_n."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=float)
    return np.interp(x_new, x_old, a)


# ----------------------------------------------------------------------
# Large‑scale hill‑climbing (greedy + simulated annealing)
# ----------------------------------------------------------------------


def _hill_climb(a0: np.ndarray,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Greedy / SA hill‑climbing that works directly on the full convolution.
    Moves a tiny amount of mass from one index to another and updates the
    convolution in O(n) using slice operations.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _full_convolution(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_score = cur_max

    start = time.time()
    total_time = deadline - start
    if total_time <= 0:
        return best_a, best_score

    # step‑size schedule (from 5 % of typical entry to 0.1 %)
    step_start = 0.05
    step_end = 0.001
    # temperature schedule (controls occasional uphill moves)
    temp_start = 0.008
    temp_end = 1e-6

    while time.time() < deadline:
        # progress [0,1]
        prog = (time.time() - start) / total_time
        step_factor = step_start * (1 - prog) + step_end * prog
        temp = temp_start * (1 - prog) + temp_end * prog

        # pick source index i (must have mass) and destination j
        i = rng.integers(0, n)
        if a[i] <= 0.0:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # amount to transfer – a small fraction of a[i]
        eps = a[i] * step_factor * rng.random()
        if eps <= 0.0:
            continue

        # --------------------------------------------------------------
        # build a candidate convolution (copy + updates)
        # --------------------------------------------------------------
        cand_conv = conv.copy()

        # linear contributions (2 * delta * shifted a)
        delta_i = -eps
        delta_j = eps

        cand_conv[i:i + n] += 2.0 * delta_i * a
        cand_conv[j:j + n] += 2.0 * delta_j * a

        # quadratic terms (only three positions change)
        cand_conv[2 * i] += delta_i * delta_i
        cand_conv[2 * j] += delta_j * delta_j
        cand_conv[i + j] += 2.0 * delta_i * delta_j

        cand_max = cand_conv.max()
        delta_score = cand_max - cur_max

        # acceptance test
        if delta_score <= 0.0 or rng.random() < np.exp(-delta_score / (temp + 1e-12)):
            # accept move
            a[i] += delta_i
            a[j] += delta_j
            conv = cand_conv
            cur_max = cand_max

            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function (vector) that minimises
    ``evaluate_sequence``.  The procedure consists of:

      1. A quick Adam + sub‑gradient optimisation on a medium size
         problem (n≈2000).
      2. Upsampling to a larger size (default n=8000) and a short Adam
         refinement.
      3. An extensive hill‑climbing stage that updates the full
         convolution in O(n) per move.

    The best vector found during the whole run is returned as a plain
    Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered construction if available
    # ------------------------------------------------------------------
    best_seq = None
    if 'height_sequence_1' in globals():
        try:
            best_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            best_seq = None

    best_score = np.inf
    last_improve = start_time

    # ------------------------------------------------------------------
    # Stage 1 – medium problem size
    # ------------------------------------------------------------------
    n_small = 2000
    stage1_deadline = min(time.time() + (deadline - time.time()) * 0.20,
                          deadline)

    if best_seq is not None:
        a0 = _upsample(best_seq, n_small)
    else:
        a0 = rng.random(n_small)

    # Tiny random jitter to avoid exact repetitions
    jitter = rng.normal(scale=a0.mean() * 1e-4 + 1e-12, size=n_small)
    a0 = np.clip(a0 + jitter, 0.0, None)
    a0 = _project_simplex(a0, np.sqrt(2.0 * n_small))

    a_opt, sc_opt = _optimize_combined_adam(a0, stage1_deadline, rng,
                                            fraction_adam=0.78)

    if sc_opt < best_score:
        best_score = sc_opt
        best_seq = a_opt.copy()
        last_improve = time.time()
        print(f"[stage‑1] n={n_small}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – upsample + short Adam refinement on a larger size
    # ------------------------------------------------------------------
    n_big = 8000
    stage2_deadline = min(time.time() + (deadline - time.time()) * 0.30,
                          deadline)

    a_big = _upsample(best_seq, n_big)
    a_big = _project_simplex(a_big, np.sqrt(2.0 * n_big))

    a_big_opt, sc_big_opt = _optimize_combined_adam(a_big,
                                                    stage2_deadline,
                                                    rng,
                                                    fraction_adam=0.75)

    if sc_big_opt < best_score:
        best_score = sc_big_opt
        best_seq = a_big_opt.copy()
        last_improve = time.time()
        print(f"[stage‑2] n={n_big}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – extensive hill‑climbing on the large problem
    # ------------------------------------------------------------------
    if time.time() < deadline:
        hill_deadline = deadline
        a_hill_best, sc_hill_best = _hill_climb(best_seq, hill_deadline, rng)
        if sc_hill_best < best_score:
            best_score = sc_hill_best
            best_seq = a_hill_best.copy()
            last_improve = time.time()
            print(f"[hill‑climb] n={n_big}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check & external evaluation
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
    else:
        # final projection (eliminates tiny drift)
        final_seq = _project_simplex(best_seq,
                                     np.sqrt(2.0 * best_seq.shape[0])).tolist()

    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} "
          f"(internal best = {best_score:.6f}) length = {len(final_seq)}")
    return final_seq
