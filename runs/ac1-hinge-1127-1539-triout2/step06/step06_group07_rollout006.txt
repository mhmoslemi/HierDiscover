"""
Simple simulated‑annealing refinement for the self‑convolution max‑norm.
Starts from the best known sequence (height_sequence_1 if present),
upsamples to a larger size (4000 coefficients) and performs many
mass‑transfer moves that keep the L¹‑norm fixed while decreasing the
maximum entry of the convolution.  The algorithm runs until the
allocated time budget (≈1000 s) expires and returns the best vector
found.
"""

import time
import numpy as np

def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _refine_sa(a: np.ndarray,
               b: np.ndarray,
               cur_score: float,
               deadline: float,
               rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Hill‑climbing / simulated‑annealing that moves mass between two positions."""
    n = a.shape[0]
    best_a = a.copy()
    best_score = cur_score
    start = time.time()
    it = 0
    no_improve = 0
    # parameters for the move size schedule
    total_time = deadline - start
    while time.time() < deadline:
        elapsed = time.time() - start
        # move size slowly shrinks; at the start we allow large jumps
        move_factor = max(0.001, 0.25 * (1.0 - elapsed / total_time))
        # pick a donor with some mass
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        # pick a different recipient
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        # amount to move – a fraction of the donor's mass
        delta = a[i] * rng.random() * move_factor
        if delta <= 0.0:
            continue

        # construct the candidate convolution using the analytical update
        new_b = b.copy()
        # shift contributions of the donor and recipient
        new_b[i:i + n] -= 2.0 * delta * a
        new_b[j:j + n] += 2.0 * delta * a
        # the quadratic Δ∗Δ term
        new_b[2 * i] += delta * delta
        new_b[2 * j] += delta * delta
        new_b[i + j] -= 2.0 * delta * delta

        new_score = new_b.max()
        # accept only if the maximum went down (strict improvement)
        if new_score < cur_score - 1e-12:
            # commit the move
            a[i] -= delta
            a[j] += delta
            b = new_b
            cur_score = new_score
            no_improve = 0
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[SA] iteration {it:7d}  new best = {best_score:.6f}")
        else:
            no_improve += 1
            # occasional ``temperature'' acceptance of worsening moves
            # (a very mild simulated‑annealing step)
            # temperature linearly decays to 0
            T = 0.001 * (1.0 - elapsed / total_time)
            if T > 0 and rng.random() < np.exp(-(new_score - cur_score) / T):
                a[i] -= delta
                a[j] += delta
                b = new_b
                cur_score = new_score
                no_improve = 0

        # if we have been stuck for a while, inject a small random jitter
        if no_improve >= 20000:
            idx = rng.integers(0, n)
            eps = a[idx] * rng.random() * 0.01
            if eps > 0:
                a[idx] -= eps
                target = rng.integers(0, n)
                a[target] += eps
                # recompute the full convolution from scratch – cheap because it happens rarely
                b = np.convolve(a, a)
                cur_score = b.max()
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[SA] jitter improved to {best_score:.6f}")
            no_improve = 0

        it += 1
        # periodic progress report
        if it % 50000 == 0:
            now = time.time()
            print(f"[SA] it={it:,}  time={now - start:.1f}s  cur={cur_score:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises the maximum entry
    of its self‑convolution.  The routine starts from the best known vector

    simulated‑annealing / hill‑climbing refinement that conserves the total
    mass while trying to flatten the convolution.  The best sequence found
    before the budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin for the external evaluation call
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Load a warm‑start, otherwise fall back to a random vector
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            init_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            init_seq = None
    else:
        init_seq = None

    if init_seq is None or init_seq.size == 0:
        # random initial point (length 2000) – later we will upscale
        init_seq = rng.random(2000)
    # ------------------------------------------------------------------
    # Target problem size – larger sizes give a better theoretical floor
    # ------------------------------------------------------------------
    n_target = 4000
    # Upsample (exact repeat if the size divides evenly, otherwise linear)
    if n_target % init_seq.size == 0:
        repeat = n_target // init_seq.size
        a = np.repeat(init_seq, repeat)
    else:
        # linear interpolation fallback
        x_old = np.arange(init_seq.size, dtype=float)
        x_new = np.linspace(0, init_seq.size - 1, n_target, dtype=float)
        a = np.interp(x_new, x_old, init_seq)

    # Make the vector non‑negative and project onto the simplex Σa = √(2·n)
    S = np.sqrt(2.0 * n_target)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full_like(a, 1.0)
    a = a / a.sum() * S

    # Initial convolution and score
    b = np.convolve(a, a)
    cur_score = b.max()
    print(f"[init] n={n_target}, initial max = {cur_score:.6f}")

    # ------------------------------------------------------------------
    # Simulated‑annealing / hill‑climb refinement
    # ------------------------------------------------------------------
    a_best, best_score = _refine_sa(a, b, cur_score, deadline, rng)

    # Final projection to clean up possible tiny numerical drift
    a_best = _project_simplex(a_best, S)
