def full_conv(a):
    n = a.shape[0]
    conv_len = 2*n-1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa*Fa, n=fft_len)[:conv_len]
    return conv

def shape_uniform(n): return np.ones(n)

def shape_reciprocal(n, p):
    idx = np.arange(n)
    base = 1.0/(idx+1) + 1.0/(n-idx)
    return base**p

def shape_product_inv(n, p):
    idx = np.arange(n)
    base = 1.0/((idx+1)*(n-idx))
    return base**p

def shape_sine_power(n, p):
    angles = np.pi * (np.arange(n)+0.5) / n
    base = np.sin(angles)
    return base**p

def shape_geometric(n, rate):
    left = rate**np.arange(n)
    right = left[::-1]
    return (left+right)/2.0

"""
Propose a step‑function (non‑negative vector) with a very low
self‑convolution peak.

Approach
--------
1. **Parametric families** – generate many symmetric shapes
   (uniform, reciprocal, product‑inverse, sine‑power, geometric decay)
   for a selection of lengths.  For each shape a single exponent / rate
   parameter is swept (or sampled) and the vector is normalised to
   Σa = √(2·n).  Because the evaluation metric equals the maximal entry of
   the convolution, we can compute the objective directly with a fast
   FFT convolution, avoiding calls to the black‑box ``evaluate_sequence``.

2. **Local refinement** – starting from the best shape found, a projected
   sub‑gradient descent (multiplicative / KL‑projection) is performed.
   The sub‑gradient corresponds to the worst convolution shift.
   Occasional random mass‑swaps and optional symmetry enforcement help
   escape flat regions.

3. **Multi‑scale up‑sampling** – the refined vector is up‑sampled to a
   slightly larger length (linear interpolation) and refined again.
   Repeating this while time permits often yields a smaller peak.

The routine is time‑aware: the first phase consumes about 30 % of the
budget, the second about 55 %, and the remaining time is spent on the
multi‑scale phase.  The final sequence is returned as a plain list of
floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast full convolution using FFT (linear convolution a*a)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * a (length 2*n‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())      # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex with total mass s."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Symmetrisation helper
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Enforce exact mirror symmetry: v[i] = v[-1-i]."""
    return 0.5 * (v + v[::-1])


# ----------------------------------------------------------------------
# Parametric shape generators (all produce symmetric vectors)
# ----------------------------------------------------------------------
def _shape_uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=float)


def _shape_reciprocal(n: int, p: float) -> np.ndarray:
    """a_i ∝ (1/(i+1) + 1/(n-i))^p."""
    idx = np.arange(n, dtype=float)
    base = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
    return base ** p


def _shape_product_inv(n: int, p: float) -> np.ndarray:
    """a_i ∝ (1/((i+1)*(n-i)))^p."""
    idx = np.arange(n, dtype=float)
    base = 1.0 / ((idx + 1.0) * (n - idx))
    # avoid division by zero at the very ends (should not happen)
    base[base == 0] = np.finfo(float).tiny
    return base ** p


def _shape_sine_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n)^p."""
    theta = np.pi * (np.arange(n, dtype=float) + 0.5) / n
    base = np.sin(theta)
    # sin is strictly positive inside (0,π), but guard against very small values
    base[base <= 0] = np.finfo(float).tiny
    return base ** p


def _shape_geometric(n: int, rate: float) -> np.ndarray:
    """a_i ∝ rate^i + rate^{n‑1‑i}."""
    left = rate ** np.arange(n, dtype=float)
    right = left[::-1]
    return (left + right) / 2.0


# ----------------------------------------------------------------------
# Linear interpolation (upsampling) of an existing sequence
# ----------------------------------------------------------------------
def _upsample_seq(base: np.ndarray, new_n: int) -> np.ndarray:
    if base.size == new_n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Sub‑gradient refinement (multiplicative updates + random swaps)
# ----------------------------------------------------------------------
def _refine_subgradient(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    enforce_sym: bool = False,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = _full_convolution(a).max()

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of max entry
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        # multiplicative (KL‑projection) update
        eta = 0.6 / np.sqrt(it)          # step size schedule
        # Clip gradient to avoid overflow in exponent
        exp_factor = np.exp(-eta * g)
        a = a * exp_factor
        # Renormalise to the simplex
        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0)
            total = a.sum()
        a = a * (S / total)

        # optional symmetry enforcement
        if enforce_sym:
            a = _symmetrize(a)
            a = a * (S / a.sum())

        cur_score = conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional random mass‑swap (simulated‑annealing style)
        if rng.random() < 0.04:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = 0.025 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = a * (S / a.sum())
                if enforce_sym:
                    a = _symmetrize(a)
                    a = a * (S / a.sum())

    return best_a, best_score


# ----------------------------------------------------------------------

    """
    Search for a non‑negative step‑function whose self‑convolution peak is

    # A few representative lengths (the optimum is often around 8–12 k)

    families = ["uniform", "reciprocal", "product_inv", "sine", "geometric"]

    while time.time() < phase1_deadline:

            if fam == "reciprocal":

                for p in p_vals:

                    a = a / a.sum() * S
                    score = _eval_vector(a)
                    if score < best_score:

            elif fam == "product_inv":
                # exponent p ∈ [0.1, 4.0]
                p_vals = np.linspace(0.1, 4.0, 30) + rng.normal(scale=0.03, size=30)
                for p in p_vals:
                    a = _shape_product_inv(n, max(p, 0.01))
                    a = a / a.sum() * S
                    score = _eval_vector(a)
                    if score < best_score:

            elif fam == "sine":

                    a = a / a.sum() * S
                    score = _eval_vector(a)
                    if score < best_score:
                        best_score, best_seq, best_n = score, a.copy(), n
            elif fam == "geometric":

                for rate in rate_vals:
                    a = _shape_geometric(n, rate)
                    a = a / a.sum() * S
                    score = _eval_vector(a)
                    if score < best_score:
                        best_score, best_seq, best_n = score, a.copy(), n

    # Phase 2 – local sub‑gradient refinement (≈55 % of budget)
    # ------------------------------------------------------------------
