"""
Propose a step‑function (non‑negative sequence) with a very small
max‑convolution value.

The algorithm works in three stages for each candidate length:

1. **Initialisation** – a random Dirichlet vector, a power‑law or a
   short exponential tail.  The vector is projected onto the ℓ¹‑simplex
   with Σa = √(2·n) (the objective is scale‑invariant, this normalisation
   only stabilises the optimiser).

2. **Flattening (variance reduction)** – minimise the variance of the
   convolution `c = a*a`.  The smooth objective
       J(a) = ∑ₖ (cₖ - μ)²   (μ = Σa²/(2n-1))
   has gradient
       ∇J = 2 * (a_rev  * (c-μ))               (linear convolution)
   which is evaluated by FFTs.  Projected gradient steps drive the
   convolution towards a flat profile, thereby decreasing its peak.

3. **Peak‑reduction** – a projected sub‑gradient descent on the
   *actual* max‑convolution.  For the index `k*` attaining the current
   maximum, a sub‑gradient is
       gᵢ = 2·a_{k*‑i}   (zero outside the valid range).
   Updating `a ← a - η·g` and re‑projecting reduces the worst entry.

Both stages are run repeatedly within the time budget; the best
sequence over all tried lengths is kept.  A short simulated‑annealing
refinement is performed at the end to escape possible shallow minima.

The whole search respects the 1000 s limit.  All helper functions are
top‑level, use only NumPy and the provided `evaluate_sequence`, and no
I/O is performed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, Σx=s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Stage 1 – Flattening (variance reduction)
# ----------------------------------------------------------------------


def _flatten_gradient(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Compute gradient of the variance of the convolution and also
    return the current max‑convolution value.
    """
    n = a.shape[0]
    # full convolution a*a
    c = _conv_fft(a)
    max_c = float(c.max())

    # desired flat level (the average of the convolution)
    S = a.sum()
    mu = S * S / (2 * n - 1)               # = mean(c)

    # diff = c - mu
    d = c - mu

    # gradient = 2 * (a_rev * d)
    rev_a = a[::-1]
    size = 1 << ((rev_a.shape[0] + d.shape[0] - 1).bit_length())
    rev_fft = np.fft.rfft(rev_a, n=size)
    d_fft = np.fft.rfft(d, n=size)
    grad_full = np.fft.irfft(rev_fft * d_fft, n=size)

    # extract the slice that corresponds to the gradient (length n)
    start = rev_a.shape[0] - 1            # = n-1
    grad = 2.0 * grad_full[start:start + n]
    return grad, max_c


def _flatten_descent(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the variance of the convolution.
    Stops when time runs out.
    Returns the best sequence seen during the run and its max‑convolution.
    """
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)

    # ensure feasibility from the start
    a = _project_to_simplex(a, target)

    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    rng = np.random.default_rng()
    it = 0
    # step‑size schedule ηₜ = α / √(t+1)
    alpha = target * 0.2

    while time.time() < deadline:
        g, cur_max = _flatten_gradient(a)
        norm = np.linalg.norm(g)
        if norm > 0.0:
            eta = alpha / (np.sqrt(it + 1) * norm)
            a -= eta * g
            a = _project_to_simplex(a, target)

        # occasionally add a tiny random jitter to avoid stagnation
        if it % 400 == 0:
            jitter = rng.normal(scale=target * 1e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] flatten iter {it:5d}  max={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 2 – Sub‑gradient descent on the actual max‑convolution
# ----------------------------------------------------------------------


def _subgradient(a: np.ndarray, k_star: int) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution at index k_star.
    g_i = 2 * a_{k_star-i} for all admissible i.
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    lo = max(0, k_star - (n - 1))
    hi = min(k_star, n - 1)

    idx = np.arange(lo, hi + 1, dtype=np.int64)
    g[idx] = 2.0 * a[k_star - idx]
    return g


def _max_descent(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the max‑convolution value.
    """
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)
    a = _project_to_simplex(a, target)

    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    rng = np.random.default_rng()
    it = 0
    alpha = target * 0.15                     # base step size

    while time.time() < deadline:
        c = _conv_fft(a)
        k_star = int(np.argmax(c))
        cur_max = float(c[k_star])

        g = _subgradient(a, k_star)
        norm = np.linalg.norm(g)
        if norm > 0.0:
            eta = alpha / (np.sqrt(it + 1) * norm)
            a -= eta * g
            a = _project_to_simplex(a, target)

        # occasional jitter to escape plateaux
        if it % 300 == 0:
            jitter = rng.normal(scale=target * 5e-5, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] max‑gd iter {it:5d}  max={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 3 – Light simulated‑annealing polish
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Very simple SA that transfers a tiny amount of mass from a
    “high‑impact” index to a “low‑impact” one.
    """
    a = a0.copy()
    n = a.size
    cur_val = _max_convolution_fft(a)

    best_a = a.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    start = time.time()
    total = max(deadline - start, 1e-6)

    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / total
        temperature = 0.4 * (0.01 / (tau + 1e-12))   # fast cooling

        # locate the current worst convolution entry
        c = _conv_fft(a)
        k = int(np.argmax(c))

        # contribution weights for the worst entry
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[k - idx]

        if contrib.sum() == 0.0:
            donor = rng.integers(n)
        else:
            donor = rng.choice(idx, p=contrib / contrib.sum())

        # choose a receiver far from the donor in terms of contribution
        low_thr = np.percentile(contrib, 20)
        low_idx = idx[contrib <= low_thr]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # move a small fraction of mass
        delta = a[donor] * rng.random() * 0.04
        a[donor] -= delta
        a[receiver] += delta

        # re‑project onto the simplex (fast, keeps sum constant)
        a = _project_to_simplex(a, np.sqrt(2.0 * n))

        new_val = _max_convolution_fft(a)
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA new best {best_val:.6f}")

        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta

    return best_a, best_val


# ----------------------------------------------------------------------
# main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises the evaluation function.
    The routine runs several independent optimisation attempts at
    different lengths, each consisting of:
        * flattening (variance reduction) → max‑gradient descent → SA
    The best sequence found before the budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5        # small safety margin

    best_seq = None
    best_score = float("inf")

    # problem sizes – we allocate more time to the larger instances
    sizes = [800, 1500, 3000, 5000]

    for n in sizes:
        if time.time() >= deadline - 5.0:
            break

        # allocate a slice of the remaining budget to this size
        remaining = deadline - time.time()
        slice_time = min(remaining * 0.35, 250.0)

        # pick a random initialisation strategy
        init_mode = rng.choice(["dirichlet", "power", "exponential"])

        # build the initial vector
        target_sum = np.sqrt(2.0 * n)
        if init_mode == "dirichlet":
            seq = rng.random(n)
            seq = seq / seq.sum() * target_sum
        elif init_mode == "power":
            p = rng.uniform(0.3, 0.9)
            seq = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
            seq = seq * (target_sum / seq.sum())
        else:   # exponential tail
            # a_i = C * (1 - α/n)^i   with α∈[0.8,4.5]
            alpha = rng.uniform(0.8, 4.5)
            r = 1.0 - alpha / n
            if r <= 0.0:
                r = 0.5
            base = r ** np.arange(n, dtype=np.float64)
            seq = base
            seq = seq * (target_sum / seq.sum())

        seq = _project_to_simplex(seq, target_sum)

        # ----- stage 1 : flattening -----
        flatten_deadline = time.time() + max(5.0, slice_time * 0.35)
        seq_f, score_f = _flatten_descent(seq, flatten_deadline)

        # ----- stage 2 : max‑gradient descent -----
        maxgd_deadline = time.time() + max(5.0, slice_time * 0.30)
        seq_m, score_m = _max_descent(seq_f, maxgd_deadline)

        # ----- stage 3 : simulated annealing -----
        sa_deadline = time.time() + max(5.0, slice_time * 0.30)
        seq_s, score_s = _simulated_annealing(seq_m, sa_deadline)

        # pick the best of the three
        final_seq, final_score = min(
            [(seq_f, score_f), (seq_m, score_m), (seq_s, score_s)],
            key=lambda x: x[1],
        )

        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(
                f"[{time.time():.1f}] size={n} init={init_mode} → "
                f"new global best {best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # final polishing of the best candidate (if any time left)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        final_deadline = time.time() + (deadline - time.time()) * 0.25
        best_seq, _ = _max_descent(best_seq, final_deadline)

    # strip trailing zeros (they do not influence the score)
    if best_seq is not None:
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[: nz[-1] + 1]
        else:
            best_seq = np.array([], dtype=float)

    print(f"[{time.time():.1f}] final best score≈{best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
