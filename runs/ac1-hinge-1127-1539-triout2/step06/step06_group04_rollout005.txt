"""
Hybrid optimiser for the self‑convolution max‑ratio problem.

The core idea is to minimise the ℓ₂‑error between the convolution
a⋆a and the flat target (its average value) which directly
encourages a nearly constant autocorrelation.  A projected
gradient descent on this smooth objective is combined with a fast
targeted hill‑climber (exact incremental convolution updates) for
fine tuning the ℓ∞‑peak.  The search runs a length schedule,
upsampling the current best solution when the dimension is
increased, and respects the supplied wall‑clock budget (default
1000 s).  The best sequence found is returned as a plain list of
floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Duchi‑et‑al. (2008) algorithm.
    """
    n = v.shape[0]
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Gradient of the ℓ₂‑error objective (constant‑target)
# ----------------------------------------------------------------------


def _gradient_l2_error(a: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Return the gradient of   L(a) = Σ_k ( (a*a)_k - μ )²,
    where μ = target_sum² / (2n‑1) is the flat target.
    The gradient is 2·(a_rev ⋆ (b-μ)) restricted to the original n entries.
    """
    n = a.shape[0]

    # full convolution b = a*a
    size = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # flat target μ (average of b)
    mu = (target_sum * target_sum) / (2 * n - 1)
    c = b - mu                     # shape (2n‑1,)

    # correlation g_i = 2 * Σ_k c_{i+k} a_k
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1            # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    C_fft = np.fft.rfft(c, n=size2)
    corr = np.fft.irfft(Ar_fft * C_fft, n=size2)[:conv_len]

    g = 2.0 * corr[n - 1: n - 1 + n]           # slice of length n
    return g


def _gradient_descent_l2(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the ℓ₂‑error objective.
    Uses a simple Nesterov‑style momentum and occasional jitter
    to escape plateaux.
    """
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)

    # hyper‑parameters (empirically chosen)
    step_base = target * 0.12
    rho = 0.92                       # momentum decay
    momentum = np.zeros_like(a)

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g = _gradient_l2_error(a, target)

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m + 1e-12)
            a -= lr * momentum

        # projection onto the ℓ¹‑simplex Σa = target
        a = _project_to_simplex(a, target)

        # jitter every few hundred iterations
        if it % 250 == 0:
            jitter = rng.normal(scale=target * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

        it += 1
    return a

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = np.sqrt(2.0 * n)

    # pick a random “family’’ – a few simple parametric shapes
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                               # power‑law
        p = rng.uniform(0.05, 1.3)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                             # exponential decay
        lam = rng.exponential(scale=0.02)
        a = np.exp(-lam * idx)

    elif family == 2:                             # mixed power‑law × exponential
        p = rng.uniform(0.1, 1.0)
        lam = rng.exponential(scale=0.02)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                             # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                             # piecewise exponential
        cut = rng.integers(int(0.2 * n), int(0.8 * n))
        lam1 = rng.exponential(scale=0.025)
        lam2 = rng.exponential(scale=0.025)
        a = np.concatenate([np.exp(-lam1 * np.arange(cut)),
                            np.exp(-lam2 * np.arange(cut, n))])

    else:                                         # small Gaussian bump
        centre = rng.uniform(0, 1)
        width = rng.exponential(scale=0.12) + 0.04
        a = np.exp(-((idx / n - centre) ** 2) / (2 * width ** 2))

    # tiny noise, enforce non‑negativity and project to simplex
    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

def _hill_climb_targeted(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Simple hill‑climber that always looks at the current maximal convolution entry.
    Moves a small amount of mass from an index that strongly contributes to the peak
    toward an index that contributes little.
    Returns the best sequence seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    T = cur_max * 0.02 + 1e-9          # initial temperature
    cooling = 0.9995

    while time.time() < deadline:
        # index of the current peak
        k_max = int(np.argmax(b))

        # contribution of each index to that peak:  a[i] * a[k_max-i]
        j_arr = k_max - np.arange(n)
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[mask] * a[j_arr[mask]]

        # source (high contribution) – weighted by contribution
        total = contrib.sum()
        if total > 0.0:
            prob_src = contrib / total
            i = rng.choice(n, p=prob_src)
        else:
            prob_src = a / a.sum()
            i = rng.choice(n, p=prob_src)

        # destination (low contribution) – weighted towards small contribution
        inv = contrib.max() - contrib + 1e-12
        prob_dst = inv / inv.sum()
        j = rng.choice(n, p=prob_dst)

        if i == j:
            continue

        # amount of mass to move (up to 50 % of source)
        max_delta = a[i] * 0.5
        if max_delta < 1e-16:
            continue
        delta = rng.random() * max_delta

        # ---- analytical update of the convolution (O(n)) ----
        b_new = b.copy()
        # linear terms: -2·δ·a shifted by i  + 2·δ·a shifted by j
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a

        # quadratic correction terms
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        # ---- acceptance test ----
        accept = False
        if new_max < cur_max - 1e-12:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            a[i] = max(a[i], 0.0)
            a[j] = max(a[j], 0.0)

            b = b_new
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_targeted improvement: {best_max:.6f}")

        T *= cooling

    return best_a, best_max

        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        best_seq = a0.copy()
        best_score = _max_convolution_fft(a0)
        print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) "
              f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Length schedule – start modest and enlarge gradually.
    # ------------------------------------------------------------------
    lengths = [2000, 5000, 8000, 12000, 18000, 25000]

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)

        # ------------------------------------------------------------------
        # Initialise candidate for this length.
        # ------------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)

        a_init = _project_to_simplex(a_init, target_sum)

        # ------------------------------------------------------------------
        # Phase A – projected gradient descent on the ℓ₂‑error (main work).
        # ------------------------------------------------------------------
        time_left = deadline - time.time()
        phaseA_dead = time.time() + min(120.0, 0.55 * time_left)
        a_gd = _gradient_descent_l2(a_init, phaseA_dead)

        score_gd = _max_convolution_fft(a_gd)
        if score_gd < best_score:
            best_score = score_gd
            best_seq = a_gd.copy()
            print(f"[{time.time():.1f}] GD ℓ₂ improvement (n={n}) score={score_gd:.6f}")

        # ------------------------------------------------------------------
        # Phase B – targeted hill‑climbing to polish the ℓ∞‑peak.
        # ------------------------------------------------------------------
        if time.time() < deadline:
            b_full = _full_convolution(a_gd)
            time_left = deadline - time.time()
            phaseB_dead = time.time() + min(80.0, 0.35 * time_left)
            a_hc, max_hc = _hill_climb_targeted(a_gd, b_full, phaseB_dead)
            if max_hc < best_score:
                best_score = max_hc
                best_seq = a_hc.copy()
                print(f"[{time.time():.1f}] Hill‑climb improvement (n={n}) "
                      f"score={max_hc:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with any remaining budget.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a_final = best_seq.copy()
        b_final = _full_convolution(a_final)
        a_final, max_final = _hill_climb_targeted(a_final, b_final, deadline)
        if max_final < best_score:
            best_score = max_final
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final polishing improvement: "
                  f"score={max_final:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries for a neat output.
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform sequence of moderate size
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log the official evaluation value (if evaluate_sequence is available).
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:   # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
