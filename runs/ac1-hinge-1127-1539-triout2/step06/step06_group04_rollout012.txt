def _gradient_and_convolution(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = _next_pow2(2*n-1)
    a_fft = np.fft.rfft(a, n=size)
    b_full = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]

    max_b = np.max(b_full)
    exps = np.exp(beta * (b_full - max_b))
    # avoid overflow: subtract max_b
    sum_exps = np.sum(exps)
    if sum_exps == 0.0:
        # handle extreme case: all exps zero? then p uniform?
        p = np.full_like(b_full, 1.0 / b_full.size)
    else:
        p = exps / sum_exps

    a_rev_fft = np.fft.rfft(a_rev, n=size2)

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Search for a non‑negative step function minimizing the
    ``evaluate_sequence`` objective using Adam‑based smoothed max descent and
    targeted hill‑climbing.  The routine respects the supplied wall‑clock
    budget (default 1000 s) and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5

    # initialization from known sequence or random
    seed_seq = globals().get('height_sequence_1')
    if isinstance(seed_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(seed_seq, dtype=float)
    else:
        a0 = None

    # target length
    target_n = 20000  # can be tuned
    if a0 is not None:
        n0 = a0.size
        if n0 != target_n:
            a0 = _upsample_sequence(a0, target_n)
    else:
        # random draw
        a0 = _random_initial_sequence(target_n, rng)

    # ensure simplex projection
    target_sum = np.sqrt(2.0 * target_n)
    a0 = _project_to_simplex(a0, target_sum)

    # Phase 1: Adam gradient descent (≈ 65 % of time)
    time_left = deadline - time.time()
    gd_dead = time.time() + max(5.0, 0.65 * time_left)
    best_a, best_max = _adam_gradient_descent(a0, gd_dead)

    # Phase 2: targeted hill‑climbing (≈ 15 % of time)
    if time.time() < deadline:
        b_full = _full_convolution(best_a)
        hc_dead = time.time() + max(2.0, 0.15 * (deadline - time.time()))
        a_hc, max_hc = _hill_climb_targeted(best_a, b_full, hc_dead)
        if max_hc < best_max:
            best_a, best_max = a_hc, max_hc
            print(f"[{time.time():.1f}] Hill‑climb improvement: {best_max:.6f}")

    # Phase 3: second round of Adam descent (≈ 15 % of time)
    if time.time() < deadline:
        gd2_dead = time.time() + max(2.0, 0.15 * (deadline - time.time()))
        a_gd2, max_gd2 = _adam_gradient_descent(best_a, gd2_dead)
        if max_gd2 < best_max:
            best_a, best_max = a_gd2, max_gd2
            print(f"[{time.time():.1f}] Second GD improvement: {best_max:.6f}")

    # Phase 4: final hill‑climb (remaining time)
    if time.time() < deadline - 2.0:
        b_full = _full_convolution(best_a)
        a_final, max_final = _hill_climb_targeted(best_a, b_full, deadline)
        if max_final < best_max:
            best_a, best_max = a_final, max_final
            print(f"[{time.time():.1f}] Final hill‑climb improvement: {best_max:.6f}")

    # Trim zeros
    nz = np.where(best_a > 1e-12)[0]
    if nz.size:
        best_a = best_a[: nz[-1] + 1]
    else:
        best_a = best_a[:1]

    # Evaluate
    try:
        val = evaluate_sequence(best_a.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_a.size})")
    except Exception as exc:
        print(f"Error calling evaluate_sequence: {exc}")

    return best_a.tolist()

"""
Search for a non‑negative step function that minimises the
``evaluate_sequence`` objective.

The algorithm consists of three stages:

1. **Adam‑based smoothed‑max descent** – a projected Adam optimiser
   minimises a log‑sum‑exp surrogate of the convolution maximum.  The
   surrogate is gradually sharpened (β‑annealing) while the iterate is
   always projected onto the ℓ¹‑simplex Σa = √(2 n).

2. **Targeted hill‑climbing** – given the current peak of the exact
   self‑convolution, a small amount of mass is transferred from an
   index that contributes heavily to that peak towards an index that
   contributes little.  The convolution update is performed analytically,
   allowing many cheap Metropolis‑style moves.

3. **A second Adam pass and a final hill‑climb** – the remaining budget
   is spent refining the solution.

The routine respects the supplied wall‑clock budget (default 1000 s) and
returns the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Duchi‑et‑al. (2008) algorithm.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_and_convolution(a: np.ndarray, beta: float):
    """
    Returns (grad, conv) where:

    * conv is the full linear self‑convolution a*a (length 2 n‑1).
    * grad = ∇_a ( (1/β)·log Σ_k exp(β·conv_k) ), i.e. 2·(a ⋆ p)
      with p_k ∝ exp(β·conv_k).
    """
    n = a.shape[0]

    # ---------- convolution ----------
    size = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # ---------- soft‑max probabilities ----------
    max_c = np.max(conv)
    exps = np.exp(beta * (conv - max_c))
    sum_exps = np.sum(exps)
    if sum_exps == 0.0:
        p = np.full_like(conv, 1.0 / conv.size)
    else:
        p = exps / sum_exps

    # ---------- gradient ----------
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3 n‑2
    size2 = _next_pow2(conv_len)
    a_rev_fft = np.fft.rfft(a_rev, n=size2)
    p_fft = np.fft.rfft(p, n=size2)
    corr = np.fft.irfft(a_rev_fft * p_fft, n=size2)[:conv_len]
    grad = 2.0 * corr[n - 1 : n - 1 + n]

    return grad, conv


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2 n‑1)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]
    return conv


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 7)

    idx = np.arange(n, dtype=float)

    if family == 0:                       # pure power‑law
        p = rng.uniform(0.10, 1.2)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                     # exponential decay
        lam = rng.exponential(scale=0.018)
        a = np.exp(-lam * idx)

    elif family == 2:                     # mixed power‑law × exponential
        p = rng.uniform(0.12, 1.0)
        lam = rng.exponential(scale=0.018)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                     # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                     # piecewise exponential
        cut = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(cut)),
             np.exp(-lam2 * np.arange(cut, n))]
        )

    elif family == 5:                     # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    else:                                 # random Gaussian‑like bump
        centre = rng.uniform(0, 1)
        width = rng.exponential(scale=0.15) + 0.05
        a = np.exp(-((idx / n - centre) ** 2) / (2 * width ** 2))

    # small Gaussian noise and projection
    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a

# Adam‑based smoothed‑max descent
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation of the log‑sum‑exp surrogate of the maximal
    self‑convolution entry.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = math.sqrt(2.0 * n)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    alpha = 0.5                     # base step size
    it = 1

    # β‑annealing for the smooth max
    beta = 0.5
    beta_max = 4096.0
    beta_factor = 1.02

    rng = np.random.default_rng()

    # initial score
    _, conv = _gradient_and_convolution(a, beta)
    best_max = conv.max()
    best_a = a.copy()
    print(f"[{time.time():.1f}] GD start – init max = {best_max:.6f}")

    while time.time() < deadline:
        # gradient & current convolution
        grad, conv = _gradient_and_convolution(a, beta)
        cur_max = conv.max()
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
            print(f"[{time.time():.1f}] GD iter {it} – new best max = {best_max:.6f}")

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # keep feasibility
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β‑annealing
        if beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1

    return best_a, best_max

def _hill_climb_targeted(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Hill‑climbing that always attacks the current maximal convolution entry.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size

    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    T = cur_max * 0.02 + 1e-9          # starting temperature
    cooling = 0.9998

    while time.time() < deadline:
        k_max = int(np.argmax(b))

        # contributions of each index to the peak at k_max
        idx = np.arange(n)
        j_arr = k_max - idx
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[mask] * a[j_arr[mask]]

        total = contrib.sum()
        if total > 0.0:
            prob_src = contrib / total
            src = rng.choice(n, p=prob_src)
        else:
            prob_src = a / a.sum()
            src = rng.choice(n, p=prob_src)

        # destination – favour small contribution
        inv = contrib.max() - contrib + 1e-12
        prob_dst = inv / inv.sum()
        dst = rng.choice(n, p=prob_dst)

        if src == dst:
            continue

        # move at most 30 % of the source mass
        max_delta = a[src] * 0.3
        if max_delta < 1e-16:
            continue
        delta = rng.random() * max_delta

        # analytical update of the convolution
        b_new = b.copy()
        b_new[src:src + n] -= 2.0 * delta * a
        b_new[dst:dst + n] += 2.0 * delta * a

        # quadratic correction terms
        k2i = 2 * src
        k2j = 2 * dst
        kij = src + dst
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        # acceptance test (always accept improvement, occasional worsening)
        accept = False
        if new_max < cur_max - 1e-12:

        else:

            if cur_max < best_max:

                print(f"[{time.time():.1f}] hill_targeted improvement: {best_max:.6f}")

    # Initialise from a known construction if available
    # ------------------------------------------------------------------
