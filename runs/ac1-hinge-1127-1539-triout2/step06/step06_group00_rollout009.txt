def spectral_flatten(a, iters=5):
    n=len(a)
    S=np.sqrt(2.0*n)

def _lp_refine(a, deadline, rng):
    # returns refined a
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = np.maximum(a, 0.0)
    # ensure sum
    if a.sum() == 0:
        a = np.full(n, S / n)
    else:
        a = a / a.sum() * S

    best_a = a.copy()
    best_M = _full_convolution_fft(best_a).max()
    t = 0.2

    # Precompute convolution matrix for the current a maybe updated each iteration
    while time.time() + 0.5 < deadline:
        # Build convolution matrix for current a
        Fmat = _convolution_matrix(a)  # shape (2n-1,n)

        # Solve LP: maximize sum(g) subject to Fmat @ g <= best_M
        # with g >= 0
        c = -np.ones(n)
        b_ub = np.full(Fmat.shape[0], best_M)
        bounds = [(0, None) for _ in range(n)]

        # attempt solution
        try:
            from scipy.optimize import linprog
            res = linprog(c, A_ub=Fmat, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'disp': False, 'presolve': True})
        except Exception as e:
            # If LP fails, break
            print("LP solve failed:", e)
            break

        if not res.success:
            # No feasible improvement
            break

        g = res.x
        sum_g = g.sum()
        if sum_g == 0:
            break
        # Scale to correct L1 norm
        g = g * (S / sum_g)
        # Blend
        a_new = (1 - t) * a + t * g
        # Ensure nonnegative and renorm (just in case)
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum() * S

        M_new = _full_convolution_fft(a_new).max()

        if M_new < best_M - 1e-9:
            best_M = M_new
            best_a = a_new.copy()
            a = a_new
            # optionally adjust step size
            t = max(0.05, t * 0.9)

            # Might also reduce M for next LP to the new value
        else:
            # reduce step size, or break
            t *= 0.8
            if t < 0.01:
                break
    return best_a, best_M

def _edge_initial(n, rng):
    S = np.sqrt(2.0 * n)
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.05, 2.0)    # broaden exponent
    offset = rng.uniform(0.0, 30.0)  # broaden offset
    mix = rng.uniform(0.0, 1.0)   # mixture
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1 - mix) * base
    raw = np.clip(raw, 0.0, None)
    raw = raw / raw.sum() * S
    return raw

def _exp_initial(n, rng):
    S = np.sqrt(2.0 * n)
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    alpha = rng.uniform(0.001, 0.05)  # small decay for smoother shape
    exp_part = np.exp(-alpha * d)
    base = np.ones(n)
    mix = rng.uniform(0.0, 1.0)
    raw = mix * exp_part + (1 - mix) * base
    raw = np.clip(raw, 0.0, None)
    raw = raw / raw.sum() * S
    return raw

"""
Step‑function optimizer (≈ 1 500 ms budget → ≈ 1 503 target)

The search consists of three independent phases:

1. **Diverse random restarts** – a fast Adam optimiser (log‑softmax
   parametrisation) is run from many initial vectors.
   Initialisations include uniform, Dirichlet, edge‑heavy power‑law and
   exponential shapes with a very wide range of parameters.  The best
   candidate (its convolution peak = internal score) is kept.

2. **Linear‑programming mass re‑allocation** – starting from the current
   best vector, we repeatedly solve a small LP:
        maximise   Σ g_j
        subject to   (f * g)_k ≤ M   (for every convolution index)
                     g_j ≥ 0,
   where `M` is the current maximal convolution value.
   The solution `g` can carry strictly more L1‑mass while respecting the
   same peak bound.  After scaling `g` to the required ℓ₁‑norm we blend it
   with `f`.  The peak is recomputed and the process repeats, steadily
   lowering `M`.

3. **Final short Adam polishing** – the vector obtained from the LP
   phase is refined once more with Adam to squeeze out the last bit of
   improvement.

All vectors are projected onto the simplex (non‑negative, ℓ₁‑norm
√(2 n)) after every operation, guaranteeing that the external
`evaluate_sequence` sees a feasible step function.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Low‑level utilities
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.ceil(pos).astype(int)
    hi = np.clip(hi, 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# --------------------------------------------------------------
# Gradient of the τ‑soft‑max approximation of max(b) where b = a*a
# --------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """Return (grad, max_b) where grad = ∂ τ‑soft‑max(b) / ∂a."""
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        w = exps / np.sum(exps)
        mask = w

    # gradient w.r.t a: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


# --------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# --------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Minimise the τ‑soft‑max approximation of the convolution peak.
    Returns the best vector ever seen (projected on the simplex) and the
    *hard* peak (max of the exact convolution) corresponding to it.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)          # required ℓ₁‑norm

    # initialise logits so that softmax(p) * S == a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)               # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.3                       # a bit larger than the reference value

    # initialise temperature for soft‑max
    tau = max(_full_convolution_fft(a0).max(), 1.0)

    best_a = a0.copy()
    best_peak = _full_convolution_fft(a0).max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ---- recover a from logits (log‑softmax) ----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ---- convolution & gradient (soft‑max) ----
        g, peak = _softmax_gradient(a, tau)

        # keep best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ---- gradient w.r.t. logits (chain rule) ----
        mu = np.dot(a, g) / S           # ⟨a,∂b/∂a⟩/S
        grad_p = a * (g - mu)           # shape (n,)

        # optional gradient clipping
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- schedules ----
        tau *= 0.9997               # tighter soft‑max
        lr *= 0.9999                # very slow decay

        # occasional jitter to escape plateaus
        if it % 800 == 0:
            p += rng.normal(scale=0.015, size=n)

    return best_a, best_peak

# Parameterised edge‑heavy and exponential initial shapes
# --------------------------------------------------------------
def _edge_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Power‑law edge‑heavy shape with very wide parameter ranges."""
    S = np.sqrt(2.0 * n)
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0          # distance to the nearer edge (1‑based)

    # broadened random parameters
    p = rng.uniform(0.05, 2.0)          # exponent
    offset = rng.uniform(0.0, 30.0)     # denominator shift
    mix = rng.uniform(0.0, 1.0)         # mixture with a flat background

    edge_part = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge_part + (1.0 - mix) * base
    raw = np.clip(raw, 0.0, None)
    raw = raw / raw.sum() * S
    return raw


def _exp_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Exponential decay from the edges, again mixed with a uniform part."""
    S = np.sqrt(2.0 * n)
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0

    alpha = rng.uniform(0.001, 0.05)   # decay rate
    mix = rng.uniform(0.0, 1.0)

    exp_part = np.exp(-alpha * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    raw = np.clip(raw, 0.0, None)
    raw = raw / raw.sum() * S
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_start: np.ndarray | None) -> np.ndarray:
    """Return a feasible initial vector of length *n*."""
    # occasional warm‑start from an existing construction
    if warm_start is not None and rng.random() < 0.12:
        a = _scale_sequence(warm_start, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * np.sqrt(2.0 * n)
        return a

    style = rng.choice(['uniform', 'dirichlet', 'edge', 'exp'])
    S = np.sqrt(2.0 * n)

    if style == 'uniform':
        a = np.full(n, S / n)
    elif style == 'dirichlet':
        a = rng.random(n)
        a = a / a.sum() * S
    elif style == 'edge':
        a = _edge_initial(n, rng)
    else:   # exponential
        a = _exp_initial(n, rng)

    return a


# --------------------------------------------------------------
# Convolution matrix (dense) – used for the LP formulation
# --------------------------------------------------------------
def _convolution_matrix(f: np.ndarray) -> np.ndarray:
    """Return the dense matrix M such that M @ g == conv(f, g)."""
    n = len(f)
    total = 2 * n - 1
    idx = np.arange(total)[:, None] - np.arange(n)[None, :]   # shape (total, n)
    mask = (idx >= 0) & (idx < n)
    M = np.zeros((total, n), dtype=float)
    M[mask] = f[idx[mask]]
    return M


# --------------------------------------------------------------
# Linear‑programming based refinement
# --------------------------------------------------------------
def _lp_refine(a: np.ndarray,
               deadline: float,
               rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Perform a few LP updates that move mass while never raising the
    current peak.  Returns an improved sequence and its new peak.
    """
    try:
        from scipy.optimize import linprog
    except Exception:   # pragma: no cover
        # SciPy not available – skip the LP phase
        return a, _full_convolution_fft(a).max()

    n = len(a)
    S = np.sqrt(2.0 * n)

    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full(n, S / n)
    else:
        a = a / a.sum() * S

    best_a = a.copy()
    best_M = _full_convolution_fft(best_a).max()
    t = 0.25                     # blending factor

    while time.time() + 0.5 < deadline:
        # Build the convolution matrix for the *current* a
        Fmat = _convolution_matrix(a)                 # (2n‑1) × n

        # Linear programme:
        #   maximise   Σ g_j   ⇔ minimise   -Σ g_j
        c = -np.ones(n)
        b_ub = np.full(Fmat.shape[0], best_M)         # all entries ≤ current peak
        bounds = [(0, None) for _ in range(n)]

        res = linprog(c, A_ub=Fmat, b_ub=b_ub,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'disp': False})

        if not res.success:
            break

        g = res.x
        if g.sum() == 0:
            break
        g = g * (S / g.sum())                         # enforce the exact ℓ₁‑norm

        # Blend a tiny amount of the new direction
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum() * S

        M_new = _full_convolution_fft(a_new).max()

        if M_new < best_M - 1e-9:
            best_M = M_new
            best_a = a_new.copy()
            a = a_new
            # tighten the LP bound for the next iteration
            t = max(0.05, t * 0.9)
        else:
            # no progress – shrink the step size
            t *= 0.8
            if t < 0.01:
                break

    return best_a, best_M
