"""
Propose a step‑function (non‑negative vector) that minimises the
maximum entry of its self‑convolution.  The algorithm works in three
stages (increasing length).  For each length we alternately:

1.  Compute the current worst convolution value M.
2.  Solve a linear program that **maximises the total mass** of a new
    vector `b` under the linear constraints `conv(f, b) ≤ M`.
    (The convolution is linear in `b`; the constraints are therefore
    linear.  The LP variables are the entries of `b`.)
3.  Scale `b` to the required L¹‑norm and move a fraction `t` towards it
    (`f ← (1‑t)·f + t·b`).  The sum of the entries stays fixed, so the
    evaluation score equals the maximum convolution value.

Repeating this “best‑response” step quickly drives the maximal
convolution down.  After the LP phase we optionally run a short
gradient‑based refinement (the Adam/sub‑gradient code from the previous
baseline) to escape possible plateaus.

The whole search respects the 1000 s budget and prints progress
information.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# Scipy is allowed and provides a fast LP solver.
from scipy.optimize import linprog
import scipy.sparse as sp


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (full linear convolution via FFT)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = rfft(a, n=fft_len)
    conv = irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Upsampling helper (linear interpolation)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a to length new_n."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=float)
    return np.interp(x_new, x_old, a)


# ----------------------------------------------------------------------
# Pre‑compute sparse structure for the convolution matrix A_f
# (A_f @ b = conv(f, b) )
# ----------------------------------------------------------------------
def _precompute_sparse_structure(n: int):
    """
    Return row and column index arrays for the matrix that maps b → conv(f,b).
    For each column j the rows i = j … j+n‑1 get the coefficient f_j.
    """
    # rows = j + k,   k = 0..n‑1  (flattened)
    row_idx = np.tile(np.arange(n), n) + np.repeat(np.arange(n), n)
    col_idx = np.repeat(np.arange(n), n)
    return row_idx.astype(np.int32), col_idx.astype(np.int32)


# ----------------------------------------------------------------------
# Linear‑programming based improvement for a fixed length
# ----------------------------------------------------------------------
def _optimize_lp_stage(f0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       mix_frac: float = 0.25,
                       max_iter: int = 80) -> tuple[np.ndarray, float]:
    """
    Repeatedly:
        * compute M = max(conv(f,f))
        * solve   maximise Σ b_i   s.t.   conv(f, b) ≤ M ,  b ≥ 0
        * scale b to the target L¹‑norm and move a fraction `mix_frac`
          towards it.
    Returns the best vector found and its score (= max(conv)).
    """
    n = f0.shape[0]
    S = np.sqrt(2.0 * n)          # required L¹‑norm
    f = _project_simplex(f0.copy(), S)

    # pre‑computed sparsity pattern for the convolution matrix
    row_idx, col_idx = _precompute_sparse_structure(n)

    best_f = f.copy()
    best_score = np.max(_full_convolution(f))

    it = 0
    while time.time() < deadline - 0.1 and it < max_iter:
        it += 1

        # current worst entry
        conv_f = _full_convolution(f)
        M = conv_f.max()

        # ----- linear program : maximise sum(b)  subject to conv(f,b) ≤ M -----
        # A_f @ b  produces conv(f,b)
        data = np.repeat(f, n)                     # length n²
        A_f = sp.csr_matrix((data, (row_idx, col_idx)),
                            shape=(2 * n - 1, n))

        c = -np.ones(n)                            # minimise -sum(b)  → maximise sum(b)
        b_ub = M * np.ones(2 * n - 1)             # RHS of the inequalities
        # generous upper bound – values >1000 would be clipped later anyway
        bounds = [(0.0, 1000.0) for _ in range(n)]

        lp_res = linprog(c,
                         A_ub=A_f,
                         b_ub=b_ub,
                         bounds=bounds,
                         method='highs',
                         options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})
        if not lp_res.success:
            # if the LP fails, abort the LP phase for this length
            break

        b_opt = lp_res.x
        sum_b = b_opt.sum()
        if sum_b <= 1e-12:
            break

        # scale to the required L¹‑norm
        g = b_opt * (S / sum_b)

        # convex combination with the current vector
        f_new = (1.0 - mix_frac) * f + mix_frac * g
        f_new = _project_simplex(f_new, S)

        # evaluate
        new_score = np.max(_full_convolution(f_new))

        if new_score < best_score:
            best_score = new_score
            best_f = f_new.copy()

        f = f_new

        # occasional random jitter to help escape plateaus
        if rng.random() < 0.07:
            i, j = rng.integers(0, n, size=2)
            if i != j and f[i] > 0.0:
                delta = f[i] * rng.random() * 0.03
                f[i] -= delta
                f[j] += delta
                f = _project_simplex(f, S)

    return best_f, best_score


# ----------------------------------------------------------------------
# Gradient based fine‑tuner (copy of the previous Adam → sub‑grad code)
# ----------------------------------------------------------------------
def _optimize_combined_adam(a0: np.ndarray,
                            deadline: float,
                            rng: np.random.Generator,
                            fraction_adam: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Run Adam for a fraction of the allocated time, then finish with sub‑gradient.
    """
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    adam_deadline = now + fraction_adam * remaining
    a_s, sc_s = _optimize_adam(a0, adam_deadline, rng)

    if time.time() < deadline:
        a_f, sc_f = _optimize_subgrad(a_s, deadline, rng)
        if sc_f < sc_s:
            return a_f, sc_f
    return a_s, sc_s


# (The Adam / sub‑gradient helpers are the same as in the reference code.
#  They are reproduced here unchanged for completeness.)

def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    lr = 0.4
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    alpha = 0.5
    alpha_max = 1500.0

    while time.time() < deadline:
        t += 1
        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_simplex(a, S)

        if rng.random() < 0.03:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.02
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        alpha = min(alpha * 1.05, alpha_max)

    return best_a, best_score


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = rfft(a, n=fft_len)
    Fw = rfft(w_rev, n=fft_len)
    full = irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.5
    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        if rng.random() < 0.07:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.03
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = np.max(_full_convolution(cand))
                if cand_score < cur_score or rng.random() < 0.2:
                    a = cand
                    cur_score = cand_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function (non‑negative vector) that minimises
    ``evaluate_sequence``.  The search proceeds through three increasing
    problem sizes using a linear‑programming based “best‑response” update,
    followed by a short Adam/sub‑gradient refinement.  The best sequence
    found before the time limit is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered construction if available
    # ------------------------------------------------------------------
    best_seq = None
    if 'height_sequence_1' in globals():
        try:
            best_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            best_seq = None

    best_score = np.inf
    sizes = [800, 2000, 4000]               # increasing lengths
    time_fractions = [0.20, 0.30, 0.45]     # how to split the overall budget

    total_budget = deadline - start_time

    for stage, n in enumerate(sizes):
        if time.time() >= deadline:
            break

        stage_time = total_budget * time_fractions[stage]
        stage_deadline = min(time.time() + stage_time, deadline)

        # initialise vector for this stage
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        else:
            a0 = rng.random(n)

        a0 += rng.normal(scale=a0.mean() * 1e-4 + 1e-12, size=n)   # tiny jitter
        a0 = np.clip(a0, 0.0, None)

        S = np.sqrt(2.0 * n)
        if a0.sum() == 0.0:
            a0 = np.full(n, 1.0)
        a0 = a0 / a0.sum() * S

        # ----- LP based optimisation -----
        a_opt, sc_opt = _optimize_lp_stage(a0, stage_deadline, rng,
                                           mix_frac=0.25, max_iter=80)

        # ----- short Adam / sub‑gradient refinement (if time permits) -----
        remaining = stage_deadline - time.time()
        if remaining > 2.0:
            a_opt, sc_opt = _optimize_combined_adam(a_opt,
                                                    stage_deadline,
                                                    rng,
                                                    fraction_adam=0.75)

        # update global best
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[stage] n={n}  new global best score={best_score:.6f}")

        total_budget = deadline - time.time()

    # ------------------------------------------------------------------
    # Final sanity check & external evaluation
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()

    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} "
          f"(internal best = {best_score:.6f}) length = {len(final_seq)}")
    return final_seq
