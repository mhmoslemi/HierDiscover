"""
Propose a step‑function sequence that minimises the benchmark
`evaluate_sequence`.  The algorithm works in four increasing scales
(≈2000 → 4000 → 8000 → 12000).  For each scale we run three optimisation
phases:

1. **Projected Adam** on a soft‑max approximation of the maximum entry of
   the self‑convolution.  Adam gives fast, stable descent while the
   projection onto the ℓ¹‑simplex keeps the sum normalised
   (Σa_i = √(2·n)).  The soft‑max temperature β is increased slowly up to
   8192 so that the surrogate becomes tighter.

2. **Sub‑gradient descent** that follows only the current maximal
   convolution entry.  This “sharpening’’ step often removes leftover
   spikes that Adam does not eliminate.

3. **Enhanced simulated‑annealing** with several move types
   (mass transfer, local smoothing, random swaps and a Gaussian‑blur
   move).  Temperature and the allowed transfer fraction decay
   exponentially, and every accepted move updates the convolution via
   FFT.

All intermediate iterates are projected onto the simplex and the best
sequence seen across all scales is finally returned.  The routine respects
the 1000 s time budget; it halves a second as a safety margin.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σ x_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (computed via FFT)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective value = max(a*a) because Σa = √(2n)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Generate (i+1)^(-p) and normalise to the requested ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(new_x, old_x, seq)
    return new_seq.astype(np.float64)


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max surrogate of max(convolution).
    Returns (grad, conv) where conv is the full convolution a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # grad = 2 * (reverse(a) * p)   (convolution of reversed a with p)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]      # keep the central n entries
    return grad, b

# Optimisation primitives

def _adam_optimize(a0: np.ndarray, deadline: float):
    """Projected Adam on the soft‑max surrogate."""
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = target_sum * 0.02               # base learning rate
    t = 0
    beta = 1.0
    max_beta = 8192.0

    while time.time() < deadline:
        t += 1
        # increase soft‑max temperature gradually
        if t % 2000 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        grad, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr / (np.sqrt(v_hat) + eps)
        a -= step * m_hat

        # projection back onto the simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape tiny plateaus
        if t % 1000 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={t:5d} β={beta:6.1f} score={best_val:.6f}")

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray, deadline: float):
    """Simple sub‑gradient descent that follows the current maximal convolution entry."""
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        # compute full convolution once
        conv = _conv_fft(a)
        k_max = int(np.argmax(conv))

        # gradient of the maximal entry (sub‑gradient)
        grad = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad[idx] = 2.0 * a[k_max - idx]

        # step size decays with √iteration count
        norm = np.linalg.norm(grad)
        if norm > 0:
            step = target_sum * 0.5 / (norm * (np.sqrt(it + 1) + 1.0))
            a -= step * grad

        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


def _enhanced_simulated_annealing(a0: np.ndarray, deadline: float,
                                 start_temp: float = 1.0,
                                 end_temp: float = 1e-5,
                                 max_frac_start: float = 0.35,
                                 max_frac_end: float = 0.01,
                                 max_window: int = 70) -> tuple[np.ndarray, float]:
    """SA with extra move types (Gaussian blur, split/merge)."""
    a = a0.copy()
    n = a.size
    cur_score = _score(a)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-9)
    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        # temperature schedule
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # index of current maximal convolution entry
        k_max = int(np.argmax(conv))

        # sensitivities (derivative of the max entry)
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]

        # choose a move
        move_type = None
        if rng.random() < 0.55:                     # mass transfer (biased)
            weight = a * sens
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight / total_w)
            else:
                donor = rng.integers(n)

            # receiver tends to have low contribution to the max entry
            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            if low_idx.size:
                receiver = rng.choice(low_idx)
            else:
                receiver = rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        elif rng.random() < 0.85:                  # local smoothing
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                old_seg = a[start:end].copy()
                avg = old_seg.mean()
                a[start:end] = avg
                move_type = "smooth"
                saved_seg = old_seg
                saved_start, saved_end = start, end

        elif rng.random() < 0.95:                  # Gaussian blur on a short window
            start = rng.integers(0, n)
            length = rng.integers(3, max_window + 1)
            end = min(start + length, n)
            if end - start >= 3:
                seg = a[start:end]
                # 1‑D Gaussian kernel (σ≈ length/4)
                sigma = max(1.0, (end - start) / 4.0)
                x = np.arange(end - start) - (end - start - 1) / 2.0
                g = np.exp(-(x ** 2) / (2 * sigma ** 2))
                g /= g.sum()
                blurred = np.convolve(seg, g, mode='same')
                saved = seg.copy()
                a[start:end] = blurred
                move_type = "blur"
                saved_seg = saved
                saved_start, saved_end = start, end

        else:                                      # random swap
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # recompute convolution and score (FFT based)
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = _score(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it:5d} new best={best_score:.6f}")
        else:
            # revert the move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "blur":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter to re‑inject diversity
        if it % 10000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            cur_sum = a.sum()
            if cur_sum > 0:
                a = a * (np.sqrt(2.0 * n) / cur_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


def _run_one_instance(n: int, init_seq: np.ndarray, deadline: float):
    """Run Adam → Sub‑gradient → SA on a single length‑n problem."""
    target_sum = np.sqrt(2.0 * n)
    a = init_seq.astype(np.float64, copy=True)
    if a.size != n:
        a = _resize_sequence(a, n)
    a = _project_to_simplex(a, target_sum)

    # Split remaining time among the three phases
    now = time.time()
    remaining = deadline - now
    # 30 % Adam, 30 % sub‑gradient, rest SA
    adam_dead = now + remaining * 0.30
    a, _ = _adam_optimize(a, adam_dead)

    now = time.time()
    remaining = deadline - now
    sub_dead = now + remaining * 0.30
    a, _ = _subgradient_optimize(a, sub_dead)

    # SA gets whatever time is left
    a, final_score = _enhanced_simulated_annealing(a, deadline)
    return a, final_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (height sequence) with low
    `evaluate_sequence` score.  The method works on four increasing scales
    (≈2000 → 4000 → 8000 → 12000), re‑using the best result of a scale as a
    warm‑start for the next.  The best sequence across all scales is
    returned.
    """
    start = time.time()
    deadline = start + budget_s - 1.0          # 1 s safety margin

    # Global fallback sequence if the judge has stored a previous construction
    if "height_sequence_1" in globals():
        fallback = np.array(height_sequence_1, dtype=np.float64)
    else:
        fallback = None

    best_seq = None
    best_score = float("inf")
    cur_seq = None

    # Scales (length, fraction of the remaining budget)
    scales = [(2000, 0.20), (4000, 0.25), (8000, 0.35), (12000, 0.20)]

    for n, frac in scales:
        now = time.time()
        if now >= deadline - 5.0:
            break
        remaining = deadline - now
        stage_deadline = now + remaining * frac

        # Initialise the sequence for this scale
        if cur_seq is not None:
            init = _resize_sequence(cur_seq, n)
        elif fallback is not None:
            init = _resize_sequence(fallback, n)
        else:
            # randomised power‑law exponent gives diverse starts
            exp = np.random.uniform(0.35, 0.75)
            init = _power_law_sequence(n, exp, np.sqrt(2.0 * n))

        seq, score = _run_one_instance(n, init, stage_deadline)

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Scale n={n} improved best score={best_score:.6f}")

        cur_seq = seq  # warm‑start for the next (larger) scale

    # Trim trailing near‑zeros for neatness
    if best_seq is None:
        best_seq = np.array([], dtype=float)
    else:
        best_seq = np.array(best_seq, dtype=float)

    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    final_seq = best_seq.tolist()

    # One final validation/evaluation (does not affect the result)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
