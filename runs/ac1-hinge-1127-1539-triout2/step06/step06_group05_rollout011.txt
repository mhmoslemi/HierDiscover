    import time, numpy as np, math, random
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 5.0

    best_seq = None
    best_score = np.inf

    # Predefine shape families functions
    def shape_uniform(n):
        a = np.ones(n)
        a = a / a.sum() * math.sqrt(2*n)
        return a

    def shape_geom(n):
        rate = rng.uniform(0.90, 0.99999)
        a = rate ** np.arange(n)
        a = a / a.sum() * math.sqrt(2*n)
        return a

    def shape_power(n, power):
        idx = np.arange(n)
        a = 1.0 / ((idx+1) * (n - idx))**(power/2) # maybe not valid at edges
        a = np.where(np.isfinite(a), a, 0.0)
        a = a / a.sum() * math.sqrt(2*n)
        return a

    def shape_gaussian(n, sigma):
        x = np.arange(n) - (n-1)/2
        a = np.exp(-0.5 * (x / sigma)**2)
        a = a / a.sum() * math.sqrt(2*n)
        return a

    # shape generation function that picks random parameters
    def generate_initial(n):
        typ = rng.integers(0,5)
        if typ == 0:
            # uniform Dirichlet random
            a = rng.random(n)
        elif typ == 1:
            a = shape_geom(n)
        elif typ == 2:
            # power-law random
            p = rng.uniform(0.2, 1.5)
            idx = np.arange(n)
            a = (1.0 / ( (idx+1)*(n-idx) ))**p
        elif typ == 3:
            sigma = rng.uniform(0.2*n, 0.5*n)
            a = shape_gaussian(n, sigma)
        elif typ == 4:
            # triangular
            centre = (n-1)/2
            a = 1.0 - np.abs(np.arange(n)-centre)/centre
        else:
            # mixture of random bases
            cand = []
            cand.append(rng.random(n))  # uniform
            rate = rng.uniform(0.9, 0.9999)
            left = rate ** np.arange(n)
            cand.append((left + left[::-1])/2)
            idx = np.arange(n)
            rec = 1.0/ (idx+1) + 1.0/(n - idx)
            power = rng.uniform(0.2,1.0)
            cand.append(rec**power)
            w = rng.random(len(cand)); w /= w.sum()
            a = sum(wi*ci for wi,ci in zip(w,cand))
        # project to simplex
        S = math.sqrt(2.0*n)
        # maybe negative values due to exponent? but they are nonnegative.
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full(n, 1.0)
            a_sum = a.sum()
        a = a / a_sum * S
        # optionally symmetrize
        if rng.random() < 0.5:
            a = 0.5*(a + a[::-1])
        return a

"""
Multi‑phase optimizer for the self‑convolution peak problem.

The algorithm proceeds in three stages:

1. **Exploration** – generate a diverse set of initial vectors using several
   analytic families (uniform, geometric, power‑law, Gaussian, triangular,
   random convex combos).  Each vector is projected onto the simplex
   Σa = √(2·n) and optionally symmetrised.

2. **Refinement** – the current best vector is repeatedly refined with a
   sequence of optimisers that respect the simplex constraint:
      • Projected Adam on a soft‑max (“log‑sum‑exp”) surrogate of the max.
      • Sub‑gradient descent on the true max entry with a simple back‑tracking
        line‑search for the step‑size.
      • A cheap local “swap” hill‑climber that moves tiny amounts of mass
        between random positions.
   The time budget is split so that most of the allotted 1000 s is spent
   on refining the best candidate, while occasional fresh restarts keep
   exploration alive.

3. **Final check** – the best sequence is returned as a Python list.  The
   global variable ``height_sequence_1`` is updated so that later runs can
   up‑sample it.

All updates are projected onto the simplex (Euclidean projection) and,
when the ``enforce_sym`` flag is set, exact mirror symmetry is enforced.
The implementation uses only NumPy (no external solvers) and respects the
time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x>=0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex with total mass s."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2*n‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate w.r.t. a.
    w must be a probability vector (soft‑max weights).
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3*n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights (log‑sum‑exp)
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Compute soft‑max probabilities over convolution entries."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry (v[i] = v[-1-i])."""
    return 0.5 * (v + v[::-1])


def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to new length n."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Initial sequence generators
# ----------------------------------------------------------------------
def _initial_sequence(base_seq: np.ndarray | None,
                     n: int,
                     rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n with ℓ¹‑norm √(2·n).
    Several families are mixed to provide diversity.
    """
    S = math.sqrt(2.0 * n)

    typ = rng.integers(0, 6)   # 0–5 inclusive

    if typ == 0 and base_seq is not None and base_seq.size > 10:
        a = _upsample_seq(base_seq, n)
        a += rng.normal(scale=a.mean() * 1e-3, size=n)
        a = np.clip(a, 0.0, None)

    elif typ == 1:
        # uniform Dirichlet(1)
        a = rng.random(n)

    elif typ == 2:
        # geometric decay from both ends
        rate = rng.uniform(0.90, 0.9999)
        left = rate ** np.arange(n)
        right = left[::-1]
        a = (left + right) / 2.0

    elif typ == 3:
        # edge‑heavy reciprocal (power‑law) shape
        idx = np.arange(n, dtype=float)
        a = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.9)
        a = a ** power

    elif typ == 4:
        # triangular shape (peak in centre)
        centre = (n - 1) / 2.0
        a = 1.0 - np.abs(np.arange(n) - centre) / centre
        a = np.clip(a, 0.0, None)

    else:
        # random convex combination of a few bases
        cand = []

        cand.append(rng.random(n))                     # uniform
        rate = rng.uniform(0.90, 0.9999)
        left = rate ** np.arange(n)
        cand.append((left + left[::-1]) / 2.0)          # geometric
        idx = np.arange(n, dtype=float)
        rec = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.9)
        cand.append(rec ** power)                       # reciprocal

        w = rng.random(len(cand))
        w /= w.sum()
        a = sum(wi * ci for wi, ci in zip(w, cand))

    # Normalisation to the required ℓ¹‑norm
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0)
        a_sum = a.sum()
    a = a / a_sum * S
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smooth log‑sum‑exp approximation
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Projected Adam minimising the soft‑max approximation of the convolution peak.
    Returns (best_vector, best_peak_value).
    """
    a = a0.copy()
    n = a.shape[0]
    S = math.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.45
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule for the soft‑max
    alpha = 0.5
    alpha_mul = 1.025

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # soft‑max weights and gradient wrt smooth objective
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # optional symmetry enforcement
        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional random jitter to escape plateaus
        if rng.random() < 0.02:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.04 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < best_score or rng.random() < 0.15:
                    a = cand

        # temperature and learning‑rate schedules
        alpha = min(alpha * alpha_mul, 80.0)
        lr *= 0.99996

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true maximum convolution entry
# ----------------------------------------------------------------------
def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent focusing on the current worst shift.  An adaptive
    back‑tracking line‑search is used to choose the step size.
    """
    a = a0.copy()
    n = a.shape[0]
    S = math.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.17
    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient for the max entry
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]                      # derivative ≈ a_j

        # simple back‑tracking line‑search
        step = eta0 / math.sqrt(it)
        improved = False
        for _ in range(4):
            cand = a - step * g
            if enforce_sym:
                cand = _symmetrize(cand)
            cand = _project_simplex(cand, S)

            cand_max = np.max(_full_convolution(cand))
            if cand_max < best_score:
                a = cand
                best_score = cand_max
                best_a = cand.copy()
                improved = True
                break
            step *= 0.5

        if not improved:
            # fall back to a tiny step to keep moving
            a = cand

        # occasional random mass‑swap (simulated‑annealing style)
        if rng.random() < 0.05:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                if enforce_sym:
                    cand = _symmetrize(cand)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < best_score or rng.random() < 0.25:
                    a = cand
                    best_score = cand_max
                    best_a = cand.copy()

        # early exit if stagnating
        if it % 6000 == 0 and best_score == np.max(_full_convolution(a)):
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Short local random‑swap search (fine‑tuning)
# ----------------------------------------------------------------------
def _local_swap_search(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """Very cheap hill‑climbing that swaps tiny masses between random positions."""
    a = a0.copy()
    n = a.shape[0]
    S = math.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.02 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = _symmetrize(cand)
        cand_score = np.max(_full_convolution(cand))
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand
            a = cand
    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimisation pipeline (Adam → sub‑gradient → local search)
# ----------------------------------------------------------------------
def _optimize_combined(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Runs the three optimisation phases within the supplied time budget.
    The split is 55 % Adam – 30 % sub‑gradient – 15 % local search.
    """
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    t_adam = now + 0.55 * remaining
    a_adam, sc_adam = _optimize_adam(a0, t_adam, rng, enforce_sym)

    t_sub = min(time.time() + 0.30 * remaining, deadline)
    a_sub, sc_sub = _optimize_subgrad(a_adam, t_sub, rng, enforce_sym)

    if time.time() < deadline:
        a_loc, sc_loc = _local_swap_search(a_sub, deadline, rng, enforce_sym)
    else:
        a_loc, sc_loc = a_sub, sc_sub

    # choose the best among the three phases
    candidates = [(a_adam, sc_adam), (a_sub, sc_sub), (a_loc, sc_loc)]
    best_seq, best_score = min(candidates, key=lambda kv: kv[1])
    return best_seq, best_score

# Main entry point for the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function with a minimal self‑convolution peak.
    Returns the best sequence (list of floats) found before the time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin for the final external evaluation
    deadline = start_time + budget_s - 5.0

    # use previous construction if available
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    best_n = None
    attempts = 0
    last_improve = start_time

    # ------------------------------------------------------------------
    # Main loop – a mix of fresh random restarts and intensive refinement
    # ------------------------------------------------------------------
    while time.time() < deadline:
        attempts += 1

        # Decide whether to start from scratch or to refine the current best
        if best_seq is not None and rng.random() < 0.73:
            # Refine mode – perturb the current best
            n = best_n
            # Occasionally vary the length a bit
            if rng.random() < 0.12:
                scale = rng.uniform(0.9, 1.1)
                n = max(600, int(best_n * scale))

            a0 = best_seq.copy()
            if n != a0.shape[0]:
                a0 = _upsample_seq(a0, n)

            # small random jitter
            jitter = rng.normal(scale=a0.mean() * 0.03, size=n)
            a0 += jitter
            a0 = np.clip(a0, 0.0, None)
            a0 = _project_simplex(a0, math.sqrt(2.0 * n))

            enforce_sym = rng.random() < 0.55
            if enforce_sym:
                a0 = _symmetrize(a0)

            slice_deadline = min(time.time() + rng.uniform(22.0, 34.0), deadline)
            a_opt, sc = _optimize_combined(a0, slice_deadline, rng, enforce_sym)

        else:
            # Exploration mode – fresh random initialisation
            n = int(rng.integers(1800, 12000))
            a0 = _initial_sequence(base_seq, n, rng)
            enforce_sym = rng.random() < 0.48
            if enforce_sym:
                a0 = _symmetrize(a0)
            slice_deadline = min(time.time() + rng.uniform(12.0, 20.0), deadline)
            a_opt, sc = _optimize_combined(a0, slice_deadline, rng, enforce_sym)

        attempts += 0  # silence unused‑variable warning

        if sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            best_n = n
            last_improve = time.time()
            print(f"[global] attempt={attempts:3d} n={n:5d} sym={enforce_sym} "
                  f"new best={best_score:.6f}")

        # If we have not improved for a while, discard the stored base_seq
        if time.time() - last_improve > 45.0:
            base_seq = None

        # Early stop if we cross the target bound (some slack for final eval)
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Final sanity check via the black‑box evaluator.
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()

    external_score = evaluate_sequence(final_seq)
    print(f"[final] external={external_score:.6f} internal_best={best_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)

    return final_seq
