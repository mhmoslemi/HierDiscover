import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

def solve_lp_for_g(f, M, max_iter=1000):
    n = f.shape[0]
    A = sp.diags([f] * n, np.arange(n), shape=(2*n-1, n), format='csc')
    # Note: constraints are A @ g <= M (vector of size 2n-1)
    b = np.full(2*n-1, M)
    # objective: maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    # Using scipy.optimize.linprog with method='highs', which supports sparse.
    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        print("LP not solved:", res.message)
        return None
    g = res.x
    return g
```

"""
Propose a step‑function sequence that minimizes
    2·n·max(a*a) / (sum(a)^2)

The algorithm uses an iterative *LP‑mixing* scheme:
1. Start from a feasible (non‑negative, ℓ¹‑normalised) sequence.
2. Compute M = max(convolution(a, a)).
3. Solve a linear program:
       maximize   Σ g_i
       subject to  (a * g)_k ≤ M   for all k
                   Σ g_i ≤ target_sum
                   g_i ≥ 0
   The LP finds a distribution g that places as much mass as possible
   without increasing any convolution entry beyond the current worst value M.
4. Mix the current sequence with the LP solution:
       a ← (1‑t)·a + t·g      (t≈0.5, then reduced if no progress)
   Project the mixture back onto the ℓ¹‑simplex Σ a_i = sqrt(2·n).
5. Repeat steps 2‑4 a few times (or until the time budget expires).
6. Keep the best sequence encountered among several problem sizes
   (1000‒2500 entries) and several random restarts.

The LP is built with a sparse Toeplitz matrix (one diagonal per shift),
so that even for n≈2500 the memory footprint stays modest.
All work is done within the 1000 s budget; the code prints progress
and finally returns the best sequence (trailing zeros stripped).
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Implements Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> tuple[np.ndarray, float]:
    """Full linear convolution a*a via FFT and its maximum entry."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv, float(conv.max())


def _score(a: np.ndarray) -> float:
    """Fast evaluation of the objective (under the ℓ¹‑normalisation)."""
    n = a.size
    _, max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _build_convolution_matrix(f: np.ndarray) -> sp.csc_matrix:
    """
    Build the (2n‑1) × n sparse matrix A such that (A @ g)[k] = (f * g)[k].
    Each diagonal of A (offset d) contains the vector f.
    """
    n = f.shape[0]
    # diags expects a list of arrays, one per diagonal offset.
    # Every diagonal gets the same data = f.
    data = [f] * n
    offsets = np.arange(n)                     # 0,1,…,n‑1
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csc")
    return A


def _solve_lp(f: np.ndarray, M: float, target_sum: float) -> np.ndarray | None:
    """
    Solve
        max Σ g_i
        s.t. (f * g)_k ≤ M   (k = 0 … 2n‑2)
             Σ g_i ≤ target_sum
             g_i ≥ 0
    Returns the optimal g (may be all zeros) or None if the solver fails.
    """
    n = f.shape[0]
    A = _build_convolution_matrix(f)                     # (2n‑1) × n
    # Append the Σ g_i ≤ target_sum row.
    ones_row = sp.csr_matrix(np.ones((1, n)))
    A_ub = sp.vstack([A, ones_row], format="csc")
    b_ub = np.concatenate([np.full(2 * n - 1, M), [target_sum]])

    c = -np.ones(n)   # maximize Σ g_i  → minimize -Σ g_i
    bounds = [(0, None)] * n

    try:
        res = opt.linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": 30.0},
        )
    except Exception as e:  # pragma: no cover
        print("LP exception:", e)
        return None

    if not res.success:
        # very often the high‑level solver reports failure for numerically
        # difficult instances; we just skip this iteration.
        return None
    g = res.x
    # Clip tiny negatives caused by numerical noise.
    g = np.maximum(g, 0.0)
    return g


def _lp_mixing_step(
    a: np.ndarray,
    target_sum: float,
    mix_factor: float,
    time_deadline: float,
) -> tuple[np.ndarray, float] | None:
    """
    Perform a single LP‑mixing iteration.
    Returns the new sequence and its score, or None if the LP could not be solved.
    """
    n = a.size
    conv, M = _max_convolution_fft(a)

    g = _solve_lp(a, M, target_sum)
    if g is None:
        return None

    # Mix and re‑project onto the simplex.
    a_new = (1.0 - mix_factor) * a + mix_factor * g
    a_new = _project_to_simplex(a_new, target_sum)
    new_score = _score(a_new)
    return a_new, new_score


# ----------------------------------------------------------------------
# Main optimisation driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function sequence.
    The routine tries several problem sizes (1000‑2500) and a few random
    restarts, applying the LP‑mixing scheme described above.
    The best sequence found before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # Problem sizes (larger → more degrees of freedom, higher per‑iteration cost)
    sizes = [1000, 1500, 2000, 2500]

    for n in sizes:
        if time.time() >= deadline - 5.0:
            break

        target_sum = np.sqrt(2.0 * n)

        # Allocate a slice of the remaining budget to this size.
        remaining = deadline - time.time()
        # Keep a small reserve for final polishing.
        per_instance_deadline = time.time() + min(remaining * 0.35, 120.0)

        # --------------------------------------------------------------
        # Multiple random restarts for robustness
        # --------------------------------------------------------------
        restarts = 3
        for r in range(restarts):
            if time.time() >= per_instance_deadline - 2.0:
                break

            # ----- initialise -------------------------------------------------
            # Prefer the globally‑available construction if it exists and is long enough.
            if "height_sequence_1" in globals():
                seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
                if seq.size > n:
                    seq = seq[:n]
                else:
                    seq = np.pad(seq, (0, n - seq.size), mode="constant")
            else:
                # Dirichlet‑style random start (exponential draws → positive entries)
                seq = rng.exponential(scale=1.0, size=n)
            seq = _project_to_simplex(seq, target_sum)

            cur_score = _score(seq)
            mix_factor = 0.5  # start with a fairly aggressive mixing

            # ----- iterative LP‑mixing ---------------------------------------
            iter_limit = 12
            for it in range(iter_limit):
                if time.time() >= per_instance_deadline - 2.0:
                    break

                result = _lp_mixing_step(seq, target_sum, mix_factor, per_instance_deadline)
                if result is None:
                    # LP failed – try a smaller mixing factor and continue.
                    mix_factor *= 0.7
                    if mix_factor < 0.05:
                        break
                    continue

                new_seq, new_score = result
                # Simple acceptance rule: keep the improvement.
                if new_score < cur_score - 1e-7:
                    seq, cur_score = new_seq, new_score
                    print(
                        f"[{time.time():.1f}] n={n} restart={r} iter={it} "
                        f"mix={mix_factor:.3f} score={cur_score:.6f}"
                    )
                    # If progress stalls, reduce the mixing strength.
                    mix_factor = max(0.1, mix_factor * 0.9)
                else:
                    # No improvement → reduce aggressiveness.
                    mix_factor *= 0.7
                    if mix_factor < 0.03:
                        break

            # ----- keep the best for this size --------------------------------
            if cur_score < best_score:
                best_score = cur_score
                best_seq = seq.copy()
                print(
                    f"[{time.time():.1f}] ** NEW GLOBAL BEST ** "
                    f"size={n} restart={r} score={best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # Final polishing (light projected gradient descent) if time permits
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 3.0:
        # Very short gradient step to fine‑tune the result.
        n = best_seq.size
        target_sum = np.sqrt(2.0 * n)
        a = best_seq.copy()
        beta = 256.0
        step_size = target_sum * 0.08
        for _ in range(80):
            # gradient of the smooth max (log‑sum‑exp)
            size_fft = 1 << ((2 * n - 1).bit_length())
            a_fft = np.fft.rfft(a, n=size_fft)
            conv = np.fft.irfft(a_fft * a_fft, n=size_fft)[: 2 * n - 1]
            max_c = conv.max()
            exps = np.exp(beta * (conv - max_c))
            p = exps / exps.sum()
            # gradient w.r.t a (derived analytically)
            a_rev = a[::-1]
            Pr_fft = np.fft.rfft(p, n=size_fft)
            Ar_fft = np.fft.rfft(a_rev, n=size_fft)
            grad_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size_fft)[: 2 * n - 1]
            grad = 2.0 * grad_rev[n - 1 : n - 1 + n]

            # simple step + projection
            a = a - step_size * grad / (np.linalg.norm(grad) + 1e-12)
            a = _project_to_simplex(a, target_sum)

        final_score = _score(a)
        if final_score < best_score:
            best_score = final_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] final polishing improved score to {best_score:.6f}")

    # Strip trailing near‑zero entries (they do not affect the evaluation).
    if best_seq is not None:
        best_seq = np.maximum(best_seq, 0.0)
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[: nz[-1] + 1]
        else:
            best_seq = np.array([], dtype=float)

    print(f"[{time.time():.1f}] DONE – best score ≈ {best_score:.6f}, length = {len(best_seq)}")
    return best_seq.tolist()
