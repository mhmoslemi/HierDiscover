"""
Search for a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.  The approach builds on the previous solution but
adds

* a broader set of parametric families (including a tapered power‑law)
* extensive random sampling of the family parameters
* a finer grid around the previously‑found optimum (p≈0.5, shift≈0.6)
* more elite candidates and a longer refinement phase
* a final “any‑time’’ random‑restart loop that continues until the deadline.

All heavy work is done with FFT‑based convolution, projection onto the
ℓ¹‑simplex (sum = √(2 n)) and a smooth log‑sum‑exp surrogate of the maximal
convolution entry.  After the gradient phase a permutation‑annealing step
optimises the ordering of the multiset.  The routine respects the overall
time budget (default 1000 s) and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the full linear convolution a*a."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex Σx = s , x≥0  (Duchi et al. 2008)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = s}."""
    if s <= 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) surrogate of max(b) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b) where

        b = a*a (full convolution)
        g = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a ⋆ p) ,   p_k ∝ exp(β·b_k)

    The implementation uses FFT for speed.
    """
    n = a.shape[0]

    # ----- full convolution a*a -----
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # ----- soft‑max probabilities -----
    max_b = np.max(b)
    # shift for numerical stability, clamp exponent to avoid overflow in negative direction
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)

    # ----- gradient g = 2·(a ⋆ p) -----
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1           # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]   # keep only the “central” n entries
    return g, b


# ----------------------------------------------------------------------
# Projected gradient descent on the smoothed max (log‑sum‑exp)
# ----------------------------------------------------------------------


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Annealed projected gradient descent on the smooth surrogate."""
    a = a0.copy()
    n = a.size
    target_sum = math.sqrt(2.0 * n)

    beta = 1.0
    max_beta = 1e7
    momentum = np.zeros_like(a)
    rho = 0.9                     # exponential decay for momentum
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum step
        momentum = rho * momentum + (1.0 - rho) * g
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            lr = 0.4 * target_sum / (norm * (beta + 1.0))
            a -= lr * momentum

        # simplex projection
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow plateaus
        if it % 200 == 0:
            jitter = rng.normal(
                scale=target_sum * 5e-4 / (beta + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal beta
        if it % 150 == 0 and beta < max_beta:
            beta = min(beta * 1.07, max_beta)

        it += 1
    return a


# ----------------------------------------------------------------------
# Permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------


def _permutation_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Improve the ordering of a fixed multiset by random swaps.
    The L¹‑norm stays unchanged; only the permutation varies.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # initialise temperature proportional to the current score
    T = best_score * 0.015 + 1e-9
    cooling = 0.9993
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]          # propose swap
        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        if cur_score <= best_score:
            accept = True
        else:
            accept = rng.random() < math.exp(-dE / T)

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # revert
            a[i], a[j] = a[j], a[i]

        T *= cooling

        # occasional “global” move: shuffle a small random subset
        if rng.random() < 0.0015:
            mask = rng.random(n) < 0.025
            if np.any(mask):
                sub = a[mask]
                rng.shuffle(sub)
                a[mask] = sub
                cur_score = _max_convolution_fft(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()

    return best


# ----------------------------------------------------------------------
# Normalisation helper (scale to a given ℓ¹ sum)
# ----------------------------------------------------------------------


def _normalize_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector so that its ℓ¹‑norm equals *target*."""
    s = np.sum(v)
    if s == 0.0:
        # fallback: uniform tiny mass
        v = np.full_like(v, target / v.size)
    else:
        v = v * (target / s)
    return np.maximum(v, 0.0)


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    target = math.sqrt(2.0 * n)
    return _normalize_to_sum(raw, target)


def _exponential_sequence(n: int, rate: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.exp(-rate * (idx + shift))
    target = math.sqrt(2.0 * n)
    return _normalize_to_sum(raw, target)


def _linear_sequence(n: int, slope: float, intercept: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, intercept - slope * idx)
    # prevent degenerate all‑zero case
    if raw.sum() == 0.0:
        raw[0] = 1.0
    target = math.sqrt(2.0 * n)
    return _normalize_to_sum(raw, target)


def _tapered_powerlaw_sequence(n: int, p: float, shift: float, gamma: float) -> np.ndarray:
    """(i+shift)^{-p} * (1 - i/(n‑1))^{γ}  – a power‑law with a smooth tail."""
    idx = np.arange(n, dtype=float)
    tail = np.maximum(0.0, 1.0 - idx / (n - 1))
    raw = (idx + shift) ** (-p) * (tail ** gamma)
    target = math.sqrt(2.0 * n)
    return _normalize_to_sum(raw, target)


# ----------------------------------------------------------------------
# Ordering heuristics
# ----------------------------------------------------------------------


def _candidate_orderings(base: np.ndarray, rng: np.random.Generator) -> list[np.ndarray]:
    """
    Generate a small collection of orderings for the same multiset.
    The aim is to reduce large‑by‑large overlaps.
    """
    n = base.size
    asc = np.sort(base)               # smallest → largest
    desc = asc[::-1]                  # largest → smallest
    out = [asc, desc]

    # alternating small–large (small first)
    alt = np.empty_like(base)
    lo, hi = 0, n - 1
    for i in range(n):
        if i % 2 == 0:
            alt[i] = asc[lo]
            lo += 1
        else:
            alt[i] = asc[hi]
            hi -= 1
    out.append(alt)

    # alternating large–small (large first)
    alt2 = np.empty_like(base)
    lo, hi = 0, n - 1
    for i in range(n):
        if i % 2 == 0:
            alt2[i] = asc[hi]
            hi -= 1
        else:
            alt2[i] = asc[lo]
            lo += 1
    out.append(alt2)

    # step‑permutations with coprime strides (few small primes)
    for step in (2, 3, 5, 7, 11, 13, 17, 19, 23):
        if math.gcd(step, n) != 1:
            continue
        perm = np.empty_like(base)
        pos = 0
        for i in range(n):
            perm[pos] = asc[i]
            pos = (pos + step) % n
        out.append(perm)

    # a handful of random shuffles
    for _ in range(5):
        shuffled = asc.copy()
        rng.shuffle(shuffled)
        out.append(shuffled)

    return out

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget,
    explores several parametric families, keeps an elite set,
    runs gradient‑descent + permutation‑annealing refinements,
    and finally returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    best_seq = None
    best_obj = np.inf
    elite = []                                      # list of (score, ndarray)

    # ------------------------------------------------------------------
    # 1. Broad parametric sweep (randomised + a fine grid around the old optimum)
    # ------------------------------------------------------------------
    lengths = [4000, 6000, 8000, 12000, 16000]
    elite_sz = 12

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # ----- power‑law family (random sampling) -----
        for _ in range(35):
            if time.time() >= deadline:
                break
            p = rng.uniform(0.38, 0.62)
            shift = rng.uniform(0.05, 5.0)
            base = _powerlaw_sequence(n, p, shift)
            for cand in _candidate_orderings(base, rng):
                score = _max_convolution_fft(cand)
                if score < best_obj:
                    best_obj, best_seq = score, cand.copy()
                    print(f"[{time.time():.1f}] New global best (pow) n={n} p={p:.4f} shift={shift:.3f} obj={score:.6f}")
                if len(elite) < elite_sz or score < elite[-1][0]:
                    elite.append((score, cand))
                    elite.sort(key=lambda x: x[0])
                    elite = elite[:elite_sz]

        # ----- tapered power‑law (adds a tail exponent) -----
        for _ in range(20):
            if time.time() >= deadline:
                break
            p = rng.uniform(0.38, 0.62)
            shift = rng.uniform(0.05, 5.0)
            gamma = rng.uniform(0.0, 3.0)
            base = _tapered_powerlaw_sequence(n, p, shift, gamma)
            for cand in _candidate_orderings(base, rng):
                score = _max_convolution_fft(cand)
                if score < best_obj:
                    best_obj, best_seq = score, cand.copy()
                    print(f"[{time.time():.1f}] New global best (tpl) n={n} p={p:.4f} shift={shift:.3f} γ={gamma:.2f} obj={score:.6f}")
                if len(elite) < elite_sz or score < elite[-1][0]:
                    elite.append((score, cand))
                    elite.sort(key=lambda x: x[0])
                    elite = elite[:elite_sz]

        # ----- exponential family -----
        for _ in range(30):
            if time.time() >= deadline:
                break
            rate = rng.uniform(0.0004, 0.006)
            shift = rng.uniform(0.0, 2.5)
            base = _exponential_sequence(n, rate, shift)
            for cand in _candidate_orderings(base, rng):
                score = _max_convolution_fft(cand)
                if score < best_obj:
                    best_obj, best_seq = score, cand.copy()
                    print(f"[{time.time():.1f}] New global best (exp) n={n} rate={rate:.5f} shift={shift:.3f} obj={score:.6f}")
                if len(elite) < elite_sz or score < elite[-1][0]:
                    elite.append((score, cand))
                    elite.sort(key=lambda x: x[0])
                    elite = elite[:elite_sz]

        # ----- linear (truncated) family -----
        for _ in range(30):
            if time.time() >= deadline:
                break
            slope = rng.uniform(0.0, 0.006)
            intercept = rng.uniform(0.2, 4.0)
            base = _linear_sequence(n, slope, intercept)
            for cand in _candidate_orderings(base, rng):
                score = _max_convolution_fft(cand)
                if score < best_obj:
                    best_obj, best_seq = score, cand.copy()
                    print(f"[{time.time():.1f}] New global best (lin) n={n} slope={slope:.5f} int={intercept:.2f} obj={score:.6f}")
                if len(elite) < elite_sz or score < elite[-1][0]:
                    elite.append((score, cand))
                    elite.sort(key=lambda x: x[0])
                    elite = elite[:elite_sz]

        # ----- random‑uniform vectors -----
        rnd_end = min(deadline, time.time() + 1.8)
        while time.time() < rnd_end:
            rand = rng.random(n)
            rand = _normalize_to_sum(rand, target_sum)
            score = _max_convolution_fft(rand)
            if score < best_obj:
                best_obj, best_seq = score, rand.copy()
                print(f"[{time.time():.1f}] New global best (rand) n={n} obj={score:.6f}")
            if len(elite) < elite_sz or score < elite[-1][0]:
                elite.append((score, rand))
                elite.sort(key=lambda x: x[0])
                elite = elite[:elite_sz]

        # ----- fine grid around previously‑found optimal parameters (if n is the largest) -----
        if n == max(lengths):
            p_vals = np.linspace(0.48, 0.52, 11)
            shift_vals = np.linspace(0.4, 0.8, 9)
            for p in p_vals:
                for shift in shift_vals:
                    if time.time() >= deadline:
                        break
                    base = _powerlaw_sequence(n, p, shift)
                    for cand in _candidate_orderings(base, rng):
                        score = _max_convolution_fft(cand)
                        if score < best_obj:
                            best_obj, best_seq = score, cand.copy()
                            print(f"[{time.time():.1f}] New global best (grid) n={n} p={p:.3f} shift={shift:.3f} obj={score:.6f}")
                        if len(elite) < elite_sz or score < elite[-1][0]:
                            elite.append((score, cand))
                            elite.sort(key=lambda x: x[0])
                            elite = elite[:elite_sz]

    # ------------------------------------------------------------------
    # 2.  Refine elite candidates (gradient descent + permutation annealing)
    # ------------------------------------------------------------------
    # Ensure we also keep the best sequence found so far as an elite entry
    if best_seq is not None:
        elite.append((_max_convolution_fft(best_seq), best_seq.copy()))
        elite.sort(key=lambda x: x[0])
        elite = elite[:elite_sz]

    for base_score, seed_seq in elite:
        if time.time() >= deadline:
            break
        n = seed_seq.size
        # allocate up to 20 s for the gradient phase (or the remaining budget)
        sub_dead = min(deadline, time.time() + 20.0)
        refined = _gradient_descent_opt(seed_seq, sub_dead)

        # use whatever time is left for permutation annealing
        refined = _permutation_annealing(refined, deadline)

        obj = _max_convolution_fft(refined)
        if obj < best_obj:
            best_obj, best_seq = obj, refined.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL BEST (refined) n={n} obj={obj:.6f}")

    # ------------------------------------------------------------------
    # 3.  Any‑time random‑restart loop: continue until budget exhausted
    # ------------------------------------------------------------------
    while time.time() < deadline - 4.0:          # keep a few seconds as final margin
        # pick a length randomly (biased towards larger ones)
        n = int(rng.choice([8000, 12000, 16000, 20000]))
        target_sum = math.sqrt(2.0 * n)

        # random start
        rnd_vec = rng.random(n)
        rnd_vec = _normalize_to_sum(rnd_vec, target_sum)

        # quick gradient phase (≤12 s)
        sub_dead = min(deadline, time.time() + 12.0)
        refined = _gradient_descent_opt(rnd_vec, sub_dead)

        # finish with permutation annealing
        refined = _permutation_annealing(refined, deadline)

        obj = _max_convolution_fft(refined)
        if obj < best_obj:
            best_obj, best_seq = obj, refined.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL BEST (random‑restart) n={n} obj={obj:.6f}")

    # ------------------------------------------------------------------
    # 4.  Final sanity & output
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform distribution of length 4000
        n = 4000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n)
        best_obj = _max_convolution_fft(best_seq)

    # trim trailing near‑zeros (purely cosmetic)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # optional verification using the provided evaluate_sequence (may be omitted)
    try:
        eval_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {eval_val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective = {best_obj:.6f}  length = {best_seq.size}")
    return best_seq.tolist()
