Strategy

*   Work on several lengths (8000 → 20000).  Larger n allows a better

        – one‑sided exponential   a_i ∝ exp(-α i / n)
        – symmetric exponential   a_i ∝ exp(-α·min(i,n‑1-i)/n)
        – power‑law               a_i ∝ (i+1)^(-p)

*   The best candidate found so far is refined by the *targeted hill‑climber*

*   Finally the sequence is trimmed (removing trailing near‑zeros) and

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

def _hill_climb_targeted(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Hill‑climbing that focuses on the current maximal convolution entry.
    Returns the best sequence seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    T = cur_max * 0.02 + 1e-12          # initial temperature
    cooling = 0.9995

    n = a.size
    while time.time() < deadline:
        k_max = int(np.argmax(b))

        # contributions of each index to the current peak
        idx = np.arange(n)
        j_arr = k_max - idx
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[j_arr[mask]]

        total = contrib.sum()
        if total > 0.0:
            prob_src = contrib / total
            i = rng.choice(n, p=prob_src)
        else:
            prob_src = a / a.sum()
            i = rng.choice(n, p=prob_src)

        inv = contrib.max() - contrib + 1e-12
        prob_dst = inv / inv.sum()
        j = rng.choice(n, p=prob_dst)
        if i == j:
            continue

        max_delta = a[i] * 0.5
        if max_delta < 1e-16:
            continue
        delta = rng.random() * max_delta

        # analytic update of the convolution
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a

        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        accept = False
        if new_max < cur_max - 1e-12:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            a[i] = max(a[i], 0.0)
            a[j] = max(a[j], 0.0)

            b = b_new
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_targeted improvement: {best_max:.6f}")

        T *= cooling

    return best_a, best_max

def _target_sum(n: int) -> float:
    """Desired ℓ¹‑norm for a length‑n sequence."""
    return np.sqrt(2.0 * n)


def _exp_one_sided(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ exp(-α·i/n).  α≥0."""
    idx = np.arange(n, dtype=float)
    a = np.exp(-alpha * idx / n)
    a *= _target_sum(n) / a.sum()
    return a


def _exp_symmetric(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ exp(-α·min(i,n‑1‑i)/n) – high at both ends, low in the middle."""
    idx = np.arange(n, dtype=float)
    dist = np.minimum(idx, n - 1 - idx) / n
    a = np.exp(-alpha * dist)
    a *= _target_sum(n) / a.sum()
    return a


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=float)
    a = idx ** (-p)
    a *= _target_sum(n) / a.sum()
    return a


def _hybrid_exp_pow(n: int, alpha: float, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}·exp(-α·i/n)."""
    idx = np.arange(1, n + 1, dtype=float)
    a = idx ** (-p) * np.exp(-alpha * (idx - 1) / n)
    a *= _target_sum(n) / a.sum()
    return a

def _grid_search_one_family(
    n: int,
    deadline: float,
    gen_func,
    param_grid,
    refine_rounds: int = 3,
):
    """
    Coarse‑to‑fine grid search for a single parametric family.
    Returns (best_sequence, best_score).
    """
    best_seq = None
    best_score = np.inf
    best_params = None

    # --- coarse grid -------------------------------------------------
    for params in param_grid:
        if time.time() >= deadline:
            return best_seq, best_score
        seq = gen_func(n, *params)
        score = _max_convolution_fft(seq)
        if score < best_score:
            best_score = score
            best_seq = seq
            best_params = params

    # --- refinement -------------------------------------------------
    # Only refine if we have a candidate and time permits
    if best_seq is None:
        return best_seq, best_score

    for _ in range(refine_rounds):
        if time.time() >= deadline:
            break
        # build a tighter neighbourhood around the current best parameters
        if isinstance(best_params, tuple):
            # assume each parameter is a float
            lo = [max(0.0, p - 0.2 * abs(p) - 1e-6) for p in best_params]
            hi = [p + 0.2 * abs(p) + 1e-6 for p in best_params]
            # for a parameter that is exactly zero we still explore a small interval
            lo = [max(0.0, p - 0.05) if p == 0 else l for p, l in zip(best_params, lo)]
            hi = [p + 0.05 if p == 0 else h for p, h in zip(best_params, hi)]
            # generate a small mesh (2 samples per dimension)
            meshes = []
            for l, h in zip(lo, hi):
                meshes.append(np.linspace(l, h, 3)[1:-1])  # middle point only
            # Cartesian product (few points)
            import itertools
            for cand in itertools.product(*meshes):
                if time.time() >= deadline:
                    break
                seq = gen_func(n, *cand)
                score = _max_convolution_fft(seq)
                if score < best_score:
                    best_score = score
                    best_seq = seq
                    best_params = cand
        else:
            # single‑parameter families
            p = best_params
            lo = max(0.0, p - 0.2 * abs(p) - 1e-6)
            hi = p + 0.2 * abs(p) + 1e-6
            if p == 0:
                lo = max(0.0, p - 0.05)
                hi = p + 0.05
            mid = (lo + hi) / 2.0
            cand = (mid,)
            seq = gen_func(n, *cand)
            score = _max_convolution_fft(seq)
            if score < best_score:
                best_score = score
                best_seq = seq
                best_params = cand

    return best_seq, best_score

        )
        if seq is not None and score < best_score:
            best_score, best_seq = score, seq.copy()
            print(f"[{time.time():.1f}] new best (power‑law) n={n} score={score:.6f}")

        # ----------------------------------------------------------------
        # 4) Hybrid (exp·pow) – 2‑parameter search
        # ----------------------------------------------------------------
        alphas = np.linspace(0.0, 4.0, 9)
        ps = np.linspace(0.0, 2.0, 9)
        param_grid = [(a, p) for a in alphas for p in ps]
        seq, score = _grid_search_one_family(
            n,
            deadline,
            _hybrid_exp_pow,
            param_grid,
            refine_rounds=2,
        )
        if seq is not None and score < best_score:
            best_score, best_seq = score, seq.copy()
            print(f"[{time.time():.1f}] new best (hybrid) n={n} score={score:.6f}")

        # ----------------------------------------------------------------
        # Quick local polishing with a few gradient‑descent steps (optional)
        # ----------------------------------------------------------------
        # If enough time remains we take a very short projected GD run
        # starting from the current best of this length.
        time_left = deadline - time.time()
        if best_seq is not None and time_left > 30.0:
            # reuse the gradient descent code from the previous solution
            # (copied locally to avoid external dependencies)
            from math import sqrt

            def _project_to_simplex(v, s):
                """Euclidean projection onto the simplex Σx_i = s, x_i ≥ 0."""
                n = v.shape[0]
                u = np.sort(v)[::-1]
                cssv = np.cumsum(u) - s
                ind = np.arange(1, n + 1)
                cond = u - cssv / ind > 0
                if np.any(cond):
                    rho = np.nonzero(cond)[0][-1]
                    theta = cssv[rho] / (rho + 1.0)
                else:
                    theta = 0.0
                w = np.maximum(v - theta, 0.0)
                return w

            def _gradient_softmax(a, beta):
                """Gradient of the smooth max (log‑sum‑exp) of a*a."""
                n = a.shape[0]
                size1 = _next_pow2(2 * n - 1)
                a_fft = np.fft.rfft(a, n=size1)
                b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

                max_b = np.max(b)
                exps = np.exp(beta * (b - max_b))
                p = exps / np.sum(exps)

                # correlation a ⋆ p
                a_rev = a[::-1]
                conv_len = n + (2 * n - 1) - 1  # = 3n‑2
                size2 = _next_pow2(conv_len)
                Ar_fft = np.fft.rfft(a_rev, n=size2)
                Pr_fft = np.fft.rfft(p, n=size2)
                conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
                g = 2.0 * conv_rev[n - 1: n - 1 + n]
                return g, b

            def _gradient_descent_opt(a0, deadline_inner):
                a = a0.copy()
                n = a.size
                target_sum = np.sqrt(2.0 * n)
                beta = 0.5
                beta_max = 16384.0
                beta_factor = 1.5
                step_base = target_sum * 0.12
                momentum = np.zeros_like(a)
                rho = 0.92
                it = 0
                while time.time() < deadline_inner:
                    g, _ = _gradient_softmax(a, beta)
                    momentum = rho * momentum + (1.0 - rho) * g
                    norm_m = np.linalg.norm(momentum)
                    if norm_m > 0.0:
                        lr = step_base / (norm_m * (beta + 1.0))
                        a -= lr * momentum
                    a = _project_to_simplex(a, target_sum)

                    if it % 300 == 0:
                        jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                            size=n)
                        a += jitter
                        a = np.maximum(a, 0.0)
                        a = _project_to_simplex(a, target_sum)

                    if it % 400 == 0 and beta < beta_max:
                        beta = min(beta * beta_factor, beta_max)

                    it += 1
                return a

            gd_deadline = time.time() + min(15.0, 0.2 * time_left)
            a_gd = _gradient_descent_opt(best_seq, gd_deadline)
            gd_score = _max_convolution_fft(a_gd)
            if gd_score < best_score:
                best_score, best_seq = gd_score, a_gd.copy()
                print(f"[{time.time():.1f}] GD polish improvement: score={gd_score:.6f}")

    # ------------------------------------------------------------------
    # Final hill‑climbing refinement (uses all remaining budget)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a_final = best_seq.copy()
        b_final = _full_convolution(a_final)
        a_final, max_final = _hill_climb_targeted(a_final, b_final, deadline)
        if max_final < best_score:
            best_score, best_seq = max_final, a_final.copy()
            print(f"[{time.time():.1f}] Final hill‑climb improvement: score={max_final:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries for neatness
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform vector (should never happen)
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log the official evaluation value (optional – may be omitted)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
