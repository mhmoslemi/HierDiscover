def generate_shape(n, rng):
    typ = rng.integers(0,5)
    if typ == 0:
        # exponential decay from left
        c = rng.uniform(0.0005, 0.005) * n # scaling factor? Actually want shape exp(-c * i) where c moderate
        i = np.arange(n)
        a = np.exp(-c * i)
    elif typ == 1:
        # power law left heavy
        p = rng.uniform(0.3, 2.0)
        i = np.arange(1, n+1)
        a = 1.0 / (i ** p)
    elif typ == 2:
        # linear decreasing (triangular)
        peak = rng.uniform(0.5, 1.5)
        slope = rng.uniform(0.0, peak/(n-1))
        a = np.maximum(peak - slope * np.arange(n), 0.0)
    elif typ == 3:
        # step function: non-zero in first m entries
        m = rng.integers(int(n*0.05), int(n*0.5))
        val = rng.uniform(0.1, 2.0)
        a = np.zeros(n)
        a[:m] = val
    elif typ == 4:
        # mixture of two exponentials (left and right)
        c1 = rng.uniform(0.0005, 0.003) * n
        c2 = rng.uniform(0.0005, 0.003) * n
        i = np.arange(n)
        a = np.exp(-c1 * i) + np.exp(-c2 * (n-1 - i))
    elif typ == 5:
        # random convex combination of exponentials and power law
        shapes = []
        weights = []
        # generate 3 random shapes
        for _ in range(3):
            sub_typ = rng.integers(0,3)
            if sub_typ == 0:
                c = rng.uniform(0.0005, 0.004) * n
                shape = np.exp(-c * np.arange(n))
            elif sub_typ == 1:
                p = rng.uniform(0.4, 1.5)
                shape = 1.0 / ((np.arange(1, n+1)) ** p)
            else:
                shape = rng.random(n)
            shapes.append(shape)
            weights.append(rng.random())
        weights = np.array(weights)
        weights /= weights.sum()
        a = sum(w * s for w, s in zip(weights, shapes))
    else: # fallback uniform
        a = rng.random(n)
    # Normalize
    a_sum = a.sum()
    if a_sum == 0:
        a = np.ones(n)
        a_sum = a.sum()
    a = a / a_sum * np.sqrt(2*n)
    return a

def lp_improve(a, S, step=0.2, max_iter=10, deadline=None):
    best_a = a.copy()
    best_val = np.max(_full_convolution(a))
    n = a.shape[0]

    while max_iter>0 and (deadline is None or time.time()<deadline):
        M = np.max(_full_convolution(a))
        # Build convolution matrix
        rows = 2*n-1
        # Use np.zeros and assign
        F = np.zeros((rows, n))
        for j in range(n):
            F[j:j+n, j] = a  # Place copy
        # Solve LP
        c = -np.ones(n)  # maximize sum
        ub = np.full(rows, M)
        bounds = [(0, None)] * n
        # Use scipy.optimize.linprog
        res = linprog(c, A_ub=F, b_ub=ub, bounds=bounds, method='highs', options={'presolve':True, 'time_limit': max(0, deadline - time.time()) if deadline else None})
        if not res.success:
            break
        g = res.x
        sum_g = g.sum()
        if sum_g == 0:
            break
        scale = S / sum_g
        g = g * scale
        # mixture
        a_new = (1-step) * a + step * g
        a_new = _project_simplex(a_new, S)  # Ensure feasibility (should be)
        val_new = np.max(_full_convolution(a_new))
        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
        a = a_new
        max_iter -= 1
    return best_a, best_val
```

"""
Hybrid LP + gradient optimiser for the self‑convolution peak problem.

The optimiser works in three stages:

1. **Coarse construction (small n, LP phase).**
   For n ≈ 1‑2 k we solve a linear programme
        max  Σ g_j   s.t.   (f * g)[k] ≤ M_f,  g ≥ 0
   where M_f is the current peak of f*f.  The LP returns an extreme
   point g that is “as flat as possible” under the same bound.
   After scaling g to the required ℓ¹‑norm we mix it with f
   (f←(1‑t)f + t g).  Repeating a few iterations drives the peak down
   dramatically.

2. **Refinement (medium to large n, smooth Adam).**
   The best vector from the LP (or a random analytic shape) is up‑sampled
   to the target length and further optimised with Adam on a smooth
   log‑sum‑exp surrogate of the max‑convolution.  The optimiser works
   on the simplex (Euclidean projection) and optionally adds a tiny
   stochastic jitter to escape plateaus.

3. **Final polishing (sub‑gradient + tiny swap search).**
   A few sub‑gradient steps on the true max entry followed by a cheap
   hill‑climbing that swaps little amounts of mass between random indices
   tighten the solution.

The whole routine is time aware: it repeatedly restarts with different
lengths (≈1 k–12 k) while keeping the best sequence found.  The final
sequence is returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x>=0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex with total mass s."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2*n‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities over convolution entries."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    return exp_shifted / exp_shifted.sum()


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of the log‑sum‑exp surrogate w.r.t. a."""
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Upsampling – linear interpolation from old to new length
# ----------------------------------------------------------------------
def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to new length n."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Analytic families for rapid random initialisation
# ----------------------------------------------------------------------
def _random_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a non‑negative vector of length n, ℓ¹‑norm = √(2 n)."""
    S = np.sqrt(2.0 * n)
    typ = rng.integers(0, 6)

    if typ == 0:
        # left‑hand exponential decay
        decay = rng.uniform(0.001, 0.008) * n
        a = np.exp(-decay * np.arange(n))
    elif typ == 1:
        # power‑law heavy on the left
        p = rng.uniform(0.3, 2.0)
        a = 1.0 / (np.arange(1, n + 1) ** p)
    elif typ == 2:
        # single step (first m entries equal)
        m = rng.integers(int(0.05 * n), int(0.4 * n))
        val = rng.uniform(0.2, 2.0)
        a = np.zeros(n)
        a[:m] = val
    elif typ == 3:
        # mixed left & right exponentials
        d1 = rng.uniform(0.001, 0.006) * n
        d2 = rng.uniform(0.001, 0.006) * n
        a = np.exp(-d1 * np.arange(n)) + np.exp(-d2 * (n - 1 - np.arange(n)))
    elif typ == 4:
        # random convex combination of a few bases
        bases = []
        # uniform
        bases.append(rng.random(n))
        # exponential
        d = rng.uniform(0.001, 0.005) * n
        bases.append(np.exp(-d * np.arange(n)))
        # power‑law
        p = rng.uniform(0.4, 1.5)
        bases.append(1.0 / (np.arange(1, n + 1) ** p))
        w = rng.random(len(bases))
        w /= w.sum()
        a = sum(wi * bi for wi, bi in zip(w, bases))
    else:
        # pure uniform (fallback)
        a = rng.random(n)

    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0)
        a_sum = a.sum()
    a = a / a_sum * S
    return a


# ----------------------------------------------------------------------
# Linear‑programme improvement (small n only)
# ----------------------------------------------------------------------
def _lp_improve(a: np.ndarray,
                S: float,
                step: float = 0.5,
                max_iter: int = 6,
                deadline: float = None) -> tuple[np.ndarray, float]:
    """
    Perform a few LP–based improvement steps.
    Each step solves:
        max  Σ g_j   subject to   (a * g)[k] ≤ M_a  for all k,  g ≥ 0
    The resulting g is scaled to have ℓ¹‑norm S and mixed with a.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = np.max(_full_convolution(a))

    for _ in range(max_iter):
        if deadline is not None and time.time() >= deadline:
            break

        M = np.max(_full_convolution(a))

        # Build the convolution matrix A (rows = 2n‑1, cols = n)
        rows = 2 * n - 1
        F = np.zeros((rows, n), dtype=float)
        for j in range(n):
            F[j:j + n, j] = a

        # Linear programme: maximise sum(g)  ⇔ minimise -sum(g)
        c = -np.ones(n, dtype=float)      # minimise -1·g
        ub = np.full(rows, M, dtype=float)
        bounds = [(0.0, None)] * n

        res = linprog(c,
                      A_ub=F,
                      b_ub=ub,
                      bounds=bounds,
                      method='highs',
                      options={'presolve': True})
        if not res.success:
            break

        g = res.x
        sum_g = g.sum()
        if sum_g == 0.0:
            break
        # Scale to the correct ℓ¹‑norm
        g = g * (S / sum_g)

        # Convex combination
        a_new = (1.0 - step) * a + step * g
        a_new = _project_simplex(a_new, S)

        val_new = np.max(_full_convolution(a_new))
        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()

        a = a_new

    return best_a, best_val


# ----------------------------------------------------------------------
# Adam optimiser on the smooth log‑sum‑exp approximation
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """Projected Adam minimising the soft‑max approximation of the peak."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    lr = 0.45
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    alpha = 0.5
    alpha_mul = 1.02

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # soft‑max weights and gradient of the smoothed objective
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = 0.5 * (a + a[::-1])

        a = _project_simplex(a, S)

        # occasional tiny random jitter – helps to escape flat regions
        if rng.random() < 0.02:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.04 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < best_score or rng.random() < 0.15:
                    a = cand

        alpha = min(alpha * alpha_mul, 50.0)
        lr *= 0.99996

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true maximum entry
# ----------------------------------------------------------------------
def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent focused on the worst shift, followed by tiny swaps."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.13
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient for the active shift
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        if enforce_sym:
            a = 0.5 * (a + a[::-1])
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # occasional random mass‑swap (simulated‑annealing style)
        if rng.random() < 0.045:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.22:
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()

        if it % 8000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Very cheap hill‑climbing that swaps tiny masses between random positions
# ----------------------------------------------------------------------
def _local_swap_search(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """Fine‑grained hill climbing – swaps a few percent of mass each step."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.02 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = 0.5 * (cand + cand[::-1])
        cand_score = np.max(_full_convolution(cand))
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand
            a = cand
    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimiser (LP → Adam → sub‑grad → local)
# ----------------------------------------------------------------------
def _optimize_combined(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Runs a short LP phase (if n is modest) followed by Adam,
    sub‑gradient and final local refinement.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    # ------------------------------------------------------------------
    # LP phase – use only when the problem is small enough
    # ------------------------------------------------------------------
    if n <= 2500:
        t_lp = now + 0.20 * remaining
        a_lp, sc_lp = _lp_improve(a0, S, step=0.5, max_iter=6, deadline=t_lp)
    else:
        a_lp, sc_lp = a0, np.max(_full_convolution(a0))

    # ------------------------------------------------------------------
    # Adam phase
    # ------------------------------------------------------------------
    t_adam = min(time.time() + 0.55 * remaining, deadline)
    a_adam, sc_adam = _optimize_adam(a_lp, t_adam, rng, enforce_sym)

    # ------------------------------------------------------------------
    # Sub‑gradient phase
    # ------------------------------------------------------------------
    t_sub = min(time.time() + 0.20 * remaining, deadline)
    a_sub, sc_sub = _optimize_subgrad(a_adam, t_sub, rng, enforce_sym)

    # ------------------------------------------------------------------
    # Final local search (if any time left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_loc, sc_loc = _local_swap_search(a_sub, deadline, rng, enforce_sym)
    else:
        a_loc, sc_loc = a_sub, sc_sub

    # Pick the best among the three phases
    candidates = [(a_lp, sc_lp), (a_adam, sc_adam), (a_sub, sc_sub), (a_loc, sc_loc)]
    best_seq, best_score = min(candidates, key=lambda kv: kv[1])
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point – orchestrates many restarts under the time budget
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,

    """

    Returns the best sequence discovered before the time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin for the external evaluation
    deadline = start_time + budget_s - 5.0

    # reuse the previously best construction, if present
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_overall_seq = None
    best_overall_score = np.inf
    best_overall_n = None
    attempts = 0
    last_improve = start_time

    while time.time() < deadline:
        attempts += 1

        # ------------------------------------------------------------------
        # Choose a problem size.
        # ------------------------------------------------------------------
        if best_overall_seq is not None and rng.random() < 0.45:
            # try to slightly enlarge the current best
            scale = rng.uniform(0.9, 1.1)
            n = max(800, int(best_overall_n * scale))
        else:
            # random length – small lengths are cheaper for the LP phase
            if rng.random() < 0.6:
                n = int(rng.integers(1200, 2400))   # LP‑friendly regime
            else:

        # ------------------------------------------------------------------

        # ------------------------------------------------------------------

            a0 = np.clip(a0 + jitter, 0.0, None)
            a0 = _project_simplex(a0, np.sqrt(2.0 * n))
        else:
            a0 = _random_shape(n, rng)

                # mix a little of the historic shape (interpolated)
                b = _upsample_seq(base_seq, n)
                mix = rng.random()
                a0 = mix * a0 + (1.0 - mix) * b
                a0 = _project_simplex(a0, np.sqrt(2.0 * n))

        # ------------------------------------------------------------------

        # ------------------------------------------------------------------

        # ------------------------------------------------------------------

            best_overall_seq = a_opt.copy()

            last_improve = time.time()
            print(f"[global] attempt={attempts} n={n} new best={best_overall_score:.6f}")

        # ------------------------------------------------------------------
        # Early exit if we have already reached the target region.
        # ------------------------------------------------------------------
        if best_overall_score <= 1.48:   # a comfortable safety margin
            break

        # ------------------------------------------------------------------
        # Forget the historic base shape if we have been idle for a while.
        # ------------------------------------------------------------------
        if time.time() - last_improve > 50.0:

    # ----------------------------------------------------------------------

    if best_overall_seq is None:
        final_seq = [1.0]
    else:

          f"length={len(final_seq)} attempts={attempts}")

    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)

    return final_seq
