import time

def _project_to_simplex(v, s):
    """Euclidean projection onto simplex {x >=0, sum x = s} using Duchi et al."""
    # Input v: numpy vector
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, len(v)+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _max_convolution_fft(a):
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())

def _conv_fft(a):
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]

    """Return evaluation of current a (max convolution) after projection condition."""
    # a should already be projected such that sum is target, else we compute proper
    n = a.shape[0]

    """
    Compute gradient of log-sum-exp of convolution entries w.r.t. a.
    Returns gradient and the convolution array.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]
    # softmax weights
    max_c = np.max(conv)
    exps = np.exp(beta * (conv - max_c))
    p = exps / np.sum(exps)
    # convolution of a_rev and p
    a_rev = a[::-1]
    p_fft = np.fft.rfft(p, n=size)
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[: 2 * n - 1]
    # gradient
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, conv

def _subgradient_max(a):
    """
    Subgradient of the max entry of the convolution.
    Returns gradient direction (negative subgradient) and current max value.
    """
    n = a.shape[0]
    conv = _conv_fft(a)
    max_idx = int(np.argmax(conv))
    max_val = float(conv[max_idx])
    # compute gradient of conv[max_idx] w.r.t a:
    # g_i = 2*a[max_idx - i] for indices i within [0,n-1] intersect [max_idx-(n-1), max_idx]
    left = max(0, max_idx - (n - 1))
    right = min(max_idx, n - 1)
    idx = np.arange(left, right + 1)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[max_idx - idx]
    return grad, max_val, max_idx

def _simulated_annealing(a, deadline,
                         start_temp=0.5,
                         end_temp=1e-6,
                         max_frac_start=0.4,
                         max_frac_end=0.01,
                         max_window=30):
    """
    Simulated annealing refinement with several move types.
    """
    n = a.shape[0]
    target = np.sqrt(2.0 * n)
    cur = a.copy()
    cur_score = _score(cur)

    best = cur.copy()
    best_score = cur_score

    start = time.time()
    total = max(deadline - start, 1e-6)
    rng = np.random.default_rng()
    size_fft = 1 << ((2 * n - 1).bit_length())
    conv_cur = _conv_fft(cur)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / total
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # locate worst entry
        max_val = conv_cur.max()
        k_max = int(np.argmax(conv_cur))

        # compute contribution weight of each index to that entry
        left = max(0, k_max - (n - 1))
        right = min(k_max, n - 1)
        idx = np.arange(left, right + 1)
        # for each i in idx, contribution = a[i] * a[k_max - i]
        weight = a[idx] * a[k_max - idx]

        # ---- propose a move
        move = None
        if rng.random() < 0.7:   # mass transfer
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(idx, p=weight / total_w)
            else:
                donor = rng.integers(n)
            # pick receiver among low contributors
            low_thr = np.percentile(weight, 10)
            low_idx = idx[weight <= low_thr]
            if low_idx.size > 0:
                receiver = rng.choice(low_idx)
            else:
                receiver = rng.integers(n)
            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move = ('transfer', donor, receiver, delta)
        elif rng.random() < 0.9:   # smoothing
            start_win = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end_win = min(start_win + length, n)
            if end_win - start_win >= 2:
                saved = a[start_win:end_win].copy()
                avg = saved.mean()
                a[start_win:end_win] = avg
                move = ('smooth', start_win, end_win, saved)
        else:  # swap
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move = ('swap', i, j)

        # evaluate new state
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = float(2.0 * n * conv_new.max() / (a.sum() ** 2))

        # acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # Metropolis
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = a.copy()
            cur_score = new_score
            conv_cur = conv_new
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA new best {best_score:.6f}")
        else:
            # revert
            if move is not None:
                if move[0] == 'transfer':
                    _, donor, receiver, delta = move
                    a[donor] += delta
                    a[receiver] -= delta
                elif move[0] == 'smooth':
                    _, start_win, end_win, saved = move
                    a[start_win:end_win] = saved
                elif move[0] == 'swap':
                    i, j = move[1], move[2]
                    a[i], a[j] = a[j], a[i]

        # occasional global jitter to escape deep minima
        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)
            conv_cur = _conv_fft(a)
            cur_score = _score(a)

        it += 1
    return best, best_score

def _initialize_sequence(n, mode, rng):

    if mode == "global" and "height_sequence_1" in globals():

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative sequence minimizing max(a*a) under ℓ¹‑normalisation.
    The algorithm runs a projected gradient descent on a soft‑max approximation,
    followed by a sub‑gradient descent on the actual max, and finishes with a
    simulated‑annealing refinement.  Several random restarts on a single
    problem size (default n=5000) are performed within the time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.2   # safety margin
    best_seq = None
    best_score = float("inf")
    # Choose a problem size; we focus resources on a relatively large n.
    n = 5000
    # If time is very limited, reduce n.
    if budget_s < 200:
        n = 2000
    # Number of independent runs we will attempt
    max_runs = 5
    for run in range(max_runs):
        now = time.time()
        if now >= deadline - 30.0:
            break
        # allocate some time for this run
        remaining = deadline - now
        # allocate 70% to optimization phases, keep 30% for possible final refinement after loop
        run_deadline = now + remaining * 0.65
        # random init mode
        mode = rng.choice(["global", "power", "random"])
        a0 = _initialize_sequence(n, mode, rng)
        # Phase 1: softmax gradient descent
        gd_deadline = time.time() + min(30.0, (run_deadline - time.time()) * 0.4)
        a_gd, gd_score = _gradient_descent_softmax(a0, gd_deadline)
        # Phase 2: subgradient descent on max
        sub_deadline = time.time() + min(30.0, (run_deadline - time.time()) * 0.3)
        a_sub, sub_score = _subgradient_descent(a_gd, sub_deadline)
        # Phase 3: simulated annealing
        sa_deadline = run_deadline
        a_sa, sa_score = _simulated_annealing(a_sub, sa_deadline)

        # Record best
        final_score = min(gd_score, sub_score, sa_score)
        if final_score < best_score:
            best_score = final_score
            best_seq = a_sa.copy()
            print(f"[{time.time():.1f}] Run {run+1}/{max_runs} mode={mode} new global best {best_score:.6f}")

    # Final refinement on the best sequence using a short softmax GD
    if best_seq is not None and time.time() < deadline - 10.0:
        final_deadline = time.time() + (deadline - time.time()) * 0.25
        best_seq, _ = _gradient_descent_softmax(best_seq, final_deadline)

    # Strip trailing near‑zeros for compactness
    if best_seq is not None:
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[: nz[-1] + 1]
        else:
            best_seq = np.array([], dtype=float)
    else:
        best_seq = np.array([], dtype=float)

    print(f"[{time.time():.1f}] Finished. Best score≈{best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()

    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # soft‑max weights
    max_c = np.max(conv)
    exps = np.exp(beta * (conv - max_c))
    p = exps / np.sum(exps)

    # gradient via convolution of reversed a and the weight vector p
    a_rev = a[::-1]
    p_fft = np.fft.rfft(p, n=size)
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, conv


def _subgradient_max(a: np.ndarray):
    """
    Sub‑gradient of the max entry of the convolution.
    Returns (gradient, max_value, max_index).
    """
    n = a.shape[0]
    conv = _conv_fft(a)
    idx_max = int(np.argmax(conv))
    max_val = float(conv[idx_max])

    left = max(0, idx_max - (n - 1))
    right = min(idx_max, n - 1)
    idx = np.arange(left, right + 1)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[idx_max - idx]
    return grad, max_val, idx_max

def _gradient_descent_softmax(a0: np.ndarray, deadline: float):
    """
    Projected gradient descent on the soft‑max (log‑sum‑exp) approximation.
    Returns the best sequence encountered and its score.
    """
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)

    best = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 4096.0
    step_base = target * 0.12
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        norm = np.linalg.norm(grad)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a_new = a - lr * grad
        else:
            a_new = a.copy()

        a_new = _project_to_simplex(a_new, target)
        new_val = _score(a_new)

        if new_val < best_val:
            best_val = new_val
            best = a_new.copy()
            print(f"[{time.time():.1f}] Soft‑max GD β={beta:.2f} score={best_val:.6f}")

        if new_val <= _score(a):
            a = a_new

        # temperature schedule
        if it % 300 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        # occasional jitter to escape plateaus
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target * 0.001 / (beta + 1.0), size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

        it += 1

    return best, best_val


def _subgradient_descent(a0: np.ndarray, deadline: float):
    """
    Sub‑gradient descent directly on the max‑convolution entry.
    Returns the best sequence encountered and its score.
    """
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)

    best = a.copy()
    best_val = _score(a)

    step0 = target * 0.05
    it = 0

    while time.time() < deadline:
        grad, cur_max, k_max = _subgradient_max(a)
        norm = np.linalg.norm(grad)
        if norm == 0:
            break

        step = step0 / (np.sqrt(it + 1) + 1.0)
        a_new = a - step * grad
        a_new = _project_to_simplex(a_new, target)

        new_val = _score(a_new)

        if new_val < best_val:
            best_val = new_val
            best = a_new.copy()
            print(f"[{time.time():.1f}] Sub‑grad it {it} score={best_val:.6f}")

        if new_val < _score(a):
            a = a_new
        else:
            step0 *= 0.95  # shrink step when not improving

        it += 1

    return best, best_val


def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_temp: float = 0.5,
                         end_temp: float = 1e-6,
                         max_frac_start: float = 0.4,
                         max_frac_end: float = 0.01,
                         max_window: int = 30):
    """
    Simulated‑annealing refinement with three move types.
    Returns the best sequence encountered and its score.
    """
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)

    cur_score = _score(a)
    best = a.copy()
    best_score = cur_score

    start = time.time()
    total = max(deadline - start, 1e-9)
    rng = np.random.default_rng()
    size_fft = 1 << ((2 * n - 1).bit_length())
    conv_cur = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / total
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # worst convolution entry and its contribution weights
        k_max = int(np.argmax(conv_cur))
        left = max(0, k_max - (n - 1))
        right = min(k_max, n - 1)
        idx = np.arange(left, right + 1)
        weight = a[idx] * a[k_max - idx]

        move = None
        # 70 % mass transfer
        if rng.random() < 0.7:
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(idx, p=weight / total_w)
            else:
                donor = rng.integers(n)

            low_thr = np.percentile(weight, 10)
            low_idx = idx[weight <= low_thr]
            receiver = rng.choice(low_idx) if low_idx.size else rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move = ("transfer", donor, receiver, delta)

        # 20 % local smoothing
        elif rng.random() < 0.9:
            start_win = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end_win = min(start_win + length, n)
            if end_win - start_win >= 2:
                saved = a[start_win:end_win].copy()
                avg = saved.mean()
                a[start_win:end_win] = avg
                move = ("smooth", start_win, end_win, saved)

        # 10 % swap
        else:
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move = ("swap", i, j)

        # evaluate the new state (single FFT)
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = float(2.0 * n * conv_new.max() / (a.sum() ** 2))

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a.copy()
            cur_score = new_score
            conv_cur = conv_new
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA new best {best_score:.6f}")
        else:
            # revert move
            if move is not None:
                if move[0] == "transfer":
                    _, donor, receiver, delta = move
                    a[donor] += delta
                    a[receiver] -= delta
                elif move[0] == "smooth":
                    _, s_w, e_w, saved = move
                    a[s_w:e_w] = saved
                elif move[0] == "swap":
                    i, j = move[1], move[2]
                    a[i], a[j] = a[j], a[i]

        # occasional global jitter
        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)
            conv_cur = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best, best_score


def _initialize_sequence(n: int, mode: str, rng: np.random.Generator) -> np.ndarray:
    """Create an initial non‑negative sequence of length n."""
    target = np.sqrt(2.0 * n)

    if mode == "global" and "height_sequence_1" in globals():
        seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if seq.size > n:
            seq = seq[:n]
        else:
            seq = np.pad(seq, (0, n - seq.size), "constant")
    elif mode == "power":
        p = rng.uniform(0.4, 0.8)          # random exponent for diversity
        seq = np.arange(1, n + 1, dtype=np.float64) ** (-p)
        seq *= target / seq.sum()
    elif mode == "random":
        seq = rng.random(n)
        seq *= target / seq.sum()
    else:  # fallback
        p = 0.55
        seq = np.arange(1, n + 1, dtype=np.float64) ** (-p)
        seq *= target / seq.sum()

    seq = _project_to_simplex(seq, target)
    return seq

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Optimise a step‑function sequence under the 1000 s budget.
    The routine performs several independent runs (default 5) on a
    relatively large problem size (n≈5000).  Each run consists of
    soft‑max gradient descent, sub‑gradient descent on the true max,
    and simulated‑annealing refinement.  The best sequence found is
    returned.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.2  # small safety margin

    best_seq = None
    best_score = float("inf")

    # problem size – bigger gives more freedom but costs more per iteration
    n = 5000 if budget_s >= 250 else 2000

    max_runs = 5
    for run in range(max_runs):
        if time.time() >= deadline - 30.0:
            break

        # allocate a slice of the remaining time to this run
        now = time.time()
        remaining = deadline - now
        run_deadline = now + remaining * 0.65

        init_mode = rng.choice(["global", "power", "random"])
        a0 = _initialize_sequence(n, init_mode, rng)

        # Phase 1 – soft‑max gradient descent
        gd_deadline = time.time() + min(30.0, (run_deadline - time.time()) * 0.4)
        a_gd, score_gd = _gradient_descent_softmax(a0, gd_deadline)

        # Phase 2 – sub‑gradient descent on the actual max
        sub_deadline = time.time() + min(30.0, (run_deadline - time.time()) * 0.3)
        a_sub, score_sub = _subgradient_descent(a_gd, sub_deadline)

        # Phase 3 – simulated annealing
        sa_deadline = run_deadline
        a_sa, score_sa = _simulated_annealing(a_sub, sa_deadline)

        run_best_score = min(score_gd, score_sub, score_sa)
        if run_best_score < best_score:
            best_score = run_best_score
            best_seq = a_sa.copy()
            print(f"[{time.time():.1f}] Run {run+1}/{max_runs} mode={init_mode} → new global best {best_score:.6f}")

    # Final short refinement on the overall best candidate
    if best_seq is not None and time.time() < deadline - 5.0:
        final_deadline = time.time() + (deadline - time.time()) * 0.25
        best_seq, _ = _gradient_descent_softmax(best_seq, final_deadline)

    # Strip trailing near‑zeros for a compact representation
    if best_seq is not None:
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[: nz[-1] + 1]
        else:
            best_seq = np.array([], dtype=float)
    else:
        best_seq = np.array([], dtype=float)

    print(f"[{time.time():.1f}] Finished. Best score≈{best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
