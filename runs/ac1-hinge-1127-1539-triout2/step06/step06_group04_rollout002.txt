def _gradient_descent_opt(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 65536.0
    beta_factor = 1.5

    step_base = target_sum * 0.2  # increased

    momentum = np.zeros_like(a)
    rho = 0.9

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # jitter more frequently
        if it % 100 == 0:
            jitter_scale = target_sum * 0.003 / (beta + 1.0)
            jitter = rng.normal(scale=jitter_scale, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        if it % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1

    return a
```

"""
Propose a step‑function (non‑negative sequence) that minimises the
``evaluate_sequence`` objective.

The algorithm follows a simple multi‑stage strategy:

1.  **Seed** – starts from the previously discovered best construction
    (``height_sequence_1``) when available.
2.  **Length schedule** – repeatedly up‑samples the current best sequence
    to larger lengths (2000 → 5000 → 8000 → 12000 → 18000 → 24000) so that
    more degrees of freedom become available.
3.  **Projected gradient descent** – optimises a smooth log‑sum‑exp
    surrogate of the max‑convolution using FFT‑based gradients.  A higher
    learning‑rate, a larger momentum decay and more frequent jitter are
    used compared with the reference code.
4.  **Targeted hill‑climbing** – repeatedly moves a small amount of mass
    from indices that contribute most to the current convolution peak to
    indices that contribute least.  The move is accepted if it lowers the
    peak or with a Metropolis probability that slowly cools.
5.  **Final polishing** – a last hill‑climbing run consumes any remaining
    time budget.

All steps respect the ℓ¹‑simplex constraint (∑a = √(2 n)), which makes the
objective equal to the maximal self‑convolution entry.
The routine stops exactly at the supplied wall‑clock budget and returns
the best sequence found as a plain list.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using an FFT (linear convolution)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Duchi‑et‑al. (2008) algorithm.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of   (1/β)·log Σ exp(β·b)   where   b = a*a .
    Returns (g, b) with   g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smoothed maximum.
    Uses a higher step size, larger momentum decay and more frequent jitter
    than the reference implementation.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 65536.0
    beta_factor = 1.5

    # more aggressive base step (relative to target ℓ¹‑norm)
    step_base = target_sum * 0.20
    momentum = np.zeros_like(a)
    rho = 0.90                               # stronger momentum

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        # projection onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # jitter – more frequent than the reference code
        if it % 100 == 0:
            jitter_scale = target_sum * 0.003 / (beta + 1.0)
            jitter = rng.normal(scale=jitter_scale, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β‑annealing (slower increase)
        if it % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 8)                     # eight families now

    idx = np.arange(n, dtype=float)

    if family == 0:                                 # pure power‑law
        p = rng.uniform(0.10, 1.2)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                               # exponential decay
        lam = rng.exponential(scale=0.018)
        a = np.exp(-lam * idx)

    elif family == 2:                               # mixed power‑law × exponential
        p = rng.uniform(0.12, 1.0)
        lam = rng.exponential(scale=0.018)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                               # Dirichlet (uniform)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                               # piecewise exponential
        cut = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(cut)),
             np.exp(-lam2 * np.arange(cut, n))]
        )

    elif family == 5:                               # near‑uniform with jitter
        a = np.full(n, target / n)

    elif family == 6:                               # sparse Dirichlet (alpha<1)
        a = rng.dirichlet(alpha=np.full(n, 0.3)) * target

    else:                                            # random spikes
        k = max(1, int(n * 0.02))
        a = np.zeros(n)
        idx_spike = rng.choice(n, size=k, replace=False)
        a[idx_spike] = rng.uniform(0.1, 1.0, size=k)
        a = a / (a.sum() + 1e-30) * target

    # Small Gaussian noise and projection onto the simplex
    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


def _hill_climb_targeted(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Hill‑climbing that always looks at the current maximal convolution entry.
    Moves a modest amount of mass from a source contributing heavily to that
    entry to a destination that contributes little.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size

    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    # start with a slightly larger temperature to explore more
    T = cur_max * 0.05 + 1e-9
    cooling = 0.9999

    while time.time() < deadline:
        # index of the current peak
        k_max = int(np.argmax(b))

        # contributions of each index to that peak:
        j_arr = k_max - np.arange(n)
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[mask] * a[j_arr[mask]]

        total = contrib.sum()
        if total > 0.0:
            prob_src = contrib / total
            i = rng.choice(n, p=prob_src)
        else:
            prob_src = a / (a.sum() + 1e-30)
            i = rng.choice(n, p=prob_src)

        # destination – favour small contribution
        inv = contrib.max() - contrib + 1e-12
        prob_dst = inv / inv.sum()
        j = rng.choice(n, p=prob_dst)
        if i == j:
            continue

        # amount of mass to move (up to 80 % of source)
        max_delta = a[i] * 0.8
        if max_delta < 1e-16:
            continue
        delta = rng.random() * max_delta

        # ---- analytical update of the convolution --------------------
        b_new = b.copy()
        # linear part  (‑2 δ·a shifted by i, +2 δ·a shifted by j)
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a

        # quadratic correction terms
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        # ---- acceptance test -----------------------------------------
        accept = False
        if new_max < cur_max - 1e-12:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            a[i] = max(a[i], 0.0)
            a[j] = max(a[j], 0.0)

            b = b_new
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_targeted improvement: {best_max:.6f}")

        T *= cooling

    return best_a, best_max
