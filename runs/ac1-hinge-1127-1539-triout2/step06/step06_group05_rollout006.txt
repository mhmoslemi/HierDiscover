def _solve_lp_for_g(f, M):
    n = f.shape[0]
    # build indices
    i = np.repeat(np.arange(n), n)
    j = np.tile(np.arange(n), n)
    rows = i + j
    cols = j
    data = f[i]
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b = np.full(2*n-1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback: return uniform small vector
        g = np.full(n, 1e-8)
    else:
        g = res.x
    return g
```

def _frank_wolfe_optimize(a0, deadline, rng, enforce_sym=False):
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # ensure positivity
    eps = 1e-10
    a = np.maximum(a, eps)
    a = _project_simplex(a, S)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    iteration = 0
    while time.time() < deadline:
        iteration += 1
        conv = _full_convolution(a)
        M = conv.max()
        # solve LP
        g = _solve_lp_for_g(a, M)
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g_scaled = g * (S / sum_g)

        # direction
        d = g_scaled - a
        # step size
        gamma = 2.0/(iteration + 2)
        # update
        a = a + gamma * d

        # enforce positivity
        a = np.maximum(a, 0.0)
        # optional symmetry
        if enforce_sym:
            a = _symmetrize(a)

        # re-project onto simplex just in case
        a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional jitter
        if rng.random() < 0.02:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.05 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        # early stop if not improve for many iterations - optional
        if iteration > 150 and best_score >= M - 1e-6:
            # no improvement
            break

    return best_a, best_score

"""
Hybrid Frank‑Wolfe optimizer for the self‑convolution peak problem.

The code builds a linear program that, for a given vector f,
maximises the ℓ¹‑norm of a feasible partner g under the
constraint ‖f⋆g‖∞ ≤ M where M = max(f⋆f).  Scaling g to the
required ℓ¹‑norm and taking a (diminishing) convex combination
f←(1‑γ)f+γg yields a descent step on the peak value.
A simple projected simplex implementation, FFT based convolutions
and a lightweight simulated‑annealing style jitter are used.
Multiple restarts with random lengths (≈1 k–4 k) explore the
search space within the 1000 s budget; the best sequence found
is returned.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x>=0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based full convolution (linear, not circular)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·n‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Symmetrisation helper
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry v[i] = v[-1‑i]."""
    return 0.5 * (v + v[::-1])


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) – used for seeding larger runs
# ----------------------------------------------------------------------
def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linearly interpolate a sequence to a new length n."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Rich family of randomized initialisers
# ----------------------------------------------------------------------
def _initial_sequence(base_seq: np.ndarray | None,
                     n: int,
                     rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n with ℓ¹‑norm √(2·n).
    The implementation mixes several analytic families to keep the
    search diverse.
    """
    S = np.sqrt(2.0 * n)

    typ = rng.integers(0, 6)

    if typ == 0 and base_seq is not None and base_seq.size >= 10:
        a = _upsample_seq(base_seq, n)
        a += rng.normal(scale=a.mean() * 1e-3, size=n)
        a = np.clip(a, 0.0, None)

    elif typ == 1:
        a = rng.random(n)                     # uniform

    elif typ == 2:
        rate = rng.uniform(0.9, 0.9995)
        left = rate ** np.arange(n)
        a = (left + left[::-1]) / 2.0        # symmetric geometric decay

    elif typ == 3:
        idx = np.arange(n, dtype=float)
        a = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.8)
        a = a ** power                       # edge‑heavy reciprocal

    elif typ == 4:
        centre = (n - 1) / 2.0
        a = 1.0 - np.abs(np.arange(n) - centre) / centre
        a = np.clip(a, 0.0, None)            # triangular

    else:
        # random convex combination of a few bases
        cand = []
        cand.append(rng.random(n))                     # uniform
        rate = rng.uniform(0.9, 0.9995)
        left = rate ** np.arange(n)
        cand.append((left + left[::-1]) / 2.0)          # geometric
        idx = np.arange(n, dtype=float)
        rec = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.8)
        cand.append(rec ** power)                       # reciprocal
        w = rng.random(len(cand))
        w /= w.sum()
        a = sum(wi * ci for wi, ci in zip(w, cand))

    # ensure no zero entries (helps the LP from becoming unbounded)
    eps = 1e-9
    a = np.maximum(a, eps)

    # normalise to prescribed ℓ¹‑norm
    a = a / a.sum() * S
    return a


# ----------------------------------------------------------------------
# Linear‑program that maximises Σg_j subject to f⋆g ≤ M (element‑wise)
# ----------------------------------------------------------------------
def _solve_lp_for_g(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   max Σ g_j   s.t.   f⋆g ≤ M,   g ≥ 0.
    Returns the optimal g (may be all‑zero if the LP fails).
    """
    n = f.shape[0]
    # build the convolution matrix A (dense in a sparse format)
    # rows = i + j, cols = j, data = f[i] for all i,j
    i = np.repeat(np.arange(n), n)          # shape (n*n,)
    j = np.tile(np.arange(n), n)            # shape (n*n,)
    rows = i + j
    cols = j
    data = f[i]                             # f_i repeated n times
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    b = np.full(2 * n - 1, M, dtype=float)
    c = -np.ones(n, dtype=float)            # maximise Σg ⇔ minimise -Σg
    bounds = [(0.0, None)] * n

    # Use HiGHS – works with sparse A_ub
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        # fall back to a tiny uniform vector (guarantees feasibility)
        g = np.full(n, 1e-9)
    else:
        g = res.x
    return g


# ----------------------------------------------------------------------
# Frank‑Wolfe style optimiser that uses the LP above
# ----------------------------------------------------------------------
def _frank_wolfe_optimize(a0: np.ndarray,
                          deadline: float,
                          rng: np.random.Generator,
                          enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Perform a few Frank‑Wolfe iterations (LP → convex combination)
    until the local time budget expires.  Returns the best vector
    encountered and its convolution peak.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # keep a tiny floor to avoid an unbounded LP
    a = np.maximum(a, 1e-9)
    a = _project_simplex(a, S)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        M = conv.max()

        # ---- LP step -------------------------------------------------
        g = _solve_lp_for_g(a, M)
        sum_g = g.sum()
        if sum_g == 0.0:
            break
        g_scaled = g * (S / sum_g)          # enforce the ℓ¹‑norm

        # ---- convex combination (Frank‑Wolfe step) -------------------
        # diminishing step size γ = 2/(t+2)  (standard FW schedule)
        gamma = 2.0 / (it + 2.0)
        a = a + gamma * (g_scaled - a)

        # optional symmetry enforcement
        if enforce_sym:
            a = _symmetrize(a)

        # re‑project onto the simplex (numerical safety)
        a = _project_simplex(a, S)

        # occasional random jitter to escape shallow plateaus
        if rng.random() < 0.018:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = 0.04 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # early exit if we have stagnated for a while
        if it % 800 == 0 and best_score >= cur_score - 1e-8:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function whose self‑convolution peak is minimal.
    The routine repeatedly restarts a Frank‑Wolfe optimiser (LP based)
    on random initial shapes, occasionally re‑using the best sequence
    discovered so far (up‑sampled to a new length).  The best vector
    encountered before the overall deadline is returned.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # keep a small safety margin for the external evaluator
    deadline = start + budget_s - 5.0

    best_seq = None
    best_score = np.inf
    attempts = 0

    # reuse a previously discovered construction if it exists
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    # allowed length range – small enough for the LP to stay tractable
    MIN_N = 1500
    MAX_N = 3500

    while time.time() < deadline:
        attempts += 1

        # ------------------------------------------------------------------
        # Choose a problem size.
        # ------------------------------------------------------------------
        if best_seq is not None and rng.random() < 0.45:
            # perturb the length of the current best by ±10 %
            scale = rng.uniform(0.9, 1.1)
            n = max(MIN_N, int(best_seq.shape[0] * scale))
        else:
            n = int(rng.integers(MIN_N, MAX_N + 1))

        enforce_sym = rng.random() < 0.48

        # ------------------------------------------------------------------
        # Initialise a fresh vector (optionally seeded from the global best).
        # ------------------------------------------------------------------
        a0 = _initial_sequence(base_seq, n, rng)
        if enforce_sym:
            a0 = _symmetrize(a0)

        # ------------------------------------------------------------------
        # Allocate a slice of the remaining budget for this attempt.
        # ------------------------------------------------------------------
        slice_deadline = min(time.time() + rng.uniform(10.0, 18.0), deadline)

        # ------------------------------------------------------------------
        # Optimise.
        # ------------------------------------------------------------------
        a_opt, score = _frank_wolfe_optimize(a0, slice_deadline, rng,
                                             enforce_sym=enforce_sym)

        # ------------------------------------------------------------------
        # Update the global best.
        # ------------------------------------------------------------------
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[global] attempt={attempts:3d} n={n:4d} sym={enforce_sym} "
                  f"new best={best_score:.6f}")

        # occasionally discard the current base to increase diversity
        if attempts % 12 == 0:
            base_seq = None

    # ----------------------------------------------------------------------
    # Final sanity check via the supplied black‑box evaluator.
    # ----------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()

    external_score = evaluate_sequence(final_seq)
    print(f"[final] external={external_score:.6f} internal_best={best_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    # expose the best sequence for potential up‑sampling in later runs
    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)

    return final_seq
