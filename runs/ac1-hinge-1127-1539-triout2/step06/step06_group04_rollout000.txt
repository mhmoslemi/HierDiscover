def _gradient_descent_opt(a0, deadline, symmetrize=False, rng=None):
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0*n)
    beta = 0.5; beta_max = 16384.; beta_factor = 1.5
    step_base = target_sum * 0.12
    momentum = np.zeros_like(a)
    rho = 0.92
    it = 0
    if rng is None:
        rng = np.random.default_rng()
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = rho*momentum + (1.0 - rho)*g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0:
            lr = step_base/(norm_m*(beta+1.0))
            a -= lr*momentum
        # projection
        a = _project_to_simplex(a, target_sum)
        if symmetrize:
            a = _symmetrize(a)
        # jitter
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum*0.001/(beta+1.0), size=n)
            a += jitter
            a = np.maximum(a,0.0)
            a = _project_to_simplex(a, target_sum)
            if symmetrize:
                a = _symmetrize(a)
        # beta schedule
        if it % 300 == 0 and beta < beta_max:
            beta = min(beta*beta_factor, beta_max)
        it += 1
    return a

"""
Propose a step‑function (non‑negative sequence) that minimises
``evaluate_sequence``.  The objective after the ℓ¹‑normalisation
Σ a_i = √(2 n) equals the maximum entry of the self‑convolution
b = a * a, so we directly minimise max(b).

Algorithm (≈ 1000 s budget)
--------------------------------
1. Optional warm‑start from the global ``height_sequence_1``.
2. Length schedule – we grow the vector size gradually
   [2000, 5000, 8000, 12000, 18000, 25000]; each size is
   obtained by linear up‑sampling of the current best solution.
3. For each length
   * initialise a *symmetric* distribution (mirror the random vector)
     and project onto the ℓ¹‑simplex.
   * **Phase A** – projected (symmetrised) gradient descent on a
     smooth log‑sum‑exp surrogate of the max (β‑annealing, Nesterov
     momentum, occasional jitter).  The iterate is kept symmetric
     after every step.
   * optional light smoothing of the result (local averaging) and a
     re‑projection.
   * **Phase B** – targeted hill‑climbing with exact convolution updates.
     The move selects a source index that contributes strongly to the
     current peak and a destination that contributes weakly; a
     random amount of mass (up to 80 % of the source) is transferred.
     A Metropolis acceptance with a slowly cooling temperature lets the
     search escape shallow plateaus.
   * If the new solution improves the best bound, update the global best.
4. Final polishing – a last hill‑climb on the best sequence for the
   remaining time budget.
5. Trim trailing near‑zero entries, print the final evaluation and
   return the sequence as a plain list.

All heavy work uses FFT‑based convolutions (O(n log n)) and the
exact update formulas of the hill‑climber (O(n) per move).  The code
stays well inside the 1 CPU limit and respects the time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a using FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Duchi‑et‑al. (2008) algorithm.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Make a symmetric vector: a <- (a + a[::-1]) / 2."""
    return (a + a[::-1]) * 0.5


def _smooth_sequence(a: np.ndarray, alpha: float = 0.1) -> np.ndarray:
    """
    Simple neighbour smoothing: a' = (1‑α)·a + α·(a * [½,½]) (same mode).
    The sum is preserved by a subsequent projection onto the ℓ¹‑simplex.
    """
    conv = np.convolve(a, np.array([0.5, 0.5]), mode="same")
    return (1.0 - alpha) * a + alpha * conv


# ----------------------------------------------------------------------
# Random initialisers (symmetric)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 7)

    idx = np.arange(n, dtype=float)

    if family == 0:  # pure power‑law
        p = rng.uniform(0.10, 1.2)
        a = (idx + 1.0) ** (-p)

    elif family == 1:  # exponential decay
        lam = rng.exponential(scale=0.018)
        a = np.exp(-lam * idx)

    elif family == 2:  # mixed power‑law × exponential
        p = rng.uniform(0.12, 1.0)
        lam = rng.exponential(scale=0.018)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:  # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:  # piecewise exponential
        cut = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(cut)),
             np.exp(-lam2 * np.arange(cut, n))]
        )

    elif family == 5:  # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    else:  # random Gaussian‑like bump
        centre = rng.uniform(0, 1)
        width = rng.exponential(scale=0.15) + 0.05
        a = np.exp(-((idx / n - centre) ** 2) / (2 * width ** 2))

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with g = 2·(a ⋆ p) and p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # b = a*a (FFT)
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p) (correlation)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _gradient_descent_opt(a0: np.ndarray,
                          deadline: float,
                          symmetrize: bool = False,
                          rng=None) -> np.ndarray:
    """
    Projected gradient descent on the smoothed maximum.
    Uses Nesterov‑style momentum, β‑annealing and occasional jitter.
    If ``symmetrize`` is True the iterate is forced to be symmetric
    after each projection.
    """
    if rng is None:
        rng = np.random.default_rng()
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 16384.0
    beta_factor = 1.5

    step_base = target_sum * 0.12
    momentum = np.zeros_like(a)
    rho = 0.92  # momentum decay

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        # projection onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        if symmetrize:
            a = _symmetrize(a)

        # jitter every few hundred steps
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            if symmetrize:
                a = _symmetrize(a)

        # β‑annealing
        if it % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a


# ----------------------------------------------------------------------
# Targeted hill‑climbing (exact convolution updates)
# ----------------------------------------------------------------------


def _hill_climb_targeted(a0: np.ndarray,
                         b0: np.ndarray,
                         deadline: float):
    """
    Mass‑transfer hill‑climbing that always looks at the current
    maximal convolution entry.  A source index that contributes heavily
    to that peak transfers a random fraction (up to 80 %) of its mass
    to a destination with a weak contribution.  Metropolis acceptance
    with a slowly cooling temperature is used.
    Returns the best sequence seen and its max‑convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size

    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    # initialise a temperature proportional to the current peak
    T = cur_max * 0.05 + 1e-9
    cooling = 0.9998  # very slow cooling → thorough exploration

    while time.time() < deadline:
        # index of the current peak
        k_max = int(np.argmax(b))

        # contribution of each index to that peak: a[i] * a[k_max-i]
        j_arr = k_max - np.arange(n)
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[mask] * a[j_arr[mask]]

        total = contrib.sum()
        if total > 0.0:
            prob_src = contrib / total
            i = rng.choice(n, p=prob_src)
        else:
            prob_src = a / a.sum()
            i = rng.choice(n, p=prob_src)

        # destination – weight towards small contribution
        inv = contrib.max() - contrib + 1e-12
        prob_dst = inv / inv.sum()
        j = rng.choice(n, p=prob_dst)

        if i == j:
            continue

        # amount of mass to move (up to 80 % of source)
        max_delta = a[i] * 0.8
        if max_delta < 1e-16:
            continue
        delta = rng.random() * max_delta

        # ---- analytical update of the convolution --------------------
        b_new = b.copy()
        # linear part  (‑2 δ·a shifted by i, +2 δ·a shifted by j)
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a

        # quadratic correction terms
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        # ---- acceptance test -----------------------------------------
        accept = False
        if new_max < cur_max - 1e-14:
            accept = True
        else:
            # Metropolis step
            if T > 0.0 and rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            a[i] = max(a[i], 0.0)
            a[j] = max(a[j], 0.0)

            b = b_new
            cur_max = new_max

            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_targeted improvement: "
                      f"{best_max:.6f}")

        T *= cooling

    return best_a, best_max


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the supplied
    wall‑clock budget (default 1000 s) and returns the best sequence
    found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered construction, if present.
    # ------------------------------------------------------------------
    earlier = globals().get("height_sequence_1")
    if isinstance(earlier, (list, tuple, np.ndarray)):
        a0 = np.asarray(earlier, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        best_seq = a0.copy()
        best_score = _max_convolution_fft(a0)
        print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) "
              f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Length schedule – we progressively enlarge the search space.
    # ------------------------------------------------------------------
    lengths = [2000, 5000, 8000, 12000, 18000, 25000]

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Initialise a candidate for this length (symmetrised)
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)

        a_init = _symmetrize(a_init)
        a_init = _project_to_simplex(a_init, target_sum)

        # --------------------------------------------------------------
        # Phase A – projected gradient descent (major share of time)
        # --------------------------------------------------------------
        time_left = deadline - time.time()
        phaseA_dead = time.time() + min(180.0, 0.55 * time_left)

        a_gd = _gradient_descent_opt(a_init, phaseA_dead,
                                     symmetrize=True, rng=rng)

        score_gd = _max_convolution_fft(a_gd)
        if score_gd < best_score:
            best_score = score_gd
            best_seq = a_gd.copy()
            print(f"[{time.time():.1f}] GD improvement (n={n}) score={score_gd:.6f}")

        # --------------------------------------------------------------
        # Optional light smoothing to flatten small ripples
        # --------------------------------------------------------------
        a_smooth = a_gd.copy()
        for _ in range(3):
            a_smooth = _smooth_sequence(a_smooth, alpha=0.1)
            a_smooth = _symmetrize(a_smooth)
            a_smooth = _project_to_simplex(a_smooth, target_sum)
        score_smooth = _max_convolution_fft(a_smooth)
        if score_smooth < best_score:
            best_score = score_smooth
            best_seq = a_smooth.copy()
            print(f"[{time.time():.1f}] smoothing improvement (n={n}) "
                  f"score={score_smooth:.6f}")

        # --------------------------------------------------------------
        # Phase B – targeted hill climbing
        # --------------------------------------------------------------
        if time.time() < deadline:
            b_full = _full_convolution(best_seq if best_seq.size == n else a_gd)
            phaseB_dead = time.time() + min(120.0,
                                            0.35 * (deadline - time.time()))
            a_hc, max_hc = _hill_climb_targeted(best_seq if best_seq.size == n else a_gd,
                                                b_full,
                                                phaseB_dead)
            if max_hc < best_score:
                best_score = max_hc
                best_seq = a_hc.copy()
                print(f"[{time.time():.1f}] Hill‑climb improvement (n={n}) "
                      f"score={max_hc:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with any remaining budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a_final = best_seq.copy()
        b_final = _full_convolution(a_final)
        a_final, max_final = _hill_climb_targeted(a_final, b_final, deadline)
        if max_final < best_score:
            best_score = max_final
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final polishing improvement: "
                  f"score={max_final:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries for neatness
    # ------------------------------------------------------------------
    if best_seq is None:
        # should never happen – fall back to a uniform vector
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log the official evaluation value
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:   # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
