def subgradient_descent(a, deadline):
    target_sum = sqrt(2 * n)
    best_a = a.copy()
    best_val = _score(a)
    step = target_sum * 0.1 # scaling
    itr = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        max_idx = int(conv.argmax())
        # compute gradient
        a_rev = a[::-1]
        w = np.zeros(2*n-1)
        w[max_idx] = 1.0
        # conv of a_rev and w: using FFT
        size = 1 << ((2*n-1).bit_length())
        wa_fft = np.fft.rfft(w, n=size)
        ar_fft = np.fft.rfft(a_rev, n=size)
        grad_rev = np.fft.irfft(ar_fft * wa_fft, n=size)[: 2*n-1]
        g = 2.0 * grad_rev[n-1:n-1+n]
        # step update
        a = a - step * g
        # project to simplex
        a = _project_to_simplex(a, target_sum)
        # maybe adapt step: if new score > old, reduce step size
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
        # adapt step
        if val > best_val:
            step *= 0.5
        else:
            step *= 1.02
        itr += 1
    return best_a, best_val

"""
Propose a step‑function sequence that (hopefully) improves the current best
upper bound on the objective

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where `conv(a, a)` is the full linear convolution of `a` with itself.
The algorithm combines:

* a fast sub‑gradient descent on the exact ℓ∞‑norm (the maximum convolution
  entry) – the sub‑gradient is simply a shifted copy of the current vector;
* a simulated‑annealing phase that moves mass away from the indices that
  contribute most to the current worst convolution entry;
* a final short sub‑gradient refinement.

All iterates are projected onto the ℓ¹‑simplex {x ≥ 0, Σx = √(2·n)} so the
scale‑invariance of the objective is respected.  The code adapts the problem
size to the available time budget (≈1000 s) and prints intermediate scores.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using FFT (returns length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a (computed via FFT)."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective value for a (assumes a is already non‑negative)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, Σx = s}.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Generate (i+1)^(-p) of length n and renormalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution (ℓ∞) term
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent for the objective
    max_i (a*a)_i, i.e. we minimise the ℓ∞‑norm of the convolution.
    The sub‑gradient for the maximal entry at index k is
    g_j = 2·a_{k‑j} (if the index exists, otherwise 0).
    """
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    # Initial step size – a fraction of the typical entry size.
    step = target * 0.2
    it = 0
    idx = np.arange(n)

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_idx = int(conv.argmax())

        # Sub‑gradient (vectorised)
        j = max_idx - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # Gradient step + projection onto the ℓ¹‑simplex
        a -= step * g
        a = _project_to_simplex(a, target)

        # Evaluate
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] Sub‑grad iter {it:6d}  best={best_val:.6f}")

        # Simple step‑size adaptation
        if val > best_val:
            step *= 0.5
        else:
            step *= 1.01

        # Occasionally add a tiny jitter to escape plateaus
        if it % 500 == 0 and it > 0:
            jitter = np.random.default_rng().normal(scale=target * 1e-4, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)
            conv = _conv_fft(a)  # recompute for next iteration

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing refinement (mass‑transfer + short‑window smoothing)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Heuristic SA that repeatedly moves probability mass away from the indices
    that contribute most to the current worst convolution entry.
    """
    a = a0.copy()
    n = a.size
    cur_score = _score(a)

    best_a = a.copy()
    best_score = cur_score

    start = time.time()
    total = deadline - start
    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / total
        # Exponential temperature schedule
        temp = 0.8 * (1e-5 / 0.8) ** tau
        max_frac = 0.30 * (0.001 / 0.30) ** tau  # proportion of donor's mass to move

        # Locate current worst entry and its contributing weights
        max_val = conv.max()
        k_max = int(conv.argmax())
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        # Contribution of a_i to the worst entry (a_i * a_{k_max-i})
        contrib = a[idx] * a[k_max - idx]

        # ------------------------------------------------------------------
        # Propose a move
        # ------------------------------------------------------------------
        move_type = None
        # 70 % donor→receiver mass transfer
        if rng.random() < 0.70:
            total_c = contrib.sum()
            if total_c > 0:
                donor = rng.choice(idx, p=contrib / total_c)
            else:
                donor = rng.integers(n)

            # Choose a receiver that currently contributes little
            low_thr = np.percentile(contrib, 10)
            low_idx = idx[contrib <= low_thr]
            if low_idx.size:
                receiver = rng.choice(low_idx)
            else:
                receiver = rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        # 20 % local smoothing of a short random window
        elif rng.random() < 0.90:
            start_win = rng.integers(0, n)
            length = rng.integers(2, 31)  # up to 30
            end_win = min(start_win + length, n)
            if end_win - start_win >= 2:
                saved = a[start_win:end_win].copy()
                a[start_win:end_win] = saved.mean()
                move_type = "smooth"
                saved_range = (start_win, end_win)

        # 10 % random swap of two entries
        else:
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # ------------------------------------------------------------------
        # Evaluate new state (single FFT)
        # ------------------------------------------------------------------
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = _score(a)

        # ------------------------------------------------------------------
        # Metropolis acceptance
        # ------------------------------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-14))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA iter {it:6d}  new best={best_score:.6f}")
        else:
            # revert the move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                s, e = saved_range
                a[s:e] = saved
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # Occasional global jitter (helps to escape deep local minima)
        if it % 4000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, np.sqrt(2.0 * n))
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function sequence.
    The routine adapts its problem size to the remaining time,
    runs a sub‑gradient phase, a simulated‑annealing refinement,
    and a final sub‑gradient polish before returning the best sequence.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # a small safety margin

    # ------------------------------------------------------------------
    # Problem size selection (larger n → more degrees of freedom)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 800:
        n = 10000
    elif remaining > 500:
        n = 8000
    elif remaining > 300:
        n = 6000
    elif remaining > 150:
        n = 4000
    else:
        n = 2500

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Initialise a promising starting point (power law + slight jitter)
    # ------------------------------------------------------------------
    p_exp = rng.uniform(0.4, 0.7)          # exponent of the power law
    a0 = _power_law_sequence(n, p_exp, target_sum)
    a0 += rng.normal(scale=target_sum * 0.01, size=n)  # small random perturbation
    a0 = np.maximum(a0, 0.0)
    a0 = _project_to_simplex(a0, target_sum)

    # Warm‑start from a previous global solution if available
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if old.size >= n:
            a0 = old[:n].copy()
        else:
            a0 = np.pad(old, (0, n - old.size), "constant")
        a0 = _project_to_simplex(a0, target_sum)

    best_seq = a0.copy()
    best_score = _score(best_seq)

    # ------------------------------------------------------------------
    # Phase 1 – sub‑gradient descent (rough optimisation)
    # ------------------------------------------------------------------
    phase1_deadline = time.time() + min(180.0, (deadline - time.time()) * 0.25)
    seq_sg, score_sg = _subgradient_descent(best_seq, phase1_deadline)
    if score_sg < best_score:
        best_score, best_seq = score_sg, seq_sg.copy()
        print(f"[{time.time():.1f}] After SG1 best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – simulated annealing (exploration & refinement)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        seq_sa, score_sa = _simulated_annealing(best_seq, deadline - 10.0)
        if score_sa < best_score:
            best_score, best_seq = score_sa, seq_sa.copy()
            print(f"[{time.time():.1f}] After SA best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final short sub‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        final_deadline = deadline - 1.0
        seq_final, score_final = _subgradient_descent(best_seq, final_deadline)
        if score_final < best_score:
            best_score, best_seq = score_final, seq_final.copy()
            print(f"[{time.time():.1f}] Final polish best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑process: strip trailing zeros (they do not affect the score)
    # ------------------------------------------------------------------
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = np.array([], dtype=float)

    print(f"[{time.time():.1f}] DONE – best score≈{best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
