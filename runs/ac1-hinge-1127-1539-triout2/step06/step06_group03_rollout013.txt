def _order_zigzag(v):
    n = v.size
    sorted_desc = np.sort(v)[::-1]  # descending
    result = np.empty_like(v)
    left = 0
    right = n-1
    idx = 0
    while left <= right:
        result[left] = sorted_desc[idx]
        idx += 1
        left += 1
        if idx < n:
            result[right] = sorted_desc[idx]
            idx += 1
            right -= 1
    return result

"""
Propose a non‑negative step‑function that minimises the
``evaluate_sequence`` objective.

Algorithm (≈ 1000 s budget)
--------------------------------
* **Initial seeds** – start from the provided ``height_sequence_1`` (if any)
  and from a few hand‑crafted power‑law families.  Values are normalised
  to the ℓ¹‑simplex Σa = √(2 n).

* **Shape optimisation** – a projected gradient descent on a smooth
  log‑sum‑exp surrogate of the maximal convolution entry.  The smoothing
  parameter β is annealed up to 1e⁸, momentum is used and occasional jitter
  keeps the iterates away from plateaus.

* **Permutation annealing** – with the multiset of values fixed,
  random swaps are performed while the current maximum convolution entry is
  monitored (FFT‑based).  Unimproved swaps are accepted with a temperature
  schedule that slowly cools.

* **Continuous annealing** – finally a simulated‑annealing phase on the
  *values* themselves: a small random multiplicative perturbation is applied
  to a randomly chosen sub‑set of entries, the vector is projected back to
  the ℓ¹‑simplex and the new objective is evaluated.  This can escape the
  basin of the previous local optimum.

* The best sequence seen throughout the run is returned as a plain Python
  list.  All steps respect the global deadline; if the deadline is reached
  early the current best is output.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a * a (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())            # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex Σx = s, x≥0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection onto {x_i ≥ 0, Σx_i = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (power‑law, exponential, truncated linear)
# ----------------------------------------------------------------------


def _normalize_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale positive vector to have ℓ¹ ‑sum = target."""
    scale = target / np.sum(v)
    a = v * scale
    return np.maximum(a, 0.0)


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _exponential_sequence(n: int, rate: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.exp(-rate * (idx + shift))
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _linear_sequence(n: int, slope: float, intercept: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, intercept - slope * idx)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


# ----------------------------------------------------------------------
# Ordering heuristics
# ----------------------------------------------------------------------


def _order_zigzag(v: np.ndarray) -> np.ndarray:
    """
    Place the largest values at the two ends and walk towards the centre:
    e.g. [max, second_max, 3rd, …] → [max, … , second_max] pattern.
    """
    n = v.size
    desc = np.sort(v)[::-1]          # descending
    out = np.empty_like(v)
    left, right = 0, n - 1
    i = 0
    while left <= right:
        out[left] = desc[i]
        i += 1
        left += 1
        if i < n:
            out[right] = desc[i]
            i += 1
            right -= 1
    return out


def _order_alt_small_first(v: np.ndarray) -> np.ndarray:
    """alternating small – large (small first) – classic baseline."""
    n = v.size
    asc = np.sort(v)          # smallest → largest
    out = np.empty_like(v)
    lo, hi = 0, n - 1
    for i in range(n):
        if i % 2 == 0:
            out[i] = asc[lo]
            lo += 1
        else:
            out[i] = asc[hi]
            hi -= 1
    return out


def _order_descending(v: np.ndarray) -> np.ndarray:
    """pure descending order (largest → smallest)."""
    return np.sort(v)[::-1]


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) surrogate of max(b) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b) where
        b = a*a (full convolution)
        g = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a ⋆ p),   p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]

    # full convolution a*a
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Projected gradient descent on the smoothed maximum
# ----------------------------------------------------------------------


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the log‑sum‑exp surrogate of the max
    convolution entry.  The smoothing parameter β is annealed up to 1e8.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 1.0
    max_beta = 1e8
    momentum = np.zeros_like(a)
    rho = 0.9                     # momentum decay
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum step
        momentum = rho * momentum + (1.0 - rho) * g
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            # step size that roughly keeps the ℓ¹‑norm stable
            lr = 0.5 * target_sum / (norm * (beta + 1.0))
            a -= lr * momentum

        # projection onto the simplex (keeps Σa fixed)
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if it % 200 == 0:
            jitter = rng.normal(
                scale=target_sum * 5e-4 / (beta + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal β
        if it % 150 == 0 and beta < max_beta:
            beta = min(beta * 1.07, max_beta)

        it += 1
    return a


# ----------------------------------------------------------------------
# Permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------


def _permutation_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Improves the ordering of a fixed multiset by random swaps.
    Accepts worsening moves with a slowly cooling temperature.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    T = best_score * 0.03 + 1e-9          # initial temperature
    cooling = 0.9996
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose swap
        a[i], a[j] = a[j], a[i]

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        accept = False
        if cur_score <= best_score:
            accept = True
        elif rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # revert swap
            a[i], a[j] = a[j], a[i]

        T *= cooling

        # occasional tiny reshuffle of 1‑2 % of entries
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.015
            if np.any(mask):
                shuffled = a[mask].copy()
                rng.shuffle(shuffled)
                a[mask] = shuffled
                cur_score = _max_convolution_fft(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()

    return best


# ----------------------------------------------------------------------
# Continuous annealing on the values (multiplicative jitter)
# ----------------------------------------------------------------------


def _continuous_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simulated annealing on the real values.
    At each step a random subset is multiplied by exp(ε) with ε~N(0,σ²).
    The vector is projected back onto the ℓ¹‑simplex after each move.
    """
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)

    best = a.copy()
    best_score = _max_convolution_fft(best)

    T = best_score * 0.12 + 1e-9          # start a bit hotter
    cooling = 0.99993
    rng = np.random.default_rng()

    while time.time() < deadline:
        # pick a modest fraction of indices (≈1‑2 %)
        num = max(1, int(0.015 * n))
        idx = rng.choice(n, size=num, replace=False)

        # multiplicative log‑normal perturbation
        factors = np.exp(rng.normal(scale=0.08, size=num))
        a_candidate = a.copy()
        a_candidate[idx] *= factors

        # project back to the simplex
        a_candidate = _project_to_simplex(a_candidate, target)

        cur_score = _max_convolution_fft(a_candidate)
        dE = cur_score - best_score
        accept = False
        if cur_score <= best_score:
            accept = True
        elif rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            a = a_candidate
            if cur_score < best_score:
                best_score = cur_score
                best = a_candidate.copy()

        T *= cooling

    return best


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the target
    ``evaluate_sequence`` objective.  The routine respects the allotted
    time budget, starts from a few solid seeds (including the optional
    global ``height_sequence_1``), refines them with a gradient descent,
    then improves ordering with a long simulated‑annealing phase, and
    finally performs a value‑annealing stage.  The best sequence found is
    returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_obj = np.inf

    # ------------------------------------------------------------------
    # 0.  Incorporate the optional pre‑computed sequence
    # ------------------------------------------------------------------
    seq0 = globals().get('height_sequence_1')
    if isinstance(seq0, list):
        a0 = np.asarray(seq0, dtype=float)
        if a0.size > 0:
            a0 = _project_to_simplex(a0, np.sqrt(2.0 * a0.size))
            sc = _max_convolution_fft(a0)
            best_seq = a0.copy()
            best_obj = sc
            print(f"[{time.time():.1f}] Seed from height_sequence_1, obj={sc:.6f}")

    # ------------------------------------------------------------------
    # 1.  Generate a few strong seeds from a power‑law family
    # ------------------------------------------------------------------
    # lengths that provide a good balance between expressivity and speed
    lengths = [5000, 6000, 7000, 8000]
    for n in lengths:
        if time.time() >= deadline:
            break
        target = np.sqrt(2.0 * n)

        # --- deterministic seed (the one that gave the previous best) ---
        base = _powerlaw_sequence(n, p=0.5, shift=0.60)
        for ordering in (_order_zigzag(base),
                         _order_descending(base),
                         _order_alt_small_first(base)):
            cand = _project_to_simplex(ordering, target)
            sc = _max_convolution_fft(cand)
            if sc < best_obj:
                best_obj = sc
                best_seq = cand.copy()
                print(f"[{time.time():.1f}] Seed (det) n={n} obj={sc:.6f}")

        # --- randomised parameter sweep (tight budget) ---
        # allocate a small slice of remaining time for this inner loop
        inner_dead = min(deadline, time.time() + 1.5)
        while time.time() < inner_dead:
            p = rng.uniform(0.45, 0.55)
            shift = rng.uniform(0.4, 0.8)
            base = _powerlaw_sequence(n, p=p, shift=shift)
            # use the edge‑biased ordering
            cand = _project_to_simplex(_order_zigzag(base), target)
            sc = _max_convolution_fft(cand)
            if sc < best_obj:
                best_obj = sc
                best_seq = cand.copy()
                print(f"[{time.time():.1f}] Seed (rand) n={n} p={p:.4f} sh={shift:.3f} obj={sc:.6f}")

    # ------------------------------------------------------------------
    # 2.  Refine the current best with projected gradient descent
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        # spend roughly half of the remaining budget on gradient descent
        gd_dead = time.time() + 0.45 * (deadline - time.time())
        refined = _gradient_descent_opt(best_seq, gd_dead)
        sc = _max_convolution_fft(refined)
        if sc < best_obj:
            best_obj = sc
            best_seq = refined.copy()
            print(f"[{time.time():.1f}] After gradient descent obj={sc:.6f}")

    # ------------------------------------------------------------------
    # 3.  Permutation annealing (ordering optimisation)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        # use the rest of the time for a long swap‑based annealing run
        refined = _permutation_annealing(best_seq, deadline)
        sc = _max_convolution_fft(refined)
        if sc < best_obj:
            best_obj = sc
            best_seq = refined.copy()
            print(f"[{time.time():.1f}] After permutation annealing obj={sc:.6f}")

    # ------------------------------------------------------------------
    # 4.  Final continuous annealing on the values themselves
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        refined = _continuous_annealing(best_seq, deadline)
        sc = _max_convolution_fft(refined)
        if sc < best_obj:
            best_obj = sc
            best_seq = refined.copy()
            print(f"[{time.time():.1f}] After continuous annealing obj={sc:.6f}")

    # ------------------------------------------------------------------
    # 5.  Clean up tiny trailing entries (purely cosmetic)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform distribution (should never happen)
        n = 3000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_obj = _max_convolution_fft(best_seq)

    nz = np.where(best_seq > 1e-12)[0]
    if nz.size > 0:
        best_seq = best_seq[: nz[-1] + 1]

    # ------------------------------------------------------------------
    # 6.  Final verification with the supplied objective function
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective = {best_obj:.6f} "
          f"length = {best_seq.size}")
    return best_seq.tolist()
