def propose_candidate(seed=42, budget_s=1000, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 0.5
    rng = np.random.default_rng(seed)
    # get initial best_seq maybe from height_sequence_1 else uniform
    best_seq = None
    best_score = np.inf

    # initial candidate: if height_sequence_1 exists
    if exists: a0 = ... ; Compute score; set best.

    lengths = [2000, 5000, 8000, 12000, 18000]
    # Optionally extend to 26000 if time.

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # get initial candidate by upsampling
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)

        a_init = _project_to_simplex(a_init, target_sum)

        # Now run a few GD attempts (maybe 2-3)
        time_per_length = deadline - time.time()
        # allocate maybe 0.6 of remaining for GD, rest for HC
        gd_time_budget = min(200.0, 0.6 * time_per_length) # adapt
        # split into runs
        runs = max(1, int(gd_time_budget // 30.0)) # ensure at least
        for run in range(runs):
            # add small jitter to init for each run
            a_guess = a_init.copy()
            jitter = rng.normal(scale=target_sum * 0.005, size=n)
            a_guess = np.maximum(a_guess + jitter, 0)
            a_guess = _project_to_simplex(a_guess, target_sum)

            sub_deadline = time.time() + min(30.0, gd_time_budget / runs)
            a_gd = _gradient_descent_opt(a_guess, sub_deadline)
            score_gd = _max_convolution_fft(a_gd)
            if score_gd < best_score:
                best_score = score_gd
                best_seq = a_gd.copy()

        # Hill climbing
        if time.time() < deadline:
            # compute full convolution of current best (or a_gd)

            hc_deadline = deadline - 0.05

                best_score = max_hc

"""
Hybrid multi‑restart gradient descent + aggressive hill‑climbing.

* For each chosen length we (re)seed from the best known sequence,
  up‑sample it (or generate a fresh random start) and run a few
  independent projected gradient‑descent runs.  The smoothed‑max
  surrogate uses a log‑sum‑exp with a rapidly annealed β, and a
  simple Nesterov‑style momentum update.  Each GD run respects the
  ℓ¹‑simplex Σa = √(2 n).

* After GD we apply a targeted hill‑climber that repeatedly moves a
  small amount of mass from an index that contributes a lot to the
  current peak towards an index that contributes little.  The move
  is accepted if the maximal convolution entry decreases, otherwise
  it is accepted with a Metropolis probability (temperature starts
  at 2 % of the current peak and cools slowly).

* The whole search is time‑bounded (default 1000 s).  Progress and the
  current best objective (which equals the maximal self‑convolution
  entry for properly normalised sequences) are printed.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Duchi‑et‑al. (2008) algorithm.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed maximum (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of   (1/β)·log Σ exp(β·b)   where   b = a*a .
    Returns (g, b) with   g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # b = a*a  (via FFT)
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)  (correlation)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smoothed maximum.
    The schedule is deliberately aggressive (long GD, larger β).
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # β‑annealing
    beta = 0.5
    beta_max = 1.5e5         # much larger than before – sharper surrogate
    beta_factor = 1.4

    # step‑size and momentum
    step_base = target_sum * 0.09
    momentum = np.zeros_like(a)
    rho = 0.93          # momentum decay

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        # projection onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β‑annealing
        if it % 350 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = np.sqrt(2.0 * n)

    idx = np.arange(n, dtype=float)

    family = rng.integers(0, 8)

    if family == 0:                       # pure power‑law
        p = rng.uniform(0.08, 1.3)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                     # exponential decay
        lam = rng.exponential(scale=0.018)
        a = np.exp(-lam * idx)

    elif family == 2:                     # mixed power‑law × exponential
        p = rng.uniform(0.12, 1.0)
        lam = rng.exponential(scale=0.018)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                     # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                     # piecewise exponential
        cut = rng.integers(int(0.25 * n), int(0.75 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(cut)),
             np.exp(-lam2 * np.arange(cut, n))]
        )

    elif family == 5:                     # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    elif family == 6:                     # single‑spike with small tail
        a = np.exp(-0.003 * idx) * (rng.random() * 0.5 + 0.5)

    else:                                 # random Gaussian‑like bump
        centre = rng.uniform(0, 1)
        width = rng.exponential(scale=0.15) + 0.03
        a = np.exp(-((idx / n - centre) ** 2) / (2 * width ** 2))

    # small Gaussian noise and projection
    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Targeted mass‑transfer hill climbing (exact convolution updates)
# ----------------------------------------------------------------------


def _hill_climb_targeted(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Hill‑climbing that repeatedly moves a small amount of mass from a
    source that contributes much to the current peak towards a destination
    that contributes little.  The move is accepted if the peak improves,
    otherwise with a Metropolis probability (temperature cools slowly).
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size

    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    T = cur_max * 0.02 + 1e-9          # start temperature (2 % of peak)
    cooling = 0.9999                  # very slow cooling

    while time.time() < deadline:
        # index of current peak
        k_max = int(np.argmax(b))

        # contribution of each index to that peak:
        #    contrib[i] = a[i] * a[k_max-i]   (zero if out of bounds)
        j_arr = k_max - np.arange(n)
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[mask] * a[j_arr[mask]]

        total = contrib.sum()
        if total > 0.0:
            prob_src = contrib / total               # source prefers large contrib
            i = rng.choice(n, p=prob_src)
        else:
            prob_src = a / a.sum()
            i = rng.choice(n, p=prob_src)

        # destination prefers *small* contribution
        inv = contrib.max() - contrib + 1e-12
        prob_dst = inv / inv.sum()
        j = rng.choice(n, p=prob_dst)
        if i == j:
            continue

        # move up to 90 % of the source mass
        max_delta = a[i] * 0.9
        if max_delta < 1e-16:
            continue
        delta = rng.random() * max_delta

        # ---- analytical update of the convolution --------------------
        b_new = b.copy()
        # linear part (‑2 δ·a shifted by i, +2 δ·a shifted by j)
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a

        # quadratic correction terms
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        # ---- acceptance test -----------------------------------------
        accept = False
        if new_max < cur_max - 1e-14:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            a[i] = max(a[i], 0.0)
            a[j] = max(a[j], 0.0)

            b = b_new
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_targeted improvement: {best_max:.6f}")

        T *= cooling

    return best_a, best_max

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimizes

    budget (default 1000 s) and returns the best sequence found.
    """

    # ------------------------------------------------------------------

        n0 = a0.size

        best_score = _max_convolution_fft(a0)
        best_seq = a0.copy()
        print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) "
              f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Length schedule – a handful of representative sizes.
    # ------------------------------------------------------------------
    lengths = [2000, 5000, 8000, 12000, 18000]

    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Initialise a candidate for this length
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:

            sub_dead = time.time() + min(30.0, (phaseA_dead - time.time()) / max(1, runs - run))
            a_gd = _gradient_descent_opt(a_guess, sub_dead)

            score_gd = _max_convolution_fft(a_gd)
            if score_gd < best_score:
                best_score = score_gd
                best_seq = a_gd.copy()
                print(f"[{time.time():.1f}] GD improvement (n={n}) score={score_gd:.6f}")

        # --------------------------------------------------------------

        # --------------------------------------------------------------
        if time.time() < deadline:
            # use the best currently known vector of this length

            else:
                a_cur = best_seq.copy()

            b_full = _full_convolution(a_cur)
            time_left = deadline - time.time()
            phaseB_dead = time.time() + min(40.0, 0.30 * time_left)

            a_hc, max_hc = _hill_climb_targeted(a_cur, b_full, phaseB_dead)
            if max_hc < best_score:
                best_score = max_hc
                best_seq = a_hc.copy()
                print(f"[{time.time():.1f}] Hill‑climb improvement (n={n}) "
                      f"score={max_hc:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with any remaining budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:

        b_final = _full_convolution(a_final)

        if max_final < best_score:

    # ------------------------------------------------------------------

    else:
        best_seq = best_seq[:1]

    # Log the official evaluation value
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:   # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
