"""
Proposes a step‑function vector that minimizes the maximum entry of its
self‑convolution.  The algorithm combines three ideas:

1. Exponential‑gradient descent on a soft‑max approximation of the
   infinity‑norm (high‑temperature → low‑temperature schedule).
2. Projected sub‑gradient descent that directly reduces the worst
   convolution entry.
3. A lightweight steady‑state genetic algorithm for final refinement.

The optimisation proceeds over increasing problem sizes (1000 → 2000 →
4000 → 6000) using up‑sampling between stages and many independent
restarts.  All iterates are kept on the L¹‑simplex Σaᵢ = √(2·n) so that
the evaluation function reduces to ``2·n·max(conv)/sum²``.
"""
import time
import numpy as np

def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a * a (length 2·n‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max of the convolution.
    Implements   g_j = 2 Σ_i w_{i+j} a_i .
    """
    n = a.shape[0]
    # reverse w because convolution with reversed weights
    w_rev = w[::-1]
    # length of full convolution of a (len n) with w_rev (len 2n‑1) is 3n‑2
    conv_len = n + (2 * n - 1) - 1   # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)   # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Stable soft‑max weighting: w_i = exp(alpha * conv_i) / Σ exp(alpha * conv_i)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a to length new_n."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=float)
    return np.interp(x_new, x_old, a)

                          alpha_factor: float = 1.05,
                          alpha_max: float = 3000.0,
                          eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Exponential‑gradient descent on the soft‑max approximation of the max‑norm.
    Keeps the iterate on the L1‑simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    alpha = alpha_start
    eta = eta0
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        # soft‑max weights
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # exponential update – shift by max to avoid overflow
        maxg = np.max(grad)
        factor = np.exp(-eta * (grad - maxg))
        a = a * factor
        # re‑normalize onto the simplex
        a = _project_simplex(a, S)

        # occasional random mass swap for exploration
        if rng.random() < 0.04:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.025
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # schedule updates
        alpha = min(alpha * alpha_factor, alpha_max)
        if it % 1500 == 0:
            eta *= 0.85   # slowly reduce step‑size

    return best_a, best_score


def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent that directly reduces the worst convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of (a*a)_k with respect to a
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional random perturbation
        if rng.random() < 0.06:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.03
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = np.max(_full_convolution(cand))
                if cand_score < cur_score or rng.random() < 0.2:
                    a = cand
                    cur_score = cand_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()

        # cheap early‑stop test (no actual break, just progress)
        if it % 8000 == 0 and cur_score >= best_score * (1 + 1e-9):
            pass

    return best_a, best_score


def _optimize_combined_exponential(a0: np.ndarray,
                                   deadline: float,
                                   rng: np.random.Generator,
                                   fraction_exp: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Run exponential‑gradient for a fraction of the time, then finish with sub‑gradient.
    """
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    exp_deadline = now + fraction_exp * remaining
    a_exp, sc_exp = _optimize_exponential(a0, exp_deadline, rng)

    if time.time() < deadline:
        a_sub, sc_sub = _optimize_subgrad(a_exp, deadline, rng)
        if sc_sub < sc_exp:
            return a_sub, sc_sub
    return a_exp, sc_exp


def _run_genetic_optimizer(initial_a: np.ndarray,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple steady‑state genetic algorithm on the simplex.
    """
    n = initial_a.shape[0]
    S = np.sqrt(2.0 * n)

    pop_size = 20 if n <= 2000 else 12
    pop = []
    scores = []
    for i in range(pop_size):
        if i == 0:
            a = initial_a.copy()
        else:
            a = rng.random(n)
        a = _project_simplex(a, S)
        pop.append(a)
        scores.append(np.max(_full_convolution(a)))
    pop = np.array(pop)
    scores = np.array(scores)

    best_idx = int(np.argmin(scores))
    best_a = pop[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(pop_size):
            idxs = list(range(pop_size))
            idxs.remove(i)
            p1, p2 = rng.choice(idxs, size=2, replace=False)

            alpha = rng.random()
            child = alpha * pop[p1] + (1.0 - alpha) * pop[p2]

            if rng.random() < 0.35:
                noise = rng.normal(scale=child.mean() * 0.05 + 1e-12, size=n)
                child += noise

            child = np.maximum(child, 0.0)
            child = _project_simplex(child, S)

            child_score = np.max(_full_convolution(child))

            if child_score < scores[i]:
                pop[i] = child
                scores[i] = child_score
                if child_score < best_score:
                    best_score = child_score
                    best_a = child.copy()

        if rng.random() < 0.12:
            worst_idx = int(np.argmax(scores))
            a_new = rng.random(n)
            a_new = _project_simplex(a_new, S)
            new_score = np.max(_full_convolution(a_new))
            pop[worst_idx] = a_new
            scores[worst_idx] = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

        if time.time() + 0.1 >= deadline:
            break

    return best_a, best_score

"""
Proposes a non‑negative step‑function (vector) that minimises the
maximum entry of its self‑convolution, i.e. the value returned by
``evaluate_sequence``.  The method combines three complementary
techniques:

1. **Exponential‑gradient descent** on a soft‑max approximation of the
   infinity‑norm.  The update ``a ← a·exp(-η·grad)`` stays in the
   non‑negative orthant and is followed by a Euclidean projection onto
   the simplex Σaᵢ = √(2·n).

2. **Projected sub‑gradient descent** that directly reduces the worst
   convolution entry (the true max‑norm).

3. **Steady‑state genetic algorithm** for a short polishing phase.

The optimisation is performed on a sequence of increasing problem sizes
(1000 → 2000 → 4000 → 6000).  For each size several independent restarts
are executed, the best vector is up‑sampled to the next size and the
process repeats.  All iterates honour the L¹‑simplex constraint, which
means the objective value returned by ``evaluate_sequence`` is exactly
the maximum convolution entry.

The routine respects the 1000 s budget (a 5 s safety margin) and prints
progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient term for the smooth max‑norm:
        g_j = 2 Σ_i w_{i+j} a_i .
    Both ``a`` (len n) and ``w`` (len 2·n‑1) are real vectors.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights (smooth approximation of max)
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """w_i = exp(alpha·conv_i) / Σ_j exp(alpha·conv_j)  (stable)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Upsampling helper (linear interpolation)
# ----------------------------------------------------------------------


def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a to length new_n."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=float)
    return np.interp(x_new, x_old, a)


# ----------------------------------------------------------------------
# Exponential‑gradient optimiser (soft‑max)
# ----------------------------------------------------------------------


def _optimize_exponential(a0: np.ndarray,
                          deadline: float,
                          rng: np.random.Generator,
                          alpha_start: float = 0.5,
                          alpha_factor: float = 1.05,
                          alpha_max: float = 3000.0,
                          eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Exponential‑gradient descent on a smooth approximation of the max‑norm.
    Keeps the iterate on the L¹‑simplex Σa = √(2·n).
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    alpha = alpha_start
    eta = eta0
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # exponential update – shift by max to keep numbers ≤1
        maxg = np.max(grad)
        factor = np.exp(-eta * (grad - maxg))
        a = a * factor
        a = _project_simplex(a, S)

        # occasional random mass swap for exploration
        if rng.random() < 0.04:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.025
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # schedules
        alpha = min(alpha * alpha_factor, alpha_max)
        if it % 1500 == 0:
            eta *= 0.85

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient optimiser (true max‑norm)
# ----------------------------------------------------------------------


def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent that directly reduces the worst entry
    of the self‑convolution.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of (a*a)_kmax
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional random perturbation
        if rng.random() < 0.06:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.03
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = np.max(_full_convolution(cand))
                if cand_score < cur_score or rng.random() < 0.2:
                    a = cand
                    cur_score = cand_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()

        # cheap early‑stop test (no early break, just a placeholder)
        if it % 8000 == 0 and cur_score >= best_score * (1 + 1e-9):
            pass

    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimiser (exponential‑gradient → sub‑gradient)
# ----------------------------------------------------------------------


def _optimize_combined_exponential(a0: np.ndarray,
                                   deadline: float,
                                   rng: np.random.Generator,
                                   fraction_exp: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Run the exponential‑gradient phase for a fraction of the allocated
    time, then finish with the projected sub‑gradient method.
    """
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    exp_deadline = now + fraction_exp * remaining
    a_exp, sc_exp = _optimize_exponential(a0, exp_deadline, rng)

    if time.time() < deadline:
        a_sub, sc_sub = _optimize_subgrad(a_exp, deadline, rng)
        if sc_sub < sc_exp:
            return a_sub, sc_sub
    return a_exp, sc_exp


# ----------------------------------------------------------------------
# Light‑weight steady‑state genetic algorithm
# ----------------------------------------------------------------------


def _run_genetic_optimizer(initial_a: np.ndarray,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple GA that works on the simplex.  Returns the best vector found
    before ``deadline``.
    """
    n = initial_a.shape[0]
    S = np.sqrt(2.0 * n)

    pop_size = 20 if n <= 2000 else 12
    pop = []
    scores = []
    for i in range(pop_size):
        a = initial_a.copy() if i == 0 else rng.random(n)
        a = _project_simplex(a, S)
        pop.append(a)
        scores.append(np.max(_full_convolution(a)))
    pop = np.array(pop)
    scores = np.array(scores)

    best_idx = int(np.argmin(scores))
    best_a = pop[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(pop_size):
            # select two distinct parents (different from target)
            idxs = list(range(pop_size))
            idxs.remove(i)
            p1, p2 = rng.choice(idxs, size=2, replace=False)

            # convex recombination
            alpha = rng.random()
            child = alpha * pop[p1] + (1.0 - alpha) * pop[p2]

            # mutation
            if rng.random() < 0.35:
                noise = rng.normal(scale=child.mean() * 0.05 + 1e-12, size=n)
                child += noise

            child = np.maximum(child, 0.0)
            child = _project_simplex(child, S)

            child_score = np.max(_full_convolution(child))

            if child_score < scores[i]:
                pop[i] = child
                scores[i] = child_score
                if child_score < best_score:
                    best_score = child_score
                    best_a = child.copy()

        # inject diversity occasionally
        if rng.random() < 0.12:
            worst_idx = int(np.argmax(scores))
            a_new = rng.random(n)
            a_new = _project_simplex(a_new, S)
            new_score = np.max(_full_convolution(a_new))
            pop[worst_idx] = a_new
            scores[worst_idx] = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

        if time.time() + 0.1 >= deadline:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function (non‑negative vector) that minimises
    ``evaluate_sequence``.  The routine mixes exponential‑gradient,
    sub‑gradient descent and a short GA, iterating over progressively
    larger problem sizes.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin for the external call
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered construction if available
    # ------------------------------------------------------------------
    best_seq = None
    try:
        if 'height_sequence_1' in globals():
            best_seq = np.asarray(height_sequence_1, dtype=float)
    except Exception:
        best_seq = None

    best_score = np.inf
    total_budget = deadline - start_time

    # ------------------------------------------------------------------
    # Progressive optimisation over increasing sizes
    # ------------------------------------------------------------------
    sizes = [1000, 2000, 4000, 6000]          # problem dimensions
    # fractions of the total budget (must sum < 1)
    fractions = [0.15, 0.25, 0.35, 0.20]

    for stage, n in enumerate(sizes):
        if time.time() >= deadline:
            break

        stage_time = total_budget * fractions[stage]
        stage_deadline = min(time.time() + stage_time, deadline)

        # initialise starting vector for this stage
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        else:
            a0 = rng.random(n)

        a0 += rng.normal(scale=a0.mean() * 1e-4 + 1e-12, size=n)
        a0 = np.clip(a0, 0.0, None)
        S = np.sqrt(2.0 * n)
        if a0.sum() == 0.0:
            a0 = np.full(n, 1.0)
        a0 = a0 / a0.sum() * S

        # number of independent restarts for this size
        restarts = 5
        for r in range(restarts):
            if time.time() >= stage_deadline:
                break
            # split remaining stage time equally among remaining restarts
            remain = stage_deadline - time.time()
            sub_deadline = time.time() + remain / (restarts - r)

            # 1️⃣  Combined exponential + sub‑gradient optimisation
            a_opt, sc_opt = _optimize_combined_exponential(
                a0, sub_deadline, rng, fraction_exp=0.65)

            # 2️⃣  Quick GA polishing if time permits
            remain = sub_deadline - time.time()
            if remain > 2.0:
                ga_deadline = time.time() + min(5.0, remain)
                a_ga, sc_ga = _run_genetic_optimizer(a_opt, ga_deadline, rng)
                if sc_ga < sc_opt:
                    a_opt, sc_opt = a_ga, sc_ga

            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"[stage] n={n}, restart={r} new best score={best_score:.6f}")

        # shrink remaining total budget for later stages
        total_budget = deadline - time.time()

    # ------------------------------------------------------------------
    # Final sanity check & external evaluation
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist() if best_seq is not None else [1.0]
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} "
          f"(internal best = {best_score:.6f}) length = {len(final_seq)}")
    return final_seq
