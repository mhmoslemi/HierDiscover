"""
Proposes a non‑negative step‑function (height sequence) that minimizes the
benchmark defined by `evaluate_sequence`.

Algorithm
---------
1.  Work on a few increasing lengths (≈1500 → 3000 → 6000).  
    The best solution of a smaller scale is linearly interpolated and used
    as a warm start for the next scale.

2.  For a fixed length `n` we optimise the smooth approximation

        L_β(a) = max(a*a) + (1/β)·log Σ exp(β·(a*a‑max(a*a)))

    of the true objective (`max(a*a)`).  The approximation becomes exact as
    β → ∞.  The gradient of `L_β` is obtained analytically:
        ∂L/∂a = 2·(a_rev ⊛ p)               (⊛ = convolution)
    where `p` is the soft‑max over the convolution entries.
    The optimisation is performed on the *simplex* using the
    re‑parameterisation

        a_i = S·exp(y_i) / Σ_j exp(y_j)   with   S = √(2·n)

    which guarantees non‑negativity and the ℓ¹‑constraint.
    An L‑BFGS‑B optimiser (from `scipy.optimize`) updates the unrestricted
    variables `y`.  A schedule of increasing β values (1 → … → 8192)
    continuously sharpens the approximation.

3.  After the L‑BFGS phase a very short simulated‑annealing (SA) polishing
    step (≈10 000 moves) moves mass from “sensitive’’ indices (large
    contribution to the current maximal convolution entry) to low‑sensitivity
    ones, occasionally smoothing short windows and swapping entries.  The SA
    uses the true max‑convolution (no softmax) and accepts worsening moves
    with a temperature that decays exponentially.

4.  The best sequence seen over all scales is returned.  Very small trailing
    values (≤1e‑12) are removed for neatness.

The whole search is time‑bounded (default budget = 1000 s).  The code never
violates the security checks of `evaluate_sequence` – only non‑negative
floats are produced and the ℓ¹‑norm is exactly √(2·n).
"""

# ----------------------------------------------------------------------


def _next_pow_two(k: int) -> int:
    """Smallest power of two ≥ k."""
    return 1 << (k - 1).bit_length()


def _conv_full(a: np.ndarray, size: int) -> np.ndarray:
    """Full linear convolution a*a using FFT of given size."""
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * a.size - 1]
    return conv

    """
    Gradient of the soft‑max loss w.r.t. `a`:
        g = 2 * (a_rev ⊛ p)[n‑1 : n‑1+n]
    """
    a_rev = a[::-1]
    ar = np.fft.rfft(a_rev, n=size)
    pr = np.fft.rfft(p, n=size)
    rev = np.fft.irfft(ar * pr, n=size)[: 2 * a.size - 1]
    g = 2.0 * rev[a.size - 1 : a.size - 1 + a.size]
    return g


class SoftmaxLoss:
    """
    Callable object used by ``scipy.optimize.minimize``.
    Represents the smooth approximation of the max‑convolution objective
    for a fixed β.

    Parameters
    ----------
    n : int
        Length of the sequence.
    S : float
        Target ℓ¹‑norm (√(2·n)).
    beta : float
        Soft‑max temperature (larger ⇒ sharper).
    """

    def __init__(self, n: int, S: float, beta: float):
        self.n = n
        self.S = S
        self.beta = beta
        self.fft_size = _next_pow_two(2 * n - 1)

    def __call__(self, y: np.ndarray):
        # --- reconstruct a from y (simplex parameterisation) ---
        y_shift = y - np.max(y)                     # stability
        exp_y = np.exp(y_shift)
        a = self.S * exp_y / exp_y.sum()            # Σ a_i = S

        # --- convolution ---
        conv = _conv_full(a, self.fft_size)         # length 2n‑1
        max_c = conv.max()

        # --- soft‑max probabilities ---
        shifted = conv - max_c                      # ≤ 0
        exp_shift = np.exp(self.beta * shifted)
        Z = exp_shift.sum()
        p = exp_shift / Z

        # --- loss (smooth max) ---
        loss = max_c + np.log(Z) / self.beta

        # --- gradient w.r.t a ---
        g_a = _grad_wrt_a(conv, a, self.fft_size, p)

        # --- chain rule to obtain gradient w.r.t y ---
        dot = np.dot(a, g_a)        # Σ a_i g_i
        avg = dot / self.S
        grad_y = a * (g_a - avg)

        return loss, grad_y


def _l_bfgs_optimize(initial_a: np.ndarray,
                     n: int,
                     deadline: float,
                     betas: list[float],
                     iters_per_beta: int = 200) -> np.ndarray:
    """
    Runs a staged L‑BFGS optimisation on the smooth loss.
    Returns the final sequence `a` (already projected onto the ℓ¹‑simplex).
    """
    S = np.sqrt(2.0 * n)

    # initialise y from the current a (log‑map, avoids division by zero)
    a = np.maximum(initial_a, 1e-30)
    y = np.log(a)

    for beta in betas:
        if time.time() >= deadline:
            break
        loss_obj = SoftmaxLoss(n, S, beta)
        # L‑BFGS‑B with a modest iteration budget
        res = minimize(loss_obj,
                       y,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': iters_per_beta,
                                'disp': False})
        y = res.x

    # final reconstruction
    y_shift = y - np.max(y)
    exp_y = np.exp(y_shift)
    a_opt = S * exp_y / exp_y.sum()
    return a_opt


def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_temp: float = 1.0,
                         end_temp: float = 1e-4,
                         max_frac_start: float = 0.30,
                         max_frac_end: float = 0.01,
                         max_window: int = 50) -> np.ndarray:
    """
    Very short SA polishing (≈10 k iterations).  Works directly on the true
    objective ``max(a*a)``.  Returns the best sequence found.
    """
    a = a0.copy()
    n = a.size
    S = np.sqrt(2.0 * n)
    size = _next_pow_two(2 * n - 1)
    rng = np.random.default_rng()
    best = a.copy()
    best_score = _max_convolution(a, size)  # cached below

    start = time.time()
    total = max(deadline - start, 1e-9)

    while time.time() < deadline:
        # temperature schedule
        t = (time.time() - start) / total
        temp = start_temp * (end_temp / start_temp) ** t
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** t

        # current convolution & worst index
        conv = _conv_full(a, size)
        k_max = int(np.argmax(conv))

        # sensitivity vector (derivative of the current max entry)
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]

        # ----- mass transfer move (dominant) -----
        weight = a * sens
        if weight.sum() > 0 and rng.random() < 0.70:
            donor = rng.choice(n, p=weight / weight.sum())
            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            if low_idx.size:
                receiver = rng.choice(low_idx)
            else:
                receiver = rng.integers(n)
            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                a[donor] -= delta
                a[receiver] += delta
        else:
            # occasional smoothing or swap
            if rng.random() < 0.80:
                start_i = rng.integers(0, n)
                length = rng.integers(2, max_window + 1)
                end_i = min(start_i + length, n)
                seg = a[start_i:end_i]
                avg = seg.mean()
                a[start_i:end_i] = avg
            else:
                i, j = rng.choice(n, size=2, replace=False)
                a[i], a[j] = a[j], a[i]

        # re‑normalise to keep Σa = S (tiny drift can accumulate)
        cur_sum = a.sum()
        if cur_sum > 0:
            a *= S / cur_sum

        new_score = _max_convolution(a, size)

        # Metropolis acceptance
        if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / max(temp, 1e-12)):
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
        else:
            # reject – revert to the best known (cheap because moves are small)
            a = best.copy()

    return best


def _max_convolution(a: np.ndarray, size: int) -> float:
    """Maximum entry of a*a (computed with FFT)."""
    conv = _conv_full(a, size)
    return float(conv.max())


def _resize_linear(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation from old_len to new_len."""
    old_len = seq.shape[0]
    if old_len == new_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _power_law(n: int, p: float, S: float) -> np.ndarray:
    """Create a (i+1)^{-p} sequence and normalise to ℓ¹‑norm S."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= S / a.sum()
    return a

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a height sequence with low benchmark score.
    The routine works on three increasing lengths (≈1500 → 3000 → 6000)
    re‑using the best solution of the previous scale as a warm start.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # 1 s safety margin

    best_seq: np.ndarray | None = None
    best_score = float("inf")
    cur_seq: np.ndarray | None = None

    # (length, fraction of remaining time)
    scales = [(1500, 0.20), (3000, 0.30), (6000, 0.50)]

    # β schedule for the smooth loss (gradually sharpen)
    beta_schedule = [1.0, 5.0, 20.0, 80.0, 320.0, 1280.0, 5120.0, 8192.0]

    for n, frac in scales:
        if time.time() >= deadline - 5.0:
            break

        now = time.time()
        remaining = deadline - now
        stage_deadline = now + remaining * frac

        # ---- initialise candidate for this scale ----
        if cur_seq is not None:
            init = _resize_linear(cur_seq, n)
        elif "height_sequence_1" in globals():
            init = _resize_linear(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
        else:
            init = _power_law(n, 0.55, np.sqrt(2.0 * n))

        # ---- L‑BFGS optimisation on the smooth loss ----
        a_opt = _l_bfgs_optimize(init,
                                 n,
                                 stage_deadline,
                                 beta_schedule,
                                 iters_per_beta=250)

        # ---- short simulated‑annealing polish (if time permits) ----
        if time.time() < stage_deadline:
            a_opt = _simulated_annealing(a_opt, stage_deadline)

        # ---- evaluate (true objective) ----
        try:
            score = evaluate_sequence(a_opt.tolist())
        except Exception:                                 # safety net
            score = float("inf")

        print(f"[{time.time():.2f}] Scale n={n}  score={score:.6f}")

        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] *** New global best: {best_score:.6f}")

        cur_seq = a_opt                                   # warm start for next scale

    # ----- post‑processing: trim trailing near‑zeros -----
    if best_seq is None:
        return []

    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final = arr.tolist()

    # final verification (optional, never raises)
    try:
        val = evaluate_sequence(final)          # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final
