def lp_optimize(f, S, max_iter, t=0.2, time_deadline):
    best_f = f.copy()
    best_M = max(conv_self(f))
    while iter < max_iter and time.time() < deadline:
        # Build A matrix using toeplitz
        A = toeplitz(np.concatenate([f, np.zeros(len(f)-1)]), np.concatenate([np.array([f[0]]), np.zeros(len(f)-1)]))
        ub = np.full(A.shape[0], best_M)  # constraints conv <= best_M
        res = linprog(-np.ones(len(f)), A_ub=A, b_ub=ub, bounds=[(0, None)]*len(f), method='highs')
        if res.status != 0:
            # LP didn't solve, break
            break
        g0 = res.x
        sum0 = g0.sum()
        if sum0 == 0:
            break
        scale = S / sum0
        # ensure scale <=1: if >1, set scale=1 (no scaling)
        if scale > 1:
            scale = 1.0

        # try steps

def improve_by_lp(f, S, t, max_iter, deadline):
    best_f = f.copy()
    best_M = M
    n = len(f)
    iter = 0

def _scale_sequence(seq, new_n):
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    old_positions = np.linspace(0, 1, old_n)
    new_positions = np.linspace(0, 1, new_n)

def _lp_step(f, S, M, t):
    n = len(f)
    # Build A matrix
    # Build using toeplitz
    c = np.concatenate([f, np.zeros(n-1, dtype=f.dtype)])
    r = np.concatenate([np.array([f[0]], dtype=f.dtype), np.zeros(n-1, dtype=f.dtype)])

"""
Hybrid LP‑Adam optimizer for the step‑function self‑convolution problem.

The algorithm repeatedly:
    1. Samples a random length n (moderate size so that a dense LP is
       still affordable) and creates a feasible start vector a0 with
       sum √(2 n) (edge‑heavy, Dirichlet or uniform).
    2. Performs a few linear‑programming updates that maximise the ℓ₁‑norm
       of a vector g under the constraint (a * g) ≤ max(a * a).  The
       update a ← (1‑t)a + t g reduces the convolution peak while
       preserving the ‖a‖₁‑norm.
    3. Refines the result with a projected‑Adam optimiser that works on
       the simplex via a log‑softmax parametrisation and a τ‑soft‑max
       approximation of the max‑convolution.
The best sequence found during the run is stored in the global variable
``height_sequence_1`` and returned as a plain list.  The external
``evaluate_sequence`` routine (already imported in the harness) is used
only for a final sanity check.
"""

import time
import numpy as np
from scipy.linalg import toeplitz
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _softmax_gradient(a: np.ndarray, tau: float):
    """Gradient of the τ‑soft‑max approximation of max(b) where b = a*a."""
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        w = exps / np.sum(exps)
        mask = w

    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[len(a) - 1:len(a) - 1 + len(a)]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Minimise the τ‑soft‑max approximation of the convolution peak.
    Returns the best vector encountered and the corresponding hard peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    # initialise logits so that softmax(p) * S == a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)

    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.3
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 1.0)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # recover a from logits (log‑softmax)
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # convolution & gradient (soft‑max)
        g, peak = _softmax_gradient(a, tau)

        # keep best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # gradient w.r.t. logits (chain rule)
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # optional gradient clipping
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # schedules
        tau *= 0.9997
        lr *= 0.9999

        # occasional jitter
        if it % 800 == 0:
            p += rng.normal(scale=0.015, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Parameterised edge‑heavy initial shapes
# ----------------------------------------------------------------------
def _edge_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Build a feasible start vector of length n.
    The shape is (d + offset)^{-p} optionally mixed with a uniform
    background; finally it is scaled to the required ℓ₁‑norm.
    """
    S = np.sqrt(2.0 * n)

    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0

    p = rng.uniform(0.2, 1.0)
    offset = rng.uniform(0.0, 5.0)
    mix = rng.uniform(0.0, 1.0)

    edge_part = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge_part + (1.0 - mix) * base

    raw = np.clip(raw, 0.0, None)
    raw = raw / raw.sum() * S
    return raw


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    old_pos = np.linspace(0, 1, old_n)
    new_pos = np.linspace(0, 1, new_n)
    return np.interp(new_pos, old_pos, seq)


def _initial_vector(base_seq: np.ndarray | None,
                    rng: np.random.Generator,
                    n: int) -> np.ndarray:
    """Create a feasible start vector of length n."""
    S = np.sqrt(2.0 * n)

    # warm‑start from the global best a few percent of the time
    if base_seq is not None and rng.random() < 0.15:
        a = _scale_sequence(base_seq, n)
        a = np.maximum(a, 0.0)
        a = a / a.sum() * S
        return a

    style = rng.choice(['uniform', 'dirichlet', 'edge'])
    if style == 'uniform':
        a = np.full(n, S / n)
    elif style == 'dirichlet':
        a = rng.random(n)
        a = a / a.sum() * S
    else:   # edge‑heavy
        a = _edge_initial(n, rng)  # already normalised to S
    return a


# ----------------------------------------------------------------------
# Linear‑programming improvement step
# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray, S: float, M: float, t: float) -> tuple[np.ndarray, float]:
    """
    One LP‑based update.
    Solve max ∑ g_j  s.t. (f * g)_k ≤ M, g ≥ 0.
    The solution g is scaled to ‖g‖₁ = S (but never increased,
    i.e. scale = min(1, S / ∑g₀)), then a convex combination
    a ← (1‑t) f + t g is tried (with back‑tracking on t) and the
    resulting peak is returned.
    """
    n = len(f)

    # build the (2n‑1)×n convolution matrix with toeplitz
    c = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    r = np.concatenate([np.array([f[0]], dtype=f.dtype), np.zeros(n - 1, dtype=f.dtype)])
    A = toeplitz(c, r)

    # linear program: maximise ∑g  ⇔ minimise -∑g
    res = linprog(-np.ones(n),
                  A_ub=A,
                  b_ub=np.full(2 * n - 1, M, dtype=f.dtype),
                  bounds=[(0, None)] * n,
                  method='highs')

    if res.status != 0:
        # LP failed – return unchanged
        return f, M

    g0 = res.x
    s0 = g0.sum()
    if s0 <= 0:
        return f, M

    scale = S / s0
    if scale > 1.0:          # do not increase – would break the bound
        scale = 1.0
    g = g0 * scale

    # back‑track on the step size
    tt = t
    while tt > 1e-6:
        f_new = (1.0 - tt) * f + tt * g
        M_new = _full_convolution_fft(f_new).max()
        if M_new < M - 1e-12:
            return f_new, M_new
        tt *= 0.5
    return f, M


def _lp_improve(f: np.ndarray,
                S: float,
                M: float,
                t: float,
                steps: int,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeated LP updates (at most *steps* times or until the deadline).
    The step size t is slowly reduced.
    """
    for _ in range(steps):
        if time.time() >= deadline:
            break
        f, M = _lp_step(f, S, M, t)
        t *= 0.85
    return f, M


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Hybrid optimiser that searches for a non‑negative step‑function
    minimising ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # safety margin

    # Try to reuse a previously discovered construction
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_val = np.inf
    attempt = 0

    # ------------------------------------------------------------------
    # Phase 1 – many short restarts with LP + Adam
    # ------------------------------------------------------------------
    short_budget = 12.0                     # seconds per restart
    while time.time() + short_budget < end_time - 30.0:
        attempt += 1
        # moderate length (LP gets expensive for very large n)
        n = rng.integers(1500, 3000)
        S = np.sqrt(2.0 * n)

        a_start = _initial_vector(base_seq, rng, n)
        a_start = np.maximum(a_start, 0.0)
        a_start = a_start / a_start.sum() * S

        # initial convolution peak
        M0 = _full_convolution_fft(a_start).max()

        # LP improvement (a few steps)
        lp_deadline = min(time.time() + 8.0, end_time - 30.0)
        a_lp, M_lp = _lp_improve(a_start, S, M0, t=0.3, steps=5, deadline=lp_deadline)

        # remaining time of the slice goes to Adam
        slice_deadline = min(end_time - 30.0, time.time() + short_budget)
        if time.time() >= slice_deadline:
            break
        a_opt, peak = _adam_optimize(a_lp, slice_deadline, rng)

        if peak < best_val:
            best_val = peak
            best_seq = a_opt.copy()
            print(f"[restart] attempt={attempt} n={n}  new best value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – final long refinement of the best candidate
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = end_time - time.time()
        if remaining > 5.0:
            S = np.sqrt(2.0 * len(best_seq))
            best_seq = np.maximum(best_seq, 0.0)
            best_seq = best_seq / best_seq.sum() * S
            a_ref, peak_ref = _adam_optimize(best_seq, end_time, rng)
            if peak_ref < best_val:
                best_val = peak_ref
                best_seq = a_ref.copy()
                print(f"[refine] final refined value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the black‑box evaluator
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()
        external = evaluate_sequence(final_seq)
        print(f"[final] external score = {external:.6f}  internal max = {best_val:.6f}  length = {len(final_seq)}")
        globals()['height_sequence_1'] = final_seq

    return final_seq
