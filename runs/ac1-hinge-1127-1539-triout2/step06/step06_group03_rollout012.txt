def _targeted_permutation_annealing(a0, deadline):
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)
    best_b = _full_convolution_fft(best)
    # temperature schedule:
    T = best_score * 0.1 + 1e-9  # scaled
    cooling = 0.9997
    rng = np.random.default_rng()
    while time.time() < deadline:
        # compute current b for selection
        b = _full_convolution_fft(a)
        k = int(np.argmax(b))
        # choose i from indices contributing to k
        i_low = max(0, k - (n-1))
        i_high = min(n-1, k)
        idxs = np.arange(i_low, i_high+1)
        # contributions
        contrib = a[idxs] * a[k - idxs]
        if contrib.sum() == 0:
            i = rng.integers(i_low, i_high+1)
        else:

        # choose j random not equal i
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)
        a_new = a.copy()
        a_new[i], a_new[j] = a_new[j], a_new[i]

"""
Propose a non‑negative step function that minimises the provided
``evaluate_sequence`` objective.

Algorithm outline
-----------------
1. **Generate candidates** from several parametric families (power‑law,
   exponential, truncated linear) for a few different lengths.  For each
   family we also try several simple orderings (ascending, descending,
   alternating, random).
2. **Keep an elite pool** (≈15 best candidates) measured by the
   convolution L∞‑norm (the true objective is proportional to this).
3. **Refinement** – each elite candidate is improved by
   *projected gradient descent* on a smooth log‑sum‑exp surrogate of the
   maximal convolution entry, followed by a
   *targeted permutation annealing* that swaps entries which contribute
   most to the current maximal bin.  The process is repeated while
   time permits.
4. **Return the best sequence** found before the 1000‑second deadline.

All heavy work is performed with FFT‑based convolutions (≈ O(n log n))
and the code respects the global time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n−1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n−1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex Σx = s, x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection onto {x_i ≥ 0, Σx_i = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) surrogate of max(b) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b) where
        b = a*a (full convolution)
        g = ∇_a ( (1/β)·log Σ_k exp(β·b_k) ) = 2·(a ⋆ p)
    with p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution a*a
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Projected gradient descent on the smoothed objective
# ----------------------------------------------------------------------


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Annealed projected gradient descent on the log‑sum‑exp surrogate.
    The ℓ¹‑norm is kept fixed at the theoretical optimum √(2·n).
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 1.0
    max_beta = 1e8
    momentum = np.zeros_like(a)
    rho = 0.9                     # momentum decay
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum step
        momentum = rho * momentum + (1.0 - rho) * g
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            lr = 0.5 * target_sum / (norm * (beta + 1.0))
            a -= lr * momentum

        # project onto simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 8e-4 / (beta + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal beta
        if it % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.08, max_beta)

        it += 1
    return a


# ----------------------------------------------------------------------
# Targeted permutation annealing (swap moves that aim at the max bin)
# ----------------------------------------------------------------------


def _targeted_permutation_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simulated‑annealing on the ordering of a fixed multiset.
    In each move we identify the current maximal convolution bin, pick a
    contributing index (weighted by a_i·a_{k-i}) and swap it with a random
    partner.  The temperature decays slowly; occasional random shuffles
    help to escape deep local minima.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # temperature loosely linked to the current objective
    T = best_score * 0.07 + 1e-9
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        # current full convolution (needed for move selection)
        b = _full_convolution_fft(a)
        k = int(np.argmax(b))

        # indices that contribute to bin k
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        idxs = np.arange(i_low, i_high + 1)
        contrib = a[idxs] * a[k - idxs]

        if contrib.sum() == 0:
            i = rng.integers(i_low, i_high + 1)
        else:
            probs = contrib / contrib.sum()
            i = rng.choice(idxs, p=probs)

        # random partner
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        a_new = a.copy()
        a_new[i], a_new[j] = a_new[j], a_new[i]

        cur_score = _max_convolution_fft(a_new)
        dE = cur_score - best_score
        accept = False
        if cur_score <= best_score:
            accept = True
        elif T > 0 and rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            a = a_new
            if cur_score < best_score:
                best_score = cur_score
                best = a_new.copy()

        T *= cooling

        # occasional global reshuffle of a tiny fraction
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.015
            if np.any(mask):
                shuffled = a[mask]
                rng.shuffle(shuffled)
                a[mask] = shuffled
                cur_score = _max_convolution_fft(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()

    return best


# ----------------------------------------------------------------------
# Parametric families (power‑law, exponential, truncated linear)
# ----------------------------------------------------------------------


def _normalize_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale an array to have ℓ¹‑sum equal to *target* (preserving non‑negativity)."""
    scale = target / np.sum(v)
    a = v * scale
    a = np.maximum(a, 0.0)
    return a


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _exponential_sequence(n: int, rate: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.exp(-rate * (idx + shift))
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _linear_sequence(n: int, slope: float, intercept: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, intercept - slope * idx)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


# ----------------------------------------------------------------------
# Ordering heuristics
# ----------------------------------------------------------------------


def _candidate_orderings(base: np.ndarray, rng: np.random.Generator) -> list[np.ndarray]:
    """Yield a handful of different permutations of *base*."""
    n = base.size
    asc = np.sort(base)          # smallest → largest
    desc = asc[::-1]             # largest → smallest
    out = [asc, desc]

    # alternating small–large (small first)
    alt = np.empty_like(base)
    lo, hi = 0, n - 1
    for i in range(n):
        if i % 2 == 0:
            alt[i] = asc[lo]
            lo += 1
        else:
            alt[i] = asc[hi]
            hi -= 1
    out.append(alt)

    # a few random shuffles
    for _ in range(5):
        shuffled = asc.copy()
        rng.shuffle(shuffled)
        out.append(shuffled)

    return out

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the allotted time budget
    and returns the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    deadline = time.time() + budget_s - 0.5  # safety margin

    best_seq = None
    best_obj = np.inf
    elite: list[tuple[float, np.ndarray]] = []   # (score, vector)

    # ------------------------------------------------------------------
    # 1. Coarse parametric sweep (collect elite candidates)
    # ------------------------------------------------------------------
    lengths = [5000, 8000, 12000]   # larger n gives a lower theoretical bound
    power_ps = np.linspace(0.44, 0.62, 12)          # finer grid around 0.5
    power_shifts = np.linspace(0.2, 1.5, 10)

    exp_rates = np.linspace(0.0004, 0.006, 12)
    exp_shifts = np.linspace(0.0, 2.0, 6)

    lin_slopes = np.linspace(0.0001, 0.006, 10)
    lin_intercepts = np.linspace(0.5, 3.0, 8)

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # ----- power‑law -----
        for p in power_ps:
            for shift in power_shifts:
                base = _powerlaw_sequence(n, p, shift)
                for cand in _candidate_orderings(base, rng):
                    score = _max_convolution_fft(cand)
                    if score < best_obj:
                        best_obj = score
                        best_seq = cand.copy()
                        print(f"[{time.time():.1f}] New global best (pow) n={n} p={p:.3f} shift={shift:.2f} obj={best_obj:.6f}")
                    if len(elite) < 12 or score < elite[-1][0]:
                        elite.append((score, cand.copy()))
                        elite.sort(key=lambda x: x[0])
                        elite = elite[:12]

        # ----- exponential -----
        for rate in exp_rates:
            for shift in exp_shifts:
                base = _exponential_sequence(n, rate, shift)
                for cand in _candidate_orderings(base, rng):
                    score = _max_convolution_fft(cand)
                    if score < best_obj:
                        best_obj = score
                        best_seq = cand.copy()
                        print(f"[{time.time():.1f}] New global best (exp) n={n} rate={rate:.5f} shift={shift:.2f} obj={best_obj:.6f}")
                    if len(elite) < 12 or score < elite[-1][0]:
                        elite.append((score, cand.copy()))
                        elite.sort(key=lambda x: x[0])
                        elite = elite[:12]

        # ----- linear (truncated) -----
        for slope in lin_slopes:
            for intercept in lin_intercepts:
                base = _linear_sequence(n, slope, intercept)
                for cand in _candidate_orderings(base, rng):
                    score = _max_convolution_fft(cand)
                    if score < best_obj:
                        best_obj = score
                        best_seq = cand.copy()
                        print(f"[{time.time():.1f}] New global best (lin) n={n} slope={slope:.5f} int={intercept:.2f} obj={best_obj:.6f}")
                    if len(elite) < 12 or score < elite[-1][0]:
                        elite.append((score, cand.copy()))
                        elite.sort(key=lambda x: x[0])
                        elite = elite[:12]

        # ----- random samples for diversity -----
        rnd_end = min(deadline, time.time() + 1.2)
        while time.time() < rnd_end:
            rnd = rng.random(n)
            rnd = _normalize_to_sum(rnd, target_sum)
            score = _max_convolution_fft(rnd)
            if score < best_obj:
                best_obj = score
                best_seq = rnd.copy()
                print(f"[{time.time():.1f}] New global best (rnd) n={n} obj={best_obj:.6f}")
            if len(elite) < 12 or score < elite[-1][0]:
                elite.append((score, rnd.copy()))
                elite.sort(key=lambda x: x[0])
                elite = elite[:12]

    # If a previous construction is available, inject it into the elite pool
    try:
        prev = globals()["height_sequence_1"]
        prev_arr = np.array(prev, dtype=float)
        prev_score = _max_convolution_fft(prev_arr)
        elite.append((prev_score, prev_arr.copy()))
        elite.sort(key=lambda x: x[0])
        elite = elite[:12]
        print(f"[{time.time():.1f}] Imported previous construction, obj={prev_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2. Refinement of elite candidates
    # ------------------------------------------------------------------
    for score0, seed_seq in elite:
        if time.time() >= deadline:
            break
        n = seed_seq.size
        # allocate a slice of the remaining budget (but not more than 35 s)
        remaining = deadline - time.time()
        sub_dead = time.time() + min(35.0, remaining * 0.6)

        # gradient descent on the smooth surrogate
        refined = _gradient_descent_opt(seed_seq, sub_dead)

        # permutation annealing targeting the maximal bin
        refined = _targeted_permutation_annealing(refined, deadline)

        # a short second gradient phase to polish the final ordering
        refined = _gradient_descent_opt(refined, deadline)

        obj = _max_convolution_fft(refined)
        if obj < best_obj:
            best_obj = obj
            best_seq = refined.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL BEST (refined) n={n} obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 3. Final sanity check & conversion
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform distribution (should never happen)
        n = 8000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)

    # Trim trailing near‑zero entries (purely cosmetic)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Verify with the official evaluator
    try:
        eval_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {eval_val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective ≈ {best_obj:.6f}, length = {best_seq.size}")
    return best_seq.tolist()
