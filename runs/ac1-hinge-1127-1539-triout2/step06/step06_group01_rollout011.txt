def hill_climb(a, b, deadline):
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()
    best_b = b.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.05 + 1e-9  # temperature
    cooling = 0.99995

    while time.time() < deadline:
        # find index of max, maybe use tie-breaking random among maxes.
        max_val = cur_score
        max_idxs = np.flatnonzero(b == max_val)
        # random choice to avoid deterministic loops
        k = rng.choice(max_idxs)

        # contributions
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        i_range = np.arange(i_low, i_high+1, dtype=int)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]
        # optionally weight
        # pick src with max contribution
        src_idx = i_range[np.argmax(contrib)]

        if a[src_idx] < 1e-12:
            continue

        # pick dest minimizing counterpart
        offsets = k - np.arange(n)
        valid = (offsets >= 0) & (offsets < n)
        counterpart = np.zeros(n)
        counterpart[valid] = a[offsets[valid]]
        # add small random noise to break ties, also penalize large a[dest]
        # compute score for dest: basis = counterpart + 0.5 * a (or just counterpart)
        # Choose dest with smallest composite metric
        # For simplicity: choose dest = np.argmin(counterpart)
        dest_candidates = np.where(counterpart == counterpart.min())[0]
        dest_idx = rng.choice(dest_candidates)
        # ensure dest != src
        if dest_idx == src_idx:
            # choose another dest, maybe random among low.
            other = np.where(np.arange(n) != src_idx)[0]
            if other.shape[0]==0: continue
            dest_idx = rng.choice(other)

        # amount to move
        max_delta = a[src_idx] * 0.9
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # compute new convolution via temporary copy
        # update b in place
        b_new = b.copy()
        # linear updates (use original a)
        # use slicing for i and j
        b_new[src_idx:src_idx+n] -= 2.0 * delta * a
        b_new[dest_idx:dest_idx+n] += 2.0 * delta * a
        # quadratic corrections
        kk = src_idx + src_idx
        if kk < b_new.size:
            b_new[kk] += delta * delta
        kk = dest_idx + dest_idx
        if kk < b_new.size:
            b_new[kk] += delta * delta
        kk = src_idx + dest_idx
        if kk < b_new.size:
            b_new[kk] -= 2.0 * delta * delta

        new_score = b_new.max()
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                accept = True
        if accept:
            # apply updates to a and b
            a[src_idx] -= delta
            a[dest_idx] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                best_b = b.copy()
                print(f"[{time.time():.1f}] hill_climb improvement: {best_score:.6f}")
        T *= cooling
        # optional early stop if temperature small
        if T < 1e-12:
            T = 1e-12

    return best_a, best_b

"""
Propose a candidate step‑function that minimises ``evaluate_sequence``.

The search consists of three stages:

1.  Initialise a distribution on the ℓ¹‑simplex (sum = √(2 n)).  
    If a previous construction ``height_sequence_1`` is available we up‑sample it,
    otherwise we draw a Dirichlet sample.

2.  A fast hill‑climbing routine that repeatedly transfers a random amount of
    mass from a source index (chosen with probability proportional to its
    contribution to the *current* worst convolution entry) to a destination
    index that minimises the contribution to that worst entry.
    The convolution is updated in O(n) time using the linear‑time formula from
    the earlier baseline.  Moves are accepted greedily or with a small
    simulated‑annealing temperature to escape plateaus.

3.  When the wall‑clock budget (default 1000 s) is exhausted the best
    sequence found is trimmed (trailing near‑zeros removed) and returned as a
    plain list.  The official ``evaluate_sequence`` value is printed for
    reference.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(_full_convolution(a).max())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x | x_i ≥ 0, Σ x_i = s}.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a onto a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Hill‑climbing with exact convolution updates
# ----------------------------------------------------------------------


def _hill_climb_improved(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Mass‑transfer hill climbing that explicitly reduces the current worst
    convolution entry.  For each move:

        * pick the index k of the maximal entry of b,
        * pick a source i with probability proportional to a_i·a_{k‑i},
        * pick a destination j that minimises a_{k‑j} (i.e. contributes
          the least to the chosen worst entry),
        * move a random fraction of a_i to a_j and update the full
          convolution in O(n).

    A simple exponential cooling schedule provides occasional uphill
    moves.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()
    best_b = b.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.05 + 1e-9          # initial temperature
    cooling = 0.99995                    # slow cooling

    it = 0
    while time.time() < deadline:
        # ----- current worst index ------------------------------------------------
        max_val = cur_score
        # pick a random occurrence when there are ties
        max_idxs = np.flatnonzero(b == max_val)
        k = rng.choice(max_idxs)

        # ----- source index – weighted by its contribution to entry k -------------
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        i_range = np.arange(i_low, i_high + 1, dtype=int)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]

        # pick the index with the largest contribution (deterministic)
        src_idx = i_range[np.argmax(contrib)]

        # skip if source essentially empty
        if a[src_idx] < 1e-12:
            it += 1
            continue

        # ----- destination index – minimise contribution to the same entry k -----
        # counterpart[t] = a_{k‑t} if that index exists, else 0
        counterpart = np.zeros(n, dtype=float)
        low_j = max(0, k - (n - 1))
        high_j = min(n - 1, k)
        # for j in [low_j, high_j] we have counterpart = a[k‑j]
        if low_j <= high_j:
            js = np.arange(low_j, high_j + 1, dtype=int)
            counterpart[low_j:high_j + 1] = a[k - js]

        # break ties randomly
        min_val = counterpart.min()
        dest_candidates = np.flatnonzero(counterpart == min_val)
        dest_idx = rng.choice(dest_candidates)

        # avoid a no‑op
        if dest_idx == src_idx:
            # pick a random index different from src
            other = np.flatnonzero(np.arange(n) != src_idx)
            if other.size == 0:
                continue
            dest_idx = rng.choice(other)

        # ----- amount of mass to move ------------------------------------------------
        max_delta = a[src_idx] * 0.9                 # keep a few atoms at the source
        delta = rng.random() * max_delta
        if delta <= 0.0:
            it += 1
            continue

        # ----- update the convolution (exact, O(n)) ---------------------------------
        b_new = b.copy()
        # linear terms (remember a is the *old* vector)
        b_new[src_idx:src_idx + n] -= 2.0 * delta * a
        b_new[dest_idx:dest_idx + n] += 2.0 * delta * a
        # quadratic corrections
        k2src = src_idx + src_idx
        if k2src < b_new.size:
            b_new[k2src] += delta * delta
        k2dest = dest_idx + dest_idx
        if k2dest < b_new.size:
            b_new[k2dest] += delta * delta
        kij = src_idx + dest_idx
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # ----- acceptance -----------------------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # Metropolis step
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            # apply the move
            a[src_idx] -= delta
            a[dest_idx] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                best_b = b.copy()
                print(f"[{time.time():.1f}] hill_climb improvement: {best_score:.6f}")

        # temperature schedule
        T *= cooling
        it += 1

    return best_a, best_b


# ----------------------------------------------------------------------
# Random initialiser
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Draw a point uniformly at random from the simplex Σa = √(2 n)."""
    target = np.sqrt(2.0 * n)
    a = rng.dirichlet(alpha=np.ones(n)) * target
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the wall‑clock budget (default 1000 s) and returns
    the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Choose a length schedule (increase resolution gradually)
    # ------------------------------------------------------------------
    length_schedule = [5000, 10000, 20000, 30000]

    for n in length_schedule:
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)

        # ------------------------------------------------------------------
        # Initialise a candidate for this length
        # ------------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        elif "height_sequence_1" in globals():
            # use the stored construction if it has the right size; otherwise upsample
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            if prev.size == n:
                a_init = prev.copy()
            else:
                a_init = _upsample_sequence(prev, n)
        else:
            a_init = _random_initial_sequence(n, rng)

        a_init = _project_to_simplex(a_init, target_sum)

        # ------------------------------------------------------------------
        # Compute the convolution once (used by the hill‑climber)
        # ------------------------------------------------------------------
        b_init = _full_convolution(a_init)

        # ------------------------------------------------------------------
        # Allocate a chunk of the remaining budget to hill climbing
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        # give the current length at most 70 % of the remaining time
        hill_deadline = time.time() + remaining * 0.70

        a_opt, b_opt = _hill_climb_improved(a_init, b_init, hill_deadline)
        cur_score = b_opt.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] Length {n} improvement: {cur_score:.6f}")

        # small warm‑up: a few quick random restarts if time allows
        # (helps escape pathological basins)
        while time.time() < deadline - 5.0:
            # allocate a short burst (≈10 s) for a fresh random start
            burst_deadline = time.time() + 10.0
            a_rand = _random_initial_sequence(n, rng)
            a_rand = _project_to_simplex(a_rand, target_sum)
            b_rand = _full_convolution(a_rand)
            a_local, b_local = _hill_climb_improved(a_rand, b_rand, burst_deadline)
            score_local = b_local.max()
            if score_local < best_score:
                best_score = score_local
                best_seq = a_local.copy()
                print(f"[{time.time():.1f}] Random restart improvement (n={n}): {score_local:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with whatever budget is left
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        n = best_seq.size
        target_sum = np.sqrt(2.0 * n)
        a = _project_to_simplex(best_seq.copy(), target_sum)
        b = _full_convolution(a)
        a_final, b_final = _hill_climb_improved(a, b, deadline)
        final_score = b_final.max()
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final polishing improvement: {final_score:.6f}")

    # ------------------------------------------------------------------
    # Trim insignificant trailing zeros (purely cosmetic)
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback: uniform sequence of modest length
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log the official evaluation (objective = max of self‑convolution)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
