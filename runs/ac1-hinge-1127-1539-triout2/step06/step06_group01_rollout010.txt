"""
Propose a step‑function (non‑negative sequence) that minimises the
``evaluate_sequence`` objective (equivalently the maximum entry of the
self‑convolution, the scale is irrelevant).  The algorithm is a hybrid
of three ideas:

1.  **Linear‑programming direction** – for moderate lengths (≤ 2000) we
    solve the LP

        maximise   Σ g_i
        subject to g_i ≥ 0,
                   (f * g)_k ≤ M  for all k,

    where ``M`` is the current worst self‑convolution value of ``f``.
    The LP yields a direction that moves mass from “heavy’’ to “light’’
    parts without increasing any convolution entry.  We then take the
    best convex combination ``(1‑t) f + t g`` (searching a few values of
    ``t``) which strictly reduces the maximum.

2.  **Projected gradient descent on a smooth ∞‑norm surrogate** – a
    log‑sum‑exp approximation of the maximum together with momentum,
    temperature annealing and occasional jitter.  This efficiently
    smooths the landscape and finds good local minima.

3.  **Exact mass‑transfer hill‑climbing** – a simulated‑annealing style
    move that transfers a random amount of mass between two indices
    while updating the full convolution in O(n).  It refines the
    solution after the previous phases.

The search proceeds on an increasing schedule of lengths
[2000, 5000, 10000, 20000, 30000]; for the first length we exploit the
LP step, for the larger lengths we rely on gradient descent and hill
climbing.  All vectors are projected onto the ℓ¹‑simplex
{a_i ≥ 0, Σ a_i = √(2 n)} after every stochastic move, which makes the
objective scale‑invariant and guarantees that ``evaluate_sequence`` sees
a valid step function.

The routine respects the wall‑clock budget (default 1000 s) and
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _full_convolution_pair(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two (possibly different) vectors."""
    n = a.size
    m = b.size
    size = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: n + m - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed ∞‑norm (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # b = a*a  (full convolution)
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities over b
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Projected gradient descent on the smooth max."""
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 8192.0
    beta_factor = 1.4

    step_base = target_sum * 0.15
    momentum = np.zeros_like(a)
    rho = 0.90

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β annealing
        if it % 350 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative vector on the ℓ¹‑simplex Σa = √(2 n).
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                             np.exp(-lam2 * np.arange(b, n))])

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length, followed by a
    projection onto the appropriate simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Linear‑programming direction (LP) for moderate lengths
# ----------------------------------------------------------------------


def _solve_lp_for_g(f: np.ndarray, M: float):
    """Solve  max Σ g_i  s.t.  g≥0, (f * g)_k ≤ M  for all k."""
    try:
        from scipy.optimize import linprog
    except Exception:         # SciPy not available – fall back to identity
        return f.copy()

    n = f.size
    m = 2 * n - 1
    # Build the Toeplitz matrix A where A[k, j] = f[k-j] if 0≤k-j<n else 0.
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    c = -np.ones(n, dtype=np.float64)          # minimise -Σg  (i.e. maximise Σg)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=np.full(m, M, dtype=np.float64),
                  bounds=bounds, method='highs', options={'disp': False})
    if res.success:
        return res.x
    # In the unlikely event that linprog fails, return the current vector.
    return f.copy()


def _lp_iterate(a: np.ndarray, lp_deadline: float, rng) -> np.ndarray:
    """
    Repeatedly (a) solve the LP for a direction g,
    (b) pick the best convex mixing parameter t∈{0.01,…,0.6}
    that minimises the max entry of the mixed self‑convolution,
    (c) update a ← (1‑t) a + t g.
    Stops when the wall‑clock deadline is reached.
    """
    n = a.size
    target = np.sqrt(2.0 * n)

    while time.time() < lp_deadline:
        # current worst convolution value
        cur_conv = _full_convolution(a)
        M = cur_conv.max()

        # direction from LP
        g = _solve_lp_for_g(a, M)
        g = _project_to_simplex(g, target)

        # pre‑compute all three convolutions needed for mixing
        conv_aa = cur_conv                               # a*a
        conv_ag = _full_convolution_pair(a, g)          # a*g  (= g*a)
        conv_gg = _full_convolution(g)                  # g*g

        # try a short list of candidate mixings
        best_t = 0.0
        best_val = M
        for t in (0.01, 0.03, 0.07, 0.15, 0.30, 0.45, 0.60):
            mixed = ((1.0 - t) ** 2) * conv_aa + \
                    2.0 * t * (1.0 - t) * conv_ag + \
                    (t ** 2) * conv_gg
            val = mixed.max()
            if val < best_val:
                best_val = val
                best_t = t

        if best_t == 0.0:
            # No improvement found – inject a tiny jitter to escape stagnation.
            jitter = rng.normal(scale=target * 5e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target)
        else:
            a = (1.0 - best_t) * a + best_t * g
            a = _project_to_simplex(a, target)

    return a


# ----------------------------------------------------------------------
# Mass‑transfer hill climbing (exact convolution updates)
# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Simulated‑annealing style hill climbing that transfers a random amount
    of mass from a source index to a destination index while keeping the
    full self‑convolution up‑to‑date in O(n) per move.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.1 + 1e-9          # initial temperature
    cooling = 0.9995                    # slower cooling

    while time.time() < deadline:
        # Pick a worst‑convolution index
        max_val = cur_score
        max_idxs = np.flatnonzero(b == max_val)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        # Source index weighted by its contribution to entry M
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # Destination index (biased towards small entries)
        if rng.random() < 0.3:
            inv_a = 1.0 / (a + 1e-12)
            probs_j = inv_a / inv_a.sum()
            j = rng.choice(n, p=probs_j)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        # Amount of mass to move (up to 80 % of the source)
        max_delta = a[i] * 0.8
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # Update the full convolution efficiently (linear time)
        b_new = b.copy()
        # linear terms
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        # quadratic correction terms
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # Metropolis acceptance
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_climb improvement: {best_score:.6f}")

        T *= cooling

    return best_a, b

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the provided wall‑clock budget and returns the
    best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Seed from a previously stored construction if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        earlier = globals()["height_sequence_1"]
        if isinstance(earlier, (list, tuple, np.ndarray)):
            a0 = np.asarray(earlier, dtype=float)
            n0 = a0.size
            a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
            cur = _max_convolution_fft(a0)
            best_seq = a0.copy()
            best_score = cur
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) score={cur:.6f}")

    # ------------------------------------------------------------------
    # Length schedule – we gradually increase the dimension
    # ------------------------------------------------------------------
    schedule = [2000, 5000, 10000, 20000, 30000]

    for n in schedule:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Initialise a candidate for this length
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a = _upsample_sequence(best_seq, n)
        else:
            a = _random_initial_sequence(n, rng)
        a = _project_to_simplex(a, target_sum)

        # --------------------------------------------------------------
        # Phase 0 – LP‑based direction (only for modest lengths)
        # --------------------------------------------------------------
        if n <= 2000 and time.time() < deadline - 5.0:
            lp_budget = min((deadline - time.time()) * 0.25, 120.0)
            lp_deadline = time.time() + lp_budget
            a = _lp_iterate(a, lp_deadline, rng)
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] LP phase improvement for n={n} score={cur_score:.6f}")

        # --------------------------------------------------------------
        # Phase A – multi‑restart projected gradient descent
        # --------------------------------------------------------------
        if time.time() >= deadline - 5.0:
            break
        gd_time = min((deadline - time.time()) * 0.35, 200.0)
        gd_deadline = time.time() + gd_time
        n_restarts = 4
        local_best_a = None
        local_best_score = np.inf

        for r in range(n_restarts):
            if time.time() >= gd_deadline:
                break
            # allocate a slice of the remaining time for this restart
            per_restart = (gd_deadline - time.time()) / (n_restarts - r)
            restart_deadline = time.time() + per_restart
            a_start = a.copy() if r == 0 else _random_initial_sequence(n, rng)
            a_gd = _gradient_descent_opt(a_start, restart_deadline)
            score = _max_convolution_fft(a_gd)
            if score < local_best_score:
                local_best_score = score
                local_best_a = a_gd.copy()

        if local_best_a is not None and local_best_score < best_score:
            best_score = local_best_score
            best_seq = local_best_a.copy()
            print(f"[{time.time():.1f}] GD improvement for n={n} score={local_best_score:.6f}")

        # --------------------------------------------------------------
        # Phase B – hill climbing with exact convolution updates
        # --------------------------------------------------------------
        if time.time() >= deadline - 5.0:
            break
        hc_time = min((deadline - time.time()) * 0.30, 250.0)
        hc_deadline = time.time() + hc_time

        # Ensure we have a vector of the correct length
        if best_seq.size != n:
            a_cur = _upsample_sequence(best_seq, n)
        else:
            a_cur = best_seq.copy()
        a_cur = _project_to_simplex(a_cur, target_sum)
        b_full = _full_convolution(a_cur)

        a_hc, b_hc = _hill_climbing(a_cur, b_full, hc_deadline)
        score_hc = b_hc.max()
        if score_hc < best_score:
            best_score = score_hc
            best_seq = a_hc.copy()
            print(f"[{time.time():.1f}] Hill‑climb improvement for n={n} score={score_hc:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with any remaining budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        target_sum = np.sqrt(2.0 * n)
        a = _project_to_simplex(best_seq.copy(), target_sum)
        b = _full_convolution(a)
        a_final, b_final = _hill_climbing(a, b, deadline)
        final_score = b_final.max()
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final polishing improvement: score={final_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros for neatness
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log the official evaluation (objective = max of the self‑convolution)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
