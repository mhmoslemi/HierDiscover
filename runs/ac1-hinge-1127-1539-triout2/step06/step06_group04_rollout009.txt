"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

The search combines several complementary optimisation phases:

* **Random initialisation** – a variety of simple families (power‑law,
  exponential, Dirichlet, piece‑wise) are sampled on the ℓ¹‑simplex
  ``Σ a = sqrt(2·n)``.
* **Sub‑gradient descent** – a direct descent on the exact piece‑wise
  linear objective ``max(a∗a)`` using the sub‑gradient
  ``2·(a ⋆ 1_{peak})`` (correlation with the peak mask).  This quickly reduces
  large spikes.
* **Smoothed‑max gradient descent** – the same objective is approximated by a
  log‑sum‑exp surrogate; an accelerated projected gradient method with
  Nesterov momentum and β‑annealing refines the solution.
* **Targeted hill‑climbing** – a fast analytical update moves a small amount
  of mass from an index that contributes strongly to the current convolution
  peak to an index that contributes little.  A modest temperature schedule
  enables escaping shallow plateaus.
* **Length schedule & up‑sampling** – the search starts at modest lengths
  (2000) and gradually enlarges the vector (up to ≈30 000).  Each new length
  is seeded by linear interpolation of the best solution found so far.
* **Final polishing** – after the length schedule the remaining budget is
  spent on an extended hill‑climb followed by a short sub‑gradient run.

All iterates are projected onto the ℓ¹‑simplex after each update, guaranteeing
that the evaluation function reduces to the peak of the self‑convolution.
The routine respects the supplied wall‑clock budget (default 1000 s) and
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Duchi‑et‑al. (2008) algorithm.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of   (1/β)·log Σ exp(β·b)   where   b = a*a .
    Returns (g, b) with   g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)  (correlation)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smoothed maximum.
    Momentum, β‑annealing and occasional jitter are used.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 16384.0
    beta_factor = 1.5

    step_base = target_sum * 0.12
    momentum = np.zeros_like(a)
    rho = 0.92                       # Nesterov‑style momentum decay

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        # projection onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to leave plateaus
        if it % 300 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β‑annealing
        if it % 400 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max
# ----------------------------------------------------------------------


def _subgradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simple sub‑gradient descent on  max(a∗a)  using the exact sub‑gradient
    2·(a ⋆ 1_{peak}) .  After each step the vector is projected onto the
    ℓ¹‑simplex Σa = sqrt(2·n).
    """
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)
    rng = np.random.default_rng()
    base_lr = target * 0.02
    it = 0
    while time.time() < deadline:
        b = _full_convolution(a)
        max_b = b.max()
        # mask of all entries that are (almost) maximal
        tol = max_b * 1e-12 + 1e-12
        mask = (b >= max_b - tol).astype(float)

        # correlation a ⋆ mask  (full mode)
        corr = np.correlate(a, mask, mode='full')
        g = 2.0 * corr[n - 1 : n - 1 + n]

        lr = base_lr / np.sqrt(it + 1.0)
        a -= lr * g

        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target)

        if it % 200 == 0:
            jitter = rng.normal(scale=target * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

        it += 1
    return a


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 7)

    idx = np.arange(n, dtype=float)

    if family == 0:                       # pure power‑law
        p = rng.uniform(0.10, 1.2)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                     # exponential decay
        lam = rng.exponential(scale=0.018)
        a = np.exp(-lam * idx)

    elif family == 2:                     # mixed power‑law × exponential
        p = rng.uniform(0.12, 1.0)
        lam = rng.exponential(scale=0.018)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                     # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                     # piecewise exponential
        cut = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(cut)),
             np.exp(-lam2 * np.arange(cut, n))]
        )

    elif family == 5:                     # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    else:                                 # random Gaussian‑like bump
        centre = rng.uniform(0, 1)
        width = rng.exponential(scale=0.15) + 0.05
        a = np.exp(-((idx / n - centre) ** 2) / (2 * width ** 2))

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Targeted mass‑transfer hill climbing (exact convolution updates)
# ----------------------------------------------------------------------


def _hill_climb_targeted(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Hill‑climbing that always looks at the current maximal convolution entry.
    Moves a small amount of mass from a source that contributes a lot to that
    entry to a destination that contributes little.
    Returns the best sequence found and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size

    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    T = cur_max * 0.02 + 1e-9          # initial temperature for Metropolis steps
    cooling = 0.9995

    while time.time() < deadline:
        # index of the current peak
        k_max = int(np.argmax(b))

        # contribution of each index to that peak:
        #   contrib[i] = a[i] * a[k_max-i]   (zero if out of bounds)
        j_arr = k_max - np.arange(n)
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[mask] * a[j_arr[mask]]

        # source index – weighted by contribution (remove from “bad” places)
        total = contrib.sum()
        if total > 0.0:
            prob_src = contrib / total
            i = rng.choice(n, p=prob_src)
        else:
            prob_src = a / a.sum()
            i = rng.choice(n, p=prob_src)

        # destination index – weighted towards *small* contribution
        inv = contrib.max() - contrib + 1e-12
        prob_dst = inv / inv.sum()
        j = rng.choice(n, p=prob_dst)
        if i == j:
            continue

        # amount of mass to move (up to 50 % of source)
        max_delta = a[i] * 0.5
        if max_delta < 1e-16:
            continue
        delta = rng.random() * max_delta

        # ---- analytical update of the convolution --------------------
        b_new = b.copy()
        # linear part  (‑2 δ·a shifted by i, +2 δ·a shifted by j)
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a

        # quadratic correction terms
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        # ---- acceptance test -----------------------------------------
        accept = False
        if new_max < cur_max - 1e-12:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            a[i] = max(a[i], 0.0)
            a[j] = max(a[j], 0.0)

            b = b_new
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                # limit printing frequency to avoid massive output
                print(f"[{time.time():.1f}] hill_targeted improvement: {best_max:.6f}")

        T *= cooling

    return best_a, best_max


# ----------------------------------------------------------------------
# Main search entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Optional seed from a previously discovered construction.
    # ------------------------------------------------------------------
    earlier = globals().get("height_sequence_1")
    if isinstance(earlier, (list, tuple, np.ndarray)):
        a0 = np.asarray(earlier, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        best_seq = a0.copy()
        best_score = _max_convolution_fft(a0)
        best_len = n0
        print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) "
              f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Length schedule – start modest, then enlarge gradually.
    # ------------------------------------------------------------------
    lengths = [2000, 5000, 8000, 12000, 18000, 25000, 30000]

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)
        # Allocate a (soft) time budget for this length
        time_remaining = deadline - time.time()
        max_len_time = min(150.0, time_remaining * 0.25)
        length_deadline = time.time() + max_len_time

        # ------------------------------------------------------------------
        # Perform a few independent restarts for this length.
        # ------------------------------------------------------------------
        for restart in range(3):
            if time.time() >= length_deadline:
                break

            # Initialise candidate
            if best_seq is not None and best_seq.size == n:
                a_init = best_seq.copy()
            elif best_seq is not None:
                a_init = _upsample_sequence(best_seq, n)
            else:
                a_init = _random_initial_sequence(n, rng)

            a_init = _project_to_simplex(a_init, target_sum)

            # -------------------- sub‑gradient phase --------------------
            sub_dead = time.time() + min(30.0, (length_deadline - time.time()) * 0.4)
            a_sub = _subgradient_descent_opt(a_init, sub_dead)
            score_sub = _max_convolution_fft(a_sub)
            if score_sub < best_score:
                best_score = score_sub
                best_seq = a_sub.copy()
                best_len = n
                print(f"[{time.time():.1f}] Sub‑grad restart {restart} (n={n}) "
                      f"score={score_sub:.6f}")

            # -------------------- gradient‑descent phase --------------------
            gd_dead = time.time() + min(30.0, (length_deadline - time.time()) * 0.4)
            a_gd = _gradient_descent_opt(a_sub, gd_dead)
            score_gd = _max_convolution_fft(a_gd)
            if score_gd < best_score:
                best_score = score_gd
                best_seq = a_gd.copy()
                best_len = n
                print(f"[{time.time():.1f}] GD restart {restart} (n={n}) "
                      f"score={score_gd:.6f}")

            # -------------------- targeted hill‑climb --------------------
            if time.time() < length_deadline:
                b_full = _full_convolution(a_gd)
                hc_dead = time.time() + min(20.0, (length_deadline - time.time()) * 0.2)
                a_hc, max_hc = _hill_climb_targeted(a_gd, b_full, hc_dead)
                if max_hc < best_score:
                    best_score = max_hc
                    best_seq = a_hc.copy()
                    best_len = n
                    print(f"[{time.time():.1f}] Hill‑climb restart {restart} (n={n}) "
                          f"score={max_hc:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with any remaining budget.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a_final = best_seq.copy()
        b_full = _full_convolution(a_final)
        a_final, max_final = _hill_climb_targeted(a_final, b_full, deadline)

        # a brief sub‑gradient polishing pass if time permits
        if time.time() < deadline - 1.0:
            a_final = _subgradient_descent_opt(a_final, deadline)

        final_score = _max_convolution_fft(a_final)
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final polishing improvement: {final_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries for neatness.
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback: uniform vector (should never happen)
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log the official evaluation value.
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:   # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
